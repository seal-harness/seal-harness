    2 -- | Source-control clone seam — the W2 no-disk design (design §4.1, §4.4,
    3 -- §4.6, §5). The credential mechanism (rev 3): the encrypted keyfile lives
    4 -- on the harness disk (ciphertext); the passphrase lives in the vault;
    5 -- per-op @ssh-agent@ decrypts the keyfile into memory using the passphrase
    6 -- piped to @ssh-add@'s stdin; @ssh -A@ forwards the @SSH_AUTH_SOCK@ to the
    7 -- untrusted machine; after the op, @ssh-add -D@ + @ssh-agent -k@.
    8 --
    9 -- Security invariants (enforced by construction — the W2 self-reviewed
   10 -- checkpoint audits these):
   11 --
   12 -- 1. **No un-encrypted secret on disk, on either machine.** The encrypted
   13 --    keyfile under @cdKeyfilesDir@ IS permitted (ciphertext — same category
   14 --    as the age-encrypted vault file). The cleartext key exists only in:
   15 --    the vault (age-encrypted at rest) → harness process memory (@vhGet@)
   16 --    → ssh-agent memory (@ssh-add@ decrypts the encrypted keyfile using the
   17 --    passphrase piped to its stdin) → signing requests over the forwarded
   18 --    agent socket. The untrusted machine only ever sees the forwarded
   19 --    @SSH_AUTH_SOCK@.
   20 --
   21 -- 2. **Per-op agent lifecycle.** Each clone/git-op gets a fresh
   22 --    @ssh-agent@ via 'withCloneTarget' (bracket semantics): start →
   23 --    @sahAddKey <enc-keyfile> <passphrase>@ → run → @sahDeleteAll@ +
   24 --    @sahKill@. Exactly one identity is live at forwarding time (the
   25 --    security-critical scoping — design §4.6).
   26 --
   27 -- 3. **PAT (fallback) uses @http.extraHeader@ in argv (memory, no file).**
   28 --    The token lives only in the git config args (in-memory argv); it is
   29 --    NEVER written to disk. The token-in-untrusted-memory residual is
   30 --    documented (deploy keys preferred).
   31 --
   32 -- 4. **Pinned @known_hosts@ (public data).** 'cdPinnedKnownHosts' is the
   33 --    compile-time-embedded GitHub host keys (tamper-resistant via
   34 --    'file-embed'). Per-op it is written to a temp @known_hosts@ file
   35 --    (0644, public data, bracket-cleaned) and referenced by
   36 --    @UserKnownHostsFile=@ in @GIT_SSH_COMMAND@. @StrictHostKeyChecking=yes@
   37 --    (NOT @accept-new@ / @/dev/null@ — §5.3: MITM defense).
   38 --
   39 -- 5. **The encrypted keyfile never leaves the harness.** The untrusted
   40 --    machine never sees the keyfile (encrypted or otherwise) — only the
   41 --    forwarded @SSH_AUTH_SOCK@ via @ssh -A@.
   42 --
   43 -- 6. **'CloneGitFailed' carries the exit code ONLY — no stderr** (§5.4:
   44 --    git can echo a token-bearing URL in stderr on auth failure).
   45 --
   46 -- 7. **'withCloneTarget' is CPS** (mirrors 'Seal.Security.Secrets.withApiKey'):
   47 --    the authenticated bits are scoped to the continuation and the cleanup
   48 --    always runs (bracket semantics).

   99 -- | The clone seam's error type. 'CloneGitFailed' carries the exit code ONLY
  100 -- (no stderr — §5.4: git can echo a token-bearing URL in stderr on auth
  101 -- failure; a "redacted one-liner" heuristic is fragile, so stderr is dropped
  102 -- entirely for this pass).
  103 data CloneError
  104   = CloneVaultError VaultError
  105     -- ^ A vault access failure. Preserves the 'VaultError' so the caller can
  106     -- distinguish 'VaultLocked' from 'VaultKeyNotFound'.
  107   | CloneNoCredentialForUrl Text
  108     -- ^ The URL is malformed (host unparseable) — no credential can be
  109     -- resolved.
  110   | CloneUnsupportedVcs VcsKind
  111     -- ^ The 'VcsKind' has no credential-injection path in this pass
  112     -- (only 'VcsGitHub' is supported).
  113   | CloneHostNotSupported Text
  114     -- ^ The parsed host is not in the allow-list (defense-in-depth on top of
  115     -- write-time validation).
  116   | CloneAgentError Text
  117     -- ^ The per-op ssh-agent lifecycle failed (start, add-key, etc.). The
  118     -- cleartext key never made it to the agent — fail closed.
  119   | CloneGitFailed Int
  120     -- ^ @git@ exited non-zero. Carries the exit code ONLY — NO stderr (§5.4).
  121   deriving stock (Eq, Show)
  122 
  123 -- | Render a 'CloneError' to the exact user-facing message used by the slash
  124 -- command (W5) and a future @CLONE@ opcode. The mapping is total and stable:
  125 -- each variant maps to a distinct, actionable message.
  126 renderCloneError :: CloneError -> Text
  127 renderCloneError = \case
  128   CloneVaultError VaultLocked ->
  129     "vault locked — run /vault unlock"
  130   CloneVaultError (VaultKeyNotFound k) ->
  131     "vault key " <> k <> " not found"
  132   CloneVaultError v ->
  133     "vault error: " <> T.pack (show v)
  134   CloneNoCredentialForUrl u ->
  135     "no credential resolvable for " <> u
  136   CloneUnsupportedVcs v ->
  137     "unsupported VCS: " <> T.pack (show v)
  138   CloneHostNotSupported h ->
  139     "host " <> h <> " not supported (only github.com is supported in this pass)"
  140   CloneAgentError msg ->
  141     "ssh-agent error: " <> msg
  142   CloneGitFailed n ->
  143     "git failed (exit " <> T.pack (show n) <> ")"

  149 -- | The pure credential-injection strategy for a 'SourceRepo'. Carries only
  150 -- public data (URLs + vault key /names/) — never secret bytes.
  151 data ClonePlan
  152   = ClonePlanExtraHeader Text Text
  153     -- ^ Token-free HTTPS URL + the vault key name (PAT / MachineUser). The
  154     -- token is fetched at clone time and passed via @http.extraHeader@ in
  155     -- the git argv (memory, no file).
  156   | ClonePlanSshKey Text Text
  157     -- ^ SSH URL (host-bound) + the vault key name (DeployKey). The
  158     -- /passphrase/ is fetched at clone time and piped to @ssh-add@'s stdin;
  159     -- the encrypted keyfile is read from @cdKeyfilesDir@ by @ssh-add@.
  160   deriving stock (Eq, Show)
  161 
  162 -- | Decide the clone strategy for a 'SourceRepo' (pure, no IO, no vault).
  163 -- Re-parses the URL host and asserts the allow-list (§5.2 defense-in-depth),
  164 -- then routes on 'srVcsKind' / 'srCredential'.
  165 --
  166 -- * Malformed URL (host unparseable) → 'CloneNoCredentialForUrl'.
  167 -- * Parsed host not in the allow-list → 'CloneHostNotSupported'.
  168 -- * 'VcsGit' (non-GitHub git) → 'CloneUnsupportedVcs' (no
  169 --   credential-injection path this pass).
  170 -- * 'CredPat' / 'CredMachineUser' → 'ClonePlanExtraHeader' (SSH URL
  171 --   rewritten to token-free HTTPS).
  172 -- * 'CredDeployKey' → 'ClonePlanSshKey' (URL unchanged, SSH).
  173 planClone :: SourceRepo -> Either CloneError ClonePlan
  174 planClone repo =
  175   case parseRepoHost (srUrl repo) of
  176     Left _err -> Left (CloneNoCredentialForUrl (srUrl repo))
  177     Right host
  178       | not (hostAllowed host) -> Left (CloneHostNotSupported host)
  179       | srVcsKind repo == VcsGit -> Left (CloneUnsupportedVcs VcsGit)
  180       | otherwise -> case srCredential repo of
  181           CredPat vaultKey ->
  182             Right (ClonePlanExtraHeader (sshToHttps (srUrl repo)) vaultKey)
  183           CredMachineUser vaultKey _username ->
  184             Right (ClonePlanExtraHeader (sshToHttps (srUrl repo)) vaultKey)
  185           CredDeployKey vaultKey ->
  186             Right (ClonePlanSshKey (srUrl repo) vaultKey)
  187 
  188 -- | Rewrite an SSH GitHub URL (@git\@github.com:owner\/repo.git@) to a
  189 -- token-free HTTPS URL (@https:\/\/github.com\/owner\/repo.git@). An
  190 -- already-HTTPS URL is returned unchanged. The result NEVER contains a
  191 -- credential fragment.
  192 sshToHttps :: Text -> Text
  193 sshToHttps url
  194   | "https://" `T.isPrefixOf` url = url
  195   | "http://" `T.isPrefixOf` url  = url
  196   | "git@" `T.isPrefixOf` url =
  197       let afterAt = T.drop (T.length "git@") url
  198       in case T.breakOn ":" afterAt of
  199            (host, rest)
  200              | T.null rest -> url
  201              | otherwise  ->
  202                  "https://" <> host <> "/" <> T.drop (T.length ":") rest
  203   | otherwise = url

  206 -- CloneDeps — the closed-over opcode param (W2)
  207 ----------------------------------------------------------------------------
  208 
  209 -- | The per-op credential dependencies, passed as a closed-over param to
  210 -- 'resolveCloneTarget' / 'cloneRepo' / 'lsRemoteRepo' (mirrors
  211 -- @secretGetOp (cdVault deps)@ at @Channels/Loop.hs:1094@ — NOT via
  212 -- 'Env'/'mkEnv'; the codebase proved 'Env' is the wrong vehicle).
  213 --
  214 -- Carries:
  215 --
  216 -- * @cdVault@ — the 'VaultRuntime' (yields the live 'VaultHandle' via
  217 --   'vrHandleRef' at runtime; fail-closed to 'CloneVaultError VaultLocked'
  218 --   if the vault is unconfigured/locked). Mirrors @secretGetOp rt@ which
  219 --   takes 'VaultRuntime', not the raw handle.
  220 -- * @cdSshAgent@ — the 'SshAgentHandle' seam (real or fake). Shared agent:
  221 --   started once (lazily on first use, cached in 'cdAgentEnvRef'),
  222 --   reused across all ops. Per-op: @sahAddKey@ → run → @sahDeleteAll@
  223 --   (the per-op key scoping — exactly one key live at forwarding time).
  224 -- * @cdAgentEnvRef@ — caches the shared 'SshAgentEnv' (the socket + PID
  225 --   from the first @sahStart@ call). 'Nothing' until the first deploy-key
  226 --   op starts the agent. Subsequent ops reuse the cached env (no
  227 --   re-start).
  228 -- * @cdPinnedKnownHosts@ — compile-time-embedded GitHub host keys (public
  229 --   data, tamper-resistant via 'file-embed'). Written per-op to a temp
  230 --   @known_hosts@ file.
  231 -- * @cdKeyfilesDir@ — the harness-private dir holding the encrypted
  232 --   keyfiles (@~\/.seal\/state\/repos\/keys\/\<repo-id\>@). The encrypted
  233 --   keyfile lives only on the harness disk (ciphertext).
  234 data CloneDeps = CloneDeps
  235   { cdVault           :: VaultRuntime
  236   , cdRepoReg         :: RepoRegistryHandle
  237   , cdSshAgent        :: SshAgentHandle
  238   , cdAgentRegistry   :: AgentRegistryHandle
  239     -- ^ One-agent-per-repo registry, keyed by the repo's vault key. Each
  240     -- unique repo gets its own @ssh-agent@ process, started lazily on
  241     -- first use. The agent + key persist for the harness lifetime AND
  242     -- across @seal serve@ restarts (the registry is persisted to disk —
  243     -- #88). At startup, 'arProbeAndSweep' GCs dead agents from crashed
  244     -- processes; live agents are reused (the key is still loaded).
  245   , cdPinnedKnownHosts :: ByteString
  246   , cdKeyfilesDir     :: FilePath
  247   , cdIsRemote        :: Bool
  248     -- ^ Whether the untrusted executor runs commands over SSH (remote
  249     -- mode). When 'True', the deploy-key clone path uses @ssh -A@ agent
  250     -- forwarding with no @GIT_SSH_COMMAND@ (the remote @git clone@ uses
  251     -- the default @ssh@ + the forwarded agent + the remote machine's
  252     -- default @known_hosts@). When 'False', the local path writes a
  253     -- per-op @known_hosts@ temp file + uses @GIT_SSH_COMMAND@.
  254   }
  255

  276 -- | The authenticated bits for a single clone, scoped to 'withCloneTarget'.
  277 -- The constructors are NOT exported — callers observe the bits only via the
  278 -- CPS continuation, mirroring 'Seal.Security.Secrets.withApiKey'. 'Show' is
  279 -- redacted so a stray log/exception cannot leak the env extras.
  280 data CloneTarget = CloneTarget
  281   { ctEnv :: CloneEnv
  282   , ctCleanup :: IO ()
  283   }
  284 
  285 instance Show CloneTarget where
  286   show _ = "CloneTarget <redacted>"

  288 -- | The environment passed to @git@. 'ceUrl' is the TOKEN-FREE URL;
  289 -- 'ceEnvExtras' carries only non-secret values (@SSH_AUTH_SOCK@ +
  290 -- @GIT_SSH_COMMAND@ for deploy keys, @GIT_TERMINAL_PROMPT=0@ always). The
  291 -- secret bytes (the passphrase for deploy keys, the token for PATs) live
  292 -- only in the ssh-agent's memory (deploy keys) or the git argv
  293 -- (@http.extraHeader@ for PATs — memory, no file).
  294 data CloneEnv = CloneEnv
  295   { ceUrl :: Text
  296     -- ^ The token-free URL passed to @git clone@ / @git ls-remote@.
  297   , ceGitConfigArgs :: [Text]
  298     -- ^ Extra @-c@ config args. PAT/MachineUser: @["-c",
  299     -- "http.extraHeader=Authorization: Basic \<base64\>"]@ (memory, no
  300     -- file). DeployKey: @[]@ (uses the agent, not extraHeader).
  301   , ceSshCommand :: Maybe Text
  302     -- ^ @GIT_SSH_COMMAND@ value (deploy-key only). Carries the pinned
  303     -- @known_hosts@ PATH, not the key bytes (the key is in the agent).
  304   , ceEnvExtras :: [(String, String)]
  305     -- ^ Env overrides MERGED over the inherited environment. Carries
  306     -- @SSH_AUTH_SOCK@ / @GIT_SSH_COMMAND@ / @GIT_TERMINAL_PROMPT@ — only
  307     -- non-secret values (paths + "0"). On the remote executor (deploy
  308     -- key), @SSH_AUTH_SOCK@ / @SSH_AGENT_PID@ are OMITTED (the forwarded
  309     -- agent via @ssh -A@ replaces them) and @GIT_SSH_COMMAND@ carries no
  310     -- @UserKnownHostsFile=@ path (the remote arm writes the known_hosts
  311     -- to a remote temp file + rewrites the command).
  312   , ceKnownHostsContent :: Maybe ByteString
  313     -- ^ The pinned @known_hosts@ content (public data) for the REMOTE
  314     -- deploy-key path. 'Just' the bytes when @cdIsRemote@ + deploy key;
  315     -- 'Nothing' otherwise (the local arm writes its own temp file). The
  316     -- remote arm of 'uioShellExecGitEnv' writes this to a remote temp
  317     -- file + rewrites @GIT_SSH_COMMAND@ to reference it.
  318   , ceRawToken :: Maybe ByteString
  319     -- ^ The raw PAT/MachineUser token bytes for the @gh@ credential
  320     -- injection path (design §3.4 Option A). 'Just' the raw bytes for
  321     -- PAT/MachineUser (the @gh@ path injects @GH_TOKEN@ from these
  322     -- bytes); 'Nothing' for deploy keys (@gh@ can't use SSH, falls
  323     -- through to plain exec). The @git@ path ignores this field (it
  324     -- uses 'ceGitConfigArgs' for the base64 @http.extraHeader@). Same
  325     -- security category as 'ceGitConfigArgs' (which carries the
  326     -- base64-encoded header — trivially reversible); CPS-scoped via
  327     -- 'withCloneTarget' (the value is only obtainable inside the
  328     -- continuation). 'CloneEnv' has NO 'Show' instance by design — a
  329     -- stray @show@/log/exception cannot leak any field.
  330   , ceCleanup :: IO ()
  331     -- ^ Removes the per-op @known_hosts@ temp file + kills the agent. Run
  332     -- by 'withCloneTarget' after the continuation (bracket — on success AND
  333     -- failure).
  334   }
  335 
  336 -- | Scope the authenticated bits + cleanup to the continuation (mirrors
  337 -- 'Seal.Security.Secrets.withApiKey'). The cleanup ALWAYS runs, even if the
  338 -- continuation throws — 'bracket' semantics.
  339 withCloneTarget :: CloneTarget -> (CloneEnv -> IO r) -> IO r
  340 withCloneTarget target k =
  341   bracket (pure target) ctCleanup (k . ctEnv)

  344 -- resolveCloneTarget

  347 -- | Resolve a 'SourceRepo' to a 'CloneTarget' — fetches the secret bytes
  348 -- from the vault, runs the per-op ssh-agent lifecycle (deploy keys), and
  349 -- writes the per-op @known_hosts@ temp file (public data, bracket-cleaned).
  350 --
  351 -- For deploy keys: @sahStart@ → @sahAddKey <enc-keyfile> <passphrase>@
  352 -- (passphrase from the vault, piped to @ssh-add@'s stdin via the fake/real
  353 -- 'SshAgentHandle'). The agent decrypts the encrypted keyfile into memory.
  354 -- On any agent failure → 'CloneAgentError' (fail closed — the cleartext key
  355 -- never made it to the agent).
  356 --
  357 -- For PAT/MachineUser: the token is fetched from the vault and base64-encoded
  358 -- into @http.extraHeader=Authorization: Basic \<base64(user:token)\>@ (for
  359 -- MachineUser, the username is the public @cUsername@). The header lives
  360 -- only in 'ceGitConfigArgs' (memory, no file).
  361 --
  362 -- The per-op @known_hosts@ temp file is written under a random-suffix name
  363 -- (§5.5) and @bracket@-cleaned via 'withCloneTarget'.
  364 resolveCloneTarget
  365   :: CloneDeps -> SourceRepo -> IO (Either CloneError CloneTarget)
  366 resolveCloneTarget deps repo =
  367   case planClone repo of
  368     Left e -> pure (Left e)
  369     Right plan -> case plan of
  370       ClonePlanExtraHeader httpsUrl vaultKey -> do
  371         eVault <- resolveVaultHandle deps
  372         case eVault of
  373           Left e -> pure (Left e)
  374           Right vault -> do
  375             mtoken <- vhGet vault vaultKey
  376             case mtoken of
  377               Left ve -> pure (Left (CloneVaultError ve))
  378               Right tokenBytes -> do
  379                 let header = renderPatHeader repo tokenBytes
  380                     envExtras = [("GIT_TERMINAL_PROMPT", "0")]
  381                     env = CloneEnv
  382                       { ceUrl = httpsUrl
  383                       , ceGitConfigArgs = ["-c", "http.extraHeader=" <> header]
  384                       , ceSshCommand = Nothing
  385                       , ceEnvExtras = envExtras
  386                       , ceKnownHostsContent = Nothing
  387                       , ceRawToken = Just tokenBytes
  388                       , ceCleanup = pure ()
  389                       }
  390                 pure (Right CloneTarget { ctEnv = env, ctCleanup = pure () })
  391       ClonePlanSshKey sshUrl _vaultKey -> do
  392         -- The encrypted keyfile path: <keyfilesDir>/<repo-id>.
  393         let keyfilePath = cdKeyfilesDir deps </> keyfileBaseName repo
  394             vaultKey = cVaultKey (srCredential repo)
  395         eVault <- resolveVaultHandle deps
  396         case eVault of
  397           Left e -> pure (Left e)
  398           Right vault -> do
  399             -- Fetch the passphrase from the vault (decrypts the keyfile).
  400             mpass <- vhGet vault vaultKey
  401             case mpass of
  402               Left ve -> pure (Left (CloneVaultError ve))
  403               Right passphrase -> do
  404                 -- One-agent-per-repo: look up the persistent registry by
  405                 -- vault key. If a live agent is cached (in this process or
  406                 -- persisted from a prior seal process), reuse it — the key
  407                 -- is still loaded. If the persisted agent is dead (crashed
  408                 -- seal), remove the stale entry and start a fresh agent.
  409                 -- If no entry exists, start a new agent + load the key
  410                 -- once, then persist (#88).
  411                 registry <- arLoad (cdAgentRegistry deps)
  412                 agentEnv <- case Map.lookup vaultKey registry of
  413                   Just cachedEnv -> do
  414                     -- If the agent was started in THIS process, reuse it
  415                     -- directly (no probe — it's alive). If it was loaded
  416                     -- from a prior process's persisted registry, probe to
  417                     -- check liveness (#88).
  418                     isLive <- arIsLive (cdAgentRegistry deps) vaultKey
  419                     if isLive
  420                       then pure cachedEnv  -- reuse (key still loaded)
  421                       else do
  422                         status <- probeAgent cachedEnv
  423                         case status of
  424                           AgentAlive -> pure cachedEnv  -- reuse across restart
  425                           AgentDead -> do
  426                             -- GC the stale entry + start a fresh agent.
  427                             arRemove (cdAgentRegistry deps) vaultKey
  428                             startAndLoadAgent deps vaultKey keyfilePath passphrase
  429                   Nothing -> startAndLoadAgent deps vaultKey keyfilePath passphrase
  430                 -- Build the CloneEnv. The key stays loaded (no per-op
  431                 -- sahDeleteAll); the cleanup is a no-op (the agent is
  432                 -- killed at harness shutdown, not per-op).
  433                 if cdIsRemote deps
  434                   then do
  435                     -- REMOTE deploy-key path (simple approach matching the
  436                     -- verified working command): no GIT_SSH_COMMAND, no
  437                     -- remote known_hosts file. The remote git clone uses
  438                     -- the default ssh + the forwarded agent (via ssh -A)
  439                     -- + the remote machine's default known_hosts. The
  440                     -- SSH_AUTH_SOCK/SSH_AGENT_PID are in ceEnvExtras for
  441                     -- the LOCAL ssh -A process (the remote arm extracts
  442                     -- them for local env + strips from remote env prefix).
  443                     let authEnv = sahGetAuthEnv (cdSshAgent deps) agentEnv
  444                         envExtras =
  445                           authEnv ++
  446                           [ ("GIT_TERMINAL_PROMPT", "0")
  447                           ]
  448                         env = CloneEnv
  449                           { ceUrl = sshUrl
  450                           , ceGitConfigArgs = []
  451                           , ceSshCommand = Nothing
  452                           , ceEnvExtras = envExtras
  453                           , ceKnownHostsContent = Nothing
  454                           , ceRawToken = Nothing
  455                           , ceCleanup = pure ()
  456                           }
  457                     pure (Right CloneTarget { ctEnv = env, ctCleanup = pure () })
  458                   else do
  459                     -- LOCAL deploy-key path: write the per-op
  460                     -- @known_hosts@ temp file (public data) to the
  461                     -- harness-private @cdKeyfilesDir@.
  462                     (knownHostsPath, knownHostsCleanup) <-
  463                       writeKnownHostsTemp (cdKeyfilesDir deps)
  464                                            (cdPinnedKnownHosts deps)
  465                     -- NOTE: IdentitiesOnly is intentionally omitted from
  466                     -- GIT_SSH_COMMAND. On macOS, the bundled /usr/bin/ssh
  467                     -- suppresses agent-offered keys when
  468                     -- IdentitiesOnly=yes is set and no -i flag is passed
  469                     -- — it only tries identity files, not the agent.
  470                     -- Without this flag, ssh tries agent keys first (which
  471                     -- is what we want — the per-op agent holds exactly one
  472                     -- deploy key). The per-op agent scoping invariant
  473                     -- (exactly one key live at forwarding time) replaces
  474                     -- what IdentitiesOnly would enforce.
  475                     let authEnv = sahGetAuthEnv (cdSshAgent deps) agentEnv
  476                         sshCmd = T.pack
  477                           ( "ssh"
  478                           <> " -o StrictHostKeyChecking=yes"
  479                           <> " -o BatchMode=yes"
  480                           <> " -o UserKnownHostsFile=" <> knownHostsPath
  481                           )
  482                         envExtras =
  483                           [ ("SSH_AUTH_SOCK", saeAuthSock agentEnv)
  484                           , ("SSH_AGENT_PID", saeAgentPid agentEnv)
  485                           , ("GIT_SSH_COMMAND", T.unpack sshCmd)
  486                           , ("GIT_TERMINAL_PROMPT", "0")
  487                           ]
  488                         env = CloneEnv
  489                           { ceUrl = sshUrl
  490                           , ceGitConfigArgs = []
  491                           , ceSshCommand = Just sshCmd
  492                           , ceEnvExtras = authEnv ++ envExtras
  493                           , ceKnownHostsContent = Nothing
  494                           , ceRawToken = Nothing
  495                           , ceCleanup = knownHostsCleanup
  496                           }
  497                     pure (Right CloneTarget { ctEnv = env, ctCleanup = knownHostsCleanup })

  499 -- | Start a fresh ssh-agent + load the deploy key once, then persist the
  500 -- agent env to the registry so it's reused across ops (and across seal
  501 -- restarts — #88). On failure, fail closed (delete-all + error).
  502 startAndLoadAgent :: CloneDeps -> Text -> FilePath -> ByteString -> IO SshAgentEnv
  503 startAndLoadAgent deps vaultKey keyfilePath passphrase = do
  504   eStart <- sahStart (cdSshAgent deps)
  505   case eStart of
  506     Left msg -> error (T.unpack msg)
  507     Right env -> do
  508       eAdd <- sahAddKey (cdSshAgent deps) env keyfilePath passphrase
  509       case eAdd of
  510         Left msg -> do
  511           -- Fail closed: delete-all even on add failure.
  512           sahDeleteAll (cdSshAgent deps) env
  513           error (T.unpack msg)
  514         Right () -> do
  515           -- Persist the agent env so a second seal process (or a restart)
  516           -- reuses this agent instead of spawning a new one (#88).
  517           arUpsert (cdAgentRegistry deps) vaultKey env
  518           pure env

  522 -- defense-in-depth). The keyfile lives at
  523 -- @\<cdKeyfilesDir\>\/\<repo-id\>@.
  524 keyfileBaseName :: SourceRepo -> FilePath

  527 -- | Resolve the live 'VaultHandle' from the 'VaultRuntime' (mirrors
  528 -- @secretGetOp@'s pattern at @Seal.ISA.Ops.Secret:73@). Fail-closed to
  529 -- 'CloneVaultError VaultLocked' if the vault is unconfigured/locked (the
  530 -- 'IORef' holds 'Nothing'). This is the W3 evolution: @cdVault@ is now
  531 -- 'VaultRuntime' (not the raw 'VaultHandle') so the opcode can be built once
  532 -- at startup and the vault-locked state is surfaced per-op at run time
  533 -- (rather than crashing at startup or silently using a stale handle).
  534 resolveVaultHandle :: CloneDeps -> IO (Either CloneError VaultHandle)
  535 resolveVaultHandle deps = do
  536   mh <- readIORef (vrHandleRef (cdVault deps))
  537   pure $ case mh of
  538     Nothing -> Left (CloneVaultError VaultLocked)
  539     Just h  -> Right h

  541 -- | Render the @http.extraHeader@ value for a PAT/MachineUser: @Authorization:
  542 -- Basic \<base64(user:token)\>@. For 'CredPat' the "user" is
  543 -- @x-access-token@; for 'CredMachineUser' it's the public @cUsername@. The
  544 -- header lives only in the git argv (memory, no file).
  545 --
  546 -- The base64 is computed over the RAW @user:token@ bytes (NOT a UTF-8
  547 -- round-trip through 'Text') — a vault token containing non-UTF-8 bytes
  548 -- (rare for GitHub PATs which are ASCII, but possible for MachineUser
  549 -- tokens or future cred types) is base64-encoded verbatim, avoiding the
  550 -- U+FFFD corruption that 'TE.decodeUtf8Lenient' would introduce before
  551 -- base64. The base64 output is pure ASCII, so the final
  552 -- 'TE.decodeUtf8Lenient' is lossless.
  553 renderPatHeader :: SourceRepo -> ByteString -> Text
  554 renderPatHeader repo tokenBytes =
  555   let userBytes = case srCredential repo of
  556         CredPat _            -> "x-access-token"
  557         CredMachineUser _ u  -> TE.encodeUtf8 u
  558         CredDeployKey _       -> "x-access-token"
  559       basic = userBytes <> ":" <> tokenBytes
  560       encoded = B64.encode basic
  561   in "Authorization: Basic " <> TE.decodeUtf8Lenient encoded

  567 -- | Write a temp @known_hosts@ file (public data — the pinned GitHub host
  568 -- keys) under the harness-private @parentDir@ (the 0700 @cdKeyfilesDir@)
  569 -- with a random-suffix name via 'openBinaryTempFile' (which uses @O_EXCL@
  570 -- under the hood — prevents a symlink-race / TOCTOU attack on the path git
  571 -- reads: an attacker cannot pre-plant a symlink at the chosen path because
  572 -- @O_EXCL@ fails if the file already exists). The file is 0600 (private —
  573 -- the content is public but the path git reads must be tamper-resistant, so
  574 -- we don't make it world-readable). Returns the path + a cleanup action
  575 -- (removes the file). 'bracket'-cleaned via the post-add bracket in
  576 -- 'resolveCloneTarget'.
  577 --
  578 -- Security rationale: the path git reads via @UserKnownHostsFile=@ must be
  579 -- integrity-protected, not just the content. A @/tmp@ location would be
  580 -- world-writable + symlink-raceable; a local attacker could swap the file
  581 -- between our write and git's read, planting an attacker-controlled host
  582 -- key → MITM the next git SSH connection. The 0700 @cdKeyfilesDir@ closes
  583 -- the race (only the harness user can write there) + @O_EXCL@ closes the
  584 -- symlink plant (the open fails if the path already exists).
  585 writeKnownHostsTemp :: FilePath -> ByteString -> IO (FilePath, IO ())
  586 writeKnownHostsTemp parentDir contents = do
  587   createDirectoryIfMissing True parentDir
  588   (path, h) <- openBinaryTempFile parentDir "seal-known-hosts-"
  589   BS.hPutStr h contents
  590   hClose h
  591   setFileMode path 0o600
  592   -- Idempotent cleanup: the bracket release in 'resolveCloneTarget' AND
  593   -- 'withCloneTarget' both call this (the bracket release runs first on
  594   -- success, then 'ctCleanup' runs again — the second 'removeFile' would
  595   -- throw NoSuchThing). Swallow the IOException so double-cleanup is safe.
  596   let cleanup = do
  597         _ <- try @IOError (removeFile path)
  598         pure ()
  599   pure (path, cleanup)

  605 -- | @git clone@ a 'SourceRepo' into @destDir@ using the resolved credential.
  606 -- The token/passphrase NEVER appears in argv, the URL, or the env (only
  607 -- @SSH_AUTH_SOCK@ / @GIT_SSH_COMMAND@ / @http.extraHeader@ — non-secret
  608 -- paths + the base64-encoded Basic header which is in argv, in memory). On
  609 -- non-zero exit → 'CloneGitFailed' with the exit code ONLY (stderr is
  610 -- DROPPED — §5.4). Cleanup runs via 'withCloneTarget' (bracket — on success
  611 -- AND failure).
  612 cloneRepo
  613   :: CloneDeps -> FilePath -> SourceRepo -> IO (Either CloneError ())
  614 cloneRepo deps destDir repo =
  615   resolveCloneTarget deps repo >>= \case
  616     Left e -> pure (Left e)
  617     Right target -> withCloneTarget target $ \env -> do
  618       let gitArgs = ceGitConfigArgsList env
  619                      <> ["clone", "--", T.unpack (ceUrl env), destDir]
  620       (ec, _out, _err) <-
  621         readProcessBinaryCwdEnv Nothing (ceEnvExtras env) "git" gitArgs BS.empty
  622       pure $ case ec of
  623         ExitSuccess   -> Right ()
  624         ExitFailure n -> Left (CloneGitFailed n)
  625 
  626 -- | @git ls-remote@ a 'SourceRepo' (for @/repo test@). Same seam as
  627 -- 'cloneRepo'; on success returns the first line of stdout
  628 -- (@<sha>\\t<ref>@ — W5 echoes it). On non-zero exit → 'CloneGitFailed' with
  629 -- the exit code ONLY (no stderr).
  630 lsRemoteRepo
  631   :: CloneDeps -> SourceRepo -> IO (Either CloneError Text)
  632 lsRemoteRepo deps repo =
  633   resolveCloneTarget deps repo >>= \case
  634     Left e -> pure (Left e)
  635     Right target -> withCloneTarget target $ \env -> do
  636       let gitArgs = ceGitConfigArgsList env
  637                      <> ["ls-remote", "--", T.unpack (ceUrl env)]
  638       (ec, out, _err) <-
  639         readProcessBinaryCwdEnv Nothing (ceEnvExtras env) "git" gitArgs BS.empty
  640       pure $ case ec of
  641         ExitSuccess   -> Right (firstLine out)
  642         ExitFailure n -> Left (CloneGitFailed n)
  643   where
  644     firstLine bs =
  645       let l = BS.takeWhile (/= nl) bs
  646       in TE.decodeUtf8Lenient l
  647     nl = 10  -- '\n'

  652 ceGitConfigArgsList env = map T.unpack (ceGitConfigArgs env)

    1 -- | The append-only audit-entry model. Integrity comes from the append-only
    2 -- handle plus keeping untrusted actions off the box that holds the log — not
    3 -- from a hash chain. 'encodeEntryRaw' guarantees the on-disk JSONL line is the
    4 -- canonical encoding, so a future "view raw" hides nothing.
    5 module Seal.Transcript.Types
    6   ( Direction (..)
    7   , TranscriptEntry (..)
    8   , encodeEntryRaw
    9   ) where
   10 
   11 import Data.Aeson (FromJSON, ToJSON, Value)
   12 import Data.Aeson qualified as A
   13 import Data.ByteString (ByteString)
   14 import Data.ByteString.Lazy qualified as BL
   15 import Data.Map.Strict (Map)
   16 import Data.Text (Text)
   17 import Data.Time (UTCTime)
   18 import GHC.Generics (Generic)
   19 
   20 import Seal.Core.Types (ModelId)
   21 import Seal.Util.AesonUtils (stripPrefixToJSON, stripPrefixParseJSON)
   22 
   23 data Direction = Request | Response
   24   deriving stock (Eq, Show, Generic)
   25 
   26 instance ToJSON Direction where toJSON = stripPrefixToJSON
   27 instance FromJSON Direction where parseJSON = stripPrefixParseJSON
   28 
   29 -- | JSON keys (via 'stripPrefixToJSON'): @teId@→@id@, @teTimestamp@→@timestamp@,
   30 -- @teModel@→@model@, @teDirection@→@direction@, @tePayload@→@payload@,
   31 -- @teDurationMs@→@durationMs@, @teCorrelation@→@correlation@, @teMeta@→@meta@.
   32 data TranscriptEntry = TranscriptEntry
   33   { teId :: Text
   34   , teTimestamp :: UTCTime
   35   , teModel :: Maybe ModelId
   36   , teDirection :: Direction
   37   , tePayload :: Value
   38   , teDurationMs :: Maybe Int
   39   , teCorrelation :: Maybe Text
   40   , teMeta :: Map Text Value
   41   } deriving stock (Eq, Show, Generic)
   42 
   43 instance ToJSON TranscriptEntry where toJSON = stripPrefixToJSON
   44 instance FromJSON TranscriptEntry where parseJSON = stripPrefixParseJSON
   45 
   46 -- | One JSONL line: the canonical aeson encoding, strict, no trailing newline.
   47 -- The daemon appends the newline when writing.
   48 encodeEntryRaw :: TranscriptEntry -> ByteString
   49 encodeEntryRaw = BL.toStrict . A.encode

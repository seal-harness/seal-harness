    2 -- | BIN_EXEC (Untrusted): run a named binary with a list of argv
    3 -- arguments, via 'System.Process.proc' (RawCommand — no shell
    4 -- interpreter) — on the local or remote backend, selected at wiring time
    5 -- via the 'UntrustedIO' capability. The binary name and args are
    6 -- validated 'BinName' / 'BinArg' (reject empty, NUL). An optional
    7 -- operator-configured allow-list (a 'Set' of permitted binary names)
    8 -- gates the binary; when the allow-list is 'Nothing' the binary is
    9 -- permitted by the gate (the autonomy policy still applies). An optional
   10 -- @cwd@ parameter selects the working directory: a relative path is
   11 -- SafePath-confined to the workspace root (the session's workdir); an
   12 -- absolute path is passed through verbatim. When omitted, the executor
   13 -- defaults to the workspace root. All IO through the 'UntrustedIO' seam;
   14 -- this module never imports 'System.Process'.
   15 --
   16 -- **Git credential injection:** when @binary == "git"@, the opcode
   17 -- resolves the cwd's @remote.origin.url@ (via a pre-flight @git config
   18 -- --get remote.origin.url@ — no auth needed, reads @.git/config@), looks
   19 -- it up in the repo registry, and — if the URL matches a registered repo
   20 -- with a credential — resolves the credential via 'resolveCloneTarget'
   21 -- (deploy key: starts the per-repo ssh-agent + injects @SSH_AUTH_SOCK@ /
   22 -- @GIT_SSH_COMMAND@; PAT: injects @GIT_TERMINAL_PROMPT=0@). The git
   23 -- command then runs via 'uioBinExecGitEnv' (agent forwarding for remote
   24 -- deploy keys) or 'uioBinExecEnv' (PAT) instead of 'uioBinExec', so
   25 -- @git fetch@ / @git pull@ / @git push@ authenticate without the model
   26 -- needing to know the credential mechanism. Unregistered repos and
   27 -- non-git binaries fall through to the plain 'uioBinExec' path.
   28 --
   29 -- **gh credential injection:** when @binary == "gh"@, the opcode mirrors
   30 -- the git path's pre-flight + registry lookup, but differs in the final
   31 -- exec step: for a PAT/MachineUser repo it injects @GH_TOKEN@ (the raw
   32 -- token bytes from 'ceRawToken') into the env via 'uioBinExecEnv' (gh
   33 -- authenticates via @GH_TOKEN@, not @http.extraHeader@ or SSH). Deploy-key
   34 -- repos fall through to plain exec (gh can't use SSH). When the argv
   35 -- contains @-R@ / @--repo@ (gh acts on a different repo than the cwd's
   36 -- @origin@), credential injection is skipped (plain exec + a non-secret
   37 -- NOTE in 'orParts' so the agent learns the skip was deliberate).

   70 -- | BIN_EXEC opcode. Input: @{ binary: BinName, args: [BinArg, ...],
   71 -- cwd?: Text }@. The @args@ field is optional (defaults to @[]@); the
   72 -- @binary@ field is required; the @cwd@ field is optional (defaults to
   73 -- the workspace root). When 'sbAllowList' is 'Nothing', any 'BinName'
   74 -- passes the authorize gate (the autonomy policy still applies); when it
   75 -- is @'Just' set@, the binary must be in @set@ or the gate returns
   76 -- 'Denied'.
   77 binExecOp
   78   :: WorkspaceRoot
   79   -> SecurityPolicy
   80   -> Maybe (Set Text)
   81   -> Opcode
   82 binExecOp wsRoot policy mAllowList = UntrustedOpcode
   83   { uoName = OpName "BIN_EXEC"
   84   , uoDesc = "Run a named binary with argv args (no shell, optional allow-list). Git and gh binaries get credential injection from registered repos."
   85   , uoInSchema = binExecSchema
   86   , uoOutSchema = object []
   87   , uoAuthorize = \v ->
   88       case (binaryField v, argsField v) of
   89         (Nothing, _) -> Left "BIN_EXEC requires {binary:string, args:[string]}"
   90         (Just binText, mArgsText) ->
   91           case spAutonomy policy of
   92             Deny -> Left "BIN_EXEC denied by autonomy policy"
   93             _    -> case mAllowList of
   94               Just allowList
   95                 | binText `Set.notMember` allowList ->
   96                   Left ("BIN_EXEC: binary \"" <> binText <> "\" not in the allow-list")
   97               _ -> case mkBinName binText of
   98                      Left _err -> Left "BIN_EXEC: invalid binary name"
   99                      Right _   -> case traverse mkBinArg <$> mArgsText of
  100                                     Just (Left _err) -> Left "BIN_EXEC: invalid arg"
  101                                     _                -> authorizeCwd wsRoot v
  102   , uoRun = \v -> do
  103       let mBin  = binaryField v
  104           mArgs = argsField v
  105           mCwd  = cwdField v
  106           recorded = object
  107             [ "binary" .= mBin
  108             , "arg_count" .= (fmap length mArgs :: Maybe Int)
  109             , "cwd" .= mCwd
  110             ]
  111       case mBin of
  112         Nothing -> pure (OpResult [TrpText "BIN_EXEC: missing binary"] True recorded)
  113         Just binText ->
  114           case mkBinName binText of
  115             Left err -> pure (OpResult [TrpText ("BIN_EXEC: invalid binary: " <> err)] True recorded)
  116             Right bin ->
  117               case traverse mkBinArg (fromMaybe [] mArgs) of
  118                    Left err -> pure (OpResult [TrpText ("BIN_EXEC: invalid arg: " <> err)] True recorded)
  119                    Right args ->
  120                       case resolveCwd wsRoot mCwd of
  121                         Left _err ->
  122                           pure (OpResult [TrpText "BIN_EXEC: invalid cwd"] True recorded)
  123                         Right mCwdPath -> do
  124                            if textBinName bin == "git"
  125                              then runGitWithCredentials bin args mCwdPath recorded
  126                               else if textBinName bin == "gh"
  127                                 then case args of
  128                                   -- Block dangerous `gh auth` subcommands
  129                                   -- — they write secrets to disk on the
  130                                   -- untrusted machine (gh auth login stores
  131                                   -- a token in ~/.config/gh/hosts.yml; gh
  132                                   -- auth token prints the token to stdout →
  133                                   -- transcript; gh auth setup-git configures
  134                                   -- a credential helper that writes to disk).
  135                                   -- `gh auth status` is read-only and allowed.
  136                                   (firstArg : secondArg : _)
  137                                     | textBinArg firstArg == "auth"
  138                                     , let sub = textBinArg secondArg
  139                                     , sub `elem` ["login", "token", "setup-git", "refresh", "git-credential"] ->
  140                                       pure (OpResult
  141                                       [ TrpText ("BIN_EXEC: `gh auth " <> sub <> "` is blocked — it writes secrets to disk on the untrusted machine (gh auth login stores a token in ~/.config/gh/hosts.yml; gh auth token prints the token to stdout → transcript). The harness injects GH_TOKEN from the vault automatically. Use gh pr create / gh repo clone / etc. directly — they authenticate via the injected GH_TOKEN.")
  142                                       ] True recorded)
  143                                   _ -> runGhWithCredentials bin args mCwdPath recorded
  144                                else do
  145                                  res <- uioBinExec bin args mCwdPath
  146                                  pure $ case res of
  147                                    Left err   -> OpResult [TrpText (renderUntrustedErr err)] True recorded
  148                                    Right out -> OpResult [TrpText out] False recorded
  149   }

  177 -- | Run a @git@ binary with credential injection when the cwd is inside a
  178 -- registered repo. Pre-flight: resolve the repo root + remote URL (no
  179 -- auth needed — reads @.git/config@), look up the URL in the repo
  180 -- registry, and — if found — resolve the credential via
  181 -- 'resolveCloneTarget' and execute via 'uioBinExecGitEnv' (deploy key:
  182 -- agent forwarding) or 'uioBinExecEnv' (PAT). If the URL is not registered
  183 -- or no credential is resolved, fall through to 'uioBinExec'. Errors
  184 -- from the pre-flight or credential resolution are surfaced (not
  185 -- silently swallowed).
  186 runGitWithCredentials
  187   :: BinName -> [BinArg] -> Maybe RemotePath -> Value -> UIO OpResult
  188 runGitWithCredentials bin args mCwdPath recorded = do
  189   -- Pre-flight: resolve the remote URL from the cwd's .git/config.
  190   -- `git config --get remote.origin.url` reads local config only — no
  191   -- network, no auth needed. Parse -C <path> / --git-dir=<path> from the
  192   -- git args so the pre-flight reads config from the right repo (the
  193   -- agent may pass `git -C subdir push ...`).
  194   mRemoteUrl <- resolveRemoteUrl bin args mCwdPath
  195   case mRemoteUrl of
  196     Left err -> pure (OpResult [TrpText err] True recorded)
  197     Right Nothing -> do
  198       -- Not a git repo (or no remote) — fall through to plain exec.
  199       res <- uioBinExec bin args mCwdPath
  200       pure $ case res of
  201         Left err   -> OpResult [TrpText (renderUntrustedErr err)] True recorded
  202         Right out -> OpResult [TrpText out] False recorded
  203     Right (Just remoteUrl) -> do
  204       -- Look up the URL in the repo registry.
  205       eRepos <- uioCdRepoRegList
  206       case eRepos of
  207         Left err -> pure (OpResult [TrpText ("BIN_EXEC: repo registry error: " <> err)] True recorded)
  208         Right repos -> do
  209           let registry = RepoRegistry (Map.fromList [(srId r, r) | r <- repos])
  210               mRepo = lookupRepoByUrl remoteUrl registry
  211           case mRepo of
  212             Nothing -> do
  213               -- URL not registered — fall through to plain exec (public repo).
  214               res <- uioBinExec bin args mCwdPath
  215               pure $ case res of
  216                 Left err   -> OpResult [TrpText (renderUntrustedErr err)] True recorded
  217                 Right out -> OpResult [TrpText out] False recorded
  218             Just repo -> do
  219               -- Resolve the credential via the clone seam.
  220               eTarget <- uioResolveClone repo
  221               case eTarget of
  222                 Left cloneErr ->
  223                   pure (OpResult [TrpText ("BIN_EXEC: credential resolution failed: " <> renderCloneError cloneErr)] True recorded)
  224                 Right target ->
  225                   uioWithClone target $ \cloneEnv -> do
  226                     let envExtras = ceEnvExtras cloneEnv
  227                         mKnownHosts = ceKnownHostsContent cloneEnv
  228                         -- Deploy keys use uioBinExecGitEnv (agent forwarding);
  229                         -- PATs use uioBinExecEnv (no agent, just env overrides
  230                         -- + the http.extraHeader git config args). The
  231                         -- distinction: deploy keys have SSH_AUTH_SOCK in
  232                         -- ceEnvExtras; PATs don't. For PATs, ceGitConfigArgs
  233                         -- carries ["-c", "http.extraHeader=Authorization:
  234                         -- Basic <base64>"] — these MUST be prepended to the
  235                         -- git argv or git push/fetch/pull won't authenticate.
  236                         hasAgent = any (\(k, _) -> k == "SSH_AUTH_SOCK") envExtras
  237                     res <- if hasAgent
  238                              then uioBinExecGitEnv envExtras mKnownHosts bin args mCwdPath
  239                              else do
  240                                let -- Prepend ceGitConfigArgs to the git argv
  241                                    -- so the http.extraHeader (PAT auth) is
  242                                    -- applied. ceGitConfigArgs is a list of
  243                                    -- complete argv tokens (e.g. ["-c",
  244                                    -- "http.extraHeader=..."]); convert each
  245                                    -- to a BinArg via mkBinArg (the literal
  246                                    -- is safe — it's harness-built, not
  247                                    -- user-supplied).
  248                                    configArgs = case traverse mkBinArg (ceGitConfigArgs cloneEnv) of
  249                                      Right as -> as
  250                                      Left _   -> []
  251                                uioBinExecEnv envExtras bin (configArgs ++ args) mCwdPath
  252                     pure $ case res of
  253                       Left err   -> OpResult [TrpText (renderUntrustedErr err)] True recorded
  254                       Right out -> OpResult [TrpText out] False recorded

  256 -- | Run a @gh@ binary with credential injection when the cwd is inside a
  257 -- registered repo with a PAT/MachineUser credential. Mirrors
  258 -- 'runGitWithCredentials' in shape (same pre-flight + registry lookup +
  259 -- credential resolution) but differs in the final exec step: gh
  260 -- authenticates via @GH_TOKEN@ in the env (not @http.extraHeader@ in
  261 -- argv, and not SSH). When the argv contains @-R@ / @--repo@ (gh acts on
  262 -- a different repo than the cwd's @origin@), credential injection is
  263 -- skipped (plain exec + a non-secret NOTE in 'orParts'). Deploy-key repos
  264 -- fall through to plain exec (gh can't use SSH). Unregistered repos and
  265 -- non-git-cwd repos fall through to plain exec. Errors from the
  266 -- pre-flight or credential resolution are surfaced (not silently
  267 -- swallowed). The pre-flight always runs @git@ (not @gh@) — @gh@ doesn't
  268 -- expose @git config@.
  269 runGhWithCredentials
  270   :: BinName -> [BinArg] -> Maybe RemotePath -> Value -> UIO OpResult
  271 runGhWithCredentials bin args mCwdPath recorded =
  272   case extractGhRepoFlag (map textBinArg args) of
  273     Just _ -> do
  274       -- -R/--repo present: gh acts on a different repo than the cwd's
  275       -- origin. Skip injection (plain exec) + surface a non-secret NOTE
  276       -- so the agent learns the skip was deliberate (not an opaque auth
  277       -- failure).
  278       res <- uioBinExec bin args mCwdPath
  279       pure $ case res of
  280         Left err -> OpResult [TrpText (renderUntrustedErr err)] True recorded
  281         Right out -> OpResult
  282           [ TrpText out
  283           , TrpText "BIN_EXEC: gh -R/--repo detected — credential injection skipped (gh uses ambient auth). Run gh from the target repo's workdir for credential injection."
  284           ] False recorded
  285     Nothing -> do
  286       -- Pre-flight: resolve the remote URL from the cwd's .git/config.
  287       -- Uses `git` (not `gh`) — gh doesn't expose `git config`.
  288       mRemoteUrl <- resolveRemoteUrl gitBin args mCwdPath
  289       case mRemoteUrl of
  290         Left err -> pure (OpResult [TrpText err] True recorded)
  291         Right Nothing -> do
  292           -- Not a git repo (or no origin) — fall through to plain exec.
  293           res <- uioBinExec bin args mCwdPath
  294           pure $ case res of
  295             Left err   -> OpResult [TrpText (renderUntrustedErr err)] True recorded
  296             Right out -> OpResult [TrpText out] False recorded
  297         Right (Just remoteUrl) -> do
  298           eRepos <- uioCdRepoRegList
  299           case eRepos of
  300             Left err -> pure (OpResult [TrpText ("BIN_EXEC: repo registry error: " <> err)] True recorded)
  301             Right repos -> do
  302               let registry = RepoRegistry (Map.fromList [(srId r, r) | r <- repos])
  303                   mRepo = lookupRepoByUrl remoteUrl registry
  304               case mRepo of
  305                 Nothing -> do
  306                   -- URL not registered — fall through to plain exec.
  307                   res <- uioBinExec bin args mCwdPath
  308                   pure $ case res of
  309                     Left err   -> OpResult [TrpText (renderUntrustedErr err)] True recorded
  310                     Right out -> OpResult [TrpText out] False recorded
  311                 Just repo -> do
  312                   eTarget <- uioResolveClone repo
  313                   case eTarget of
  314                     Left cloneErr ->
  315                       pure (OpResult [TrpText ("BIN_EXEC: credential resolution failed: " <> renderCloneError cloneErr)] True recorded)
  316                     Right target ->
  317                       uioWithClone target $ \cloneEnv ->
  318                         case ceRawToken cloneEnv of
  319                           Nothing -> do
  320                             -- Deploy key: gh can't use SSH. Fall through
  321                             -- to plain exec (gh surfaces its own auth error).
  322                             res <- uioBinExec bin args mCwdPath
  323                             pure $ case res of
  324                               Left err   -> OpResult [TrpText (renderUntrustedErr err)] True recorded
  325                               Right out -> OpResult [TrpText out] False recorded
  326                           Just tokenBytes -> do
  327                             -- PAT/MachineUser: inject GH_TOKEN (byte-accurate
  328                             -- — map each Word8 to the Char with that
  329                             -- codepoint via toEnum . fromIntegral, avoiding
  330                             -- the U+FFFD corruption of decodeUtf8Lenient on
  331                             -- non-UTF-8 tokens; mirrors renderPatHeader's
  332                             -- raw-bytes discipline).
  333                             let token = map (toEnum . fromIntegral) (BS.unpack tokenBytes)
  334                                 envExtras = ("GH_TOKEN", token)
  335                                           : ceEnvExtras cloneEnv
  336                             res <- uioBinExecEnv envExtras bin args mCwdPath
  337                             pure $ case res of
  338                               Left err   -> OpResult [TrpText (renderUntrustedErr err)] True recorded
  339                               Right out -> OpResult [TrpText out] False recorded
  340   where
  341     -- The pre-flight always runs `git` (not `gh`) — gh doesn't expose
  342     -- `git config`. The literal "git" always parses as a valid BinName.
  343     gitBin = either (error "unreachable: mkBinName rejected a literal") id (mkBinName "git")

  345 -- | Pre-flight: resolve the @remote.origin.url@ from the cwd's git
  346 -- config. Runs @git config --get remote.origin.url@ via 'uioBinExec'
  347 -- (no auth needed — reads @.git/config@). Returns:
  348 --
  349 --   * @Left err@ — the pre-flight git call itself failed (surfacable error).
  350 --   * @Right Nothing@ — the cwd is not inside a git repo, or the repo
  351 --     has no @remote.origin.url@ (fall through to plain exec).
  352 --   * @Right (Just url)@ — the remote URL, trimmed.
  353 --
  354 -- Parses @-C <path>@ and @--git-dir=<path>@ from the git args (the agent
  355 -- may pass @git -C subdir push ...@) so the pre-flight reads config from
  356 -- the right repo, not the workdir root.
  357 resolveRemoteUrl :: BinName -> [BinArg] -> Maybe RemotePath -> UIO (Either Text (Maybe Text))
  358 resolveRemoteUrl bin args mCwdPath = do
  359   let arg t = case mkBinArg t of
  360         Right a -> a
  361         Left _  -> error "unreachable: mkBinArg rejected a literal"
  362       -- Extract -C <path> or --git-dir=<path> from the git args so the
  363       -- pre-flight reads config from the right repo.
  364       mGitDir = extractGitDir (map textBinArg args)
  365       preflightArgs = case mGitDir of
  366         Just dir -> [arg "-C", arg dir, arg "config", arg "--get", arg "remote.origin.url"]
  367         Nothing  -> [arg "config", arg "--get", arg "remote.origin.url"]
  368   res <- uioBinExec bin preflightArgs mCwdPath
  369   pure $ case res of
  370     Left err -> Left ("BIN_EXEC: pre-flight git config failed: " <> renderUntrustedErr err)
  371     Right out ->
  372       let trimmed = T.strip (T.filter (/= '\n') out)
  373       in if T.null trimmed
  374            then Right Nothing
  375            else Right (Just trimmed)

  392 -- | Validate the @cwd@ at the authorize gate. A relative path is
  393 -- SafePath-confined (rejects @..@ escapes + blocked names); an absolute
  394 -- path is accepted verbatim (the operator explicitly chose an
  395 -- out-of-workspace directory). 'mkRemotePath' defends against
  396 -- option-injection (leading dash) + control chars for both kinds.
  397 authorizeCwd :: WorkspaceRoot -> Value -> Either Text ()
  398 authorizeCwd wsRoot v =
  399   case cwdField v of
  400     Nothing -> Right ()
  401     Just cwdText ->
  402       case mkRemotePath cwdText of
  403         Left _err -> Left "BIN_EXEC: cwd must not start with '-'"
  404         Right rp ->
  405           let rel = T.unpack (getRemotePath rp)
  406           in if isAbsolute rel
  407                then Right ()
  408                else case mkSafePathRemote wsRoot rel of
  409                       Left pe  -> Left (cwdEscapeErr pe)
  410                       Right _  -> Right ()
  411 
  412 -- | Resolve the @cwd@ to a 'Maybe RemotePath' for the executor. 'Nothing'
  413 -- means the caller omitted @cwd@ → the executor defaults to the workspace
  414 -- root. @'Just' rp@ carries the validated 'RemotePath' (the arm applies
  415 -- SafePath confinement for relative paths, pass-through for absolute).
  416 resolveCwd :: WorkspaceRoot -> Maybe Text -> Either Text (Maybe RemotePath)
  417 resolveCwd _ Nothing = Right Nothing
  418 resolveCwd wsRoot (Just cwdText) =
  419   case mkRemotePath cwdText of
  420     Left _err -> Left "BIN_EXEC: cwd must not start with '-'"
  421     Right rp ->
  422       let rel = T.unpack (getRemotePath rp)
  423       in if isAbsolute rel
  424            then Right (Just rp)
  425            else case mkSafePathRemote wsRoot rel of
  426                   Left pe  -> Left (cwdEscapeErr pe)
  427                   Right _  -> Right (Just rp)

  429 -- | Render a 'PathError' from cwd validation as a user-facing 'Text'.
  430 -- Maps the two confinement failures to stable messages the authorize
  431 -- gate returns (so the test suite can assert on them).
  432 cwdEscapeErr :: PathError -> Text
  433 cwdEscapeErr (PathEscapesWorkspace _) = "BIN_EXEC: cwd escapes the workspace"
  434 cwdEscapeErr (PathIsBlocked _)        = "BIN_EXEC: cwd touches a blocked location"
  435 cwdEscapeErr (PathDoesNotExist _)     = "BIN_EXEC: cwd does not exist"
  436 cwdEscapeErr (PathInsecureMode _)     = "BIN_EXEC: cwd has insecure mode"

  438 -- | Extract a @-C <path>@ or @--git-dir=<path>@ value from a git argv
  439 -- list. Returns 'Nothing' when neither flag is present (the pre-flight
  440 -- reads config from the workdir root / cwd). Handles both the
  441 -- space-separated form (@-C subdir@) and the joined form (@-Csubdir@,
  442 -- @--git-dir=subdir@). Only the first occurrence is used (git itself
  443 -- takes the last, but for a single git invocation the first is
  444 -- sufficient — the agent doesn't pass multiple -C flags).
  445 extractGitDir :: [Text] -> Maybe Text
  446 extractGitDir = go
  447   where
  448     go [] = Nothing
  449     go (x : xs)
  450       | x == "-C" = case xs of
  451           (p : _) -> Just p
  452           []      -> Nothing
  453       | "-C" `T.isPrefixOf` x
  454       , x /= "-C"
  455       = Just (T.drop 2 x)
  456       | "--git-dir=" `T.isPrefixOf` x
  457       = Just (T.drop (T.length "--git-dir=") x)
  458       | otherwise = go xs

  460 -- | Extract the @-R owner\/repo@ or @--repo owner\/repo@ value from a
  461 -- @gh@ argv list. Returns 'Just' the first value or 'Nothing' when
  462 -- neither flag is present. @gh@ (cobra/pflag) accepts four forms:
  463 --
  464 --   * @-R value@ (space-separated short)
  465 --   * @-Rvalue@ (joined short — pflag supports this for string
  466 --     shorthand flags)
  467 --   * @--repo value@ (space-separated long)
  468 --   * @--repo=value@ (joined long with @=@)
  469 --
  470 -- Scans the ENTIRE argv (global flags may appear after the subcommand:
  471 -- @gh pr create -R owner\/repo@ is valid). Only the first occurrence is
  472 -- used (the return value is only used for the "is @-R@ present?" skip
  473 -- decision, not for credential lookup — first-vs-last is irrelevant).
  474 -- Mirrors 'extractGitDir' (which handles both space and joined forms for
  475 -- @-C@ / @--git-dir=@).
  476 extractGhRepoFlag :: [Text] -> Maybe Text
  477 extractGhRepoFlag = go
  478   where
  479     go [] = Nothing
  480     go (x : xs)
  481       | x == "-R" = case xs of
  482           (v : _) -> Just v
  483           []      -> Nothing
  484       | "-R" `T.isPrefixOf` x
  485       , x /= "-R"
  486       = Just (T.drop 2 x)
  487       | x == "--repo" = case xs of
  488           (v : _) -> Just v
  489           []      -> Nothing
  490       | "--repo=" `T.isPrefixOf` x
  491       = Just (T.drop (T.length "--repo=") x)
  492       | otherwise = go xs

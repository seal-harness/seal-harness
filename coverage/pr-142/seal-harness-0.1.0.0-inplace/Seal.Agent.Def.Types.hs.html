    2 -- | The agent-definition model. 'AgentDefId' is a smart-constructed newtype with
    3 -- the same charset predicate as 'Seal.Core.Types.SessionId'
    4 -- (@[A-Za-z0-9_-]+@, non-empty, no leading dot). An 'AgentDef' is a named
    5 -- configuration (provider + model + system prompt + tool exposure) that a
    6 -- running agent instance is bound to. The definition store is canonical in the
    7 -- Audited log; this module's backend is a materialized view.

   13   , AgentDef (..)
   14   ) where

   29 -- | Opaque agent-definition key. Smart-constructed via 'mkAgentDefId'.
   30 newtype AgentDefId = AgentDefId Text
   31   deriving stock (Eq, Ord, Show)
   32   deriving newtype (ToJSON, FromJSON)
   33 
   34 -- | @[A-Za-z0-9_-]+@, non-empty, no leading dot. Mirrors 'isValidMemoryId'.
   35 isValidAgentDefId :: Text -> Bool
   36 isValidAgentDefId t =
   37   not (T.null t)
   38     && T.head t /= '.'
   39     && T.all (`elem` chars) t
   40   where
   41     chars = ['A' .. 'Z'] <> ['a' .. 'z'] <> ['0' .. '9'] <> "_-"
   42 
   43 mkAgentDefId :: Text -> Either Text AgentDefId
   44 mkAgentDefId t
   45   | isValidAgentDefId t = Right (AgentDefId t)
   46   | otherwise           = Left ("invalid agent def id: " <> T.pack (show t))
   47 
   48 agentDefIdText :: AgentDefId -> Text
   49 agentDefIdText (AgentDefId t) = t

   51 -- | One agent definition. 'adProvider' is a provider label (e.g. @\"ollama\"@);
   52 -- 'adTools' is the opcode allow-list (which opcodes this agent may call). The
   53 -- system prompt and tool list are agent-visible data (not vault secrets); they
   54 -- are recorded in full in both the session transcript and the Audited log.
   55 -- 'adSession' is the originating session (provenance). 'adUpdatedAt' is bumped
   56 -- on each 'AGENT_DEF_UPDATE'.
   57 --
   58 -- 'adGroup' is an optional category for display grouping, mirroring 'skGroup'
   59 -- on 'Seal.Skills.Types.Skill'. It is derived from the on-disk parent
   60 -- directory name (@config\/agents\/\<group\>\/\<id\>.md@) and may be
   61 -- overridden via a @group:@ frontmatter key. 'Nothing' means the def belongs
   62 -- to the default (ungrouped) section. The def /id/ stays flat regardless of
   63 -- group (the charset predicate forbids @\/@), so group is purely display
   64 -- metadata — opcodes, file mapping, and lookups all key on 'adId'.
   65 data AgentDef = AgentDef
   66   { adId        :: AgentDefId
   67   , adName      :: Text
   68   , adProvider  :: Text
   69   , adModel     :: ModelId
   70   , adSystem    :: Maybe Text
   71   , adTools     :: AllowList OpName
   72   , adGroup     :: Maybe Text
   73   , adCreatedAt :: UTCTime
   74   , adUpdatedAt :: UTCTime
   75   , adSession   :: SessionId
   76   } deriving stock (Eq, Show, Generic)
   77 
   78 -- | Encode an 'AllowList OpName' as a JSON value: @\"all\"@ for 'AllowAll', or
   79 -- an array of opcode-name strings for 'AllowOnly'.
   80 allowListToValue :: AllowList OpName -> Value
   81 allowListToValue AllowAll       = String "all"
   82 allowListToValue (AllowOnly xs) = toJSON (map unOpNameText (Set.toList xs))
   83   where
   84     unOpNameText (OpName t) = t
   85 
   86 -- | Decode an 'AllowList OpName' from @\"all\"@ or an array of opcode-name
   87 -- strings. Unknown shapes default to 'AllowAll' (fail-closed would block the
   88 -- agent from calling anything; the wiring layer validates input before
   89 -- construction, so a malformed stored def is treated permissively).
   90 allowListFromValue :: Value -> AllowList OpName
   91 allowListFromValue (String "all") = AllowAll
   92 allowListFromValue (Array xs)     = AllowOnly (Set.fromList [ OpName t | String t <- V.toList xs ])
   93 allowListFromValue _               = AllowAll
   94 
   95 instance ToJSON AgentDef where
   96   toJSON d = object
   97     [ "id"         .= adId d
   98     , "name"       .= adName d
   99     , "provider"   .= adProvider d
  100     , "model"      .= adModel d
  101     , "system"     .= adSystem d
  102     , "tools"      .= allowListToValue (adTools d)
  103     , "group"      .= adGroup d
  104     , "created_at" .= adCreatedAt d
  105     , "updated_at" .= adUpdatedAt d
  106     , "session"    .= adSession d
  107     ]
  108 
  109 instance FromJSON AgentDef where
  110   parseJSON = withObject "AgentDef" $ \o -> AgentDef
  111     <$> o .:  "id"
  112     <*> o .:  "name"
  113     <*> o .:  "provider"
  114     <*> o .:  "model"
  115     <*> o .:? "system"
  116     <*> (allowListFromValue <$> o .: "tools")
  117     <*> o .:? "group"
  118     <*> o .:  "created_at"
  119     <*> o .:  "updated_at"
  120     <*> o .:  "session"

   28 -- | Build a JSON-Schema object with a single required string property.
   29 singleStringSchema :: Text -> Text -> Value
   30 singleStringSchema fieldName fieldDesc =
   31   object
   32     [ "type" .= ("object" :: Text)
   33     , "properties" .= object
   34         [fromText fieldName .= object
   35            [ "type" .= ("string" :: Text)
   36            , "description" .= fieldDesc
   37            ]]
   38     , "required" .= ([fieldName] :: [Text])
   39     ]
   40 
   41 -- | Extract the @name@ string field from a JSON object.
   42 nameField :: Value -> Maybe Text
   43 nameField = parseMaybe (withObject "in" (.: "name"))

   45 -- | SECRET_GET opcode: reads a named secret from the unlocked vault and
   46 -- returns the value to the model. The recorded payload carries only the key
   47 -- name — never the secret value.
   48 secretGetOp :: VaultRuntime -> Opcode
   49 secretGetOp rt = TrustedOpcode
   50   { toName = OpName "SECRET_GET"
   51   , toTrust = Trusted
   52   , toDesc = "Fetch a secret value from the vault by key name."
   53   , toInSchema = singleStringSchema "name" "The vault key name of the secret to fetch."
   54   , toOutSchema = object []
   55   , toAuthorize =
   56       maybe (Left "SECRET_GET requires {name:string}") (const (Right ())) . nameField
   57   , toRun = \_ v -> do
   58       let key = fromMaybe "" (nameField v)
   59       val <- liftIO (vaultGetByName rt key)
   60       pure $ case val of
   61         Left err ->
   62           OpResult [TrpText err] True (object ["name" .= key])
   63         Right secret ->
   64           OpResult [TrpText secret] False (object ["name" .= key])
   65   }

   67 -- | Look up a secret by name from the vault handle in the runtime.
   68 -- Returns 'Left' with a human-readable message if the vault is unconfigured
   69 -- or the key is absent; 'Right' with the UTF-8-decoded value otherwise.
   70 -- The decoded 'Text' is passed directly to 'orParts' and must not reach
   71 -- 'orRecorded'.
   72 vaultGetByName :: VaultRuntime -> Text -> IO (Either Text Text)
   73 vaultGetByName rt key = do
   74   mh <- readIORef (vrHandleRef rt)
   75   case mh of
   76     Nothing -> pure (Left "vault not configured — run /vault setup")
   77     Just h -> do
   78       r <- vhGet h key
   79       pure $ case r of
   80         Left e -> Left (T.pack (show e))
   81         Right bs -> Right (TE.decodeUtf8Lenient bs)

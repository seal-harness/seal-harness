    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | TEXT_TO_SPEECH (Untrusted): a thin abstraction over a pluggable TTS
    3 -- provider (a real TTS model in a future phase). The default provider is
    4 -- fail-closed (@noTtsProvider@).
    5 module Seal.Media.Tts
    6   ( textToSpeechOp
    7   , TtsProvider (..)
    8   , noTtsProvider
    9   ) where
   10 
   11 import Data.Aeson (Value, object, withObject, (.:), (.=))
   12 import Data.Aeson.Types (parseMaybe)
   13 import Data.Maybe (fromMaybe)
   14 import Data.Text (Text)
   15 import Data.Text qualified as T
   16 
   17 import Seal.Core.Types (OpName (..))
   18 import Seal.ISA.Opcode
   19 import Seal.Providers.Class (ToolResultPart (..))
   20 
   21 -- | The TTS-provider interface.
   22 newtype TtsProvider = TtsProvider
   23   { tpSynthesize :: Text -> IO (Either Text Text)   -- ^ text → audio data (base64 or URL)
   24   }
   25 
   26 -- | The fail-closed default provider.
   27 noTtsProvider :: TtsProvider
   28 noTtsProvider = TtsProvider
   29   { tpSynthesize = \_ -> pure (Left "TEXT_TO_SPEECH: no TTS provider configured") }
   30 
   31 -- | TEXT_TO_SPEECH opcode. Input: @{ text: Text }@.
   32 textToSpeechOp :: TtsProvider -> Opcode
   33 textToSpeechOp _prov = UntrustedOpcode
   34   { uoName = OpName "TEXT_TO_SPEECH"
   35   , uoDesc = "Synthesize speech from text (provider-pluggable, fail-closed default)."
   36   , uoInSchema = ttsSchema
   37   , uoOutSchema = object []
   38   , uoAuthorize = \v ->
   39       case textField v of
   40         Nothing -> Left "TEXT_TO_SPEECH requires {text:string}"
   41         Just t | T.null t -> Left "TEXT_TO_SPEECH: text is empty"
   42                | otherwise -> Right ()
   43   , uoRun = \v -> do
   44       let t = fromMaybe "" (textField v)
   45           recorded = object [ "text" .= t ]
   46       pure (OpResult [TrpText "TEXT_TO_SPEECH: no TTS provider configured"] True recorded)
   47   }
   48 
   49 ttsSchema :: Value
   50 ttsSchema =
   51   object
   52     [ "type" .= ("object" :: Text)
   53     , "properties" .= object
   54         [ "text" .= object
   55             [ "type" .= ("string" :: Text)
   56             , "description" .= ("The text to synthesize." :: Text)
   57             ]
   58         ]
   59     , "required" .= (["text"] :: [Text])
   60     ]
   61 
   62 textField :: Value -> Maybe Text
   63 textField = parseMaybe (withObject "in" (.: "text"))

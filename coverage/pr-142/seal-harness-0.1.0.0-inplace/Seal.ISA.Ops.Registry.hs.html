    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The Registry introspection opcode group: OPCODE_DESCRIBE and
    3 -- OPCODE_LIST. These are trusted, read-only opcodes that let the model
    4 -- discover an opcode's full input/output JSON Schema on demand. They are
    5 -- registered (alongside stub @input_schema@s emitted by
    6 -- 'Seal.ISA.Registry.registryToolDefs'') only when the
    7 -- @on_demand_schemas@ config flag is set, so sessions that keep full
    8 -- schemas inline never see them.
    9 --
   10 -- 'OPCODE_DESCRIBE' mirrors the 'SKILL_LOAD' pattern: take an opcode
   11 -- name, render name + description + input schema + output schema as
   12 -- Markdown-ish text the model can read into its context before calling
   13 -- the tool. 'OPCODE_LIST' mirrors 'SKILL_LIST': a cheap name +
   14 -- description summary of every registered opcode (no schema bodies),
   15 -- so the model can pick which opcode to describe without burning the
   16 -- full schema set up front.
   17 module Seal.ISA.Ops.Registry
   18   ( opcodeDescribeOp
   19   , opcodeListOp
   20   ) where
   21 
   22 import Data.Aeson
   23   ( Value, encode, object, withObject, (.:), (.=) )
   24 import Data.Aeson.Key (fromText)
   25 import Data.Aeson.Types (parseMaybe)

   27 import Data.Text (Text)

   35 -- | The full schema rendering used by 'OPCODE_DESCRIBE': a compact,
   36 -- model-readable block with the opcode's name, description, and the
   37 -- compact JSON input/output Schemas. The schemas are encoded as compact
   38 -- JSON (no trailing newline) so the block is easy for the model to parse
   39 -- and small to carry in-context.
   40 describeText :: Opcode -> Text
   41 describeText op =
   42   let OpName n = opName op
   43       d = opDesc op
   44       inJson  = encode (opInSchema op)
   45       outJson = encode (opOutSchema op)
   46   in T.intercalate "\n"
   47        [ "# " <> n
   48        , ""
   49        , d
   50        , ""
   51        , "input_schema:"
   52        , T.pack (BLC.unpack inJson)
   53        , ""
   54        , "output_schema:"
   55        , T.pack (BLC.unpack outJson)
   56        ]
   57 
   58 -- | OPCODE_DESCRIBE: return one opcode's full name, description, and
   59 -- input/output JSON Schemas by name. Trusted — introspection is a
   60 -- read-only control-plane action. The result is a 'TrpText' the model
   61 -- reads before calling the described tool. 'orRecorded' carries the
   62 -- secret-free name + a flag for whether the opcode was found.
   63 opcodeDescribeOp :: Registry -> Opcode
   64 opcodeDescribeOp reg = TrustedOpcode
   65   { toName = OpName "OPCODE_DESCRIBE"
   66   , toTrust = Trusted
   67   , toDesc = "Retrieve the full input/output JSON Schema for one opcode by name. Call this before invoking a tool when its input schema was not included inline."
   68   , toInSchema = object
   69       [ "type" .= ("object" :: Text)
   70       , "properties" .= object
   71           [ fromText "name" .= object
   72               [ "type" .= ("string" :: Text)
   73               , "description" .= ("The opcode name (e.g. FILE_READ)." :: Text)
   74               ]
   75           ]
   76       , "required" .= (["name"] :: [Text])
   77       ]
   78   , toOutSchema = object
   79       [ "type" .= ("object" :: Text)
   80       , "properties" .= object
   81           [ fromText "name" .= object ["type" .= ("string" :: Text)]
   82           , fromText "description" .= object ["type" .= ("string" :: Text)]
   83           , fromText "input_schema" .= object []
   84           , fromText "output_schema" .= object []
   85           ]
   86       ]
   87   , toAuthorize = maybe (Left "OPCODE_DESCRIBE requires {name:string}") (const (Right ())) . nameField
   88   , toRun = \_ v -> do
   89       let mName = nameField v
   90       case mName of
   91         Nothing -> pure (OpResult [TrpText "missing name"] True (object []))
   92         Just nm -> case lookupOp reg (OpName nm) of
   93           Nothing -> pure
   94             ( OpResult [TrpText ("opcode not found: " <> nm)] True
   95             (object ["name" .= nm, "found" .= False]) )
   96           Just op -> do
   97             let rendered = describeText op
   98                 OpName n = opName op
   99                 recorded = object
  100                   [ "name" .= n
  101                   , "found" .= True
  102                   ]
  103             pure (OpResult [TrpText rendered] False recorded)
  104   }

  106 -- | OPCODE_LIST: enumerate every registered opcode's name + description
  107 -- (no schema bodies). Cheap summary so the model can decide which
  108 -- opcode to describe with 'OPCODE_DESCRIBE' without loading all
  109 -- schemas. Trusted and read-only.
  110 opcodeListOp :: Registry -> Opcode
  111 opcodeListOp reg = TrustedOpcode
  112   { toName = OpName "OPCODE_LIST"
  113   , toTrust = Trusted
  114   , toDesc = "List all registered opcodes with their names and one-line descriptions (no schemas). Use OPCODE_DESCRIBE to fetch a specific opcode's full schema."
  115   , toInSchema = object
  116       [ "type" .= ("object" :: Text)
  117       , "properties" .= object []
  118       ]
  119   , toOutSchema = object
  120       [ "type" .= ("array" :: Text)
  121       , "items" .= object
  122           [ "type" .= ("object" :: Text)
  123           , "properties" .= object
  124               [ fromText "name" .= object ["type" .= ("string" :: Text)]
  125               , fromText "description" .= object ["type" .= ("string" :: Text)]
  126               ]
  127           ]
  128       ]
  129   , toAuthorize = const (Right ())
  130   , toRun = \_ _ -> do
  131       let defs = registryToolDefs' False reg
  132           rendered = case defs of
  133             [] -> "(no opcodes registered)"
  134             _  -> T.intercalate "\n"
  135                     [ let OpName n = tdName d in n <> ": " <> tdDescription d
  136                     | d <- defs
  137                     ]
  138           recorded = object
  139             [ "count" .= length defs
  140             , "names" .= [ let OpName n = tdName d in n | d <- defs ]
  141             ]
  142       pure (OpResult [TrpText rendered] False recorded)
  143   }

  149 -- | Extract the @name@ string field from a JSON object.
  150 nameField :: Value -> Maybe Text
  151 nameField = parseMaybe (withObject "in" (.: "name"))

    2 -- | The in-process agent runtime registry. Tracks live subagent instances
    3 -- spawned by AGENT_START so the parent (or operator) can list them, read
    4 -- status, and interrupt by id.
    5 --
    6 -- The registry is keyed by 'SubagentId' (def-id + random suffix), so multiple
    7 -- concurrent children of the same def don't collide. Each 'AgentInstance'
    8 -- binds a 'SubagentId' to its def id, fresh 'SessionId', status, the forked
    9 -- worker's 'ThreadId' (for interrupt), and the parent's delegation depth.
   10 --
   11 -- Scope: in-process instances only — no tmux/harness integration (that is
   12 -- the separate Phase 3 Harnesses group, a different concern). Lifecycle ops
   13 -- (@AGENT_INSTANCES@ / @AGENT_START@ / @AGENT_STATUS@ / @AGENT_STOP@ /
   14 -- @AGENT_INTERRUPT@) are Trusted (not Audited) because running an instance is
   15 -- harness-internal, not an evolutionary mutation.
   16 --
   17 -- 'startAgent' takes the worker action @IO ()@ as a parameter so the
   18 -- registry is testable without dragging in 'Seal.Agent.Loop.runTurn' /
   19 -- 'Seal.Types.Env'. The 'Seal.ISA.Ops.Agent' opcode builds the worker from a
   20 -- fresh 'AgentEnv' over a def's provider/model/system/tools and passes it in.

   23   , AgentInstance (..)

   25   , newAgentRuntime

   27   , registerCompletedAgent

   48 -- | The lifecycle status of a running agent instance. 'Crashed' carries the
   49 -- exception message; 'Interrupted' means the parent/operator requested a
   50 -- stop via 'interruptAgent' (sets a flag the worker polls between turns).
   51 data AgentStatus = Starting | Running | Idle | Stopped | Interrupted | Crashed Text
   52   deriving stock (Eq, Show)
   53 
   54 -- | One running agent instance, bound to a definition, a fresh session, and
   55 -- a subagent id (the registry key).
   56 data AgentInstance = AgentInstance
   57   { aiId         :: AgentDefId
   58     -- ^ The def the child was spawned from (for display / lookup).
   59   , aiSubagentId :: SubagentId
   60     -- ^ The registry key (def-id + random suffix). Unique per spawn.
   61   , aiSession    :: SessionId
   62     -- ^ The fresh session the child runs under.
   63   , aiStatus     :: AgentStatus
   64   , aiThreadId   :: ThreadId
   65     -- ^ The forked worker thread (killed on stop / interrupt).
   66   , aiDepth      :: Int
   67     -- ^ The parent's delegation depth (0 = top-level parent). The child's
   68     -- depth is @aiDepth + 1@; checked against @max_spawn_depth@ before
   69     -- spawning.
   70   } deriving stock (Eq, Show)
   71 
   72 -- | The STM-backed registry of running instances, keyed by 'SubagentId'. No
   73 -- two running instances share a subagent id: 'startAgent' generates a fresh
   74 -- one per spawn. 'stopAgent' removes the entry.
   75 newtype AgentRuntime = AgentRuntime (TVar (Map SubagentId AgentInstance))
   76 
   77 -- | Build an empty runtime.
   78 newAgentRuntime :: IO AgentRuntime
   79 newAgentRuntime = AgentRuntime <$> newTVarIO Map.empty
   80 
   81 -- | Start a new agent instance bound to the given def id + fresh session +
   82 -- subagent id. The worker action is forked; its 'ThreadId' is recorded.
   83 -- Returns @Left err@ if an instance is already running for this subagent id
   84 -- (shouldn't happen since subagent ids are freshly minted, but the check is
   85 -- race-safe). The worker's status transitions to 'Running' once the fork
   86 -- succeeds, or 'Crashed' on exception.
   87 startAgent :: AgentRuntime -> AgentDefId -> SubagentId -> SessionId -> Int -> IO () -> IO (Either Text AgentInstance)
   88 startAgent (AgentRuntime tv) aid subagentId session depth worker = do
   89   tid <- forkIO (runWorker tv subagentId worker)
   90   mInst <- atomically $ do
   91     insts <- readTVar tv
   92     if Map.member subagentId insts
   93       then pure Nothing  -- lost the race; shouldn't happen with random ids
   94       else do
   95         let inst = AgentInstance aid subagentId session Running tid depth
   96         writeTVar tv (Map.insert subagentId inst insts)
   97         pure (Just inst)
   98   case mInst of
   99     Just inst -> pure (Right inst)
  100     Nothing   -> do
  101       killThread tid
  102       pure (Left "agent already running for this subagent id")
  103 
  104 -- | Register a synchronously-completed child in the runtime registry. The
  105 -- synchronous delegation model runs the worker to completion BEFORE this is
  106 -- called (the 'ChildResult' payload IS the post-hoc record), so the instance
  107 -- is recorded with status 'Stopped' (the child has already finished) and the
  108 -- @ThreadId@ of the registering thread (the parent's). This makes the child
  109 -- visible to 'AGENT_INSTANCES' / 'AGENT_STATUS' / 'AGENT_STOP' after the
  110 -- synchronous run completes. 'AGENT_STOP' removes it; 'AGENT_INTERRUPT'
  111 -- flips its status to 'Interrupted' (a no-op for an already-stopped child,
  112 -- but harmless). Idempotent: re-registering the same subagent id overwrites
  113 -- the prior entry (the second registration wins).
  114 registerCompletedAgent
  115   :: AgentRuntime -> AgentDefId -> SubagentId -> SessionId -> Int -> IO ()
  116 registerCompletedAgent (AgentRuntime tv) aid subagentId session depth = do
  117   tid <- myThreadId
  118   atomically $ do
  119     insts <- readTVar tv
  120     let inst = AgentInstance aid subagentId session Stopped tid depth
  121     writeTVar tv (Map.insert subagentId inst insts)
  122 
  123 -- | Run the worker action, transitioning the instance to 'Crashed' on
  124 -- exception. A normal completion leaves the status as 'Running' until the
  125 -- registry's 'stopAgent' is called (the worker is the child turn; reaching
  126 -- its end without exception is a clean completion that the orchestrator
  127 -- observes via 'runDelegate').
  128 runWorker :: TVar (Map SubagentId AgentInstance) -> SubagentId -> IO () -> IO ()
  129 runWorker tv subagentId worker =
  130   catch worker $ \(e :: SomeException) ->
  131     atomically (modifyTVar' tv (Map.adjust (\i -> i { aiStatus = Crashed (T.pack (show e)) }) subagentId))
  132 
  133 -- | Stop a running agent instance: kill the thread, remove the entry.
  134 -- Idempotent (stopping a non-running subagent id is a success with a
  135 -- \"not running\" message, not an error). Always leaves the registry without
  136 -- an entry for this subagent id, so a fresh 'startAgent' can proceed.
  137 stopAgent :: AgentRuntime -> SubagentId -> IO (Either Text ())
  138 stopAgent (AgentRuntime tv) subagentId = do
  139   mInst <- atomically $ do
  140     insts <- readTVar tv
  141     case Map.lookup subagentId insts of
  142       Nothing -> pure Nothing
  143       Just i  -> do
  144         writeTVar tv (Map.delete subagentId insts)
  145         pure (Just i)
  146   case mInst of
  147     Nothing -> pure (Right ())
  148     Just i  -> do
  149       killThread (aiThreadId i)
  150       pure (Right ())
  151 
  152 -- | Request that a single running subagent stop at its next iteration
  153 -- boundary. Does NOT hard-kill the worker thread (Haskell can't safely); sets
  154 -- the instance's status to 'Interrupted' so the worker (which polls its
  155 -- 'crhInterrupted' hook between turns) can exit cleanly. Returns 'True' if a
  156 -- matching subagent was found.
  157 --
  158 -- The worker-builder is responsible for polling the 'crhInterrupted' IORef
  159 -- between turn iterations and returning 'CerInterrupted' when it flips. The
  160 -- registry cannot force the worker to stop — only the worker can decide to
  161 -- exit. 'interruptAgent' /sets the flag and the status/; the worker /observes
  162 -- the flag/.
  163 interruptAgent :: AgentRuntime -> SubagentId -> IO Bool
  164 interruptAgent (AgentRuntime tv) subagentId = do
  165   mInst <- atomically $ do
  166     insts <- readTVar tv
  167     case Map.lookup subagentId insts of
  168       Nothing -> pure Nothing
  169       Just i  -> do
  170         let i' = i { aiStatus = Interrupted }
  171         writeTVar tv (Map.insert subagentId i' insts)
  172         pure (Just i')
  173   pure (isJust mInst)
  174 
  175 -- | Snapshot all running instances.
  176 listAgents :: AgentRuntime -> IO [AgentInstance]
  177 listAgents (AgentRuntime tv) = Map.elems <$> readTVarIO tv

  179 -- | Read one instance's status. @Nothing@ if not running.
  180 agentStatus :: AgentRuntime -> SubagentId -> IO (Maybe AgentStatus)
  181 agentStatus (AgentRuntime tv) subagentId =
  182   fmap aiStatus . Map.lookup subagentId <$> readTVarIO tv

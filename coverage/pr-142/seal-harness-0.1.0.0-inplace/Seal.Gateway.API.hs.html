  151 apiApp :: ApiDeps -> Application
  152 apiApp deps req respond =
  153   case (requestMethod req, pathInfo req) of
  154     (m', ["api", "health"]) | m' == methodGet ->
  155       respond (jsonOk (object
  156         [ "status" .= ("ok" :: Text)
  157         , "wsPort" .= adWsPort deps
  158         ]))

  160       tl <- snapshotTabs (adTabsHandle deps)

  169       thinkingSids <- case adBroker deps of
  170         Just broker -> thinkingSessions broker
  171         Nothing    -> pure Set.empty
  172       snap <- buildListsSnapshot (adTabsHandle deps) (srPaths (adSessionRuntime deps)) thinkingSids
  173       respond (jsonLBS status200 (A.encode snap))

  175     -- pure UI hint persisted as an @archived@ marker file in the session
  176     -- directory (the transcript + session.json stay on disk); 'listSessions'
  177     -- filters those out, and 'listArchivedSessions' returns them via
  178     -- @/api/sessions/archived@.
  179     (m', ["api", "sessions"]) | m' == methodGet -> do
  180       metas <- listSessions (srPaths (adSessionRuntime deps))
  181       infos <- mapM (sessionInfoJsonWithSnippet (srPaths (adSessionRuntime deps))) metas
  182       respond (jsonLBS status200 (A.encode infos))

  186       metas <- listArchivedSessions (srPaths (adSessionRuntime deps))
  187       infos <- mapM (sessionInfoJsonWithSnippet (srPaths (adSessionRuntime deps))) metas
  188       respond (jsonLBS status200 (A.encode infos))

  195     -- ('adSend' = 'Just'), route the message through the real agent loop
  196     -- (slash registry vs plain turn) and return the outcome. When 'adSend'
  197     -- is 'Nothing' (tests without the full runtime), fall back to the stub
  198     -- response so the contract is preserved.
  199     (m', ["api", "sessions", sid, "send"]) | m' == methodPost -> do
  200       body <- collectBody req
  201       case adSend deps of
  202         Nothing -> respond (jsonOk (object ["kind" .= ("assistant" :: Text)]))

  204           let msg = parseSendMessage body
  205           case mkSessionId sid of
  206             Left e   -> respond (errJson status400 ("invalid session id: " <> e))
  207             Right sId -> do
  208               outcome <- handleSend sendDeps sId msg
  209               let (code, val) = sendOutcomeJson outcome
  210               respond (jsonLBS (statusFromInt code) (A.encode val))
  211     -- POST /api/sessions/:id/stop — abort the session's in-flight tool call
  212     -- (and any future tool call until the next turn starts). Returns 200
  213     -- {aborted: true, pending: bool} where pending=true means no turn was
  214     -- in flight (the flag is set but will be cleared at the next runTurn
  215     -- entry). The abort flag is looked up from the SessionAbortRegistry
  216     -- (design Blocker Resolution #2 — ApiDeps has no AgentEnv at this call
  217     -- site). Trust boundary: loopback-only, matching /api/repos.
  218     (m', ["api", "sessions", sid, "stop"]) | m' == methodPost -> do
  219       case mkSessionId sid of
  220         Left e   -> respond (errJson status400 ("invalid session id: " <> e))
  221         Right sId -> do
  222           setSessionAbort (adAbortReg deps) sId
  223           -- pending: whether a turn is currently in flight. Determined via a
  224           -- non-blocking tryReadMVar on the session's lock (from sdLocks if
  225           -- adSend is wired; else assume not-in-flight). True = no turn in
  226           -- flight (the abort will be a no-op, cleared at next runTurn entry).
  227           pending <- case adSend deps of
  228             Just sendDeps -> sessionTurnInFlight (sdLocks sendDeps) sId
  229             Nothing       -> pure True
  230           respond (jsonOk (object
  231             [ "aborted" .= True
  232             , "pending" .= pending
  233             ]))

  235     -- workdir before the first turn (the web "set up repo" combo box calls
  236     -- this). Shares the clone logic with the SETUP_REPO opcode via
  237     -- 'cloneRepoIO'. Returns 200 {ok:true, target, status} on
  238     -- cloned/noop, 409 on a conflict, 400 on an invalid url, 503 on a
  239     -- clone failure. When 'adSend' is Nothing (tests without the full
  240     -- runtime), returns a stub 200.
  241     (m', ["api", "sessions", sid, "setup-repo"]) | m' == methodPost -> do
  242       body <- collectBody req
  243       case parseRepoUrl body of
  244         Nothing -> respond (errJson status400 "missing or invalid 'url' field")
  245         Just url -> case adSend deps of
  246           Nothing -> respond (jsonOk (object ["ok" .= True, "target" .= ("" :: Text), "status" .= ("stubbed" :: Text)]))
  247           Just sendDeps -> case mkSessionId sid of
  248             Left e -> respond (errJson status400 ("invalid session id: " <> e))
  249             Right sId -> do
  250               -- Broadcast agent-defs-changed after a successful SETUP_REPO so
  251               -- the frontend's session-scoped Agent dropdown re-fetches (the
  252               -- clone may have introduced repo-local .agents/ defs). The
  253               -- existing BeAgentDefsChanged event is reused (§3.6).
  254               eRes <- handleSetupRepo sendDeps sId url
  255               when (isRight eRes) $
  256                 broadcastAgentDefsChanged (adBroker deps)
  257               respond (handleSetupRepoResponse eRes)

  259     -- (workdir ⊕ user, workdir-wins). Returns 200 + AgentInfo[] with
  260     -- isDefault per §3.2 (user default_agent > repo agents.md > none);
  261     -- 404 for an unknown session (no session.json). The frontend's
  262     -- Session setup Agent dropdown populates from this (W3).
  263     (m', ["api", "sessions", sid, "agents"]) | m' == methodGet -> do

  265         Left e -> respond (errJson status400 ("invalid session id: " <> e))

  267     -- PUT /api/sessions/:id/description — set or clear the user-set
  268     -- display title for a session (the chat-header pencil). Body:
  269     -- {"description":"<text>"} (empty/missing/whitespace clears it).
  270     -- Persists via 'updateSessionDescription' (a field on session.json)
  271     -- and triggers a lists broadcast so the sidebar picks up the new
  272     -- label without a refresh. Returns 200 {ok:true} on success, 404
  273     -- when the session has no session.json on disk, 400 on invalid JSON.
  274     (m', ["api", "sessions", sid, "description"]) | m' == methodPut -> do
  275       body <- collectBody req
  276       case mkSessionId sid of
  277         Left e  -> respond (errJson status400 ("invalid session id: " <> e))
  278         Right sId -> respond =<< handleSessionDescription deps sId body
  279     -- PUT /api/sessions/:id/archived — set or clear the archive flag. Body:
  280     -- {"archived": true|false}. Persists via 'updateSessionArchived' (a marker
  281     -- file in the session directory). Returns 200 {ok:true} on success, 404
  282     -- when the session has no session.json on disk, 400 on invalid JSON or a
  283     -- missing/non-boolean @archived@ field.
  284     (m', ["api", "sessions", sid, "archived"]) | m' == methodPut -> do
  285       body <- collectBody req
  286       case mkSessionId sid of
  287         Left e  -> respond (errJson status400 ("invalid session id: " <> e))
  288         Right sId -> case parseArchivedFlag body of
  289           Nothing       -> respond (errJson status400 "missing or invalid 'archived' field")

  292     -- override for the session (the Session setup screen's "Use a one-off
  293     -- agent file" upload). Body: {"prompt":"<text>"} (empty/missing
  294     -- clears). When set, 'plainTurn' uses this verbatim as the system

  296     -- on success, 404 when the session doesn't exist.

  299       case mkSessionId sid of
  300         Left e  -> respond (errJson status400 ("invalid session id: " <> e))

  306     -- the session doesn't exist, 400 on an invalid agent id.

  310         Left e  -> respond (errJson status400 ("invalid session id: " <> e))

  323     -- unblocking the agent-loop thread. Body: {"scope":"once|for_session|

  344               eRes <- handleAnswerTextDelivery sendDeps sId qid answerText
  345               case eRes of
  346                 Left e     -> respond (errJson status400 e)
  347                 Right acc -> respond (jsonOk (object ["ok" .= True, "accepted" .= acc]))

  700   case parseKind v of
  701     Nothing       -> pure (errJson status400 "missing or invalid 'kind' field")
  702     Just "shell"  -> pure (errJson status501 "shell/ssh tabs require Phase 4 untrusted execution")
  703     Just "ssh"    -> pure (errJson status501 "shell/ssh tabs require Phase 4 untrusted execution")

  733 -- and return its id. The "Recent Sessions +" button calls this. Body is

  735 -- — so a user who configured `default_provider = "anthropic"` gets
  736 -- anthropic as the default, not a hardcoded fallback. When the body omits

  739 handleSessionNew :: ApiDeps -> A.Value -> IO Response

  764 -- | Handle POST /api/sessions/:id/new — rebind the tab (if any) bound to
  765 -- :id to a fresh session, and return the new sid + tab_index + rebound
  766 -- flag. The old session is kept on disk (still in /session list). 404 when
  767 -- :id doesn't resolve to an on-disk session (prevents orphan-session spam;
  768 -- matches handleSessionPrompt/handleSessionAgent).
  769 handleSessionRebindNew :: ApiDeps -> Text -> IO Response
  770 handleSessionRebindNew deps sidTxt =
  771   case mkSessionId sidTxt of
  772     Left e  -> pure (errJson status400 ("invalid session id: " <> e))
  773     Right oldSid -> do
  774       let paths = srPaths (adSessionRuntime deps)
  775           mp = sessionMetaPath paths oldSid
  776       exists <- doesFileExist mp
  777       if not exists
  778         then pure (errJson status404 "session not found")
  779         else do
  780           mOldMeta <- decodeFileStrict mp :: IO (Maybe SessionMeta)
  781           case mOldMeta of
  782             Nothing -> pure (errJson status404 "session not found")
  783             Just oldMeta -> do
  784               meta <- newSession paths (smProvider oldMeta) (smModel oldMeta) "web" (smAgent oldMeta)
  785               let newSid = smId meta
  786               snap <- snapshotTabs (adTabsHandle deps)
  787               case [ t | t <- tlTabs snap, tRef t == BoundSession oldSid ] of
  788                 []       -> do
  789                   triggerBroadcast deps
  790                   pure (jsonOk (object
  791                     [ "session_id" .= sessionIdText newSid
  792                     , "tab_index" .= (Nothing :: Maybe Int)
  793                     , "rebound" .= False
  794                     ]))
  795                 (tab : _) -> do
  796                   r <- rebindTabH (adTabsHandle deps) (tIndex tab) (BoundSession newSid)

  798                     Left e -> pure (errJson status400 ("tab rebind failed: " <> e))
  799                     Right _ -> do
  800                       triggerBroadcast deps
  801                       pure (jsonOk (object
  802                         [ "session_id" .= sessionIdText newSid
  803                         , "tab_index" .= tabIndexToInt (tIndex tab)
  804                         , "rebound" .= True

  907       -- default_agent is the fallback when no repo agents.md is present.

  910             | Just agentsMd <- agentsMdInUnion defs = Just agentsMd

  927     rightToMaybe (Left _) = Nothing

 1358 -- | Handle POST /api/repos. The body carries @id@, @url@, @vcs_kind@, and a
 1359 -- @credential@ object (@{kind, vault_key, username?}@). Validates the id
 1360 -- ('mkRepoId'), URL shape ('urlShapeValid'), host allow-list
 1361 -- ('parseRepoHost' + 'hostAllowed'), VCS kind, and credential kind. An
 1362 -- existing repo with the same id is REPLACED (idempotent upsert, mirrors
 1363 -- 'handleSkillCreate'). Returns 201 + the created descriptor. After a
 1364 -- successful write, a best-effort @gitCommitAll@ audit-commits
 1365 -- @repos.toml@ (a commit failure is swallowed — the registry write already
 1366 -- succeeded) and a @repos-changed@ WS broadcast is fired.
 1367 handleRepoCreate :: ApiDeps -> BL.ByteString -> IO Response
 1368 handleRepoCreate deps body =
 1369   case A.decode body :: Maybe A.Value of
 1370     Nothing -> pure (errJson status400 "invalid JSON body")
 1371     Just v  -> case parseRepoFromBody v of
 1372       Left e -> pure (errJson status400 e)
 1373       Right repo -> do
 1374         -- Check for generate_key: true (iff credential.kind == deploy_key).
 1375         let genKey = case v of
 1376               A.Object o -> case KeyMap.lookup (Key.fromText "generate_key") o of
 1377                 Just (A.Bool True) -> True
 1378                 _                  -> False
 1379               _ -> False
 1380             isDeployKey = case srCredential repo of
 1381               CredDeployKey _ -> True
 1382               _              -> False
 1383         if genKey && isDeployKey
 1384           then do
 1385             -- Generate the keypair + store the repo with the public key.
 1386             mh <- readIORef (vrHandleRef (adVault deps))
 1387             case mh of
 1388               Nothing -> pure (errJson status500 "vault locked — run /vault unlock")
 1389               Just vh -> do
 1390                 let keyfilePath = repoKeysDir (adPaths deps) </> T.unpack (repoIdText (srId repo))
 1391                     vaultKey = cVaultKey (srCredential repo)
 1392                 eGen <- generateDeployKey vh vaultKey keyfilePath (repoIdText (srId repo))
 1393                 case eGen of
 1394                   Left err -> pure (errJson status500 err)
 1395                   Right pub -> do
 1396                     let repo' = repo { srDeployKeyPublic = Just pub
 1397                                      , srKeyfilePath = Just (T.pack keyfilePath) }
 1398                     eRes <- rrhMutate (adRepoRegistry deps) (upsertRepo repo')
 1399                     case eRes of
 1400                       Left err -> pure (errJson status500 err)
 1401                       Right _  -> do
 1402                         bestEffortCommitRepos deps
 1403                         broadcastReposChanged (adBroker deps)
 1404                         pure (jsonCreated (repoInfoJson repo'))
 1405           else case extractToken v of
 1406             Just _ | isDeployKeyCred (srCredential repo) ->
 1407               -- A token is only meaningful for PAT/MachineUser credentials
 1408               -- (the two token-bearing kinds); a deploy key's secret is
 1409               -- server-generated via generate_key. Fail closed (400).
 1410               pure (errJson status400 "token is only valid for pat/machine_user credentials")
 1411             Just token -> do
 1412               -- PAT/MachineUser: store the token in the vault under the
 1413               -- repo's vault key BEFORE upserting the descriptor (mirrors
 1414               -- generateDeployKey's vhPut-before-upsert ordering). The
 1415               -- descriptor carries only the vault key NAME — never the token.
 1416               mh <- readIORef (vrHandleRef (adVault deps))
 1417               case mh of
 1418                 Nothing -> pure (errJson status500 "vault locked — run /vault unlock")
 1419                 Just vh -> do
 1420                   ePut <- vhPut vh (cVaultKey (srCredential repo)) (TE.encodeUtf8 token)
 1421                   case ePut of
 1422                     Left ve -> pure (errJson status500 ("vault put failed: " <> T.pack (show ve)))
 1423                     Right _ -> do
 1424                       eRes <- rrhMutate (adRepoRegistry deps) (upsertRepo repo)
 1425                       case eRes of
 1426                         Left err -> pure (errJson status500 err)
 1427                         Right _  -> do
 1428                           bestEffortCommitRepos deps
 1429                           broadcastReposChanged (adBroker deps)
 1430                           pure (jsonCreated (repoInfoJson repo))
 1431             Nothing -> do
 1432               -- Standard upsert (no token, no key generation). The vault is
 1433               -- untouched — the operator may have pre-populated the token
 1434               -- out-of-band via /vault add.
 1435               eRes <- rrhMutate (adRepoRegistry deps) (upsertRepo repo)
 1436               case eRes of
 1437                 Left err -> pure (errJson status500 err)
 1438                 Right _  -> do
 1439                   bestEffortCommitRepos deps
 1440                   broadcastReposChanged (adBroker deps)
 1441                   pure (jsonCreated (repoInfoJson repo))

 1443 -- | Handle PUT /api/repos/:id. The path id is authoritative; the body must
 1444 -- carry @url@, @vcs_kind@, and @credential@ (the id may be omitted — it is
 1445 -- taken from the path). The id must ALREADY exist (404 otherwise); ids are
 1446 -- stable (no @new_id@ rename). Returns 200 + the updated descriptor on
 1447 -- success. Same validation + best-effort commit + broadcast as Create.
 1448 handleRepoUpdate :: ApiDeps -> Text -> BL.ByteString -> IO Response
 1449 handleRepoUpdate deps ridTxt body =
 1450   case mkRepoId ridTxt of
 1451     Left e  -> pure (errJson status400 ("invalid repo id: " <> e))
 1452     Right rid -> do
 1453       eRepos <- rrhList (adRepoRegistry deps)
 1454       case eRepos of
 1455         Left err -> pure (errJson status500 err)
 1456         Right rs
 1457           | not (any ((== rid) . srId) rs) -> pure (errJson status404 "repo not found")
 1458           | otherwise -> case A.decode body :: Maybe A.Value of
 1459               Nothing -> pure (errJson status400 "invalid JSON body")
 1460               Just v  -> case parseRepoWithId rid v of
 1461                 Left e -> pure (errJson status400 e)
 1462                 Right repo -> do
 1463                   -- Preserve server-managed deploy-key state (the public
 1464                   -- key + encrypted keyfile path) from the existing entry
 1465                   -- when the PUT body keeps the same deploy-key credential.
 1466                   -- The frontend's edit-mode Save never sends these fields
 1467                   -- (they are generated server-side); without this, the
 1468                   -- upsert would wipe them and the key would "disappear"
 1469                   -- from the UI even though the keyfile still exists on
 1470                   -- disk. Credential vault key names are also stable
 1471                   -- identifiers — match on credential equality.
 1472                   let mOld = findRepo rid rs
 1473                       repo' = case mOld of
 1474                         Just old
 1475                           | srCredential repo == srCredential old
 1476                           , CredDeployKey _ <- srCredential old
 1477                           -> repo { srDeployKeyPublic = srDeployKeyPublic old
 1478                                   , srKeyfilePath     = srKeyfilePath old }
 1479                         _ -> repo
 1480                       kindChanged = case mOld of
 1481                         Just old -> srCredential repo' /= srCredential old
 1482                         Nothing  -> False
 1483                       mOldVaultKey = case mOld of
 1484                         Just old | kindChanged -> Just (cVaultKey (srCredential old))
 1485                         _                       -> Nothing
 1486                   case extractToken v of
 1487                     Just _ | isDeployKeyCred (srCredential repo') ->
 1488                       pure (errJson status400 "token is only valid for pat/machine_user credentials")
 1489                     Just token -> do
 1490                       -- Token present + PAT/MachineUser. If the credential
 1491                       -- kind changed, best-effort delete the old vault key
 1492                       -- (the old entry is inert without a repo descriptor,
 1493                       -- but cleaning it up avoids orphan accumulation). Then
 1494                       -- vhPut the new token under the new vault key BEFORE
 1495                       -- upserting the descriptor (fail-closed 500 on vault
 1496                       -- error, mirroring POST).
 1497                       mh <- readIORef (vrHandleRef (adVault deps))
 1498                       case mh of
 1499                         Nothing -> pure (errJson status500 "vault locked — run /vault unlock")
 1500                         Just vh -> do
 1501                           case mOldVaultKey of
 1502                             Just oldKey -> void (vhDelete vh oldKey)
 1503                             Nothing     -> pure ()
 1504                           ePut <- vhPut vh (cVaultKey (srCredential repo')) (TE.encodeUtf8 token)
 1505                           case ePut of
 1506                             Left ve -> pure (errJson status500 ("vault put failed: " <> T.pack (show ve)))
 1507                             Right _ -> do
 1508                               eRes <- rrhMutate (adRepoRegistry deps) (upsertRepo repo')
 1509                               case eRes of
 1510                                 Left err -> pure (errJson status500 err)
 1511                                 Right _  -> do
 1512                                   bestEffortCommitRepos deps
 1513                                   broadcastReposChanged (adBroker deps)
 1514                                   pure (jsonOk (repoInfoJson repo'))
 1515                     Nothing
 1516                       | kindChanged, not (isDeployKeyCred (srCredential repo')) ->
 1517                         -- Kind changed to a token-bearing credential (PAT or
 1518                         -- MachineUser) but no token was supplied: the new vault
 1519                         -- key has no entry. Fail closed (400) — the operator
 1520                         -- must paste a token to re-authenticate under the new
 1521                         -- kind. (Switching TO a deploy key needs no token —
 1522                         -- its secret is server-generated via generate_key, so
 1523                         -- that falls through to the standard upsert.)
 1524                         pure (errJson status400
 1525                                ("credential kind changed but no token supplied — paste a "
 1526                               <> repoCredentialKindText (srCredential repo')
 1527                               <> " token to re-authenticate"))
 1528                       | otherwise -> do
 1529                         -- No token + kind unchanged (or switching to a deploy
 1530                         -- key): preserve the existing vault entry (the PUT
 1531                         -- doesn't touch the vault).
 1532                         eRes <- rrhMutate (adRepoRegistry deps) (upsertRepo repo')
 1533                         case eRes of
 1534                           Left err -> pure (errJson status500 err)
 1535                           Right _  -> do
 1536                             bestEffortCommitRepos deps
 1537                             broadcastReposChanged (adBroker deps)
 1538                             pure (jsonOk (repoInfoJson repo'))

 1540 -- | Handle DELETE /api/repos/:id. Idempotent: 204 whether or not the repo
 1541 -- existed. 400 on a malformed id. Best-effort @gitCommitAll@ + broadcast
 1542 -- after a successful mutation.
 1543 handleRepoDelete :: ApiDeps -> Text -> IO Response
 1544 handleRepoDelete deps ridTxt =
 1545   case mkRepoId ridTxt of
 1546     Left e  -> pure (errJson status400 ("invalid repo id: " <> e))
 1547     Right rid -> do
 1548       -- Look up the repo BEFORE deleting (to get the keyfile path + vault key
 1549       -- for cleanup).
 1550       eRepos <- rrhList (adRepoRegistry deps)
 1551       case eRepos of
 1552         Left err -> pure (errJson status500 err)
 1553         Right rs -> do
 1554           let mRepo = findRepo rid rs
 1555           -- Delete the encrypted keyfile + .pub (if present) + the vault
 1556           -- passphrase (best-effort — swallow IO errors so a missing file
 1557           -- doesn't block the delete).
 1558           case mRepo of
 1559             Just repo | CredDeployKey vk <- srCredential repo -> do
 1560               -- Delete the encrypted keyfile + .pub.
 1561               case srKeyfilePath repo of
 1562                 Just p -> void (try @SomeException (do
 1563                   let path = T.unpack p
 1564                   _ <- try @SomeException (removeFile path) :: IO (Either SomeException ())
 1565                   _ <- try @SomeException (removeFile (path <> ".pub")) :: IO (Either SomeException ())
 1566                   pure ()))
 1567                 Nothing -> pure ()
 1568               -- Delete the passphrase from the vault (best-effort).
 1569               mh <- readIORef (vrHandleRef (adVault deps))
 1570               case mh of
 1571                 Just vh -> void (try @SomeException (vhDelete vh vk))
 1572                 Nothing -> pure ()
 1573             _ -> pure ()
 1574           -- Delete the repo from the registry.
 1575           _eRes <- rrhMutate (adRepoRegistry deps) (removeRepo rid)
 1576           case _eRes of
 1577             Left err -> pure (errJson status500 err)
 1578             Right _  -> do
 1579               bestEffortCommitRepos deps
 1580               broadcastReposChanged (adBroker deps)
 1581               pure noContent

 1849 -- | T11: handle GET /api/sessions/:id/transcript. Returns the session's
 1850 -- transcript as the frontend's @TranscriptEntry@ shape
 1851 -- (@id@/@timestamp@/@direction@/@payload@/@model@/@harness@/@raw@).
 1852 --
 1853 -- Two on-disk formats are supported:
 1854 --   1. The legacy @transcript.jsonl@ (one Haskell 'TranscriptEntry' per line,
 1855 --      @te*@-prefixed fields). Each line is mapped to the frontend shape.
 1856 --   2. The new two-file @conversation.jsonl@ (one 'Message' per line,
 1857 --      @msgRole@/@msgContent@). Each line is synthesized into a
 1858 --      @TranscriptEntry@-shaped JSON (User → request, Assistant → response);
 1859 --      timestamps/model are not available in this file so they're set to
 1860 --      the session's @smCreatedAt@/@smModel@ from @session.json@.
 1861 --
 1862 -- Missing files -> @[]@; unparseable lines are skipped.
 1863 --
 1864 -- Emits a @Server-Timing@ header (per <https://www.w3.org/TR/server-timing/>)
 1865 -- with per-phase durations (file-read / parse / reconstruct / rewrite / total)
 1866 -- and the transcript source (legacy / conv+entries / conv-only / missing) +
 1867 -- entry count. The frontend parses this via @performance.getEntriesByName@ or
 1868 -- by reading the @Server-Timing@ response header directly to direct
 1869 -- optimization work without needing a separate tracing harness.
 1870 handleTranscript :: ApiDeps -> Text -> IO Response
 1871 handleTranscript deps sidTxt =
 1872   case mkSessionId sidTxt of
 1873     Left _  -> pure (jsonLBS status200 (A.encode ([] :: [Value])))
 1874     Right sid -> do
 1875       let paths = srPaths (adSessionRuntime deps)
 1876       active <- readIORef (srActive (adSessionRuntime deps))
 1877       tReadStart <- getCurrentTime
 1878       (entries, tt0) <- readTranscriptEntriesTimed paths (smModel active) (showIso (smCreatedAt active)) sid
 1879       tEncStart <- getCurrentTime
 1880       let body = A.encode entries
 1881       tEncEnd <- getCurrentTime
 1882       -- Fold the encode duration + the gap between read-complete and
 1883       -- encode-start (negligible) into the timings so the @en@ token
 1884       -- reflects the JSON-serialization cost and @tt@ reflects the full
 1885       -- read+encode cost. This is where the slow tab's time is hiding —
 1886       -- A.encode of a deeply-nested Value tree (web_fetch tool_result
 1887       -- carrying full page HTML) is pathologically slow on large strings.
 1888       let tt = setEncodeMs (msDiff tEncStart tEncEnd) (msDiff tReadStart tEncEnd) tt0
 1889           timingHeader = (mkHN "Server-Timing", renderServerTiming tt)
 1890       pure (responseLBS status200 (corsHeaders <> [jsonHeader, timingHeader]) body)
 1891   where
 1892     msDiff a b = round (realToFrac (b `diffUTCTime` a) * 1000 :: Double)

 2115 -- | Generate a new deploy keypair: random 32-byte passphrase (base64) →
 2116 -- vhPut the passphrase under the vault key → ssh-keygen -t ed25519 -f
 2117 -- <keyfilePath> -N "<passphrase>" -C "seal-deploy-key:<id>" → read the .pub
 2118 -- file. Returns 'Just <pubkey>' on success, 'Left <err>' on failure. The
 2119 -- encrypted keyfile is ciphertext (stays on the harness disk; the
 2120 -- passphrase lives in the vault). The .pub is public data.
 2121 generateDeployKey :: VaultHandle -> Text -> FilePath -> Text -> IO (Either Text Text)
 2122 generateDeployKey vh vaultKey keyfilePath repoId = do
 2123   -- Ensure the parent directory exists (ssh-keygen won't create it).
 2124   let keyfilesDir = takeDirectory keyfilePath
 2125   createDirectoryIfMissing True keyfilesDir
 2126   -- Remove any existing keyfile + .pub so ssh-keygen does not prompt
 2127   -- "Overwrite (y/n)?" interactively (the server pipes empty stdin, so
 2128   -- ssh-keygen would exit 1 on the prompt). Rotation semantics: the new
 2129   -- keypair replaces the old.
 2130   let pubPath = keyfilePath <> ".pub"
 2131   ignoringDoesNotExist $ removeFile keyfilePath
 2132   ignoringDoesNotExist $ removeFile pubPath
 2133   -- Generate a random 32-byte passphrase, base64-encoded.
 2134   rawPass <- randomByteString 32
 2135   let passphrase = TE.decodeUtf8Lenient (B64.encode rawPass)
 2136   -- Store the passphrase in the vault.
 2137   ePut <- vhPut vh vaultKey (TE.encodeUtf8 passphrase)
 2138   case ePut of
 2139     Left ve -> pure (Left ("vault put failed: " <> T.pack (show ve)))
 2140     Right _ -> do
 2141       -- Run ssh-keygen to write the encrypted keyfile + .pub.
 2142       let args = ["-t", "ed25519", "-f", keyfilePath, "-N", T.unpack passphrase
 2143                  , "-C", "seal-deploy-key-" <> T.unpack repoId]
 2144       (ec, _out, err) <- readCreateProcessWithExitCode (proc "ssh-keygen" args) ""
 2145       case ec of
 2146         ExitSuccess -> do
 2147           -- Read the .pub file.
 2148           pubExists <- doesFileExist pubPath
 2149           if pubExists
 2150             then do
 2151               pubBytes <- BS.readFile pubPath
 2152               pure (Right (TE.decodeUtf8Lenient pubBytes))
 2153             else pure (Left "ssh-keygen succeeded but .pub file not found")
 2154         ExitFailure n -> pure (Left ("ssh-keygen failed (exit " <> T.pack (show n) <> "): " <> T.pack err))
 2155 
 2156 -- | Remove a file, swallowing 'isDoesNotExistError' so callers don't need
 2157 -- to pre-check existence. Avoids a TOCTOU race between doesFileExist and
 2158 -- removeFile (two processes could both delete the same keyfile).
 2159 ignoringDoesNotExist :: IO () -> IO ()
 2160 ignoringDoesNotExist act = void (try act :: IO (Either SomeException ()))

 2164 randomByteString n = BS.pack <$> replicateM n (randomRIO (0, 255))

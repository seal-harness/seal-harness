   83 -- | Handle to the legacy single-file transcript writer. All fields are
   84 -- functions so the type is uniform between the real (file-backed) and fake

   96 -- | A unit of work for the single-writer daemon. 'Write' carries an entry and
   97 -- an optional ack slot (filled after fsync). 'Shutdown' tells the daemon to
   98 -- drain any remaining queued writes, fill its ack slot, and exit.
   99 data Item
  100   = Write TranscriptEntry (Maybe (TMVar ()))
  101   | Shutdown (TMVar ())

  103 -- | Write the strict 'ByteString' to 'fd' using the Posix primitive, which
  104 -- bypasses any user-space buffer. Loops until the whole buffer is written,
  105 -- advancing past each (possibly short) write; 'fdWriteBuf' itself retries the
  106 -- underlying syscall on @EINTR@, so an interrupted write resumes rather than
  107 -- truncating the line.
  108 writeFd :: Fd -> BS.ByteString -> IO ()
  109 writeFd fd bs =
  110   BSU.unsafeUseAsCStringLen bs $ \(ptr, len) ->
  111     go (castPtr ptr) len
  112   where
  113     go :: Ptr Word8 -> Int -> IO ()
  114     go _ 0 = pure ()
  115     go ptr remaining = do
  116       written <- fromIntegral <$> fdWriteBuf fd ptr (fromIntegral remaining)
  117       go (ptr `plusPtr` written) (remaining - written)

  119 -- | Open the transcript file in O_APPEND mode, spawn the single-writer daemon,
  120 -- run @action@, then close. Every 'recordAndAck' blocks until 'fileSynchronise'
  121 -- returns, guaranteeing the entry is durable before the caller proceeds.
  122 withTranscript :: FilePath -> (TranscriptHandle -> IO a) -> IO a
  123 withTranscript path action = do
  124   q <- newTQueueIO
  125   let flags = defaultFileFlags
  126         { append = True
  127         , creat = Just (0o600 :: FileMode)
  128         }
  129       ack = maybe (pure ()) (\tv -> atomically (putTMVar tv ()))
  130       writeEntry fd e = do
  131         writeFd fd (encodeEntryRaw e <> "\n")
  132         fileSynchronise fd
  133       -- Drain every currently-queued write (used at shutdown), fsyncing and
  134       -- acking each, so nothing is lost before the fd is closed.
  135       drain fd = do
  136         next <- atomically (tryReadTQueue q)
  137         case next of
  138           Nothing -> pure ()
  139           Just (Write e mack) -> writeEntry fd e >> ack mack >> drain fd
  140           Just (Shutdown done) -> atomically (putTMVar done ()) >> drain fd
  141       -- Single-writer daemon: process writes; on Shutdown, drain the queue,
  142       -- signal completion, and exit (no 'forever', so it stops deterministically).
  143       daemon fd = do
  144         item <- atomically (readTQueue q)
  145         case item of
  146           Write e mack -> writeEntry fd e >> ack mack >> daemon fd
  147           Shutdown done -> drain fd >> atomically (putTMVar done ())
  148       -- Cleanup: tell the daemon to drain+exit and WAIT for it before closing
  149       -- the fd, so no queued write races (or writes to) a closed fd.
  150       shutdown fd = do
  151         done <- newEmptyTMVarIO
  152         atomically (writeTQueue q (Shutdown done))
  153         atomically (takeTMVar done)
  154         closeFd fd
  155   bracket (openFd path WriteOnly flags) shutdown $ \fd -> do
  156     void $ forkIO (daemon fd)
  157     let enqueue e = atomically (writeTQueue q (Write e Nothing))
  158         ackWrite e = do
  159           tv <- newEmptyTMVarIO
  160           atomically (writeTQueue q (Write e (Just tv)))
  161           atomically (takeTMVar tv)
  162     action TranscriptHandle
  163       { recordAndAck = ackWrite
  164       , recordAsync = enqueue
  165       , closeTranscript = pure ()
  166       }

  187 -- | A structured write request for the two-file writer. The writer diffs
  188 -- @tfwMessages@ against the conversation as it exists on disk and appends only
  189 -- the new lines (with @CbToolResult@ parts redacted), then appends the entry
  190 -- record. Crash semantics: messages first, then the entry line, so a torn tail
  191 -- leaves at most orphan message lines / a malformed last line — both already
  192 -- tolerated by the skip-malformed decode path.
  193 data TwoFileWrite = TwoFileWrite
  194   { tfwMessages :: [Message]
  195   -- ^ The full message list in effect at this turn (the conversation prefix +
  196   -- any new messages). The writer diffs against the on-disk conversation and
  197   -- appends only the new lines.
  198   , tfwEntry    :: EntryRecord
  199   -- ^ The event line to append to @entries.jsonl@.
  200   }

  202 -- | Handle to the two-file writer. 'tfwRecordAndAck' blocks until both files
  203 -- are fsync'd; 'tfwRecordAsync' is fire-and-forget.
  204 data TwoFileHandle = TwoFileHandle
  205   { tfwRecordAndAck :: TwoFileWrite -> IO ()
  206   , tfwRecordAsync  :: TwoFileWrite -> IO ()
  207   , tfwReadConversation :: IO [Message]
  208   -- ^ Read back the on-disk conversation (for tests / replay).
  209   , tfwReadEntries     :: IO [EntryRecord]
  210   -- ^ Read back the on-disk entry log.
  211   , tfwSetSecretOps    :: Set OpName -> IO ()
  212   -- ^ Set the set of opcode names whose tool results should be redacted from
  213   -- the on-disk conversation file. Call this after building the ISA registry
  214   -- so the writer knows which 'CbToolResult's may carry secrets. Results from
  215   -- opcodes NOT in this set (e.g. SHELL_EXEC, FILE_READ) pass through verbatim.
  216   , tfwCloseTranscript :: IO ()
  217   , tfwIsAlive :: IO Bool
  218   -- ^ Is the single-writer daemon still running? Returns 'False' after a
  219   -- write error killed it. Callers that depend on durability (the dispatch
  220   -- ACK-before-execute gate) should check this before trusting a prior write.
  221   }

  223 -- | Default ACK timeout (30 seconds). If the daemon does not acknowledge a
  224 -- write within this window, 'tfwRecordAndAck' raises 'TranscriptError' instead
  225 -- of hanging forever. Tuned generously — normal writes complete in
  226 -- milliseconds; 30s is a fail-safe for a stuck/dead daemon, not a performance
  227 -- bound.
  228 defaultAckTimeoutUs :: Int
  229 defaultAckTimeoutUs = 30_000_000

  231 -- | The daemon's accumulated state: the conversation lines written so far (so
  232 -- the diff can be computed without re-reading the file each turn) and the
  233 -- fsync'd fds.
  234 data TwoFileState = TwoFileState
  235   { tfsConvFd :: Fd
  236   , tfsEntriesFd :: Fd
  237   , tfsWritten :: [Message]
  238   -- ^ The conversation as it exists on disk, in order. Grown by each write
  239   -- so the next diff can be computed in-memory.
  240   , tfsSecretOpsRef :: IORef (Set OpName)
  241   -- ^ The set of opcode names whose tool results may carry secrets and must
  242   -- be redacted from the on-disk conversation file. An IORef so the caller
  243   -- can set it after building the registry (the registry construction may
  244   -- depend on values only available inside the 'withTwoFileTranscript'
  245   -- callback, e.g. session id or worker functions).
  246   , tfsPriorEnv :: Maybe Envelope
  247   -- ^ The effective envelope at the most recent 'EKRequest' entry. Used by
  248   -- 'writeOne' to compute a minimal 'EnvelopeDelta' — only fields that
  249   -- differ from the prior envelope are emitted, so a stable system prompt
  250   -- + tools array isn't re-serialized on every turn (a 146-turn session
  251   -- with a 36KB system prompt would otherwise waste ~5MB). 'Nothing' until
  252   -- the first request entry lands.
  253   }

  255 -- | A work item for the two-file daemon.
  256 data TwoFileItem
  257   = TFWWrite TwoFileWrite (Maybe (TMVar ()))
  258   | TFWShutdown (TMVar ())

  260 -- | Open both files in O_APPEND mode, spawn the single-writer daemon, run
  261 -- @action@, then close. Every 'tfwRecordAndAck' blocks until both files are
  262 -- fsync'd.
  263 --
  264 -- If the session directory already contains a legacy @transcript.jsonl@ and no
  265 -- @conversation.jsonl@, the legacy file is left untouched (the legacy read path
  266 -- handles it); the new writer simply creates the two new files alongside it.
  267 withTwoFileTranscript :: FilePath -> (TwoFileHandle -> IO a) -> IO a
  268 withTwoFileTranscript dir action = do
  269   q <- newTQueueIO
  270   aliveRef <- newTVarIO True
  271   let convPath     = dir </> "conversation.jsonl"
  272       entriesPath  = dir </> "entries.jsonl"
  273       flags = defaultFileFlags
  274         { append = True
  275         , creat = Just (0o600 :: FileMode)
  276         }
  277       ack = maybe (pure ()) (\tv -> atomically (putTMVar tv ()))
  278       writeOne st (TwoFileWrite msgs entry) = do
  279         secretOps <- readIORef (tfsSecretOpsRef st)
  280         let new = diffMessages msgs (tfsWritten st)
  281             redacted = redactMessages secretOps msgs new
  282         -- 1. Append new conversation lines, fsync.
  283         mapM_ (\m -> writeFd (tfsConvFd st) (encodeConvLine (ConvLine m) <> "\n")) redacted
  284         fileSynchronise (tfsConvFd st)
  285         -- 2. Compute a minimal envelope delta for EKRequest entries. The
  286         -- caller passes a FULL delta (every field set to 'Just'); the writer
  287         -- compares it against the prior effective envelope and emits only
  288         -- the fields that differ. This keeps a stable system prompt + tools
  289         -- array from being re-serialized on every turn (a 146-turn session
  290         -- with a 36KB system prompt would otherwise waste ~5MB). The first
  291         -- request (no prior envelope) emits the full delta so
  292         -- reconstruction has a baseline to fold from.
  293         let (entry', mNextEnv) = case erKind entry of
  294               EKRequest -> case erEnvelope entry of
  295                 Just delta ->
  296                   let baseline = fromMaybe defaultEnv (tfsPriorEnv st)
  297                       nextEnv = applyDelta baseline delta
  298                       minimal = minimalDelta (tfsPriorEnv st) nextEnv
  299                   in (entry { erEnvelope = minimal }, Just nextEnv)
  300                 Nothing -> (entry, tfsPriorEnv st)
  301               _ -> (entry, tfsPriorEnv st)
  302         -- 3. Append the entry line, fsync.
  303         writeFd (tfsEntriesFd st) (encodeEntryRecordRaw entry' <> "\n")
  304         fileSynchronise (tfsEntriesFd st)
  305         pure st { tfsWritten = tfsWritten st <> redacted, tfsPriorEnv = mNextEnv }
  306       drain st = do
  307         next <- atomically (tryReadTQueue q)
  308         case next of
  309           Nothing -> pure st
  310           Just (TFWWrite w mack) -> do
  311             st' <- writeOne st w
  312             ack mack
  313             drain st'
  314           Just (TFWShutdown done) -> do
  315             atomically (putTMVar done ())
  316             drain st
  317       daemon st = do
  318         item <- atomically (readTQueue q)
  319         case item of
  320           TFWWrite w mack -> do
  321             st' <- writeOne st w
  322             ack mack
  323             daemon st'
  324           TFWShutdown done -> do
  325             st' <- drain st
  326             atomically (putTMVar done ())
  327             pure st'
  328       shutdown convFd entriesFd done = do
  329         atomically (writeTQueue q (TFWShutdown done))
  330         atomically (takeTMVar done)
  331         closeFd convFd
  332         closeFd entriesFd
  333   -- Read existing conversation (if any) so the diff starts from the right
  334   -- baseline. A torn legacy file or a fresh session yields [].
  335   existingConv <- do
  336     exists <- doesFileExist convPath
  337     if exists
  338       then readConversation <$> BS.readFile convPath
  339       else pure []
  340   bracket
  341     (do convFd     <- openFd convPath    WriteOnly flags
  342         entriesFd  <- openFd entriesPath WriteOnly flags
  343         pure (convFd, entriesFd))
  344     (\(convFd, entriesFd) -> do
  345         done <- newEmptyTMVarIO
  346         shutdown convFd entriesFd done)
  347     $ \(convFd, entriesFd) -> do
  348         secretOpsRef <- newIORef Set.empty
  349         let st0 = TwoFileState convFd entriesFd existingConv secretOpsRef Nothing
  350             -- The daemon with an exception handler: if writeOne throws, mark
  351             -- the daemon dead, fail every queued ACK (so callers blocked on
  352             -- takeTMVar unblock immediately), and log to stderr. The daemon
  353             -- then exits — no further writes are attempted.
  354             safeDaemon st =
  355               daemon st `catch` \e -> do
  356                 atomically (writeTVar aliveRef False)
  357                 globalLogIO ErrorS ("[transcript] writer daemon died: " <> ls (T.pack (show (e :: SomeException))))
  358                 -- Drain remaining queue items WITHOUT filling their ACK slots
  359                 -- (the writes never completed). Callers blocked on takeTMVar
  360                 -- unblock via the aliveRef check in ackWrite's orElse. Shutdown
  361                 -- items still get their done signal so the bracket cleanup
  362                 -- completes.
  363                 let drainRemaining = do
  364                       item <- atomically (tryReadTQueue q)
  365                       case item of
  366                         Nothing -> pure ()
  367                         Just (TFWWrite _ _mack) -> drainRemaining
  368                         Just (TFWShutdown done) -> do
  369                           atomically (putTMVar done ())
  370                 drainRemaining
  371                 pure st
  372         void $ forkIO (void (safeDaemon st0))
  373         let enqueue w = atomically (writeTQueue q (TFWWrite w Nothing))
  374             ackWrite w = do
  375               alive <- readTVarIO aliveRef
  376               if not alive
  377                 then throwIO (TranscriptError "transcript writer daemon is dead")
  378                 else do
  379                   tv <- newEmptyTMVarIO
  380                   timeoutVar <- newEmptyTMVarIO
  381                   atomically (writeTQueue q (TFWWrite w (Just tv)))
  382                   void $ forkIO $ do
  383                     threadDelay defaultAckTimeoutUs
  384                     void $ atomically (tryPutTMVar timeoutVar ())
  385                   mResult <- atomically $
  386                     (Right <$> takeTMVar tv)
  387                       `orElse` (Left <$> takeTMVar timeoutVar)
  388                       `orElse` (do
  389                          stillAlive <- readTVar aliveRef
  390                          if stillAlive then retry else pure (Left ()))
  391                   case mResult of
  392                     Right () -> pure ()
  393                     Left ()  -> do
  394                       stillAlive <- readTVarIO aliveRef
  395                       if stillAlive
  396                         then throwIO (TranscriptError "transcript write timed out (daemon stuck)")
  397                         else throwIO (TranscriptError "transcript writer daemon died during write")
  398         action TwoFileHandle
  399           { tfwRecordAndAck = ackWrite
  400           , tfwRecordAsync  = enqueue
  401           , tfwReadConversation = readConversation <$> BS.readFile convPath
  402           , tfwReadEntries     = readEntries entriesPath
  403           , tfwSetSecretOps    = writeIORef secretOpsRef
  404           , tfwCloseTranscript = pure ()
  405           , tfwIsAlive = readTVarIO aliveRef
  406           }

  449 -- | The new-message suffix beyond the written conversation prefix. Mirrors
  450 -- 'Seal.Transcript.Conv.diffNew' (kept local so this module is self-contained).
  451 diffMessages :: [Message] -> [Message] -> [Message]
  452 diffMessages incoming written = fromMaybe incoming (stripPrefixMsg written incoming)
  453   where
  454     stripPrefixMsg [] is             = Just is
  455     stripPrefixMsg _  []             = Nothing
  456     stripPrefixMsg (w:ws) (i:is)
  457       | w == i    = stripPrefixMsg ws is
  458       | otherwise = Nothing

  460 -- | Redact tool-result parts that may carry secret values. Only results from
  461 -- secret-producing opcodes (those whose name is in the provided set) are
  462 -- redacted — Untrusted opcodes (SHELL_EXEC, FILE_READ, etc.) produce non-secret
  463 -- output that passes through verbatim so the frontend can display it.
  464 --
  465 -- The @allMsgs@ list (the full conversation) is scanned for 'CbToolUse' blocks
  466 -- whose 'cbName' is in the secret-opcode-name set; their 'cbId's form the set
  467 -- of tool-call ids whose results should be redacted. Only the @newMsgs@ are
  468 -- returned (after redaction); @allMsgs@ is only used for the scan.
  469 redactMessages :: Set OpName -> [Message] -> [Message] -> [Message]
  470 redactMessages secretOps allMsgs newMsgs =
  471   let secretIds = buildSecretIds secretOps allMsgs
  472   in map (redactMessage secretIds) newMsgs
  473   where
  474     buildSecretIds :: Set OpName -> [Message] -> Set ToolCallId
  475     buildSecretIds ops = foldr go mempty
  476       where
  477         go (Message _ blocks) acc = acc <> foldr goBlock mempty blocks
  478           where
  479             goBlock (CbToolUse tcid name _) a
  480               | name `Set.member` ops = tcid `Set.insert` a
  481               | otherwise             = a
  482             goBlock _ a = a
  483 
  484     redactMessage :: Set ToolCallId -> Message -> Message
  485     redactMessage secretIds (Message role blocks) =
  486       Message role (map (redactBlock secretIds) blocks)
  487       where
  488         redactBlock ids (CbToolResult tcid parts isErr)
  489           | tcid `Set.member` ids = CbToolResult tcid (map redactPart parts) isErr
  490           | otherwise             = CbToolResult tcid parts isErr
  491         redactBlock _ other = other
  492 
  493         redactPart (TrpText t)
  494           | not (T.null t) = TrpText "<redacted:secret>"
  495           | otherwise      = TrpText t

  497 -- | A neutral baseline envelope used when no prior envelope exists (the
  498 -- first request). Mirrors 'Seal.Transcript.Reconstruct.defaultEnv'.
  499 defaultEnv :: Envelope
  500 defaultEnv = Envelope (ModelId "") Nothing [] ToolAuto 0

  502 -- | Compute the minimal 'EnvelopeDelta' that, when applied to the prior
  503 -- envelope, yields the next envelope. 'Nothing' fields mean "inherit" — so
  504 -- a field that's unchanged from the prior is emitted as 'Nothing' and
  505 -- omitted from the on-disk JSON entirely. When there is no prior envelope
  506 -- (the first request), the full delta is returned so reconstruction has a
  507 -- baseline to fold from.
  508 minimalDelta :: Maybe Envelope -> Envelope -> Maybe EnvelopeDelta
  509 minimalDelta mPrior next = case mPrior of
  510   Nothing -> Just EnvelopeDelta
  511     { edModel = Just (envModel next)
  512     , edSystem = Just (envSystem next)
  513     , edTools = Just (envTools next)
  514     , edToolChoice = Just (envToolChoice next)
  515     , edMaxTokens = Just (envMaxTokens next)
  516     }
  517   Just prior ->
  518     let d = EnvelopeDelta
  519           { edModel = if envModel next == envModel prior then Nothing else Just (envModel next)
  520           , edSystem = if envSystem next == envSystem prior then Nothing else Just (envSystem next)
  521           , edTools = if envTools next == envTools prior then Nothing else Just (envTools next)
  522           , edToolChoice = if envToolChoice next == envToolChoice prior then Nothing else Just (envToolChoice next)
  523           , edMaxTokens = if envMaxTokens next == envMaxTokens prior then Nothing else Just (envMaxTokens next)
  524           }
  525     in if d == emptyEnvelopeDelta then Nothing else Just d

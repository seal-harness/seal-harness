    3 -- | The SSH-agent capability seam — a record-of-IO-actions (mirrors
    4 -- 'Seal.Tools.Exec.Remote.RemoteRunner' and 'Seal.Security.Vault.VaultHandle')
    5 -- so the ssh-agent lifecycle is testable without spawning a real
    6 -- @ssh-agent@ / @ssh-add@ (unit-test-speed; no process spawning in the
    7 -- test suite).
    8 --
    9 -- Security model (one agent per repo, key persists for the harness lifetime):
   10 --
   11 --   * Each unique repo (identified by its vault key) gets its own
   12 --     @ssh-agent -s@ process, started lazily on first use. The agent
   13 --     persists until the harness shuts down (no @-t@ lifetime, no
   14 --     per-op @sahDeleteAll@).
   15 --   * @sahAddKey@ runs @ssh-add <keyfile>@ once per repo (when the agent
   16 --     is first started). The passphrase is supplied via an
   17 --     @SSH_ASKPASS@ helper script + @SSH_ASKPASS_REQUIRE=force@. The
   18 --     helper script reads the passphrase from the @SEAL_ASKPASS_SECRET@
   19 --     environment variable — the script itself contains NO secret, so it
   20 --     can be written to a temp file without violating the "no secret on
   21 --     disk" requirement. The passphrase flows: vault (encrypted at rest)
   22 --     → Haskell process memory → @ssh-add@'s env → askpass helper's
   23 --     inherited env → stdout → @ssh-add@. Never touches disk.
   24 --   * @sahGetAuthEnv@ yields the @SSH_AUTH_SOCK@ / @SSH_AGENT_PID@ env to
   25 --     forward to the untrusted git run (via @ssh -A@ agent forwarding).
   26 --
   27 -- The untrusted machine NEVER sees the keyfile (encrypted or otherwise) —
   28 -- only the forwarded @SSH_AUTH_SOCK@ via @ssh -A@. The agent keeps
   29 -- passwords in memory, not on disk.

   64 -- | The shared agent environment — the @SSH_AUTH_SOCK@ / @SSH_AGENT_PID@
   65 -- values to forward to the untrusted git run (via the env-override seam on
   66 -- 'UntrustedIO' / the @-A@ flag on 'sshExecArgv'). Built once when the
   67 -- agent starts; reused across all ops for that repo.
   68 data SshAgentEnv = SshAgentEnv
   69   { saeAuthSock :: String
   70   , saeAgentPid :: String
   71   }
   72   deriving stock (Eq, Show)
   73 
   74 -- | JSON codec for persistence (the agent registry serializes these to
   75 -- @~\/.seal\/state\/ssh-agents\/registry.json@). The values are non-secret
   76 -- (a socket path + PID); the key material lives only in the agent's memory.
   77 instance ToJSON SshAgentEnv where
   78   toJSON env = object [ "auth_sock" .= saeAuthSock env
   79                       , "agent_pid" .= saeAgentPid env ]
   80 
   81 instance FromJSON SshAgentEnv where
   82   parseJSON = withObject "SshAgentEnv" $ \o -> do
   83     sock <- o .: "auth_sock"
   84     pid  <- o .: "agent_pid"
   85     pure SshAgentEnv { saeAuthSock = sock, saeAgentPid = pid }

   87 -- | The capability seam. Each field is an IO action the clone/git opcode
   88 -- calls; the smart constructors wire the real or fake implementation.
   89 -- The constructor is NOT exported — only the smart constructors are.
   90 --
   91 -- The lifecycle is one-agent-per-repo:
   92 --   @sahStart@ (once per repo) → @sahAddKey@ (once per repo) → run.
   93 -- The agent process + key persist for the harness lifetime; @sahKill@
   94 -- is called at shutdown.
   95 data SshAgentHandle = SshAgentHandle
   96   { sahStart      :: IO (Either Text SshAgentEnv)
   97     -- ^ Start @ssh-agent -s@ (once per repo, lazily on first use). Parse
   98     -- @SSH_AUTH_SOCK@ / @SSH_AGENT_PID@ from its stdout.
   99   , sahAddKey     :: SshAgentEnv -> FilePath -> ByteString -> IO (Either Text ())
  100     -- ^ @ssh-add <keyfile>@: the passphrase (the 'ByteString') is supplied
  101     -- via an @SSH_ASKPASS@ helper script that reads it from the
  102     -- @SEAL_ASKPASS_SECRET@ environment variable (no secret on disk). The
  103     -- agent decrypts the encrypted keyfile into memory. Called once per
  104     -- repo (when the agent is first started).
  105   , sahDeleteAll  :: SshAgentEnv -> IO ()
  106     -- ^ @ssh-add -D@ — forget all keys. Not called per-op in the
  107     -- one-agent-per-repo model; retained for fail-closed cleanup on
  108     -- @sahAddKey@ failure.
  109   , sahKill       :: SshAgentEnv -> IO ()
  110     -- ^ @ssh-agent -k@ — terminate the agent. Called once at harness
  111     -- shutdown.
  112   , sahGetAuthEnv :: SshAgentEnv -> [(String, String)]
  113     -- ^ The @SSH_AUTH_SOCK@ / @SSH_AGENT_PID@ env to forward.
  114   }
  115

  120 -- | Render the env extras for forwarding to the untrusted git run.
  121 saeGetAuthEnv :: SshAgentEnv -> [(String, String)]
  122 saeGetAuthEnv env =
  123   [ ("SSH_AUTH_SOCK", saeAuthSock env)
  124   , ("SSH_AGENT_PID", saeAgentPid env)
  125   ]
  126 
  127 -- | Parse @ssh-agent -s@ stdout for @SSH_AUTH_SOCK@ and @SSH_AGENT_PID@.
  128 parseAgentEnv :: String -> Maybe SshAgentEnv
  129 parseAgentEnv out = do
  130   sock <- findAssign "SSH_AUTH_SOCK"
  131   pid  <- findAssign "SSH_AGENT_PID"
  132   Just SshAgentEnv { saeAuthSock = sock, saeAgentPid = pid }
  133   where
  134     ls = lines out
  135     findAssign name =
  136       let prefix = name <> "="
  137       in case filter (prefix `isPrefixOf`) ls of
  138            (l : _) -> Just (takeWhile (/= ';') (drop (length prefix) l))
  139            []      -> Nothing

  142 -- Real implementation (spawns ssh-agent / ssh-add)
  143 ----------------------------------------------------------------------------
  144 
  145 -- | The real @ssh-agent@ / @ssh-add@ implementation. Spawns a real agent
  146 -- process per repo (no @-t@ lifetime — the key persists for the harness
  147 -- lifetime). @sahAddKey@ loads the key once per repo using an env-var
  148 -- askpass helper (no secret on disk).
  149 mkRealSshAgentHandle :: SshAgentHandle
  150 mkRealSshAgentHandle = SshAgentHandle
  151   { sahStart = do
  152       let agentArgv = ["ssh-agent", "-s"]
  153       eRes <- try @IOError (readProcessCollect agentArgv "")
  154       pure $ case eRes of
  155         Left _ioErr -> Left "ssh-agent: failed to start (unreachable)"
  156         Right (ExitSuccess, out, _err) -> case parseAgentEnv out of
  157           Nothing -> Left "ssh-agent: could not parse SSH_AUTH_SOCK/SSH_AGENT_PID"
  158           Just env -> Right env
  159         Right (ExitFailure _n, _out, _err) -> Left "ssh-agent: non-zero exit"
  160   , sahAddKey = \env keyfile passphrase -> do
  161       -- The askpass helper script contains NO secret — it reads the
  162       -- passphrase from the SEAL_ASKPASS_SECRET environment variable.
  163       -- The passphrase flows: Haskell memory → ssh-add's env → askpass
  164       -- helper's inherited env → stdout → ssh-add. Never touches disk.
  165       -- The script is written to a 0700 temp file, used once, then removed.
  166       (askpassPath, askpassCleanup) <- writeAskpassHelper
  167       let addEnv =
  168             [ ("SSH_ASKPASS", askpassPath)
  169             , ("SSH_ASKPASS_REQUIRE", "force")
  170             , ("SEAL_ASKPASS_SECRET", T.unpack (TE.decodeUtf8Lenient passphrase))
  171             ] ++ saeGetAuthEnv env
  172       eRes <- try @IOError (runProcessEnv addEnv "ssh-add" [keyfile] "")
  173       askpassCleanup
  174       pure $ case eRes of
  175         Left _ioErr -> Left "ssh-add: failed to start"
  176         Right (ExitSuccess, _out, _err) -> Right ()
  177         Right (ExitFailure _n, _out, err)
  178           | "Enter passphrase" `isInfixOf` T.unpack err ->
  179               Left "ssh-add: passphrase rejected (wrong passphrase?)"
  180           | T.null err -> Left "ssh-add: non-zero exit"
  181           | otherwise -> Left ("ssh-add: " <> err)
  182   , sahDeleteAll = \env -> do
  183       _ <- try @IOError (runProcessEnv (saeGetAuthEnv env) "ssh-add" ["-D"] "")
  184       pure ()
  185   , sahKill = \env -> do
  186       _ <- try @IOError (runProcessEnv (saeGetAuthEnv env) "ssh-agent" ["-k"] "")
  187       pure ()
  188   , sahGetAuthEnv = saeGetAuthEnv
  189   }

  191 -- | The 'CreateProcess' for @ssh-agent -s@ (sahStart). Exposed for the
  192 -- FD-leak regression test (#88): @close_fds = True@ ensures ssh-agent does
  193 -- NOT inherit seal's TCP sockets (the WS stream port) + transcript file
  194 -- handles. Without this, a crashed seal leaves ssh-agent holding the
  195 -- gateway's ports hostage.
  196 agentCreateProcess :: [String] -> CreateProcess
  197 agentCreateProcess argv =
  198   case argv of
  199     (p : as) -> mkCloseFds (proc p as)
  200     []       -> error "agentCreateProcess: empty argv (unreachable)"
  201   where
  202     mkCloseFds cp = cp { close_fds = True }
  203 
  204 -- | The 'CreateProcess' for @ssh-add@ (sahAddKey / sahDeleteAll).
  205 addKeyCreateProcess :: [(String, String)] -> String -> [String] -> CreateProcess
  206 addKeyCreateProcess env program args =
  207   (proc program args) { close_fds = True, std_in = CreatePipe
  208                       , std_out = CreatePipe, std_err = CreatePipe
  209                       , env = Just env }
  210 
  211 -- | The 'CreateProcess' for @ssh-agent -k@ (sahKill).
  212 killCreateProcess :: [(String, String)] -> CreateProcess
  213 killCreateProcess env =
  214   (proc "ssh-agent" ["-k"]) { close_fds = True
  215                              , std_in = CreatePipe
  216                              , std_out = CreatePipe
  217                              , std_err = CreatePipe
  218                              , env = Just env }
  219 
  220 -- | Test-visible projections of the 'close_fds' field (the FD-leak
  221 -- regression test asserts these are 'True').
  222 agentCreateProcessCloseFds :: Bool
  223 agentCreateProcessCloseFds =
  224   close_fds (agentCreateProcess ["ssh-agent", "-s"])
  225 
  226 addKeyCreateProcessCloseFds :: Bool
  227 addKeyCreateProcessCloseFds =
  228   close_fds (addKeyCreateProcess [] "ssh-add" ["-D"])
  229 
  230 killCreateProcessCloseFds :: Bool
  231 killCreateProcessCloseFds = close_fds (killCreateProcess [])

  233 -- | Run a process with a given env + stdin 'ByteString', capturing
  234 -- stdout/stderr as 'Text'.
  235 runProcessEnv
  236   :: [(String, String)] -> String -> [String] -> ByteString
  237   -> IO (ExitCode, Text, Text)
  238 runProcessEnv env program args stdinBytes = do
  239   let cp = addKeyCreateProcess env program args
  240   withCreateProcess cp $ \mIn mOut mErr ph -> do
  241     case mIn of
  242       Just hIn -> do
  243         BS.hPutStr hIn stdinBytes
  244         hFlush hIn
  245         hClose hIn
  246       Nothing -> pure ()
  247     out <- readHandle mOut
  248     err <- readHandle mErr
  249     ec  <- waitForProcess ph
  250     pure (ec, out, err)
  251   where
  252     readHandle :: Maybe Handle -> IO Text
  253     readHandle Nothing  = pure ""
  254     readHandle (Just h) = TE.decodeUtf8Lenient <$> BS.hGetContents h
  255 
  256 -- | Run a process with inherited env + a 'String' stdin, capturing
  257 -- stdout/stderr as 'String' (used by 'sahStart' to parse the agent env).
  258 readProcessCollect :: [String] -> String -> IO (ExitCode, String, String)
  259 readProcessCollect argv stdinStr =
  260   case argv of
  261     (_ : _) -> readCreateProcessWithExitCode (agentCreateProcess argv) stdinStr
  262     []       -> error "readProcessCollect: empty argv (unreachable)"

  264 ----------------------------------------------------------------------------
  265 -- Askpass helper (generic script that reads passphrase from env var)
  266 ----------------------------------------------------------------------------
  267 
  268 -- | Write a generic @SSH_ASKPASS@ helper script that reads the passphrase
  269 -- from the @SEAL_ASKPASS_SECRET@ environment variable. The script contains
  270 -- NO secret — only the env var *name*. It can be written to a temp file
  271 -- without violating the "no secret on disk" requirement. The temp file is
  272 -- 0700 (private), used once by @ssh-add@, then removed.
  273 writeAskpassHelper :: IO (FilePath, IO ())
  274 writeAskpassHelper = do
  275   let script = "#!/bin/sh\necho $SEAL_ASKPASS_SECRET\n"
  276   (path, h) <- openBinaryTempFile "/tmp" "seal-askpass-"
  277   BS.hPutStr h (TE.encodeUtf8 (T.pack script))
  278   hClose h
  279   setFileMode path 0o700
  280   let cleanup = do
  281         _ <- try @IOError (removeFile path)
  282         pure ()
  283   pure (path, cleanup)

  298 -- | A fake 'SshAgentHandle' that records every call into the 'IORef'
  299 -- (oldest-last) and returns a canned 'SshAgentEnv'. No real process is
  300 -- spawned.
  301 mkFakeSshAgentHandle :: IORef [FakeAgentCall] -> SshAgentEnv -> SshAgentHandle
  302 mkFakeSshAgentHandle ref cannedEnv = SshAgentHandle
  303   { sahStart = do

  307       modifyIORef' ref (++ [SahAddKey keyfile pass])
  308       pure (Right ())
  309   , sahDeleteAll = \_env -> do
  310       modifyIORef' ref (++ [SahDeleteAll])
  311       pure ()
  312   , sahKill = \_env -> do
  313       modifyIORef' ref (++ [SahKill])

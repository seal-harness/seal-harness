    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Derive Telegram BotCommand menu entries from the Seal command 'Registry'.
    3 -- Each 'CommandSpec' becomes a 'BotCommand' (name + short description) so
    4 -- Telegram's native @\/@-command auto-completion menu is populated from the
    5 -- same single source of truth that drives @\/help@ and the CLI dispatch.
    6 -- Mirrors hermes-agent's @telegram_bot_commands()@ approach: one menu entry
    7 -- per canonical command (aliases skipped), names sanitized for Telegram's
    8 -- charset, descriptions truncated to Telegram's 256-char limit.
    9 module Seal.Channels.Telegram.Commands
   10   ( telegramBotCommands
   11   , sanitizeTelegramName
   12   , maxTelegramCommandNameLen
   13   , maxTelegramDescriptionLen
   14   ) where
   15 
   16 import Data.Char (isAscii, isLower, isDigit)
   17 import Data.Text (Text)
   18 import Data.Text qualified as T
   19 
   20 import Seal.Channels.Telegram.Transport (BotCommand (..))
   21 import Seal.Command.Spec
   22   ( CommandName (..), CommandSpec (..)
   23   , Registry (..), registrySpecs )
   24 
   25 -- | Telegram's limits: command names ≤ 32 chars (lowercase @a-z0-9_@),
   26 -- descriptions ≤ 256 chars.
   27 maxTelegramCommandNameLen :: Int
   28 maxTelegramCommandNameLen = 32
   29 
   30 maxTelegramDescriptionLen :: Int
   31 maxTelegramDescriptionLen = 256
   32 
   33 -- | Derive the BotCommand menu from the Registry. Includes all commands
   34 -- (both 'AlwaysAvailable' and 'InteractiveOnly') — interactive commands
   35 -- still work over Telegram: their handlers respond with usage text or
   36 -- defer via 'ChannelCaps' when selected. Skips aliases, sanitizes names
   37 -- for Telegram's charset, and truncates descriptions. The synthetic
   38 -- @\/help@ is always included first.
   39 telegramBotCommands :: Registry -> [BotCommand]
   40 telegramBotCommands reg =
   41   helpCommand : map specToBotCommand (filter isMenuEligible (registrySpecs reg))
   42   where
   43     isMenuEligible spec = isTelegramSafeName (csName spec)
   44     helpCommand = BotCommand
   45       { bcName = "help"
   46       , bcDescription = "Show available commands"
   47       }
   48 
   49 -- | Convert a CommandSpec to a BotCommand.
   50 specToBotCommand :: CommandSpec -> BotCommand
   51 specToBotCommand spec =
   52   let CommandName n = csName spec
   53   in BotCommand
   54        { bcName = sanitizeTelegramName n
   55        , bcDescription = T.take maxTelegramDescriptionLen (csSynopsis spec)
   56        }
   57 
   58 -- | Sanitize a command name for Telegram: lowercase, replace @-@ with @_@,
   59 -- strip invalid chars (only @a-z0-9_@ allowed), collapse double underscores,
   60 -- trim leading/trailing underscores, clamp to 32 chars.
   61 sanitizeTelegramName :: Text -> Text
   62 sanitizeTelegramName =
   63   clampName
   64   . T.dropWhileEnd (== '_')
   65   . T.dropWhile (== '_')
   66   . collapseUnderscores
   67   . T.filter isValidChar
   68   . T.toLower
   69   . T.replace "-" "_"
   70   where
   71     isValidChar c = (isAscii c && isLower c) || isDigit c || c == '_'
   72     collapseUnderscores = T.intercalate "_" . T.splitOn "__"
   73     clampName t =
   74       if T.length t > maxTelegramCommandNameLen
   75         then T.take maxTelegramCommandNameLen t
   76         else t
   77 
   78 -- | Check if a command name is safe for Telegram (after sanitization it would
   79 -- be non-empty). The terse grammar @\/N@ is excluded (single uppercase letter
   80 -- doesn't sanitize to a meaningful command).
   81 isTelegramSafeName :: CommandName -> Bool
   82 isTelegramSafeName (CommandName n) =
   83   not (T.null (sanitizeTelegramName n)) && n /= "N"

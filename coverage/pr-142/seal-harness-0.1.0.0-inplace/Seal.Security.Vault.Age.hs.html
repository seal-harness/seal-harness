   27 -- | Run a process with a strict 'ByteString' stdin, capturing stdout and
   28 -- stderr as strict 'ByteString's. Used by the age encrypt/decrypt seam
   29 -- where the payload is binary (ciphertext) and must not be round-tripped
   30 -- through 'String'. Closes all handles and waits for the child before
   31 -- returning.
   32 readProcessBinary :: FilePath -> [String] -> ByteString
   33                  -> IO (ExitCode, ByteString, ByteString)
   34 readProcessBinary cmdPath args input =
   35   withCreateProcess
   36     ( (proc cmdPath args)
   37         { std_in = CreatePipe, std_out = CreatePipe, std_err = CreatePipe }
   38     ) $ \mIn mOut mErr ph -> do
   39       (hIn, hOut, hErr) <- case (mIn, mOut, mErr) of
   40         (Just a, Just b, Just c) -> pure (a, b, c)
   41         _ -> error "readProcessBinary: pipe creation failed (unreachable)"
   42       -- Feed stdin on a child thread so we can read stdout/stderr
   43       -- concurrently without deadlocking on a large payload.
   44       BS.hPutStr hIn input
   45       hFlush hIn
   46       hClose hIn
   47       -- Read stdout/stderr fully after stdin is closed (age reads the
   48       -- whole plaintext before emitting ciphertext, so this is safe).
   49       out <- BS.hGetContents hOut
   50       err <- BS.hGetContents hErr
   51       ec  <- waitForProcess ph
   52       -- Force the lazy hGetContents results so the handles are fully read
   53       -- before withCreateProcess closes them.
   54       let !_ = BS.length out
   55           !_ = BS.length err
   56       pure (ec, out, err)

   58 -- | The vault's error type. The first four constructors are matched on to
   59 -- drive control flow (so they earn being a sum type per the haskell-coder
   60 -- skill); 'VaultBackendError' is the single catch-all for failures that are
   61 -- only reported — @age@ stderr, "age not installed", corrupt vault data.
   62 data VaultError
   63   = VaultLocked
   64   | VaultNotFound
   65   | VaultAlreadyExists
   66   | VaultKeyNotFound Text
   67   | VaultBackendError Text
   68   deriving stock (Eq, Show)
   69 
   70 newtype AgeRecipient = AgeRecipient Text deriving stock (Eq, Show)
   71 newtype AgeIdentity  = AgeIdentity  Text deriving stock (Eq, Show)

   73 -- | Encrypt/decrypt with credentials already captured in the closure.
   74 data VaultEncryptor = VaultEncryptor
   75   { veEncrypt :: ByteString -> IO (Either VaultError ByteString)
   76   , veDecrypt :: ByteString -> IO (Either VaultError ByteString)
   77   }
   78 
   79 -- | Build a real encryptor backed by @age@. Preflights @age --version@ and
   80 -- reports a 'VaultBackendError' install hint if the binary is absent.
   81 mkAgeEncryptor :: AgeRecipient -> AgeIdentity -> IO (Either VaultError VaultEncryptor)
   82 mkAgeEncryptor (AgeRecipient recipient) (AgeIdentity identity) = do
   83   versionResult <- try @IOException (callProcessNoOutput "age" ["--version"])
   84   case versionResult of
   85     Right ExitSuccess ->
   86       pure (Right VaultEncryptor
   87         { veEncrypt = run ["--encrypt", "--recipient", T.unpack recipient]
   88         , veDecrypt = run ["--decrypt", "--identity", T.unpack identity]
   89         })
   90     _ -> pure notInstalled  -- IOException (binary absent) OR ExitFailure
   91   where
   92     notInstalled =
   93       Left (VaultBackendError "age not installed; see https://age-encryption.org")
   94     run :: [String] -> ByteString -> IO (Either VaultError ByteString)
   95     run args input = do
   96       (code, out, err) <- readProcessBinary "age" args input
   97       pure $ case code of
   98         ExitSuccess   -> Right out
   99         ExitFailure _ -> Left (VaultBackendError (TE.decodeUtf8Lenient err))
  100 
  101 -- | Run a process, ignoring its stdout/stderr, returning only the exit code.
  102 -- Used for the @age --version@ preflight. Catches a launch failure as
  103 -- 'ExitFailure 127' (consistent with the git seam's convention).
  104 callProcessNoOutput :: FilePath -> [String] -> IO ExitCode
  105 callProcessNoOutput cmdPath args =
  106   withCreateProcess ((proc cmdPath args) { std_out = NoStream, std_err = NoStream }) $
  107     \_ _ _ ph -> waitForProcess ph

  109 -- | XOR-with-0xAB mock; reversible, no binary required.
  110 mkMockEncryptor :: VaultEncryptor
  111 mkMockEncryptor = VaultEncryptor
  112   { veEncrypt = pure . Right . BS.map (`xor` 0xAB)

    2 -- | The @/model@ command group: list known providers/models and set the active
    3 -- session's provider+model (persisted to session.json). Provider+model are named
    4 -- explicitly (unambiguous once providers host arbitrary model names).
    5 --
    6 -- Two constructors mirror 'Seal.Command.Stop':
    7 --
    8 -- * 'modelCommandSpec' — single-session channels (CLI, TUI). Reads
    9 --   'srActive' at dispatch time and saves back to it + its on-disk
   10 --   @session.json@.
   11 -- * 'modelCommandSpecForSession' — multi-session surfaces (the per-request web
   12 --   gateway rebuild, the inbox channel-loop shadow). Closes over an explicit
   13 --   'SessionId' resolver + 'SealPaths' so a @\/model use@ typed in tab N (or
   14 --   from a Telegram conversation) targets THAT session's @session.json@, not
   15 --   the process-global @srActive@ — which on those surfaces points at an
   16 --   unrelated session and would silently lose the change on the next turn.
   17 --
   18 -- @\/model use@ records an @EKResponse@ audit entry in the target session's
   19 -- transcript (via 'ModelTranscriptWriter', twin of 'StopTranscriptWriter') so
   20 -- the change is visible cross-channel and survives a @seal serve@ restart. It
   21 -- makes NO LLM request — it is pure session-state mutation + one transcript
   22 -- line, like @\/stop@.
   23 module Seal.Command.Model
   24   ( modelCommandSpec
   25   , modelCommandSpecForSession
   26   , ModelTranscriptWriter
   27   , noModelTranscriptWriter
   28   , mkModelTranscriptWriter
   29   ) where

   62 -- | A function that records the @\/model use@ confirmation to the target
   63 -- session's transcript and broadcasts the new entry to WS subscribers. Built
   64 -- at the wiring site from 'SealPaths' + the broker (twin of
   65 -- 'Seal.Command.Stop.StopTranscriptWriter'). 'Nothing' (via
   66 -- 'noModelTranscriptWriter') skips the write — used in tests or contexts
   67 -- without a transcript.
   68 newtype ModelTranscriptWriter = ModelTranscriptWriter (Maybe (SessionId -> Text -> IO ()))
   69 
   70 -- | The no-op writer (no transcript write). Used in tests or contexts without
   71 -- a transcript.
   72 noModelTranscriptWriter :: ModelTranscriptWriter
   73 noModelTranscriptWriter = ModelTranscriptWriter Nothing
   74 
   75 -- | Build a 'ModelTranscriptWriter' from the shared dependencies. Opens the
   76 -- session's transcript, writes the confirmation as an assistant 'EKResponse'
   77 -- entry, and broadcasts the new entry via the broker. Mirrors
   78 -- 'Seal.Command.Stop.mkStopTranscriptWriter'.
   79 mkModelTranscriptWriter :: SealPaths -> Maybe StreamBroker -> ModelTranscriptWriter
   80 mkModelTranscriptWriter paths mBroker =
   81   ModelTranscriptWriter $ Just $ \sid msg -> do
   82     let dir = sessionDir paths sid
   83     mMeta <- loadSessionMeta paths sid
   84     now <- getCurrentTime
   85     let model = maybe "" smModel mMeta
   86         createdAt = maybe now smCreatedAt mMeta
   87     withTwoFileTranscript dir $ \h -> do
   88       let assistantMsg = Message Assistant [CbText msg]
   89           entry = EntryRecord
   90             { erId = ""
   91             , erTimestamp = now
   92             , erKind = EKResponse
   93             , erConvLen = 1
   94             , erEnvelope = Nothing
   95             , erUsage = Nothing
   96             , erStop = Nothing
   97             , erDurationMs = Nothing
   98             , erHarness = Nothing
   99             , erCorrelation = Nothing
  100             , erMeta = mempty
  101             }
  102       tfwRecordAndAck h (TwoFileWrite [assistantMsg] entry)
  103       broadcastNewEntries mBroker paths sid model createdAt

  105 -- | The @\/model@ command spec for single-session channels (CLI, TUI).
  106 -- Reads 'srActive' at dispatch time and saves back to it + its on-disk
  107 -- @session.json@.
  108 modelCommandSpec :: ProviderRuntime -> SessionRuntime -> ModelTranscriptWriter -> CommandSpec
  109 modelCommandSpec pr sr (ModelTranscriptWriter mw) =
  110   modelCommandSpecWith pr (Just <$> readIORef (srActive sr)) save mw
  111   where
  112     save m = do
  113       writeIORef (srActive sr) m
  114       saveSessionMeta (srPaths sr) m
  115 
  116 -- | The @\/model@ command spec for multi-session surfaces (web per-request,
  117 -- channel loops). Closes over an explicit 'SessionId' resolver + 'SealPaths'.
  118 modelCommandSpecForSession
  119   :: ProviderRuntime -> SealPaths -> IO SessionId -> ModelTranscriptWriter -> CommandSpec
  120 modelCommandSpecForSession pr paths resolveSid (ModelTranscriptWriter mw) =
  121   modelCommandSpecWith pr resolveMeta save mw
  122   where
  123     resolveMeta = do
  124       sid <- resolveSid
  125       loadSessionMeta paths sid
  126     save = saveSessionMeta paths

  128 -- | The core spec builder, parameterized by a session-meta resolver, a save
  129 -- action, and an optional transcript writer. Both public constructors
  130 -- delegate here.
  131 modelCommandSpecWith
  132   :: ProviderRuntime
  133   -> IO (Maybe SessionMeta)
  134   -> (SessionMeta -> IO ())
  135   -> Maybe (SessionId -> Text -> IO ())
  136   -> CommandSpec
  137 modelCommandSpecWith pr resolveMeta save mWriter = CommandSpec
  138   { csName         = CommandName "model"
  139   , csAliases      = []
  140   , csGroup        = GroupModel
  141   , csSynopsis     = "List models and set the active session's model"
  142   , csParserInfo   = modelParserInfo pr resolveMeta save mWriter
  143   , csAvailability = InteractiveOnly
  144   }

  146 modelParserInfo
  147   :: ProviderRuntime -> IO (Maybe SessionMeta) -> (SessionMeta -> IO ())
  148   -> Maybe (SessionId -> Text -> IO ()) -> ParserInfo CommandAction
  149 modelParserInfo pr resolveMeta save mWriter =
  150   info (modelParser pr resolveMeta save mWriter <**> helper)
  151     (  progDesc "List known providers/models and choose the session's model"
  152     <> header   "model — inspect and set the active session's model"
  153     )

  155 modelParser
  156   :: ProviderRuntime -> IO (Maybe SessionMeta) -> (SessionMeta -> IO ())
  157   -> Maybe (SessionId -> Text -> IO ()) -> Parser CommandAction
  158 modelParser pr resolveMeta save mWriter = hsubparser
  159   (  command "list"
  160        (info (listCmd pr resolveMeta <$> optional listProvArg)
  161              (progDesc "List known providers, or a provider's live models"))
  162   <> command "use"
  163        (info (useCmd pr resolveMeta save mWriter <$> provArg <*> optional modelArg)
  164              (progDesc "Set the session's provider and model (model optional)"))
  165   <> command "default"
  166        (info (defaultCmd pr <$> provArg <*> modelArg)
  167              (progDesc "Set a provider's default model"))
  168   <> metavar "COMMAND"
  169   )

  181 listCmd
  182   :: ProviderRuntime -> IO (Maybe SessionMeta) -> Maybe Text -> CommandAction
  183 listCmd pr resolveMeta Nothing = CommandAction $ \caps -> do
  184   eCfg <- loadRuntimeConfig (prConfigPath pr)
  185   let cfg = fromRight defaultRuntimeConfig eCfg
  186   mh <- readIORef (vrHandleRef (prVault pr))
  187   configured <- configuredProviders mh cfg
  188   if null configured
  189     then ccSend caps "no providers configured — run /provider add <provider>"
  190     else mapM_ (reportProviderQuiet pr caps cfg) configured
  191   mActive <- resolveMeta
  192   let activeLine = case mActive of
  193         Just a  -> "active: " <> smProvider a <> "/" <> smModel a
  194         Nothing -> "active: (no session)"
  195   ccSend caps activeLine
  196 listCmd pr _ (Just provLbl) = CommandAction $ \caps ->
  197   case parseProvider provLbl of
  198     Nothing -> ccSend caps (unknownProviderMsg provLbl)
  199     Just kp -> do
  200       eCfg <- loadRuntimeConfig (prConfigPath pr)
  201       let cfg = fromRight defaultRuntimeConfig eCfg
  202       reportProviderLoud pr caps cfg kp

  204 -- | Render one provider for the bare @/model list@: its label followed by
  205 -- either the live models reachable through a resolved provider, or — if
  206 -- resolution or the live listing fails — the configured default model as a
  207 -- fallback so the line is never empty and never noisy. Unconfigured
  208 -- providers are filtered out before this runs, so a failure here means the
  209 -- daemon is down, not that the provider is unconfigured.
  210 reportProviderQuiet
  211   :: ProviderRuntime -> ChannelCaps -> RuntimeConfig -> KnownProvider -> IO ()
  212 reportProviderQuiet pr caps cfg kp = do
  213   let lbl   = providerLabel kp
  214       baseUrl = fromMaybe defaultOllamaBaseUrl (providerBaseUrl cfg lbl)
  215       ModelId dm = resolveDefaultModel (providerDefaultModel cfg lbl) lbl
  216   mh <- readIORef (vrHandleRef (prVault pr))
  217   eProv <- resolveProvider mh (prManager pr) baseUrl kp (defaultModelFor kp) (prCallCounter pr)
  218   eModels <- case eProv of
  219     Left _    -> pure (Left "unresolved")
  220     Right sp -> listSome sp
  221   case eModels of
  222     Left _   -> ccSend caps (lbl <> " (default model: " <> dm <> ")")
  223     Right [] -> ccSend caps (lbl <> " (default model: " <> dm <> ")")
  224     Right ms -> do
  225       ccSend caps (lbl <> " models (live):")
  226       mapM_ (\(ModelId m) -> ccSend caps ("  " <> m)) ms
  227 
  228 -- | Render one provider for the explicit @/model list <provider>@: surfaces
  229 -- resolution and listing failures as actionable error lines (the user asked
  230 -- for this specific provider, so a silent fallback would hide the problem).
  231 reportProviderLoud
  232   :: ProviderRuntime -> ChannelCaps -> RuntimeConfig -> KnownProvider -> IO ()
  233 reportProviderLoud pr caps cfg kp = do
  234   let lbl   = providerLabel kp
  235       baseUrl = fromMaybe defaultOllamaBaseUrl (providerBaseUrl cfg lbl)
  236   mh <- readIORef (vrHandleRef (prVault pr))
  237   eProv <- resolveProvider mh (prManager pr) baseUrl kp (defaultModelFor kp) (prCallCounter pr)
  238   case eProv of
  239     Left e   -> ccSend caps ("could not list " <> lbl <> " models: " <> e)
  240     Right sp -> do
  241       eModels <- listSome sp
  242       case eModels of
  243         Left e   -> ccSend caps ("could not list " <> lbl <> " models: " <> e)
  244         Right [] -> ccSend caps (lbl <> " has no models available")
  245         Right ms -> do
  246           ccSend caps (lbl <> " models (live):")
  247           mapM_ (\(ModelId m) -> ccSend caps ("  " <> m)) ms

  249 useCmd
  250   :: ProviderRuntime
  251   -> IO (Maybe SessionMeta)
  252   -> (SessionMeta -> IO ())
  253   -> Maybe (SessionId -> Text -> IO ())
  254   -> Text
  255   -> Maybe Text
  256   -> CommandAction
  257 useCmd pr resolveMeta save mWriter provLbl mModel = CommandAction $ \caps ->
  258   case parseProvider provLbl of
  259     Nothing -> ccSend caps (unknownProviderMsg provLbl)
  260     Just kp -> do
  261       model <- case mModel of
  262         Just m  -> pure m
  263         Nothing -> do
  264           eCfg <- loadRuntimeConfig (prConfigPath pr)
  265           let cfg = fromRight defaultRuntimeConfig eCfg
  266               ModelId m = resolveDefaultModel (providerDefaultModel cfg (providerLabel kp)) (providerLabel kp)
  267           pure m
  268       mActive <- resolveMeta
  269       case mActive of
  270         Nothing -> ccSend caps "no active session to update"
  271         Just m0 -> do
  272           let lbl = providerLabel kp
  273               m1 = m0 { smProvider = lbl, smModel = model }
  274               confirm = "session model set to " <> lbl <> "/" <> model
  275           save m1
  276           case mWriter of
  277             Just w  -> w (smId m1) confirm
  278             Nothing -> pure ()
  279           ccSend caps confirm
  280 
  281 defaultCmd :: ProviderRuntime -> Text -> Text -> CommandAction
  282 defaultCmd pr provLbl model = CommandAction $ \caps ->
  283   case parseProvider provLbl of
  284     Nothing -> ccSend caps (unknownProviderMsg provLbl)
  285     Just kp -> do
  286       res <- updateRuntimeConfig (prConfigPath pr)
  287                (upsertProvider (providerLabel kp) (\p -> p { pcDefaultModel = Just model }))
  288       case res of
  289         Left e   -> ccSend caps e
  290         Right () -> ccSend caps (providerLabel kp <> " default model set to " <> model)
  291 
  292 unknownProviderMsg :: Text -> Text
  293 unknownProviderMsg lbl =
  294   "unknown provider: " <> lbl <> " (known: "
  295     <> T.intercalate ", " (map providerLabel knownProviders) <> ")"

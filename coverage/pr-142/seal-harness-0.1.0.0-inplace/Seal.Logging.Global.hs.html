    1 -- | Process-global logger ref for library modules that don't have a
    2 -- 'SealLogger' threaded through their dependencies (e.g. 'Seal.Tabs',
    3 -- 'Seal.Handles.Transcript', 'Seal.Web.UiState'). Set once at startup via
    4 -- 'setGlobalLogger'; read via 'globalLogIO'. This avoids threading the
    5 -- logger through dozens of function signatures in deep library code.
    6 --
    7 -- The primary logging path is the threaded 'SealLogger' (via 'ChannelDeps',
    8 -- 'SendDeps', 'Env'). This global ref is a fallback for modules that are
    9 -- called from many call sites and can't easily receive the logger as a
   10 -- parameter. If 'setGlobalLogger' has not been called, 'globalLogIO' is a
   11 -- no-op (best-effort — the logger must never crash the caller).
   12 module Seal.Logging.Global
   13   ( setGlobalLogger
   14   , unsetGlobalLogger
   15   , globalLogIO
   16   ) where
   17 
   18 import Control.Exception (catch, SomeException)
   19 import Data.IORef (IORef, newIORef, readIORef, writeIORef)
   20 import System.IO.Unsafe (unsafePerformIO)
   21 
   22 import Seal.Logging.Logger (SealLogger, logIO)
   23 
   24 import Katip (Severity (..), LogStr)
   25 
   26 -- | The process-global logger ref. Initialized to 'Nothing' (no-op).
   27 -- Set once at startup via 'setGlobalLogger'.
   28 globalLoggerRef :: IORef (Maybe SealLogger)
   29 globalLoggerRef = unsafePerformIO (newIORef Nothing)
   30 {-# NOINLINE globalLoggerRef #-}
   31 
   32 -- | Set the process-global logger. Called once at startup (in
   33 -- 'Seal.AppMain.dispatch', inside the 'withSealLogger' bracket).
   34 setGlobalLogger :: SealLogger -> IO ()
   35 setGlobalLogger = writeIORef globalLoggerRef . Just
   36 
   37 -- | Unset the process-global logger (set the ref to 'Nothing'). Used by
   38 -- tests that install a capture logger via 'setGlobalLogger' and want to
   39 -- restore the no-op default after the test so subsequent tests are
   40 -- unaffected. Not called in production (the logger lives for the process
   41 -- lifetime).
   42 unsetGlobalLogger :: IO ()
   43 unsetGlobalLogger = writeIORef globalLoggerRef Nothing
   44 
   45 -- | Emit a log line via the process-global logger. If no logger has been
   46 -- set (e.g. in tests that don't call 'setGlobalLogger'), this is a no-op.
   47 -- Best-effort: if the logger throws, the error is silently swallowed.
   48 globalLogIO :: Severity -> LogStr -> IO ()
   49 globalLogIO sev msg = do
   50   mLogger <- readIORef globalLoggerRef
   51   case mLogger of
   52     Nothing    -> pure ()
   53     Just logger -> logIO logger sev msg
   54       `catch` \(_ :: SomeException) -> pure ()

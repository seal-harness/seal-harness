    1 -- | The per-process session execution + discovery cache.
    2 --
    3 -- Two memoization layers over the per-session workdir machinery, shared by
    4 -- every surface that touches a session's workspace (turn engine, /call
    5 -- dispatcher, gateway API):
    6 --
    7 --   1. __Session exec__ ('cachedSessionExec') — 'mkSessionExec' runs a
    8 --      remote @mkdir -p@ bootstrap on every construction; in remote mode
    9 --      that is one SSH round trip per turn / per API call. The cache keys
   10 --      the built 'SessionExec' by 'SessionId' and rebuilds only when the
   11 --      security config's untrusted-exec fingerprint changes (a local↔remote
   12 --      or workspace switch) or after an explicit 'invalidateExec'.
   13 --      Fail-closed builds are NEVER cached (a later retry must be able to
   14 --      succeed).
   15 --
   16 --   2. __Workdir discovery__ ('cachedWorkdirScan') — the agent-def +
   17 --      skill scans ('listWorkdirAgentDefs', 'listWorkdirSkills') run once
   18 --      per session within a TTL window and are invalidated explicitly when
   19 --      the workdir's structure changes (@SETUP_REPO@ clone). Each cached
   20 --      entry records its workspace root, so a config switch re-scans even
   21 --      inside the TTL.
   22 --
   23 -- Staleness contract: within the TTL, out-of-band workdir mutations (the
   24 -- model writing files via untrusted opcodes, manual edits on the remote
   25 -- machine) are NOT observed. The TTL bounds that window; SETUP_REPO — the
   26 -- only structural mutation the harness itself drives — invalidates eagerly.
   27 module Seal.Session.ExecCache
   28   ( SessionExecCache
   29   , newSessionExecCache
   30   , newSessionExecCacheWith
   31   , cachedSessionExec
   32   , invalidateExec
   33   , cachedWorkdirScan
   34   , invalidateWorkdirScan
   35   ) where

   60 -- | The shared cache handle. Create ONE per process (per gateway /
   61 -- channel loop / CLI invocation) via 'newSessionExecCache'.
   62 data SessionExecCache = SessionExecCache
   63   { sceExecs :: IORef (Map SessionId CachedExec)
   64   , sceScans :: IORef (Map SessionId CachedScan)
   65   , sceNow   :: IO UTCTime
   66     -- ^ The clock seam (tests inject a fake).
   67   }
   68 
   69 data CachedExec = CachedExec
   70   { ceFingerprint :: !Text       -- ^ Untrusted-exec config fingerprint at build time
   71   , ceExec        :: !SessionExec
   72   }
   73 
   74 data CachedScan = CachedScan
   75   { csAt    :: !UTCTime          -- ^ When the scan ran (for the TTL)
   76   , csRoot  :: !Text             -- ^ Workspace root text (config-switch guard)
   77   , csDefs  :: ![AgentDef]
   78   , csSkills:: ![Skill]
   79   }
   80 
   81 -- | How long a discovery scan stays fresh before the next consumer re-scans.
   82 defaultDiscoveryTTL :: NominalDiffTime
   83 defaultDiscoveryTTL = 60
   84 
   85 -- | Create a cache with the real clock.
   86 newSessionExecCache :: IO SessionExecCache
   87 newSessionExecCache = newSessionExecCacheWith getCurrentTime
   88 
   89 -- | Create a cache with an injected clock (tests).
   90 newSessionExecCacheWith :: IO UTCTime -> IO SessionExecCache
   91 newSessionExecCacheWith now =
   92   SessionExecCache <$> newIORef Map.empty <*> newIORef Map.empty <*> pure now

   94 -- | A stable fingerprint of the part of the security config that determines
   95 -- session-exec behavior (mode local/remote + SSH coordinates). 'Show'-based:
   96 -- the config types are small, pure records with 'Eq'/'Show' pinned for tests.
   97 execFingerprint :: SecurityConfig -> Text
   98 execFingerprint = T.pack . show . untrustedExecConfigFromSecurity
   99 
  100 -- | Fetch-or-build the session's 'SessionExec'. Successes are memoized by
  101 -- 'SessionId'; rebuilt on a fingerprint mismatch or explicit
  102 -- 'invalidateExec'. Fail-closed builds are never stored.
  103 cachedSessionExec
  104   :: SessionExecCache -> SealPaths -> SecurityConfig -> SessionId
  105   -> CloneDeps -> RemoteRunner -> IO SessionExec
  106 cachedSessionExec cache paths secCfg sid cloneDeps runner = do
  107   execs <- readIORef (sceExecs cache)
  108   let fp = execFingerprint secCfg
  109   case Map.lookup sid execs of
  110     Just ce | ceFingerprint ce == fp -> pure (ceExec ce)
  111     _ -> do
  112       exec <- mkSessionExec paths secCfg sid cloneDeps runner
  113       if isFailClosedSessionExec exec
  114         then pure exec   -- never cache failures: a later retry must succeed
  115         else do
  116           modifyIORef' (sceExecs cache) (Map.insert sid (CachedExec fp exec))
  117           pure exec
  118 
  119 -- | Drop one session's memoized exec (the next 'cachedSessionExec' call
  120 -- rebuilds it). Used when the workdir is known to have been recreated.
  121 invalidateExec :: SessionExecCache -> SessionId -> IO ()
  122 invalidateExec cache sid =
  123   modifyIORef' (sceExecs cache) (Map.delete sid)
  124 
  125 -- | Fetch-or-scan the session's workdir discovery results: the agent defs
  126 -- + skills, computed together from ONE 'wfsSnapshot' so a cache miss costs
  127 -- exactly one snapshot round trip. Memoized by 'SessionId' within
  128 -- 'defaultDiscoveryTTL'; a stored entry whose workspace root differs from
  129 -- the caller's re-scans immediately (config-switch guard). A FAILED
  130 -- snapshot is returned fail-soft-to-empty and never cached (the next
  131 -- consumer retries).
  132 cachedWorkdirScan
  133   :: SessionExecCache -> SessionId -> WorkdirFs -> WorkspaceRoot
  134   -> Maybe MetaCacheEnv
  135   -- ^ When 'Just' (remote mode), @.agents/@ reads are served from the
  136   -- content-addressed local snapshot cache ('Seal.Session.AgentMetaCache')
  137   -- after a per-repo URL+content probe; repos that fail the snapshot path
  138   -- keep flowing through the underlying (remote) fs. 'Nothing' is the
  139   -- legacy all-remote crawl.
  140   -> IO ([AgentDef], [Skill])
  141 cachedWorkdirScan cache sid fs wsRoot mMetaEnv = do
  142   now <- sceNow cache
  143   scans <- readIORef (sceScans cache)
  144   let rootText = rootTextOf wsRoot
  145   case Map.lookup sid scans of
  146     Just cs | csRoot cs == rootText && diffUTCTime now (csAt cs) < defaultDiscoveryTTL ->
  147       pure (csDefs cs, csSkills cs)
  148     _ -> do
  149       eSnap <- wfsSnapshot fs
  150       case eSnap of
  151         -- Fail-soft, uncached: don't freeze empty results for the TTL.
  152         Left _ -> pure ([], [])
  153         Right snap -> do
  154           -- Structure comes from the live remote snapshot; only CONTENT
  155           -- reads may switch lanes to local snapshots.
  156           fs' <- case mMetaEnv of
  157             Nothing   -> pure fs
  158             Just env  -> buildRoutingWorkdirFs env fs rootText (snapTopDirs snap)
  159           defs <- listWorkdirAgentDefsSnap snap fs'
  160           skills <- listWorkdirSkillsSnap snap fs'
  161           modifyIORef' (sceScans cache)
  162             (Map.insert sid (CachedScan now rootText defs skills))
  163           pure (defs, skills)
  164 
  165 -- | Drop one session's memoized discovery scan. Called after SETUP_REPO
  166 -- succeeds (the clone changed the workdir's structure) so the next consumer
  167 -- sees the cloned repo's defs/skills.
  168 invalidateWorkdirScan :: SessionExecCache -> SessionId -> IO ()
  169 invalidateWorkdirScan cache sid =
  170   modifyIORef' (sceScans cache) (Map.delete sid)
  171 
  172 rootTextOf :: WorkspaceRoot -> Text
  173 rootTextOf (WorkspaceRoot p) = T.pack p

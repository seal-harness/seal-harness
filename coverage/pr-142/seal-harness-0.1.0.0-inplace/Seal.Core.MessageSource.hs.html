    4 -- | The authenticated-transport-derived identity of an inbound message.
    5 --
    6 -- The critical security property: the 'ConversationId' is **server-derived
    7 -- from transport metadata, never read from a message body**, so a sender
    8 -- cannot forge it to hijack another conversation's tab cursor. This is
    9 -- enforced structurally — 'mkMessageSource' takes a 'ConversationId' (which
   10 -- itself is smart-constructed) and never reads a conversation id from the
   11 -- open field map. The open map is also forbidden from carrying a
   12 -- @conversationId@ key, so a future caller cannot smuggle a second id in.
   13 module Seal.Core.MessageSource
   14   ( ConversationId (..)
   15   , mkConversationId
   16   , conversationIdText
   17   , UserId (..)
   18   , mkUserId
   19   , userIdText
   20   , MessageSource (..)
   21   , mkMessageSource
   22   , maxConversationIdLen
   23   , maxUserIdLen
   24   , maxOpenEntries
   25   , maxOpenFieldLen
   26   ) where

   44 -- | Maximum length of a 'ConversationId' or 'UserId' (in characters).
   45 maxConversationIdLen :: Int
   46 maxConversationIdLen = 256
   47 
   48 maxUserIdLen :: Int
   49 maxUserIdLen = 256
   50 
   51 -- | Maximum number of entries in the open metadata map. Bounds the
   52 -- attacker-controlled metadata size so a malicious sender cannot bloat the
   53 -- transcript or exhaust a cursor map.
   54 maxOpenEntries :: Int
   55 maxOpenEntries = 32
   56 
   57 -- | Maximum length of any single open-map key or value (in characters).
   58 maxOpenFieldLen :: Int
   59 maxOpenFieldLen = 256

   61 -- ---------------------------------------------------------------------------
   62 -- ConversationId
   63 -- ---------------------------------------------------------------------------
   64 
   65 -- | A server-derived, transport-scoped conversation key. NEVER read from a
   66 -- message body — always minted from authenticated transport metadata (e.g.
   67 -- the Signal peer's phone number + UUID, the web session's authenticated
   68 -- principal). Smart-constructed; the predicate bounds length and charset so
   69 -- an attacker cannot bloat the transcript or exhaust a cursor map.
   70 newtype ConversationId = ConversationId Text
   71   deriving stock (Eq, Ord, Show)
   72   deriving newtype (ToJSON, FromJSON)
   73 
   74 -- | Charset: @A-Za-z0-9_-:@. Colon is included for composite keys like
   75 -- @phone:uuid@. Leading dot is rejected (path-join defense-in-depth, even
   76 -- though conversation ids are never path-joined today).
   77 mkConversationId :: Text -> Either Text ConversationId
   78 mkConversationId t
   79   | T.null t              = Left "conversation id is empty"
   80   | T.length t > maxConversationIdLen
   81                           = Left ("conversation id too long (>" <> T.pack (show maxConversationIdLen) <> ")")
   82   | T.head t == '.'       = Left "conversation id must not start with '.'"
   83   | not (T.all validConvChar t)
   84                           = Left "conversation id has invalid characters"
   85   | otherwise             = Right (ConversationId t)
   86 
   87 validConvChar :: Char -> Bool
   88 validConvChar c =
   89   c `Set.member` convChars
   90   where
   91     convChars :: Set Char
   92     convChars = Set.fromList
   93       $ ['A'..'Z'] <> ['a'..'z'] <> ['0'..'9'] <> "_-:+"
   94 
   95 conversationIdText :: ConversationId -> Text
   96 conversationIdText (ConversationId t) = t
   97 
   98 -- ---------------------------------------------------------------------------
   99 -- UserId
  100 -- ---------------------------------------------------------------------------
  101 
  102 -- | An authenticated user identity on a channel (e.g. a Signal phone number
  103 -- or UUID, a Telegram user id, a web principal). Optional — some channels
  104 -- (Background) have no user. Smart-constructed with the same predicate shape
  105 -- as 'ConversationId', plus @+@ (common in E.164 phone numbers). No
  106 -- leading-dot rule (user ids are transport-minted, never path-joined).
  107 newtype UserId = UserId Text
  108   deriving stock (Eq, Ord, Show)
  109   deriving newtype (ToJSON, FromJSON)
  110 
  111 mkUserId :: Text -> Either Text UserId
  112 mkUserId t
  113   | T.null t              = Left "user id is empty"
  114   | T.length t > maxUserIdLen
  115                           = Left ("user id too long (>" <> T.pack (show maxUserIdLen) <> ")")
  116   | not (T.all validUserChar t)
  117                           = Left "user id has invalid characters"
  118   | otherwise             = Right (UserId t)
  119 
  120 validUserChar :: Char -> Bool
  121 validUserChar c = c `Set.member` userChars
  122   where
  123     userChars :: Set Char
  124     userChars = Set.fromList
  125       $ ['A'..'Z'] <> ['a'..'z'] <> ['0'..'9'] <> "_+-:"
  126 
  127 userIdText :: UserId -> Text
  128 userIdText (UserId t) = t

  131 -- MessageSource
  132 -- ---------------------------------------------------------------------------

  134 -- | The authenticated-transport-derived identity of an inbound message.
  135 -- Constructed ONLY via 'mkMessageSource', which strips control characters
  136 -- and bounds the length of every attacker-controlled string leaf.
  137 data MessageSource = MessageSource
  138   { msConversationId :: ConversationId   -- ^ required; server-derived
  139   , msChannelKind    :: ChannelKind
  140   , msUserId         :: Maybe UserId     -- ^ optional; absent on Background
  141   , msOpen           :: Map Text Text    -- ^ bounded, control-char-stripped
  142   } deriving stock (Eq, Show, Generic)
  143   deriving anyclass (ToJSON, FromJSON)
  144 
  145 -- | Smart constructor. Strips control characters from every open-map key and
  146 -- value, rejects any key or value over 'maxOpenFieldLen', rejects more than
  147 -- 'maxOpenEntries' entries, and rejects a @conversationId@ key in the open
  148 -- map (it is a structural field, not an open one — preventing a caller from
  149 -- smuggling a second conversation id).
  150 mkMessageSource
  151   :: ConversationId
  152   -> ChannelKind
  153   -> Maybe UserId
  154   -> Map Text Text
  155   -> Either Text MessageSource
  156 mkMessageSource cid kind mUid open0
  157   | Map.size open0 > maxOpenEntries
  158       = Left ("open metadata map has too many entries (>" <> T.pack (show maxOpenEntries) <> ")")
  159   | Map.member "conversationId" open0
  160       = Left "open field key 'conversationId' is reserved"
  161   | otherwise =
  162       let open1 = Map.mapKeys stripControl open0
  163           open2 = Map.map stripControl open1
  164       in if anyTooLong open2
  165            then Left ("open field key/value too long (>" <> T.pack (show maxOpenFieldLen) <> ")")
  166            else Right MessageSource
  167                   { msConversationId = cid
  168                   , msChannelKind    = kind
  169                   , msUserId         = mUid
  170                   , msOpen           = open2
  171                   }
  172   where
  173     anyTooLong m = any (\(k,v) -> T.length k > maxOpenFieldLen || T.length v > maxOpenFieldLen)
  174                        (Map.toList m)
  175 
  176 -- | Strip control characters from a Text leaf.
  177 stripControl :: Text -> Text
  178 stripControl = T.filter (not . isControl)

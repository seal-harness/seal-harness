    2 -- | The @/new@ command: start a fresh session in a new tab.
    3 --
    4 -- @/new@ mints a new 'SessionMeta' from the config defaults (or from
    5 -- optional @-p@\/@-m@ overrides) and inserts a new tab into the
    6 -- 'TabsHandle' — the same action the web frontend's "Start a new tab"
    7 -- form performs. The old session (if any) is kept on disk, untouched —
    8 -- still listed in @\/session list@, still resumable with @\/tab resume
    9 -- \<id\>@. When @-r@\/@--repo@ is given, the repo is cloned into the
   10 -- new session's workdir via SETUP_REPO before the first turn (same as
   11 -- the web form's "Set up repo" field).
   12 --
   13 -- The @-r@\/@--repo@ argument accepts either a registered repo ID (looked
   14 -- up in the repo registry @repos.toml@) or a raw git URL. When a registered
   15 -- ID is found, the registered URL is used for the clone. When no ID
   16 -- matches, the value is treated as a raw URL.
   17 --
   18 -- == Channel wiring
   19 --
   20 -- The command is registered as a 'CommandSpec' for the CLI and web-gateway
   21 -- paths (both track "current" via an active-session ref + a 'TabsHandle').
   22 -- The inbox channels (Signal, Telegram) handle @/new@ at the loop level
   23 -- ('Seal.Channels.Loop') instead, because the per-conversation cursor +
   24 -- conversation key aren't available to a registry 'CommandAction' — see the
   25 -- design doc.
   26 module Seal.Command.New

   28   , newCommandSpec

   30   , mintNewSession

   58 -- | Optional arguments for @/new@. All fields are 'Maybe' — 'Nothing' means
   59 -- "use the config default" (for provider/model) or "no repo" (for repo).
   60 data NewArgs = NewArgs
   61   { naProvider :: Maybe Text
   62   , naModel    :: Maybe Text
   63   , naRepo     :: Maybe Text
   64   } deriving stock (Eq, Show)
   65 
   66 -- | The empty 'NewArgs' — no overrides, use all config defaults.
   67 emptyNewArgs :: NewArgs
   68 emptyNewArgs = NewArgs { naProvider = Nothing, naModel = Nothing, naRepo = Nothing }
   69 
   70 -- | Parse the raw argument text from the routing layer into 'NewArgs'.
   71 -- This is a simple manual parse used by the inbox-channel loop path
   72 -- ('Seal.Channels.Loop.handleNewSession'), which doesn't go through
   73 -- optparse-applicative. Supports @-p@\/@--provider@, @-m@\/@--model@, and
   74 -- @-r@\/@--repo@ with a single value each. Unknown flags are ignored.
   75 parseNewArgs :: Text -> NewArgs
   76 parseNewArgs raw =
   77   go (splitArgs raw) emptyNewArgs
   78   where
   79     go [] acc = acc
   80     go (flag : val : rest) acc
   81       | flag `elem` ["-p", "--provider"] = go rest acc { naProvider = Just val }
   82       | flag `elem` ["-m", "--model"]    = go rest acc { naModel    = Just val }
   83       | flag `elem` ["-r", "--repo"]     = go rest acc { naRepo     = Just val }
   84       | otherwise                        = go (val : rest) acc
   85     go (_flag : rest) acc = go rest acc
   86 
   87     -- Split on whitespace, preserving non-empty tokens.
   88     splitArgs t = case T.words t of
   89       [] -> []
   90       ws -> ws

   92 -- | Resolve the @-r@\/@--repo@ value to a git URL. First tries the value
   93 -- as a registered repo ID (looked up via 'RepoRegistryHandle'); if found,
   94 -- returns the registered 'srUrl'. If no ID matches, falls back to treating
   95 -- the value as a raw git URL. This lets the user type @\/new -r myrepo@
   96 -- (a registered id) or @\/new -r https:\/\/github.com\/foo\/bar.git@ (a raw
   97 -- URL) interchangeably.
   98 resolveRepoUrl :: RepoRegistryHandle -> Text -> IO Text
   99 resolveRepoUrl regH raw = do
  100   eRepos <- rrhList regH
  101   case eRepos of
  102     Right repos | Just repo <- matchById repos -> pure (srUrl repo)
  103     _ -> pure raw
  104   where
  105     matchById repos = do
  106       rid <- either (const Nothing) Just (mkRepoId (T.strip raw))
  107       listToMaybe [ r | r <- repos, srId r == rid ]

  109 -- | The deps a channel needs to run @/new@. Built once at startup.
  110 --
  111 -- 'ndInsertTab' is the seam where each channel plugs in its tab-insertion
  112 -- + active-session-swap logic. It receives the freshly-minted
  113 -- 'SessionMeta' and the 'ChannelCaps', inserts a new tab bound to the new
  114 -- session, swaps the active-session ref to the new meta, and returns the
  115 -- old sid (so the caller can render the confirmation line naming it).
  116 data NewDeps = NewDeps
  117   { ndPaths        :: SealPaths
  118   , ndCfg          :: IO RuntimeConfig
  119   , ndAgentDefs    :: Backends
  120   , ndChannelLabel :: Text
  121   , ndOldMeta      :: IO SessionMeta
  122     -- ^ Read the current (pre-swap) active session. Used to (a) determine
  123     -- the old sid for the confirmation line, and (b) as the source of
  124     -- default provider/model when no overrides are given.
  125   , ndInsertTab     :: ChannelCaps -> SessionMeta -> IO SessionId
  126     -- ^ Insert a new tab bound to the new session, swap the active-session
  127     -- ref, and return the old sid.
  128   , ndSetupRepo    :: Maybe (SessionId -> Text -> IO (Either Text Text))
  129     -- ^ Optional repo-setup seam. When 'Just', the @/new@ command calls it
  130     -- with the new session id + repo URL to clone a repo into the session
  131     -- workdir (dispatching SETUP_REPO). 'Nothing' in contexts without a
  132     -- dispatcher (standalone CLI, tests).
  133   , ndRepoReg      :: Maybe RepoRegistryHandle
  134     -- ^ Optional repo-registry handle for resolving @-r@ repo IDs to URLs.
  135     -- 'Nothing' in contexts without a registry (tests); the @-r@ value is
  136     -- then treated as a raw URL.
  137   }

  139 -- | The @/new@ command spec. Grouped under Sessions in @/help@. Always
  140 -- available (the operator typed it; no autonomy gate).
  141 newCommandSpec :: NewDeps -> CommandSpec
  142 newCommandSpec deps = CommandSpec
  143   { csName         = CommandName "new"
  144   , csAliases      = []
  145   , csGroup        = GroupSession
  146   , csSynopsis     = "Start a new tab with a fresh session (optional -p/-m/-r overrides)"
  147   , csParserInfo   = newParserInfo deps
  148   , csAvailability = AlwaysAvailable
  149   }
  150 
  151 newParserInfo :: NewDeps -> ParserInfo CommandAction
  152 newParserInfo deps =
  153   info (newCmd deps <$> optional providerOpt <*> optional modelOpt <*> optional repoOpt <**> helper)
  154     (  progDesc "Start a new tab with a fresh session"
  155     <> header   "new — start a new tab with a fresh session (old session kept in /session list)"
  156     )

  182 -- | The @/new@ action: read the old session, mint a fresh session using
  183 -- the given args (or config defaults when not specified), insert a new
  184 -- tab, optionally clone a repo, and send the confirmation line.
  185 newCmd :: NewDeps -> Maybe Text -> Maybe Text -> Maybe Text -> CommandAction
  186 newCmd deps mProvider mModel mRepo = CommandAction $ \caps -> do
  187   oldMeta <- ndOldMeta deps
  188   let args = NewArgs { naProvider = mProvider, naModel = mModel, naRepo = mRepo }
  189   meta <- mintNewSessionWith args oldMeta deps
  190   oldSid <- ndInsertTab deps caps meta
  191   case (naRepo args, ndSetupRepo deps) of
  192     (Just repoVal, Just setupFn) -> do
  193       repoUrl <- resolveRepoUrlMaybe deps repoVal
  194       eRes <- setupFn (smId meta) repoUrl
  195       case eRes of
  196         Left err -> ccSend caps ("repo setup failed: " <> err)
  197         Right _  -> pure ()
  198     _ -> pure ()
  199   ccSend caps (renderNewConfirmation meta oldSid)
  200 
  201 -- | Resolve the @-r@ value to a URL using the repo registry (if wired),
  202 -- falling back to the raw value.
  203 resolveRepoUrlMaybe :: NewDeps -> Text -> IO Text
  204 resolveRepoUrlMaybe deps raw =
  205   case ndRepoReg deps of
  206     Just regH -> resolveRepoUrl regH raw
  207     Nothing   -> pure raw

  209 -- | Mint a fresh 'SessionMeta' using the given args, falling back to the
  210 -- old session's provider/model when no override is given (so mid-session
  211 -- @\/model use@ changes survive @\/new@), and ultimately to config
  212 -- defaults. Persisted to disk via 'newSession' so @/session list@ picks it
  213 -- up. Shared by the registry path (CLI/web) and the loop-level inbox path.
  214 mintNewSessionWith :: NewArgs -> SessionMeta -> NewDeps -> IO SessionMeta
  215 mintNewSessionWith args oldMeta deps =
  216   newSession (ndPaths deps) provider model (ndChannelLabel deps) (smAgent oldMeta)
  217   where
  218     provider = fromMaybe (smProvider oldMeta) (naProvider args)
  219     model    = fromMaybe (smModel oldMeta) (naModel args)
  220 
  221 -- | Mint a fresh 'SessionMeta' from the config defaults (or the given
  222 -- overrides). Used when there is no "old session" to inherit from (e.g.
  223 -- the loop-level path on first conversation message).
  224 mintNewSession :: NewArgs -> NewDeps -> IO SessionMeta
  225 mintNewSession args deps = do
  226   cfg <- ndCfg deps
  227   (mAgent, mProv, mModel) <- resolveDefaultAgent (bAgentDefs (ndAgentDefs deps)) cfg
  228   let (cfgProv, cfgModel) = defaultSessionSelection cfg
  229       provider = fromMaybe (fromMaybe cfgProv mProv) (naProvider args)
  230       model    = fromMaybe (fromMaybe cfgModel mModel) (naModel args)
  231   newSession (ndPaths deps) provider model (ndChannelLabel deps) mAgent
  232 
  233 -- | Render the one-line confirmation. Names the old session + resume hint
  234 -- so the safety net (old session kept on disk) is user-visible, per
  235 -- PM/designer review.
  236 renderNewConfirmation :: SessionMeta -> SessionId -> Text
  237 renderNewConfirmation meta oldSid =
  238   "new session " <> sessionIdText (smId meta)
  239     <> " (" <> smProvider meta <> "/" <> smModel meta <> ")"
  240     <> " — prior session " <> sessionIdText oldSid
  241     <> " kept in /session list"

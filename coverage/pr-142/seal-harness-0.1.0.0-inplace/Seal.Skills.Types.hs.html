    2 -- | The skill entry model. A *skill* is a named Markdown bundle (name +
    3 -- description + body) the agent can read into a prompt or update. 'SkillId' is
    4 -- a smart-constructed newtype with the same charset predicate as
    5 -- 'Seal.Core.Types.SessionId' (@[A-Za-z0-9_-]+@, non-empty, no leading dot). The
    6 -- skill body is agent-visible data (not a vault secret); it is recorded in full
    7 -- in both the session transcript and the Audited log.
    8 module Seal.Skills.Types
    9   ( SkillId (..)
   10   , mkSkillId
   11   , isValidSkillId
   12   , skillIdText
   13   , Skill (..)
   14   ) where

   24 -- | Opaque skill key. Smart-constructed via 'mkSkillId'; the charset predicate
   25 -- guards every Audited-log / path / SQL-parameter position.
   26 newtype SkillId = SkillId Text
   27   deriving stock (Eq, Ord, Show)
   28   deriving newtype (ToJSON, FromJSON)
   29 
   30 -- | @[A-Za-z0-9_-]+@, non-empty, no leading dot. Mirrors 'isValidMemoryId'.
   31 isValidSkillId :: Text -> Bool
   32 isValidSkillId t =
   33   not (T.null t)
   34     && T.head t /= '.'
   35     && T.all (`elem` chars) t
   36   where
   37     chars = ['A' .. 'Z'] <> ['a' .. 'z'] <> ['0' .. '9'] <> "_-"
   38 
   39 mkSkillId :: Text -> Either Text SkillId
   40 mkSkillId t
   41   | isValidSkillId t = Right (SkillId t)
   42   | otherwise        = Left ("invalid skill id: " <> T.pack (show t))
   43 
   44 skillIdText :: SkillId -> Text
   45 skillIdText (SkillId t) = t

   47 -- | One agent skill. The body is agent-visible data (not a vault secret); it is
   48 -- recorded in full in both the session transcript and the Audited log.
   49 -- 'skSession' is the originating session (provenance).
   50 --
   51 -- 'skGroup' is an optional category for display grouping in the
   52 -- @\<available_skills\>@ catalog. It is derived from the on-disk parent
   53 -- directory name (@config\/skills\/\<group\>\/\<id\>.md@) and may be
   54 -- overridden via a @group:@ frontmatter key. 'Nothing' means the skill
   55 -- belongs to the default (ungrouped) section. The skill /id/ stays flat
   56 -- regardless of group (the charset predicate forbids @\/@), so group is
   57 -- purely display metadata — opcodes, file mapping, and lookups all key
   58 -- on 'skId'.
   59 data Skill = Skill
   60   { skId          :: SkillId
   61   , skDescription :: Text
   62   , skBody        :: Text
   63   , skGroup       :: Maybe Text
   64   , skCreatedAt   :: UTCTime
   65   , skUpdatedAt   :: UTCTime
   66   , skSession     :: SessionId
   67   } deriving stock (Eq, Show, Generic)
   68 
   69 instance ToJSON Skill
   70 instance FromJSON Skill

    3 -- | The wire snapshot for the WS @lists@ frame and the REST @GET /api/lists@
    4 -- endpoint. Carries the partitioned session lists (mutually exclusive by
    5 -- construction via 'partitionSessions') plus the set of sessions currently
    6 -- in a @thinking@ turn (so a freshly-connected web client can hydrate its
    7 -- sidebar without waiting for the next harness-status event). The WS frame
    8 -- wraps this with @{"type": "lists", ...}@ (added by the broadcast path);
    9 -- the REST body is the bare record (no @type@ field).
   10 --
   11 -- Takes 'TabsHandle' + 'SealPaths' directly (NOT 'ApiDeps') so this module
   12 -- does NOT import 'Seal.Gateway.API' — avoids a source-level import cycle
   13 -- ('Seal.Gateway.API' imports this module for the /api/lists route).
   14 module Seal.Gateway.ListsSnapshot
   15   ( ListsSnapshotWire (..)
   16   , buildListsSnapshot
   17   ) where

   34 -- | The partitioned snapshot. Haskell record fields use the @lsw@ prefix;
   35 -- the 'ToJSON' instance drops it (wire keys: @tabs@, @recentSessions@,
   36 -- @archivedSessions@, @tabSessions@, @thinkingSessionIds@).
   37 data ListsSnapshotWire = ListsSnapshotWire
   38   { lswTabs             :: [A.Value]
   39   , lswRecentSessions   :: [A.Value]
   40   , lswArchivedSessions :: [A.Value]
   41   , lswTabSessions      :: [A.Value]
   42   , lswThinkingSessionIds :: [Text]
   43   } deriving stock (Eq, Show, Generic)
   44 
   45 instance ToJSON ListsSnapshotWire where
   46   toJSON s = object
   47     [ "tabs"             .= lswTabs s
   48     , "recentSessions"   .= lswRecentSessions s
   49     , "archivedSessions" .= lswArchivedSessions s
   50     , "tabSessions"      .= lswTabSessions s
   51     , "thinkingSessionIds" .= lswThinkingSessionIds s
   52     ]
   53 
   54 -- | Build the partitioned snapshot. Takes the components directly (not
   55 -- 'ApiDeps') so this module stays free of a cycle with 'Seal.Gateway.API'.
   56 -- The @thinking@ set is the broker's current in-memory view of which
   57 -- sessions are mid-turn, so a freshly-connected web client can hydrate
   58 -- its sidebar (the live activity stream only carries forward from here).
   59 buildListsSnapshot :: TabsHandle -> SealPaths -> Set SessionId -> IO ListsSnapshotWire
   60 buildListsSnapshot tabsH paths thinkingSids = do
   61   tl <- snapshotTabs tabsH
   62   let tabsJson = map tabToJson (tlTabs tl)
   63   recent   <- listSessions paths
   64   archived <- listArchivedSessions paths
   65   let ps = partitionSessions tl recent archived
   66   recentJson   <- mapM (sessionInfoJsonWithSnippet paths) (psRecentSessions ps)
   67   archivedJson <- mapM (sessionInfoJsonWithSnippet paths) (psArchivedSessions ps)
   68   tabbedJson   <- mapM (sessionInfoJsonWithSnippet paths) (psTabSessions ps)
   69   pure ListsSnapshotWire
   70     { lswTabs = tabsJson
   71     , lswRecentSessions = recentJson
   72     , lswArchivedSessions = archivedJson
   73     , lswTabSessions = tabbedJson
   74     , lswThinkingSessionIds = map sessionIdText (Set.toList thinkingSids)
   75     }

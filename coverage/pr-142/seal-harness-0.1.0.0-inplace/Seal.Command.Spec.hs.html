   51 -- | NOTE: /help is NOT a registered spec — it is a meta-operation handled by
   52 -- Seal.Command.Help / Seal.Command.Parse over the Registry (avoids the
   53 -- registry-needs-itself knot). Feature modules build their own CommandSpec
   54 -- and the startup wiring assembles the Registry.
   55 newtype Registry = Registry { registrySpecs :: [CommandSpec] }

   57 mkRegistry :: [CommandSpec] -> Registry

   60 -- | Case-insensitive lookup by primary name or any alias.
   61 lookupSpec :: Registry -> CommandName -> Maybe CommandSpec
   62 lookupSpec (Registry specs) (CommandName needle) =
   63   find matchesAny specs
   64   where
   65     lower = T.toCaseFold needle
   66     nameEq (CommandName n) = T.toCaseFold n == lower
   67     matchesAny spec = nameEq (csName spec) || any nameEq (csAliases spec)

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The @/bg@ command: run a prompt in a fresh session using the default
    3 -- model + agent, and send the LLM response back on the channel the command
    4 -- was invoked from.
    5 --
    6 -- This module is deliberately thin: the @/bg@ 'CommandAction' just delegates
    7 -- to a channel-supplied 'BgRunner'. Each channel implements its own
    8 -- 'BgRunner' (in 'Seal.Channels.Loop' for inbox-driven channels,
    9 -- 'Seal.Channel.Cli' for the TUI) so that:
   10 --
   11 --   * the turn is forked (the receive loop keeps running so confirmation
   12 --     answers can be delivered asynchronously);
   13 --   * the confirmation prompt (@ccPrompt@) routes through the channel's
   14 --     own approval UX — the same seam a normal turn uses — so the reply
   15 --     and any @Allow SHELL_EXEC ...? [y/N]@ prompts land on the
   16 --     originating channel;
   17 --   * no tab or cursor state is mutated (the fresh @/bg@ session runs
   18 --     headless relative to the tab list).
   19 --
   20 -- The fresh session is persisted to disk (like @/tab new@) so the run
   21 -- appears in @/session list@ and is resumable.
   22 module Seal.Command.Background
   23   ( BgRunner(..)
   24   , backgroundCommandSpec
   25   , runBackground
   26   ) where
   27 
   28 import Data.Text (Text)
   29 import Data.Text qualified as T
   30 import Options.Applicative
   31   ( Parser, ParserInfo, (<**>), header, help, helper, info, many, metavar
   32   , progDesc, strArgument )
   33 
   34 import Seal.Channel.Caps (ChannelCaps (..))
   35 import Seal.Command.Spec
   36   ( Availability (..), CommandAction (..), CommandGroup (..)
   37   , CommandName (..), CommandSpec (..) )

   39 -- | A channel-supplied runner for a @/bg@ prompt. The channel builds this
   40 -- closure (minting the fresh session, forking the turn, wiring @ccPrompt@
   41 -- to its own approval UX, and delivering the reply to the invoking
   42 -- 'ChannelCaps'); the @/bg@ action just invokes it. Keeping this as a
   43 -- parameter decouples the command from the channel-specific turn +
   44 -- confirmation machinery so each channel can tailor the approval UX (e.g.
   45 -- Telegram's inline buttons, the CLI's @>@ prompt).
   46 --
   47 -- The confirmation ask is keyed to the /originating conversation's/ active
   48 -- session id (NOT the fresh bg session's) so the channel loop's per-session
   49 -- 'deliverNextAnswer' short-circuit consumes the next inbound message as
   50 -- the answer — producing a modal "answer the pending question before
   51 -- resuming normal turns" state scoped to that conversation. The runner
   52 -- resolves that sid itself (the @/bg@ 'CommandAction' only receives
   53 -- 'ChannelCaps', not conversation context, so the sid cannot be passed in
   54 -- at action-invocation time); channels typically capture a mutable
   55 -- "current conversation sid" cell that the loop updates each turn.
   56 newtype BgRunner = BgRunner { runBg :: Text -> IO () }
   57 
   58 -- | The @/bg@ command spec. The action closes over the channel's
   59 -- 'BgRunner'; optparse supplies the prompt as one or more positional
   60 -- words (joined with spaces).
   61 backgroundCommandSpec :: BgRunner -> CommandSpec
   62 backgroundCommandSpec runner = CommandSpec
   63   { csName         = CommandName "bg"
   64   , csAliases      = []
   65   , csGroup        = GroupGeneral
   66   , csSynopsis     = "Run a prompt in a fresh session and reply with the result"
   67   , csParserInfo   = bgParserInfo runner
   68   , csAvailability = AlwaysAvailable
   69   }
   70 
   71 bgParserInfo :: BgRunner -> ParserInfo CommandAction
   72 bgParserInfo runner =
   73   info (bgParser runner <**> helper)
   74     (  progDesc "Run a prompt in a fresh session and reply with the result"
   75     <> header "bg — run a prompt in a fresh background session"
   76     )
   77 
   78 -- | One or more positional arguments joined with spaces to form the prompt.
   79 -- The tokenizer already joins quoted spans into a single token, so
   80 -- @/bg "tell me a joke"@ yields one token; @/bg tell me a joke@ yields
   81 -- several tokens that we re-join. A blank prompt (no tokens) is rejected at
   82 -- run time with a usage line.
   83 bgParser :: BgRunner -> Parser CommandAction
   84 bgParser runner =
   85   runBackground runner . T.intercalate " " <$> many wordArg
   86   where
   87     wordArg = strArgument (metavar "PROMPT..." <> help "The prompt to run")
   88 
   89 -- | The @/bg@ action: reject a blank prompt, otherwise delegate to the
   90 -- channel's 'BgRunner'. The runner owns session creation, forking,
   91 -- confirmation routing, and reply delivery.
   92 runBackground :: BgRunner -> Text -> CommandAction
   93 runBackground runner prompt = CommandAction $ \caps ->
   94   if T.null (T.strip prompt)
   95     then ccSend caps "usage: /bg <prompt>"
   96     else runBg runner prompt

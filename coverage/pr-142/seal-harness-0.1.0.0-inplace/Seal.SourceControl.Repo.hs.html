   79 -- | The credential bound to a 'SourceRepo'. Carries only vault key /names/ and
   80 -- (for 'CredMachineUser') a public username — never secret bytes.
   81 data RepoCredential
   82   = CredPat         { cVaultKey :: Text }
   83     -- ^ A personal access token (or fine-grained PAT); vault key → token bytes.
   84   | CredDeployKey   { cVaultKey :: Text }
   85     -- ^ An SSH deploy key; vault key → private key bytes.
   86   | CredMachineUser { cVaultKey :: Text, cUsername :: Text }
   87     -- ^ A machine-user HTTPS clone: vault key → token, plus a public bot handle.
   88     -- @cUsername@ is a public handle (a GitHub bot account name), NOT a secret.
   89   deriving stock (Eq, Show)

   91 data SourceRepo = SourceRepo
   92   { srId               :: RepoId
   93   , srUrl              :: Text
   94   , srVcsKind          :: VcsKind
   95   , srCredential       :: RepoCredential
   96   , srDeployKeyPublic  :: Maybe Text
   97     -- ^ The deploy-key public key (present iff 'srCredential' is
   98     -- 'CredDeployKey' and the key was generated). Public data — safe to

  100   , srKeyfilePath      :: Maybe Text
  101     -- ^ The path to the encrypted keyfile on the harness disk
  102     -- (@repoKeysDir </> repo-id>@). The file is CIPHERTEXT (encrypted by

  116 -- | Opaque repo key. Smart-constructed via 'mkRepoId'; the charset predicate
  117 -- guards every Map position. 'RepoId' is never used to construct a 'FilePath'

  119 newtype RepoId = RepoId Text

  122 -- | @[A-Za-z0-9_-]+@, non-empty, no leading @\.@. Mirrors
  123 -- 'Seal.Skills.Types.isValidSkillId'.
  124 isValidRepoId :: Text -> Bool
  125 isValidRepoId t =
  126   not (T.null t)
  127     && T.head t /= '.'
  128     && T.all (`elem` chars) t
  129   where
  130     chars = ['A' .. 'Z'] <> ['a' .. 'z'] <> ['0' .. '9'] <> "_-"
  131 
  132 mkRepoId :: Text -> Either Text RepoId
  133 mkRepoId t
  134   | isValidRepoId t = Right (RepoId t)
  135   | otherwise       = Left ("invalid repo id: " <> T.pack (show t))

  153 repoCredentialKindText :: RepoCredential -> Text
  154 repoCredentialKindText CredPat{}         = "pat"
  155 repoCredentialKindText CredDeployKey{}   = "deploy_key"
  156 repoCredentialKindText CredMachineUser{} = "machine_user"

  158 -- | Build a 'RepoCredential' from a wire kind + a vault key name + an optional
  159 -- username. Validates that @machine_user@ carries a non-empty username and
  160 -- that the vault key is non-empty. Used by the REST API (W4) and slash UI (W5).
  161 --
  162 -- @account_key@ is a /reserved/ credential kind (a future constructor may add
  163 -- it; until then it fails-closed so a repo registered with @account_key@
  164 -- cannot decode — the registry MUST NOT silently accept an unknown kind).
  165 parseCredentialKind :: Text -> Text -> Maybe Text -> Either Text RepoCredential
  166 parseCredentialKind kind vaultKey mUsername
  167   | T.null vaultKey = Left "credential vault_key must be non-empty"
  168   | kind == "pat"          = Right (CredPat vaultKey)
  169   | kind == "deploy_key"   = Right (CredDeployKey vaultKey)
  170   | kind == "machine_user" =
  171       case mUsername of
  172         Just u | not (T.null u) -> Right (CredMachineUser vaultKey u)
  173         _                       -> Left "machine_user requires a non-empty username"
  174   | kind == "account_key" = Left "account_key credential kind is reserved (not yet supported)"
  175   | otherwise = Left ("unknown credential_kind: " <> kind)

  181 -- | Bidirectional tomland codec for ONE @[repos.\<id\>]@ table. The @id@
  182 -- comes from the enclosing 'Toml.tableMap' key (via 'repoRegistryCodec') and
  183 -- is injected as the fixed 'RepoId'. Fail-closed on unknown @vcs_kind@ /
  184 -- @credential_kind@, missing @vault_key@, and @machine_user@ without
  185 -- @username@ (each is a decode 'Failure'). Extra/unknown fields are ignored
  186 -- (tolmland default).
  187 repoCodec :: RepoId -> TomlCodec SourceRepo
  188 repoCodec srId = SourceRepo srId
  189   <$> Toml.text "url"                                       .= srUrl
  190   <*> validatedTextCodec "vcs_kind" parseVcsKind vcsKindText .= srVcsKind
  191   <*> credentialCodec                                       .= srCredential
  192   <*> Toml.dioptional (Toml.text "deploy_key_public")       .= srDeployKeyPublic
  193   <*> Toml.dioptional (Toml.text "keyfile_path")            .= srKeyfilePath
  194 
  195 -- | Bidirectional tomland codec for the whole @[repos]@ table — a keyed-by-id
  196 -- map. Runs 'mkRepoId' on each decoded key (fail-closed: a bad key id → decode
  197 -- 'Failure'). Returns an empty 'Map' when the @[repos]@ table is absent.
  198 repoRegistryCodec :: TomlCodec (Map RepoId SourceRepo)
  199 repoRegistryCodec = Toml.tableMap keyRepoBiMap repoCodecByKey "repos"
  200 
  201 -- | Adapter the 'Toml.tableMap' value-codec slot requires (a @Key -> TomlCodec@
  202 -- function). Converts the raw TOML 'Key' to a 'RepoId' (fail-closed via
  203 -- 'mkRepoId') and nests 'repoCodec' under a sub-table named by the key —
  204 -- mirroring @Toml.tableMap Toml._KeyText (Toml.table providerConfigCodec)
  205 -- "providers"@ in "Seal.Config.File". The 'keyRepoBiMap' already validates
  206 -- the key, so the 'Left' branch is unreachable in practice but kept for
  207 -- totality.
  208 repoCodecByKey :: Key -> TomlCodec SourceRepo
  209 repoCodecByKey rawKey =
  210   case mkRepoId (prettyKey rawKey) of
  211     Left _    -> empty
  212     Right rid -> Toml.table (repoCodec rid) rawKey
  213 
  214 -- | Bidirectional 'TomlBiMap' between TOML 'Key's and 'RepoId's. @forward@
  215 -- (Key → RepoId) runs 'mkRepoId' on the rendered key text; a bad id is a
  216 -- 'Failure' (fail-closed). @backward@ (RepoId → Key) re-parses the id text
  217 -- (which is guaranteed valid by 'mkRepoId').
  218 keyRepoBiMap :: TomlBiMap Key RepoId
  219 keyRepoBiMap = BiMap
  220   { forward  = first (ArbitraryError . ("invalid repo id key: " <>) . T.pack . show)
  221                 . mkRepoId . prettyKey
  222   , backward = first (ArbitraryError . unTomlParseError) . parseKey . repoIdText
  223   }

  229 -- | Codec for a required text field that is parsed/validated into a typed
  230 -- value. On decode: looks up the key, decodes it as 'Text', runs the parser;
  231 -- on @Left@ returns a 'Failure' (fail-closed). On encode: renders the value
  232 -- back to 'Text' and writes it. A missing key is a 'KeyNotFound' 'Failure'.
  233 validatedTextCodec :: forall a. Key -> (Text -> Either Text a) -> (a -> Text) -> TomlCodec a
  234 validatedTextCodec key parse render = Codec input output
  235   where
  236     input :: TOML -> Validation [TomlDecodeError] a
  237     input toml = case HashMap.lookup key (tomlPairs toml) of
  238       Nothing -> Failure [KeyNotFound key]
  239       Just v  -> case backward Toml._Text v of
  240         Left err -> Failure [BiMapError key err]
  241         Right txt -> case parse txt of
  242           Left msg -> Failure [BiMapError key (ArbitraryError msg)]
  243           Right a  -> Success a
  244 
  245     output :: a -> TomlState a
  246     output a = do
  247       anyVal <- eitherToTomlState (forward Toml._Text (render a))
  248       a <$ modify (insertKeyAnyVal key anyVal)

  251 -- Credential codec (fail-closed on unknown kind / missing username)

  254 -- | Codec for the @credential_kind@ + @vault_key@ + optional @username@
  255 -- fields, decoded into a 'RepoCredential'. Fail-closed on unknown
  256 -- @credential_kind@, missing @vault_key@, and @machine_user@ without
  257 -- @username@.
  258 credentialCodec :: TomlCodec RepoCredential
  259 credentialCodec = Codec input output
  260   where
  261     input :: TOML -> Validation [TomlDecodeError] RepoCredential
  262     input toml =
  263       let vKind  = fieldText          "credential_kind" toml
  264           vVault = fieldText          "vault_key"        toml
  265           vUser  = fieldOptionalText  "username"         toml
  266       in case (vKind, vVault, vUser) of
  267            (Success kind, Success vault, Success mUser) ->
  268              either (\msg -> Failure [BiMapError "credential_kind" (ArbitraryError msg)])
  269                     Success
  270                     (parseCredentialKind kind vault mUser)
  271            (e1, e2, e3) -> Failure (concatErrors e1 <> concatErrors e2 <> concatErrors e3)
  272 
  273     output :: RepoCredential -> TomlState RepoCredential
  274     output cred = do
  275       writeText "credential_kind" (repoCredentialKindText cred)
  276       writeText "vault_key"        (cVaultKey cred)
  277       case cred of
  278         CredMachineUser _ u -> writeText "username" u
  279         _                   -> pure ()
  280       pure cred
  281 
  282 -- | Collect the error list from a 'Validation' branch (used to merge
  283 -- fail-closed field errors in 'credentialCodec').
  284 concatErrors :: Validation [e] a -> [e]
  285 concatErrors (Failure es) = es
  286 concatErrors (Success _)  = []

  288 -- | Read a required text field from a 'TOML' (fail-closed on missing/typed
  289 -- mismatch). Used inside the custom 'credentialCodec' read.
  290 fieldText :: Key -> TOML -> Validation [TomlDecodeError] Text
  291 fieldText key toml = case HashMap.lookup key (tomlPairs toml) of
  292   Nothing -> Failure [KeyNotFound key]
  293   Just v  -> case backward Toml._Text v of
  294     Left err -> Failure [BiMapError key err]
  295     Right t  -> Success t
  296 
  297 -- | Read an optional text field from a 'TOML' (missing → 'Nothing').
  298 fieldOptionalText :: Key -> TOML -> Validation [TomlDecodeError] (Maybe Text)
  299 fieldOptionalText key toml = case HashMap.lookup key (tomlPairs toml) of
  300   Nothing -> Success Nothing
  301   Just v  -> case backward Toml._Text v of
  302     Left err -> Failure [BiMapError key err]
  303     Right t  -> Success (Just t)
  304 
  305 -- | Write a text key/value into the encoded TOML state.
  306 writeText :: Key -> Text -> TomlState ()
  307 writeText key txt = do
  308   anyVal <- eitherToTomlState (forward Toml._Text txt)
  309   void (modify (insertKeyAnyVal key anyVal))

  315 -- | tomland's 'Toml.tableMap' \/ 'Toml.table' combinators look up the @repos@
  316 -- node by requiring it to carry an explicit value in the parsed AST. A TOML
  317 -- file that declares only @[repos.\<id\>]@ sub-tables (the idiomatic style,
  318 -- and the one every hand-written @repos.toml@ uses) never writes a bare
  319 -- @[repos]@ header, so that node is /implicit/ in tomland's prefix tree and
  320 -- the lookup silently returns an empty map instead of the sub-tables'
  321 -- contents.
  322 --
  323 -- This walks the parsed AST once before decoding and makes that node explicit
  324 -- — without disturbing anything else — so both styles decode correctly.
  325 -- Round-tripped files (written by the encoder) already have an explicit node
  326 -- and pass through unchanged. Direct sibling of 'normalizeProvidersTable'.
  327 normalizeReposTable :: TOML -> TOML
  328 normalizeReposTable t =
  329   t { tomlTables = HashMap.adjust explicitNode "repos" (tomlTables t) }
  330   where
  331     explicitNode :: Toml.PrefixTree TOML -> Toml.PrefixTree TOML
  332     explicitNode tree = case tree of
  333       Toml.Branch pref Nothing children ->
  334         Toml.Leaf pref (mempty { tomlTables = children })
  335       Toml.Branch _ (Just _) _ -> tree
  336       Toml.Leaf (_ :|| [])  _  -> tree
  337       Toml.Leaf (_ :|| (p : ps)) v ->
  338         Toml.Leaf ("repos" :|| []) (mempty { tomlTables = Toml.single (p :|| ps) v })

  344 -- | The GitHub-first host allow-list. This pass restricts clones to
  345 -- @github.com@; a per-credential @cHost@ may widen this in a follow-up.
  346 -- @localhost@ / @127.0.0.1@ are included for integration testing (a local

  349 githubHosts :: [Text]
  350 githubHosts = ["github.com", "localhost", "127.0.0.1"]

  353 hostAllowed :: Text -> Bool
  354 hostAllowed h = h `elem` githubHosts

  356 -- | Parse the host from an SSH (@git@github.com:owner\/repo.git@) or HTTPS
  357 -- (@https://github.com/owner/repo.git@) URL. Returns @Left@ with a clear
  358 -- error on malformed URLs. Returns 'Either' 'Text' (NOT @Either CloneError@)
  359 -- so this module does not depend on W3's 'CloneError'; W3's @planClone@ lifts
  360 -- the error.
  361 parseRepoHost :: Text -> Either Text Text
  362 parseRepoHost url
  363   | T.null url = Left "empty repo URL"
  364   | Just afterAt <- stripUserAt url = parseSsh afterAt
  365   | "https://" `T.isPrefixOf` url = parseHttps
  366   | "http://" `T.isPrefixOf` url = parseHttps
  367   | otherwise = Left "URL is neither SSH (user@<host>:...) nor HTTPS (https://<host>/...)"
  368   where
  369     -- Strip the leading user@ from an SCP-form URL (git@github.com:...,
  370     -- zoe@localhost:..., etc). Returns 'Nothing' when the URL is not
  371     -- SCP-form (no colon, or no @ before the colon).
  372     stripUserAt :: Text -> Maybe Text
  373     stripUserAt u =
  374       case T.breakOn ":" u of
  375         (_, "") -> Nothing  -- no colon → not SCP-form
  376         (beforeColon, _) ->
  377           case T.breakOn "@" beforeColon of
  378             (_, rest) | not (T.null rest) -> Just (T.drop 1 rest <> T.drop (T.length beforeColon) u)
  379             _ -> Nothing
  380 
  381     parseSsh :: Text -> Either Text Text
  382     parseSsh afterUser =
  383       case T.breakOn ":" afterUser of
  384         (host, rest)
  385           | T.null rest -> Left "missing ':' in scp-form URL"
  386           | T.null host -> Left "empty host in scp-form URL"
  387           | otherwise   -> Right host
  388 
  389     parseHttps :: Either Text Text
  390     parseHttps =
  391       let rest = fromMaybe url (T.stripPrefix "https://" url <|> T.stripPrefix "http://" url)
  392           host = T.takeWhile (\c -> c /= '/' && c /= '?' && c /= '#') rest
  393       in if T.null host
  394            then Left "empty host in HTTPS URL"
  395            else Right host
  396 
  397 -- | True iff the URL is non-empty AND matches one of the two supported shapes
  398 -- (SSH @git\@\<host\>:...@ or HTTPS @https://\<host\>/...@).
  399 urlShapeValid :: Text -> Bool
  400 urlShapeValid url = not (T.null url) && (isSsh url || isHttps url)
  401   where
  402     isSsh u   = hasUserAt u && T.isInfixOf ":" u
  403     isHttps u = "https://" `T.isPrefixOf` u || "http://" `T.isPrefixOf` u
  404     hasUserAt u = case T.breakOn ":" u of
  405       (beforeColon, rest) -> not (T.null rest) && T.isInfixOf "@" beforeColon

  411 -- | Normalize a repo URL to a canonical form for cross-scheme comparison.
  412 -- The same repo reached via different schemes (e.g.
  413 -- @git\@github.com:foo/bar.git@ vs @https://github.com/foo/bar.git@) should
  414 -- compare equal so 'lookupRepoByUrl' matches across forms. Normalization:
  415 --
  416 --   * strip a scheme (@https://@, @http://@, @git://@, @ssh://@, @file://@);
  417 --   * strip a leading @user\@@ (SCP-style);
  418 --   * replace the SCP-style @:@ separator with @/@;
  419 --   * strip a trailing @.git@;
  420 --   * strip trailing slashes;
  421 --   * lowercase (GitHub URLs are case-insensitive in host/path).
  422 --
  423 -- Returns the empty string only if the input is empty after trimming.
  424 -- Idempotent: @normalizeRepoUrl (normalizeRepoUrl u) == normalizeRepoUrl u@.
  425 normalizeRepoUrl :: Text -> Text
  426 normalizeRepoUrl raw =
  427   let t1 = T.strip raw
  428       -- Strip a scheme.
  429       t2 = foldr (\s acc -> fromMaybe acc (T.stripPrefix s acc))
  430                  t1 [ "https://", "http://", "git://", "ssh://", "file://" ]
  431       -- Strip a leading user@ (SCP-style: git@host:path).
  432       t3 = case T.breakOn "@" t2 of
  433              (_, rest) | not (T.null rest) -> T.drop 1 rest
  434              _ -> t2
  435       -- Replace the SCP ':' separator with '/'. Only the FIRST ':'
  436       -- (after the host) is the separator; later ':' would be port-like,
  437       -- but a bare SCP url has exactly one ':'. If there's no '/', the
  438       -- first ':' is host/path; if there's a '/' before the ':', it's
  439       -- not SCP-style (e.g. ssh://host:port/path already had scheme
  440       -- stripped, so unlikely) — leave it.
  441       t4 = case T.breakOn ":" t3 of
  442              (host, rest) | not (T.null rest), not (T.any (== '/') host) ->
  443                host <> "/" <> T.drop 1 rest
  444              _ -> t3
  445       -- Strip trailing slashes, then a trailing .git, then any slash
  446       -- that the .git-strip might have exposed (e.g. ".../repo.git/"
  447       -- → ".../repo.git" → ".../repo"). Apply twice in case of a
  448       -- ".../repo.git/."-like edge; the composition is idempotent.
  449       stripTail t = T.dropWhileEnd (== '/') (fromMaybe t (T.stripSuffix ".git" (T.dropWhileEnd (== '/') t)))
  450       t5 = stripTail t4
  451   in T.toLower t5
  452 
  453 -- | Look up a 'SourceRepo' in a 'RepoRegistry' by URL, matching across
  454 -- scheme forms (SSH @git\@host:o/r.git@ ↔ HTTPS @https://host/o/r.git@ ↔
  455 -- @https://host/o/r@). Both the query URL and each 'srUrl' are normalized
  456 -- via 'normalizeRepoUrl' before comparison. Pure.
  457 lookupRepoByUrl :: Text -> RepoRegistry -> Maybe SourceRepo
  458 lookupRepoByUrl query (RepoRegistry repos) =
  459   let q = normalizeRepoUrl query
  460   in listToMaybe
  461        [ r | r <- Map.elems repos, normalizeRepoUrl (srUrl r) == q ]

  477 instance FromJSON SourceRepo where
  478   parseJSON = withObject "SourceRepo" $ \o -> do
  479     rid    <- o .:  "id"
  480     url    <- o .:  "url"
  481     kind   <- o .:  "vcs_kind"
  482     cred   <- o .:  "credential"
  483     mPub   <- o .:? "deploy_key_public"
  484     mPath  <- o .:? "keyfile_path"
  485     i  <- either (fail . T.unpack) pure (mkRepoId rid)
  486     vk <- either (fail . T.unpack) pure (parseVcsKind kind)
  487     cred' <- parseCredentialFromObject cred
  488     pure SourceRepo
  489       { srId = i
  490       , srUrl = url
  491       , srVcsKind = vk
  492       , srCredential = cred'
  493       , srDeployKeyPublic = mPub
  494       , srKeyfilePath = mPath
  495       }
  496 
  497 credentialToObject :: RepoCredential -> Value
  498 credentialToObject c = object $ case c of
  499   CredPat k ->
  500     [ "kind" Aeson..= ("pat" :: Text), "vault_key" Aeson..= k ]
  501   CredDeployKey k ->
  502     [ "kind" Aeson..= ("deploy_key" :: Text), "vault_key" Aeson..= k ]
  503   CredMachineUser k u ->
  504     [ "kind" Aeson..= ("machine_user" :: Text), "vault_key" Aeson..= k, "username" Aeson..= u ]
  505 
  506 parseCredentialFromObject :: Value -> Parser RepoCredential
  507 parseCredentialFromObject = withObject "RepoCredential" $ \o -> do
  508   kind  <- o .:  "kind"
  509   vault <- o .:  "vault_key"
  510   mUser <- o .:? "username"
  511   either (fail . T.unpack) pure (parseCredentialKind kind vault mUser)

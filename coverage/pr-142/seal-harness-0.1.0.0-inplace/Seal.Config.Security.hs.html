    3 -- | The security-critical, boot-only configuration that lives in
    4 -- @~\/.seal\/security.toml@ — a separate file from @config.toml@ (which holds
    5 -- the agent/operator-tunable 'RuntimeConfig'). This file is read once at boot
    6 -- and never re-read; it is not exposed via any opcode or the HTTP Gateway.
    7 --
    8 -- Moving @untrusted_execution@ and the vault settings here means a future
    9 -- @CONFIG_UPDATE@ opcode (which operates on 'RuntimeConfig') and the Gateway's
   10 -- @updateRuntimeConfig@ caller physically cannot express a change to these
   11 -- fields — the type split makes it a compile error (design §4 Approach E).
   12 --
   13 -- Every field is optional; a missing key decodes as 'Nothing' and a
   14 -- 'Nothing' value is omitted from the encoded output.

   44 -- | The boot-only, agent-immutable security configuration. Carries the
   45 -- @untrusted_execution@ section (remote-only enforcement) and the vault
   46 -- settings. Loaded once at boot from @security.toml@; never passed to any
   47 -- opcode or the Gateway.
   48 data SecurityConfig = SecurityConfig
   49   { scVaultPath :: Maybe Text
   50     -- ^ Absolute path to the vault file (default: @~\/.seal\/config\/vault\/vault.age@).
   51   , scVaultRecipient :: Maybe Text
   52     -- ^ age public key: @age1…@ or @age1yubikey1…@.
   53   , scVaultIdentity :: Maybe Text
   54     -- ^ Path to the identity file under @keys\/@, or a user-supplied path.
   55   , scVaultUnlock :: Maybe Text
   56     -- ^ @\"startup\"@ | @\"on_demand\"@ | @\"per_access\"@.
   57   , scVaultKeyType :: Maybe Text
   58     -- ^ Display label: @\"x25519\"@ | @\"yubikey\"@ | @\"user\"@.
   59   , scUntrustedExec :: Maybe UntrustedExecFileConfig
   60     -- ^ Optional @[untrusted_execution]@ section (remote-only untrusted
   61     -- execution). Absent means @mode=local@ (default). @mode=remote@
   62     -- fail-closes at call time when the remote block is absent or
   63     -- unreachable (boot still succeeds).
   64   } deriving stock (Eq, Show)
   65 
   66 -- | The @[untrusted_execution]@ section (spec §3 Layer A). @mode@ is
   67 -- @\"local\"@ (default) or @\"remote\"@ (fail-closed). The optional
   68 -- @[untrusted_execution.remote]@ sub-table carries the SSH coordinates;
   69 -- absent under @mode=local@ (no remote needed), and @mode=remote@ parses
   70 -- OK without it (fail-closed is at call time, not parse time — spec §7
   71 -- row 1).
   72 data UntrustedExecFileConfig = UntrustedExecFileConfig
   73   { uefcMode   :: Text
   74     -- ^ @\"local\"@ | @\"remote\"@
   75   , uefcRemote :: Maybe UntrustedExecRemoteFileConfig
   76   } deriving stock (Eq, Show)
   77 
   78 -- | The @[untrusted_execution.remote]@ sub-table (spec §5). Every field is
   79 -- optional at the TOML layer; the call site validates that the required
   80 -- fields (@host@/@user@/@known_hosts@/@workspace@) are present when
   81 -- @mode=remote@ (absent ⇒ fail-closed at call time).
   82 data UntrustedExecRemoteFileConfig = UntrustedExecRemoteFileConfig
   83   { uerfcHost       :: Maybe Text
   84   , uerfcUser       :: Maybe Text
   85   , uerfcPort       :: Maybe Int
   86   , uerfcIdentity   :: Maybe FilePath
   87   , uerfcKnownHosts :: Maybe FilePath
   88   , uerfcWorkspace  :: Maybe Text
   89   } deriving stock (Eq, Show)
   90 
   91 -- | Starting state: all fields absent.
   92 defaultSecurityConfig :: SecurityConfig
   93 defaultSecurityConfig = SecurityConfig
   94   { scVaultPath      = Nothing
   95   , scVaultRecipient = Nothing
   96   , scVaultIdentity  = Nothing
   97   , scVaultUnlock    = Nothing
   98   , scVaultKeyType   = Nothing
   99   , scUntrustedExec  = Nothing
  100   }

  106 -- | Bidirectional tomland codec for 'SecurityConfig'.
  107 securityConfigCodec :: Toml.TomlCodec SecurityConfig
  108 securityConfigCodec = SecurityConfig
  109   <$> Toml.dioptional (Toml.text "vault_path")     .= scVaultPath
  110   <*> Toml.dioptional (Toml.text "vault_recipient") .= scVaultRecipient
  111   <*> Toml.dioptional (Toml.text "vault_identity")  .= scVaultIdentity
  112   <*> Toml.dioptional (Toml.text "vault_unlock")    .= scVaultUnlock
  113   <*> Toml.dioptional (Toml.text "vault_key_type")  .= scVaultKeyType
  114   <*> Toml.dioptional (Toml.table untrustedExecConfigCodec "untrusted_execution") .= scUntrustedExec

  116 -- | Bidirectional tomland codec for the @[untrusted_execution]@ section.
  117 untrustedExecConfigCodec :: Toml.TomlCodec UntrustedExecFileConfig
  118 untrustedExecConfigCodec = UntrustedExecFileConfig
  119   <$> Toml.text "mode" .= uefcMode
  120   <*> Toml.dioptional (Toml.table untrustedExecRemoteConfigCodec "remote") .= uefcRemote

  122 -- | Bidirectional tomland codec for the @[untrusted_execution.remote]@
  123 -- sub-table. Every field is optional at the TOML layer.
  124 untrustedExecRemoteConfigCodec :: Toml.TomlCodec UntrustedExecRemoteFileConfig
  125 untrustedExecRemoteConfigCodec = UntrustedExecRemoteFileConfig
  126   <$> Toml.dioptional (Toml.text "host")        .= uerfcHost
  127   <*> Toml.dioptional (Toml.text "user")        .= uerfcUser
  128   <*> Toml.dioptional (Toml.int "port")         .= uerfcPort
  129   <*> Toml.dioptional (Toml.string "identity")  .= uerfcIdentity
  130   <*> Toml.dioptional (Toml.string "known_hosts") .= uerfcKnownHosts
  131   <*> Toml.dioptional (Toml.text "workspace")   .= uerfcWorkspace

  139 -- | Load the security config file at @path@.
  140 --
  141 -- * File absent  → @Right 'defaultSecurityConfig'@
  142 -- * Parse error  → @Left@ with the rendered tomland diagnostics
  143 loadSecurityConfig :: FilePath -> IO (Either Text SecurityConfig)
  144 loadSecurityConfig path = do
  145   exists <- doesFileExist path
  146   if not exists
  147     then pure (Right defaultSecurityConfig)
  148     else do
  149       contents <- TIO.readFile path
  150       pure $ case Toml.parse contents of
  151         Left err   -> Left (Toml.unTomlParseError err)
  152         Right toml -> case Toml.runTomlCodec securityConfigCodec toml of
  153           Success cfg  -> Right cfg
  154           Failure errs -> Left (Toml.prettyTomlDecodeErrors errs)
  155 
  156 -- | Save @cfg@ to @path@ atomically: write @.tmp@, chmod 0600, rename over
  157 -- @path@. The file is chmod-restricted because @security.toml@ carries the
  158 -- vault recipient/identity (security-critical, agent-immutable).
  159 saveSecurityConfig :: FilePath -> SecurityConfig -> IO ()
  160 saveSecurityConfig path cfg = do
  161   let encoded = Toml.encode securityConfigCodec cfg
  162       tmp     = path <> ".tmp"
  163   TIO.writeFile tmp encoded
  164   setFileMode tmp 0o600
  165   renameFile tmp path

  167 -- | Process-wide lock serializing security config writes (design V7).
  168 {-# NOINLINE securityWriteLock #-}
  169 securityWriteLock :: MVar ()
  170 securityWriteLock = unsafePerformIO (newMVar ())
  171 
  172 -- | Load the security config at @path@, apply @f@, save. Propagates any load
  173 -- error as @Left Text@ without writing. This is the admin/boot path (e.g.
  174 -- @\/vault setup@); it is NEVER called by an opcode or the HTTP Gateway
  175 -- (those operate on 'RuntimeConfig' and physically cannot reach
  176 -- 'SecurityConfig' — design §4 E). Serialized behind 'securityWriteLock'
  177 -- to prevent lost-update races (design V7).
  178 updateSecurityConfig :: FilePath -> (SecurityConfig -> SecurityConfig) -> IO (Either Text ())
  179 updateSecurityConfig path f = withMVar securityWriteLock $ \_ -> do
  180   result <- loadSecurityConfig path
  181   case result of
  182     Left err  -> pure (Left err)
  183     Right cfg -> saveSecurityConfig path (f cfg) >> pure (Right ())

  185 -- | Resolve the 'SecurityConfig'\'s @[untrusted_execution]@ section to the
  186 -- typed 'UntrustedExecConfig' the dispatcher consumes. Returns
  187 -- 'Nothing' when the section is absent OR @mode=local@ (the default —
  188 -- no remote needed, the local executor is wired by the call site).
  189 -- Returns @Just (UemRemote, ...remote...)@ when @mode=remote@,
  190 -- with the remote 'SshConfig' if the remote block is fully present
  191 -- (host/user/known_hosts/workspace all set). A @mode=remote@ with a
  192 -- missing/incomplete remote block returns @Just (UemRemote, Nothing)@
  193 -- — fail-closed is at call time (spec §7 row 1), not at config resolution.
  194 untrustedExecConfigFromSecurity :: SecurityConfig -> Maybe UntrustedExecConfig
  195 untrustedExecConfigFromSecurity cfg =
  196   resolveByMode =<< scUntrustedExec cfg
  197 
  198 resolveByMode :: UntrustedExecFileConfig -> Maybe UntrustedExecConfig
  199 resolveByMode uec =
  200   let mode = if uefcMode uec == "remote" then UemRemote else UemLocal
  201   in case mode of
  202        UemRemote -> Just (UntrustedExecConfig UemRemote (resolveRemoteSsh uec))
  203        UemLocal  -> localModeResult uec
  204 
  205 -- | Resolve the [untrusted_execution.remote] sub-table to an 'SshConfig'.
  206 -- Shared by both the UemRemote branch and the hardened-build localModeResult.
  207 resolveRemoteSsh :: UntrustedExecFileConfig -> Maybe SshConfig
  208 resolveRemoteSsh uec = uefcRemote uec >>= \r ->
  209   uerfcHost r >>= \host ->
  210   uerfcUser r >>= \user ->
  211   uerfcKnownHosts r >>= \knownHosts ->
  212   uerfcWorkspace r >>= \workspace ->
  213   let port = fromMaybe 22 (uerfcPort r)
  214       identity = uerfcIdentity r
  215   in case ( mkSshHost host
  216           , mkSshUser user
  217           , mkRemotePath workspace
  218           ) of
  219        (Right h, Right u, Right w) -> Just SshConfig
  220          { scHost       = h
  221          , scUser       = u
  222          , scPort       = port
  223          , scIdentity   = identity
  224          , scKnownHosts = knownHosts
  225          , scWorkspace  = w
  226          }
  227        _ -> Nothing
  228 
  229 #if defined(REMOTE_ONLY_UNTRUSTED)
  230 localModeResult :: UntrustedExecFileConfig -> Maybe UntrustedExecConfig
  231 localModeResult uec =
  232   Just (UntrustedExecConfig UemRemote (resolveRemoteSsh uec))
  233 #else
  234 localModeResult :: UntrustedExecFileConfig -> Maybe UntrustedExecConfig
  235 localModeResult _ = Nothing
  236 #endif

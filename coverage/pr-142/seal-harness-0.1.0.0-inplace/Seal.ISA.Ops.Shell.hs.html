    2 -- | SHELL_EXEC (Untrusted): run a validated shell command via the
    3 -- 'UntrustedIO' capability. The command is a validated 'ShellCommand'
    4 -- (smart-constructed — NUL rejected, option-injection defense at the
    5 -- executor's @\/bin\/sh -c@ single-arg boundary). The cwd is
    6 -- SafePath-confined to the workspace root. The 'SecurityPolicy' gates the
    7 -- call: a 'Deny' autonomy level rejects before any execution.
    8 -- ACK-before-execute is inherited from the dispatcher (the opcode is
    9 -- 'UntrustedOpcode'). All side-effecting IO is funnelled through
   10 -- 'UntrustedIO'; this module never imports 'System.Process'.

   13   , shellExecSchema

   30 -- | @gh auth@ subcommands that write secrets to disk or print them to

   38   , "gh auth git-credential"

   41 -- | SHELL_EXEC opcode: run a validated shell command. Input: @{
   42 -- command: ShellCommand, cwd?: RemotePath }@. Authorize: the
   43 -- 'SecurityPolicy' must not be 'Deny' and the command must parse as a
   44 -- 'ShellCommand'. Run: via 'uioShellExec' (Local or Remote SSH, backend-
   45 -- selected at wiring time). 'orRecorded': the command + cwd (secret-free
   46 -- metadata).
   47 shellExecOp :: WorkspaceRoot -> SecurityPolicy -> Opcode
   48 shellExecOp _wsRoot policy = UntrustedOpcode
   49   { uoName = OpName "SHELL_EXEC"
   50   , uoDesc = "Run a shell command (validated, SafePath-confined, AuthorizedCommand-gated)."
   51   , uoInSchema = shellExecSchema
   52   , uoOutSchema = object []
   53   , uoAuthorize = \v ->
   54       case commandField v of
   55         Nothing -> Left "SHELL_EXEC requires {command:string}"
   56         Just cmd -> case mkShellCommand cmd of
   57           Left _err -> Left "SHELL_EXEC: invalid command"
   58           Right _ -> case spAutonomy policy of
   59             Deny -> Left "SHELL_EXEC denied by autonomy policy"
   60             _   -> if any (`T.isInfixOf` cmd) dangerousGhAuthCmds
   61                      then Left "SHELL_EXEC: `gh auth login/token/setup-git` is blocked — it writes secrets to disk on the untrusted machine (gh auth login stores a token in ~/.config/gh/hosts.yml; gh auth token prints the token to stdout → transcript). The harness injects GH_TOKEN from the vault automatically. Use `gh pr create` / `gh repo clone` / etc. directly."
   62                      else Right ()
   63   , uoRun = \v -> do
   64       let mCmd = commandField v
   65           mCwd = cwdField v
   66           recorded = object [ "command" .= (mCmd :: Maybe Text), "cwd" .= mCwd ]
   67       case mCmd of
   68         Nothing -> pure (OpResult [TrpText "SHELL_EXEC requires {command:string}"] True recorded)
   69         Just cmdText ->
   70           case mkShellCommand cmdText of
   71             Left err -> pure (OpResult [TrpText ("SHELL_EXEC: invalid command: " <> err)] True recorded)
   72             Right cmd -> do
   73               let mCwdPath = case mCwd of
   74                     Nothing -> Nothing
   75                     Just t -> case mkRemotePath t of
   76                       Right rp -> Just rp
   77                       Left _   -> Nothing
   78               res <- uioShellExec cmd mCwdPath
   79               pure $ case res of
   80                 Left err   -> OpResult [TrpText (renderUntrustedErr err)] True recorded
   81                 Right out -> OpResult [TrpText out] False recorded
   82   }

   98             [ "type" .= ("integer" :: Text)

  105 commandField :: Value -> Maybe Text
  106 commandField = parseMaybe (withObject "in" (.: "command"))

  108 cwdField :: Value -> Maybe Text
  109 cwdField v = case parseMaybe (withObject "in" (.:? "cwd")) v :: Maybe (Maybe Text) of
  110   Just (Just t) -> Just t
  111   _             -> Nothing

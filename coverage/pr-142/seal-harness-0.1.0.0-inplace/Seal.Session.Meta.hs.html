    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The on-disk session metadata record ('session.json'). Holds the session's
    3 -- selected provider label + model id (never a key), its channel of origin, the
    4 -- bound agent definition (if any — set at init from @default_agent@), and
    5 -- timestamps. The 'FromJSON' is tolerant (missing 'channel' defaults to
    6 -- @\"cli\"@, missing 'agent' defaults to 'Nothing') so older/partial files
    7 -- still load.
    8 module Seal.Session.Meta
    9   ( SessionMeta (..)
   10   ) where
   11 
   12 import Data.Aeson
   13   ( FromJSON (..), ToJSON (..), object, withObject, (.:), (.:?), (.!=), (.=) )
   14 import Data.Text (Text)
   15 import Data.Time (UTCTime)
   16 
   17 import Seal.Agent.Def.Types (AgentDefId (..))
   18 import Seal.Core.Types (SessionId)
   19 
   20 data SessionMeta = SessionMeta
   21   { smId         :: SessionId
   22   , smProvider   :: Text      -- ^ Provider label, e.g. @\"anthropic\"@.
   23   , smModel      :: Text      -- ^ Model id, e.g. @\"claude-opus-4-8\"@.
   24   , smChannel    :: Text      -- ^ Channel that created the session, e.g. @\"cli\"@.
   25   , smAgent      :: Maybe AgentDefId
   26     -- ^ The agent definition bound to this session at init (from
   27     -- @default_agent@ in @config.toml@). 'Nothing' means no agent is bound.
   28   , smSystemOverride :: Maybe Text
   29     -- ^ An ad-hoc system prompt supplied via the Session setup screen's
   30     -- \"Use a one-off agent file\" upload. When 'Just', 'plainTurn' prefers
   31     -- this over the bound agent's 'adSystem'. 'Nothing' means no override
   32     -- (the agent's prompt, if any, is used).
   33   , smAgentName :: Maybe Text
   34     -- ^ Display label for the session's active agent (shown in the
   35     -- sidebar / chat header as the @agent@ field of 'SessionInfo').
   36     -- Populated whenever the session has an effective agent — either the
   37     -- bound 'smAgent' (set to the agent def's id) or a one-off uploaded
   38     -- file (set to the file's frontmatter @id@, or the filename when the
   39     -- file has no frontmatter). 'Nothing' when no agent is active.
   40   , smDescription :: Maybe Text
   41     -- ^ A user-set display title for the session (the chat-header pencil).
   42     -- 'Nothing' means no explicit title; the UI then falls back to the
   43     -- auto-summary, first-message snippet, or agent name. Persisted to
   44     -- @session.json@ so it survives restarts and propagates to the
   45     -- sidebar (which reads the live session list, not a client-side
   46     -- override).
   47   , smCreatedAt  :: UTCTime
   48   , smLastActive :: UTCTime
   49   } deriving stock (Eq, Show)
   50 
   51 instance ToJSON SessionMeta where
   52   toJSON m = object
   53     [ "id"          .= smId m
   54     , "provider"    .= smProvider m
   55     , "model"       .= smModel m
   56     , "channel"     .= smChannel m
   57     , "agent"       .= smAgent m
   58     , "system_override" .= smSystemOverride m
   59     , "agent_name"  .= smAgentName m
   60     , "description" .= smDescription m
   61     , "created_at"  .= smCreatedAt m
   62     , "last_active" .= smLastActive m
   63     ]
   64 
   65 instance FromJSON SessionMeta where
   66   parseJSON = withObject "SessionMeta" $ \o -> SessionMeta
   67     <$> o .:  "id"
   68     <*> o .:  "provider"
   69     <*> o .:  "model"
   70     <*> o .:? "channel" .!= "cli"
   71     <*> o .:? "agent"
   72     <*> o .:? "system_override"
   73     <*> o .:? "agent_name"
   74     <*> o .:? "description"
   75     <*> o .:  "created_at"
   76     <*> o .:  "last_active"

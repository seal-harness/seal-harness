    1 -- | The pure partition step: given the current tab list and the on-disk
    2 -- session lists (non-archived + archived), produce three mutually-exclusive
    3 -- lists. The backend's source of truth for the sidebar's three sections.
    4 --
    5 -- Haddock note: Haskell record fields use the @ps@ prefix; the wire keys
    6 -- (in 'Seal.Gateway.ListsSnapshot') drop the prefix
    7 -- (@tabSessions@\/@recentSessions@\/@archivedSessions@).
    8 module Seal.Tabs.Partition
    9   ( PartitionedSessions (..)
   10   , partitionSessions
   11   ) where
   12 
   13 import Data.List (partition)
   14 import Data.Set qualified as Set
   15 
   16 import Seal.Session.Meta (SessionMeta (..))
   17 import Seal.Tabs.Types (Tab (..), TabList (..), TabRef (..))
   18 
   19 -- | The three mutually-exclusive session lists backing the sidebar sections.
   20 data PartitionedSessions = PartitionedSessions
   21   { psTabSessions      :: [SessionMeta]
   22     -- ^ wire key: @tabSessions@ — sessions with an open tab.
   23   , psRecentSessions   :: [SessionMeta]
   24     -- ^ wire key: @recentSessions@ — non-archived, no open tab.
   25   , psArchivedSessions :: [SessionMeta]
   26     -- ^ wire key: @archivedSessions@ — archived, no open tab.
   27   } deriving stock (Eq, Show)
   28 
   29 -- | Partition the session space into three mutually-exclusive lists.
   30 --
   31 -- A session @s@ goes into 'psTabSessions' iff some tab's 'tRef' is
   32 -- @BoundSession (smId s)@; else into 'psRecentSessions' iff not archived;
   33 -- else into 'psArchivedSessions'.
   34 --
   35 -- Harness tabs (@'BoundHarness'@) carry no session and pull nothing into
   36 -- 'psTabSessions', so a harness's backing session (if it exists on disk)
   37 -- stays in 'psRecentSessions' — the intentional harness dual-listing is
   38 -- preserved.
   39 --
   40 -- An archived + tab-bound session goes into 'psTabSessions' (the tab wins;
   41 -- the archive flag stays on disk and resurfaces when the tab closes). Both
   42 -- the recent and archived lists are filtered by @tabSids@, and the
   43 -- tab-bound entries from both are merged into 'psTabSessions'.
   44 partitionSessions :: TabList -> [SessionMeta] -> [SessionMeta] -> PartitionedSessions
   45 partitionSessions tl recent archived =
   46   let tabSids = Set.fromList [ sid | t <- tlTabs tl, BoundSession sid <- [tRef t] ]
   47       (tabbedFromRecent, recent') = partition (`belongsInTabs` tabSids) recent
   48       (tabbedFromArchived, archived') = partition (`belongsInTabs` tabSids) archived
   49   in PartitionedSessions
   50        { psTabSessions = tabbedFromRecent <> tabbedFromArchived
   51        , psRecentSessions = recent'
   52        , psArchivedSessions = archived'
   53        }
   54   where
   55     belongsInTabs s tabSids = smId s `Set.member` tabSids

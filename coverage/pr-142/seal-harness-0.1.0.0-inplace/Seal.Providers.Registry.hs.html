   49 -- | Every provider Seal can build. M1 ships Anthropic only; later milestones
   50 -- extend this sum (the totality of the functions below forces each addition to
   51 -- be handled everywhere).
   52 data KnownProvider = AnthropicProvider | OllamaProvider
   53   deriving stock (Eq, Show, Enum, Bounded)

   55 knownProviders :: [KnownProvider]
   56 knownProviders = [minBound .. maxBound]

   58 -- | The user-facing id typed at the @/provider@ prompt.
   59 providerLabel :: KnownProvider -> Text
   60 providerLabel AnthropicProvider = "anthropic"

   66 -- | Parse a label (case-insensitive) back to a 'KnownProvider'.
   67 parseProvider :: Text -> Maybe KnownProvider
   68 parseProvider t =
   69   let needle = T.toCaseFold (T.strip t)
   70   in lookup needle [(T.toCaseFold (providerLabel p), p) | p <- knownProviders]

   82 -- | A provider's default model: the configured value if given, else the
   83 -- provider's hardcoded default (Anthropic's for an unrecognized label).
   84 resolveDefaultModel :: Maybe Text -> Text -> ModelId
   85 resolveDefaultModel (Just m) _   = ModelId m
   86 resolveDefaultModel Nothing  lbl =
   87   maybe (defaultModelFor AnthropicProvider) defaultModelFor (parseProvider lbl)

   89 -- | Build a live provider. For Anthropic, stored OAuth tokens take precedence
   90 -- over an API key; a present-but-corrupt OAuth blob fails loudly (the user must
   91 -- re-run @\/provider login@) rather than silently falling back. The @callCounter@
   92 -- is shared across turns so Ollama-synthesized tool-call ids stay unique across
   93 -- a whole session (Ollama responses carry no id, so Seal synthesizes
   94 -- @call_<i>@; the counter must outlive any single 'Ollama' value, which is
   95 -- rebuilt each turn — see 'ProviderRuntime.prCallCounter').
   96 resolveProvider
   97   :: Maybe VaultHandle -> Manager -> Text -> KnownProvider -> ModelId
   98   -> IORef Int -> IO (Either Text SomeProvider)
   99 resolveProvider Nothing _ _ AnthropicProvider _ _ =
  100   pure (Left "vault not configured \x2014 run /vault setup")
  101 resolveProvider (Just vh) mgr _baseUrl AnthropicProvider model _ = do
  102   eOAuth <- vhGet vh "ANTHROPIC_OAUTH_TOKENS"
  103   case eOAuth of
  104     Right blob -> case deserializeTokens blob of
  105       Left e     -> pure (Left ("stored Anthropic OAuth tokens are unreadable ("
  106                                   <> e <> ") — run /provider login anthropic"))
  107       Right toks -> do
  108         ref <- newIORef toks
  109         let sess = OAuthSession
  110               { osTokens  = ref
  111               , osRefresh = refreshTokens mgr
  112               , osPersist = void . vhPut vh "ANTHROPIC_OAUTH_TOKENS" . serializeTokens
  113               }
  114         _ <- ensureFresh sess
  115         pure (Right (SomeProvider (mkAnthropicOAuth mgr sess model)))
  116     Left _ -> do
  117       eKey <- vhGet vh (vaultKeyName AnthropicProvider)
  118       pure $ case eKey of
  119         Left e         -> Left (credErr AnthropicProvider e)
  120         Right keyBytes -> Right (SomeProvider (mkAnthropic mgr (mkApiKey keyBytes) model))
  121 resolveProvider mvh mgr baseUrl OllamaProvider model callCounter
  122   | not (ollamaNeedsKey baseUrl) =
  123       -- local/custom host: keyless, never touch the vault
  124       Right . SomeProvider <$> mkOllama mgr baseUrl Nothing model callCounter
  125   | otherwise = case mvh of
  126       Nothing ->
  127         pure (Left "Ollama Cloud needs an API key \x2014 run /vault setup then /provider add ollama")
  128       Just vh -> do
  129         eKey <- vhGet vh (vaultKeyName OllamaProvider)
  130         case eKey of
  131           Right kb -> Right . SomeProvider <$> mkOllama mgr baseUrl (Just (mkApiKey kb)) model callCounter
  132           Left e   -> pure (Left (vaultErrText e))

  142 -- | Provider-aware credential error: a missing key points the user at the
  143 -- exact @/provider add@ they need.
  144 credErr :: KnownProvider -> VaultError -> Text
  145 credErr kp = \case
  146   VaultKeyNotFound _ ->
  147     "no credential for " <> providerLabel kp
  148       <> " — run /provider add " <> providerLabel kp
  149   e -> vaultErrText e
  150 
  151 -- | Human-readable rendering of a vault error.
  152 vaultErrText :: VaultError -> Text
  153 vaultErrText = \case
  154   VaultLocked         -> "vault is locked — run /vault unlock"
  155   VaultNotFound       -> "vault not found — run /vault setup"
  156   VaultAlreadyExists  -> "vault already exists"
  157   VaultKeyNotFound k  -> "no such secret: " <> k
  158   VaultBackendError t -> "backend error: " <> t
  159 
  160 -- | The set of providers that are *configured* (have a usable credential or
  161 -- are usable keyless). A provider appears in @/provider list@ and in the
  162 -- frontend's provider select only when this returns 'True'. The rules mirror
  163 -- the auth labels in 'Seal.Command.Provider.listCmd':
  164 --
  165 --   * __Ollama__ is configured when its base URL needs no key (local daemon)
  166 --     OR an API key / OAuth blob is stored in the vault. A cloud URL
  167 --     (@ollama.com@) without a key is NOT configured.
  168 --   * __Anthropic__ is configured when either an OAuth blob or an API key is
  169 --     stored in the vault.
  170 --
  171 -- A missing vault ('Nothing') means no provider has credentials, so only a
  172 -- keyless local Ollama would survive the filter (and only when its base URL
  173 -- is the default local URL).
  174 configuredProviders :: Maybe VaultHandle -> RuntimeConfig -> IO [KnownProvider]
  175 configuredProviders mvh cfg = filterM isConfigured knownProviders
  176   where
  177     isConfigured kp = case kp of
  178       OllamaProvider -> do
  179         let baseUrl = fromMaybe defaultOllamaBaseUrl (providerBaseUrl cfg "ollama")
  180         if not (ollamaNeedsKey baseUrl)
  181           then pure True
  182           else hasVaultKey mvh (vaultKeyName OllamaProvider)
  183       AnthropicProvider ->
  184         hasVaultKey mvh (vaultKeyName AnthropicProvider)
  185           ||^ hasVaultKey mvh anthropicOAuthKey
  186 
  187 -- | Does the vault hold a value for @k@? 'Nothing' (no vault) or a vault error
  188 -- (locked / key missing) → 'False'; a present key → 'True'.
  189 hasVaultKey :: Maybe VaultHandle -> Text -> IO Bool
  190 hasVaultKey Nothing _ = pure False
  191 hasVaultKey (Just vh) k = do
  192   r <- vhGet vh k
  193   pure $ case r of
  194     Right _ -> True
  195     Left _  -> False
  196 
  197 -- | Short-circuit OR for two 'IO' booleans (so the second is only run when the
  198 -- first is 'False'). Avoids touching OAuth when an API key is already present.
  199 infixr 2 ||^
  200 (||^) :: IO Bool -> IO Bool -> IO Bool
  201 a ||^ b = do
  202   x <- a
  203   if x then pure True else b

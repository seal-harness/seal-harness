    6 -- @UIO OpResult@ won't compile. Constructed ONLY by 'mkLocalUIO'/
    7 -- 'mkRemoteUIO'\/'mkUIOStub'; the constructor + 'unUIO' + 'runUIO' +

   76 -- | The restricted monad. Exposes 'Functor', 'Applicative', 'Monad' — and
   77 -- deliberately **no** 'MonadIO', no 'MonadReader', no 'MonadThrow', no
   78 -- 'Katip'. An opcode body typed @UIO OpResult@ literally cannot call
   79 -- 'liftIO', 'ask', 'throwM', or import 'System.Directory' (compile error).
   80 -- The only IO is the module-level 'uio*' functions below.
   81 newtype UIO a = UIO { unUIO :: ReaderT UIOEnv IO a }
   82   deriving newtype (Functor, Applicative, Monad)
   83   -- deliberately NOT: MonadIO, MonadReader, MonadThrow
   84 
   85 -- | The environment carried by 'UIO'. Holds the 'UntrustedIO' handle + the
   86 -- Git 'CloneDeps' surface. NOT exported (opaque — no caller can forge
   87 -- one); only 'mkLocalUIO'\/'mkRemoteUIO'\/'mkUIOStub'\/'mkTestUIOEnv'
   88 -- construct it.
   89 data UIOEnv = UIOEnv
   90   { uieUntrustedIO :: !UntrustedIO
   91   , uieCloneDeps   :: !CloneDeps
   92   }
   93 
   94 -- | Run a 'UIO' action with a pre-built 'UIOEnv'. Exported to the
   95 -- dispatcher + 'WorkdirFs' (NOT to opcode modules — W6 grep check). The
   96 -- 'UIOEnv' is opaque; callers obtain one from 'mkSessionExec'
   97 -- ('seUIOEnv') or 'mkTestUIOEnv'.
   98 runUIOWithEnv :: UIOEnv -> UIO a -> IO a
   99 runUIOWithEnv env (UIO action) = runReaderT action env
  100 
  101 -- | Build a 'UIOEnv' from an 'UntrustedIO' + 'CloneDeps'. Exported with
  102 -- the 'Test' prefix as the naming gate (matching
  103 -- 'mkFakeRemoteRunnerRecording' — the @Fake@\/@Test@ prefix is the
  104 -- convention; no @#ifdef TESTING@). Used by the 'uoRunLegacy' back-compat
  105 -- wrapper (W-A3) + tests (Git specs seed a stub 'CloneDeps' into the
  106 -- 'UIOEnv' via this helper).
  107 mkTestUIOEnv :: UntrustedIO -> CloneDeps -> UIOEnv
  108 mkTestUIOEnv = UIOEnv

  114 -- | @mode=local@: the workdir is a local path; 'UntrustedIO' is local.
  115 -- Builds a local 'UntrustedIO' + a local 'CloneDeps' internally, runs the
  116 -- action via 'runUIOWithEnv'.
  117 mkLocalUIO :: WorkspaceRoot -> CloneDeps -> UIO a -> IO a
  118 mkLocalUIO wsRoot cloneDeps action =
  119   let uio = mkLocalUntrustedIO wsRoot
  120       env = UIOEnv uio cloneDeps
  121   in runUIOWithEnv env action
  122 
  123 -- | @mode=remote@: the workdir is a remote workspace path; 'UntrustedIO'
  124 -- is remote (SSH via the 'RemoteRunner'). The runner is shared with
  125 -- 'WorkdirFs' (§3.12, via 'runUIOWithEnv').
  126 mkRemoteUIO :: SshConfig -> RemoteRunner -> WorkspaceRoot -> CloneDeps
  127             -> UIO a -> IO a
  128 mkRemoteUIO sshCfg runner _wsRoot cloneDeps action =
  129   let uio = mkRemoteUntrustedIO sshCfg runner
  130       env = UIOEnv uio cloneDeps
  131   in runUIOWithEnv env action
  132 
  133 -- | Fail-closed: every operation returns 'Left'\/stub. For misconfigured
  134 -- or unreachable remotes; mirrors 'mkRemoteUntrustedIOStub'.
  135 mkUIOStub :: CloneDeps -> UIO a -> IO a
  136 mkUIOStub cloneDeps action =
  137   let env = UIOEnv mkRemoteUntrustedIOStub cloneDeps
  138   in runUIOWithEnv env action

  233 -- | Internal: lift an 'IO' action into the 'ReaderT' carrier. This is the
  234 -- ONE place 'liftIO' is used in this module — it is NOT exposed to opcode
  235 -- modules (the 'UIO' newtype hides 'ReaderT' and its 'MonadIO' instance).
  236 -- Opcode bodies call the 'uio*' functions above, never 'liftIO' directly.
  237 liftIO' :: IO a -> ReaderT UIOEnv IO a
  238 liftIO' = liftIO

  240 -- | The escape hatch for untrusted opcodes that perform IO NOT through
  241 -- 'UntrustedIO' (e.g. 'WEB_FETCH' uses an HTTP @Manager@, 'IMAGE_ANALYZE'
  242 -- uses a vision API). These opcodes live outside @src\/Seal\/ISA\/Ops\/@
  243 -- (in @src\/Seal\/Web\/@, @src\/Seal\/Media\/@) — the W-A2 compile-fail
  244 -- fixture asserts that @liftIO@ (from 'MonadIO') does NOT work in a 'UIO'
  245 -- body, and the ISA opcode modules (File\/Shell\/Search\/Bin\/Process\/
  246 -- Git\/Repo) use the typed 'uio*' surface instead. This function is the
  247 -- explicit, audited escape hatch for the Web\/Media opcodes that need
  248 -- direct IO access.
  249 uioLiftIO :: IO a -> UIO a
  250 uioLiftIO = UIO . liftIO'
  251 
  252 -- | Read the 'UntrustedIO' handle from the 'UIOEnv'. Escape hatch for
  253 -- opcodes that delegate to functions taking a raw 'UntrustedIO' (e.g.
  254 -- 'cloneRepoIO' in 'Seal.ISA.Ops.Repo', which is shared between the opcode
  255 -- path and the API endpoint). W6 will fully migrate these to the 'uio*'
  256 -- surface.
  257 uioUntrustedIO :: UIO UntrustedIO
  258 uioUntrustedIO = UIO $ asks uieUntrustedIO

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | System-prompt construction for skills. This module renders the
    3 -- @\<available_skills\>@ catalog — a grouped listing of every skill's id +
    4 -- description — and appends it to the resolved system prompt so the model
    5 -- discovers and uses skills. The catalog is built from 'sbList' (the
    6 -- unioned backend: user ⊕ builtin, and later workdir ⊕ user ⊕ builtin),
    7 -- grouped by 'skGroup', and truncated to a token-safe budget.
    8 --
    9 -- The catalog is /teaching/, not enforcement: it tells the model which
   10 -- skills exist and that it should call @SKILL_LOAD@ to read one. It does
   11 -- not auto-load skill bodies (only the operator-configured @autoload@ id
   12 -- is auto-injected, via 'Seal.Skills.Autoload').
   13 module Seal.Skills.Prompt
   14   ( availableSkillsBlock
   15   , injectAvailableSkills
   16   , availableSkillsBudget
   17   ) where
   18 
   19 import Data.List (groupBy, sortOn)
   20 import Data.Maybe (fromMaybe)
   21 import Data.Text (Text)
   22 import Data.Text qualified as T
   23 
   24 import Seal.Skills.Backend (SkillBackend (..))
   25 import Seal.Skills.Types (Skill (..), skillIdText)
   26 
   27 -- | A soft character budget for the whole @\<available_skills\>@ block.
   28 -- Keeps the catalog token-safe even when a deployment has many skills;
   29 -- once the budget is exhausted, the block is truncated with a count of the
   30 -- remaining skills. Large enough that a typical deployment's full catalog
   31 -- fits intact.
   32 availableSkillsBudget :: Int
   33 availableSkillsBudget = 4096
   34 
   35 -- | Render the @\<available_skills\>@ catalog from a list of skills.
   36 -- Skills are grouped by 'skGroup' (a header per group; ungrouped skills
   37 -- fall under a default \"Skills\" header), each entry is
   38 -- @\- \<id\>: \<description\>@, and the whole block is wrapped in
   39 -- @\<available_skills\>@ \/ @\<\/available_skills\>@ tags. Returns the
   40 -- empty 'Text' when the list is empty (the caller skips injection in that
   41 -- case so no empty tags are emitted). The block is truncated to
   42 -- 'availableSkillsBudget' characters with a tail count when exceeded.
   43 --
   44 -- The block ends with a one-line note nudging the model to load a skill's
   45 -- full body via @SKILL_LOAD@ before relying on it.
   46 availableSkillsBlock :: [Skill] -> Text
   47 availableSkillsBlock [] = ""
   48 availableSkillsBlock skills =
   49   truncateBlock availableSkillsBudget (fullBlock skills)
   50 
   51 -- | The untruncated catalog block.
   52 fullBlock :: [Skill] -> Text
   53 fullBlock skills =
   54   "<available_skills>\n"
   55   <> T.intercalate "\n\n" (map renderGroup grouped)
   56   <> "\n\nCall SKILL_LOAD with an id to read a skill's full body before relying on it."
   57   <> "\n</available_skills>"
   58   where
   59     -- Sort by (group, id) then group by group. Ungrouped (Nothing) sorts
   60     -- first under a generic header; grouped skills follow in alphabetical
   61     -- group order.
   62     sorted = sortOn (\s -> (groupKey s, skillIdText (skId s))) skills
   63     grouped = groupBy (\a b -> groupKey a == groupKey b) sorted
   64     groupKey s = fromMaybe "" (skGroup s)
   65 
   66 -- | Render one group's section: a header line (the group name, or
   67 -- \"Skills\" for the ungrouped section) followed by the @\- id: desc@
   68 -- lines for each skill in the group.
   69 renderGroup :: [Skill] -> Text
   70 renderGroup [] = ""
   71 renderGroup group@(s:_) =
   72   header <> "\n" <> T.intercalate "\n" [ "- " <> skillIdText (skId x) <> ": " <> desc x | x <- group ]
   73   where
   74     header = case skGroup s of
   75       Just g  -> "## " <> g
   76       Nothing -> "## Skills"
   77     desc x = let d = skDescription x in if T.null d then "(no description)" else d
   78 
   79 -- | Truncate @block@ to @budget@ characters, appending a tail notice with
   80 -- the number of characters elided when truncation occurs. Strings at or
   81 -- under the budget are returned as-is. Truncation happens on a character
   82 -- boundary (Haskell 'Text' is Unicode-correct).
   83 truncateBlock :: Int -> Text -> Text
   84 truncateBlock budget block
   85   | T.length block <= budget = block
   86   | otherwise =
   87       T.take budget block
   88         <> "\n[...catalog truncated at "
   89         <> T.pack (show budget)
   90         <> " chars; "
   91         <> T.pack (show (T.length block - budget))
   92         <> " more chars elided...]"
   93 
   94 -- | Append the @\<available_skills\>@ catalog to the resolved system
   95 -- prompt. Calls 'sbList' on the backend, renders the block, and appends it
   96 -- after any existing prompt. Returns the prompt unchanged (no catalog)
   97 -- when the backend lists no skills, so no empty tags are ever emitted.
   98 --
   99 -- The catalog is appended /after/ the auto-loaded skill body (the caller
  100 -- runs 'Seal.Skills.Autoload.injectAutoloadSkill' first, then this). This
  101 -- ordering keeps the stable identity prefix first, the auto-loaded
  102 -- orientation next, and the dynamic catalog last — so prefix-cache hits
  103 -- are maximized and the most volatile content is at the tail.
  104 injectAvailableSkills
  105   :: SkillBackend
  106   -- ^ The skill store (typically the unioned backend).
  107   -> Maybe Text
  108   -- ^ The resolved system prompt so far (base + autoload body, or
  109   -- 'Nothing' when no agent is bound and no autoload skill injected).
  110   -> IO (Maybe Text)
  111 injectAvailableSkills backend mPrompt = do
  112   skills <- sbList backend
  113   let block = availableSkillsBlock skills
  114   if T.null block
  115     then pure mPrompt
  116     else pure (Just (case mPrompt of
  117                         Nothing  -> block
  118                         Just base -> base <> "\n\n" <> block))

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Skill auto-injection at session start. The harness reads a configured
    3 -- skill id (default @"seal-usage"@, the fresh-workdir contract), loads its
    4 -- body from the skill backend, and appends it to the resolved system prompt
    5 -- so the model is oriented to its per-session workspace from turn one.
    6 --
    7 -- The injected skill is /teaching/, not enforcement: it tells the model the
    8 -- workdir contract so it cooperates. The operator can disable auto-injection
    9 -- by setting @[skills] autoload = ""@ in @config.toml@, or override the
   10 -- injected skill id with any other value.
   11 module Seal.Skills.Autoload
   12   ( injectAutoloadSkill
   13   , renderSkillForPrompt
   14   ) where
   15 
   16 import Data.Text (Text)
   17 
   18 import Seal.Skills.Backend (SkillBackend (..))
   19 import Seal.Skills.Types (Skill (..), mkSkillId, skillIdText)
   20 
   21 -- | Append the auto-loaded skill body to the resolved system prompt.
   22 --
   23 -- * @mSkillId = Nothing@ → no auto-injection; return the prompt unchanged.
   24 -- * @mSkillId = Just sid@ → load the skill body from @backend@; if found,
   25 --   append it under an @## Auto-loaded skill: <id>@ header. If the skill is
   26 --   missing from the backend, fall back to the prompt unchanged (a missing
   27 --   autoload skill is a soft failure — the session proceeds, the model
   28 --   just doesn't get the orientation note).
   29 --
   30 -- The skill body is appended /after/ any existing system prompt (the bound
   31 -- agent's @adSystem@). The header separates the agent's identity/prompt
   32 -- from the harness-injected orientation.
   33 injectAutoloadSkill
   34   :: SkillBackend
   35   -- ^ The skill store (reads from @~/.seal/config/skills@).
   36   -> Maybe Text
   37   -- ^ The skill id to auto-inject ('Nothing' disables).
   38   -> Maybe Text
   39   -- ^ The resolved system prompt (the bound agent's @adSystem@, or
   40   -- 'Nothing' when no agent is bound).
   41   -> IO (Maybe Text)
   42 injectAutoloadSkill _ Nothing prompt = pure prompt
   43 injectAutoloadSkill backend (Just skillIdText_) prompt = do
   44   case mkSkillId skillIdText_ of
   45     Left _err -> pure prompt  -- invalid id → soft-fail (no injection)
   46     Right sid -> do
   47       mSkill <- sbRead backend sid
   48       case mSkill of
   49         Nothing    -> pure prompt  -- skill missing → soft-fail
   50         Just skill -> pure (Just (renderSkillForPrompt prompt skill))
   51 
   52 -- | Render the system prompt + auto-loaded skill body as a single text.
   53 -- The skill body is appended under a header; if the existing prompt is
   54 -- 'Nothing', the skill body becomes the entire prompt.
   55 renderSkillForPrompt :: Maybe Text -> Skill -> Text
   56 renderSkillForPrompt mPrompt skill =
   57   case mPrompt of
   58     Nothing    -> rendered
   59     Just base  -> base <> "\n\n" <> rendered
   60   where
   61     rendered =
   62       "# Auto-loaded skill: " <> skillIdText (skId skill) <> "\n\n"
   63       <> skBody skill

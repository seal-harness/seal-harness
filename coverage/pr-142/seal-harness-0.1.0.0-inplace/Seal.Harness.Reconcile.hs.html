    2 -- | The background reconcile loop: a server sweep ('readMarkers' + a screen
    3 -- capture) per session, classified by a per-flavour observer into
    4 -- 'Liveness', merged into the registry via 'mergeReconcile', then the
    5 -- 'defaultOrphanGraceTicks' wall-clock-free grace policy auto-evicts an
    6 -- entry after N consecutive Orphaned ticks (never touches @session.json@).
    7 module Seal.Harness.Reconcile
    8   ( reconcileTick
    9   , tickOrphans
   10   , defaultOrphanGraceTicks
   11   ) where

   28 -- | The default wall-clock-free grace policy: auto-evict an entry after
   29 -- this many consecutive Orphaned ticks. Never touches @session.json@.
   30 defaultOrphanGraceTicks :: Int
   31 defaultOrphanGraceTicks = 3
   32 
   33 -- | One reconcile tick: read markers for the session, classify via the
   34 -- observer, merge into the registry, then tick orphans and evict those
   35 -- over the grace limit. Returns the post-tick snapshot.
   36 --
   37 -- The observed list is built from the @seal_id@ marker (which carries the
   38 -- 'HarnessId') + the classified liveness. In 6a's single-harness-per-session
   39 -- model there's at most one observed harness per tick.
   40 reconcileTick
   41   :: HarnessRegistry -> TmuxRunner -> TmuxIdent -> HarnessFlavour
   42   -> Int  -- ^ grace ticks (defaultOrphanGraceTicks)
   43   -> IO [HarnessEntry]
   44 reconcileTick reg runner session flavour graceTicks = do
   45   markersR <- readMarkers runner session
   46   captureR <- captureWindowNamed runner session
   47   observed <- case (markersR, captureR) of
   48     (Right markers, Right screenLines) ->
   49       let screen = T.intercalate "\n" screenLines
   50       in pure (buildObserved flavour screen markers)
   51     _ -> pure []
   52   _ <- atomically (mergeReconcile reg observed)
   53   let observedIds = Set.fromList (map ohId observed)
   54   _evicted <- atomically (tickOrphans reg observedIds graceTicks)
   55   snapshot reg
   56 
   57 -- | Build the observed-harness list from the marker map + the classified
   58 -- liveness. The @seal_id@ marker carries the 'HarnessId'; the coord is the
   59 -- session ident. Pure.
   60 buildObserved :: HarnessFlavour -> Text -> Map.Map Text Text -> [ObservedHarness]
   61 buildObserved flavour screen markers =
   62   case Map.lookup "seal_id" markers of
   63     Nothing -> []  -- no seal_id => nothing to observe (the observer's
   64                    -- LvOrphaned/LvExited classification is for entries that
   65                    -- ARE in the registry but lost their marker; that's
   66                    -- handled by the unobserved-tick path)
   67     Just sealIdText -> case parseHarnessId sealIdText of
   68       Right hid ->
   69         let liveness = classify flavour screen markers
   70         in [ ObservedHarness
   71                { ohId = hid
   72                , ohLiveness = liveness
   73                , ohTmuxCoord = Nothing
   74                , ohFlavour = Just (flavourText flavour)
   75                } ]
   76       Left _ -> []
   77 
   78 -- | Increment orphan-ticks for entries NOT in the observed set, evict those
   79 -- over the limit. STM. Returns the evicted entries.
   80 tickOrphans :: HarnessRegistry -> Set HarnessId -> Int -> STM [HarnessEntry]
   81 tickOrphans (HarnessRegistry tv) observedIds graceTicks = do
   82   m <- readTVar tv
   83   let (m', evicted) = foldr step (m, []) (Map.toList m)
   84   writeTVar tv m'
   85   pure evicted
   86   where
   87     step (hid, entry) (m', ev)
   88       | hid `Set.member` observedIds = (m', ev)  -- observed; untouched here
   89       | otherwise =
   90           let ticks = heOrphanTicks entry + 1
   91               entry' = entry { heLiveness = LvOrphaned, heOrphanTicks = ticks }
   92           in if ticks > graceTicks
   93                then (Map.delete hid m', entry' : ev)  -- evict
   94                else (Map.insert hid entry' m', ev)
   95 
   96 classify :: HarnessFlavour -> Text -> Map.Map Text Text -> Liveness
   97 classify HfClaudeCode screen markers = observeClaudeCode screen markers
   98 classify _ _ _ = LvIdle
   99 
  100 flavourText :: HarnessFlavour -> Text
  101 flavourText HfClaudeCode = "claude-code"
  102 flavourText HfCodex      = "codex"
  103 flavourText HfGeneric    = "generic"

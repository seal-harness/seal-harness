   75 -- | Extract the @path@ string field from a JSON object.
   76 pathField :: Value -> Maybe Text
   77 pathField = parseMaybe (withObject "in" (.: "path"))

   79 -- | Leniently parse an optional integer field that the model may emit as a
   80 -- JSON number OR as a numeric string (e.g. @"72"@). Returns 'Nothing' when
   81 -- the field is missing, malformed, non-numeric, or negative — the caller
   82 -- then applies its own default. The pager's clamp is the second line of
   83 -- defense. A strictly-typed @.:?@ parse would silently drop a numeric
   84 -- string to the default, trapping the model in a re-read loop (see session
   85 -- 20260724-133418-292: @offset="72"@ returned the first window again).
   86 intField :: Text -> Value -> Maybe Int
   87 intField key v =
   88   case parseMaybe (withObject "in" (.:? fromText key)) v :: Maybe (Maybe Int) of
   89     Just (Just n) | n >= 0 -> Just n
   90     Just{}                 -> Nothing      -- negative number, or absent
   91     Nothing                ->               -- present but not a JSON number; try numeric string
   92       parseMaybe (withObject "in" (.: fromText key)) v >>= textToInt
   93 
   94 -- | Parse a 'Text' as a non-negative decimal integer. Rejects negatives,
   95 -- leading '+', whitespace, and trailing garbage (the model's value must be
   96 -- a clean integer).
   97 textToInt :: Text -> Maybe Int
   98 textToInt t =
   99   case TR.decimal t of
  100     Right (n, rest) | T.null rest && n >= 0 -> Just n
  101     _                                      -> Nothing

  106 offsetField :: Value -> Int
  107 offsetField v = fromMaybe 0 (intField "offset" v)

  112 limitField :: Value -> Maybe Int

  115 -- | Leniently parse the optional @max_scan_bytes@ integer field. Missing,
  116 -- malformed, or non-positive values fall back to the operator-configured
  117 -- ceiling; the clamp in 'opRun' is the second line of defense. Returns
  118 -- 'Nothing' when the model did not supply a usable value.
  119 scanBytesField :: Value -> Maybe Int
  120 scanBytesField v = case intField "max_scan_bytes" v of
  121   Just n | n >= 1 -> Just n
  122   _               -> Nothing

  124 -- | FILE_READ opcode: reads a UTF-8 text file at a workspace-relative path,
  125 -- confined by 'mkSafePath', returning a bounded window of lines. Trust level
  126 -- is 'Untrusted'; all IO is funnelled through the 'UntrustedIO' capability
  127 -- handle.
  128 --
  129 -- The @operatorCeiling@ argument is the operator-configured upper bound on
  130 -- bytes scanned per call (resolved from the @[retrieval]@ config section by
  131 -- the wiring site). The model's per-call @max_scan_bytes@ request is clamped
  132 -- to @[1, operatorCeiling]@ — it can narrow the scan but can never widen the
  133 -- memory bound above what the operator configured.
  134 --
  135 -- 'orRecorded' captures the requested path / offset / limit / resolved
  136 -- max_scan_bytes (secret-free metadata); file contents flow only to 'orParts'
  137 -- (model-visible).
  138 fileReadOp :: WorkspaceRoot -> Int -> Opcode
  139 fileReadOp _root operatorCeiling = UntrustedOpcode
  140   { uoName = OpName "FILE_READ"
  141   , uoDesc = "Read a UTF-8 text file from the workspace (path is workspace-relative)."
  142   , uoInSchema = fileReadSchema
  143   , uoOutSchema = object []
  144   , uoAuthorize =
  145       maybe (Left "FILE_READ requires {path:string}") (const (Right ())) . pathField
  146   , uoRun = \v -> do
  147       let rel       = maybe "" T.unpack (pathField v)
  148           offset    = offsetField v
  149           mLimit    = limitField v
  150           scanBytes = clampScanBytes operatorCeiling (scanBytesField v)
  151       let recorded = object
  152             [ "path" .= rel
  153             , "offset" .= offset
  154             , "limit" .= mLimit
  155             , "max_scan_bytes" .= scanBytes
  156             ]
  157       case mkRemotePath (T.pack rel) of
  158         Left _ ->
  159           -- An empty/invalid path: surface as an error (the authorize gate
  160           -- already rejects a missing path, but a malformed one falls here).
  161           pure $ OpResult
  162             [TrpText "FILE_READ: invalid path"]
  163             True
  164             recorded
  165         Right rp -> do
  166           res <- uioRead rp scanBytes
  167           case res of
  168             Left err ->
  169               pure $ OpResult
  170                 [TrpText (renderUntrustedErr err)]
  171                 True
  172                 recorded
  173             Right fullWin -> do
  174               -- Character ceiling: if the file content exceeds maxReadChars
  175               -- and the caller did not supply an explicit offset or limit,
  176               -- reject outright (the model must paginate). When the caller
  177               -- IS paginating, return the window normally. This mirrors
  178               -- Hermes' read_file behavior.
  179               let contentChars = sum (map T.length (lwLines fullWin))
  180                   isPaginating = offset /= 0 || case mLimit of Just _ -> True; Nothing -> False
  181               if contentChars > maxReadChars && not isPaginating
  182                 then pure $ OpResult
  183                   [TrpText (T.pack (charCeilingMsg contentChars maxReadChars))]
  184                   True
  185                   recorded
  186                 else do
  187                   -- Apply the offset/limit windowing purely (the capability
  188                   -- returned a dynamic-page-sized window; the opcode re-windows
  189                   -- to the requested offset/limit). Preserve the total +
  190                   -- truncated flags from the capability's bounded read (so the
  191                   -- footer reflects the true file size + byte ceiling).
  192                   let win = windowLines defaultPageParams offset mLimit (lwLines fullWin)
  193                       win' = win { lwTotal     = lwTotal fullWin
  194                                  , lwHasMore   = lwEnd win < lwTotal fullWin
  195                                               || lwTruncated fullWin
  196                                  , lwTruncated = lwTruncated fullWin
  197                                  }
  198                   pure $ OpResult
  199                     [TrpText (renderWindow win')]
  200                     False
  201                     recorded
  202   }

  204 -- | Error message for the character-ceiling rejection. Tells the model the
  205 -- file is too large for a single read and it must use offset/limit to chunk.
  206 charCeilingMsg :: Int -> Int -> String
  207 charCeilingMsg contentChars ceiling' =
  208   "FILE_READ: file content (" <> show contentChars <> " chars) exceeds the "
  209     <> show ceiling' <> "-char ceiling; use offset/limit to read it in chunks"
  210 
  211 -- | Resolve the per-call scan byte ceiling. The model's request (if any) is
  212 -- clamped to @[1, operatorCeiling]@; missing/malformed requests fall back to
  213 -- the operator ceiling. The operator's configured bound is a hard upper
  214 -- bound — the model can only narrow the scan, never widen it.
  215 clampScanBytes :: Int -> Maybe Int -> Int
  216 clampScanBytes operatorCeiling mReq =
  217   case mReq of
  218     Nothing  -> operatorCeiling
  219     Just req -> max 1 (min operatorCeiling req)

  225 -- | FILE_WRITE opcode: write or append content to a workspace-relative file,
  226 -- confined by 'mkSafePath'. The @operatorWriteCeiling@ is the hard upper
  227 -- bound on bytes written per call; content exceeding it is rejected
  228 -- (bounded write, mirrors FILE_READ's @max_scan_bytes@ clamp). The
  229 -- @mode@ field is @\"write\"@ (default, truncate + create) or @\"append\"@.
  230 -- 'orRecorded' captures the path + mode + byte count (NOT the content —
  231 -- content may be large; the transcript records metadata only).
  232 fileWriteOp :: WorkspaceRoot -> Int -> Opcode
  233 fileWriteOp _root operatorWriteCeiling = UntrustedOpcode
  234   { uoName = OpName "FILE_WRITE"
  235   , uoDesc = "Write or append to a workspace file (path is workspace-relative, bounded)."
  236   , uoInSchema = fileWriteSchema
  237   , uoOutSchema = object []
  238   , uoAuthorize =
  239       maybe (Left "FILE_WRITE requires {path:string, content:string}") (const (Right ()))
  240         . pathField
  241   , uoRun = \v -> do
  242       let rel     = maybe "" T.unpack (pathField v)
  243           content = fromMaybe "" (contentField v)
  244           mode    = case modeField v of
  245             Just m | m == "append" -> WMAppend
  246             _                     -> WMWrite
  247           byteCount = BS.length (TE.encodeUtf8 content)  -- capability re-measures; this is the recorded estimate
  248           recorded = object
  249             [ "path" .= rel
  250             , "mode" .= (case mode of WMAppend -> ("append" :: Text); WMWrite -> "write")
  251             , "bytes" .= byteCount
  252             ]
  253       if byteCount > operatorWriteCeiling
  254         then pure $ OpResult
  255           [TrpText ("FILE_WRITE: content (" <> T.pack (show byteCount) <> " bytes) exceeds operator ceiling ("
  256                     <> T.pack (show operatorWriteCeiling) <> " bytes)")]
  257           True
  258           recorded
  259         else case mkRemotePath (T.pack rel) of
  260           Left _ ->
  261             pure $ OpResult
  262               [TrpText "FILE_WRITE: invalid path"]
  263               True
  264               recorded
  265           Right rp -> do
  266             res <- uioWrite rp content mode operatorWriteCeiling
  267             pure $ case res of
  268               Left err ->
  269                 OpResult
  270                   [TrpText (renderUntrustedErr err)]
  271                   True
  272                   recorded
  273               Right n ->
  274                 OpResult
  275                   [TrpText ("wrote " <> T.pack (show n) <> " bytes to " <> T.pack rel)]
  276                   False
  277                   recorded
  278   }

  313 -- | FILE_PATCH opcode: apply a unified diff to a workspace-relative file,
  314 -- confined by 'SafePath'. The patch is applied via a pure Haskell diff-
  315 -- apply (no subprocess — the patch is parsed and applied in-process; an
  316 -- atomic temp+rename write lands the result). 'orRecorded' captures the
  317 -- path + patch line count + applied flag (NOT the patch body — it may be
  318 -- large; the transcript records metadata only).
  319 filePatchOp :: WorkspaceRoot -> Opcode
  320 filePatchOp _root = UntrustedOpcode
  321   { uoName = OpName "FILE_PATCH"
  322   , uoDesc = "Apply a unified diff to a workspace file (SafePath-confined, atomic write). \
  323              \Input fields are {path: string, patch: string} — the patch field is named \
  324              \'patch' (not 'diff'). The patch is a standard unified diff as @git diff@ \
  325              \would emit; both the long hunk header form (@@@ -1,2 +1,2 @@@) and the \
  326              \short form (@@@ -1 +1 @@@, length 1 implied) are accepted. No shell is \
  327              \invoked — the diff is parsed and applied in-process."
  328   , uoInSchema = filePatchSchema
  329   , uoOutSchema = object []
  330   , uoAuthorize = \v ->
  331       case (pathField v, patchField v) of
  332         (Nothing, _)      -> Left "FILE_PATCH requires {path:string, patch:string}"
  333         (_, Nothing)      -> Left "FILE_PATCH requires {path:string, patch:string}"
  334         (_, Just p)
  335           | T.null p      -> Left "FILE_PATCH requires a non-empty patch"
  336           | otherwise     -> Right ()
  337   , uoRun = \v -> do
  338       let rel   = maybe "" T.unpack (pathField v)
  339           patch = fromMaybe "" (patchField v)
  340           recorded = object [ "path" .= rel, "patch_lines" .= length (T.lines patch) ]
  341       case mkRemotePath (T.pack rel) of
  342         Left _ -> pure $ OpResult [TrpText "FILE_PATCH: invalid path"] True recorded
  343         Right rp -> do
  344           -- Defense-in-depth: a non-empty patch must reach the applier. The
  345           -- authorize gate already rejects an empty/missing 'patch', but if
  346           -- the opcode is ever invoked with the gate bypassed, refuse to
  347           -- silently no-op here rather than rewrite the file with its own
  348           -- contents and report success.
  349           if T.null patch
  350             then pure $ OpResult
  351                    [TrpText "FILE_PATCH requires a non-empty patch"] True recorded
  352             else do
  353               res <- uioPatch rp patch
  354               pure $ case res of
  355                 Left err ->
  356                   OpResult [TrpText ("FILE_PATCH: " <> renderUntrustedErr err)] True recorded
  357                 Right _  ->
  358                   OpResult [TrpText ("patched " <> T.pack rel)] False recorded
  359   }

  380 patchField :: Value -> Maybe Text
  381 -- | Accept the canonical 'patch' field, falling back to 'diff' (a name the
  382 -- model frequently guesses first). Both map to the same unified-diff payload;
  383 -- preferring 'patch' preserves the schema contract, while the 'diff' alias
  384 -- avoids a round-trip through OPCODE_DESCRIBE just to learn the field name.
  385 patchField v = parseMaybe (withObject "in" (.: "patch")) v
  386            <|> parseMaybe (withObject "in" (.: "diff")) v

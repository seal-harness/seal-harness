    2 -- | The @entries.jsonl@ model: one line per event, 1:1 with the old
    3 -- 'TranscriptEntry' stream but payload-free. The giant 'tePayload' is replaced
    4 -- by a @convLen@ prefix pointer (into @conversation.jsonl@) plus, for 'Request'
    5 -- events, an envelope-delta recorded only when changed from the prior request
    6 -- (omit-if-unchanged). 'Response' events store all 'CompletionResponse' fields
    7 -- /except/ the content blocks (those live in @conversation.jsonl@ and are
    8 -- re-inserted on reconstruction). 'Harness' and 'compaction' events keep their
    9 -- metadata columns and a @convLen@.
   10 --
   11 -- Integrity, like the conversation file, rests on the append-only single-writer
   12 -- + fsync — not a hash chain.

   42 -- | Event discriminator. Mirrors the old 'Direction' plus the two call-less
   43 -- kinds ('EKHarness', 'EKCompaction').
   44 data EntryKind
   45   = EKRequest
   46   | EKResponse
   47   | EKHarness
   48   | EKCompaction
   49   deriving stock (Eq, Show, Generic)
   50 
   51 instance ToJSON EntryKind where
   52   toJSON EKRequest    = "request"
   53   toJSON EKResponse   = "response"
   54   toJSON EKHarness    = "harness"
   55   toJSON EKCompaction = "compaction"
   56 
   57 instance FromJSON EntryKind where
   58   parseJSON = withText "EntryKind" $ \t ->
   59     case t of
   60       "request"    -> pure EKRequest
   61       "response"   -> pure EKResponse
   62       "harness"    -> pure EKHarness
   63       "compaction" -> pure EKCompaction
   64       _            -> fail ("unknown EntryKind: " <> T.unpack t)

   66 -- | The full envelope in effect at a given turn. Folded from deltas.
   67 data Envelope = Envelope
   68   { envModel      :: ModelId
   69   , envSystem     :: Maybe Text
   70   , envTools      :: [ToolDefinition]
   71   , envToolChoice :: ToolChoice
   72   , envMaxTokens  :: Int
   73   } deriving stock (Eq, Show)
   74 
   75 -- | A per-request change to the envelope. Each field is 'Maybe'; 'Nothing'
   76 -- means \"inherit from the prior request\" (omit-if-unchanged). For
   77 -- 'envSystem' the 'Maybe' is nested: 'Nothing' = inherit, @'Just' 'Nothing'@ =
   78 -- clear the system prompt, @'Just' ('Just' t)@ = set it to @t@.
   79 data EnvelopeDelta = EnvelopeDelta
   80   { edModel      :: Maybe ModelId
   81   , edSystem     :: Maybe (Maybe Text)
   82   , edTools      :: Maybe [ToolDefinition]
   83   , edToolChoice :: Maybe ToolChoice
   84   , edMaxTokens  :: Maybe Int
   85   } deriving stock (Eq, Show)
   86 
   87 -- | The neutral delta: inherit everything. The writer compares the current
   88 -- request's envelope against the prior effective envelope and emits a delta
   89 -- with 'Just' only on the fields that differ.
   90 emptyEnvelopeDelta :: EnvelopeDelta
   91 emptyEnvelopeDelta = EnvelopeDelta Nothing Nothing Nothing Nothing Nothing
   92 
   93 instance ToJSON EnvelopeDelta where
   94   toJSON d = object (catMaybes'
   95     [ ("model" .=)      <$> edModel d
   96     , ("tools" .=)      <$> edTools d
   97     , ("toolChoice" .=) <$> edToolChoice d
   98     , ("maxTokens" .=)  <$> edMaxTokens d
   99     ] <> systemField)
  100     where
  101       -- The nested Maybe on system distinguishes inherit (omit the key) from
  102       -- clear ("system": null) from set ("system": "text"). Omitting the key
  103       -- is the default aeson behavior for the outer Nothing; the inner cases
  104       -- are encoded explicitly so a clear is distinguishable from an inherit.
  105       systemField = case edSystem d of
  106         Nothing       -> []
  107         Just Nothing  -> ["system" .= A.Null]
  108         Just (Just t) -> ["system" .= t]
  109 
  110 instance FromJSON EnvelopeDelta where
  111   parseJSON = withObject "EnvelopeDelta" $ \o -> do
  112     -- The nested Maybe on system distinguishes inherit (key absent) from clear
  113     -- (key present, value null) from set (key present, value string). aeson's
  114     -- '.:?' conflates absent and null, so we use 'KeyMap.lookup' to detect
  115     -- presence, then decode the value when present.
  116     let mSystem = case KeyMap.lookup (Key.fromString "system") o of
  117           Nothing -> Nothing                         -- inherit
  118           Just v  -> Just (case v of
  119             A.Null     -> Nothing                     -- clear
  120             A.String t -> Just t                      -- set
  121             _          -> Nothing)                    -- malformed: treat as clear
  122     EnvelopeDelta
  123       <$> o .:? "model"
  124       <*> pure mSystem
  125       <*> o .:? "tools"
  126       <*> o .:? "toolChoice"
  127       <*> o .:? "maxTokens"
  128 
  129 -- | Apply a delta to an envelope, producing the next effective envelope.
  130 -- 'Nothing' fields inherit; the nested 'Maybe' on system distinguishes
  131 -- inherit from clear.
  132 applyDelta :: Envelope -> EnvelopeDelta -> Envelope
  133 applyDelta env d = Envelope
  134   { envModel      = fromMaybe (envModel env) (edModel d)
  135   , envSystem     = case edSystem d of
  136       Nothing     -> envSystem env
  137       Just system -> system
  138   , envTools      = fromMaybe (envTools env) (edTools d)
  139   , envToolChoice = fromMaybe (envToolChoice env) (edToolChoice d)
  140   , envMaxTokens  = fromMaybe (envMaxTokens env) (edMaxTokens d)
  141   }
  142 
  143 -- | Left-fold over the entry stream, returning the effective envelope at each
  144 -- 'EKRequest' entry. Non-request entries inherit the prior envelope (so a
  145 -- 'Response' knows the envelope in effect for its paired request). The fold
  146 -- starts from the given initial envelope (typically the first request's full
  147 -- envelope, since there is no prior to delta against).
  148 --
  149 -- Returns the pairs @(entry, effective-envelope-at-entry)@ in input order.
  150 effectiveEnvelope :: Envelope -> [EntryRecord] -> [(EntryRecord, Envelope)]
  151 effectiveEnvelope = go
  152   where
  153     go _ [] = []
  154     go env (e : es) =
  155       let env' = case erKind e of
  156             EKRequest -> case erEnvelope e of
  157               Nothing -> env
  158               Just d  -> applyDelta env d
  159             _ -> env
  160       in (e, env') : go env' es
  161 
  162 -- | One event line. Fields are kind-specific; the 'Maybe' columns are present
  163 -- only for the kinds that use them (e.g. 'erEnvelope' is 'Just' only for
  164 -- 'EKRequest', 'erUsage' only for 'EKResponse'). The on-disk encoding omits
  165 -- 'Nothing' fields (aeson's @omitIfNothing@ would do this; we encode manually to
  166 -- keep the shape explicit and stable for reconstruction).
  167 data EntryRecord = EntryRecord
  168   { erId          :: Text
  169   , erTimestamp   :: UTCTime
  170   , erKind        :: EntryKind
  171   , erConvLen     :: Int
  172   -- | 'EKRequest' only: the envelope delta vs the prior request. 'Nothing' on
  173   -- non-request kinds.
  174   , erEnvelope    :: Maybe EnvelopeDelta
  175   -- | 'EKResponse' only: token usage.
  176   , erUsage       :: Maybe Usage
  177   -- | 'EKResponse' only: stop reason.
  178   , erStop        :: Maybe StopReason
  179   -- | 'EKResponse' / 'EKHarness' only: wall-clock duration in ms.
  180   , erDurationMs  :: Maybe Int
  181   -- | 'EKHarness' only: the harness name.
  182   , erHarness     :: Maybe Text
  183   -- | Common: correlation id linking a request to its response.
  184   , erCorrelation :: Maybe Text
  185   -- | Common: extensible metadata (e.g. @op@ for opcode invocations).
  186   , erMeta        :: Map Text A.Value
  187   } deriving stock (Eq, Show)
  188 
  189 instance ToJSON EntryRecord where
  190   toJSON e = object $
  191     [ "id"          .= erId e
  192     , "ts"          .= erTimestamp e
  193     , "kind"        .= erKind e
  194     , "convLen"     .= erConvLen e
  195     ] <> envelopeField <> responseFields <> harnessField <> commonFields
  196     where
  197       envelopeField   = maybe [] (\d -> ["envelope" .= d]) (erEnvelope e)
  198       responseFields  = case erKind e of
  199         EKResponse -> catMaybes'
  200           [ ("usage" .=)      <$> erUsage e
  201           , ("stop" .=)       <$> erStop e
  202           , ("durationMs" .=) <$> erDurationMs e
  203           ]
  204         _ -> []
  205       harnessField    = maybe [] (\h -> ["harness" .= h]) (erHarness e)
  206       commonFields    =
  207         catMaybes'
  208           [ ("correlation" .=) <$> erCorrelation e
  209           ] <> ["meta" .= erMeta e | not (null (erMeta e))]
  210 
  211 instance FromJSON EntryRecord where
  212   parseJSON = withObject "EntryRecord" $ \o -> do
  213     kind <- o .: "kind"
  214     EntryRecord
  215       <$> o .:  "id"
  216       <*> o .:  "ts"
  217       <*> pure kind
  218       <*> o .:  "convLen"
  219       <*> o .:? "envelope"
  220       <*> o .:? "usage"
  221       <*> o .:? "stop"
  222       <*> o .:? "durationMs"
  223       <*> o .:? "harness"
  224       <*> o .:? "correlation"
  225       <*> o .:? "meta" .!= mempty

  227 -- | Canonical strict encoding of one entry line, no trailing newline. The

  232 -- | A small 'catMaybes' for key-value pairs that keeps the code warning-clean
  233 -- under @-Wincomplete-record-updates@.
  234 catMaybes' :: [Maybe (a, b)] -> [(a, b)]
  235 catMaybes' = foldr (\mx acc -> maybe acc (: acc) mx) []

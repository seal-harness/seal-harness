    3 -- | Built-in skills shipped with the Seal Harness binary. The markdown
    4 -- sources live under @config/skills/\<group\>/@ in the repo and are embedded
    5 -- into the binary at compile time via 'file-embed' so the harness is
    6 -- self-contained: no first-run materialization, no install-layout
    7 -- dependency, no staleness across upgrades (the embedded copy is always the
    8 -- version the binary shipped with).
    9 --
   10 -- A user can override any built-in by dropping a same-id markdown file at
   11 -- @~/.seal/config/skills/<group>/<id>.md@ (or the flat
   12 -- @~/.seal/config/skills/<id>.md@); the union backend
   13 -- ('Seal.Skills.Backend.unionSkillBackend') checks the user layer first
   14 -- and falls back to the embedded built-in. To bring in upstream changes
   15 -- after an upgrade, the user diffs their override against the new
   16 -- built-in (visible via @SKILL_LIST@ or @\/skill list@ — both surface
   17 -- built-ins) and does the desired merge manually.
   18 module Seal.Skills.Builtins
   19   ( builtinSkills
   20   , builtinSkillMap
   21   , lookupBuiltinSkill
   22   , builtinSkillIds
   23   ) where
   24 
   25 import Data.FileEmbed (embedFile)
   26 import Data.Map.Strict (Map)
   27 import Data.Map.Strict qualified as Map
   28 import Data.Maybe (mapMaybe)
   29 import Data.Text (Text)
   30 import Data.Text.Encoding qualified as TE
   31 import Data.Time (UTCTime (..))
   32 import Data.Time.Calendar (fromGregorian)
   33 import Data.Time.Clock (secondsToDiffTime)
   34 import Language.Haskell.TH.Syntax (qAddDependentFile)

   37 import Seal.Skills.Types (Skill (..), SkillId)

   39 -- | The compiled-in provenance stamp for built-in skills. A fixed epoch
   40 -- instant so the embedded data is deterministic (no per-build timestamp
   41 -- churn in the binary). A user override's @updated_at@ will always be
   42 -- later, which is the right signal for "this is a user-modified copy."
   43 builtinStamp :: UTCTime
   44 builtinStamp = UTCTime (fromGregorian 2026 7 24) (secondsToDiffTime 0)
   45 
   46 -- | The raw markdown bytes of each built-in skill source, embedded at
   47 -- compile time. Add a tuple here to ship another built-in skill. The first
   48 -- element is the skill id (must match the @id:@ frontmatter inside the
   49 -- file); the second is the embedded file content (via 'embedFile'); the
   50 -- third is the group the built-in belongs to (stamped into 'skGroup' so
   51 -- the @\<available_skills\>@ catalog groups built-ins consistently with
   52 -- their on-disk location).
   53 --
   54 -- Each 'embedFile' splice is paired with a 'qAddDependentFile' call so
   55 -- GHC recompiles this module when an embedded source changes (plain
   56 -- 'embedFile' alone does not register the file as a build dependency, so
   57 -- editing the markdown would not trigger a rebuild — a subtle footgun
   58 -- we close here). The splice must use a literal path (GHC stage
   59 -- restriction: a top-level splice cannot reference a locally-bound
   60 -- variable), so each entry inlines its own splice rather than sharing a
   61 -- single one parameterized by @path@.
   62 builtinRaw :: [(Text, Text, Text)]
   63 builtinRaw =
   64   [ ( "seal-usage"
   65     , TE.decodeUtf8 $(do
   66           qAddDependentFile "config/skills/core/seal-usage.md"
   67           embedFile "config/skills/core/seal-usage.md")
   68     , "core"
   69     )
   70   ]
   71 
   72 -- | The decoded built-in skills, with provenance stamped to 'builtinStamp'
   73 -- (the decoded @session@ defaults to @"unknown"@ per 'decodeSkill'; we
   74 -- leave it — built-ins are not user-authored, so the originating-session
   75 -- field is conventional, not meaningful) and 'skGroup' stamped to the
   76 -- group declared in 'builtinRaw' (so built-ins are always grouped
   77 -- consistently, regardless of any @group:@ frontmatter in the file).
   78 -- Malformed sources are skipped (a malformed built-in is an authoring
   79 -- error caught at the first test run, not a runtime condition the binary
   80 -- should paper over).
   81 builtinSkills :: [Skill]
   82 builtinSkills = mapMaybe decodeAndStamp builtinRaw
   83   where
   84     decodeAndStamp (_expectedId, raw, group) =
   85       case decodeSkill raw of
   86         Nothing    -> Nothing
   87         Just skill -> Just skill
   88           { skCreatedAt = builtinStamp
   89           , skUpdatedAt = builtinStamp
   90           , skGroup     = Just group
   91           }
   92 
   93 -- | 'builtinSkills' keyed by id for O(log n) lookups.
   94 builtinSkillMap :: Map SkillId Skill
   95 builtinSkillMap = Map.fromList [(skId s, s) | s <- builtinSkills]
   96 
   97 -- | The ids of all built-in skills (for listing / override detection).
   98 builtinSkillIds :: [SkillId]
   99 builtinSkillIds = map skId builtinSkills
  100 
  101 -- | Look up a built-in skill by id. 'Nothing' if no built-in has that id.
  102 lookupBuiltinSkill :: SkillId -> Maybe Skill
  103 lookupBuiltinSkill sid = Map.lookup sid builtinSkillMap

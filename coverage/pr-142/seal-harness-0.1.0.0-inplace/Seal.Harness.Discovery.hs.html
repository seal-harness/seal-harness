   15 -- | A discovered tmux window that could be adopted.
   16 data DiscoverableWindow = DiscoverableWindow
   17   { dwTmuxCoord    :: Text        -- ^ "session:window" or "%<pane>"
   18   , dwTitle        :: Text        -- ^ the window's name
   19   , dwFlavourHint  :: Maybe Text  -- ^ guessed from the title
   20   } deriving stock (Eq, Show)
   21 
   22 -- | Scan for unmanaged tmux windows (no @seal_id marker). IO via the
   23 -- 'TmuxRunner' seam: @tmux list-windows -F "#{window_id}:#{window_name}"@
   24 -- (a real call would also check each window's markers; for 6a's
   25 -- fake-runner test, the scripted stdout is parsed as one window per line).
   26 -- Windows whose title starts with @seal:@ (the harness's own naming) are
   27 -- considered managed and filtered out.
   28 scanDiscoverableIO :: TmuxRunner -> IO (Either HarnessError [DiscoverableWindow])
   29 scanDiscoverableIO runner = do
   30   r <- runTmux runner ["list-windows", "-F", "#{window_id}:#{window_name}"]
   31   pure $ case r of
   32     Left e  -> Left e
   33     Right t -> Right (parseWindows t)
   34 
   35 -- | Parse @tmux list-windows -F@ output: one line per window, format
   36 -- @<coord>:<name>@. Filter out managed harness windows (name starts with
   37 -- @seal-@ — the harness naming convention).
   38 parseWindows :: Text -> [DiscoverableWindow]
   39 parseWindows = mapMaybe parseLine . T.lines
   40   where
   41     parseLine ln =
   42       case T.breakOn ":" ln of
   43         (coord, rest)
   44           | T.null rest -> Nothing
   45           | otherwise ->
   46               let title = T.drop 1 rest  -- drop the leading ":"
   47               in if "seal-" `T.isPrefixOf` title
   48                    then Nothing  -- managed harness window
   49                    else Just (DiscoverableWindow coord title (guessFlavour title))
   50 
   51 -- | Guess a flavour from the window title.
   52 guessFlavour :: Text -> Maybe Text
   53 guessFlavour title
   54   | "claude"  `T.isInfixOf` T.toCaseFold title = Just "claude-code"
   55   | "codex"   `T.isInfixOf` T.toCaseFold title = Just "codex"
   56   | otherwise                                  = Nothing
   57 
   58 -- local helper (avoid a Data.Maybe import for one helper)
   59 mapMaybe :: (a -> Maybe b) -> [a] -> [b]
   60 mapMaybe _ []     = []
   61 mapMaybe f (x:xs) = case f x of
   62   Just y  -> y : mapMaybe f xs
   63   Nothing -> mapMaybe f xs

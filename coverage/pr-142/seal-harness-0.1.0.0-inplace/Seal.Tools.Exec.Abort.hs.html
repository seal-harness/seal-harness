    1 -- | Session-scoped abort signals for tool-call cancellation.
    2 --
    3 -- Two layers:
    4 --
    5 -- 1. 'AbortFlag' — a per-session 'IORef' 'Bool'. The channel layer (Signal,
    6 --    CLI, Web) calls 'setAbort' when the user sends a stop/interrupt; the
    7 --    dispatch wrapper polls 'isAborted' during the three-way race (see
    8 --    'Seal.Tools.Exec.Timeout.runWithTimeoutAbortRetry'). 'clearAbort' fires
    9 --    once at 'runTurn' entry (before any tool call); a mid-turn abort keeps
   10 --    the flag set until the next turn begins.
   11 --
   12 -- 2. 'SessionAbortRegistry' — a per-session map of 'AbortFlag's, keyed by
   13 --    'SessionId'. Mirrors 'Seal.Session.Lock.SessionLocks' (a @TVar (Map
   14 --    SessionId (MVar ()))@ lazily created per session). This registry is the
   15 --    vehicle for the web @POST /api/sessions/:id/stop@ endpoint: 'ApiDeps'
   16 --    has no 'AgentEnv' (it's constructed per-turn), so the abort flag must
   17 --    be looked up by session id rather than threaded through 'AgentEnv'
   18 --    alone. The CLI owns one session; the web gateway + inbox channels use
   19 --    the registry.
   20 --
   21 -- Security: the 'AbortFlag' constructor is NOT exported (design Blocker
   22 --    Resolution #6 — mirrors the 'UntrustedIO' capability-handle pattern at
   23 --    'Seal.Tools.Exec.UntrustedIO'). Only the smart constructors and
   24 --    accessors below are exported. This prevents an untrusted opcode from
   25 --    forging an 'AbortFlag' or pattern-matching out the 'IORef' to write
   26 --    directly, bypassing 'setAbort'. The 'SessionAbortRegistry' constructor
   27 --    is likewise unexported.
   28 module Seal.Tools.Exec.Abort
   29   ( AbortFlag
   30   , newAbortFlag
   31   , isAborted
   32   , setAbort
   33   , clearAbort
   34   , waitForAbort
   35   , SessionAbortRegistry
   36   , newSessionAbortRegistry
   37   , lookupOrCreateAbortFlag
   38   , setSessionAbort
   39   , clearSessionAbort
   40   ) where
   41 
   42 import Control.Concurrent (threadDelay)
   43 import Control.Concurrent.STM
   44   ( TVar, atomically, newTVarIO, readTVar, readTVarIO, writeTVar )
   45 import Data.IORef (IORef, newIORef, readIORef, writeIORef)
   46 import Data.Map.Strict (Map)
   47 import Data.Map.Strict qualified as Map
   48 import Data.Maybe (fromMaybe)
   49 
   50 import Seal.Core.Types (SessionId)

   52 -- | A session-scoped abort flag. Set by the channel/user to cancel
   53 -- in-flight tool calls. Checked by the dispatch wrapper during the
   54 -- three-way race. The constructor is NOT exported (Blocker Resolution #6).
   55 newtype AbortFlag = AbortFlag (IORef Bool)
   56 
   57 -- | Per-session abort registry (mirrors 'SessionLocks' at
   58 -- "Seal.Session.Lock"). Maps each 'SessionId' to its 'AbortFlag', lazily
   59 -- created on first lookup. STM-backed so 'lookupOrCreateAbortFlag' is
   60 -- race-safe (two concurrent lookups for a new session won't create two
   61 -- flags — uses @modifyTVar'@ insert-if-absent semantics). The constructor
   62 -- is NOT exported.
   63 newtype SessionAbortRegistry = SessionAbortRegistry (TVar (Map SessionId AbortFlag))
   64 
   65 -- | Build a fresh, non-aborted flag.
   66 newAbortFlag :: IO AbortFlag
   67 newAbortFlag = AbortFlag <$> newIORef False
   68 
   69 -- | Check whether the flag has been set.
   70 isAborted :: AbortFlag -> IO Bool
   71 isAborted (AbortFlag ref) = readIORef ref
   72 
   73 -- | Set the abort flag (idempotent).
   74 setAbort :: AbortFlag -> IO ()
   75 setAbort (AbortFlag ref) = writeIORef ref True
   76 
   77 -- | Clear the abort flag (idempotent; a no-op on a fresh flag).
   78 clearAbort :: AbortFlag -> IO ()
   79 clearAbort (AbortFlag ref) = writeIORef ref False
   80 
   81 -- | Poll 'isAborted' every @pollMicros@ microseconds. Returns 'True' as
   82 -- soon as the flag is set. Used as the third participant in the three-way
   83 -- race (worker vs timeout vs abort-poll). The poll interval is passed in
   84 -- by the caller (defaults to @ttcAbortPollMicros@ = 100ms from
   85 -- 'Seal.Tools.Timeout.ToolTimeoutConfig').
   86 waitForAbort :: AbortFlag -> Int -> IO Bool
   87 waitForAbort flag pollMicros = do
   88   aborted <- isAborted flag
   89   if aborted
   90     then pure True
   91     else do
   92       threadDelay pollMicros
   93       waitForAbort flag pollMicros
   94 
   95 -- | Build a new empty registry.
   96 newSessionAbortRegistry :: IO SessionAbortRegistry
   97 newSessionAbortRegistry = SessionAbortRegistry <$> newTVarIO Map.empty
   98 
   99 -- | Look up the 'AbortFlag' for a session, creating it on first use. The
  100 -- insert-if-absent is atomic via STM, so two concurrent lookups for a new
  101 -- session yield the same flag (no duplicate creation).
  102 lookupOrCreateAbortFlag :: SessionAbortRegistry -> SessionId -> IO AbortFlag
  103 lookupOrCreateAbortFlag (SessionAbortRegistry tv) sid = do
  104   m <- readTVarIO tv
  105   case Map.lookup sid m of
  106     Just flag -> pure flag
  107     Nothing -> do
  108       flag <- newAbortFlag
  109       atomically $ do
  110         m' <- readTVar tv
  111         -- insertIfAbsent semantics: if another thread won the race, use
  112         -- its flag; otherwise insert ours.
  113         let flag' = fromMaybe flag (Map.lookup sid m')
  114         writeTVar tv (Map.insert sid flag' m')
  115         pure flag'
  116 
  117 -- | Set the abort flag for a session (creates the flag on first use).
  118 setSessionAbort :: SessionAbortRegistry -> SessionId -> IO ()
  119 setSessionAbort reg sid = do
  120   flag <- lookupOrCreateAbortFlag reg sid
  121   setAbort flag
  122 
  123 -- | Clear the abort flag for a session (creates the flag on first use if
  124 -- it doesn't exist, then clears it — equivalent to a no-op since a fresh
  125 -- flag is already clear).
  126 clearSessionAbort :: SessionAbortRegistry -> SessionId -> IO ()
  127 clearSessionAbort reg sid = do
  128   flag <- lookupOrCreateAbortFlag reg sid
  129   clearAbort flag

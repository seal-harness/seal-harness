    3 -- | The ISA as data: an 'Opcode' is a GADT-style sum keyed by trust level.
    4 -- A 'TrustedOpcode' carries 'toRun' (no untrusted capability); an
    5 -- 'UntrustedOpcode' carries 'uoRun' (which threads an 'UntrustedIO' — the
    6 -- unified capability handle for all side-effecting IO an untrusted opcode
    7 -- can perform). This split makes the capability-scoping guarantee
    8 -- type-level (spec §4 line 129, §8): a Trusted opcode that shells out
    9 -- literally cannot be constructed — it has no 'UntrustedIO' in scope. The
   10 -- compile-fail fixture ('Seal.Tools.Exec.CapabilityScopingFail') asserts
   11 -- this.
   12 --
   13 -- Trusted opcodes use 'BackendExec' for their own non-untrusted IO (e.g.
   14 -- memory/skill file writes under @config/@); Untrusted opcodes use
   15 -- 'UntrustedIO' for ALL their side-effecting IO (files, commands, process
   16 -- management, search). The 'UntrustedIO' is the single seam — backend
   17 -- (local vs remote SSH) is selected once at wiring time; the opcode never
   18 -- sees the backend, only the capability.
   19 module Seal.ISA.Opcode

   21   , Opcode (..)
   22   , BackendExec (..)

   27   , opInSchema

   43 data OpResult = OpResult
   44   { orParts :: [ToolResultPart]  -- ^ what the model sees (may include secret values)
   45   , orIsError :: Bool
   46   , orRecorded :: Value          -- ^ what the transcript records (secret-free)
   47   } deriving stock (Eq)
   48 
   49 -- | Manual Show that never renders 'orParts': those slots may hold secret values
   50 -- (e.g. vault credentials returned by SECRET_GET) that must not appear in logs.
   51 instance Show OpResult where
   52   show r = "OpResult {orParts = <" <> show (length (orParts r)) <> " part(s) redacted>, orIsError = "
   53              <> show (orIsError r) <> ", orRecorded = " <> show (orRecorded r) <> "}"

   55 -- | The execution seam. Untrusted opcodes funnel their IO through 'runLocal';
   56 -- Phase 4 introduces a remote-SSH 'BackendExec' with the same shape.
   57 newtype BackendExec = BackendExec { runLocal :: forall a. IO a -> App a }
   58 
   59 localBackend :: BackendExec
   60 localBackend = BackendExec liftIO
   61 
   62 -- | The opcode sum. 'TrustedOpcode' covers both 'Trusted' and 'Audited'
   63 -- (Audited is treated as Trusted by the dispatcher — record async, then
   64 -- run; the 'toTrust' field carries the distinction for 'opTrust').
   65 -- 'UntrustedOpcode' is always 'Untrusted' and carries the 'UntrustedIO'
   66 -- its 'uoRun' threads. The field-name prefixes (@to@ / @uo@) keep the two
   67 -- constructors' record namespaces disjoint (Haskell requires this when
   68 -- constructors share a type but have different fields).
   69 data Opcode
   70   = TrustedOpcode
   71       { toName       :: OpName
   72       , toTrust      :: TrustLevel       -- ^ 'Trusted' or 'Audited'
   73       , toDesc       :: Text
   74       , toInSchema   :: Value
   75       , toOutSchema  :: Value
   76       , toAuthorize  :: Value -> Either Text ()
   77       , toRun        :: BackendExec -> Value -> App OpResult
   78       }
   79   | UntrustedOpcode
   80       { uoName       :: OpName
   81       , uoDesc       :: Text
   82       , uoInSchema   :: Value
   83       , uoOutSchema  :: Value
   84       , uoAuthorize  :: Value -> Either Text ()
   85       , uoRun        :: Value -> UIO OpResult
   86       }
   87 
   88 -- | Accessor: the opcode's name (works for both constructors).
   89 opName :: Opcode -> OpName
   90 opName (TrustedOpcode n _ _ _ _ _ _) = n
   91 opName (UntrustedOpcode n _ _ _ _ _) = n
   92 
   93 -- | Accessor: the trust level. 'UntrustedOpcode' is always 'Untrusted'.
   94 opTrust :: Opcode -> TrustLevel
   95 opTrust (TrustedOpcode _ tl _ _ _ _ _) = tl
   96 opTrust (UntrustedOpcode {})  = Untrusted
   97 
   98 -- | Accessor: the description (for tool definitions).
   99 opDesc :: Opcode -> Text
  100 opDesc (TrustedOpcode _ _ d _ _ _ _) = d
  101 opDesc (UntrustedOpcode _ d _ _ _ _) = d
  102 
  103 -- | Accessor: the input JSON schema.
  104 opInSchema :: Opcode -> Value
  105 opInSchema (TrustedOpcode _ _ _ s _ _ _) = s
  106 opInSchema (UntrustedOpcode _ _ s _ _ _) = s
  107 
  108 -- | Accessor: the output JSON schema.
  109 opOutSchema :: Opcode -> Value
  110 opOutSchema (TrustedOpcode _ _ _ _ s _ _) = s
  111 opOutSchema (UntrustedOpcode _ _ _ s _ _) = s
  112 
  113 -- | Accessor: the authorization gate.
  114 opAuthorize :: Opcode -> Value -> Either Text ()
  115 opAuthorize (TrustedOpcode _ _ _ _ _ a _) = a
  116 opAuthorize (UntrustedOpcode _ _ _ _ a _) = a
  117 
  118 -- | Accessor: the run action for a TRUSTED opcode. Calling this on an
  119 -- 'UntrustedOpcode' is an error (use the dispatcher, which pattern-matches
  120 -- the GADT and calls 'uoRun' for untrusted). Provided so legacy call
  121 -- sites that read 'opRun' on a known-trusted opcode keep working.
  122 opRun :: Opcode -> BackendExec -> Value -> App OpResult
  123 opRun (TrustedOpcode _ _ _ _ _ _ r) = r
  124 opRun UntrustedOpcode {} =
  125   error "opRun: UntrustedOpcode has no toRun; use the dispatcher's uoRun path"
  126 
  127 -- | Helper for the one record-update site (DispatchSpec): replace the
  128 -- authorize function on an opcode (GADTs forbid record-update on a sum
  129 -- whose fields differ per constructor).
  130 withAuthorize :: Opcode -> (Value -> Either Text ()) -> Opcode
  131 withAuthorize (TrustedOpcode n tl d is os _ r) f =
  132   TrustedOpcode n tl d is os f r
  133 withAuthorize (UntrustedOpcode n d is os _ r) f =
  134   UntrustedOpcode n d is os f r

   26 -- | The evolutionary-store backends + the in-process agent runtime, created
   27 -- once at startup and shared between the command specs (which read them via
   28 -- @\/skill@ \/ @\/agent@) and the ISA opcodes (which mutate them). The three
   29 -- store backends are disk-backed (Markdown files under @config\/@); disk is
   30 -- canonical and git is the versioning + audit layer. The agent runtime is an
   31 -- in-process STM registry (lifecycle only — not persisted). The delegation
   32 -- knobs (config, pause flag, parent-activity cell) are process-global so
   33 -- AGENT_START calls across all channels share one pause / heartbeat state.
   34 data Backends = Backends
   35   { bMemory    :: Mem.MemoryBackend
   36   , bSkills    :: Skill.SkillBackend
   37   , bAgentDefs :: Def.AgentDefBackend
   38   , bRuntime   :: AgentRuntime
   39   , bDelegationConfig :: IO DelegationConfig
   40     -- ^ Reload the [delegation] config per AGENT_START call (so config
   41     -- changes take effect without a restart). The IO action reads
   42     -- @config.toml@ and returns the resolved 'DelegationConfig'.
   43   , bSpawnPauseFlag :: SpawnPauseFlag
   44     -- ^ Process-global spawn-pause flag (operator can freeze new fan-out).
   45   , bParentActivity :: ParentActivity
   46     -- ^ Process-global parent-activity cell (heartbeat target).
   47   }
   48 
   49 -- | Construct the disk-backed backends for the given config repo. The three
   50 -- stores read their directories on demand (no startup materialization needed
   51 -- — disk is canonical, so @\/skill list@ etc. just enumerate the dir). The
   52 -- delegation knobs are process-global; the config is re-read per AGENT_START
   53 -- call so config changes take effect without a restart.
   54 newBackends :: FilePath -> ConfigRepo -> IO Backends
   55 newBackends cfgRoot repo = do
   56   let skillsDir    = cfgRoot </> "skills"
   57       agentsDir    = cfgRoot </> "agents"
   58       memoryDir    = cfgRoot </> "memory"
   59   rt          <- newAgentRuntime
   60   pauseFlag   <- newSpawnPauseFlag
   61   parentAct   <- newParentActivity
   62   Backends
   63     <$> Mem.markdownMemoryBackend memoryDir repo
   64     <*> (Skill.unionSkillBackend <$> Skill.markdownSkillBackend skillsDir repo)
   65     <*> Def.markdownAgentDefBackend agentsDir repo
   66     <*> pure rt
   67     <*> pure (pure defaultDelegationConfig)  -- overridden at call sites that have a config path
   68     <*> pure pauseFlag
   69     <*> pure parentAct

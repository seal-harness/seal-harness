   65 -- | A local/remote-agnostic, SafePath-confined filesystem vocabulary for
   66 -- workspace-derived discovery. Each field is an IO action; the smart
   67 -- constructors wire the local, in-memory-stub, or fail-closed-stub arm.

   69   { wfsReadFile            :: RemotePath -> IO (Either WorkdirFsErr Text)
   70   , wfsDoesFileExist       :: RemotePath -> IO Bool
   71   , wfsDoesDirectoryExist  :: RemotePath -> IO Bool
   72   , wfsListDirectory       :: RemotePath -> IO (Either WorkdirFsErr [Text])

   77     -- operation (one SSH round trip on the remote arm). Enumerates regular

  119 -- | The error ADT for 'WorkdirFs' methods. Pinned for exhaustive
  120 -- pattern-match under @-Wall -Werror@.
  121 data WorkdirFsErr
  122   = WfsPath !PathError
  123   | WfsNotFound
  124   | WfsOversize
  125   | WfsIo !Text
  126   | WfsExec !Text
  127   | WfsStub
  128   deriving stock (Eq, Show)

  130 -- | An entry in the in-memory stub's seeded map. Models files, symlink

  136   | Directory [Text]

  145 -- | Truncate a body to @limit@ characters, appending the exact truncation
  146 -- marker. Strings at or under the limit are returned as-is. Matches
  147 -- 'truncateSection' in "Seal.Agent.Def.Backend".
  148 truncateSection :: Int -> Text -> Text
  149 truncateSection limit txt
  150   | T.length txt <= limit = txt
  151   | otherwise =
  152       T.take limit txt
  153         <> "\n[...truncated at " <> T.pack (show limit) <> " chars...]"

  159 -- | The local arm. Reads via 'System.Directory' / 'Data.Text.IO'; every
  160 -- path is run through 'mkSafePath' (anchored at the workdir root) first.
  161 -- The @Int@ is the operator scan-byte ceiling, captured at construction.
  162 mkLocalWorkdirFs :: WorkspaceRoot -> Int -> WorkdirFs
  163 mkLocalWorkdirFs wsRoot ceilingBytes = WorkdirFs
  164   { wfsReadFile = \rp -> do
  165       let rel = T.unpack (getRemotePath rp)
  166       eSafe <- mkSafePath wsRoot rel
  167       case eSafe of
  168         Left (PathDoesNotExist _) -> pure (Left WfsNotFound)
  169         Left pe -> pure (Left (WfsPath pe))
  170         Right sp -> readLocalBounded (getSafePath sp) ceilingBytes
  171   , wfsDoesFileExist = \rp -> do
  172       let rel = T.unpack (getRemotePath rp)
  173       eSafe <- mkSafePath wsRoot rel
  174       case eSafe of
  175         Left _  -> pure False
  176         Right sp -> doesFileExist (getSafePath sp)
  177   , wfsDoesDirectoryExist = \rp -> do
  178       let rel = T.unpack (getRemotePath rp)
  179       eSafe <- mkSafePath wsRoot rel
  180       case eSafe of
  181         Left _  -> pure False
  182         Right sp -> doesDirectoryExist (getSafePath sp)
  183   , wfsListDirectory = \rp -> do
  184       let rel = T.unpack (getRemotePath rp)
  185       eSafe <- mkSafePath wsRoot rel
  186       case eSafe of
  187         Left (PathDoesNotExist _) -> pure (Right [])
  188         Left pe -> pure (Left (WfsPath pe))
  189         Right sp -> do
  190           exists <- doesDirectoryExist (getSafePath sp)
  191           if not exists
  192             then pure (Right [])
  193             else do
  194               eNames <- try (listDirectory (getSafePath sp))
  195                         :: IO (Either IOException [FilePath])
  196               pure $ case eNames of
  197                 Left ioErr -> Left (WfsIo (T.pack (show ioErr)))
  198                 Right names -> Right (map T.pack names)
  199   , wfsFileSize = \rp -> do
  200       let rel = T.unpack (getRemotePath rp)
  201       eSafe <- mkSafePath wsRoot rel
  202       case eSafe of
  203         Left (PathDoesNotExist _) -> pure (Left WfsNotFound)
  204         Left pe -> pure (Left (WfsPath pe))
  205         Right sp -> do
  206           eSize <- try (getFileSize (getSafePath sp))
  207                     :: IO (Either IOException Integer)
  208           pure $ case eSize of
  209             Left _ -> Left WfsNotFound
  210             Right size -> Right size
  211   , wfsModificationTime = \rp -> do
  212       let rel = T.unpack (getRemotePath rp)
  213       eSafe <- mkSafePath wsRoot rel
  214       case eSafe of
  215         Left (PathDoesNotExist _) -> pure (Left WfsNotFound)
  216         Left pe -> pure (Left (WfsPath pe))
  217         Right sp -> do
  218           eMtime <- try (getModificationTime (getSafePath sp))
  219                      :: IO (Either IOException UTCTime)
  220           pure $ case eMtime of
  221             Left _ -> Left WfsNotFound
  222             Right mtime -> Right mtime
  223   , wfsSnapshot = localSnapshot wsRoot
  224   }

  226 -- | The depth-limited structural walk for the local arm ('wfsSnapshot').
  227 -- Enumerates regular dirs + files only; symlinks are skipped entirely
  228 -- (never followed — mirrors the remote @find -P@ semantics so both arms
  229 -- agree on structure). Depth- and count-bounded.
  230 localSnapshot :: WorkspaceRoot -> IO (Either WorkdirFsErr WorkspaceSnapshot)
  231 localSnapshot wsRoot = do
  232   entries <- walkDir root "" snapshotMaxDepth
  233   pure $!
  234     if length entries > maxSnapshotEntries
  235       then Left WfsOversize
  236       else Right (WorkspaceSnapshot (sortOn snapRelPath entries))
  237   where
  238     root = case wsRoot of WorkspaceRoot p -> p
  239     walkDir :: FilePath -> Text -> Int -> IO [SnapshotEntry]
  240     walkDir dir relPrefix remaining
  241       | remaining <= 0 = pure []
  242       | otherwise = do
  243           eNames <- try (listDirectory dir) :: IO (Either IOException [FilePath])
  244           case eNames of
  245             Left _     -> pure []   -- unreadable subtree: fail-soft
  246             Right names -> concat <$> mapM step (sort names)
  247       where
  248         step name = do
  249           let childRel = if T.null relPrefix
  250                            then T.pack name
  251                            else relPrefix <> "/" <> T.pack name
  252               fsPath = dir </> name
  253           eLink <- try (pathIsSymbolicLink fsPath)
  254                      :: IO (Either IOException Bool)
  255           if fromRight True eLink
  256             then pure []   -- symlink (or vanished): never followed
  257             else do
  258               isDir <- doesDirectoryExist fsPath
  259               if isDir
  260                 then do
  261                   sub <- walkDir fsPath childRel (remaining - 1)
  262                   pure (SnapshotEntry childRel True : sub)
  263                 else do
  264                   isFile <- doesFileExist fsPath
  265                   pure [SnapshotEntry childRel False | isFile]

  267 -- | Stat-first (reject 'WfsOversize' if > ceiling), then read + decode to
  268 -- 'Text', truncate to 'defaultSectionCharLimit'. Returns 'WfsNotFound' if
  269 -- the file is missing.
  270 readLocalBounded :: FilePath -> Int -> IO (Either WorkdirFsErr Text)
  271 readLocalBounded path ceilingBytes = do
  272   eSize <- try (getFileSize path) :: IO (Either IOException Integer)
  273   case eSize of
  274     Left _ -> pure (Left WfsNotFound)
  275     Right size
  276       | size > fromIntegral ceilingBytes -> pure (Left WfsOversize)
  277       | otherwise -> do
  278           eTxt <- try (TIO.readFile path) :: IO (Either IOException Text)
  279           pure $ case eTxt of
  280             Left ioErr -> Left (WfsIo (T.pack (show ioErr)))
  281             Right txt ->
  282               let trimmed = T.dropWhileEnd Char.isSpace txt
  283               in if T.null (T.strip trimmed)
  284                    then Left WfsNotFound
  285                    else Right (truncateSection defaultSectionCharLimit trimmed)

  291 -- | The remote arm. Reads via SSH through 'UIO'\'s 'uioShellExec' (the shared
  292 -- 'RemoteRunner' lives in the 'UIOEnv'). Every path is run through
  293 -- 'mkSafePathRemote' (anchored at the remote workspace root) BEFORE any
  294 -- shell-exec call. 'wfsReadFile'/'wfsListDirectory'/'wfsFileSize'/
  295 -- 'wfsModificationTime' additionally run @realpath -f -- <abspath>@ on the
  296 -- remote OS and re-check containment on the resolved path, so a symlink
  297 -- escaping the workspace is rejected before any content is read or any
  298 -- directory is listed. The @Int@ is the operator scan-byte ceiling,
  299 -- captured at construction.
  300 mkRemoteWorkdirFs :: UIOEnv -> SshConfig -> WorkspaceRoot -> Int -> WorkdirFs
  301 mkRemoteWorkdirFs uioEnv sshCfg wsRoot ceilingBytes = WorkdirFs
  302   { wfsReadFile = \rp -> do
  303       let rel = T.unpack (getRemotePath rp)
  304       case mkSafePathRemote wsRoot rel of
  305         Left pe -> pure (Left (WfsPath pe))
  306         Right sp -> remoteReadFile uioEnv sshCfg wsRoot ceilingBytes (getSafePath sp)
  307   , wfsDoesFileExist = \rp -> do
  308       let rel = T.unpack (getRemotePath rp)
  309       case mkSafePathRemote wsRoot rel of
  310         Left _  -> pure False
  311         Right sp -> remoteTestFlag uioEnv sshCfg "-f" (getSafePath sp)
  312   , wfsDoesDirectoryExist = \rp -> do
  313       let rel = T.unpack (getRemotePath rp)
  314       case mkSafePathRemote wsRoot rel of
  315         Left _  -> pure False
  316         Right sp -> remoteTestFlag uioEnv sshCfg "-d" (getSafePath sp)
  317   , wfsListDirectory = \rp -> do
  318       let rel = T.unpack (getRemotePath rp)
  319       case mkSafePathRemote wsRoot rel of
  320         Left pe -> pure (Left (WfsPath pe))
  321         Right sp -> remoteListDirectory uioEnv sshCfg wsRoot (getSafePath sp)
  322   , wfsFileSize = \rp -> do
  323       let rel = T.unpack (getRemotePath rp)
  324       case mkSafePathRemote wsRoot rel of
  325         Left pe -> pure (Left (WfsPath pe))
  326         Right sp ->
  327           remoteStatRechecked uioEnv sshCfg wsRoot "%s" "%z" (getSafePath sp) >>= \case
  328             Left e -> pure (Left e)
  329             Right txt -> case parseInteger txt of
  330               Nothing -> pure (Left (WfsIo "stat size: non-numeric stdout"))
  331               Just n  -> pure (Right n)
  332   , wfsModificationTime = \rp -> do
  333       let rel = T.unpack (getRemotePath rp)
  334       case mkSafePathRemote wsRoot rel of
  335         Left pe -> pure (Left (WfsPath pe))
  336         Right sp ->
  337           remoteStatRechecked uioEnv sshCfg wsRoot "%Y" "%m" (getSafePath sp) >>= \case
  338             Left e -> pure (Left e)
  339             Right txt -> case parseInteger txt of
  340               Nothing -> pure (Left (WfsIo "stat mtime: non-numeric stdout"))
  341               Just n  -> pure (Right (posixSecondsToUTCTime (fromIntegral n)))
  342   , wfsSnapshot = remoteSnapshot uioEnv ceilingBytes
  343   }

  345 -- | The depth-limited structural snapshot for the remote arm
  346 -- ('wfsSnapshot'): ONE shell command runs two @find@ passes (directories,
  347 -- then files) over the workspace root, separated by a marker line, with
  348 -- the total output bounded by @head -c <ceiling>@.
  349 --
  350 -- /Portability/: @-maxdepth@ is not POSIX but is supported by GNU findutils,
  351 -- BSD find, macOS find, and busybox — same de-facto-universal status as the
  352 -- @readlink -f@ choice above. Symlinks are never followed (the default
  353 -- @-P@ mode matches neither @-type d@ nor @-type f@), so an escaping symlink
  354 -- can never contribute structure. Names containing newlines cannot be
  355 -- represented in the line-based output — the same limitation as the
  356 -- line-splitting of @ls -1@ elsewhere in this module.
  357 remoteSnapshot :: UIOEnv -> Int -> IO (Either WorkdirFsErr WorkspaceSnapshot)
  358 remoteSnapshot uioEnv ceilingBytes = do
  359   let cmd = T.pack $
  360         "{ find . -maxdepth " <> show snapshotMaxDepth <> " -type d -print;"
  361           <> " printf '\\n" <> T.unpack snapshotFilesMarker <> "\\n';"
  362           <> " find . -maxdepth " <> show snapshotMaxDepth <> " -type f -print; }"
  363           <> " 2>/dev/null | head -c " <> show ceilingBytes
  364   res <- remoteExecText uioEnv cmd
  365   pure $ case res of
  366     Left e  -> Left e
  367     Right out -> fmap WorkspaceSnapshot (parseSnapshotOutput ceilingBytes out)
  368 
  369 -- | Parse the remote snapshot output: @find -type d@ lines (@./@-prefixed),
  370 -- the 'snapshotFilesMarker' line, then @find -type f@ lines. Output at or
  371 -- over the byte ceiling is treated as truncated ('WfsOversize' — a silently
  372 -- truncated tree would yield wrong discovery results); a missing marker is
  373 -- malformed output ('WfsIo'). The @.@ root line and any non-@./@ line are
  374 -- skipped. Entries are sorted by path.
  375 parseSnapshotOutput :: Int -> Text -> Either WorkdirFsErr [SnapshotEntry]
  376 parseSnapshotOutput ceilingBytes out
  377   | T.length out >= ceilingBytes = Left WfsOversize
  378   | otherwise = case break (== snapshotFilesMarker) (T.lines out) of
  379       (dirLines, _marker : fileLines) -> do
  380         let entries =
  381               mapMaybe pathEntry dirLines  -- dirs
  382                 <>
  383               [ SnapshotEntry p False | Just p <- map pathLine fileLines ]
  384         if length entries > maxSnapshotEntries
  385           then Left WfsOversize
  386           else Right (sortOn snapRelPath entries)
  387       _ -> Left (WfsIo "snapshot output missing files marker")
  388   where
  389     -- A find output line "./a/b" → the relative path "a/b". The bare "."
  390     -- root line and anything unexpected are skipped.
  391     pathLine :: Text -> Maybe Text
  392     pathLine ln =
  393       case T.stripPrefix "./" ln of
  394         Just rest | not (T.null rest) -> Just rest
  395         _ -> Nothing
  396     pathEntry :: Text -> Maybe SnapshotEntry
  397     pathEntry ln = SnapshotEntry <$> pathLine ln <*> pure True
  398 
  399 -- | Run a remote shell command (a 'Text' command string) via 'uioShellExec'
  400 -- in the shared 'UIOEnv'. The command runs with no cwd override (the paths
  401 -- are absolute, SafePath-validated). Lifts the 'UntrustedErr' to a
  402 -- 'WorkdirFsErr'.
  403 --
  404 -- /Portability/: the remote commands in this module avoid GNU-coreutils-
  405 -- only flags so they work on both GNU/Linux and BSD/macOS remotes:
  406 --   * @readlink -f --@ (not @realpath -f --@, which is GNU-only).
  407 --   * @test -<flag> <path>@ (no @--@; the shell-builtin @test@ on zsh/BSD
  408 --     rejects @--@). Safe because every path passed here is absolute
  409 --     (starts with @/@, from 'mkSafePathRemote' + the workspace root), so
  410 --     it can never be mistaken for a @test@ flag.
  411 --   * @head -c <n> --@ for bounded reads (portable).
  412 --   * @ls -1 --@ for directory listings (portable).
  413 remoteExecText :: UIOEnv -> Text -> IO (Either WorkdirFsErr Text)
  414 remoteExecText uioEnv cmdText =
  415   case mkShellCommand cmdText of
  416     Left e -> pure (Left (WfsIo e))
  417     Right cmd ->
  418       runUIOWithEnv uioEnv (uioShellExec cmd Nothing) >>= \case
  419         Left (UePath pe) -> pure (Left (WfsPath pe))
  420         Left (UeExec _)  -> pure (Left (WfsExec "remote shell exec failed"))
  421         Left (UeIo msg)  -> pure (Left (WfsIo msg))
  422         Left (UeBounded n) -> pure (Left (WfsIo ("bounded: " <> T.pack (show n))))
  423         Right out -> pure (Right out)
  424 
  425 -- | Resolve a remote absolute path with @readlink -f -- <abspath>@, then
  426 -- re-check lexical containment of the resolved path against the workspace
  427 -- root. @readlink -f@ is used instead of GNU-only @realpath -f@ so the
  428 -- command works on BSD/macOS remotes as well as GNU/Linux. Returns:
  429 --
  430 --   * @Right (Just resolved)@ — the resolved absolute path, contained.
  431 --   * @Right Nothing@ — the path is missing on the remote (@readlink -f@
  432 --     exits non-zero for a non-existent / unresolvable path on both GNU
  433 --     and BSD).
  434 --   * @Left err@ — exec failure, OR the resolved path escapes the workspace
  435 --     root ('WfsPath').
  436 remoteRealpathRecheck
  437   :: UIOEnv -> SshConfig -> WorkspaceRoot -> FilePath
  438   -> IO (Either WorkdirFsErr (Maybe FilePath))
  439 remoteRealpathRecheck uioEnv _sshCfg wsRoot absPath = do
  440   let cmd = T.pack ("readlink -f -- " <> shellQuote absPath)
  441   remoteExecText uioEnv cmd >>= \case
  442     Left e -> pure (Left e)
  443     Right out
  444       | T.null (T.strip out) -> pure (Right Nothing)
  445       | otherwise ->
  446           let resolved = T.unpack (T.strip out)
  447           in case mkSafePathRemote wsRoot resolved of
  448                Left pe -> pure (Left (WfsPath pe))
  449                Right _ -> pure (Right (Just resolved))
  450 
  451 -- | @wfsReadFile@ on the remote arm: @readlink -f@ → re-check containment →
  452 -- bounded read via @head -c <ceil> --@ → raw 'Text'. A missing path's
  453 -- @readlink -f@ failure yields 'WfsNotFound'. The byte cap (@head -c@)
  454 -- replaces a prior @stat -c %s@ oversize pre-check (which was GNU-only and
  455 -- broke on BSD/macOS remotes); @head -c@ is portable and still bounds the
  456 -- read to at most @ceilingBytes@, so an oversize file is read partially
  457 -- rather than rejected outright — acceptable for discovery (the section
  458 -- truncation downstream caps content regardless).
  459 remoteReadFile
  460   :: UIOEnv -> SshConfig -> WorkspaceRoot -> Int -> FilePath
  461   -> IO (Either WorkdirFsErr Text)
  462 remoteReadFile uioEnv sshCfg wsRoot ceilingBytes absPath =
  463   remoteRealpathRecheck uioEnv sshCfg wsRoot absPath >>= \case
  464     Left e -> pure (Left e)
  465     Right Nothing -> pure (Left WfsNotFound)
  466     Right (Just resolved) -> do
  467       readRes <- remoteExecText uioEnv
  468         (T.pack ( "head -c " <> show ceilingBytes <> " -- "
  469                <> shellQuote resolved))
  470       case readRes of
  471         Left e -> pure (Left e)
  472         Right raw ->
  473           let trimmed = T.dropWhileEnd Char.isSpace raw
  474           in if T.null (T.strip trimmed)
  475                then pure (Left WfsNotFound)
  476                else pure (Right (truncateSection defaultSectionCharLimit trimmed))
  477 
  478 -- | @wfsDoesFileExist@ / @wfsDoesDirectoryExist@ on the remote arm:
  479 -- @test -<flag> <abspath>@ → parse stdout. Exempt from the @readlink -f@
  480 -- re-check (1-bit bool, no target identity). The path is NOT preceded by
  481 -- @--@ because the shell-builtin @test@ on zsh/BSD rejects @--@; this is
  482 -- safe because every path here is absolute (starts with @/@ via
  483 -- 'mkSafePathRemote' + the workspace root) and 'shellQuote'-d, so it
  484 -- cannot be parsed as a @test@ flag.
  485 remoteTestFlag :: UIOEnv -> SshConfig -> String -> FilePath -> IO Bool
  486 remoteTestFlag uioEnv _sshCfg flag absPath = do
  487   let cmd = T.pack ("test " <> flag <> " " <> shellQuote absPath
  488                     <> " && echo y")
  489   remoteExecText uioEnv cmd >>= \case
  490     Right out -> pure (T.strip out == "y")
  491     Left _ -> pure False
  492 
  493 -- | @wfsListDirectory@ on the remote arm: @readlink -f@ → re-check
  494 -- containment → @ls -1 -- <resolved>@ → split lines. A missing path's
  495 -- @readlink -f@ failure (or a missing dir) yields @Right []@ (fail-soft-to-
  496 -- empty).
  497 remoteListDirectory
  498   :: UIOEnv -> SshConfig -> WorkspaceRoot -> FilePath
  499   -> IO (Either WorkdirFsErr [Text])
  500 remoteListDirectory uioEnv sshCfg wsRoot absPath =
  501   remoteRealpathRecheck uioEnv sshCfg wsRoot absPath >>= \case
  502     Left e -> pure (Left e)
  503     Right Nothing -> pure (Right [])
  504     Right (Just resolved) -> do
  505       lsRes <- remoteExecText uioEnv
  506                  (T.pack ("ls -1 -- " <> shellQuote resolved))
  507       case lsRes of
  508         Left e -> pure (Left e)
  509         Right out -> pure (Right (filter (not . T.null) (T.lines out)))
  510 
  511 -- | A portable remote @stat@ for one numeric field. Tries GNU
  512 -- @stat -c <fmt> --@ first; on a non-numeric / empty result (the BSD
  513 -- @stat@ rejects @-c@), retries with BSD @stat -f <fmt>@ (no @--@, which
  514 -- BSD @stat@ also rejects). Returns the raw stdout 'Text' on success.
  515 -- A missing path's @readlink -f@ failure yields 'WfsNotFound'.
  516 remoteStatRechecked
  517   :: UIOEnv -> SshConfig -> WorkspaceRoot -> String -> String -> FilePath
  518   -> IO (Either WorkdirFsErr Text)
  519 remoteStatRechecked uioEnv sshCfg wsRoot gnuFmt bsdFmt absPath =
  520   remoteRealpathRecheck uioEnv sshCfg wsRoot absPath >>= \case
  521     Left e -> pure (Left e)
  522     Right Nothing -> pure (Left WfsNotFound)
  523     Right (Just resolved) -> do
  524       gnuRes <- remoteExecText uioEnv
  525                   (T.pack ("stat -c " <> gnuFmt <> " -- " <> shellQuote resolved))
  526       case gnuRes of
  527         Right out
  528           | not (T.null (T.strip out))
  529           , T.all isDigit (T.strip (T.dropWhile isSpace out)) -> pure (Right out)
  530         _ -> do
  531           bsdRes <- remoteExecText uioEnv
  532                       (T.pack ("stat -f" <> bsdFmt <> " " <> shellQuote resolved))
  533           case bsdRes of
  534             Left e -> pure (Left e)
  535             Right out -> pure (Right out)
  536 
  537 -- | Parse a non-negative integer from 'Text' stdout (the output of
  538 -- @stat -c %s@ / @stat -c %Y@). Total — returns 'Nothing' on any
  539 -- non-numeric input (including negative or empty).
  540 parseInteger :: Text -> Maybe Integer
  541 parseInteger txt =
  542   let t = T.dropWhile isSpace (T.strip txt)
  543   in if not (T.null t) && T.all isDigit t
  544        then Just (read (T.unpack t))
  545        else Nothing

  551 -- | The maximum symlink-chain depth the in-memory stub will resolve before
  552 -- rejecting with 'WfsIo'. Bounds against symlink loops.
  553 maxSymlinkDepth :: Int
  554 maxSymlinkDepth = 16
  555 
  556 -- | The in-memory stub. The stub's 'wfsReadFile' resolves 'SymlinkTarget'
  557 -- chains (depth-bounded), re-checks containment on the resolved path, and
  558 -- rejects on escape — mirroring the real remote arm's @realpath@ logic so
  559 -- the symlink-escape test is non-vacuous.
  560 mkInMemWorkdirFs :: Map RemotePath StubEntry -> WorkdirFs
  561 mkInMemWorkdirFs seed = WorkdirFs
  562   { wfsReadFile = pure . stubReadFile seed
  563   , wfsDoesFileExist = pure . stubDoesFileExist seed
  564   , wfsDoesDirectoryExist = pure . stubDoesDirectoryExist seed
  565   , wfsListDirectory = pure . stubListDirectory seed
  566   , wfsFileSize = pure . stubFileSize seed
  567   , wfsModificationTime = \_ -> pure (Right epochZero)
  568   , wfsSnapshot = pure (stubSnapshot seed)
  569   }
  570 
  571 -- | Derive the structural snapshot from the in-memory seed ('wfsSnapshot'
  572 -- on the stub arm). Mirrors the real arms' semantics:
  573 --
  574 --   * 'Directory' entries become dir entries; their listed children are
  575 --     classified by seed lookup — 'FileContent' or an absent key ⇒ file,
  576 --     another 'Directory' ⇒ dir (recursed), 'Missing' or 'SymlinkTarget'
  577 --     ⇒ skipped (structure never follows symlinks).
  578 --   * Depth- and count-bounded like the local/remote arms.
  579 stubSnapshot :: Map RemotePath StubEntry -> Either WorkdirFsErr WorkspaceSnapshot
  580 stubSnapshot seed =
  581   let rootNames = case Map.lookup (stubKey ".") seed of
  582         Just (Directory ns) -> sort ns
  583         _ -> let comps = sort [ firstComp (getRemotePath k)
  584                               | k <- Map.keys seed, getRemotePath k /= "." ]
  585              in Set.toAscList (Set.fromList comps)   -- sorted-dedup
  586       walk prefix names remaining
  587         | remaining <= 0 = []
  588         | otherwise = concatMap (step prefix remaining) names
  589       step prefix remaining name =
  590         let childRel = joinRel prefix name
  591         in case Map.lookup (stubKey childRel) seed of
  592              Just (Directory ns) ->
  593                SnapshotEntry childRel True : walk childRel (sort ns) (remaining - 1)
  594              Just Missing -> []
  595              Just (SymlinkTarget _) -> []   -- structure never follows symlinks
  596              Just (FileContent _) -> [SnapshotEntry childRel False]
  597              Nothing -> [SnapshotEntry childRel False]  -- implicit child = file
  598       entries = walk "" rootNames snapshotMaxDepth
  599   in if length entries > maxSnapshotEntries
  600        then Left WfsOversize
  601        else Right (WorkspaceSnapshot (sortOn snapRelPath entries))
  602   where
  603     stubKey t = case mkRemotePath t of
  604       Right k  -> k
  605       Left err -> error ("stubSnapshot: invalid seeded path: " <> T.unpack err
  606                          <> ": " <> T.unpack t)
  607     firstComp = T.takeWhile (/= '/')
  608     joinRel prefix name
  609       | T.null prefix = name
  610       | otherwise     = prefix <> "/" <> name
  611 
  612 -- | Resolve a 'RemotePath' through 'SymlinkTarget' chains, re-checking
  613 -- containment on the resolved path at each step. Returns 'Right' with the
  614 -- final 'RemotePath' pointing at a non-symlink entry, or 'Left' on
  615 -- escape / loop / missing.
  616 resolveSymlinkChain
  617   :: Map RemotePath StubEntry
  618   -> RemotePath
  619   -> Either WorkdirFsErr StubEntry
  620 resolveSymlinkChain seed = go 0
  621   where
  622     go depth rp
  623       | depth > maxSymlinkDepth = Left (WfsIo "symlink chain too deep")
  624       | otherwise = case Map.lookup rp seed of
  625           Nothing -> Left WfsNotFound
  626           Just Missing -> Left WfsNotFound
  627           Just (SymlinkTarget target) ->
  628             case checkContainment target of
  629               Left pe -> Left (WfsPath pe)
  630               Right _ -> go (depth + 1) target
  631           Just entry -> Right entry
  632 
  633 -- | Lexical containment check on a 'RemotePath' against the stub's
  634 -- workspace root (@\/workspace@). Mirrors 'mkSafePathRemote' — a path
  635 -- that lexically escapes the root (e.g. @\/etc\/shadow@, @..\/passwd@) is
  636 -- rejected.
  637 checkContainment :: RemotePath -> Either PathError ()
  638 checkContainment rp =
  639   case mkSafePathRemote stubWorkspaceRoot (T.unpack (getRemotePath rp)) of
  640     Left pe -> Left pe
  641     Right _ -> Right ()
  642 
  643 -- | The stub's workspace root. A fixed absolute path so that relative
  644 -- seeded keys (e.g. @agents.md@) anchor under it and absolute escaping
  645 -- paths (e.g. @\/etc\/shadow@) are rejected by 'mkSafePathRemote'.
  646 stubWorkspaceRoot :: WorkspaceRoot
  647 stubWorkspaceRoot = WorkspaceRoot "/workspace"
  648 
  649 -- | Read a file from the in-memory stub. Resolves symlinks, re-checks
  650 -- containment, returns the (truncated) content.
  651 stubReadFile :: Map RemotePath StubEntry -> RemotePath -> Either WorkdirFsErr Text
  652 stubReadFile seed rp = case checkContainment rp of
  653   Left pe -> Left (WfsPath pe)
  654   Right _ -> case resolveSymlinkChain seed rp of
  655       Left e -> Left e
  656       Right (FileContent txt) ->
  657         let trimmed = T.dropWhileEnd Char.isSpace txt
  658         in if T.null (T.strip trimmed)
  659              then Left WfsNotFound
  660              else Right (truncateSection defaultSectionCharLimit trimmed)
  661       Right (Directory _) -> Left (WfsIo "is a directory")
  662       Right Missing -> Left WfsNotFound
  663       Right (SymlinkTarget _) -> Left (WfsIo "unresolved symlink")
  664 
  665 -- | Existence check for a file in the in-memory stub. Resolves symlinks.
  666 stubDoesFileExist :: Map RemotePath StubEntry -> RemotePath -> Bool
  667 stubDoesFileExist seed rp = case checkContainment rp of
  668   Left _ -> False
  669   Right _ -> case resolveSymlinkChain seed rp of
  670       Right (FileContent _) -> True
  671       _ -> False
  672 
  673 -- | Existence check for a directory in the in-memory stub. Does NOT
  674 -- resolve symlinks (a 'SymlinkTarget' is not a directory in the stub
  675 -- model).
  676 stubDoesDirectoryExist :: Map RemotePath StubEntry -> RemotePath -> Bool
  677 stubDoesDirectoryExist seed rp = case checkContainment rp of
  678   Left _ -> False
  679   Right _ -> case Map.lookup rp seed of
  680       Just (Directory _) -> True
  681       _ -> False
  682 
  683 -- | List the immediate children of a directory in the in-memory stub.
  684 -- Returns @Right []@ on a missing directory (fail-soft-to-empty).
  685 stubListDirectory :: Map RemotePath StubEntry -> RemotePath -> Either WorkdirFsErr [Text]
  686 stubListDirectory seed rp = case checkContainment rp of
  687   Left pe -> Left (WfsPath pe)
  688   Right _ -> case Map.lookup rp seed of
  689       Nothing -> Right []
  690       Just Missing -> Right []
  691       Just (Directory names) -> Right names
  692       Just (FileContent _) -> Left (WfsIo "not a directory")
  693       Just (SymlinkTarget _) -> Left (WfsIo "not a directory")
  694 
  695 -- | File size in the in-memory stub (byte length of the 'FileContent').
  696 stubFileSize :: Map RemotePath StubEntry -> RemotePath -> Either WorkdirFsErr Integer
  697 stubFileSize seed rp = case checkContainment rp of
  698   Left pe -> Left (WfsPath pe)
  699   Right _ -> case resolveSymlinkChain seed rp of
  700       Left e -> Left e
  701       Right (FileContent txt) -> Right (fromIntegral (T.length txt))
  702       Right (Directory _) -> Left (WfsIo "is a directory")
  703       Right Missing -> Left WfsNotFound
  704       Right (SymlinkTarget _) -> Left (WfsIo "unresolved symlink")
  705

  727 -- | The workspace root-relative path of a direct child of @base@
  728 -- (@""@ = the root itself).
  729 snapJoin :: Text -> Text -> Text
  730 snapJoin base name
  731   | T.null base = name
  732   | otherwise   = base <> "/" <> name
  733 
  734 -- | Does @path@ exist in the snapshot as a directory? Convention dirs are
  735 -- dot-dirs reached by DIRECT path (e.g. @@repo/.agents@@), so lookups are
  736 -- unfiltered.
  737 snapIsDirectoryAt :: WorkspaceSnapshot -> Text -> Bool
  738 snapIsDirectoryAt snap path =
  739   any (\e -> snapRelPath e == path && snapIsDir e) (snapEntries snap)
  740 
  741 -- | Does @path@ exist in the snapshot as a regular file?
  742 snapIsFileAt :: WorkspaceSnapshot -> Text -> Bool
  743 snapIsFileAt snap path =
  744   any (\e -> snapRelPath e == path && not (snapIsDir e)) (snapEntries snap)
  745 
  746 -- | The immediate child NAMES (both kinds) of @base@ whose basename is
  747 -- visible (non-hidden — parity with @ls -1@). Sorted.
  748 snapChildNamesAt :: WorkspaceSnapshot -> Text -> [Text]
  749 snapChildNamesAt snap base =
  750   sort
  751     [ name
  752     | e <- snapEntries snap
  753     , let rel = snapRelPath e
  754     , T.isPrefixOf prefix rel
  755     , T.length rel > T.length prefix
  756     , let name = T.drop (T.length prefix) rel
  757     , not (T.isInfixOf "/" name)
  758     , not ("." `T.isPrefixOf` name)
  759     ]
  760   where
  761     prefix = if T.null base then "" else base <> "/"
  762 
  763 -- | The immediate visible child DIRECTORIES of @base@, as full relative
  764 -- paths, sorted. Hidden children are excluded (parity with @ls -1@).
  765 snapChildDirsAt :: WorkspaceSnapshot -> Text -> [Text]
  766 snapChildDirsAt snap base =
  767   [ snapJoin base n | n <- snapChildNamesAt snap base, snapIsDirectoryAt snap (snapJoin base n) ]
  768 
  769 -- | The immediate visible child FILES of @base@, as full relative paths,
  770 -- sorted. Hidden children are excluded (parity with @ls -1@).
  771 snapChildFilesAt :: WorkspaceSnapshot -> Text -> [Text]
  772 snapChildFilesAt snap base =
  773   [ snapJoin base n | n <- snapChildNamesAt snap base, snapIsFileAt snap (snapJoin base n) ]
  774 
  775 -- | The top-level visible directories of the snapshot (the cloned repo
  776 -- names), sorted.
  777 snapTopDirs :: WorkspaceSnapshot -> [Text]
  778 snapTopDirs snap = snapChildDirsAt snap ""

    1 -- | Strict, resource-safe file-reading helpers built on @conduit@ + @resourcet@.
    2 --
    3 -- The lazy 'Data.ByteString.Lazy.readFile' / 'Data.Text.IO.readFile' helpers
    4 -- keep the file handle open until the lazy structure is fully consumed, which
    5 -- is GC-dependent. The 'Seal.Gateway.ListsSnapshot.buildListsSnapshot' hot
    6 -- path reads hundreds of small files per request (111 sessions × multiple
    7 -- files each) and the lazy handles pile up faster than the GC reclaims them,
    8 -- hitting the OS soft FD limit ('EMFILE' / @resource exhausted (Too many
    9 -- open files)@).
   10 --
   11 -- @conduit@'s 'sourceFile' opens the handle inside a 'ResourceT' scope, so the
   12 -- handle is released as soon as 'runConduitRes' returns — regardless of when
   13 -- (or whether) the resulting lazy 'ByteString' is forced. The bytes are
   14 -- materialized eagerly by 'sinkLazy' before the 'ResourceT' closes the handle,
   15 -- so the decode sees the full file with the fd already released.
   16 module Seal.Util.StrictIO
   17   ( readFileStrict
   18   , readFileTextStrict
   19   , decodeFileStrict
   20   ) where
   21 
   22 import Conduit ( runConduitRes, sinkLazy, sourceFile, (.|) )
   23 import Data.Aeson (FromJSON, decode)
   24 import Data.ByteString.Lazy qualified as BL
   25 import Data.Text (Text)
   26 import Data.Text.Encoding qualified as TE
   27 
   28 -- | Read a file strictly into a lazy 'ByteString', closing the underlying
   29 -- handle before the helper returns. The handle is acquired and released by
   30 -- the 'ResourceT' scope of 'runConduitRes'; the bytes are materialized by
   31 -- 'sinkLazy' before the scope exits. Callers 'doesFileExist'-guard the path
   32 -- first (the lazy variants they replace did the same).
   33 readFileStrict :: FilePath -> IO BL.ByteString
   34 readFileStrict path = runConduitRes (sourceFile path .| sinkLazy)
   35 
   36 -- | Read a file strictly as 'Text' by reading bytes strictly then decoding.
   37 -- The handle is closed before the decode runs.
   38 readFileTextStrict :: FilePath -> IO Text
   39 readFileTextStrict path = TE.decodeUtf8 . BL.toStrict <$> readFileStrict path
   40 
   41 -- | Read and decode a JSON file strictly. The handle is closed before the
   42 -- decode runs. Returns 'Nothing' on a decode failure (the caller is
   43 -- expected to 'doesFileExist'-guard the path first).
   44 decodeFileStrict :: FromJSON a => FilePath -> IO (Maybe a)
   45 decodeFileStrict path = decode <$> readFileStrict path

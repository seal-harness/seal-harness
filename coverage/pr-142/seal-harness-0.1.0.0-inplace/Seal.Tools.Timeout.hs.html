   35 -- | A timeout value in microseconds. The newtype carries the unit in the
   36 -- type to prevent micros/seconds confusion across the JSON boundary (the
   37 -- JSON @timeout@ field is in seconds; this is the internal representation).
   38 newtype Microseconds = Microseconds
   39   { unMicroseconds :: Int
   40   } deriving stock (Eq, Ord, Show)
   41 
   42 -- | Configuration for tool call timeout/retry behavior. Loaded from
   43 -- @config.toml@ under @[tool_timeout]@ (in 'Seal.Config.File.RuntimeConfig').
   44 data ToolTimeoutConfig = ToolTimeoutConfig
   45   { ttcDefaultSeconds  :: !Int    -- ^ default per-call timeout (default: 120)
   46   , ttcMaxSeconds      :: !Int    -- ^ hard cap (default: 600)
   47   , ttcRetryMax        :: !Int    -- ^ max retry attempts (default: 3)
   48   , ttcRetryBaseMicros :: !Int   -- ^ base delay between retries (default: 2_000_000 = 2s)
   49   , ttcRetryFactor     :: !Double -- ^ backoff multiplier (default: 2.0)
   50   , ttcKillGraceMicros :: !Int   -- ^ SIGTERM→SIGKILL grace period (default: 5_000_000 = 5s)
   51   , ttcMaxOutputBytes  :: !Int    -- ^ bounded output cap (default: 50_000)
   52   , ttcAbortPollMicros :: !Int   -- ^ abort-poll interval (default: 100_000 = 100ms)
   53   } deriving stock (Eq, Show)
   54 
   55 -- | The default config — matches the design doc defaults.
   56 defaultToolTimeoutConfig :: ToolTimeoutConfig
   57 defaultToolTimeoutConfig = ToolTimeoutConfig
   58   { ttcDefaultSeconds  = 120
   59   , ttcMaxSeconds      = 600
   60   , ttcRetryMax        = 3
   61   , ttcRetryBaseMicros = 2_000_000
   62   , ttcRetryFactor     = 2.0
   63   , ttcKillGraceMicros = 5_000_000
   64   , ttcMaxOutputBytes  = 50_000
   65   , ttcAbortPollMicros = 100_000
   66   }

   68 -- | The error type for the timeout/abort/retry lifecycle. Kept separate
   69 -- from 'Seal.ISA.Dispatch.DispatchError' (which is about the dispatch
   70 -- decision — op not found, denied, exec failed); 'ToolError' is about
   71 -- the execution lifecycle (timeout, abort, retry). The dispatcher maps
   72 -- 'ToolError' → 'DispatchError (ExecFailed (renderToolError ...))' at the
   73 -- boundary.
   74 data ToolError
   75   = ToolTimeout Int          -- ^ the timeout that was in effect (seconds, for display)
   76   | ToolAborted              -- ^ user/session cancellation
   77   | ToolIOError Text         -- ^ transient IO failure (error class only in audit log — see 'errorClass')
   78   | ToolRetriesExhausted ToolError  -- ^ wrapper after max retries
   79   deriving stock (Eq, Show)
   80 
   81 -- | Extract the per-call timeout from the tool input JSON. The @timeout@
   82 -- field is in SECONDS (the JSON wire unit; converted to microseconds
   83 -- internally). Validation:
   84 --
   85 --   * absent / null / non-integer (string, float, bool) → default
   86 --   * negative / zero → default
   87 --   * positive > max → clamp to max
   88 --   * positive in [1, max] → as-is (× 1_000_000)
   89 --
   90 -- The output is always in @[1_000_000, max * 1_000_000]@ (1s to max) — never
   91 -- negative, never zero (which would cause an instant-timeout DoS or a
   92 -- @threadDelay@ crash).
   93 extractPerCallTimeout :: Value -> ToolTimeoutConfig -> Microseconds
   94 extractPerCallTimeout input cfg =
   95   case parseMaybe (A.withObject "input" (A..: "timeout")) input of
   96     Nothing -> defaultMicros
   97     Just v -> case A.fromJSON @Int v of
   98       A.Error _ -> defaultMicros
   99       A.Success n
  100         | n <= 0    -> defaultMicros
  101         | n > maxS  -> maxMicros
  102         | otherwise -> Microseconds (n * 1_000_000)
  103   where
  104     maxS = ttcMaxSeconds cfg
  105     defaultMicros = Microseconds (ttcDefaultSeconds cfg * 1_000_000)
  106     maxMicros = Microseconds (maxS * 1_000_000)
  107 
  108 -- | Compute the retry delay: @base * factor^attempt@. Pure, overflow-safe
  109 -- (guards against Int overflow at large attempt counts). For @factor >= 1.0@
  110 -- the result is monotonically increasing in the attempt number.
  111 computeRetryDelay :: ToolTimeoutConfig -> Int -> Microseconds
  112 computeRetryDelay cfg attempt =
  113   let base = fromIntegral (ttcRetryBaseMicros cfg) :: Double
  114       factor = ttcRetryFactor cfg
  115       raw = base * factor ^ attempt
  116       -- Guard against Int overflow: clamp to maxBound if the Double exceeds it.
  117       clamped = if raw > fromIntegral (maxBound :: Int)
  118                 then maxBound
  119                 else round raw
  120   in Microseconds clamped
  121 
  122 -- | Whether to retry on a given 'ToolError'. Retry only on transient
  123 -- failures (timeout, IO error). Do NOT retry on abort (user cancelled —
  124 -- respect immediately) or on retries-exhausted (already exhausted).
  125 shouldRetry :: ToolError -> Bool
  126 shouldRetry (ToolTimeout _)        = True
  127 shouldRetry (ToolIOError _)         = True
  128 shouldRetry ToolAborted             = False
  129 shouldRetry (ToolRetriesExhausted _) = False
  130 
  131 -- | Render the error CLASS for the audit log. Returns ONLY the class
  132 -- string — never the full 'ToolIOError' Text payload (which could contain
  133 -- paths/host info from an IO error). The audit log ('orRecorded') carries
  134 -- this class string; the model-visible error text ('renderToolError') is
  135 -- separate.
  136 errorClass :: ToolError -> Text
  137 errorClass (ToolTimeout _)          = "timeout"
  138 errorClass ToolAborted              = "aborted"
  139 errorClass (ToolIOError _)          = "io"
  140 errorClass (ToolRetriesExhausted _) = "retries_exhausted"
  141 
  142 -- | Render a model-visible error message. The timeout value is rendered
  143 -- in seconds (not microseconds) so the model can reason about it. The
  144 -- message is actionable: the timeout case tells the model how to retry
  145 -- with a larger timeout.
  146 renderToolError :: OpName -> ToolError -> Text
  147 renderToolError (OpName name) = \case
  148   ToolTimeout s ->
  149     name <> " timed out after " <> T.pack (show s) <> "s. If this command is expected to take longer, retry with a larger timeout value."
  150   ToolAborted ->
  151     name <> " was aborted by the user."
  152   ToolIOError msg ->
  153     name <> " failed with an IO error: " <> msg
  154   ToolRetriesExhausted innerErr ->
  155     name <> " failed after " <> T.pack (show (ttcRetryMax defaultToolTimeoutConfig)) <> " retries (last error: " <> innerRender innerErr <> ")."
  156   where
  157     innerRender (ToolTimeout s)    = "timed out after " <> T.pack (show s) <> "s"
  158     innerRender (ToolIOError m)    = "IO error: " <> m
  159     innerRender ToolAborted        = "aborted"
  160     innerRender (ToolRetriesExhausted _) = "retries exhausted"

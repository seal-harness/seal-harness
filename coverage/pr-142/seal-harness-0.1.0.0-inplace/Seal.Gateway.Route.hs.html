   68 -- | The Seal Harness route GADT. One constructor per REST route. The type
   69 -- parameters are: @caps@ (path captures), @qrys@ (query params, always @'[]@
   70 -- here — all routes are queryless), @req@ (request body bodiedness), @resp@
   71 -- (response type).
   72 --
   73 -- Captures use 'Text' for unvalidated captures and the validated capture
   74 -- newtypes ('SessionIdOrErr', etc.) for validated ones. The handler
   75 -- pattern-matches via the 'withSessionId' / 'withAgentDefId' / etc. helpers
   76 -- (in 'Seal.Gateway.Route.Codec') to emit the exact legacy error messages.

  144 -- | The 'Meta' table — one arm per 'SealRoute' constructor. Uses @./@ for
  145 -- path building, @end@ for path termination, @capture@ for path captures.
  146 -- All routes are queryless (@qend@). Body-bearing routes use
  147 -- @body (one reqCodec)@ (a dummy 'BodyCodec' — the handler builds the
  148 -- 'Response' directly; trasa's response encoding is never invoked);
  149 -- bodyless routes use @bodyless@. Response codecs are @resp (one respCodec)@
  150 -- (also a dummy — see 'freeFormBodyCodec'). 'RouteHealth' uses
  151 -- 'jsonBodyCodec' since its response type is 'Text' (which has real
  152 -- 'ToJSON'/'FromJSON' instances).
  153 routeMeta :: SealRoute caps qrys req resp -> MetaCodec caps qrys req resp

  375 -- ---------------------------------------------------------------------------
  376 -- Free-form JSON schema + dummy JSON instances for the opaque body/response
  377 -- marker types (OpenAPI generation). The handlers build 'Value's directly, so
  378 -- these instances exist solely so 'Trasa.OpenApi.Codec.openApiJsonBody' can
  379 -- produce an 'OpenApiBodyCodec' for each route. The 'ToJSON'/'FromJSON'
  380 -- instances are never called by the OpenAPI generator (it only reads the
  381 -- schema + media types); they're total dummies that satisfy the class
  382 -- constraints. The 'ToSchema' instances render a free-form object schema
  383 -- (@{"type":"object","additionalProperties":true}@).
  384 -- ---------------------------------------------------------------------------
  385 
  386 freeFormObjectSchema :: Schema
  387 freeFormObjectSchema = mempty
  388   & type_ ?~ OpenApiTypeSingle OpenApiObject
  389   & additionalProperties ?~ AdditionalPropertiesAllowed True
  390 
  391 instance ToSchema Resp where
  392   declareNamedSchema _ = pure (NamedSchema Nothing freeFormObjectSchema)
  393 instance ToSchema Req where
  394   declareNamedSchema _ = pure (NamedSchema Nothing freeFormObjectSchema)
  395 instance ToSchema AnswerReq where
  396   declareNamedSchema _ = pure (NamedSchema Nothing freeFormObjectSchema)

  398 -- | A dummy 'BodyCodec' for the opaque marker types ('Resp', 'Req',
  399 -- 'AnswerReq'). The OpenAPI generator only reads the media types from
  400 -- 'bodyCodecNames' (the encode/decode functions are never called — the
  401 -- generator builds a schema document, not a server). The functions are total
  402 -- bottoms ('Left "unused") so they can't accidentally produce a value of the
  403 -- uninhabited 'Resp'/'Req' types.
  404 freeFormBodyCodec :: BodyCodec a
  405 freeFormBodyCodec = BodyCodec
  406   { bodyCodecNames = "application/json" :| []
  407   , bodyCodecEncode = const BL.empty
  408   , bodyCodecDecode = \_ -> Left "freeFormBodyCodec: unused (OpenAPI generation only)"
  409   }

    2 -- | 'UntrustedExecBackend' — a smart-constructed type that can ONLY be
    3 -- built from an 'Ssh' backend (spec §4). A local container/VM shares the
    4 -- harness kernel and does NOT count as remote (spec §2 sharpening), so
    5 -- 'Container' cannot produce this type either.

    7 -- This is the type the remote-only arm of 'selectUntrustedBackend' returns;
    8 -- the dispatcher's full-sum selection is 'selectExecBackend' (4a-T2), which
    9 -- also yields the 'EbLocal' arm. 'UntrustedExecBackend' is the
   10 -- type-guaranteed "this came from SSH" proof.

   25 -- | A backend that untrusted dispatch is allowed to run through. Smart-
   26 -- constructed so it can ONLY come from an 'Ssh' backend (spec §4). A local
   27 -- container/VM shares the harness kernel and does NOT count as remote
   28 -- (spec §2 sharpening), so 'Container' cannot produce this type either.
   29 newtype UntrustedExecBackend = UntrustedExecBackend SshConfig
   30   deriving stock (Eq, Show)
   31 
   32 -- | The only way to obtain an 'UntrustedExecBackend' is to hand in an
   33 -- 'Ssh' 'TerminalBackend'. Local/Tmux/Container all fail with the
   34 -- appropriate 'ExecError'.
   35 mkUntrustedExecBackend :: TerminalBackend -> Either ExecError UntrustedExecBackend
   36 mkUntrustedExecBackend = \case
   37   TbSsh cfg       -> Right (UntrustedExecBackend cfg)
   38   TbLocal         -> Left ExecLocalNotPermittedForUntrusted
   39   TbTmux _       -> Left ExecNotImplemented
   40   TbContainer _  -> Left ExecNotImplemented
   41 
   42 -- | The startup check for the @remote-only-untrusted@ Cabal flag (spec §3
   43 -- Layer B). When the flag is on (compiled in via CPP — see
   44 -- 'Seal.Tools.Exec.Local'\'s absence), a @mode=local@ value in the config
   45 -- is a startup configuration error: the local untrusted executor is
   46 -- absent from the binary, so @mode=local@ is unexecutable. Returns
   47 -- @Right ()@ when @mode=remote@; @Left err@ when @mode=local@.
   48 --
   49 -- This is a pure function so the startup check is testable without boot.
   50 enforceRemoteOnly :: UntrustedExecConfig -> Either Text ()
   51 enforceRemoteOnly cfg =
   52   case uecMode cfg of
   53     UemRemote -> Right ()
   54     UemLocal  -> Left "remote-only-untrusted build: untrusted_execution.mode=local is rejected (the local executor is absent from this binary)"
   55 
   56 -- | Recover the 'SshConfig' from an 'UntrustedExecBackend' (the remote
   57 -- executor in 4g consumes this).
   58 untrustedBackendSshConfig :: UntrustedExecBackend -> SshConfig
   59 untrustedBackendSshConfig (UntrustedExecBackend cfg) = cfg

   65 -- | The runtime mode (spec §3 Layer A): @local@ (default) or @remote@
   66 -- (fail-closed). Forced @remote@ under the @-f remote-only-untrusted@
   67 -- Cabal flag (4g-T2).
   68 data UntrustedExecMode = UemLocal | UemRemote
   69   deriving stock (Eq, Show)
   70 
   71 -- | The untrusted-execution configuration the pure select functions
   72 -- consume. Carries the 'mode' and the remote 'SshConfig' (if configured).
   73 -- The full TOML parsing lands in 4b-T2; 4a-T2 only needs the type.
   74 data UntrustedExecConfig = UntrustedExecConfig
   75   { uecMode   :: UntrustedExecMode
   76   , uecRemote :: Maybe SshConfig
   77   }
   78   deriving stock (Eq, Show)
   79 
   80 -- | The spec's @selectUntrustedBackend@, narrowed to the remote arm.
   81 -- Returns 'Right' ONLY when the backend is 'Ssh' AND a remote 'SshConfig'
   82 -- is configured AND the mode permits it. NEVER yields a local-capable
   83 -- backend (the 'UntrustedExecBackend' type is Ssh-only by construction).
   84 --
   85 -- @mode=local@ always returns 'Left ExecLocalNotPermittedForUntrusted'
   86 -- (this function is the remote-only selector; the full-sum selector is
   87 -- 'selectExecBackend').
   88 selectUntrustedBackend
   89   :: UntrustedExecConfig -> TerminalBackend -> Either ExecError UntrustedExecBackend
   90 selectUntrustedBackend cfg = \case
   91   TbSsh _        -> case uecRemote cfg of
   92                       Just _  -> case uecMode cfg of
   93                                    UemRemote -> mkUntrustedExecBackend (TbSsh (sshFromCfg cfg))
   94                                    UemLocal -> Left ExecLocalNotPermittedForUntrusted
   95                       Nothing -> Left ExecRemoteRequired
   96   TbLocal        -> Left ExecLocalNotPermittedForUntrusted
   97   TbTmux _       -> Left ExecNotImplemented
   98   TbContainer _  -> Left ExecNotImplemented
   99   where
  100     -- Recover the configured SshConfig. (Only reached when uecRemote is Just.)
  101     sshFromCfg c = case uecRemote c of
  102       Just s  -> s
  103       Nothing -> error "selectUntrustedBackend: unreachable (guarded by uecRemote check)"
  104 
  105 -- | The full-sum selector the dispatcher wires. Returns 'EbLocal' when
  106 -- @mode=local@ + 'TbLocal'; 'EbRemote' when @mode=remote@ + 'TbSsh'
  107 -- configured; 'Left' otherwise (fail-closed, no local fallback under
  108 -- @mode=remote@).
  109 selectExecBackend
  110   :: UntrustedExecConfig -> TerminalBackend -> Either ExecError ExecBackend
  111 selectExecBackend cfg = \case
  112   TbLocal        -> case uecMode cfg of
  113                       UemLocal  -> Right (EbLocal mkLocalExecHandlePlaceholder)
  114                       UemRemote -> Left ExecLocalNotPermittedForUntrusted
  115   TbSsh _        -> case (uecMode cfg, uecRemote cfg) of
  116                       (UemLocal, _)       -> Left ExecLocalNotPermittedForUntrusted
  117                       (UemRemote, Just s) -> Right (EbRemote s)
  118                       (UemRemote, Nothing) -> Left ExecRemoteRequired
  119   TbTmux _       -> Left ExecNotImplemented
  120   TbContainer _  -> Left ExecNotImplemented

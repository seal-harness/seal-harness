    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The Layer-1 terse-grammar routing front-end. @\/N@ switches focus to
    3 -- tab N, @\/N payload@ injects into tab N, a bare @\/tab@ shows the current
    4 -- tab, @\/new [args]@ starts a new tab with a fresh session, @\/<other>…@
    5 -- is deferred to the @\/@-command registry (including @\/tab
    6 -- \u003csubcommand\u003e@), anything else is plain text to the focused tab.
    7 -- The grammar is a first-class synopsis entry in @\/help@ so it's
    8 -- discoverable.
    9 module Seal.Routing.Route
   10   ( ParseError (..)
   11   , RoutingDecision (..)
   12   , route
   13   , terseSynopsis
   14   ) where
   15 
   16 import Data.Char (isDigit, isAsciiLower)
   17 import Data.Text (Text)
   18 import Data.Text qualified as T
   19 
   20 import Seal.Handles.Tab (TabIndex, tabIndexFromChar)
   21 import Seal.Tabs.Types (TabSlashCommand (..))
   22 
   23 -- | A parse error from the routing grammar.
   24 newtype ParseError = ParseError Text
   25   deriving stock (Eq, Show)
   26 
   27 -- | The Layer-1 routing decision.
   28 data RoutingDecision
   29   = Focus TabIndex                 -- ^ /N
   30   | Inject TabIndex Text           -- ^ /N payload
   31   | Plain Text                     -- ^ plain text to the focused tab
   32   | TabCommand TabSlashCommand     -- ^ /tab <subcommand> … (reserved; /tab
   33                                    --   is dispatched via the registry, but
   34                                    --   this variant is kept for future
   35                                    --   Layer-1 tab-command routing)
   36   | CurrentTab                     -- ^ bare /tab — show the current tab
   37   | NewSession Text                -- ^ /new [args] — start a new tab with a
   38                                    --   fresh session. The carried 'Text' is
   39                                    --   the raw argument string (everything
   40                                    --   after @/new @), empty when bare.
   41   | SlashCommand Text              -- ^ other /commands (deferred to the registry)
   42   deriving stock (Eq, Show)
   43 
   44 -- | Route one inbound line. The Layer-1 terse grammar:
   45 --
   46 -- * @\/N@          -> 'Focus' N (N is a single char 0-9a-z, at end-of-string
   47 --                   or followed by a space)
   48 -- * @\/N payload@  -> 'Inject' N payload
   49 -- * @\/tab@        -> 'CurrentTab' (show the current tab)
   50 -- * @\/new [args]@ -> 'NewSession' (start a new tab with a fresh session;
   51 --                   @args@ is the raw text after @/new @, carried for the
   52 --                   handler to parse optional @-p@\/@-m@\/@-r@ flags)
   53 -- * @\/<other>…@   -> 'SlashCommand' (deferred to the registry — this is
   54 --                   multi-char commands like @\/vault@, @\/help@, @\/ping@,
   55 --                   and @\/tab <subcommand>@)
   56 -- * anything else  -> 'Plain'
   57 --
   58 -- The disambiguator: a single tab-char @\/N@ is the tab grammar ONLY when N
   59 -- is alone (end-of-string or followed by a space). @\/vault@ (no space after
   60 -- @v@) is a 'SlashCommand', not @Inject v ault@. A @\/@ followed by a
   61 -- non-tab, non-tab-command char is a 'SlashCommand'.
   62 route :: Text -> Either ParseError RoutingDecision
   63 route t
   64   | T.null t             = Right (Plain t)
   65   | T.head t /= '/'      = Right (Plain t)
   66   | otherwise            =
   67       let rest = T.drop 1 t  -- drop the leading '/'
   68       in case T.uncons rest of
   69            Nothing -> Right (Plain "/")  -- a bare "/" — treat as plain
   70            Just (c, after)
   71               | isTabChar c && (T.null after || T.head after == ' ') ->
   72                   -- single-char /N or /N payload (the tab grammar)
   73                   case tabIndexFromChar c of
   74                     Left e -> Left (ParseError e)
   75                     Right idx -> Right (focusOrInject idx after)
   76                | rest == "tab" ->
   77                   Right CurrentTab
   78                | rest == "new" || T.isPrefixOf "new " rest ->
   79                   Right (NewSession (stripNewPrefix rest))
   80                | otherwise ->
   81                   Right (SlashCommand rest)
   82   where
   83     isTabChar c = isDigit c || isAsciiLower c
   84 
   85 -- | Extract the argument text from the @new@ prefix. @\"new\"@ yields
   86 -- @\"\"@; @\"new -p anthropic\"@ yields @\"-p anthropic\"@.
   87 stripNewPrefix :: Text -> Text
   88 stripNewPrefix rest =
   89   if rest == "new"
   90     then ""
   91     else T.drop 1 (snd (T.breakOn " " rest))  -- everything after the first space
   92 
   93 -- | Given a valid tab index + the text after it: if the rest is empty (or
   94 -- whitespace-only), it's a Focus; otherwise it's an Inject (the payload is
   95 -- the text after the first space, preserving internal spaces verbatim).
   96 focusOrInject :: TabIndex -> Text -> RoutingDecision
   97 focusOrInject idx after
   98   | T.null (T.strip after) = Focus idx
   99   | otherwise               =
  100       let payload = T.drop 1 (snd (T.breakOn " " after))  -- everything after the first space
  101       in Inject idx payload
  102 
  103 -- | The terse-grammar synopsis (for /help). One line.
  104 terseSynopsis :: Text
  105 terseSynopsis = "/N [payload]  Switch to tab N (0-9a-z), or inject payload into it"

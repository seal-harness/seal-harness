    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Static behavioral guidance blocks injected into the system prompt.
    3 -- Each block is a few sentences of operational guidance that improves
    4 -- model behavior across providers without being model-specific. The
    5 -- blocks are config-gated (see 'Seal.Config.File': @[agent]@
    6 -- @parallel_tool_guidance@, @tool_use_enforcement@,
    7 -- @task_completion_guidance@); each defaults to injected (true).
    8 --
    9 -- The blocks are appended /before/ the dynamic @\<available_skills\>@
   10 -- catalog so the stable text precedes the volatile catalog (cache-
   11 -- friendly ordering: stable identity prefix → static guidance →
   12 -- autoload body → available-skills catalog).
   13 module Seal.Agent.PromptParts
   14   ( parallelToolGuidance
   15   , toolUseEnforcement
   16   , taskCompletionGuidance
   17   , staticGuidanceBlock
   18   , injectStaticGuidance
   19   ) where
   20 
   21 import Data.Text (Text)
   22 import Data.Text qualified as T
   23 
   24 -- | Parallel tool-call guidance. Tells the model to batch independent
   25 -- tool calls into one turn rather than running them sequentially across
   26 -- turns. A real latency/token win with no risk.
   27 parallelToolGuidance :: Text
   28 parallelToolGuidance =
   29   "## Parallel tool calls\n\n\
   30   \When several tool calls are independent (no output of one is needed to \
   31   \call another), batch them into a single turn. Do not call them \
   32   \sequentially across turns when you could have called them together."
   33 
   34 -- | Tool-use enforcement. Tells the model to actually call tools instead
   35 -- of describing actions. Improves reliability on weaker models that tend
   36 -- to narrate instead of act.
   37 toolUseEnforcement :: Text
   38 toolUseEnforcement =
   39   "## Tool use\n\n\
   40   \When a task requires a tool, actually call the tool. Do not describe \
   41   \or narrate the action you would take (\"I would run X\") and then stop \
   42   \— call it. Do not answer from memory when a current tool call could \
   43   \get the real state."
   44 
   45 -- | Task-completion / anti-fabrication guidance. Tells the model not to
   46 -- stop after a stub and not to fabricate output when a path is blocked.
   47 taskCompletionGuidance :: Text
   48 taskCompletionGuidance =
   49   "## Task completion\n\n\
   50   \Do not stop after producing a stub, TODO, or skeleton and report the \
   51   \task done. Complete the implementation through to working, tested \
   52   \code. If a path is genuinely blocked (missing input, failed tool, \
   53   \permission denied), say so explicitly and ask for what you need — do \
   54   \not fabricate output to fill the gap."
   55 
   56 -- | Render the enabled guidance blocks as a single section, joined with
   57 -- blank lines. Returns the empty 'Text' when no block is enabled (so the
   58 -- caller can skip emitting anything). Each enabled block is a
   59 -- @## header@ + body, so the whole section reads as a sequence of
   60 -- short guidance notes.
   61 staticGuidanceBlock :: Bool -> Bool -> Bool -> Text
   62 staticGuidanceBlock parallel toolUse taskCompletion =
   63   let parts = [ parallelToolGuidance    | parallel ]
   64            <> [ toolUseEnforcement      | toolUse ]
   65            <> [ taskCompletionGuidance  | taskCompletion ]
   66   in if null parts then "" else T.intercalate "\n\n" parts
   67 
   68 -- | Append the enabled static guidance blocks to the resolved system
   69 -- prompt. The blocks go /before/ the auto-loaded skill body and the
   70 -- available-skills catalog (the caller runs this first, then
   71 -- 'Seal.Skills.Autoload.injectAutoloadSkill', then
   72 -- 'Seal.Skills.Prompt.injectAvailableSkills'). Returns the prompt
   73 -- unchanged when no block is enabled.
   74 injectStaticGuidance
   75   :: Bool    -- ^ parallel tool-call guidance
   76   -> Bool    -- ^ tool-use enforcement
   77   -> Bool    -- ^ task-completion guidance
   78   -> Maybe Text
   79   -> Maybe Text
   80 injectStaticGuidance parallel toolUse taskCompletion mPrompt =
   81   let block = staticGuidanceBlock parallel toolUse taskCompletion
   82   in if T.null block
   83        then mPrompt
   84        else Just (case mPrompt of
   85                     Nothing  -> block
   86                     Just base -> base <> "\n\n" <> block)

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The @/agent@ command group: list defined agent definitions, show one
    3 -- agent def's fields, and view or set the default agent. Agent defs are stored
    4 -- as structured records materialized from the Audited log into the in-memory
    5 -- 'AgentDefBackend'; this command reads that backend (no filesystem
    6 -- discovery). The default agent is persisted in @config.toml@ as
    7 -- @default_agent@.
    8 --
    9 -- Scope (mirrors pureclaw's @\/agent@ surface): discovery over definitions
   10 -- only. The running-agent lifecycle (@AGENT_START@\/@STOP@\/@STATUS@) is
   11 -- reachable via the ISA tool calls, not via slash commands.
   12 module Seal.Command.Agent
   13   ( agentCommandSpec
   14   , renderAgentLine
   15   , renderAgentInfo
   16   ) where

   35 -- | The @/agent@ command spec. Closes over the 'AgentDefBackend' (for
   36 -- list/info) and the config path (for default get/set).
   37 agentCommandSpec :: AgentDefBackend -> FilePath -> CommandSpec
   38 agentCommandSpec backend cfgPath = CommandSpec
   39   { csName         = CommandName "agent"
   40   , csAliases      = []
   41   , csGroup        = GroupAgent
   42   , csSynopsis     = "List agent defs, show one, or view/set the default agent"
   43   , csParserInfo   = agentParserInfo backend cfgPath
   44   , csAvailability = InteractiveOnly
   45   }
   46 
   47 agentParserInfo :: AgentDefBackend -> FilePath -> ParserInfo CommandAction
   48 agentParserInfo backend cfgPath =
   49   info (agentParser backend cfgPath <**> helper)
   50     (  progDesc "Inspect agent definitions and the default agent"
   51     <> header   "agent — list defs, show one, or view/set the default"
   52     )
   53 
   54 agentParser :: AgentDefBackend -> FilePath -> Parser CommandAction
   55 agentParser backend cfgPath = hsubparser
   56   (  command "list"
   57        (info (pure (listCmd backend))
   58              (progDesc "List all defined agent defs (id + name + provider/model)"))
   59   <> command "info"
   60        (info (infoCmd backend <$> agentArg)
   61              (progDesc "Show one agent def's fields"))
   62   <> command "default"
   63        (info (defaultCmd backend cfgPath <$> optional agentArg)
   64              (progDesc "View or set the default agent"))
   65   <> metavar "COMMAND"
   66   )
   67 
   68 -- | Required agent-def-id argument.
   69 agentArg :: Parser Text
   70 agentArg = T.pack <$> strArgument (metavar "AGENT" <> help "Agent def id (e.g. worker)")
   71 
   72 listCmd :: AgentDefBackend -> CommandAction
   73 listCmd backend = CommandAction $ \caps -> do
   74   defs <- adbList backend
   75   if null defs
   76     then ccSend caps "no agent defs defined"
   77     else mapM_ (ccSend caps . renderAgentLine) (sortOn adName defs)
   78 
   79 infoCmd :: AgentDefBackend -> Text -> CommandAction
   80 infoCmd backend raw = CommandAction $ \caps ->
   81   case mkAgentDefId raw of
   82     Left err -> ccSend caps err
   83     Right aid -> do
   84       mDef <- adbRead backend aid
   85       case mDef of
   86         Nothing -> ccSend caps ("agent def not found: " <> agentDefIdText aid)
   87         Just d  -> mapM_ (ccSend caps) (renderAgentInfo d)
   88 
   89 -- | @/agent default@ with no arg views; with an arg validates-then-persists.
   90 defaultCmd :: AgentDefBackend -> FilePath -> Maybe Text -> CommandAction
   91 defaultCmd backend cfgPath mArg = CommandAction $ \caps ->
   92   case mArg of
   93     Nothing -> do
   94       eCfg <- loadRuntimeConfig cfgPath
   95       let cur = either (const Nothing) rcDefaultAgent eCfg
   96       case cur of
   97         Nothing -> ccSend caps "no default agent set. Use /agent default <id> to set one."
   98         Just a  -> ccSend caps ("default agent: " <> a)
   99     Just raw ->
  100       case mkAgentDefId raw of
  101         Left err -> ccSend caps err
  102         Right aid -> do
  103           mDef <- adbRead backend aid
  104           case mDef of
  105             Nothing -> ccSend caps ("agent def not found: " <> agentDefIdText aid)
  106             Just _ -> do
  107               res <- updateRuntimeConfig cfgPath
  108                        (\fc -> fc { rcDefaultAgent = Just (agentDefIdText aid) })
  109               case res of
  110                 Left e   -> ccSend caps e
  111                 Right () -> ccSend caps ("default agent set to: " <> agentDefIdText aid)

  113 -- | One line per def for @/agent list@. Empty provider/model (e.g. a
  114 -- DirScheme agent with no @AGENTS.md@ frontmatter) render as @default@.
  115 renderAgentLine :: AgentDef -> Text
  116 renderAgentLine d =
  117   agentDefIdText (adId d) <> "  " <> adName d
  118     <> "  (" <> providerLabel <> "/" <> modelLabel <> ")"
  119   where
  120     ModelId modelName = adModel d
  121     providerLabel = if T.null (adProvider d) then "default" else adProvider d
  122     modelLabel = if T.null modelName then "default" else modelName
  123 
  124 -- | Multi-line detail for @/agent info@.
  125 renderAgentInfo :: AgentDef -> [Text]
  126 renderAgentInfo d =
  127   [ "id:          " <> agentDefIdText (adId d)
  128   , "name:        " <> adName d
  129   , "provider:    " <> adProvider d
  130   , "model:       " <> modelName
  131   , "system:      " <> fromMaybe "(none)" (adSystem d)
  132   , "tools:       " <> renderTools (adTools d)
  133   , "updated:     " <> T.pack (show (adUpdatedAt d))
  134   , "session:     " <> T.pack (show (adSession d))
  135   ]
  136   where
  137     ModelId modelName = adModel d
  138     renderTools AllowAll       = "all"
  139     renderTools (AllowOnly xs) =
  140       T.intercalate ", " [ t | OpName t <- Set.toList xs ]

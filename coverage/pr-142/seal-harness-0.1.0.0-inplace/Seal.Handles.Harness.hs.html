    4 -- | The harness capability handle — the seam the registry, reconcile loop,
    5 -- and (Phase 6b) the tab runtime use to drive one harness. A no-op handle
    6 -- backs the registry/reconcile tests so they need no real tmux. The
    7 -- sanitize helpers ('stripAnsi'/'stripControl') strip escape sequences and
    8 -- control bytes from captured screen content (a captured screen may carry
    9 -- escape sequences that could spoof the observer or the transcript).
   10 module Seal.Handles.Harness
   11   ( HarnessStatus (..)
   12   , HarnessError (..)
   13   , HarnessHandle (..)
   14   , noOpHarnessHandle
   15   , stripAnsi
   16   , stripControl
   17   ) where
   18

   25 -- | The liveness states a harness can be in.
   26 data HarnessStatus
   27   = HsIdle | HsThinking | HsAwaitingInput | HsExited | HsOrphaned
   28   deriving stock (Eq, Show, Generic)
   29   deriving anyclass (ToJSON, FromJSON)
   30 
   31 -- | The harness error type. Matched on to drive control flow (so it earns
   32 -- being a sum type per the haskell-coder skill).
   33 data HarnessError
   34   = HeSpawnFailed Text
   35   | HeCaptureFailed Text
   36   | HeStopFailed Text
   37   | HeNotFound Text
   38   | HeTmuxMissing
   39   deriving stock (Eq, Show)
   40 
   41 -- | The capability record of IO actions for driving one harness. Every
   42 -- field is an IO action so the type is uniform between real (tmux-backed)
   43 -- and fake (no-op) variants.
   44 data HarnessHandle = HarnessHandle
   45   { hhSend     :: Text -> IO (Either HarnessError ())
   46     -- ^ Send text (keystrokes) to the harness.
   47   , hhReceive  :: IO (Either HarnessError [Text])
   48     -- ^ Capture the current screen as sanitized lines.
   49   , hhSnapshot :: IO (Either HarnessError Text)
   50     -- ^ Full screen capture (sanitized).
   51   , hhStatus   :: IO (Either HarnessError HarnessStatus)
   52     -- ^ The current liveness.
   53   , hhStop     :: IO (Either HarnessError ())
   54     -- ^ Stop the harness (kill the tmux window).
   55   }
   56 
   57 -- | A no-op handle for tests: sends succeed, receive/snapshot return empty,
   58 -- status is HsIdle, stop succeeds.
   59 noOpHarnessHandle :: HarnessHandle
   60 noOpHarnessHandle = HarnessHandle
   61   { hhSend     = \_ -> pure (Right ())
   62   , hhReceive  = pure (Right [])
   63   , hhSnapshot = pure (Right "")
   64   , hhStatus   = pure (Right HsIdle)
   65   , hhStop     = pure (Right ())
   66   }

   68 -- | Strip ANSI escape sequences (CSI: @\\x1b[\\x30-\\x3f*@\\x20-\\x2f*@\\x40-\\x7e@)
   69 -- from a captured line. Pure.
   70 stripAnsi :: Text -> Text
   71 stripAnsi = T.unfoldr go
   72   where
   73     go s
   74       | T.null s = Nothing
   75       | T.head s == '\x1b' = case T.uncons s of
   76           Just ('\x1b', s1)
   77             | Just ('[', s2) <- T.uncons s1 ->
   78                 -- CSI: ESC [ 0x30-0x3f* 0x20-0x2f* 0x40-0x7e
   79                 let rest = dropCsi s2
   80                 in go rest
   81             | otherwise -> go s1  -- drop the ESC alone
   82           _ -> Nothing  -- unreachable: s is non-null
   83       | otherwise = Just (T.head s, T.tail s)
   84 
   85     -- Drop the parameter + intermediate bytes + the final byte.
   86     dropCsi s =
   87       let afterParams = T.dropWhile (\c -> c >= '\x30' && c <= '\x3f') s
   88           afterInter  = T.dropWhile (\c -> c >= '\x20' && c <= '\x2f') afterParams
   89       in case T.uncons afterInter of
   90            Just (_final, rest) -> rest
   91            Nothing             -> T.empty  -- incomplete CSI; drop everything
   92 
   93 -- | Strip ALL control characters (a superset of 'stripAnsi' — also removes
   94 -- NUL/BEL/BS/DEL etc., not just CSI sequences). Pure.
   95 stripControl :: Text -> Text
   96 stripControl = T.filter (not . isControl)

    1 -- | The reusable allow-list family used by sender allow-listing (channels)
    2 -- and opcode-exposure gating (agent defs). Extracted from
    3 -- 'Seal.Security.Policy' so the cross-channel layer can depend on this leaf
    4 -- without pulling in the whole security policy.
    5 module Seal.Core.AllowList
    6   ( AllowList (..)
    7   , isAllowed
    8   ) where
    9 
   10 import Data.Set (Set)
   11 import Data.Set qualified as Set
   12 
   13 -- | An allow-list: either admit everything ('AllowAll') or only the given
   14 -- set ('AllowOnly').
   15 data AllowList a = AllowAll | AllowOnly (Set a)
   16   deriving stock (Eq, Show)
   17 
   18 -- | Membership test.
   19 isAllowed :: Ord a => a -> AllowList a -> Bool
   20 isAllowed _ AllowAll      = True
   21 isAllowed x (AllowOnly s) = x `Set.member` s

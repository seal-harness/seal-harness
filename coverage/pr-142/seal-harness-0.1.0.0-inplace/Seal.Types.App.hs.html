   16 -- | The application monad: @ReaderT Env (KatipContextT IO)@, matching the
   17 -- source project's application-monad shape. The 'Katip' and 'KatipContext'
   18 -- instances are hand-written, delegating through the newtype to the underlying
   19 -- 'KatipContextT' layer.
   20 newtype App a = App { unApp :: ReaderT Env (KatipContextT IO) a }
   21   deriving newtype
   22     ( Functor, Applicative, Monad, MonadIO, MonadReader Env, MonadThrow )
   23 
   24 instance Katip App where
   25   getLogEnv = App (lift getLogEnv)
   26   localLogEnv f (App m) = App (localLogEnv f m)
   27 
   28 instance KatipContext App where
   29   getKatipContext = App (lift getKatipContext)
   30   localKatipContext f (App m) = App (localKatipContext f m)
   31   getKatipNamespace = App (lift getKatipNamespace)
   32   localKatipNamespace f (App m) = App (localKatipNamespace f m)
   33 
   34 -- | Run an 'App' action: use the 'SealLogger' from 'Env' to set up the katip
   35 -- context (LogEnv + LogContexts + Namespace), then run the 'ReaderT' against
   36 -- the given 'Env'. The logger is built once at startup via 'withSealLogger'
   37 -- and threaded through 'Env' so 'App'-level 'logMsg' calls and 'IO'-level
   38 -- 'logIO' calls reach the same scribe.
   39 runApp :: Env -> App a -> IO a
   40 runApp env (App m) =
   41   runKatipContextT (slLogEnv logger) (slContext logger)
   42     (slNamespace logger) (runReaderT m env)
   43   where
   44     logger = envLogger env

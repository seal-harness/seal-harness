   89 -- | Full @seal serve@ startup wiring. Mirrors 'Seal.Tui.runTui': paths →
   90 -- config → vault → session → backends → tabsH → broker → gateway + WS
   91 -- server.
   92 runServeMain :: AutonomyLevel -> SealLogger -> IO ()
   93 runServeMain autonomy logger = do
   94   paths <- getSealPaths
   95   ensureSealDirs paths
   96   -- Probe + sweep the persistent ssh-agent registry: GC dead agents from
   97   -- a crashed/killed seal process so their stale entries don't accumulate
   98   -- (#88). Live agents are reused (the key is still loaded).
   99   startupAgentRegH <- mkAgentRegistryHandle (sshAgentsDir paths)
  100   arProbeAndSweep startupAgentRegH
  101   -- GC the content-addressed agent-metadata snapshot cache: keep the
  102   -- newest N entries, drop the rest. Best-effort — failures are ignored.
  103   gcAgentMetaCache (agentMetaCacheDir paths) agentMetaCacheKeepN
  104   migrateSecurityConfig paths
  105   let cfgPath = configFilePath paths
  106   cfg <- loadRuntimeConfig cfgPath >>= \case
  107     Left err -> do
  108       logIO logger WarningS ("could not load config: " <> ls err)
  109       pure defaultRuntimeConfig
  110     Right c  -> pure c
  111   secCfg <- loadSecurityConfig (securityFilePath paths) >>= \case
  112     Left err -> do
  113       logIO logger WarningS ("could not load security config: " <> ls err)
  114       pure defaultSecurityConfig
  115     Right c  -> pure c
  116   mHandle <- tryOpenVault paths secCfg logger
  117   ref     <- newIORef mHandle
  118   let rt = VaultRuntime
  119             { vrPaths      = paths
  120             , vrConfigPath = cfgPath
  121             , vrHandleRef  = ref
  122             }
  123   mgr <- newTlsManager
  124   callCounter <- newIORef 0
  125   let pr = ProviderRuntime
  126             { prConfigPath  = cfgPath
  127             , prVault       = rt
  128             , prManager     = mgr
  129             , prCallCounter = callCounter
  130             }
  131       cfgRoot = spConfig paths
  132   ensureConfigRepo cfgRoot
  133   let repo = openConfigRepo cfgRoot
  134   backends <- newBackends cfgRoot repo
  135   -- W4: the source-control repo registry handle (closes over
  136   -- repos.toml). Built once at startup and threaded into ApiDeps for
  137   -- /api/repos CRUD. The handle's rrhList/rrhMutate re-read the file on
  138   -- each call so mutations from other processes are reflected.
  139   repoRegH <- mkRepoRegistryHandle (reposFilePath paths)
  140   -- W5: the /repo slash command's test seam. The ls-remote arm runs real
  141   -- git against repoCloneStateDir; the vault-list arm backs the /repo info
  142   -- non-blocking vault-key advisory. The vault handle (mHandle) may be
  143   -- 'Nothing' if the vault is not configured — in that case /repo test
  144   -- surfaces 'vault locked' (fail-closed) and /repo info skips the advisory.
  145   repoSeam <- mkRepoTestSeam rt repoRegH paths
  146   -- W5: persisting tab handle. Load the persisted tab list, drop tabs whose
  147   -- session.json is missing on disk (stale), and seed the TVar. Harness tabs
  148   -- (BoundHarness) are kept as-is; the periodic reconcile sweep (run later)
  149   -- re-resolves their liveness and marks missing windows 'orphaned'. No
  150   -- broadcast at boot (no WS client connected yet); the first subscriber's
  151   -- broadcast trigger refreshes the sidebar.
  152   let tabsPath = tabListPath paths
  153   tabsH <- newPersistingTabsHandle tabsPath
  154   mPersisted <- loadTabList tabsPath
  155   case mPersisted of
  156     Nothing -> pure ()
  157     Just tl -> do
  158       kept <- filterM (sessionTabExists paths) (tlTabs tl)
  159       seedTabsHandle tabsH (renumberTabs kept)
  160   -- Persisting cursor store: load + seed so existing Telegram/Signal
  161   -- conversations re-resolve to their prior sessions (carrying the user's
  162   -- /model use choice) after a seal serve restart. A cursor pointing at a
  163   -- session whose session.json is missing degrades gracefully —
  164   -- resolveTabSession returns Nothing and the loop mints a fresh session
  165   -- on the next message (exactly as if the cursor were absent). No
  166   -- boot-time stale sweep is needed.
  167   cursorsH <- newPersistingCursorStore (cursorMapPath paths)
  168   mCursors <- loadCursorMap (cursorMapPath paths)
  169   for_ mCursors (seedCursorStore cursorsH)
  170   reg     <- newHarnessRegistry
  171   tmuxR   <- mkRealTmuxRunner
  172   uiState <- newUiStateHandle paths
  173   askReply <- newAskReplyStore 0  -- 0 = block indefinitely (no timeout); a
  174                                  -- future phase may surface a configurable
  175                                  -- per-session timeout.
  176   approvals <- newApprovalCache
  177   -- Build an in-memory active session (NOT persisted to disk) so the
  178   -- active-session ref has valid provider/model fallbacks. The session
  179   -- only lands on disk when the user sends the first message (the web send
  180   -- handler writes the transcript to the session dir). This avoids
  181   -- polluting the sessions list with an empty session on every `seal serve`.
  182   sessionMeta <- initSessionMeta paths cfg (bAgentDefs backends)
  183   activeRef   <- newIORef sessionMeta
  184   broker <- newStreamBroker 1024
  185   -- Build the shared ChannelDeps early so the reply registry + write locks
  186   -- can be shared with the web send handler (SendDeps). The cursor store,
  187   -- reply registry, and write locks are created inside newChannelDeps.
  188   let loadCfg = fromRight defaultRuntimeConfig <$> loadRuntimeConfig cfgPath
  189       isRemoteExec = case untrustedExecConfigFromSecurity secCfg of
  190         Just uec -> uecMode uec == UemRemote
  191         Nothing  -> False
  192   chanDeps <- newChannelDeps
  193         paths rt repoRegH pr backends autonomy (Just broker)
  194         reg tmuxR (Just mgr) approvals loadCfg isRemoteExec tabsH logger cursorsH
  195   let sr = SessionRuntime
  196              { srPaths      = paths
  197              , srConfigPath = cfgPath
  198              , srActive     = activeRef
  199              }
  200       -- The /new command for the web: mints a fresh session, swaps srActive,
  201       -- inserts a new tab into the TabsHandle, swaps srActive to the new
  202       -- session, and returns the old sid. When -r/--repo is given, the
  203       -- repo is cloned via SETUP_REPO (same as the web "Set up repo").
  204       newDeps = NewDeps
  205         { ndPaths = paths
  206         , ndCfg = loadCfg
  207         , ndAgentDefs = backends
  208         , ndChannelLabel = "web"
  209         , ndOldMeta = readIORef activeRef
  210         , ndInsertTab = \_caps newMeta -> do
  211             oldMeta <- readIORef activeRef
  212             let oldSid = smId oldMeta
  213             _ <- insertTabH tabsH (BoundSession (smId newMeta)) KindProvider Nothing
  214             writeIORef activeRef newMeta
  215             pure oldSid
  216         , ndSetupRepo = Just (handleSetupRepo sendDeps)
  217         , ndRepoReg = Just repoRegH
  218         }
  219       -- The slash-command registry is built from the shared
  220       -- 'coreCommandSpecs' (the channel-agnostic core: vault, provider,
  221       -- session, model, agent, tab, stop, terseGrammar, new, repo) plus
  222       -- the web's call/skill placeholders. 'runSlash' rebuilds call/skill
  223       -- AND stop per-request via 'replaceCallSkillSpecs' so those commands
  224       -- target the request's explicit 'SessionId' (the web is
  225       -- multi-session; srActive is not authoritative). The placeholders
  226       -- here never run.
  227       placeholderDispatcher _ _ = pure (Left (OpNotFound (OpName "placeholder")))
  228       coreDeps = CoreCommandDeps
  229         { ccdVault       = rt
  230         , ccdProvider    = pr
  231         , ccdSession     = sr
  232         , ccdAgentDefs   = bAgentDefs backends
  233         , ccdCfgPath     = cfgPath
  234         , ccdPaths       = paths
  235         , ccdTabs        = tabsH
  236         , ccdTabCloseNotifier = mkTabCloseNotifier (cdCursors chanDeps) (cdReplies chanDeps)
  237         , ccdAbortReg    = cdAbortReg chanDeps
  238         , ccdStopWriter  = mkStopTranscriptWriter paths (Just broker)
  239         , ccdModelWriter = mkModelTranscriptWriter paths (Just broker)
  240         , ccdRepoReg     = repoRegH
  241         , ccdRepoSeam    = Just repoSeam
  242         }
  243       registry = mkRegistry
  244         ( coreCommandSpecs coreDeps
  245           <> [ skillCommandSpec (bSkills backends) placeholderDispatcher
  246              , callCommandSpec placeholderDispatcher
  247              , newCommandSpec newDeps
  248              ]
  249         )
  250       sendDeps = SendDeps
  251         { sdPaths      = paths
  252         , sdVault      = rt
  253         , sdRepoReg    = repoRegH
  254         , sdProvider   = pr
  255         , sdSession    = sr
  256         , sdBackends   = backends
  257         , sdConfigRepo = repo
  258         , sdPreprocess = emptyChain
  259         , sdRegistry   = registry
  260         , sdResolve    = resolveSessionProvider pr
  261         , sdAutonomy   = autonomy
  262         , sdBroker     = Just broker
  263         , sdHarnessRegistry = reg
  264         , sdTmuxRunner  = tmuxR
  265         , sdHttpManager = Just mgr
  266         , sdAskReply    = askReply
  267         , sdApprovals   = approvals
  268         , sdReplies     = cdReplies chanDeps
  269         , sdLocks       = cdLocks chanDeps
  270         , sdAbortReg    = cdAbortReg chanDeps
  271         , sdTabsHandle  = tabsH
  272         , sdLogger      = logger
  273         , sdIsRemote    = isRemoteExec
  274         , sdExecCache   = cdExecCache chanDeps
  275         , sdRemoteRunner = Nothing
  276           -- ^ ONE shared instance: turns (web + channels), /call dispatches,
  277           -- and GET /api/sessions/:id/agents all hit the same cache.
  278         , sdMkWorker    = Nothing
  279           -- ^ Production: always use the real 'buildWorker' →
  280           -- 'mkDelegateWorker' path. The 'sdMkWorker' seam is for gateway
  281           -- API integration tests only.
  282         }
  283   -- Build the gateway config (from the [gateway] section or the default)
  284   let gwCfg = maybe defaultGatewayConfig withGatewayDefaults (rcGateway cfg)
  285       deps = ApiDeps
  286         { adSessionRuntime  = sr
  287         , adTabsHandle      = tabsH
  288         , adHarnessRegistry = reg
  289         , adAdoptConsent    = Just CcWeb
  290         , adAgentDefs       = bAgentDefs backends
  291         , adSkills          = bSkills backends
  292         , adProviders       = do
  293             -- The configured-provider list is computed on each request so
  294             -- newly-added credentials are reflected without a restart. The
  295             -- vault handle is read from the same ref the commands use.
  296             mh <- readIORef (vrHandleRef rt)
  297             configuredProviders mh cfg
  298         , adUiState         = uiState
  299         , adSend            = Just sendDeps
  300         , adDefaultAgent    = rcDefaultAgent <$> loadCfg
  301         , adBroker          = Just broker
  302         , adTabCloseNotifier = mkTabCloseNotifier (cdCursors chanDeps) (cdReplies chanDeps)
  303         , adRepoRegistry     = repoRegH
  304         , adConfigRepo       = repo
  305         , adVault            = rt
  306         , adPaths            = paths
  307         , adWsPort           = gcWsPort gwCfg
  308         , adSecurityConfig   = secCfg
  309         , adMkSessionExec    = Nothing
  310         , adAbortReg         = cdAbortReg chanDeps
  311         }
  312   -- Start the WS stream server on the WS port.
  313   -- The Origin allowlist is the configured list PLUS origins derived from
  314   -- the HTTP server's host + port. A wildcard host (0.0.0.0) means "bind all
  315   -- interfaces" — the browser may reach the server via any of them, so we
  316   -- can't enumerate the allowed origins ahead of time. In that case, pass an
  317   -- empty allowlist to the WS guard, which triggers wildcard mode (accept
  318   -- any Origin) — overriding even the default loopback origin. For a specific
  319   -- host, derive the origin + prepend it to the configured list.
  320   let isWildcard = gcHost gwCfg == "0.0.0.0" || gcHost gwCfg == "::"
  321       httpOrigins = [ "http://" <> gcHost gwCfg <> ":" <> T.pack (show (gcPort gwCfg))
  322                     | not isWildcard ]
  323       origins = if isWildcard
  324                   then []  -- wildcard host → empty allowlist → Stream.hs accepts any
  325                   else httpOrigins <> gcAllowedOrigins gwCfg
  326       guard = StreamGuard { sgAllowedOrigins = origins, sgGlobalCap = 1024, sgTabsHandle = tabsH, sgPaths = paths }
  327   logIO logger InfoS ("WS stream server binding to " <> ls (gcHost gwCfg <> ":" <> T.pack (show (gcWsPort gwCfg))))
  328   _ <- forkIO (runStreamServer (gcHost gwCfg) (gcWsPort gwCfg) guard broker)
  329   -- Fork channel listeners for any configured channel. Each channel gets
  330   -- its own askReply store; the tab list is shared (passed by the
  331   -- listener). The listener runs the shared 'runChannelLoop' + 'plainTurn'
  332   -- so the agent loop is identical to the standalone modes.
  333   forkSignalListener chanDeps cfg registry
  334   forkTelegramListener chanDeps cfg registry
  335   -- Run the HTTP gateway (blocks). Fail-closed on non-loopback when
  336   -- mode=remote (design V6: prevents network access to the unauthenticated
  337   -- updateRuntimeConfig caller).
  338   let isRemote = case scUntrustedExec secCfg of
  339         Just uefc -> uefcMode uefc == "remote"
  340         Nothing   -> False
  341   runGateway gwCfg isRemote deps

  343 -- | Build the /repo command's 'RepoTestSeam' from the live vault runtime +
  344 -- paths. The 'VaultRuntime' holds the 'IORef' of the (maybe) live
  345 -- 'VaultHandle'; when the vault is unconfigured/locked ('Nothing' in the
  346 -- ref), @rtsLsRemote@ fails closed to 'CloneVaultError VaultLocked' (via
  347 -- 'Clone.resolveVaultHandle') and @rtsVaultList@ returns 'Left VaultLocked'
  348 -- (/repo info shows the locked advisory). Mirrors 'vaultGetByName' in
  349 -- 'Seal.ISA.Ops.Secret'.
  350 mkRepoTestSeam :: VaultRuntime -> RepoRegistryHandle -> SealPaths -> IO RepoTestSeam
  351 mkRepoTestSeam rt repoRegH paths = do
  352   agentRegH <- mkAgentRegistryHandle (sshAgentsDir paths)
  353   pure RepoTestSeam
  354     { rtsLsRemote  = \repo -> do
  355         let deps = Clone.CloneDeps
  356               { Clone.cdVault = rt
  357               , Clone.cdRepoReg = repoRegH
  358               , Clone.cdSshAgent = mkRealSshAgentHandle
  359               , Clone.cdAgentRegistry = agentRegH
  360               , Clone.cdPinnedKnownHosts = pinnedGithubKnownHosts
  361               , Clone.cdKeyfilesDir = repoKeysDir paths
  362               , Clone.cdIsRemote = False
  363               }
  364         lsRemoteRepo deps repo
  365   , rtsVaultList = do
  366       mh <- readIORef (vrHandleRef rt)
  367       case mh of
  368         Nothing -> pure (Left VaultLocked)
  369         Just vh -> vhList vh
  370   , rtsVaultPut = \k v -> do
  371       mh <- readIORef (vrHandleRef rt)
  372       case mh of
  373         Nothing -> pure (Left VaultLocked)
  374         Just vh -> vhPut vh k v
  375   }

  377 -- | Open the vault if both recipient and identity are configured. Mirrors
  378 -- 'Seal.Tui.tryOpenVault'; duplicated to keep this module standalone.
  379 tryOpenVault :: SealPaths -> SecurityConfig -> SealLogger -> IO (Maybe VaultHandle)
  380 tryOpenVault paths fcfg logger =
  381   case (scVaultRecipient fcfg, scVaultIdentity fcfg) of
  382     (Just _, Just _) ->
  383       resolveEncryptor fcfg >>= \case
  384         Left err -> do
  385           logIO logger WarningS ("vault not available: " <> ls (T.pack (show err)))
  386           pure Nothing
  387         Right enc -> do
  388           let vcfg = VaultConfig
  389                 { vcPath    = maybe (vaultFilePath paths) T.unpack (scVaultPath fcfg)
  390                 , vcKeyType = fromMaybe "x25519" (scVaultKeyType fcfg)
  391                 , vcUnlock  = parseUnlockMode (scVaultUnlock fcfg)
  392                 }
  393           Just <$> openVault vcfg enc
  394     _ -> pure Nothing

  400 -- | Does the session.json backing a 'BoundSession' tab exist on disk? Used
  401 -- at boot to drop stale tabs (the session was deleted out-of-band). Harness
  402 -- tabs (BoundHarness) are always kept (the reconcile sweep handles them).
  403 sessionTabExists :: SealPaths -> Tab -> IO Bool
  404 sessionTabExists paths t = case tRef t of
  405   BoundSession sid -> doesFileExist (sessionMetaPath paths sid)
  406   BoundHarness _  -> pure True

  408 -- | Renumber a tab list so slots stay contiguous 0..n-1 after dropping stale
  409 -- tabs (I1: the tab list invariant). The TVar's smart constructors would
  410 -- do this via 'removeTab', but here we renumber in bulk after a boot-time
  411 -- filter, so the indices are recomputed from position.
  412 renumberTabs :: [Tab] -> TabList
  413 renumberTabs ts = TabList (zipWith renumber [0..] ts)
  414   where
  415     renumber n t = t { tIndex = mkIdx n }
  416     mkIdx n = case mkTabIndex n of
  417       Right i -> i
  418       Left _  -> error ("renumberTabs: index out of range (unreachable, n=" <> show n <> ")")

  420 -- | Fork the Signal channel listener if @[signal]@ is configured. Resolves
  421 -- the config section, spawns the signal-cli transport, and runs the shared
  422 -- inbox-driven loop in a background thread. A missing/unresolved section
  423 -- is logged to stderr and skipped (not fatal — the gateway still starts).
  424 forkSignalListener :: ChannelDeps -> RuntimeConfig -> Registry -> IO ()
  425 forkSignalListener deps cfg registry =
  426   case resolveSignalConfig (rcSignal cfg) Nothing of
  427     Left _ -> pure ()  -- not configured; skip silently
  428     Right (account, chunkLimit, allow) -> do
  429       let accountLabel = Seal.Signal.Config.signalAccountText account
  430       eTransport <- mkRealSignalTransport accountLabel
  431       case eTransport of
  432         Left err -> logIO (cdLogger deps) WarningS ("seal serve: signal channel skipped: " <> ls err)
  433         Right transport -> do
  434           let tabsH = cdTabs deps
  435           askReply <- newAskReplyStore 0
  436           let withCh = withSignalChannel (allow, chunkLimit) account transport (cdLogger deps)
  437               plainHandler h = plainTurn deps h askReply
  438           _ <- forkIO (runChannelLoop deps withCh plainHandler registry emptyChain askReply tabsH Nothing Nothing)
  439           pure ()

  441 -- | Fork the Telegram channel listener if @[telegram]@ is configured.
  442 -- Resolves the config section, spawns the Bot API transport, registers the
  443 -- bot's slash-command menu with BotFather for auto-completion, and runs the
  444 -- shared inbox-driven loop in a background thread. A missing/unresolved
  445 -- section is logged to stderr and skipped.
  446 forkTelegramListener :: ChannelDeps -> RuntimeConfig -> Registry -> IO ()
  447 forkTelegramListener deps cfg registry = do
  448   -- Read the bot token from the vault (the wizard stores it there).
  449   mh <- readIORef (vrHandleRef (cdVault deps))
  450   mVaultToken <- case mh of
  451     Nothing -> pure Nothing
  452     Just vh -> do
  453       r <- vhGet vh Seal.Telegram.Config.telegramVaultKey
  454       pure $ case r of
  455         Right bs -> Just (TE.decodeUtf8 bs)
  456         Left _   -> Nothing
  457   case resolveTelegramConfig (rcTelegram cfg) mVaultToken of
  458     Left err
  459       | isJust (rcTelegram cfg) ->
  460           -- The [telegram] section is present but unresolved (e.g. the vault
  461           -- is locked / missing the token). Surface it so the channel isn't
  462           -- silently dropped on startup.
  463           logIO (cdLogger deps) WarningS ("seal serve: telegram channel skipped: " <> ls err)
  464       | otherwise -> pure ()  -- not configured; skip silently
  465     Right (token, chunkLimit, allow) -> do
  466       mgr <- newTlsManager
  467       transport <- mkRealTelegramTransport (Seal.Telegram.Config.telegramTokenText token) mgr
  468       -- Register the bot's slash-command menu with BotFather.
  469       tgSetCommands transport (Seal.Channels.Telegram.Commands.telegramBotCommands registry)
  470       let tabsH = cdTabs deps
  471       askReply <- newAskReplyStore 0
  472       let withCh = withTelegramChannel (allow, chunkLimit) transport (cdLogger deps)

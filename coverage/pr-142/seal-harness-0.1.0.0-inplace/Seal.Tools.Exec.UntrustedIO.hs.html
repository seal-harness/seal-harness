    3 -- | The unified capability handle for Untrusted opcodes. Every side-effecting
    4 -- operation an untrusted tool call can perform is a method on this type.
    5 -- Opcode modules never import 'System.Process', 'System.Directory', or
    6 -- 'System.Posix' — they call these methods. The constructor is NOT exported;
    7 -- the two smart constructors ('mkLocalUntrustedIO', 'mkRemoteUntrustedIO')
    8 -- are the only way to obtain one.
    9 --
   10 -- Security properties preserved by construction:
   11 --
   12 --   * Capability scoping (spec §4/§8): a Trusted opcode has no 'UntrustedIO'
   13 --     in scope — it cannot call any of these methods (compile error).
   14 --   * SafePath confinement: file methods take a 'RemotePath' (a
   15 --     workspace-relative path) and internally 'mkSafePath' /
   16 --     'mkSafePathForWrite' (local arm) or 'mkSafePathRemote' (remote arm).
   17 --     The caller never sees a raw 'FilePath'.
   18 --   * Bounded: read/write/search methods carry operator ceilings.
   19 --   * Validated argv: shell/bin/search methods take validated newtypes
   20 --     ('ShellCommand', 'BinName', 'BinArg', 'SearchPattern'), never raw
   21 --     'Text'.
   22 --
   23 -- The local arm is implemented via the existing 'System.Process' /
   24 -- 'System.Directory' code (lifted out of 'Seal.Tools.Exec.Local' and the
   25 -- opcode modules). The remote arm is implemented over SSH (Option A in the
   26 -- plan: file IO piped over the SSH channel's stdin/stdout; commands via
   27 -- the existing 'RemoteRunner'). SafePath is validated LOCALLY before any
   28 -- SSH call so a @..@ escape is rejected before the network is touched.

   90 -- | The unified capability handle. Each field is an IO action the opcode
   91 -- calls; the smart constructors wire the local or remote implementation.
   92 -- The constructor is NOT exported — only the smart constructors are.
   93 --
   94 -- File methods take a 'RemotePath' (a workspace-relative path). Internally
   95 -- each arm re-anchors the path against its own 'WorkspaceRoot' and runs
   96 -- the lexical + (local arm) canonical confinement check. The remote arm
   97 -- uses 'mkSafePathRemote' (lexical-only — the file lives on the remote
   98 -- machine, so no local 'canonicalizePath').
   99 data UntrustedIO = UntrustedIO
  100   { uioReadFile    :: RemotePath -> Int -> IO (Either UntrustedErr LineWindow)
  101     -- ^ Read a workspace-relative file as a 'LineWindow' (line-oriented,
  102     -- paged, bounded by the operator scan-byte ceiling). Returns the
  103     -- window (the opcode renders it via 'renderWindow') or a structured
  104     -- error. The path is validated + confined internally.
  105 
  106   , uioWriteFile   :: RemotePath -> Text -> WriteMode -> Int
  107                    -> IO (Either UntrustedErr Int)
  108     -- ^ Write or append content to a workspace-relative file. The
  109     -- @operatorWriteCeiling@ is the hard upper bound on bytes written per
  110     -- call; the capability rejects content above the ceiling with
  111     -- 'UeBounded' (the opcode may pre-check, but the capability is the
  112     -- authoritative bound). Returns bytes written. The path is validated
  113     -- + confined internally.
  114 
  115   , uioPatchFile   :: RemotePath -> Text -> IO (Either UntrustedErr ())
  116     -- ^ Apply a unified diff to a workspace-relative file. Read (cat) →
  117     -- apply the diff in-process (the pure 'applyUnifiedDiff') → write
  118     -- atomically (temp + rename on the target plane). Returns '()' or
  119     -- a structured error.
  120 
  121   , uioShellExec   :: ShellCommand -> Maybe RemotePath
  122                    -> IO (Either UntrustedErr Text)
  123     -- ^ Run a validated shell command (single arg to @/bin/sh -c@), with
  124     -- an optional SafePath-confined cwd. Returns stdout (+ exit
  125     -- annotation) or a structured error.
  126 
  127   , uioBinExec     :: BinName -> [BinArg] -> Maybe RemotePath
  128                    -> IO (Either UntrustedErr Text)
  129     -- ^ Run a named binary (no shell, fixed argv) with an optional cwd.
  130     -- A relative 'RemotePath' is SafePath-confined to the workspace root;
  131     -- an absolute 'RemotePath' is passed through verbatim (not confined).
  132     -- Returns stdout or error.
  133 
  134   , uioProcessList :: IO (Either UntrustedErr Text)
  135     -- ^ List processes on the untrusted plane (bounded output).
  136 
  137   , uioProcessKill :: Int -> IO (Either UntrustedErr ())
  138     -- ^ Kill a process by PID (validated positive integer) on the
  139     -- untrusted plane.
  140 
  141   , uioSearchFiles :: SearchPattern -> Maybe RemotePath -> Int
  142                    -> IO (Either UntrustedErr Text)
  143     -- ^ Search workspace files for a pattern (@rg -n -- <pattern>
  144     -- <path>@). The path defaults to the workspace root. The result
  145     -- count is bounded by the operator ceiling. Returns matching lines
  146     -- or error.
  147 
  148   , uioShellExecEnv :: [(String, String)] -> ShellCommand -> Maybe RemotePath
  149                    -> IO (Either UntrustedErr Text)
  150     -- ^ Like 'uioShellExec' but with env overrides MERGED over the
  151     -- inherited environment (local arm) / @env VAR=val@-prefixed into the
  152     -- command string (remote arm — portable: @ssh -A@ only forwards the
  153     -- agent socket; @SendEnv@/@SetEnv@ need server @AcceptEnv@ which is
  154     -- @none@ by default). Used by the git-credential opcodes to inject
  155     -- @SSH_AUTH_SOCK@ / @GIT_SSH_COMMAND@ for deploy-key ops (the
  156     -- forwarded agent + pinned @known_hosts@, never the key bytes).
  157 
  158   , uioShellExecGitEnv :: [(String, String)] -> Maybe BS.ByteString
  159                    -> ShellCommand -> Maybe RemotePath
  160                    -> IO (Either UntrustedErr Text)
  161     -- ^ Like 'uioShellExecEnv' but with an optional @known_hosts@
  162     -- content payload for the REMOTE deploy-key path. When the content
  163     -- is 'Just' the remote arm: (1) writes it to a remote temp file via
  164     -- @ssh ... -- tee <path>@, (2) rewrites @GIT_SSH_COMMAND@ in the

  166     -- @ssh -A@ (agent forwarding) so the remote @git@ can sign via the
  167     -- forwarded @SSH_AUTH_SOCK@, (4) strips @SSH_AUTH_SOCK@ /

  170     -- local arm ignores the content (the local temp file is already
  171     -- written by 'resolveCloneTarget') + delegates to
  172     -- 'uioShellExecEnv'. Used by 'cloneWithCredential' + 'runGitCommand'

  177                  -> Maybe RemotePath
  178                  -> IO (Either UntrustedErr Text)
  179     -- ^ Like 'uioBinExec' but with env overrides (same merge strategy as

  193     -- is @git@ and the cwd is inside a registered repo with a deploy-key
  194     -- credential.

  202 -- | The structured error an 'UntrustedIO' method returns. Wraps the
  203 -- path-error, the bounded-overflow, and the executor-layer errors so the
  204 -- opcode can render a single, consistent error message.
  205 data UntrustedErr
  206   = UePath PathError
  207     -- ^ SafePath confinement failed (blocked name, @..@ escape, missing
  208     -- parent, symlink escape).
  209   | UeBounded Int
  210     -- ^ The call exceeded the operator ceiling (write/read/search). The
  211     -- 'Int' is the ceiling for the error message.
  212   | UeExec ExecError
  213     -- ^ The executor failed (remote unreachable, host-key mismatch, not
  214     -- implemented).
  215   | UeIo Text
  216     -- ^ An otherwise-uncategorized IO error (caught 'IOException',
  217     -- rendered as 'Text').
  218   deriving stock (Eq, Show)
  219 
  220 -- | Render an 'UntrustedErr' as a human-readable 'Text' for the opcode's
  221 -- 'orParts' (the model-visible error message).
  222 renderUntrustedErr :: UntrustedErr -> Text
  223 renderUntrustedErr = \case
  224   UePath pe   -> "path error: " <> T.pack (show pe)
  225   UeBounded n -> "content exceeds operator ceiling (" <> T.pack (show n) <> " bytes)"
  226   UeExec ee   -> "exec error: " <> T.pack (show ee)
  227   UeIo msg    -> "io error: " <> msg

  233 -- | The local untrusted executor. Workspace files live on the local FS;
  234 -- commands run via @/bin/sh -c@ (shell) or @proc@ (bin). Absent under
  235 -- @-f remote-only-untrusted@.
  236 mkLocalUntrustedIO :: WorkspaceRoot -> UntrustedIO
  237 mkLocalUntrustedIO wsRoot =
  238   -- The workdir root itself, resolved once as the SafePath for @.@. This
  239   -- is the single source of truth for the \"untrusted opcodes share one
  240   -- workspace root\" invariant: 'uioShellExec' / 'uioBinExec' default to
  241   -- it when the caller omits a cwd, so every opcode agrees on the same
  242   -- root without any opcode needing to remember to pass it.
  243   let wsRootPath = case mkSafePathRemote wsRoot "." of
  244         Right sp -> getSafePath sp
  245         Left _   -> "/nonexistent-workdir-fail-closed"
  246   in UntrustedIO
  247   { uioReadFile = \rp scanBytes -> do
  248       let rel = T.unpack (getRemotePath rp)
  249       eSafe <- mkSafePath wsRoot rel
  250       case eSafe of
  251         Left pe  -> pure (Left (UePath pe))
  252         Right sp -> do
  253           -- Bounded read: read at most scanBytes from the file, decode to
  254           -- Text, split to lines, build a LineWindow covering all read
  255           -- lines. The 'lwTruncated' flag is set when the file exceeds the
  256           -- scan ceiling (the read stopped before EOF). This matches the
  257           -- remote arm's 'lineWindowFromText' — both arms return the same
  258           -- shape, and the opcode does the offset/limit windowing purely.
  259           eContent <- try (readBoundedLocal (getSafePath sp) scanBytes)
  260                         :: IO (Either IOException (BS.ByteString, Bool))
  261           pure $ case eContent of
  262             Left ioErr   -> Left (UeIo (T.pack (show ioErr)))
  263             Right (bs, truncated) ->
  264               -- When truncated, the trailing partial line (no newline) is
  265               -- NOT counted as a complete line — matching the original
  266               -- 'readLineWindow' semantics (lwTotal is a lower bound). When
  267               -- not truncated, 'T.lines' counts a trailing partial line
  268               -- (matching Data.Text.lines).
  269               let txt = TE.decodeUtf8Lenient bs
  270                   rawLines = T.lines txt
  271                   -- Drop the trailing partial line if truncated AND it has
  272                   -- no newline (i.e. it's a partial). T.lines splits on \n;
  273                   -- a trailing partial line is one where the original bytes
  274                   -- didn't end with \n.
  275                   endsWithNl = not (BS.null bs) && BS.last bs == 0x0A
  276                   ls = if truncated && not endsWithNl
  277                          then init rawLines   -- drop the partial
  278                          else rawLines
  279                   n = length ls
  280               in Right LineWindow
  281                    { lwLines     = ls
  282                    , lwStart     = 0
  283                    , lwEnd       = n
  284                    , lwTotal     = n
  285                    , lwHasMore   = truncated
  286                    , lwTruncated = truncated
  287                    }
  288   , uioWriteFile = \rp content mode ceiling' -> do
  289       let rel       = T.unpack (getRemotePath rp)
  290           byteCount = BS.length (TE.encodeUtf8 content)
  291       if byteCount > ceiling'
  292         then pure (Left (UeBounded ceiling'))
  293         else do
  294           eSafe <- mkSafePathForWrite wsRoot rel
  295           case eSafe of
  296             Left pe  -> pure (Left (UePath pe))
  297             Right sp -> do
  298               eUnit <- try (writeLocal sp content mode) :: IO (Either IOException ())
  299               pure $ case eUnit of
  300                 Left ioErr -> Left (UeIo (T.pack (show ioErr)))
  301                 Right _    -> Right byteCount
  302   , uioPatchFile = \rp patch -> do
  303       let rel = T.unpack (getRemotePath rp)
  304       eSafe <- mkSafePath wsRoot rel
  305       case eSafe of
  306         Left pe  -> pure (Left (UePath pe))
  307         Right sp -> patchLocal sp patch
  308   , uioShellExec = \cmd mCwd ->
  309       let argv = ["/bin/sh", "-c", T.unpack (textShellCommand cmd)]
  310       in case mCwd of
  311            Nothing -> runLocalFixedArgv False argv (Just wsRootPath)
  312            Just rp -> case mkSafePathRemote wsRoot (T.unpack (getRemotePath rp)) of
  313              Left pe  -> pure (Left (UePath pe))
  314              Right sp -> runLocalFixedArgv False argv (Just (getSafePath sp))
  315   , uioBinExec = \bin bargs mCwd ->
  316       let argv = T.unpack (textBinName bin) : map (T.unpack . textBinArg) bargs
  317       in case resolveBinCwd wsRoot mCwd of
  318            Left pe       -> pure (Left (UePath pe))
  319            Right cwdAbs -> runLocalFixedArgv True argv (Just cwdAbs)
  320   , uioProcessList =
  321       case mkShellCommand "ps -o pid=,cmd=" of
  322         Left _   -> pure (Left (UeExec ExecNotImplemented))
  323         Right sh -> uioShellExec (mkLocalUntrustedIO wsRoot) sh Nothing
  324   , uioProcessKill = \pid ->
  325       case mkShellCommand ("kill " <> T.pack (show pid)) of
  326         Left _   -> pure (Left (UeExec ExecNotImplemented))
  327         Right sh -> do
  328           res <- uioShellExec (mkLocalUntrustedIO wsRoot) sh Nothing
  329           pure (const (Right ()) =<< res)
  330   , uioSearchFiles = \pat mPath _limit -> do
  331       let mRel = maybe "" (T.unpack . getRemotePath) mPath
  332       case mkSafePathRemote wsRoot mRel of
  333         Left pe  -> pure (Left (UePath pe))
  334         Right sp -> case mkShellCommand (buildRgCmd pat (Just sp)) of
  335           Left _   -> pure (Left (UeExec ExecNotImplemented))
  336           Right sh -> uioShellExec (mkLocalUntrustedIO wsRoot) sh Nothing
  337   , uioShellExecEnv = \extras cmd mCwd ->
  338       let argv = ["/bin/sh", "-c", T.unpack (textShellCommand cmd)]
  339       in case mCwd of
  340            Nothing -> runLocalFixedArgvEnv False argv (Just wsRootPath) extras
  341            Just rp -> case mkSafePathRemote wsRoot (T.unpack (getRemotePath rp)) of
  342              Left pe  -> pure (Left (UePath pe))
  343              Right sp -> runLocalFixedArgvEnv False argv (Just (getSafePath sp)) extras
  344   , uioShellExecGitEnv = \extras _knownHosts cmd mCwd ->
  345       -- Local arm: the known_hosts temp file is already written by
  346       -- resolveCloneTarget (cdIsRemote=False path); ignore the content
  347       -- + delegate to uioShellExecEnv.
  348       uioShellExecEnv (mkLocalUntrustedIO wsRoot) extras cmd mCwd
  349   , uioBinExecEnv = \extras bin bargs mCwd ->
  350       let argv = T.unpack (textBinName bin) : map (T.unpack . textBinArg) bargs
  351       in case resolveBinCwd wsRoot mCwd of
  352            Left pe       -> pure (Left (UePath pe))
  353            Right cwdAbs -> runLocalFixedArgvEnv True argv (Just cwdAbs) extras
  354   , uioBinExecGitEnv = \extras _mKnownHosts bin bargs mCwd ->
  355       -- Local arm: the known_hosts temp file is already written by
  356       -- resolveCloneTarget (cdIsRemote=False path); ignore the content
  357       -- + delegate to uioBinExecEnv.
  358       uioBinExecEnv (mkLocalUntrustedIO wsRoot) extras bin bargs mCwd
  359   }

  361 -- | Build the @rg@ command string from a validated 'SearchPattern' + an
  362 -- optional workspace-relative path (anchored to the workspace root, not
  363 -- the remote user's home CWD). Defaults to the workspace root itself.
  364 -- Both the pattern and the path are 'shellQuote'-d so a pattern
  365 -- containing spaces (e.g. @Recent Sessions@) or a single quote is passed
  366 -- to @rg@ as a single argv token, not word-split by the shell.
  367 buildRgCmd :: SearchPattern -> Maybe SafePath -> Text
  368 buildRgCmd pat mSafePath =
  369   T.pack ("rg -n -- " <> shellQuote (T.unpack (textSearchPattern pat))
  370           <> " " <> shellQuote (maybe "." getSafePath mSafePath))

  372 -- | Read at most @maxBytes@ from a local file. Returns the bytes read and
  373 -- a 'Bool' indicating whether the file was truncated (the file is larger
  374 -- than @maxBytes@ — the read stopped at the ceiling).
  375 readBoundedLocal :: FilePath -> Int -> IO (BS.ByteString, Bool)
  376 readBoundedLocal path maxBytes = do
  377   bs <- BS.readFile path
  378   let len = BS.length bs
  379   if len <= maxBytes
  380     then pure (bs, False)
  381     else pure (BS.take maxBytes bs, True)

  383 -- | Write content to the local FS via the validated 'SafePath'. The temp
  384 -- file + rename is used for @'WMWrite'@ so the write is atomic (a crash
  385 -- mid-write leaves the old file intact); @'WMAppend'@ appends directly
  386 -- (the file already exists or is created fresh by 'BS.appendFile').
  387 writeLocal :: SafePath -> Text -> WriteMode -> IO ()
  388 writeLocal sp content mode =
  389   let path  = getSafePath sp
  390       bytes = TE.encodeUtf8 content
  391   in case mode of
  392        WMAppend -> BS.appendFile path bytes
  393        WMWrite  -> do
  394          let tmp = path <> ".seal-write-tmp"
  395          BS.writeFile tmp bytes
  396          renameFile tmp path

  398 -- | Apply a unified diff to the local file at the validated 'SafePath':
  399 -- read → apply in-process → atomic temp+rename write.
  400 patchLocal :: SafePath -> Text -> IO (Either UntrustedErr ())
  401 patchLocal sp patch = do
  402   eContent <- try (BS.readFile (getSafePath sp)) :: IO (Either IOException BS.ByteString)
  403   case eContent of
  404     Left ioErr -> pure (Left (UeIo (T.pack (show ioErr))))
  405     Right content -> case applyUnifiedDiff (TE.decodeUtf8Lenient content) patch of
  406       Left applyErr -> pure (Left (UeIo applyErr))
  407       Right newContent -> do
  408         let path = getSafePath sp
  409             tmp  = path <> ".seal-patch-tmp"
  410         eUnit <- try (BS.writeFile tmp (TE.encodeUtf8 newContent) >> renameFile tmp path)
  411                    :: IO (Either IOException ())
  412         pure $ case eUnit of
  413           Left ioErr -> Left (UeIo (T.pack (show ioErr)))
  414           Right _    -> Right ()

  416 -- | Run a fixed-argv program locally, capturing stdout + stderr as Text.
  417 -- @treat127AsMissing@: a 127 exit maps to 'Left ExecNotImplemented'
  418 -- (the binary is not on PATH) when True; otherwise 127 is a normal
  419 -- command-not-found failure, returned via 'Right' with the exit code
  420 -- annotation. Any 'IOException' becomes 'Left ExecNotImplemented'.
  421 -- No env overrides — delegates to 'runLocalFixedArgvEnv' with @[]@.
  422 runLocalFixedArgv
  423   :: Bool -> [String] -> Maybe String -> IO (Either UntrustedErr Text)
  424 runLocalFixedArgv treat127AsMissing argv mCwd =
  425   runLocalFixedArgvEnv treat127AsMissing argv mCwd []

  427 -- | Run a fixed-argv program locally with env overrides. The @extras@
  428 -- are MERGED over the inherited environment ('getEnvironment'): the
  429 -- process inherits the harness env (PATH, HOME, etc.) plus the overrides
  430 -- (e.g. @SSH_AUTH_SOCK@, @GIT_SSH_COMMAND@, @GIT_TERMINAL_PROMPT=0@). This
  431 -- is the local-arm env-override seam (W2) — the secret-free auth env for
  432 -- deploy-key ops. The overrides are NEVER secret bytes (only paths +
  433 -- non-secret values); the secret lives only in the forwarded agent's
  434 -- memory.
  435 runLocalFixedArgvEnv
  436   :: Bool -> [String] -> Maybe String -> [(String, String)]
  437   -> IO (Either UntrustedErr Text)
  438 runLocalFixedArgvEnv treat127AsMissing argv mCwd extras = do
  439   let (program, args) = case argv of
  440         (p : as) -> (p, as)
  441         []       -> error "runLocalFixedArgvEnv: empty argv (unreachable)"
  442   logExecDebug "[local]" argv mCwd extras
  443   inherited <- getEnvironment
  444   let env' = Just (mergeEnv inherited extras)
  445       cp = (proc program args)
  446               { std_in = NoStream, std_out = CreatePipe, std_err = CreatePipe
  447               , cwd = mCwd, env = env'
  448               , create_group = True
  449               }
  450       -- ^ @create_group = True@ puts the child in its own POSIX process
  451       -- group so 'withManagedProcess' can kill the whole group on cleanup
  452       -- (SIGTERM → grace → SIGKILL). This prevents orphans when the
  453       -- dispatch wrapper cancels the Haskell worker thread.
  454       maxOutput = 50_000  -- bounded output cap (Task 4, matches ttcMaxOutputBytes)
  455   res <- try @IOException
  456          (withManagedProcess cp $ \ph _mIn hOut hErr -> do
  457             out <- readBounded hOut maxOutput
  458             err <- readBounded hErr maxOutput
  459             ec  <- waitForProcess ph
  460             pure (ec, out, err))
  461   pure $ case res of
  462     Left _ioErr                     -> Left (UeExec ExecNotImplemented)
  463     Right (ExitSuccess, out, _)    -> Right out
  464     Right (ExitFailure 127, _, _)
  465       | treat127AsMissing           -> Left (UeExec ExecNotImplemented)
  466     Right (ExitFailure n, out, err) -> Right (formatExitResult n out err)

  468 -- | Merge env overrides over the inherited environment. The @overrides@
  469 -- WIN on collision (per-op auth env must shadow any ambient value — e.g.
  470 -- a user's ambient @SSH_AUTH_SOCK@ must NOT override the per-op agent's
  471 -- socket, or git would sign with the user's ambient agent keys, breaking
  472 -- the per-op scoping invariant in design §4.6). The result is sorted by
  473 -- key for a deterministic order (mainly for test assertions —
  474 -- @CreateProcess.env@ is order-insensitive but a stable order makes the
  475 -- recording-fake + no-disk tests' diffs readable).
  476 --
  477 -- Implementation: @Map.union left right@ keeps @left@ on collision; putting
  478 -- @overrides@ as @left@ makes overrides win. Mirrors
  479 -- 'Seal.Git.Repo.readProcessBinaryCwdEnv' (which uses
  480 -- @Map.union (Map.fromList extras) (Map.fromList inherited)@ — same
  481 -- extras-win semantics).

  483 mergeEnv inherited overrides =
  484   Map.toList (Map.union (Map.fromList overrides) (Map.fromList inherited))

  486 -- | Format a non-zero exit result for the tool-call consumer. Combines

  509 -- | Replace the values of keys in 'secretEnvKeys' with @"<redacted>"@,
  510 -- preserving the key (so the reader sees that an env override was

  517 -- | Emit a debug-level log line showing the exact command the UntrustedIO
  518 -- executor is about to run. Covers both the local arm (a fixed-argv
  519 -- subprocess) and the remote arm (an @ssh@ argv, including @-A@ for
  520 -- agent-forwarding git ops). The @tag@ prefixes the line so the reader
  521 -- can distinguish local vs remote vs remote-with-agent at a glance:
  522 --
  523 --   * @"[local]"@        — a local subprocess (@runLocalFixedArgvEnv@)
  524 --   * @"[remote ssh]"@   — a remote SSH exec (no agent forwarding)
  525 --   * @"[remote ssh -A]"@ — a remote SSH exec with agent forwarding
  526 --                          (deploy-key git ops)
  527 --
  528 -- For the remote arm, the full SSH argv is logged (e.g.
  529 -- @ssh -o StrictHostKeyChecking=yes ... agent@exec.internal -- cd ... && git pull@),
  530 -- so the complete command — including the SSH portion — is visible when
  531 -- @mode=remote@. Env overrides (e.g. @SSH_AUTH_SOCK=...@,
  532 -- @GIT_SSH_COMMAND=...@) are shown as @KEY=VAL@ tokens. The cwd is shown
  533 -- as @cd <path> &&@ when set.
  534 --
  535 -- Secret-bearing env keys (see 'secretEnvKeys') are redacted via
  536 -- 'redactEnv' before the @KEY=VAL@ tokens are rendered, so a value like
  537 -- @GH_TOKEN=ghp_secret...@ appears as @GH_TOKEN=<redacted>@ in the log.
  538 -- This is a defense-in-depth seam on 'logExecDebug' itself, not caller
  539 -- discipline — the redaction fires regardless of which arm produced the
  540 -- call (the remote arm passes the secret extras separately via
  541 -- 'runRemoteShellTextEnv' so the same redaction path covers it).
  542 logExecDebug :: String -> [String] -> Maybe String -> [(String, String)] -> IO ()
  543 logExecDebug tag argv mCwd extras =
  544   globalLogIO DebugS (K.ls msg)
  545   where
  546     msg = T.pack (unwords (filter (not . null) parts))
  547     parts =
  548       [ tag ]
  549       <> envPart
  550       <> cwdPart
  551       <> [ unwords (map shellQuoteArgv argv) ]
  552     -- Secret values NEVER reach the log: 'redactEnv' replaces the values
  553     -- of 'secretEnvKeys' with <redacted> before rendering. Non-secret
  554     -- keys (SSH_AUTH_SOCK, GIT_TERMINAL_PROMPT, ...) pass through so
  555     -- operators can still diagnose credential-injection plumbing.
  556     envPart = case redactEnv extras of
  557       [] -> []
  558       xs -> [ unwords (map (\(k, v) -> k <> "=" <> v) xs) ]
  559     cwdPart = case mCwd of
  560       Just c  -> ["cd " <> c <> " &&"]
  561       Nothing -> []
  562     -- Quote an argv token for log readability: wrap in single quotes if
  563     -- it contains spaces or shell metacharacters. This is display-only
  564     -- (the actual subprocess argv is passed verbatim, never shell-parsed).
  565     shellQuoteArgv s
  566       | any (\c -> c == ' ' || c == '\'' || c == '"' || c == '$' || c == '`') s = "'" <> s <> "'"
  567       | otherwise = s

  573 -- | The remote SSH executor. Workspace files live on the remote machine;
  574 -- commands run via the SSH transport. File IO is implemented over SSH
  575 -- (Option A in the plan: content piped over the SSH channel's
  576 -- stdin/stdout). The 'SshConfig' is the validated, host-key-pinned config;
  577 -- the 'RemoteRunner' is the existing SSH transport.

  581 -- | Same as 'mkRemoteUntrustedIO' (the two names document the same
  582 -- constructor). Provided for the wiring site, which threads the runner
  583 -- explicitly.
  584 mkRemoteUntrustedIOFromRunner :: SshConfig -> RemoteRunner -> UntrustedIO
  585 mkRemoteUntrustedIOFromRunner sshCfg runner =
  586   -- The workdir root itself, resolved once as the SafePath for @.@. This
  587   -- is the single source of truth for the \"untrusted opcodes share one
  588   -- workspace root\" invariant on the remote arm: 'uioShellExec' /
  589   -- 'uioBinExec' default to it when the caller omits a cwd, so every
  590   -- opcode agrees on the same root (the remote @scWorkspace@), not the
  591   -- remote user's home or wherever the SSH server drops it.
  592   let wsRootPath = case mkSafePathRemote (wsRootFromCfg sshCfg) "." of
  593         Right sp -> getSafePath sp
  594         Left _   -> "/nonexistent-workdir-fail-closed"
  595   in UntrustedIO
  596   { uioReadFile = \rp scanBytes ->
  597       let rel = T.unpack (getRemotePath rp)
  598       in case mkSafePathRemote (wsRootFromCfg sshCfg) rel of
  599            Left pe  -> pure (Left (UePath pe))
  600            Right sp -> do
  601              -- ssh ... -- head -c <scanBytes> <abspath>  (bounded read).
  602              -- The SafePath is the workspace-anchored absolute path on the
  603              -- remote machine; the SSH command reads from that path, not
  604              -- the remote user's home CWD.
  605              let absPath = getSafePath sp
  606                  cmd = T.pack ("head -c " <> show scanBytes <> " " <> shellQuote absPath)
  607              res <- runRemoteShellText runner sshCfg cmd
  608              pure (Right . lineWindowFromText =<< res)
  609   , uioWriteFile = \rp content mode _ceiling' -> do
  610       let rel       = T.unpack (getRemotePath rp)
  611           byteCount = BS.length (TE.encodeUtf8 content)
  612       case mkSafePathRemote (wsRootFromCfg sshCfg) rel of
  613         Left pe  -> pure (Left (UePath pe))
  614         Right sp -> do
  615           -- ssh ... -- tee [-a] <abspath>   with content on stdin.
  616           -- The SafePath is the workspace-anchored absolute path.
  617           let absPath  = getSafePath sp
  618               teeFlag = case mode of WMWrite -> "" ; WMAppend -> "-a "
  619               cmd  = T.pack ("tee " <> teeFlag <> shellQuote absPath)
  620               argv = sshExecArgv sshCfg cmd
  621           res <- runRemoteStdin runner argv (TE.encodeUtf8 content)
  622           pure (either (Left . UeExec) (const (Right byteCount)) res)
  623   , uioPatchFile = \rp patch -> do
  624       let rel = T.unpack (getRemotePath rp)
  625       case mkSafePathRemote (wsRootFromCfg sshCfg) rel of
  626         Left pe  -> pure (Left (UePath pe))
  627         Right sp -> do
  628           let absPath = getSafePath sp
  629           -- Read remote (cat) → apply diff in-process → write remote via
  630           -- a single SSH exec with stdin. The patched content is piped to
  631           -- a remote sh -c that writes the temp + mv (atomic).
  632           let readCmd = T.pack ("cat " <> shellQuote absPath)
  633           rRead <- runRemoteShellText runner sshCfg readCmd
  634           case rRead of
  635             Left e       -> pure (Left e)
  636             Right oldTxt -> case applyUnifiedDiff oldTxt patch of
  637               Left applyErr -> pure (Left (UeIo applyErr))
  638               Right newContent -> do
  639                 let tmpPath = absPath <> ".seal-patch-tmp"
  640                     remoteSh = T.pack
  641                       ("sh -c 'cat > " <> shellQuote tmpPath
  642                        <> " && mv " <> shellQuote tmpPath
  643                        <> " " <> shellQuote absPath <> "'")
  644                     argv = sshExecArgv sshCfg remoteSh
  645                 res <- runRemoteStdin runner argv (TE.encodeUtf8 newContent)
  646                 pure (either (Left . UeExec) (const (Right ())) res)
  647   , uioShellExec = \cmd mCwd ->
  648       let prefixCd p = "cd " <> shellQuote p <> " && "
  649       in case mCwd of
  650         Nothing -> runRemoteShellText runner sshCfg
  651                      (T.pack (prefixCd wsRootPath
  652                               <> T.unpack (textShellCommand cmd)))
  653         Just rp ->
  654           case mkSafePathRemote (wsRootFromCfg sshCfg) (T.unpack (getRemotePath rp)) of
  655             Left pe  -> pure (Left (UePath pe))
  656             Right sp ->
  657               let cdCmd = "cd " <> shellQuote (getSafePath sp)
  658                           <> " && " <> T.unpack (textShellCommand cmd)
  659               in runRemoteShellText runner sshCfg (T.pack cdCmd)
  660   , uioBinExec = \bin bargs mCwd ->
  661       let argv' = T.unpack (textBinName bin) : map (T.unpack . textBinArg) bargs
  662       in case resolveBinCwd (wsRootFromCfg sshCfg) mCwd of
  663            Left pe  -> pure (Left (UePath pe))
  664            Right cwdPath ->
  665              let cmd = T.pack ("cd " <> shellQuote cwdPath <> " && "
  666                              <> T.unpack (T.intercalate " " (map (T.pack . shellQuote) argv')))
  667              in runRemoteShellText runner sshCfg cmd
  668   , uioProcessList =
  669       runRemoteShellText runner sshCfg "ps -o pid=,cmd="
  670    , uioProcessKill = \pid -> do
  671        res <- runRemoteShellText runner sshCfg ("kill " <> T.pack (show pid))
  672        pure (const (Right ()) =<< res)
  673    , uioSearchFiles = \pat mPath _limit ->
  674        let mRel = maybe "" (T.unpack . getRemotePath) mPath
  675        in case mkSafePathRemote (wsRootFromCfg sshCfg) mRel of
  676             Left pe  -> pure (Left (UePath pe))
  677             Right sp -> runRemoteShellText runner sshCfg (buildRgCmd pat (Just sp))
  678     , uioShellExecEnv = \extras cmd mCwd ->
  679        let prefixCd p = "cd " <> shellQuote p <> " && "
  680        in case mCwd of
  681          Nothing -> runRemoteShellTextEnv runner sshCfg
  682                       (T.pack (prefixCd wsRootPath)) extras
  683                       (T.pack (T.unpack (textShellCommand cmd)))
  684          Just rp ->
  685            case mkSafePathRemote (wsRootFromCfg sshCfg) (T.unpack (getRemotePath rp)) of
  686              Left pe  -> pure (Left (UePath pe))
  687              Right sp ->
  688                runRemoteShellTextEnv runner sshCfg
  689                  (T.pack ("cd " <> shellQuote (getSafePath sp) <> " && ")) extras
  690                  (T.pack (T.unpack (textShellCommand cmd)))
  691     , uioShellExecGitEnv = \extras _mKnownHosts cmd mCwd ->
  692         -- Remote deploy-key path (simple approach): ssh -A forwards the
  693         -- SEAL agent to the remote machine. The remote git clone uses the
  694         -- default ssh + the forwarded agent + the remote machine's default
  695         -- known_hosts. No GIT_SSH_COMMAND, no remote known_hosts file.
  696         -- Strip SSH_AUTH_SOCK/SSH_AGENT_PID from the REMOTE env prefix
  697         -- (the forwarded agent replaces them). Keep them in the LOCAL
  698         -- ssh -A process env so the right agent is forwarded.
  699         let localEnv = [ (k, v) | (k, v) <- extras
  700                       , k == "SSH_AUTH_SOCK" || k == "SSH_AGENT_PID"
  701                       ]
  702             remoteExtras = [ (k, v) | (k, v) <- extras
  703                            , k /= "SSH_AUTH_SOCK"
  704                            , k /= "SSH_AGENT_PID"
  705                            ]
  706             remoteEnvPrefix = T.pack (renderEnvPrefix remoteExtras)
  707         in case mCwd of
  708           Nothing ->
  709             let prefixCd = "cd " <> shellQuote wsRootPath <> " && "
  710                 fullCmd = T.pack prefixCd <> remoteEnvPrefix <> T.pack (T.unpack (textShellCommand cmd))
  711             in runRemoteShellForwardingEnvText runner sshCfg localEnv fullCmd
  712           Just rp ->
  713             case mkSafePathRemote (wsRootFromCfg sshCfg) (T.unpack (getRemotePath rp)) of
  714               Left pe  -> pure (Left (UePath pe))
  715               Right sp ->
  716                 let cdCmd = "cd " <> shellQuote (getSafePath sp) <> " && "
  717                     fullCmd = T.pack cdCmd <> remoteEnvPrefix <> T.pack (T.unpack (textShellCommand cmd))
  718                 in runRemoteShellForwardingEnvText runner sshCfg localEnv fullCmd
  719      , uioBinExecEnv = \extras bin bargs mCwd ->
  720          let argv' = T.unpack (textBinName bin) : map (T.unpack . textBinArg) bargs
  721          in case resolveBinCwd (wsRootFromCfg sshCfg) mCwd of
  722             Left pe  -> pure (Left (UePath pe))
  723             Right cwdPath ->
  724               let cdPart = T.pack ("cd " <> shellQuote cwdPath <> " && ")
  725                   cmd = T.pack (T.unpack (T.intercalate " " (map (T.pack . shellQuote) argv')))
  726               in runRemoteShellTextEnv runner sshCfg cdPart extras cmd
  727     , uioBinExecGitEnv = \extras _mKnownHosts bin bargs mCwd ->
  728         -- Remote deploy-key path: mirrors uioShellExecGitEnv. Strip
  729         -- SSH_AUTH_SOCK/SSH_AGENT_PID from the REMOTE env prefix (the
  730         -- forwarded agent replaces them). Keep them in the LOCAL ssh -A
  731         -- process env so the right agent is forwarded.
  732         let localEnv = [ (k, v) | (k, v) <- extras
  733                       , k == "SSH_AUTH_SOCK" || k == "SSH_AGENT_PID"
  734                       ]
  735             remoteExtras = [ (k, v) | (k, v) <- extras
  736                            , k /= "SSH_AUTH_SOCK"
  737                            , k /= "SSH_AGENT_PID"
  738                            ]
  739             argv' = T.unpack (textBinName bin) : map (T.unpack . textBinArg) bargs
  740             remoteEnvPrefix = T.pack (renderEnvPrefix remoteExtras)
  741         in case resolveBinCwd (wsRootFromCfg sshCfg) mCwd of
  742            Left pe  -> pure (Left (UePath pe))
  743            Right cwdPath ->
  744              let cdPart = T.pack ("cd " <> shellQuote cwdPath <> " && ")
  745                  cmd = T.pack (T.unpack (T.intercalate " " (map (T.pack . shellQuote) argv')))
  746              in runRemoteShellForwardingEnvText runner sshCfg localEnv (cdPart <> remoteEnvPrefix <> cmd)
  747    }

  769 -- | The workspace root for remote confinement. The 'SshConfig' carries

  771 -- 'WorkspaceRoot' for 'mkSafePathRemote'.
  772 wsRootFromCfg :: SshConfig -> WorkspaceRoot

  775 -- | Resolve an optional cwd ('RemotePath') for BIN_EXEC into an absolute
  776 -- 'FilePath' the executor can pass to the subprocess. Unlike
  777 -- 'uioShellExec' (which confines ALL paths — relative and absolute —
  778 -- under the workspace root via 'mkSafePathRemote'), BIN_EXEC treats an
  779 -- ABSOLUTE cwd as a pass-through (the operator explicitly chose an
  780 -- out-of-workspace directory); a RELATIVE cwd is SafePath-confined under
  781 -- the workspace root (the default untrusted-confinement contract).
  782 --
  783 --   * 'Nothing' → the workspace root itself (the one-root invariant).
  784 --   * 'Just rp' (absolute) → the path verbatim (no confinement).
  785 --   * 'Just rp' (relative) → 'mkSafePathRemote'-confined under the root;
  786 --     a @..@ escape or blocked name is rejected with 'Left'.
  787 resolveBinCwd :: WorkspaceRoot -> Maybe RemotePath -> Either PathError FilePath
  788 resolveBinCwd wsRoot mCwd =
  789   case mCwd of
  790     Nothing   -> Right wsRootPath
  791     Just rp ->
  792       let rel = T.unpack (getRemotePath rp)
  793       in if isAbsolute rel
  794            then Right rel
  795            else getSafePath <$> mkSafePathRemote wsRoot rel
  796   where
  797     wsRootPath = case mkSafePathRemote wsRoot "." of
  798       Right sp -> getSafePath sp
  799       Left _   -> "/nonexistent-workdir-fail-closed"

  801 -- | Single-quote a 'String' for the remote shell (the path is already
  802 -- SafePath-validated, but quoting is defense-in-depth against any
  803 -- metacharacters the validator permits, e.g. spaces). Embedded single
  804 -- quotes are escaped with the standard @'\''@ idiom so a value containing
  805 -- a quote cannot break out of its single-quoted argv token — e.g.
  806 -- @foo'bar@ becomes @'foo'\''bar'@, which the shell parses back to the
  807 -- literal @foo'bar@ as a single token. Applied uniformly to all remote
  808 -- paths and search patterns.
  809 shellQuote :: String -> String
  810 shellQuote s = "'" <> go s <> "'"
  811   where
  812     go []         = []
  813     go ('\'':rest) = "'\\''" <> go rest
  814     go (c:rest)    = c : go rest
  815 
  816 -- | Render the env-override prefix for the remote arm: @env VAR='val'
  817 -- VAR2='val2' @ (each value single-quoted via 'shellQuote' so a value
  818 -- containing spaces / quotes / @$()@ is passed as a single token — the
  819 -- @env@ command sets them before the command runs). The @env@ prefix is
  820 -- portable (@ssh -A@ only forwards the agent socket; @SendEnv@/@SetEnv@
  821 -- need server @AcceptEnv@ which is @none@ by default — design §4.4). The
  822 -- caller appends the command after the prefix.
  823 renderEnvPrefix :: [(String, String)] -> String
  824 renderEnvPrefix extras =
  825   if null extras
  826     then ""
  827     else "env " <> unwords [ k <> "=" <> shellQuote v | (k, v) <- extras ] <> " "

  829 -- | Smart-construct a 'ShellCommand', lifting a parse failure into an
  830 -- 'UntrustedErr' (defensive — the inputs are already validated, so this
  831 -- only fires on a NUL that slipped through). Returns 'Left' so the caller
  832 -- can short-circuit; the message is uniform across the remote arm.
  833 shellCmd :: Text -> Either UntrustedErr ShellCommand
  834 shellCmd t = case mkShellCommand t of
  835   Left _  -> Left (UeExec ExecNotImplemented)
  836   Right c -> Right c

  838 -- | Like 'runRemoteShellForwardingText' but with local env overrides
  839 -- (MERGED over the inherited environment). Used by the remote deploy-key
  840 -- clone path so the local @ssh -A@ process uses the SEAL agent's
  841 -- @SSH_AUTH_SOCK@ (not the ambient one) — the forwarded agent then has

  843 runRemoteShellForwardingEnvText
  844   :: RemoteRunner -> SshConfig -> [(String, String)] -> Text
  845   -> IO (Either UntrustedErr Text)
  846 runRemoteShellForwardingEnvText runner cfg localEnv cmdText =

  848     Left e    -> pure (Left e)
  849     Right cmd -> do
  850       let argv = sshExecArgvForwarding cfg (textShellCommand cmd)
  851       logExecDebug "[remote ssh -A]" argv Nothing localEnv
  852       res <- runRemoteEnv runner localEnv argv
  853       pure (either (Left . UeExec) Right res)

  855 -- | Run a remote shell command built from a 'Text' command string. The
  856 -- 'shellCmd' parse failure is lifted to 'UeExec'; the 'runRemoteShell'
  857 -- 'ExecError' is lifted to 'UeExec'. Uniform across the remote arm.
  858 runRemoteShellText
  859   :: RemoteRunner -> SshConfig -> Text -> IO (Either UntrustedErr Text)
  860 runRemoteShellText runner cfg cmdText =
  861   case shellCmd cmdText of
  862     Left e    -> pure (Left e)
  863     Right cmd -> do
  864       let argv = sshExecArgv cfg (textShellCommand cmd)
  865       logExecDebug "[remote ssh]" argv Nothing []
  866       res <- runRemoteShell runner cfg cmd
  867       pure (either (Left . UeExec) Right res)

  869 -- | Like 'runRemoteShellText' but with env overrides applied to the
  870 -- SUBPROCESS. The caller supplies the @cd@ prefix and the command SEPARATELY;
  871 -- the composed remote command is @\<cdPart\>env K='V' … \<cmd\>@ so the vars
  872 -- scope to the trailing command — NOT to the @cd@.
  873 --
  874 -- Placement is security-relevant (the PAT-clone bug): a POSIX shell scopes a
  875 -- leading @env@ assignment only to the single command that follows it, so an
  876 -- env prefix BEFORE @\<cd\> && \<cmd\>@ binds nothing for @\<cmd\>@. This
  877 -- mirrors the deploy-key path ('uioShellExecGitEnv'), which composes
  878 -- @cd … && env … git …@ for the same reason.
  879 --
  880 -- The 'logExecDebug' call receives the command WITHOUT the env prefix and
  881 -- the extras SEPARATELY so 'redactEnv' can redact secret-bearing keys before
  882 -- rendering (design §3.7): the logged message must not contain any value
  883 -- from a key in 'secretEnvKeys'.
  884 runRemoteShellTextEnv
  885   :: RemoteRunner -> SshConfig -> Text -> [(String, String)] -> Text
  886   -> IO (Either UntrustedErr Text)
  887 runRemoteShellTextEnv runner cfg cdPart extras cmd =
  888   let envPrefix = T.pack (renderEnvPrefix extras)
  889       fullCmdText = cdPart <> envPrefix <> cmd
  890       logCmdText = cdPart <> cmd
  891   in case shellCmd fullCmdText of
  892        Left e     -> pure (Left e)
  893        Right full -> do
  894          logCmd <- case shellCmd logCmdText of
  895            Left _   -> pure full
  896            Right l  -> pure l
  897          let argv = sshExecArgv cfg (textShellCommand logCmd)
  898          logExecDebug "[remote ssh]" argv Nothing extras
  899          res <- runRemoteShell runner cfg full
  900          pure (either (Left . UeExec) Right res)

  907 -- | Apply a minimal unified diff to the original content. Returns
  908 -- @Left errMsg@ if the patch is malformed or the context doesn't match;
  909 -- @Right newContent@ on success. This is the same applier that lived in
  910 -- 'Seal.ISA.Ops.File.applyUnifiedDiff' — lifted here so both arms share
  911 -- one implementation.
  912 --
  913 -- The applier is **content-based**, not position-based: for each hunk it
  914 -- extracts the old lines (context @ @ + removed @-@), searches for that
  915 -- sequence in the file starting near the hunk header's line-number hint,
  916 -- and splices in the new lines (context @ @ + added @+@). The line number
  917 -- is a hint to disambiguate duplicate matches, NOT a trusted insertion
  918 -- index — this is the design used by OpenCode (@seek@ in patch.ts) and
  919 -- Hermes (@fuzzy_find_and_replace@). A position-based applier that
  920 -- blindly @splitAt@s at the header's oldStart corrupts the file when:
  921 --   * the model's line number is stale (file changed since it read it),
  922 --   * an earlier hunk in the same patch shifted line numbers, or
  923 --   * the model miscounted lines.
  924 -- All three were observed in session 20260810-123133-477 ("The patch is
  925 -- inserting lines at wrong positions").
  926 applyUnifiedDiff :: Text -> Text -> Either Text Text
  927 applyUnifiedDiff original patch =
  928   let origLines  = T.lines original
  929       patchLines = T.lines patch
  930   in go origLines patchLines
  931   where
  932     -- All hunks apply against the ORIGINAL file's line numbering; we
  933     -- thread the already-patched lines through 'go' so successive hunks
  934     -- see a consistent view. The search-start hint is carried in the
  935     -- accumulator so a later hunk's line-number hint still points into
  936     -- the original's coordinate space (we never let an earlier hunk's
  937     -- insertions/deletions shift a later hunk's target).
  938     go origLines [] = Right (joinLines origLines)
  939     go origLines (h : rest)
  940       | T.isPrefixOf "@@ " h  = applyHunk origLines h rest
  941       | T.isPrefixOf "--- " h = go origLines rest
  942       | T.isPrefixOf "+++ " h = go origLines rest
  943       | T.null h              = go origLines rest
  944       | otherwise             = Left ("unexpected line in patch: " <> h)
  945 
  946     applyHunk origLines header rest =
  947       case parseHunkHeader header of
  948         Left err -> Left err
  949         Right (oldStart, mOldLen) ->
  950           let (hunkLines, remainingPatch) = span isHunkLine rest
  951               (oldLines, _newLines) = splitHunk hunkLines
  952               -- The line-number hint is a 0-based index into the file
  953               -- pointing at where the hunk's old lines BEGIN. For a
  954               -- normal hunk (oldLen >= 1) that's oldStart-1 (1-based ->
  955               -- 0-based). For a PURE-INSERT hunk (oldLen == 0) the
  956               -- header's oldStart is the line AFTER which to insert
  957               -- (1-based), so the 0-based insertion index is oldStart
  958               -- itself (insert after that line). The @@ -0,0 +1,N @@
  959               -- form (insert at start of empty file) gives oldStart=0,
  960               -- which clamps to index 0.
  961               isPureInsert = case mOldLen of
  962                 Just 0  -> True
  963                 Nothing -> null oldLines   -- short-form header; infer
  964                 _       -> False
  965               hint = if isPureInsert
  966                        then max 0 oldStart        -- insert after line oldStart
  967                        else max 0 (oldStart - 1) -- replace starting at oldStart
  968           in case findMatch origLines oldLines hint of
  969                Left err -> Left err
  970                Right idx ->
  971                  let (before, atAndAfter) = splitAt idx origLines
  972                      matched = take (length oldLines) atAndAfter
  973                      after   = drop (length oldLines) atAndAfter
  974                      -- Build the replacement: walk the hunk body and the
  975                      -- matched file lines in parallel. For context (' ')
  976                      -- lines we prefer the FILE's actual text (so a
  977                      -- fuzzy match doesn't drag the patch's trailing
  978                      -- whitespace into the file); for '+' lines we use
  979                      -- the patch's text; '-' lines are dropped.
  980                      replacement = spliceHunk hunkLines matched
  981                  in go (before ++ replacement ++ after) remainingPatch
  982 
  983     -- | Walk the hunk body and the matched file lines in parallel,
  984     -- producing the replacement line list. Context (' ') lines take
  985     -- the FILE's line (preserving the file's actual whitespace);
  986     -- '+' lines take the patch's text; '-' and '\\' lines are dropped.
  987     spliceHunk :: [Text] -> [Text] -> [Text]
  988     spliceHunk = go'
  989       where
  990         go' [] _ = []
  991         go' (l : ls) fileLines =
  992           case T.uncons l of
  993             Just (' ', _) -> case fileLines of
  994               (f : fs) -> f : go' ls fs
  995               []       -> go' ls []   -- shouldn't happen; findMatch guards
  996             Just ('+', body') -> body' : go' ls fileLines
  997             Just ('-', _) -> go' ls (drop 1 fileLines)
  998             Just ('\\', _) -> go' ls fileLines
  999             _ -> go' ls fileLines   -- empty/malformed: skip, keep file pos
 1000 
 1001     -- | Split hunk body lines into (oldLines, newLines). Context lines
 1002     -- (' ' prefix) appear in BOTH; removed lines ('-') in old only;
 1003     -- added lines ('+') in new only. '\\' (no-newline-at-eof) and empty
 1004     -- lines are ignored.
 1005     splitHunk :: [Text] -> ([Text], [Text])
 1006     splitHunk ls =
 1007       let go' []         = ([], [])
 1008           go' (l : rest) =
 1009             case T.uncons l of
 1010               Just (' ', body) -> let (o, n) = go' rest in (body : o, body : n)
 1011               Just ('-', body) -> let (o, n) = go' rest in (body : o, n)
 1012               Just ('+', body) -> let (o, n) = go' rest in (o, body : n)
 1013               Just ('\\', _)   -> go' rest
 1014               _                -> go' rest  -- empty or malformed: skip
 1015       in go' ls
 1016 
 1017     -- | Find the index in @lines@ where the @pattern@ sequence matches,
 1018     -- searching near the @hint@ first, then the whole file. Tries four
 1019     -- comparison strategies in order (exact -> rstrip -> trim ->
 1020     -- normalized) so trailing-whitespace and Unicode punctuation
 1021     -- differences don't cause a hard failure — mirrors OpenCode's
 1022     -- @seek@ fallback ladder. An empty pattern (pure-insert hunk) lands
 1023     -- at the hint index (insert after the hinted line).
 1024     findMatch :: [Text] -> [Text] -> Int -> Either Text Int
 1025     findMatch _    []     hint = Right hint
 1026     findMatch xs pat    hint =
 1027       case searchNear xs pat hint of
 1028         Just idx -> Right idx
 1029         Nothing  ->
 1030           case searchAll xs pat of
 1031             Just idx -> Right idx
 1032             Nothing  ->
 1033               Left ("FILE_PATCH: context not found near line "
 1034                       <> T.pack (show (hint + 1)) <> ":\n"
 1035                       <> T.intercalate "\n" pat)
 1036 
 1037     searchNear xs pat hint =
 1038       let n      = length xs
 1039           plen   = length pat
 1040           -- Search outward from the hint: try hint, hint±1, hint±2, ...
 1041           -- so the closest match to the claimed line wins (disambiguates
 1042           -- duplicates by proximity to the hint).
 1043           offsets = concatMap (\d -> [hint + d, hint - d]) [0..(n - 1)]
 1044           valid  = filter (\i -> i >= 0 && i <= n - plen) offsets
 1045       in firstMatch xs pat valid
 1046 
 1047     searchAll xs pat =
 1048       let n    = length xs
 1049           plen = length pat
 1050       in firstMatch xs pat [0 .. n - plen]
 1051 
 1052     firstMatch _    _   []       = Nothing
 1053     firstMatch xs pat (i : is) =
 1054       if matchesAt xs pat i
 1055         then Just i
 1056         else firstMatch xs pat is
 1057 
 1058     matchesAt xs pat i =
 1059       any (\cmp -> and (zipWith cmp (drop i xs) pat))
 1060            [exactEq, rstripEq, trimEq, normalizedEq]
 1061 
 1062     exactEq a b = a == b
 1063     rstripEq a b = T.dropWhileEnd isSpace a == T.dropWhileEnd isSpace b
 1064     trimEq a b = T.dropWhile isSpace (T.dropWhileEnd isSpace a)
 1065               == T.dropWhile isSpace (T.dropWhileEnd isSpace b)
 1066     normalizedEq a b = normalize (trimLine a) == normalize (trimLine b)
 1067       where
 1068         trimLine = T.dropWhile isSpace . T.dropWhileEnd isSpace
 1069         normalize = T.map (\c -> if c == '\t' then ' ' else c)
 1070 
 1071     isHunkLine l =
 1072       T.null l
 1073       || T.isPrefixOf " " l
 1074       || T.isPrefixOf "-" l
 1075       || T.isPrefixOf "+" l
 1076       || T.isPrefixOf "\\" l

 1078     -- | Join lines back into text, preserving the trailing newline if
 1079     -- the original had one (T.lines drops it; we re-add it so a file
 1080     -- ending in a newline round-trips).
 1081     joinLines [] = ""
 1082     joinLines ls = T.intercalate "\n" ls <> "\n"

 1084     parseHunkHeader h =
 1085       case T.stripPrefix "@@ -" h of
 1086         Nothing -> Left ("malformed hunk header: " <> h)
 1087         Just rest0 ->
 1088           let (oldStartStr, afterOld) = breakNum rest0
 1089               (oldLenStr, afterOldLen) = breakLen afterOld
 1090               afterPlus0 = T.dropWhile (/= '+') afterOldLen
 1091               afterPlus  = T.drop 1 afterPlus0
 1092               (newStartStr, _) = breakNum afterPlus
 1093               oldStart = readMaybe (T.unpack oldStartStr) :: Maybe Int
 1094               _newStart = readMaybe (T.unpack newStartStr) :: Maybe Int
 1095           in case oldStart of
 1096                Just os_ -> Right (os_, readMaybe (T.unpack oldLenStr) :: Maybe Int)
 1097                Nothing  -> Left ("malformed hunk header numbers: " <> h)
 1098       where
 1099         breakNum :: Text -> (Text, Text)
 1100         breakNum s =
 1101           let (digits, rest) = T.span isDigit s
 1102           in case T.uncons rest of
 1103                Just (',', rest') -> (digits, T.dropWhile isDigit rest')
 1104                _                 -> (digits, rest)
 1105         -- | After the oldStart number, an optional @,len@ may follow.
 1106         -- Return the len string (empty if the short form was used) and
 1107         -- the rest after the @,len@ (or after the start number if short
 1108         -- form). The length is informational (the content search is the
 1109         -- source of truth); we keep it only for diagnostics.
 1110         breakLen :: Text -> (Text, Text)
 1111         breakLen s =
 1112           case T.uncons s of
 1113             Just (',', rest) -> T.span isDigit rest
 1114             _                -> ("", s)
 1115         readMaybe :: String -> Maybe Int
 1116         readMaybe s = case reads s :: [(Int, String)] of
 1117           [(n, _)] -> Just n
 1118           _        -> Nothing

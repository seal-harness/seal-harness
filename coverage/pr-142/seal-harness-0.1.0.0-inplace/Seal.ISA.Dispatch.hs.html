    2 -- | The dispatcher. Runs the pure authorization gate, then — for Untrusted
    3 -- opcodes — durably records the invocation (tfwRecordAndAck) BEFORE executing,
    4 -- so no untrusted action runs until its audit entry is on disk. Trusted
    5 -- opcodes record concurrently with execution.
    6 --
    7 -- Invariant: for an Untrusted opcode, tfwRecordAndAck completes before opRun
    8 -- is called. This ordering is the whole point of the module.
    9 --
   10 -- There is no longer an Audited branch: the four evolutionary stores
   11 -- (memory, skills, agent-defs) are file-backed under @config\/@ and versioned
   12 -- by git; their opcodes are Trusted file writes that auto-commit. The session
   13 -- transcript (two-file format) remains the per-session record of every opcode
   14 -- invocation, recorded here as an 'EKHarness' entry.
   15 --
   16 -- Opcode invocations are recorded as 'EKHarness' entries in @entries.jsonl@:
   17 -- the opcode name goes into 'erMeta' under @"op"@, and the secret-free input
   18 -- goes into 'erMeta' under @"input"@. No conversation lines are added.

   53 -- | Dispatch an opcode invocation. The dispatcher threads a 'UIOEnv'
   54 -- for Untrusted opcodes (the restricted execution environment carrying
   55 -- the 'UntrustedIO' capability handle + the Git 'CloneDeps' surface);
   56 -- Trusted/Audited opcodes ignore it (they have no 'UIOEnv' in scope —
   57 -- type-level capability scoping, spec §4/§8).
   58 dispatch
   59   :: Registry -> TwoFileHandle -> BackendExec -> UIOEnv
   60   -> ToolTimeoutConfig -> AbortFlag
   61   -> OpName -> Value
   62   -> App (Either DispatchError OpResult)
   63 dispatch reg h backend uioEnv toolTimeout abortFlag name input =
   64   case lookupOp reg name of
   65     Nothing -> pure (Left (OpNotFound name))
   66     Just op ->
   67       case opAuthorize op input of
   68         Left why -> pure (Left (Denied why))
   69         Right () -> do
   70           entry <- liftIO (mkInvocationEntry name input)
   71           let -- Extract the per-call timeout (in seconds → microseconds)
   72               -- from the input JSON, clamped to the max.
   73               Microseconds perCallMicros = extractPerCallTimeout input toolTimeout
   74           case op of
   75             UntrustedOpcode {} -> do
   76               liftIO (tfwRecordAndAck h (TwoFileWrite [] entry))   -- ACK-before-execute
   77               -- Wrap uoRun in the timeout/abort/retry race. The untrusted
   78               -- action runs via 'runUIOWithEnv' (the UIO→IO bridge carrying
   79               -- the 'UntrustedIO' capability handle + Git 'CloneDeps'
   80               -- surface from the 'UIOEnv'); the race wrapper treats it as a
   81               -- plain 'IO' action. OpResult semantic errors are NOT
   82               -- ToolErrors (wrapped Right).
   83               let ioAction :: IO (Either ToolError OpResult)
   84                   ioAction = do
   85                     r <- runUIOWithEnv uioEnv (uoRun op input)
   86                     pure (Right r)
   87               raced <- liftIO (runWithTimeoutAbortRetry toolTimeout abortFlag (Microseconds perCallMicros) ioAction)
   88               case raced of
   89                 Right opResult -> pure (Right opResult)
   90                 Left toolErr -> do
   91                   -- Record the error class in the transcript (design Task 8:
   92                   -- orRecorded carries the error class only — never the full
   93                   -- ToolIOError Text payload, which could contain paths/host
   94                   -- info, design Blocker Resolution #13).
   95                   liftIO (recordToolError h name input toolErr perCallMicros)
   96                   pure (Left (ExecFailed (renderToolError name toolErr)))
   97             TrustedOpcode {} ->
   98               case opTrust op of
   99                 Trusted -> do
  100                   liftIO (tfwRecordAsync h (TwoFileWrite [] entry))
  101                   -- Wrap toRun in the same timeout/abort/retry race.
  102                   env <- ask
  103                   let ioAction :: IO (Either ToolError OpResult)
  104                       ioAction = do
  105                         r <- runApp env (toRun op backend input)
  106                         pure (Right r)
  107                   raced <- liftIO (runWithTimeoutAbortRetry toolTimeout abortFlag (Microseconds perCallMicros) ioAction)
  108                   case raced of
  109                     Right opResult -> pure (Right opResult)
  110                     Left toolErr -> do
  111                       liftIO (recordToolError h name input toolErr perCallMicros)
  112                       pure (Left (ExecFailed (renderToolError name toolErr)))
  113                 Audited -> do
  114                   -- No Audited log remains; treat as Trusted (record to the
  115                   -- session transcript, then run). The evolutionary-store
  116                   -- opcodes that used to be Audited are now Trusted file writes.
  117                   liftIO (tfwRecordAsync h (TwoFileWrite [] entry))
  118                   env <- ask
  119                   let ioAction :: IO (Either ToolError OpResult)
  120                       ioAction = do
  121                         r <- runApp env (toRun op backend input)
  122                         pure (Right r)
  123                   raced <- liftIO (runWithTimeoutAbortRetry toolTimeout abortFlag (Microseconds perCallMicros) ioAction)
  124                   case raced of
  125                     Right opResult -> pure (Right opResult)
  126                     Left toolErr -> do
  127                       liftIO (recordToolError h name input toolErr perCallMicros)
  128                       pure (Left (ExecFailed (renderToolError name toolErr)))
  129                 Untrusted ->
  130                   -- Unreachable: an UntrustedOpcode would have matched above.
  131                   -- Kept for exhaustiveness (the GADT already separates the
  132                   -- arms; opTrust on a TrustedOpcode returns its stored tl,
  133                   -- which the Opcode invariants guarantee is Trusted or Audited).
  134                   error "dispatch: invariant violation — Untrusted trust on a TrustedOpcode"

  136 -- | Build the 'EntryRecord' for an opcode invocation. The opcode name and the
  137 -- secret-free input are recorded in 'erMeta'; the entry kind is 'EKHarness'.
  138 -- 'erConvLen' is 0 because no conversation lines are added by an opcode
  139 -- invocation.
  140 mkInvocationEntry :: OpName -> Value -> IO EntryRecord
  141 mkInvocationEntry name input = do
  142   now <- getCurrentTime
  143   pure EntryRecord
  144     { erId = ""
  145     , erTimestamp = now
  146     , erKind = EKHarness
  147     , erConvLen = 0
  148     , erEnvelope = Nothing
  149     , erUsage = Nothing
  150     , erStop = Nothing
  151     , erDurationMs = Nothing
  152     , erHarness = Nothing
  153     , erCorrelation = Nothing
  154     , erMeta = Map.fromList [("op", object ["name" .= name]), ("input", input)]
  155     }

  157 -- | Record a second 'EKHarness' entry carrying the opcode's 'orRecorded'
  158 -- value as @result@ in @erMeta@. Called by the 'CallDispatcher' sites
  159 -- (webCallDispatcher, channelCallDispatcher, CLI callDispatcher) after
  160 -- 'dispatch' returns a successful 'Right' for the 'SKILL_LOAD' opcode.
  161 -- The invocation entry (recorded by 'dispatch' before the opcode runs)
  162 -- carries only @op.name@ + @input@; this result entry adds the
  163 -- @orRecorded@ payload (which for 'SKILL_LOAD' includes the skill id,
  164 -- description, body, updated_at, and session) so the frontend can render
  165 -- the skill body in a collapsible tool-call box without duplicating it
  166 -- in the transient slash bubble.
  167 --
  168 -- Only fires for 'SKILL_LOAD' (the v1 user-surfacing opcode). Other
  169 -- opcodes' 'orRecorded' is not surfaced to the frontend via this path.
  170 -- Error results ('orIsError' = True) are NOT recorded here — the error
  171 -- text is rendered via 'ccSend' to the slash bubble instead.
  172 --
  173 -- The skill body is ALSO appended to @conversation.jsonl@ as an 'Assistant'
  174 -- message carrying the rendered body text (harness output, not user input).
  175 -- The agent loop builds its
  176 -- next-turn context from @conversation.jsonl@ ('runTurn' at
  177 -- 'Seal.Agent.Loop' reads @tfwReadConversation@); without this write,
  178 -- a user-invoked @/skill load@ would record the body only to
  179 -- @entries.jsonl@ (the audit sidecar) and the model would never see
  180 -- the skill on the next turn — the @/skill load@ "Command output" box
  181 -- would display but the skill would not actually load. The agent's own
  182 -- tool-call path needs no such write because 'runTurn' already folds
  183 -- the 'CbToolResult' into @conversation.jsonl@; this closes the gap
  184 -- for the user-invoked path, which bypasses the agent loop.
  185 --
  186 -- The @channel@ argument is the communications-channel label of the
  187 -- session the skill was loaded into (e.g. @"telegram"@, @"web"@,
  188 -- @"cli"@), or 'Nothing' when the caller didn't supply one. It is
  189 -- recorded under @"channel"@ in @erMeta@ so the frontend can surface
  190 -- the channel provenance in the skill-load row's source label (e.g.
  191 -- @"Skill · telegram"@), making it clear how/why the skill was loaded.
  192 --
  193 -- The @input@'s optional @message@ field (a string) is appended to
  194 -- @conversation.jsonl@ as a SECOND 'User' message AFTER the skill body,
  195 -- so the model sees the skill followed by the user's trailing message on
  196 -- the next turn (e.g. @/skill load start #123@ loads the @start@ skill
  197 -- then sends @#123@ as the user's message). When @message@ is absent or
  198 -- blank, no second message is written (the load behaves as before — skill
  199 -- body only).
  200 recordSkillLoadResult :: TwoFileHandle -> OpName -> Value -> OpResult -> Maybe Text -> IO ()
  201 recordSkillLoadResult h (OpName nm) input result mChannel
  202   | nm == "SKILL_LOAD" && not (orIsError result) = do
  203       now <- getCurrentTime
  204       let channelMeta = case mChannel of
  205             Just ch  -> [("channel", A.String ch)]
  206             Nothing  -> []
  207           entry = EntryRecord
  208             { erId = ""
  209             , erTimestamp = now
  210             , erKind = EKHarness
  211             , erConvLen = 0
  212             , erEnvelope = Nothing
  213             , erUsage = Nothing
  214             , erStop = Nothing
  215             , erDurationMs = Nothing
  216             , erHarness = Nothing
  217             , erCorrelation = Nothing
  218             , erMeta = Map.fromList
  219                 ([ ("op", object ["name" .= OpName nm])
  220                  , ("input", input)
  221                  , ("result", orRecorded result)
  222                  ] <> channelMeta)
  223             }
  224           bodyText = T.intercalate "\n" [ t | TrpText t <- orParts result ]
  225           -- The optional trailing message from the /skill load <id> <message>
  226           -- invocation. Extracted from the input's @message@ field; blank or
  227           -- absent yields no second conversation message.
  228           messageText = fromMaybe "" (parseMaybe (A.withObject "input" (A..: "message")) input)
  229           trimmedMessage = T.strip messageText
  230           convMsgs =
  231             [ Message Assistant [CbText bodyText] | not (T.null bodyText) ]
  232             <> [ Message User [CbText trimmedMessage] | not (T.null trimmedMessage) ]
  233       tfwRecordAndAck h (TwoFileWrite convMsgs entry)
  234   | otherwise = pure ()

  270 -- | Record a 'ToolError' as an 'EKHarness' transcript entry carrying the
  271 -- error CLASS (secret-free — design Blocker Resolution #13: never the full
  272 -- 'ToolIOError' Text payload, which could contain paths/host info). The
  273 -- metadata includes the timeout value (seconds) and the error class string.
  274 -- Called by 'dispatch' when the timeout/abort/retry wrapper returns 'Left'.
  275 recordToolError :: TwoFileHandle -> OpName -> Value -> ToolError -> Int -> IO ()
  276 recordToolError h name input toolErr timeoutMicros = do
  277   now <- getCurrentTime
  278   let entry = EntryRecord
  279         { erId = ""
  280         , erTimestamp = now
  281         , erKind = EKHarness
  282         , erConvLen = 0
  283         , erEnvelope = Nothing
  284         , erUsage = Nothing
  285         , erStop = Nothing
  286         , erDurationMs = Nothing
  287         , erHarness = Nothing
  288         , erCorrelation = Nothing
  289         , erMeta = Map.fromList
  290             [ ("op", object ["name" .= name])
  291             , ("input", input)
  292             , ("error", A.String (errorClass toolErr))
  293             , ("timeout_s", A.toJSON (timeoutMicros `div` 1_000_000))
  294             ]
  295         }
  296   tfwRecordAndAck h (TwoFileWrite [] entry)

   40 -- ---------------------------------------------------------------------------
   41 -- TelegramToken — the validated bot token for the Telegram Bot API
   42 -- ---------------------------------------------------------------------------
   43 
   44 -- | A Telegram bot token as issued by BotFather (@<digits>:<AA-xx>@). Smart-
   45 -- constructed: non-empty, no leading dash (option-injection defense — a
   46 -- leading-dash value would be interpreted as a flag by a subprocess if the
   47 -- token ever reaches an argv), charset @[A-Za-z0-9+:-_@. The validated type
   48 -- that reaches the Bot API URL path. Not a secret in the vault sense (it is
   49 -- stored in @config.toml@); it is the channel credential, like the Signal
   50 -- account label.
   51 newtype TelegramToken = TelegramToken Text
   52   deriving stock (Eq, Show)
   53 
   54 -- | Smart-construct a 'TelegramToken'. Accepts the BotFather format
   55 -- (@<digits>:<rest>@) and rejects empty / leading-dash / invalid chars.
   56 mkTelegramToken :: Text -> Either Text TelegramToken
   57 mkTelegramToken t
   58   | T.null t        = Left "telegram token is empty"
   59   | T.head t == '-' = Left "telegram token must not start with '-' (option injection)"
   60   | not (T.all validTokenChar t)
   61                      = Left "telegram token has invalid characters"
   62   | otherwise       = Right (TelegramToken t)
   63 
   64 validTokenChar :: Char -> Bool
   65 validTokenChar c = c `Set.member` tokenChars
   66   where
   67     tokenChars :: Set Char
   68     tokenChars = Set.fromList
   69       $ ['A'..'Z'] <> ['a'..'z'] <> ['0'..'9'] <> "+-:_"
   70 
   71 telegramTokenText :: TelegramToken -> Text
   72 telegramTokenText (TelegramToken t) = t

   78 -- | The @[telegram]@ config section. All fields optional at the file level;
   79 -- @seal telegram@ fails fast via 'resolveTelegramConfig' if 'tcToken' is
   80 -- unset (no config token AND no vault-supplied token).
   81 data TelegramConfig = TelegramConfig
   82   { tcToken         :: Maybe Text       -- ^ BotFather bot token
   83   , tcTextChunkLimit :: Maybe Int        -- ^ default 'defaultTelegramChunkLimit'
   84   , tcAllowFrom     :: AllowList Text   -- ^ sender allow-list (Telegram user id)
   85   } deriving stock (Eq, Show)
   86 
   87 -- | Telegram's text-message character limit. Replies longer than this are
   88 -- chunked via 'Seal.Channels.Telegram.Transport.chunkMessage'. Telegram's
   89 -- hard limit is 4096; we leave headroom for the @model> @ prefix.
   90 defaultTelegramChunkLimit :: Int
   91 defaultTelegramChunkLimit = 3900
   92 
   93 defaultTelegramConfig :: TelegramConfig
   94 defaultTelegramConfig = TelegramConfig
   95   { tcToken = Nothing
   96   , tcTextChunkLimit = Just defaultTelegramChunkLimit
   97   , tcAllowFrom = AllowAll
   98   }

  100 -- ---------------------------------------------------------------------------
  101 -- TOML codec (bidirectional, used by Seal.Config.File)
  102 -- ---------------------------------------------------------------------------
  103 
  104 -- | Bidirectional tomland codec for the @[telegram]@ section. 'tcAllowFrom'
  105 -- is a TOML array of strings (@allow_from = ["12345", "67890"]@); an absent
  106 -- array decodes as 'AllowAll'.
  107 telegramConfigCodec :: Toml.TomlCodec TelegramConfig
  108 telegramConfigCodec = TelegramConfig
  109   <$> Toml.dioptional (Toml.text "token")            .= tcToken
  110   <*> Toml.dioptional (Toml.int "text_chunk_limit")  .= tcTextChunkLimit
  111   <*> allowListCodec                                  .= tcAllowFrom
  112 
  113 -- | Codec for the @allow_from@ array: 'AllowAll' when absent, 'AllowOnly'
  114 -- of the listed strings when present.
  115 allowListCodec :: Toml.TomlCodec (AllowList Text)
  116 allowListCodec = Toml.dimap toList fromList (Toml.dioptional (Toml.arrayOf Toml._Text "allow_from"))
  117   where
  118     toList :: AllowList Text -> Maybe [Text]
  119     toList AllowAll       = Nothing
  120     toList (AllowOnly xs) = Just (Set.toList xs)
  121     fromList :: Maybe [Text] -> AllowList Text
  122     fromList Nothing   = AllowAll
  123     fromList (Just xs) = AllowOnly (Set.fromList xs)
  124 
  125 -- ---------------------------------------------------------------------------
  126 -- Resolution — config section + vault token -> validated (token, limit, allow)
  127 -- ---------------------------------------------------------------------------
  128 
  129 -- | Resolve the @[telegram]@ section + an optional vault-supplied token
  130 -- into a validated 'TelegramToken', chunk limit, and 'AllowList UserId' for
  131 -- the channel's sender check. The vault token (if supplied) overrides the
  132 -- config token; either way it's smart-constructed here. A missing token (no
  133 -- section, no vault token) is rejected. The @allow_from@ 'AllowList Text' is
  134 -- mapped through 'mkUserId' to an 'AllowList UserId' — a malformed entry
  135 -- fails the whole resolution with a clear error.
  136 resolveTelegramConfig
  137   :: Maybe TelegramConfig -> Maybe Text
  138   -> Either Text (TelegramToken, Int, AllowList UserId)
  139 resolveTelegramConfig mSection mVaultToken =
  140   case mVaultToken <|> (mSection >>= tcToken) of
  141     Nothing -> Left "telegram: no token configured (set [telegram].token or supply via vault)"
  142     Just tokenRaw -> do
  143       token <- mkTelegramToken tokenRaw
  144       let tgCfg = fromMaybe defaultTelegramConfig mSection
  145           limit  = fromMaybe defaultTelegramChunkLimit (tcTextChunkLimit tgCfg)
  146       allowUserId <- mapAllowUserId (tcAllowFrom tgCfg)
  147       Right (token, limit, allowUserId)
  148 
  149 -- | Map an 'AllowList Text' through 'mkUserId' to an 'AllowList UserId'.
  150 -- 'AllowAll' passes through; 'AllowOnly' maps each element (a malformed
  151 -- entry fails the whole resolution).
  152 mapAllowUserId :: AllowList Text -> Either Text (AllowList UserId)
  153 mapAllowUserId AllowAll = Right AllowAll
  154 mapAllowUserId (AllowOnly txts) = do
  155   uids <- traverse mkUserIdText (Set.toList txts)
  156   Right (AllowOnly (Set.fromList uids))
  157   where
  158     mkUserIdText t = case mkUserId t of
  159       Right u  -> Right u
  160       Left err -> Left ("telegram allow_from has invalid user id " <> T.pack (show t) <> ": " <> err)

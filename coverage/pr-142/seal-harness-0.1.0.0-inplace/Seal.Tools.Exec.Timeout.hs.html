    1 -- | The IO-level three-way race + retry wrapper for tool-call timeout/abort.
    2 --
    3 -- This module is the core of the tool-call-timeout-abort-retry mechanism.
    4 -- It is IO-level (no 'App'/'Env' dependency) — the 'App'→'IO' bridge is at
    5 -- the dispatch call site (design Blocker Resolution #1): the dispatcher
    6 -- captures the 'Env' via 'ask', then re-enters 'App' inside the worker
    7 -- thread via @runApp env (uoRun ...)@. The wrapper itself never imports
    8 -- 'Seal.Types.App' — this keeps the race logic testable without a full
    9 -- 'AppEnv'.
   10 --
   11 -- = The three-way race (worker vs timeout vs abort-poll)
   12 --
   13 -- Per-attempt, three 'IO' actions race (via nested 'race'):
   14 --
   15 -- @
   16 -- (worker vs timeoutDelay) vs waitForAbort pollInterval
   17 -- @
   18 --
   19 -- Nesting order: abort wins over timeout (user intent > resource limit). On
   20 -- the worker winning: return the result. On timeout: 'cancel' the worker
   21 -- (which triggers the 'withManagedProcess' bracket cleanup → process-group
   22 -- kill), return 'ToolTimeout'. On abort: 'cancel' the worker, return
   23 -- 'ToolAborted'.
   24 --
   25 -- = Retry logic
   26 --
   27 -- Retry only on 'ToolTimeout' and 'ToolIOError' (transient). Do NOT retry on
   28 -- 'ToolAborted' (user cancelled — respect immediately) or on 'Right'
   29 -- results (success is success, even if the opcode returned a semantic error
   30 -- — that's not a transport failure). Up to 'ttcRetryMax' attempts,
   31 -- exponential backoff via 'computeRetryDelay'. The loop checks 'isAborted'
   32 -- BEFORE sleeping and AFTER sleeping (before the next attempt) so an abort
   33 -- during the retry sleep is detected promptly (design PM question).
   34 module Seal.Tools.Exec.Timeout
   35   ( runWithTimeoutAbortRetry
   36   ) where
   37 
   38 import Control.Concurrent (threadDelay)
   39 import Control.Concurrent.Async
   40   ( Async, async, cancel, race, wait, waitCatch )
   41 import Control.Exception (IOException, try)

   44 import Seal.Tools.Timeout

   49 -- | Run an IO action with timeout + abort + retry protection.
   50 --
   51 --   * @cfg@ — the timeout/retry config (default, max, retry count, backoff).
   52 --   * @flag@ — the session abort flag (polled during the race).
   53 --   * @timeout@ — the per-call timeout (microseconds, already clamped by
   54 --     'Seal.Tools.Timeout.extractPerCallTimeout').
   55 --   * @action@ — the opcode action, bridged from 'App' at the call site
   56 --     (via @runApp env (...)@). Returns 'Either ToolError a' — 'Right' for
   57 --     success (including semantic errors), 'Left' for transport failures.
   58 --
   59 -- Returns 'Right a' on success, 'Left ToolError' on timeout/abort/IO-failure
   60 -- (wrapped in 'ToolRetriesExhausted' after 'ttcRetryMax' attempts).
   61 runWithTimeoutAbortRetry
   62   :: forall a
   63    . ToolTimeoutConfig
   64   -> AbortFlag
   65   -> Microseconds
   66   -> IO (Either ToolError a)
   67   -> IO (Either ToolError a)
   68 runWithTimeoutAbortRetry cfg flag (Microseconds timeoutUs) action = go 0
   69   where
   70     go :: Int -> IO (Either ToolError a)
   71     go attempt = do
   72       -- Check-before-attempt: if already aborted, return immediately.
   73       aborted <- isAborted flag
   74       if aborted
   75         then pure (Left ToolAborted)
   76         else do
   77           -- Run the three-way race: worker vs timeout vs abort-poll.
   78           result <- raceThreeWay cfg flag timeoutUs action
   79           case result of
   80             Right _ -> pure result  -- success (or semantic error) — return immediately, no retry
   81             Left err
   82               | not (shouldRetry err) -> pure result  -- ToolAborted or already-exhausted
   83               | attempt + 1 >= ttcRetryMax cfg ->
   84                   -- Exhausted retries.
   85                   pure (Left (ToolRetriesExhausted err))
   86               | otherwise -> do
   87                   -- Check-before-sleep: abort during the sleep window?
   88                   preSleep <- isAborted flag
   89                   if preSleep
   90                     then pure (Left ToolAborted)
   91                     else do
   92                       let Microseconds delayUs = computeRetryDelay cfg attempt
   93                       threadDelay delayUs
   94                       -- Check-after-sleep: abort during the sleep?
   95                       postSleep <- isAborted flag
   96                       if postSleep
   97                         then pure (Left ToolAborted)
   98                         else go (attempt + 1)
   99 
  100 -- | The three-way race: worker vs timeout vs abort-poll.
  101 --
  102 -- Nesting: @(worker-vs-timeout) vs abort-poll@. Abort wins over timeout
  103 -- (user intent > resource limit). On the worker winning: return its
  104 -- result. On timeout: the worker async is cancelled (triggering the
  105 -- 'withManagedProcess' bracket cleanup → process-group kill) and we return
  106 -- 'ToolTimeout'. On abort: the worker async is cancelled and we return
  107 -- 'ToolAborted'.
  108 raceThreeWay
  109   :: forall a
  110    . ToolTimeoutConfig
  111   -> AbortFlag
  112   -> Int  -- ^ timeout (microseconds)
  113   -> IO (Either ToolError a)
  114   -> IO (Either ToolError a)
  115 raceThreeWay cfg flag timeoutUs action = do
  116   worker <- async action
  117   -- Outer race: (worker-vs-timeout) vs abort-poll.
  118   -- `race a b` returns Left if `a` wins, Right if `b` wins.
  119   outer <- race
  120     (race (wait worker) (threadDelay timeoutUs))  -- inner: Left=worker, Right=timeout
  121     (abortPoll (ttcAbortPollMicros cfg))
  122   case outer of
  123     -- Inner race (worker-vs-timeout) won first.
  124     Left inner -> case inner of
  125       Left r -> pure r  -- worker completed: return its result
  126       Right () -> do
  127         -- Timeout: cancel the worker (bracket cleanup → process-group kill),
  128         -- wait for cleanup to finish (bounded by reapBounded's 1s),
  129         -- return ToolTimeout (seconds for display).
  130         cancelWorker worker
  131         pure (Left (ToolTimeout (timeoutUs `div` 1_000_000)))
  132     -- Abort-poll won first: cancel the worker, return ToolAborted.
  133     Right _aborted -> do
  134       cancelWorker worker
  135       pure (Left ToolAborted)
  136   where
  137     -- The abort-poll: loop checking isAborted every @pollUs@ microseconds.
  138     -- Returns True when aborted (so the outer race's Right branch fires).
  139     abortPoll :: Int -> IO Bool
  140     abortPoll pollUs = do
  141       let loop = do
  142             a <- isAborted flag
  143             if a then pure True else do threadDelay pollUs; loop
  144       loop
  145 
  146 -- | Cancel a worker async + wait for its bracket cleanup to finish. The
  147 -- cleanup (killProcessGroup + bounded reap) is bounded, so this returns
  148 -- promptly; the @try@ swallows any async-exception IO errors.
  149 cancelWorker :: Async a -> IO ()
  150 cancelWorker worker = do
  151   cancel worker
  152   _ <- try @IOException (waitCatch worker)
  153   pure ()

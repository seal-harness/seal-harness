    2 -- | The persistent ssh-agent registry (#88). Each unique repo (identified
    3 -- by its vault key) gets an entry recording the @SSH_AUTH_SOCK@ /
    4 -- @SSH_AGENT_PID@ of the ssh-agent process that holds its deploy key. The
    5 -- registry persists to @~\/.seal\/state\/ssh-agents\/registry.json@ so
    6 -- agents survive @seal serve@ restarts — a second seal process reuses the
    7 -- running agent instead of spawning a new one.
    8 --
    9 -- At startup, 'arProbeAndSweep' probes each persisted agent
   10 -- (@ssh-add -l@ with @SSH_AUTH_SOCK@) and removes dead entries (PID gone
   11 -- or socket removed). This GCs agents from unexpected shutdowns where
   12 -- cleanup didn't happen.
   13 --
   14 -- Security: the registry file contains only @auth_sock@ + @agent_pid@
   15 -- (non-secret values — the socket path + PID are not secrets; the key
   16 -- material lives only in the agent's memory). Mode 0600 to prevent other
   17 -- users from reading the socket path (defense-in-depth).

   57 -- | The liveness of a persisted agent, determined by 'probeAgent'.
   58 data AgentStatus = AgentAlive | AgentDead

   61 -- | A JSON-serializable registry entry. The 'Text' key is the repo's vault
   62 -- key (the repo identity — see 'Seal.SourceControl.Clone.cdAgentRegistry').
   63 newtype AgentRegistry = AgentRegistry (Map Text SshAgentEnv)
   64   deriving stock (Eq, Show)
   65 
   66 instance ToJSON AgentRegistry where
   67   toJSON (AgentRegistry m) =
   68     object [ AK.fromString (T.unpack k) .= object [ "auth_sock" .= saeAuthSock v
   69                                                   , "agent_pid" .= saeAgentPid v ]
   70            | (k, v) <- Map.toList m ]
   71 
   72 instance FromJSON AgentRegistry where
   73   parseJSON = withObject "AgentRegistry" $ \o -> do
   74     let keys = AKM.keys o
   75     entries <- traverse (\k -> do
   76       v <- o .: k
   77       pure (AK.toText k, v)) keys
   78     pure (AgentRegistry (Map.fromList entries))

   84 -- | A handle to the persistent agent registry, closing over the directory
   85 -- where the registry file lives. All operations are serialized behind a
   86 -- process-wide MVar to prevent lost-update races (mirrors
   87 -- 'Seal.SourceControl.Registry.repoRegistryWriteLock').
   88 data AgentRegistryHandle = AgentRegistryHandle
   89   { arhDir    :: FilePath
   90     -- ^ The directory holding @registry.json@.
   91   , arhLock   :: MVar ()
   92     -- ^ Serializes read-modify-write cycles.
   93   , arhCache  :: IORef (Maybe AgentRegistry)
   94     -- ^ In-memory cache of the last-loaded registry (avoids re-reading the
   95     -- file on every 'arLoad' when no writes have occurred in between).
   96   , arhLive   :: IORef (Set Text)
   97     -- ^ Keys of agents started in THIS process (not loaded from disk).
   98     -- Reuse of a live agent skips the liveness probe (the agent was just
   99     -- started — it's alive). Only agents loaded from a prior process's
  100     -- persisted registry are probed.
  101   }
  102 
  103 -- | The registry file name (lives inside 'arhDir').
  104 registryFileName :: FilePath
  105 registryFileName = "registry.json"
  106 
  107 -- | Build an 'AgentRegistryHandle' closing over @dir@. The directory is
  108 -- created if it doesn't exist (the caller's 'ensureSealDirs' is responsible
  109 -- for creating it; this is a best-effort idempotent creation).
  110 mkAgentRegistryHandle :: FilePath -> IO AgentRegistryHandle
  111 mkAgentRegistryHandle dir = do
  112   lock  <- newMVar ()
  113   cache <- newIORef Nothing
  114   live  <- newIORef Set.empty
  115   pure AgentRegistryHandle
  116     { arhDir = dir
  117     , arhLock = lock
  118     , arhCache = cache
  119     , arhLive = live
  120     }

  122 ----------------------------------------------------------------------------
  123 -- Load / Save (atomic)
  124 ----------------------------------------------------------------------------
  125 
  126 -- | Load the registry from disk. Absent file → empty registry. Caches the
  127 -- result in 'arhCache'.
  128 arLoad :: AgentRegistryHandle -> IO (Map Text SshAgentEnv)
  129 arLoad h = do
  130   mCached <- readIORef (arhCache h)
  131   case mCached of
  132     Just (AgentRegistry m) -> pure m
  133     Nothing -> do
  134       m <- loadFromDisk (arhDir h)
  135       writeIORef (arhCache h) (Just (AgentRegistry m))
  136       pure m
  137 
  138 loadFromDisk :: FilePath -> IO (Map Text SshAgentEnv)
  139 loadFromDisk dir = do
  140   let path = dir </> registryFileName
  141   exists <- doesFileExist path
  142   if not exists
  143     then pure Map.empty
  144     else do
  145       bytes <- BL.readFile path
  146       case A.decode bytes of
  147         Just (AgentRegistry m) -> pure m
  148         Nothing                -> pure Map.empty
  149 
  150 -- | Save the registry to disk atomically: write @registry.json.tmp@, then
  151 -- rename. The file is created mode 0600 (defense-in-depth — the socket
  152 -- path is not a secret but shouldn't be world-readable).
  153 saveToDisk :: FilePath -> Map Text SshAgentEnv -> IO ()
  154 saveToDisk dir m = do
  155   let path = dir </> registryFileName
  156       tmp  = path <> ".tmp"
  157       encoded = A.encode (AgentRegistry m)
  158   BL.writeFile tmp encoded
  159   setFileMode tmp 0o600
  160   renameFile tmp path

  163 -- Mutation (lock-serialized)

  166 -- | Insert or overwrite an agent entry (keyed by vault key). Also marks
  167 -- the agent as live in this process (so subsequent reuse skips the liveness
  168 -- probe).
  169 arUpsert :: AgentRegistryHandle -> Text -> SshAgentEnv -> IO ()
  170 arUpsert h key env = withMVar (arhLock h) $ \_ -> do
  171   m <- loadFromDisk (arhDir h)
  172   let m' = Map.insert key env m
  173   saveToDisk (arhDir h) m'
  174   writeIORef (arhCache h) (Just (AgentRegistry m'))
  175   modifyIORef' (arhLive h) (Set.insert key)
  176 
  177 -- | Remove an agent entry (no-op if absent).
  178 arRemove :: AgentRegistryHandle -> Text -> IO ()
  179 arRemove h key = withMVar (arhLock h) $ \_ -> do
  180   m <- loadFromDisk (arhDir h)
  181   let m' = Map.delete key m
  182   saveToDisk (arhDir h) m'
  183   writeIORef (arhCache h) (Just (AgentRegistry m'))
  184   modifyIORef' (arhLive h) (Set.delete key)
  185 
  186 -- | Check if an agent was started in THIS process (live, no probe needed).
  187 arIsLive :: AgentRegistryHandle -> Text -> IO Bool
  188 arIsLive h key = Set.member key <$> readIORef (arhLive h)

  190 ----------------------------------------------------------------------------
  191 -- Probe + Sweep (liveness check + GC dead entries)
  192 ----------------------------------------------------------------------------
  193 
  194 -- | Probe a persisted agent to determine if it's still alive. Runs
  195 -- @ssh-add -l@ with @SSH_AUTH_SOCK=<sock>@:
  196 --
  197 --   * Exit 0 → alive (keys are loaded).
  198 --   * Exit 1 → alive (agent is running, but no keys loaded — still alive).
  199 --   * Exit 2 → dead (can't contact the agent — socket gone or PID dead).
  200 --   * IOException → dead (the @ssh-add@ binary couldn't run or the socket
  201 --     path is invalid).
  202 probeAgent :: SshAgentEnv -> IO AgentStatus
  203 probeAgent env = do
  204   let cp = (proc "ssh-add" ["-l"])
  205         { close_fds = True
  206         , env = Just [("SSH_AUTH_SOCK", saeAuthSock env)]
  207         , std_out = CreatePipe
  208         , std_err = CreatePipe
  209         }
  210   eRes <- try @IOError (readCreateProcessWithExitCode cp "")
  211   case eRes of
  212     Left _ioErr -> pure AgentDead
  213     Right (ec, _out, _err) -> case ec of
  214       ExitSuccess     -> pure AgentAlive
  215       ExitFailure 1   -> pure AgentAlive
  216       ExitFailure _n  -> pure AgentDead
  217 
  218 -- | Probe every persisted agent and remove dead entries. Called at seal
  219 -- startup (before serving) so stale agents from a crashed/seal-killed
  220 -- process are GC'd. Lock-serialized so a concurrent 'arUpsert' can't race
  221 -- the sweep.
  222 arProbeAndSweep :: AgentRegistryHandle -> IO ()
  223 arProbeAndSweep h = withMVar (arhLock h) $ \_ -> do
  224   m <- loadFromDisk (arhDir h)
  225   -- Probe each entry; collect the alive ones.
  226   aliveEntries <- traverse (\(k, env) -> do
  227     status <- probeAgent env
  228     pure (k, env, status)) (Map.toList m)
  229   let m' = Map.fromList [ (k, env) | (k, env, AgentAlive) <- aliveEntries ]
  230   -- Only write if something changed (avoid unnecessary disk writes).
  231   if Map.size m /= Map.size m'
  232     then do
  233       saveToDisk (arhDir h) m'
  234       writeIORef (arhCache h) (Just (AgentRegistry m'))
  235     else writeIORef (arhCache h) (Just (AgentRegistry m))

  241 -- | Parse @ssh-agent -s@ stdout for @SSH_AUTH_SOCK@ and @SSH_AGENT_PID@.
  242 -- Re-exported here so the test can start a real agent and parse its output.
  243 parseAgentEnv :: String -> Maybe SshAgentEnv
  244 parseAgentEnv out = do
  245   sock <- findAssign "SSH_AUTH_SOCK"
  246   pid  <- findAssign "SSH_AGENT_PID"
  247   Just SshAgentEnv { saeAuthSock = sock, saeAgentPid = pid }
  248   where
  249     ls = lines out
  250     findAssign name =
  251       let prefix = name <> "="
  252       in case filter (prefix `isPrefixOf'`) ls of
  253            (l : _) -> Just (takeWhile (/= ';') (drop (length prefix) l))
  254            []      -> Nothing
  255     isPrefixOf' p s = take (length p) s == p

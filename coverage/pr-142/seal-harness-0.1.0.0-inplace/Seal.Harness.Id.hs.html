    1 {-# LANGUAGE DeriveGeneric #-}
    2 {-# LANGUAGE OverloadedStrings #-}
    3 -- | A UUID-backed durable harness identity (the registry key). Minted at
    4 -- spawn time; stamped as a tmux @seal_id marker on the harness window so a
    5 -- window can be re-identified after a rename or reconnect. The UUID is
    6 -- generated in-repo from 'System.Random' (the repo has @uuid-types@ but
    7 -- not the full @uuid@ package, so v4 generation is hand-rolled from two
    8 -- random 'Word64's with the v4 version/variant bits set).
    9 module Seal.Harness.Id
   10   ( HarnessId (..)
   11   , newHarnessId
   12   , parseHarnessId
   13   , harnessIdToText
   14   , isValidHarnessIdText
   15   ) where
   16 
   17 import Data.Aeson (FromJSON, ToJSON)
   18 import Data.Bits ((.&.), (.|.), complement)
   19 import Data.Text (Text)
   20 import Data.Text qualified as T
   21 import Data.UUID.Types qualified as U
   22 import Data.Word (Word64)
   23 import GHC.Generics (Generic)
   24 import System.Random (randomIO)
   25 
   26 -- | A UUID-backed durable harness identity.
   27 newtype HarnessId = HarnessId Text
   28   deriving stock (Eq, Ord, Show, Generic)
   29   deriving newtype (ToJSON, FromJSON)
   30 
   31 -- | Mint a fresh random HarnessId (UUID v4). IO because it reads randomness.
   32 newHarnessId :: IO HarnessId
   33 newHarnessId = do
   34   w1 <- randomIO
   35   w2 <- randomIO
   36   -- Set version (4) in the high nibble of byte 6 (bits 48-51 of w1) and
   37   -- variant (10) in the high bits of byte 8 (bits 56-57 of w2).
   38   let v1 = (w1 .&. complement 0x0000F000) .|. (0x4000 :: Word64)   -- version 4
   39       v2 = (w2 .|. 0x8000000000000000) .&. complement 0x4000000000000000  -- variant 10
   40   pure (HarnessId (T.pack (U.toString (U.fromWords64 v1 v2))))
   41 
   42 -- | The text form (a UUID string) — the value stamped as the @seal_id marker.
   43 harnessIdToText :: HarnessId -> Text
   44 harnessIdToText (HarnessId t) = t
   45 
   46 -- | Parse a HarnessId from its text form. 'Left' on malformed input.
   47 parseHarnessId :: Text -> Either Text HarnessId
   48 parseHarnessId t
   49   | isValidHarnessIdText t = Right (HarnessId t)
   50   | otherwise              = Left ("invalid HarnessId: " <> t)
   51 
   52 -- | True if the text is a valid UUID (8-4-4-4-12 hex, case-insensitive).
   53 -- Used to defend the @seal_id marker stamp against a malformed id.
   54 isValidHarnessIdText :: Text -> Bool
   55 isValidHarnessIdText t =
   56   case U.fromString (T.unpack t) of
   57     Just _  -> True
   58     Nothing -> False

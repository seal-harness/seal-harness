  165 -- | Resolve the system prompt for a session turn. This is the **single**
  166 -- implementation — used by all four surfaces (Web, TUI, Telegram, Signal).
  167 --
  168 -- The resolution pipeline:
  169 -- 1. If @smSystemOverride@ is set (non-empty), use it as the base.
  170 -- 2. Otherwise, read the bound agent's @adSystem@ from the workdir-aware
  171 --    agent-def backend.
  172 -- 3. Inject static guidance (parallel / tool-use / task-completion).
  173 -- 4. Inject the autoload skill body.
  174 -- 5. Inject the available-skills catalog (if enabled).
  175 --
  176 -- The flags ('autoloadId', 'injectCatalog', 'parallel', 'toolUse',
  177 -- 'taskCompletion') are passed as explicit args — the caller computes
  178 -- them from the loaded 'RuntimeConfig'. This matches the web path's
  179 -- existing convention and avoids re-reading config inside the resolver.
  180 resolveSystemPrompt
  181   :: Def.AgentDefBackend
  182   -> Skill.SkillBackend
  183   -> Maybe Text
  184   -- ^ The resolved auto-load skill id ('Nothing' disables injection).
  185   -> Bool
  186   -- ^ Whether to inject the @\<available_skills\>@ catalog.
  187   -> Bool
  188   -- ^ Whether to inject the parallel tool-call guidance block.
  189   -> Bool
  190   -- ^ Whether to inject the tool-use enforcement guidance block.
  191   -> Bool
  192   -- ^ Whether to inject the task-completion guidance block.
  193   -> SessionMeta
  194   -> IO (Maybe Text)
  195 resolveSystemPrompt agentDefBackend skillBackend autoloadId injectCatalog
  196                    parallel toolUse taskCompletion meta = do
  197   base <- case smSystemOverride meta of
  198     Just t | not (T.null (T.strip t)) -> pure (Just t)
  199     _ -> case smAgent meta of
  200            Nothing  -> pure Nothing
  201            Just aid -> maybe Nothing adSystem <$> Def.adbRead agentDefBackend aid
  202   let withGuidance = injectStaticGuidance parallel toolUse taskCompletion base
  203   withAutoload <- injectAutoloadSkill skillBackend autoloadId withGuidance
  204   if injectCatalog
  205     then injectAvailableSkills skillBackend withAutoload
  206     else pure withAutoload

  407 -- | The adapter-owned per-turn hooks (design §5.2 step table). These are the
  408 -- ONLY medium-specific pieces the engine calls back into — the adapter
  409 -- constructs them, the engine runs them at the right points.
  410 --
  411 -- * @taCaps@ — the 'ChannelCaps' for this turn (the only medium-specific
  412 --   I/O seam).
  413 -- * @taPreTurn@ — runs BEFORE the engine broadcasts "thinking" + acquires
  414 --   the lock. Used by inbox-driven channels to 'replySubscribe' the channel
  415 --   handle and 'replyFanoutMessage' the inbound text to other subscribed
  416 --   channels (cross-channel mirroring). Web/CLI pass @pure ()@ (no
  417 --   subscription, no mirror — the web frontend sees the message directly,
  418 --   the CLI is the only surface).
  419 -- * @taChannelLabel@ — the label stamped into the transcript's @channel@
  420 --   field (e.g. @"web"@, @"cli"@, @smChannel@ for inbox-driven channels).
  421 -- * @taOnStop@ — builds the 'aeOnStop' fan-out hook for a given 'SessionId'.
  422 --   'Just' for surfaces that want cross-channel reply fan-out after the turn
  423 --   (web + channels); 'Nothing' for the CLI (no chat channels subscribe to a
  424 --   CLI session).
  425 -- * @taOnUserMessage@ — builds the 'aeOnUserMessage' hook for a given
  426 --   'SessionMeta' (e.g. the /bg 'broadcastTabs' refresh when
  427 --   'shouldAutoTab' is false). 'Nothing' for normal turns.
  428 -- * @taPostTurn@ — runs AFTER the engine's own post-turn broadcast +
  429 --   auto-tab. Used by channels to refresh the @lists@ snapshot
  430 --   ('broadcastTabs') when 'shouldAutoTab' is false (the /bg path). Web/CLI
  431 --   pass @pure ()@.
  432 -- * @taStartWiring@ — builds the 'AgentStartWiring' for this turn, given
  433 --   the per-turn 'SessionMeta' (after 'autoBindRepoAgent' may have rebound
  434 --   it). The per-surface builders (@webStartWiring@,
  435 --   @channelStartWiring@, @cliStartWiring@) are passed in here; W4
  436 --   collapses them into a single engine-owned @buildStartWiring@.

  451 newtype TurnOutcome = TurnOutcome { toError :: Maybe Text }

  453 -- | The single turn body (design §5.2). Replaces @plainTurn@ (Send.hs),
  454 -- @runTurnOnSession@ (Loop.hs), and @withCliTurn@ (Cli.hs).
  455 --
  456 -- The engine owns (per the §5.2 step table): load meta, resolve provider,
  457 -- broadcast "thinking", the per-session lock (OUTSIDE the transcript
  458 -- bracket — serializes the entire turn), the transcript bracket,
  459 -- mkSessionExec, workdir-aware backends, the system prompt, the ISA
  460 -- registry, AgentEnv construction, runTurn, broadcast "idle" + new entries,
  461 -- reply fan-out, and auto-tab. The adapter owns: 'replySubscribe' +
  462 -- 'replyFanoutMessage' (@taPreTurn@, runs before the lock), the channel
  463 -- label, the on-stop/on-user-message hooks, the start-wiring builder, and
  464 -- the post-turn @lists@ refresh (@taPostTurn@).
  465 --
  466 -- Broadcast timing + exception safety: the "thinking" signal fires before
  467 -- the lock; the "idle" signal + 'fanoutLastReply' run in a 'bracket' cleanup
  468 -- so a turn killed mid-way (success, sync exception, async 'ThreadKilled')
  469 -- cannot leave the tab stuck in "thinking". This matches the existing web +
  470 -- channel paths; the CLI converges to the same guarantee.
  471 runSessionTurn
  472   :: TurnDeps -> TurnAdapter -> SessionMeta -> Maybe MessageSource -> Text -> IO TurnOutcome
  473 runSessionTurn td adapter meta mSrc t = do
  474   let paths = tdPaths td
  475       sid   = smId meta
  476       sessionDirPath = sessionDir paths sid
  477   eprov <- tdResolve td meta
  478   case eprov of
  479     Left err -> do
  480       -- Provider resolution failed: still run the adapter's pre-turn hook so
  481       -- the adapter isn't left in an inconsistent state (e.g. a channel
  482       -- handle that subscribed but never sees the turn complete). The
  483       -- engine's bracket is NOT entered (no "thinking" signal was sent).
  484       taPreTurn adapter sid meta t
  485       pure (TurnOutcome (Just err))
  486     Right (prov, model) -> do
  487       createDirectoryIfMissing True sessionDirPath
  488       saveSessionMeta paths meta
  489       -- [adapter] Pre-turn: subscribe + cross-channel mirror (inbox-driven
  490       -- channels); no-op for web/CLI.
  491       taPreTurn adapter sid meta t
  492       -- [engine] Broadcast "thinking" (before the lock, matches the existing
  493       -- web + channel paths).
  494       broadcastHarnessStatus (tdBroker td) sid "thinking"
  495       mErr <- bracket
  496         (pure ())
  497         (\_ -> do
  498           -- [engine] Guaranteed cleanup: signal idle + fan out the last
  499           -- assistant reply to subscribed chat channels (so a turn that
  500           -- dies mid-way still releases the tab + delivers any partial
  501           -- reply).
  502           broadcastHarnessStatus (tdBroker td) sid "idle"

  517 -- + AgentEnv, run 'runTurn', broadcast new entries, return 'Just' on error.

  520 -- | Build the agent-metadata cache env for a scan, or 'Nothing' when the
  521 -- wiring can't support it (local mode, remote unconfigured, tests with no
  522 -- runner). Remote-only by design: in local mode the workdir IS local, so
  523 -- the per-file crawl has no SSH round trips to amortize.
  524 -- | Build the agent-metadata cache env for a scan, or 'Nothing' when the
  525 -- wiring can't support it (local mode, remote unconfigured/incomplete,
  526 -- security-config load failure). Remote-only by design: in local mode the
  527 -- workdir IS local, so the per-file crawl has no SSH round trips to
  528 -- amortize. Logs its decision at debug so fallbacks are diagnosable.
  529 metaCacheEnvFor
  530   :: TurnDeps -> SealPaths -> Either Text SecurityConfig
  531   -> IO (Maybe MetaCacheEnv)
  532 metaCacheEnvFor td paths eSecCfg = do
  533   let mEnv = case (tdIsRemote td, eSecCfg) of
  534         (True, Right secCfg) ->
  535           case untrustedExecConfigFromSecurity secCfg of
  536             Just (UntrustedExecConfig UemRemote (Just sshCfg)) -> Just MetaCacheEnv
  537               { mceRunner       = fromMaybe mkRealRemoteRunner (tdRemoteRunner td)
  538               , mceSshCfg       = sshCfg
  539               , mceCacheRoot    = agentMetaCacheDir paths
  540               , mceRunTransfer  = rsyncTransferIO sshCfg
  541               }
  542             _ -> Nothing
  543         _ -> Nothing
  544       reason = case mEnv of
  545         Just _  -> "Just (snapshot routing active)"
  546         Nothing -> case (tdIsRemote td, eSecCfg) of
  547           (False, _) -> "Nothing (local mode)"
  548           (_, Left err) ->
  549             "Nothing (security config load failed: " ++ T.unpack err ++ ")"
  550           (_, Right secCfg) ->
  551             case untrustedExecConfigFromSecurity secCfg of
  552               Just uec | uecMode uec == UemRemote ->
  553                 "Nothing (mode=remote but ssh config missing/incomplete)"
  554               _ -> "Nothing (mode is not remote)"
  555   globalLogIO K2.DebugS (K2.ls (T.pack ("[agent-meta] env=" <> reason)))
  556   pure mEnv

  558 runTurnBody
  559   :: TurnDeps -> TurnAdapter -> SessionMeta -> Maybe MessageSource -> Text
  560   -> SessionId -> SealPaths
  561   -> SomeProvider -> ModelId -> TwoFileHandle
  562   -> IO (Maybe Text)
  563 runTurnBody td adapter meta mSrc t sid paths prov model tHandle = do
  564   appEnv <- mkEnv (tdLogger td) defaultConfig
  565   eCfg <- loadRuntimeConfig (prConfigPath (tdProvider td))
  566   eSecCfg <- loadSecurityConfig (securityFilePath paths)
  567   let operatorCeiling = either (const defaultRetrievalMaxScanBytes) retrievalMaxScanBytes eCfg
  568   cloneDeps <- mkCloneDepsTurn td
  569   exec <- either (\_ _ _ _ -> pure (failClosedSessionExec cloneDeps))
  570                  (\sc _sid _cd runner -> cachedSessionExec (tdExecCache td) paths sc sid cloneDeps runner)
  571                  eSecCfg sid cloneDeps (fromMaybe mkRealRemoteRunner (tdRemoteRunner td))
  572   let wfs    = seWorkdirFs exec
  573       wsRoot = seWorkspaceRoot exec
  574       uioEnv = seUIOEnv exec
  575   -- [engine] One discovery scan per session (cache-backed): the result
  576   -- feeds autoBindRepoAgent AND both per-turn workdir backends, so a turn
  577   -- costs zero extra scans after the first (or after SETUP_REPO).
  578   metaEnv <- metaCacheEnvFor td paths eSecCfg
  579   (workdirDefs, workdirSkillsList) <- cachedWorkdirScan (tdExecCache td) sid wfs wsRoot metaEnv
  580   workdirSkills <- Skill.staticSkillBackend workdirSkillsList
  581   workdirAgentDefs <- Def.staticAgentDefBackend workdirDefs
  582   -- [engine] autoBindRepoAgent (design §5.2 step 3b — currently web-only;
  583   -- unifying means channels/CLI gain it, the desired behavioural
  584   -- convergence). Runs over the SAME cached defs — no extra scan.
  585   autoBindRepoAgentWith workdirDefs paths sid
  586   mMetaAfterBind <- loadSessionMeta paths sid
  587   let meta' = fromMaybe meta mMetaAfterBind
  588   -- [engine] Workdir-aware backends (per-turn merge; tdBaseBackends is the
  589   -- base, never mutated). The workdir halves are static backends over the
  590   -- cache's scan result (see above).
  591   let sessionSkills = Skill.tripleUnionSkillBackend workdirSkills (bSkills (tdBaseBackends td))
  592       sessionBackends = (tdBaseBackends td)
  593         { bAgentDefs = Def.unionAgentDefBackend workdirAgentDefs (bAgentDefs (tdBaseBackends td)) }
  594       agentDefBackend = bAgentDefs sessionBackends
  595       autoloadId = either (const Nothing) resolvedAutoloadSkill eCfg
  596       injectCatalog = either (const True) resolvedAvailableSkills eCfg
  597       parallel = either (const True) resolvedParallelToolGuidance eCfg
  598       toolUse = either (const True) resolvedToolUseEnforcement eCfg
  599       taskCompletion = either (const True) resolvedTaskCompletionGuidance eCfg
  600   -- [engine] System prompt (single resolver, honors smSystemOverride).
  601   mSystem <- resolveSystemPrompt agentDefBackend sessionSkills
  602               autoloadId injectCatalog parallel toolUse taskCompletion meta'
  603   turnAbortFlag <- lookupOrCreateAbortFlag (tdAbortReg td) sid
  604   let onDemand = either (const False) onDemandSchemas eCfg
  605       startWiring = taStartWiring adapter sessionBackends sid appEnv eCfg operatorCeiling meta'
  606       isaReg = buildSessionRegistry (tdVault td) cloneDeps sessionBackends wsRoot sid operatorCeiling
  607                  (tdAutonomy td) (either (const Nothing) rcWeb eCfg) startWiring
  608                  (tdHarnessReg td) (tdTmuxRunner td) (tdHttpManager td) (taCaps adapter) onDemand
  609   tfwSetSecretOps tHandle (ISA.secretOpNames isaReg)
  610   let onEntry = broadcastNewEntries (tdBroker td) paths sid (modelText model) (smCreatedAt meta')
  611       env = (mkSessionAgentEnv TurnEnv
  612               { teCaps          = taCaps adapter
  613               , teProvider      = prov
  614               , teProviderLabel = smProvider meta'
  615               , teModel         = model
  616               , teSession       = sid
  617               , teSystem        = mSystem
  618               , teRegistry      = isaReg
  619               , teTranscript    = tHandle
  620               , teUioEnv        = uioEnv
  621               , teDebugReqPath  = debugRequestsPath paths sid eCfg
  622               , teAutonomy      = tdAutonomy td
  623               , teApprovals     = tdApprovals td
  624               , teOnEntry       = onEntry
  625               , teOnDemand      = onDemand
  626               , teLogPath       = Just (sessionLogPath paths sid)
  627               , teMaxTurns      = either (const defaultMaxTurns) maxTurnsConfig eCfg
  628               , teOnUserMessage = taOnUserMessage adapter meta'
  629               , teChannel       = taChannelLabel adapter meta'
  630               , teOnStop        = taOnStop adapter sid
  631               , teAbortFlag     = turnAbortFlag
  632               , teToolTimeout   = either (const defaultToolTimeoutConfig) toolTimeoutConfig eCfg
  633               })
  634             { aeMessageSource = mSrc }
  635   eResult <- withExceptionLogging (tdLogger td) (Just (sessionLogPath paths sid)) "turn" $
  636     runApp appEnv (runTurn env t)
  637   case eResult of
  638     Left errMsg -> pure (Just errMsg)
  639     Right _     -> do
  640       -- [engine] Broadcast new transcript entries (one canonical post-turn
  641       -- sequence; the per-entry live broadcast already fired via aeOnEntry).
  642       broadcastNewEntries (tdBroker td) paths sid (modelText model) (smCreatedAt meta')
  643       pure Nothing

  645 -- | Record a synthetic 'EKRequest' preamble entry (convLen=0, full envelope,

  647 -- + Tools rows appear above the first message, not interleaved with it. The
  648 -- entry carries the resolved system prompt + the ISA registry's tool defs so
  649 -- the frontend can render both rows from this single entry. The preamble is
  650 -- recorded ONCE per session (the caller checks for prior EKRequest entries).

  654   now <- getCurrentTime
  655   let env0 = EnvelopeDelta
  656         { edModel = Just model
  657         , edSystem = Just mSystem
  658         , edTools = Just (ISA.registryToolDefs isaReg)
  659         , edToolChoice = Just ToolAuto
  660         , edMaxTokens = Just defaultMaxTokens
  661         }
  662       entry = EntryRecord
  663         { erId = ""
  664         , erTimestamp = now
  665         , erKind = EKRequest
  666         , erConvLen = 0
  667         , erEnvelope = Just env0
  668         , erUsage = Nothing
  669         , erStop = Nothing
  670         , erDurationMs = Nothing
  671         , erHarness = Nothing
  672         , erCorrelation = Nothing
  673         , erMeta = Map.empty
  674         }
  675   tfwRecordAndAck tHandle (TwoFileWrite [] entry)

  725 -- | Fan out the last assistant reply to subscribed chat channels, and emit
  726 -- a @reply-delivered@ signal when ≥1 chat channel received it (so the web
  727 -- sidebar marks the tab Idle Read). No-op when the session has no
  728 -- @conversation.json@ (CLI/tests).
  729 fanoutLastReply :: ReplyRegistry -> Maybe StreamBroker -> SealPaths -> SessionId -> IO ()
  730 fanoutLastReply replies mBroker paths sid = do
  731   let convPath = sessionConversationPath paths sid
  732   exists <- doesFileExist convPath
  733   if not exists
  734     then pure ()
  735     else do
  736       raw <- readFileTextStrict convPath
  737       let lines' = filter (not . T.null) (T.lines raw)
  738           msgs = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8) lines' :: [Message]
  739       for_ (lastAssistantText msgs) $ \reply -> do
  740         replyFanout replies sid reply
  741         count <- replySubscriberCount replies sid
  742         when (count > 0) (broadcastReplyDelivered mBroker sid)

  748     (m : _) -> case [t | CbText t <- msgContent m] of

  750       []      -> Nothing
  751     []      -> Nothing

  777 -- | The unified call dispatcher (design §5.3 — replaces @webCallDispatcher@,
  778 -- @channelCallDispatcher@, and the CLI's inline @callDispatcher@). Takes
  779 -- explicit 'TurnDeps' + 'SessionId' + 'ChannelCaps' + the channel label (for
  780 -- the transcript entry's @erMeta.channel@). Opens the transcript, builds the
  781 -- session's ISA registry, dispatches the opcode via
  782 -- 'Seal.ISA.Dispatch.dispatch' under 'Full' autonomy semantics (the operator
  783 -- is the approver by typing @/call@), records the result, and broadcasts new
  784 -- entries.
  785 --
  786 -- SETUP_REPO convergence (W4): the unified dispatcher runs
  787 -- 'recordSetupRepoResult' + 'autoBindRepoAgent' + 'broadcastAgentDefsChanged'
  788 -- for ALL surfaces (was web-only). This matches W3's autoBindRepoAgent
  789 -- convergence — channels/CLI gain the auto-bind behavior.
  790 callDispatcher
  791   :: TurnDeps -> ChannelCaps -> SessionId -> Text
  792   -> OpName -> Value -> IO (Either DispatchError OpResult)
  793 callDispatcher td caps sid channelLabel callOpName val = do
  794   let paths = tdPaths td
  795       sessionDirPath = sessionDir paths sid
  796   createDirectoryIfMissing True sessionDirPath
  797   mMeta <- loadSessionMeta paths sid
  798   withTwoFileTranscript sessionDirPath $ \tHandle -> do
  799     appEnv <- mkEnv (tdLogger td) defaultConfig
  800     eCfg <- loadRuntimeConfig (prConfigPath (tdProvider td))
  801     eSecCfg <- loadSecurityConfig (securityFilePath paths)
  802     let operatorCeiling = either (const defaultRetrievalMaxScanBytes) retrievalMaxScanBytes eCfg
  803     cloneDeps <- mkCloneDepsTurn td
  804     exec <- either (\_ _ _ _ -> pure (failClosedSessionExec cloneDeps))
  805                    (\sc _sid _cd runner -> cachedSessionExec (tdExecCache td) paths sc sid cloneDeps runner)
  806                    eSecCfg sid cloneDeps (fromMaybe mkRealRemoteRunner (tdRemoteRunner td))
  807     let wfs = seWorkdirFs exec
  808         wsRoot = seWorkspaceRoot exec
  809         uioEnv = seUIOEnv exec
  810     metaEnv <- metaCacheEnvFor td paths eSecCfg
  811     (workdirDefs, _) <- cachedWorkdirScan (tdExecCache td) sid wfs wsRoot metaEnv
  812     workdirAgentDefs <- Def.staticAgentDefBackend workdirDefs
  813     let onDemand = either (const False) onDemandSchemas eCfg
  814         sessionBackends = (tdBaseBackends td)
  815           { bAgentDefs = Def.unionAgentDefBackend workdirAgentDefs (bAgentDefs (tdBaseBackends td)) }
  816         startWiring = buildStartWiring td sessionBackends sid appEnv eCfg operatorCeiling channelLabel
  817         isaReg = buildSessionRegistry (tdVault td) cloneDeps sessionBackends wsRoot sid operatorCeiling
  818                    (tdAutonomy td) (either (const Nothing) rcWeb eCfg) startWiring
  819                    (tdHarnessReg td) (tdTmuxRunner td) (tdHttpManager td) caps onDemand
  820     tfwSetSecretOps tHandle (ISA.secretOpNames isaReg)
  821     callAbortFlag <- lookupOrCreateAbortFlag (tdAbortReg td) sid
  822     res <- runApp appEnv (dispatch isaReg tHandle localBackend uioEnv (either (const defaultToolTimeoutConfig) toolTimeoutConfig eCfg) callAbortFlag callOpName val)
  823     case res of
  824       Right r -> do
  825         let opNm = case callOpName of OpName n -> n
  826         if opNm == "SETUP_REPO"
  827           then do
  828             recordSetupRepoResult tHandle callOpName val r (Just channelLabel)
  829             unless (orIsError r) $ do
  830               -- The clone changed the workdir's structure: drop the cached
  831               -- discovery scan, re-scan fresh, and bind from the fresh defs.
  832               invalidateWorkdirScan (tdExecCache td) sid
  833               freshMetaEnv <- metaCacheEnvFor td paths eSecCfg
  834               (freshDefs, freshSkillsList) <-
  835                 cachedWorkdirScan (tdExecCache td) sid wfs wsRoot freshMetaEnv
  836               autoBindRepoAgentWith freshDefs paths sid
  837               -- Record a preamble entry (System Prompt + Tools) now that
  838               -- the workdir is set up and the agent is bound. This makes
  839               -- the System Prompt + Tools rows appear in the frontend
  840               -- BEFORE the first user message, not after it. Only recorded
  841               -- when no prior EKRequest entry exists (first time only).
  842               priorEntries <- tfwReadEntries tHandle
  843               unless (any (\e -> erKind e == EKRequest) priorEntries) $ do
  844                 mMetaAfterBind <- loadSessionMeta paths sid
  845                 let meta' = fromMaybe metaFallback mMetaAfterBind
  846                     metaFallback = fromMaybe (error "callDispatcher: SETUP_REPO without session meta") mMeta
  847                     autoloadId = either (const Nothing) resolvedAutoloadSkill eCfg
  848                     injectCatalog = either (const True) resolvedAvailableSkills eCfg
  849                     parallel = either (const True) resolvedParallelToolGuidance eCfg
  850                     toolUse = either (const True) resolvedToolUseEnforcement eCfg
  851                     taskCompletion = either (const True) resolvedTaskCompletionGuidance eCfg
  852                 freshSkills <- Skill.staticSkillBackend freshSkillsList
  853                 -- Rebuild sessionBackends with the FRESH agent defs (after
  854                 -- autoBindRepoAgent re-bound the agent to the repo's
  855                 -- .agents/agents.md). The original sessionBackends was built
  856                 -- BEFORE the fresh scan, so it doesn't include the repo's
  857                 -- agent definition — using it would produce a preamble with
  858                 -- a stale (or missing) system prompt.
  859                 freshAgentDefs <- Def.staticAgentDefBackend freshDefs
  860                 let freshBackends = sessionBackends
  861                       { bAgentDefs = Def.unionAgentDefBackend freshAgentDefs (bAgentDefs (tdBaseBackends td)) }
  862                     sessionSkills = Skill.tripleUnionSkillBackend freshSkills (bSkills (tdBaseBackends td))
  863                 mSystem <- resolveSystemPrompt
  864                   (bAgentDefs freshBackends) sessionSkills
  865                   autoloadId injectCatalog parallel toolUse taskCompletion meta'
  866                 let model = maybe (ModelId "") (ModelId . smModel) mMetaAfterBind
  867                 recordPreamble tHandle model mSystem isaReg
  868             broadcastAgentDefsChanged (tdBroker td)
  869           else recordSkillLoadResult tHandle callOpName val r (Just channelLabel)
  870         case mMeta of
  871           Just meta -> broadcastNewEntries (tdBroker td) paths sid (smModel meta) (smCreatedAt meta)
  872           Nothing   -> pure ()
  873       Left _  -> pure ()
  874     pure res

  907 -- | The unified AGENT_START worker-builder (design §5.3 — replaces
  908 -- @webMkWorker@, @channelMkWorker@, @cliMkWorker@). Resolves the def's
  909 -- provider+model (falling back to the parent session meta when the def
  910 -- fields are empty), opens a fresh two-file transcript under the parent
  911 -- session's agents dir, builds a narrowed child ISA registry (the unified
  912 -- 'buildChildRegistry'), and runs 'runTurn' with the goal as the first user
  913 -- message.
  914 buildWorker
  915   :: TurnDeps -> SessionId -> Env -> Either a RuntimeConfig -> Int -> Text
  916   -> AgentWorkerBuilder
  917 buildWorker td parentSid appEnv eCfg operatorCeiling channel =
  918   mkDelegateWorker DelegationWorkerDeps
  919     { dwdPaths = tdPaths td
  920     , dwdParentSid = parentSid
  921     , dwdAppEnv = appEnv
  922     , dwdMkUIOEnv = \childSid -> do
  923         childCloneDeps <- mkCloneDepsTurn td
  924         eSecCfg <- loadSecurityConfig (securityFilePath (tdPaths td))
  925         seUIOEnv <$> either (\_ _ _ _ -> pure (failClosedSessionExec childCloneDeps))
  926                             (\sc _sid _cd runner -> cachedSessionExec (tdExecCache td) (tdPaths td) sc childSid childCloneDeps runner)
  927                             eSecCfg childSid childCloneDeps (fromMaybe mkRealRemoteRunner (tdRemoteRunner td))
  928     , dwdAutonomy = tdAutonomy td
  929     , dwdApprovals = tdApprovals td
  930     , dwdOnDemand = either (const False) onDemandSchemas eCfg
  931     , dwdParentDepth = 0
  932     , dwdResolveProvider = resolveChild
  933     , dwdChildRegistry = buildChildRegistryAdapter td eCfg operatorCeiling
  934     , dwdChildSystemPrompt = childSystemPrompt td eCfg
  935     , dwdOnEntry = pure ()
  936     , dwdChannel = channel
  937     , dwdAbortFlag = lookupOrCreateAbortFlag (tdAbortReg td)
  938     , dwdToolTimeout = either (const defaultToolTimeoutConfig) toolTimeoutConfig eCfg
  939     }
  940   where
  941     resolveChild agentDef = do
  942       mParentMeta <- loadSessionMeta (tdPaths td) parentSid
  943       now <- getCurrentTime
  944       let parent = fromMaybe (fallbackMeta now) mParentMeta
  945           fallBackProvider = if T.null (adProvider agentDef) then smProvider parent else adProvider agentDef
  946           fallBackModel = case adModel agentDef of
  947             ModelId m | T.null m -> smModel parent
  948                       | otherwise -> m
  949       resolveDefProvider (tdProvider td) fallBackProvider (ModelId fallBackModel)
  950     fallbackMeta tnow = SessionMeta
  951       { smId = parentSid, smProvider = "ollama", smModel = "glm-5.2:cloud"
  952       , smChannel = "cli", smAgent = Nothing, smSystemOverride = Nothing, smAgentName = Nothing
  953       , smDescription = Nothing
  954       , smCreatedAt = tnow, smLastActive = tnow }

  956 -- | Build a narrowed child registry via the unified 'buildChildRegistry'.

  972 -- | Resolve the child agent's system prompt. Mirrors the three original
  973 -- @*ChildSystemPrompt@ helpers: the def's 'adSystem' (with the task context
  974 -- appended) + the static guidance + autoload skill + available-skills
  975 -- catalog. Uses 'tdBaseBackends' (the child registry doesn't get
  976 -- workdir-aware skills — matching the original implementations).
  977 childSystemPrompt
  978   :: TurnDeps -> Either a RuntimeConfig
  979   -> AgentDef -> ChildTask -> IO (Maybe Text)
  980 childSystemPrompt td eCfg agentDef task = do
  981   let base = adSystem agentDef
  982       ctx  = ctContext task
  983       basePrompt = case (base, ctx) of
  984         (Just b, Just c) | not (T.null c) -> Just (b <> "\n\nCONTEXT:\n" <> c)
  985         (Just b, _)                       -> Just b
  986         (Nothing, Just c)                 -> Just ("CONTEXT:\n" <> c)
  987         (Nothing, Nothing)                -> Nothing
  988       autoloadId = either (const Nothing) resolvedAutoloadSkill eCfg
  989       injectCatalog = either (const True) resolvedAvailableSkills eCfg
  990       parallel = either (const True) resolvedParallelToolGuidance eCfg
  991       toolUse = either (const True) resolvedToolUseEnforcement eCfg
  992       taskCompletion = either (const True) resolvedTaskCompletionGuidance eCfg
  993       withGuidance = injectStaticGuidance parallel toolUse taskCompletion basePrompt
  994   withAutoload <- injectAutoloadSkill (bSkills (tdBaseBackends td)) autoloadId withGuidance
  995   if injectCatalog
  996     then injectAvailableSkills (bSkills (tdBaseBackends td)) withAutoload
  997     else pure withAutoload

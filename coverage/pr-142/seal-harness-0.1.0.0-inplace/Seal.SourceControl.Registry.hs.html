    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Load and save @config\/repos.toml@ — the source-control repo registry
    3 -- ('RepoRegistry'), a keyed-by-id map of 'SourceRepo's the harness may clone
    4 -- (design §4.3, §4.8). Absent file decodes as an empty registry. Writes are
    5 -- atomic (write @.tmp@, rename). The load-decode path runs
    6 -- 'normalizeReposTable' before 'Toml.runTomlCodec' so both the idiomatic
    7 -- @[repos.\<id\>]@-only style and the encoder's explicit-@[repos]@-header
    8 -- style decode correctly (the W2 DoD).
    9 --
   10 -- This module mirrors 'Seal.Config.File' (load/save/update) but with a
   11 -- dedicated 'repoRegistryWriteLock' — it does NOT share 'configWriteLock',
   12 -- so a slow repo write can never block a config read and vice-versa.
   13 module Seal.SourceControl.Registry
   14   ( RepoRegistry(..)
   15   , loadRepoRegistry, saveRepoRegistry, updateRepoRegistry
   16   , upsertRepo, removeRepo, lookupRepo
   17   , RepoRegistryHandle(..), mkRepoRegistryHandle
   18   ) where
   19 
   20 import Control.Concurrent.MVar (MVar, newMVar, withMVar)
   21 import Data.Functor ((<&>))
   22 import Data.Map.Strict qualified as Map
   23 import Data.Text (Text)
   24 import Data.Text.IO qualified as TIO
   25 import System.Directory (doesFileExist, renameFile)

   34 ----------------------------------------------------------------------------
   35 -- Load / Save (mirror loadRuntimeConfig / saveRuntimeConfig)
   36 ----------------------------------------------------------------------------
   37 
   38 -- | Load the repo registry at @path@.
   39 --
   40 -- * File absent  → @Right ('RepoRegistry' 'Map.empty')@
   41 -- * Parse error  → @Left@ with the rendered tomland parser error
   42 -- * Decode error → @Left@ with 'Toml.prettyTomlDecodeErrors'
   43 loadRepoRegistry :: FilePath -> IO (Either Text RepoRegistry)
   44 loadRepoRegistry path = do
   45   exists <- doesFileExist path
   46   if not exists
   47     then pure (Right (RepoRegistry Map.empty))
   48     else do
   49       contents <- TIO.readFile path
   50       pure $ case Toml.parse contents of
   51         Left err   -> Left (Toml.unTomlParseError err)
   52         Right toml -> case Toml.runTomlCodec repoRegistryCodec (normalizeReposTable toml) of
   53           Success m   -> Right (RepoRegistry m)
   54           Failure errs -> Left (Toml.prettyTomlDecodeErrors errs)
   55 
   56 -- | Save @rr@ to @path@ atomically: write @path.tmp@, rename over @path@.
   57 saveRepoRegistry :: FilePath -> RepoRegistry -> IO ()
   58 saveRepoRegistry path rr = do
   59   let encoded = Toml.encode repoRegistryCodec (rrRepos rr)
   60       tmp     = path <> ".tmp"
   61   TIO.writeFile tmp encoded
   62   renameFile tmp path
   63 
   64 ----------------------------------------------------------------------------
   65 -- Process-wide write lock (mirror configWriteLock — DEDICATED, not shared)
   66 ----------------------------------------------------------------------------
   67 
   68 -- | Process-wide lock serializing repo-registry writes to prevent
   69 -- lost-update races (design V7). Multiple concurrent 'updateRepoRegistry'
   70 -- callers (e.g. REST PUT + a future REPO_UPDATE opcode) can silently
   71 -- clobber each other without this lock. The MVar is initialized once via
   72 -- 'unsafePerformIO' — idiomatic for a process-wide lock (mirrors
   73 -- 'Seal.Config.File.configWriteLock' and 'Seal.Security.Vault.stWriteLock').
   74 -- This lock is DEDICATED to the repo registry; it is NOT shared with
   75 -- 'configWriteLock' so a slow repo write cannot block a config read.
   76 {-# NOINLINE repoRegistryWriteLock #-}
   77 repoRegistryWriteLock :: MVar ()
   78 repoRegistryWriteLock = unsafePerformIO (newMVar ())
   79 
   80 -- | Load the registry at @path@, apply @f@, save. Propagates any load error
   81 -- as @Left Text@ without writing. The load-modify-save is serialized behind
   82 -- 'repoRegistryWriteLock' to prevent lost-update races (design V7).
   83 updateRepoRegistry :: FilePath -> (RepoRegistry -> RepoRegistry) -> IO (Either Text ())
   84 updateRepoRegistry path f = withMVar repoRegistryWriteLock $ \_ -> do
   85   result <- loadRepoRegistry path
   86   case result of
   87     Left err  -> pure (Left err)
   88     Right rr  -> saveRepoRegistry path (f rr) >> pure (Right ())
   89 
   90 ----------------------------------------------------------------------------
   91 -- Pure mutation helpers
   92 ----------------------------------------------------------------------------
   93 
   94 -- | Insert or overwrite a 'SourceRepo' (keyed by its 'srId').
   95 upsertRepo :: SourceRepo -> RepoRegistry -> RepoRegistry
   96 upsertRepo r (RepoRegistry m) = RepoRegistry (Map.insert (srId r) r m)
   97 
   98 -- | Delete the repo with the given 'RepoId' (no-op if absent).
   99 removeRepo :: RepoId -> RepoRegistry -> RepoRegistry
  100 removeRepo k (RepoRegistry m) = RepoRegistry (Map.delete k m)
  101 
  102 -- | Look up the repo with the given 'RepoId'.
  103 lookupRepo :: RepoId -> RepoRegistry -> Maybe SourceRepo
  104 lookupRepo k (RepoRegistry m) = Map.lookup k m
  105 
  106 ----------------------------------------------------------------------------
  107 -- Handle (closes over the path — used by the REST API in W4)
  108 ----------------------------------------------------------------------------
  109 
  110 -- | A handle bundling the two operations the REST API (W4) needs against the
  111 -- repo registry, with the registry path closed over. @rrhList@ returns
  112 -- 'Either' 'Text' so a corrupt @repos.toml@ surfaces as HTTP 500 (the AC5/S2
  113 -- mitigation), not a silent empty list. @rrhMutate@ wraps
  114 -- 'updateRepoRegistry' (atomic, lock-serialized).
  115 data RepoRegistryHandle = RepoRegistryHandle
  116   { rrhList   :: IO (Either Text [SourceRepo])
  117   , rrhMutate :: (RepoRegistry -> RepoRegistry) -> IO (Either Text ())
  118   }
  119 
  120 -- | Build a 'RepoRegistryHandle' closing over @path@.
  121 mkRepoRegistryHandle :: FilePath -> IO RepoRegistryHandle
  122 mkRepoRegistryHandle path = pure RepoRegistryHandle
  123   { rrhList   = loadRepoRegistry path <&> fmap (Map.elems . rrRepos)
  124   , rrhMutate = updateRepoRegistry path
  125   }

    3 -- under @config\/skills\/\<id\>.md@ (frontmatter + body). 'markdownSkillBackend'
    4 -- reads by enumerating the directory and writes by atomic file replace +
    5 -- auto-commit to the config git repo. 'noneBackend' (in-memory) is kept for
    6 -- tests.
    7 --
    8 -- The git repo is the versioning + audit layer; model-authored writes
    9 -- (@SKILL_CREATE@ \/ @SKILL_UPDATE@, which are Trusted file writes) auto-commit.
   10 -- Human file-drops are committed by the human via @git -C ~/.seal/config@.
   11 --
   12 -- The /workdir-scoped/ discovery backend (for @.skills\/@ discovery from
   13 -- cloned repos) operates over the 'WorkdirFs' handle via

   73 data SkillBackend = SkillBackend
   74   { sbCreate :: Skill -> IO ()
   75   -- ^ Insert or replace a skill by id (writes the file + auto-commits).
   76   , sbRead   :: SkillId -> IO (Maybe Skill)
   77   -- ^ Fetch one skill by id (reads its file).
   78   , sbList   :: IO [Skill]
   79   -- ^ All skills, sorted by id (deterministic for tests + git diffs).
   80   , sbUpdate :: Skill -> IO ()
   81   -- ^ Update an existing skill (same as 'sbCreate' for both backends).
   82   , sbDelete :: SkillId -> IO ()
   83   -- ^ Remove a skill by id (delete the file + auto-commit; idempotent).
   84   }

   86 -- | The in-memory backend: a single 'IORef' over a 'Map'. Used by tests.
   87 -- Kept as a fallback when no config repo is available.
   88 noneBackend :: IO SkillBackend
   89 noneBackend = do
   90   ref <- newIORef (Map.empty :: Map SkillId Skill)
   91   pure SkillBackend
   92     { sbCreate = \s -> modifyIORef' ref (Map.insert (skId s) s)
   93     , sbRead   = \sid -> Map.lookup sid <$> readIORef ref
   94     , sbList   = Map.elems <$> readIORef ref
   95     , sbUpdate = \s -> modifyIORef' ref (Map.insert (skId s) s)
   96     , sbDelete = modifyIORef' ref . Map.delete
   97     }

   99 -- | The Markdown backend. One file per skill under @dir@ (the @config/skills@
  100 -- directory), either flat (@\<id\>.md@) or grouped (@\<group\>\/\<id\>.md@).
  101 -- Writes are atomic (tmp → chmod 0600 → rename) and auto-committed to the
  102 -- config git repo. Reads enumerate the directory (both layouts).
  103 -- Malformed files are skipped (a partial write never breaks the list).
  104 markdownSkillBackend :: FilePath -> ConfigRepo -> IO SkillBackend
  105 markdownSkillBackend dir repo = pure SkillBackend
  106     { sbCreate = writeSkill dir repo
  107     , sbRead   = readSkill dir
  108     , sbList   = listSkills dir
  109     , sbUpdate = writeSkill dir repo
  110     , sbDelete = deleteSkill dir repo
  111     }

  113 -- | A read-layer union of a user 'SkillBackend' (the on-disk
  114 -- @~/.seal/config/skills/@ store) and the embedded built-in skills
  115 -- ('Seal.Skills.Builtins.builtinSkills'). Reads check the user layer first
  116 -- and fall back to the built-in; listing merges both, with the user copy
  117 -- winning on id collisions (so a user override shadows the built-in).
  118 -- Writes ('sbCreate'/'sbUpdate'/'sbDelete') go to the user layer only —
  119 -- built-ins are immutable from the model's perspective.
  120 --
  121 -- This is what makes Seal self-contained: the @seal-usage@ orientation
  122 -- skill is always present (shipped in the binary), and a user can override
  123 -- it by dropping @~/.seal/config/skills/seal-usage.md@. After a Seal
  124 -- upgrade, the user diffs their override against the new built-in (visible
  125 -- via @sbList@) and merges manually — no forced overwrites, no staleness.
  126 unionSkillBackend :: SkillBackend -> SkillBackend
  127 unionSkillBackend user = SkillBackend
  128     { sbCreate = sbCreate user
  129     , sbRead   = \sid -> do
  130         mUser <- sbRead user sid
  131         case mUser of
  132           Just s  -> pure (Just s)
  133           Nothing -> pure (Map.lookup sid builtinSkillMap)
  134     , sbList   = do
  135         userSkills <- sbList user
  136         let userMap = Map.fromList [(skId s, s) | s <- userSkills]
  137             merged = Map.union userMap builtinSkillMap
  138         pure (Map.elems merged)
  139     , sbUpdate = sbUpdate user
  140     , sbDelete = sbDelete user
  141     }

  143 -- | The conventional skill directories a cloned repo may carry, checked in
  144 -- order per repo. The first match per convention wins (a repo should only
  145 -- use one). Top-level only — no recursion into the repo.
  146 --
  147 -- @.skills@ follows the [agentskills.io](https://agentskills.io)
  148 -- specification: each subdirectory contains a @SKILL.md@ file with YAML
  149 -- frontmatter (@name@, @description@) + Markdown body. The other
  150 -- @.agents\/skills@ also uses the agentskills.io format (it is the skills
  151 -- sub-directory of the [.agents Protocol](https://dotagentsprotocol.com)).
  152 -- The other conventions use Seal's native flat/grouped @.md@ layout.
  153 workdirSkillConventions :: [FilePath]
  154 workdirSkillConventions = [ ".skills", ".agents/skills", ".seal/skills", ".claude/skills", "agents/skills" ]

  165 -- | A read-only 'SkillBackend' that scans a session workdir for skills
  166 -- shipped by cloned repositories. For each top-level directory in the
  167 -- workdir (a cloned repo), it checks the conventional skill locations
  168 -- (@.skills\/@, @.agents\/skills\/@, @.seal\/skills\/@,
  169 -- @.claude\/skills\/@, @agents\/skills\/@)
  170 -- and loads any skills there. The @.skills@ and @.agents\/skills@
  171 -- conventions use the agentskills.io directory-based format
  172 -- (@\<name\>\/SKILL.md@); the others
  173 -- use Seal's native flat/grouped @.md@ layout.
  174 --
  175 -- This backend is /read-only/: 'sbCreate'/'sbUpdate'/'sbDelete' are no-ops
  176 -- (repo-local skills are immutable from the model's perspective — the
  177 -- model writes skills to its user store via 'SKILL_WRITE', never into a
  178 -- cloned repo). 'sbRead' scans on every call (workdirs are small and the
  179 -- catalog is built once per turn); a future optimization could cache per
  180 -- session.
  181 --
  182 -- Within-workdir collisions (two repos ship a skill with the same id):
  183 -- the alphabetically-first repo wins (deterministic; the operator can
  184 -- control precedence by renaming a repo clone directory).
  185 --
  186 -- Every workspace read goes through the 'WorkdirFs' handle (symlink-escape
  187 -- confinement — §3.8; single chokepoint, §3.6) and is size-capped at
  188 -- 'maxScanBytes'.
  189 workdirSkillBackend :: WorkdirFs -> IO SkillBackend
  190 workdirSkillBackend fs = pure SkillBackend
  191     { sbCreate = \_ -> pure ()
  192     , sbRead   = \sid -> do
  193         skills <- listWorkdirSkills fs
  194         pure (Map.lookup sid (Map.fromList [(skId s, s) | s <- skills]))
  195     , sbList   = listWorkdirSkills fs
  196     , sbUpdate = \_ -> pure ()
  197     , sbDelete = \_ -> pure ()
  198     }

  200 -- | Enumerate every skill found under the conventional locations across
  201 -- all top-level directories (cloned repos) in the workdir anchored at the
  202 -- 'WorkdirFs'. The alphabetically-first repo wins on id collisions
  203 -- (deterministic). Missing or empty workdirs yield @[]@.
  204 --
  205 -- For the @.skills@ and @.agents\/skills@ conventions, skills are loaded from the agentskills.io
  206 -- directory format (each subdirectory contains a @SKILL.md@). For the
  207 -- other conventions, skills are loaded from Seal's native flat/grouped
  208 -- @.md@ layout.
  209 --
  210 -- /Round-trip economy/: all structural questions (which repos exist, which
  211 -- convention dirs exist, which entries are dirs or files) are answered
  212 -- from ONE 'wfsSnapshot' call; only file CONTENTS are read through
  213 -- 'wfsReadFile' (the symlink-containment chokepoint). On @mode=remote@
  214 -- this collapses a scan from one SSH round trip per probe to one round
  215 -- trip + two calls per discovered skill file.
  216 listWorkdirSkills :: WorkdirFs -> IO [Skill]
  217 listWorkdirSkills fs = do
  218   eSnap <- wfsSnapshot fs
  219   case eSnap of
  220     -- Fail-soft-to-empty: a missing/stub/failing workdir yields no skills.
  221     Left _  -> pure []
  222     Right snap -> listWorkdirSkillsSnap snap fs
  223 
  224 -- | 'listWorkdirSkills' over a caller-supplied snapshot — lets a consumer
  225 -- (the discovery cache) fetch ONE workspace snapshot and serve both
  226 -- scanners from it.
  227 listWorkdirSkillsSnap :: WorkspaceSnapshot -> WorkdirFs -> IO [Skill]
  228 listWorkdirSkillsSnap snap fs = do
  229   let repos = snapTopDirs snap
  230   perRepo <- forM repos $ \repo -> do
  231     raw <- concat <$> forM workdirSkillConventions (\conv -> do
  232       let convRp = repo <> "/" <> T.pack conv
  233       if not (snapIsDirectoryAt snap convRp)
  234         then pure []
  235         else if conv `elem` [".skills", ".agents/skills"]
  236                then do
  237                  -- agentskills.io format: <conv>/<name>/SKILL.md. The
  238                  -- SKILL.md existence check is folded into the read
  239                  -- (NotFound ⇒ skipped).
  240                  catMaybes <$> mapM (readAgentSkillDirect fs Nothing)
  241                                     (snapChildDirsAt snap convRp)
  242                else do
  243                  -- Seal native format: flat/grouped .md files
  244                  let flatMd =
  245                        [p | p <- snapChildFilesAt snap convRp
  246                           , ".md" `T.isSuffixOf` p]
  247                      groups = snapChildDirsAt snap convRp
  248                  flat <- mapM (\rel -> readAndStampGroup fs rel Nothing) flatMd
  249                  grouped <- forM groups $ \g ->
  250                    let gBase = snd (T.breakOnEnd "/" g)
  251                    in mapM (\rel -> readAndStampGroup fs rel (Just gBase))
  252                            [p | p <- snapChildFilesAt snap g
  253                               , ".md" `T.isSuffixOf` p]
  254                  pure (catMaybes (flat <> concat grouped)))
  255     -- Stamp each repo-local skill with a group derived from the repo
  256     -- directory name so the <available_skills> catalog groups them
  257     -- under a "<repo> project skills" heading.
  258     pure (mapMaybe (prefixWorkdirSkill repo . stampProjectGroup repo) raw)
  259   let merge m [] = m
  260       merge m (s:ss) = merge (Map.insertWith (\_new old -> old) (skId s) s m) ss
  261       merged = merge Map.empty (concat perRepo)
  262   pure (Map.elems merged)

  264 -- | A read-only backend serving a PRE-COMPUTED skill list (the discovery
  265 -- cache's point-in-time scan of the workdir). Mirrors
  266 -- 'workdirSkillBackend''s semantics: reads from memory, writes/deletes are
  267 -- no-ops (repo-local skills are immutable from the model's perspective).
  268 staticSkillBackend :: [Skill] -> IO SkillBackend
  269 staticSkillBackend skills = pure SkillBackend
  270   { sbCreate = \_ -> pure ()
  271   , sbRead   = \sid -> pure (Map.lookup sid (Map.fromList [(skId s, s) | s <- skills]))
  272   , sbList   = pure skills
  273   , sbUpdate = \_ -> pure ()
  274   , sbDelete = \_ -> pure ()
  275   }

  277 -- | Read one agentskills.io @\<dir\>\/SKILL.md@ skill by DIRECT read against
  278 -- a 'WorkdirFs' anchored at the workspace root (no existence probe first —
  279 -- 'WfsNotFound' ⇒ 'Nothing'). Mirrors 'readAgentSkillAt' minus the
  280 -- separate does-file-exist round trip. @dirRel@ is the full workspace-
  281 -- relative path of the skill directory.
  282 readAgentSkillDirect :: WorkdirFs -> Maybe Text -> Text -> IO (Maybe Skill)
  283 readAgentSkillDirect fs mGroup dirRel = do
  284   eContent <- wfsReadFile fs =<< rpOrDie (dirRel <> "/SKILL.md")
  285   case eContent of
  286     Left _   -> pure Nothing
  287     Right c -> pure (stampGroupDirect (decodeAgentSkillWithGroup mGroup c))
  288   where
  289     stampGroupDirect (Just s) = case skGroup s of
  290       Just _  -> Just s
  291       Nothing -> Just s { skGroup = mGroup }
  292     stampGroupDirect Nothing = Nothing
  293 
  294 -- | Stamp a repo-local skill's 'skGroup' with @"\<repo\> project skills"@
  295 -- so the @\<available_skills\>@ catalog groups them under a per-repo
  296 -- heading. A skill that already has a group (e.g. from the native grouped
  297 -- layout) keeps its existing group — only ungrouped skills are stamped.
  298 -- This mirrors how 'readAndStampGroup' fills 'skGroup' from the on-disk
  299 -- directory for user-store skills.
  300 stampProjectGroup :: Text -> Skill -> Skill
  301 stampProjectGroup repo s = case skGroup s of
  302   Just g | not (T.null (T.strip g)) -> s
  303   _ -> s { skGroup = Just (repo <> " project skills") }
  304 
  305 -- | Prefix a workdir-discovered skill's id with @\<repo\>--\<id\>@ so it
  306 -- never collides with a user-store skill of the same name (mirrors the
  307 -- agent-def pattern in 'Seal.Agent.Def.Backend.prefixWorkdirDef'). The
  308 -- @--@ separator is charset-safe per 'isValidSkillId'. If the prefixed id
  309 -- fails validation (e.g. the repo dir has a char outside the charset),
  310 -- the skill is dropped ('Nothing' — fail-closed).
  311 prefixWorkdirSkill :: Text -> Skill -> Maybe Skill
  312 prefixWorkdirSkill repo s =
  313   let prefixedIdText = repo <> "--" <> skillIdText (skId s)
  314   in case mkSkillId prefixedIdText of
  315        Left _ -> Nothing
  316        Right sid -> Just s { skId = sid }

  318 -- | A three-way union of a workdir backend (repo-local skills), a user
  319 -- backend, and the built-in skills. Workdir skill ids are namespaced
  320 -- (@\<repo\>--\<id\>@) so they never collide with user skills by design.
  321 -- On id collisions between user and built-in, user shadows built-in.
  322 -- Reads check workdir first, then user, then the built-in map. Listing
  323 -- merges all three with the same precedence. Writes go to the /user/
  324 -- backend only (repo and built-in skills are immutable from the model's
  325 -- perspective).
  326 --
  327 -- This is the backend wired into per-turn prompt construction so the
  328 -- @\<available_skills\>@ catalog and @SKILL_LOAD@ surface repo-local
  329 -- skills discovered by 'SETUP_REPO' (and the user store, and built-ins).
  330 tripleUnionSkillBackend :: SkillBackend -> SkillBackend -> SkillBackend
  331 tripleUnionSkillBackend workdir user = SkillBackend
  332     { sbCreate = sbCreate user
  333     , sbRead   = \sid -> do
  334         mWd <- sbRead workdir sid
  335         case mWd of
  336           Just s  -> pure (Just s)
  337           Nothing -> do
  338             mUser <- sbRead user sid
  339             case mUser of
  340               Just s  -> pure (Just s)
  341               Nothing -> pure (Map.lookup sid builtinSkillMap)
  342     , sbList   = do
  343         wdSkills <- sbList workdir
  344         userSkills <- sbList user
  345         let wdMap   = Map.fromList [(skId s, s) | s <- wdSkills]
  346             userMap = Map.fromList [(skId s, s) | s <- userSkills]
  347             merged = Map.union wdMap (Map.union userMap builtinSkillMap)
  348         pure (Map.elems merged)
  349     , sbUpdate = sbUpdate user
  350     , sbDelete = sbDelete user
  351     }

  353 -- | The filename for a skill in the flat layout: @\<id\>.md@.
  354 skillFile :: FilePath -> SkillId -> FilePath
  355 skillFile dir sid = dir </> T.unpack (skillIdText sid) <.> "md"
  356 
  357 -- | The full path for a skill write, honoring its group. A grouped skill
  358 -- lives at @\<root\>\/\<group\>\/\<id\>.md@; an ungrouped skill (or one
  359 -- whose group is empty) lives at @\<root\>\/\<id\>.md@ (back-compat with
  360 -- the flat layout). The group directory is created on write.
  361 skillPath :: FilePath -> Skill -> FilePath
  362 skillPath root skill =
  363   case skGroup skill of
  364     Just g | not (T.null (T.strip g)) ->
  365       root </> T.unpack (T.strip g) </> T.unpack (skillIdText (skId skill)) <.> "md"
  366     _ -> skillFile root (skId skill)

  368 -- | Write one skill to disk (atomic) and auto-commit. Honors 'skGroup':
  369 -- a grouped skill is written under @\<root\>\/\<group\>\/\<id\>.md@ (the
  370 -- group directory is created if absent); an ungrouped skill is written
  371 -- flat at @\<root\>\/\<id\>.md@. The git commit path is relative to the
  372 -- config repo root and encodes the group when present.
  373 writeSkill :: FilePath -> ConfigRepo -> Skill -> IO ()
  374 writeSkill root repo s = do
  375   let path = skillPath root s
  376       tmp  = path <.> "tmp"
  377       parent = takeDirectory path
  378   createDirectoryIfMissing True parent
  379   TIO.writeFile tmp (encodeSkill s)
  380   setFileMode tmp 0o600
  381   renameFile tmp path
  382   let rel = "skills" </> case skGroup s of
  383         Just g | not (T.null (T.strip g)) ->
  384           T.unpack (T.strip g) </> (T.unpack (skillIdText (skId s)) <.> "md")
  385         _ -> T.unpack (skillIdText (skId s)) <.> "md"
  386   _ <- gitCommitAll repo rel ("seal: SKILL write " <> skillIdText (skId s))
  387   pure ()

  389 -- | Read one skill by id. Probes, in order: the native grouped layout
  390 -- (@\<group\>\/\<id\>.md@), the agentskills.io grouped layout
  391 -- (@\<group\>\/\<id\>\/SKILL.md@), the native flat layout
  392 -- (@\<id\>.md@), then the agentskills.io top-level layout
  393 -- (@\<id\>\/SKILL.md@). Returns 'Nothing' if no file is found or every
  394 -- candidate is malformed. The returned skill's 'skGroup' is filled from the
  395 -- directory when the frontmatter omitted it (the common case), so a grouped
  396 -- file reads back as grouped regardless of whether its frontmatter
  397 -- redundantly declares a group.
  398 --
  399 -- The reads go through the 'WorkdirFs' handle (via 'userDirFs') so they
  400 -- share the single confined code path (§3.6) while remaining local-FS.
  401 readSkill :: FilePath -> SkillId -> IO (Maybe Skill)
  402 readSkill root sid = do
  403   let fs = userDirFs root
  404       base = skillIdText sid <> ".md"
  405       dir = skillIdText sid
  406   groups <- listSubdirs fs
  407   let nativeGrouped   = [ readAndStampGroup fs (g <> "/" <> base) (Just g) | g <- groups ]
  408       agentGrouped    = [ readAgentSkillAt fs (g <> "/" <> dir) (Just g) | g <- groups ]
  409       nativeFlat      = readAndStampGroup fs base Nothing
  410       agentTop        = readAgentSkillAt fs dir Nothing
  411   firstMatchM (nativeGrouped ++ agentGrouped ++ [nativeFlat, agentTop])
  412 
  413 -- | Read a skill file at @\<anchor\>\/\<rel\>@ (via the 'WorkdirFs') and,
  414 -- if it decodes, fill in 'skGroup' from the directory when the frontmatter
  415 -- omitted one. @mGroup@ is the group implied by the file's location
  416 -- ('Nothing' for the flat layout). @rel@ is a relative 'RemotePath' under
  417 -- the 'WorkdirFs' anchor.
  418 readAndStampGroup :: WorkdirFs -> Text -> Maybe Text -> IO (Maybe Skill)
  419 readAndStampGroup fs rel mGroup = do
  420   eContent <- wfsReadFile fs =<< rpOrDie rel
  421   case eContent of
  422     Left _ -> pure Nothing
  423     Right content -> case decodeSkill content of
  424       Nothing -> pure Nothing
  425       Just s  -> pure (Just (stampGroup s))
  426   where
  427     stampGroup s = case skGroup s of
  428       Just _  -> s
  429       Nothing -> s { skGroup = mGroup }

  431 -- | Enumerate the immediate sub-directories of the 'WorkdirFs' anchor
  432 -- (non-recursive, no hidden dirs), sorted for deterministic output.
  433 -- Returns the visible sub-directory names as 'Text' (suitable for building
  434 -- relative 'RemotePath's). Missing anchor yields @[]@.
  435 listSubdirs :: WorkdirFs -> IO [Text]
  436 listSubdirs fs = do
  437   eEntries <- wfsListDirectory fs =<< rpOrDie "."
  438   let entries = fromRight [] eEntries
  439       visible = [e | e <- entries, not (T.isPrefixOf "." e)]
  440   filterM' (wfsDoesDirectoryExist fs <=< rpOrDie) (sortOn id visible)

  442 -- | Like 'Control.Monad.filterM' but without the import (kept local to
  443 -- avoid widening the module's import surface for one call site).
  444 filterM' :: (a -> IO Bool) -> [a] -> IO [a]
  445 filterM' _ []     = pure []
  446 filterM' f (x:xs) = do
  447   ok <- f x
  448   rest <- filterM' f xs
  449   pure (if ok then x : rest else rest)
  450 
  451 -- | Return the first 'Just' in a list of IO actions, short-circuiting.
  452 firstMatchM :: [IO (Maybe a)] -> IO (Maybe a)
  453 firstMatchM []       = pure Nothing
  454 firstMatchM (a:as)   = do
  455   m <- a
  456   case m of
  457     Just x  -> pure (Just x)
  458     Nothing -> firstMatchM as

  460 -- | Enumerate all skills in @dir@: top-level @.md@ files (flat layout,
  461 -- back-compat) plus @.md@ files one directory down (grouped layout), plus
  462 -- agentskills.io @\<name\>\/SKILL.md@ skills at both levels (top-level and
  463 -- @\<group\>\/\<name\>\/SKILL.md@). Malformed files are skipped (a partial
  464 -- write never breaks the list). Each skill's 'skGroup' is filled from its
  465 -- directory when the frontmatter omitted one. Results are sorted by id for
  466 -- deterministic output.
  467 --
  468 -- The reads go through the 'WorkdirFs' handle (via 'userDirFs') so they
  469 -- share the single confined code path (§3.6) while remaining local-FS.
  470 listSkills :: FilePath -> IO [Skill]
  471 listSkills dir = do
  472   let fs = userDirFs dir
  473   topNative   <- listTopLevelSkills fs
  474   grouped     <- listGroupedSkills fs
  475   agentTop    <- listTopLevelAgentSkills fs
  476   agentGrouped <- listGroupedAgentSkills fs
  477   pure (sortOn (skillIdText . skId) (catMaybes (topNative ++ grouped ++ agentTop ++ agentGrouped)))
  478 
  479 -- | Read the flat-layout skills: @.md@ files directly under the 'WorkdirFs'
  480 -- anchor. Returns one 'Maybe Skill' per file (the outer list, not the inner
  481 -- Maybe, is the collection; 'Nothing' marks a malformed file to be
  482 -- 'catMaybes'-filtered).
  483 listTopLevelSkills :: WorkdirFs -> IO [Maybe Skill]
  484 listTopLevelSkills fs = do
  485   eEntries <- wfsListDirectory fs =<< rpOrDie "."
  486   let entries = fromRight [] eEntries
  487       mdFiles = [e | e <- entries, ".md" `T.isSuffixOf` e]
  488   forM mdFiles $ \e -> readAndStampGroup fs e Nothing
  489 
  490 -- | Read the grouped-layout skills: for each subdirectory @g@, read the
  491 -- @.md@ files under @g/@ and stamp 'skGroup' = @Just g@ when the
  492 -- frontmatter omitted one. Results from all groups are concatenated.
  493 listGroupedSkills :: WorkdirFs -> IO [Maybe Skill]
  494 listGroupedSkills fs = do
  495   groups <- listSubdirs fs
  496   results <- forM groups $ \g -> do
  497     let subFs = reanchorFs fs g
  498     eEntries <- wfsListDirectory subFs =<< rpOrDie "."
  499     let entries = fromRight [] eEntries
  500         mdFiles = [e | e <- entries, ".md" `T.isSuffixOf` e]
  501     forM mdFiles $ \e -> readAndStampGroup fs (g <> "/" <> e) (Just g)
  502   pure (concat results)
  503 
  504 -- | Read the agentskills.io skills at the top level: each immediate
  505 -- subdirectory @\<name\>@ of the anchor that contains a @SKILL.md@. The
  506 -- skill id comes from the @name@ frontmatter field (the spec requires it to
  507 -- match the directory name); the group is 'Nothing' (top-level skills are
  508 -- ungrouped, matching the flat native layout). Malformed skills are
  509 -- skipped ('Nothing').
  510 listTopLevelAgentSkills :: WorkdirFs -> IO [Maybe Skill]
  511 listTopLevelAgentSkills fs = do
  512   subdirs <- listSubdirs fs
  513   forM subdirs $ \subdir -> readAgentSkillAt fs subdir Nothing
  514 
  515 -- | Read the agentskills.io skills within Seal group directories: for each
  516 -- group @g@, each immediate subdirectory @\<name\>@ under @g/@ that contains
  517 -- a @SKILL.md@ is decoded and stamped with 'skGroup' = @'Just' g@ when the
  518 -- frontmatter omitted one. Results from all groups are concatenated. This
  519 -- composes the Seal grouped-layout convention with the agentskills.io
  520 -- per-skill directory convention (@\<group\>\/\<name\>\/SKILL.md@).
  521 listGroupedAgentSkills :: WorkdirFs -> IO [Maybe Skill]
  522 listGroupedAgentSkills fs = do
  523   groups <- listSubdirs fs
  524   results <- forM groups $ \g -> do
  525     let subFs = reanchorFs fs g
  526     subdirs <- listSubdirs subFs
  527     forM subdirs $ \subdir -> readAgentSkillAt fs (g <> "/" <> subdir) (Just g)
  528   pure (concat results)
  529 
  530 -- | Delete one skill file and auto-commit. Idempotent (no-op if the file is
  531 -- absent). Searches both the grouped and flat layouts: a skill written
  532 -- under a group is deleted from its group directory; a flat-layout skill
  533 -- is deleted from the root. If a file exists in both, both are removed.
  534 --
  535 -- This is a user-store /write/ operation: it stays on local 'FilePath'
  536 -- ('removeFile', 'doesFileExist') — 'WorkdirFs' is a read-only discovery
  537 -- handle. The group enumeration ('listSubdirs') goes through the
  538 -- 'WorkdirFs' handle (via 'userDirFs') so the read path stays unified.
  539 deleteSkill :: FilePath -> ConfigRepo -> SkillId -> IO ()
  540 deleteSkill dir repo sid = do
  541   let fs = userDirFs dir
  542       base = T.unpack (skillIdText sid) <.> "md"
  543   groups <- listSubdirs fs
  544   let groupedPaths = [ dir </> T.unpack g </> base | g <- groups ]
  545       flatPath     = skillFile dir sid
  546   candidates <- filterM' doesFileExist (groupedPaths ++ [flatPath])
  547   forM_ candidates $ \path -> do
  548     removeFile path
  549     let rel = "skills" </> fromMaybe base (stripRoot dir path)
  550     _ <- gitCommitAll repo rel ("seal: SKILL delete " <> skillIdText sid)
  551     pure ()
  552   where
  553     stripRoot r p =
  554       let r' = dropTrailingSlash r
  555           p' = dropTrailingSlash p
  556       in if r' `isPrefixPath` p'
  557            then Just (drop (length r' + 1) p')
  558            else Nothing
  559     dropTrailingSlash = reverse . dropWhile (== '/') . reverse
  560     isPrefixPath pre s = take (length pre) s == pre
  561

  566 -- | Enumerate skills in agentskills.io directory format: each subdirectory
  567 -- of the 'WorkdirFs' anchor contains a @SKILL.md@ file with YAML frontmatter
  568 -- (@name@, @description@) + Markdown body. Returns one 'Maybe Skill' per
  569 -- subdirectory ('Nothing' for malformed/missing @SKILL.md@). The skill id
  570 -- is taken from the @name@ frontmatter field (which the spec requires to
  571 -- match the parent directory name). The skill group is set to 'Nothing'
  572 -- (agentskills.io has no group concept; the directory name IS the id).
  573 listAgentSkillsDir :: WorkdirFs -> IO [Maybe Skill]
  574 listAgentSkillsDir fs = do
  575   subdirs <- listSubdirs fs
  576   forM subdirs $ \subdir -> readAgentSkillAt fs subdir Nothing
  577 
  578 -- | Read a @\<dir\>\/SKILL.md@ agentskills.io skill relative to a
  579 -- 'WorkdirFs' anchor, decoding it and stamping 'skGroup' = @mGroup@ when
  580 -- the frontmatter omitted one (mirrors 'readAndStampGroup' for the native
  581 -- codec). Returns 'Nothing' when the subdirectory has no @SKILL.md@ or the
  582 -- file fails to decode. Used by both the workdir scanner
  583 -- ('listAgentSkillsDir') and the user-store auto-detection
  584 -- ('listTopLevelAgentSkills' \/ 'listGroupedAgentSkills').
  585 readAgentSkillAt :: WorkdirFs -> Text -> Maybe Text -> IO (Maybe Skill)
  586 readAgentSkillAt fs dir mGroup = do
  587   let subFs = reanchorFs fs dir
  588   mdExists <- wfsDoesFileExist subFs =<< rpOrDie "SKILL.md"
  589   if not mdExists
  590     then pure Nothing
  591     else do
  592       eContent <- wfsReadFile subFs =<< rpOrDie "SKILL.md"
  593       case eContent of
  594         Left _   -> pure Nothing
  595         Right c -> pure (stampGroup (decodeAgentSkillWithGroup mGroup c))
  596   where
  597     stampGroup (Just s) = case skGroup s of
  598       Just _  -> Just s
  599       Nothing -> Just s { skGroup = mGroup }
  600     stampGroup Nothing = Nothing
  601 
  602 -- | Decode an agentskills.io @SKILL.md@ file into a 'Skill'. The frontmatter
  603 -- uses @name@ (required, maps to 'skId') and @description@ (required, maps to
  604 -- 'skDescription'). The body after the frontmatter becomes 'skBody'.
  605 -- Timestamps default to epoch zero (repo-local skills have no provenance
  606 -- timestamps). The session is set to a system "manual" session (matching
  607 -- the DirScheme agent def pattern). The 'skGroup' is set to the supplied
  608 -- stamp ('Nothing' for a top-level agentskills.io skill; @'Just' g@ for one
  609 -- discovered under a Seal group directory).
  610 decodeAgentSkill :: Text -> Maybe Skill
  611 decodeAgentSkill = decodeAgentSkillWithGroup Nothing
  612 
  613 -- | The general form of 'decodeAgentSkill' that pre-stamps 'skGroup'. Used
  614 -- internally so the user-store grouped scanner can fill the group from the
  615 -- on-disk directory (mirroring 'readAndStampGroup' for the native codec).
  616 decodeAgentSkillWithGroup :: Maybe Text -> Text -> Maybe Skill
  617 decodeAgentSkillWithGroup mGroup content =
  618   case decodeDoc content of
  619     (fm, body) -> do
  620       nameT <- fmLookup "name" fm
  621       sid  <- either (const Nothing) Just (mkSkillId nameT)
  622       Just Skill
  623         { skId = sid
  624         , skDescription = fromMaybe "" (fmLookup "description" fm)
  625         , skBody = body
  626         , skGroup = mGroup
  627         , skCreatedAt = agentSkillEpoch
  628         , skUpdatedAt = agentSkillEpoch
  629         , skSession = mkSystemSessionId "manual"
  630         }
  631   where
  632     agentSkillEpoch = UTCTime (fromGregorian 1970 1 1) (secondsToDiffTime 0)

  635 -- WorkdirFs re-anchoring + RemotePath helpers (internal)
  636 ----------------------------------------------------------------------------

  638 -- | Produce a 'WorkdirFs' "view" anchored at a sub-directory of the original
  639 -- anchor. Every 'wfs*' call on the re-anchored handle prepends @prefix@ to
  640 -- the supplied 'RemotePath'. Works for both the local and remote arms (no
  641 -- new constructor — pure record wrapper). The prefix is a relative path
  642 -- (e.g. @"my-repo\/.skills"@); the join uses @\/@ as the separator.
  643 -- (Mirrors 'Seal.Agent.Def.Workdir.reanchorFs'.)
  644 reanchorFs :: WorkdirFs -> Text -> WorkdirFs
  645 reanchorFs fs prefix = WorkdirFs
  646   { wfsReadFile            = wfsReadFile fs           <=< joinRp prefix
  647   , wfsDoesFileExist       = wfsDoesFileExist fs      <=< joinRp prefix
  648   , wfsDoesDirectoryExist  = wfsDoesDirectoryExist fs <=< joinRp prefix
  649   , wfsListDirectory       = wfsListDirectory fs      <=< joinRp prefix
  650   , wfsFileSize            = wfsFileSize fs           <=< joinRp prefix
  651   , wfsModificationTime    = wfsModificationTime fs   <=< joinRp prefix
  652   , wfsSnapshot            = wfsSnapshot fs
  653     -- ^ Whole-workspace: re-anchoring changes where relative reads land,
  654     -- never the snapshot's root-relative frame.
  655   }
  656 
  657 -- | Join a prefix and a (possibly @"\. "@) relative 'RemotePath' into a single
  658 -- 'RemotePath'. A @"\. "@ suffix collapses to the prefix (so re-anchoring at
  659 -- @"\. "@ is the identity). Crashes on invalid input — the prefix/suffix
  660 -- are internally generated from validated components, never user/LLM input.
  661 joinRp :: Text -> RemotePath -> IO RemotePath
  662 joinRp prefix rp' =
  663   let suffix = getRemotePath rp'
  664       joined
  665         | T.null prefix            = suffix
  666         | suffix == "."            = prefix
  667         | T.isPrefixOf "./" suffix = prefix <> "/" <> T.drop 2 suffix
  668         | otherwise                = prefix <> "/" <> suffix
  669   in case mkRemotePath joined of
  670        Right r  -> pure r
  671        Left err -> error ("joinRp: invalid remote path: " <> T.unpack err
  672                           <> ": " <> T.unpack joined)
  673 
  674 -- | Construct a 'RemotePath', crashing on invalid input. Used only for
  675 -- internally-generated, validated path components (never user/LLM input).
  676 rpOrDie :: Text -> IO RemotePath
  677 rpOrDie t = case mkRemotePath t of
  678   Right r  -> pure r
  679   Left err -> error ("rpOrDie: invalid remote path: " <> T.unpack err
  680                      <> ": " <> T.unpack t)

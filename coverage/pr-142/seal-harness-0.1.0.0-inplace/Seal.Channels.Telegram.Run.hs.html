   79 -- | Spawn the real Telegram transport, resolve the token + chunk limit +
   80 -- allow-list, and run the agent loop against the Telegram channel. Fails
   81 -- fast with a stderr diagnostic if the token is unresolved or the Bot API
   82 -- is unreachable.
   83 runTelegram
   84   :: ChannelDeps -> Registry -> PreprocessChain
   85   -> (TelegramToken, Int, AllowList UserId)
   86   -> AskReplyStore
   87   -> IO ()
   88 runTelegram deps registry chain (token, chunkLimit, allow) askReply = do
   89   let mgr = prManager (cdProvider deps)
   90   transport <- mkRealTelegramTransport (telegramTokenText token) mgr
   91   -- Register the bot's slash-command menu with BotFather for auto-completion.
   92   tgSetCommands transport (telegramBotCommands registry)
   93   let withCh = withTelegramChannel (allow, chunkLimit) transport (cdLogger deps)
   94       plainHandler h = plainTurnWithCaps deps h askReply (Just (mkTelegramHandleCaps transport))
   95   tabsH <- newTabsHandle
   96   runChannelLoop deps withCh plainHandler registry chain askReply tabsH
   97     (Just (mkTelegramHandleCaps transport)) (Just (onTelegramCallback askReply))

   99 -- | Full @seal telegram@ startup wiring: paths -> config -> vault -> session
  100 -- -> backends -> registry -> spawn the Telegram channel -> run the loop.
  101 -- Mirrors 'Seal.Channels.Signal.Run.runSignalMain' but drives the Telegram
  102 -- channel instead. Resolves the @[telegram]@ config section + an optional
  103 -- vault-supplied token; fails fast with a stderr diagnostic if the token is
  104 -- unresolved.
  105 runTelegramMain :: AutonomyLevel -> SealLogger -> IO ()
  106 runTelegramMain autonomy logger = do
  107   paths <- getSealPaths
  108   ensureSealDirs paths
  109   migrateSecurityConfig paths
  110   let cfgPath = configFilePath paths
  111   cfg <- loadRuntimeConfig cfgPath >>= \case
  112     Left err -> do
  113       logIO logger WarningS ("could not load config: " <> ls err)
  114       pure defaultRuntimeConfig
  115     Right c  -> pure c
  116   secCfg <- loadSecurityConfig (securityFilePath paths) >>= \case
  117     Left err -> do
  118       logIO logger WarningS ("could not load security config: " <> ls err)
  119       pure defaultSecurityConfig
  120     Right c  -> pure c
  121   mHandle <- tryOpenVault paths secCfg logger
  122   ref     <- newIORef mHandle
  123   let rt = VaultRuntime
  124             { vrPaths      = paths
  125             , vrConfigPath = cfgPath
  126             , vrHandleRef  = ref
  127             }
  128   mgr <- newTlsManager
  129   callCounter <- newIORef 0
  130   let pr = ProviderRuntime
  131             { prConfigPath  = cfgPath
  132             , prVault       = rt
  133             , prManager     = mgr
  134             , prCallCounter = callCounter
  135             }
  136   let cfgRoot = spConfig paths
  137   ensureConfigRepo cfgRoot
  138   let repo = openConfigRepo cfgRoot
  139   backends <- newBackends cfgRoot repo
  140   sessionMeta <- initSession paths cfg (bAgentDefs backends)
  141   activeRef   <- newIORef sessionMeta
  142   let sr = SessionRuntime
  143              { srPaths      = paths
  144              , srConfigPath = cfgPath
  145              , srActive     = activeRef
  146              }
  147   tabsH <- newTabsHandle
  148   cli <- mkRealSignalCli
  149   tgApi <- mkRealTelegramBotApi
  150   vaultStore <- mkRealVaultStore mHandle
  151   let channelRt = ChannelRuntime { crConfigPath = cfgPath, crSignalCli = cli
  152                                  , crTelegramBotApi = tgApi
  153                                  , crVaultStore = vaultStore }
  154   askReply <- newAskReplyStore 0
  155   approvals <- newApprovalCache
  156   harnessReg <- Seal.Harness.Registry.newHarnessRegistry
  157   tmuxR <- Seal.Harness.Tmux.mkRealTmuxRunner
  158   let loadCfg = do
  159         lc <- loadRuntimeConfig cfgPath
  160         pure (fromRight defaultRuntimeConfig lc)
  161   repoRegH <- mkRepoRegistryHandle (reposFilePath paths)
  162   cursorsH <- newPersistingCursorStore (cursorMapPath paths)
  163   mCursors <- loadCursorMap (cursorMapPath paths)
  164   for_ mCursors (seedCursorStore cursorsH)
  165   chanDeps <- newChannelDeps
  166         paths rt repoRegH pr backends autonomy Nothing
  167         harnessReg tmuxR (Just mgr) approvals loadCfg (isJust (untrustedExecConfigFromSecurity secCfg)) tabsH logger cursorsH
  168   let coreDeps = CoreCommandDeps
  169         { ccdVault       = rt
  170         , ccdProvider    = pr
  171         , ccdSession     = sr
  172         , ccdAgentDefs   = bAgentDefs backends
  173         , ccdCfgPath     = cfgPath
  174         , ccdPaths       = paths
  175         , ccdTabs        = tabsH
  176         , ccdTabCloseNotifier = mkTabCloseNotifier (cdCursors chanDeps) (cdReplies chanDeps)
  177         , ccdAbortReg    = cdAbortReg chanDeps
  178         , ccdStopWriter  = mkStopTranscriptWriter paths (cdBroker chanDeps)
  179         , ccdModelWriter = mkModelTranscriptWriter paths (cdBroker chanDeps)
  180         , ccdRepoReg     = repoRegH
  181         , ccdRepoSeam    = Nothing
  182         }
  183       registry = mkRegistry
  184         ( coreCommandSpecs coreDeps
  185           <> [ channelCommandSpec channelRt
  186              ]
  187         )
  188   -- Read the bot token from the vault (the wizard stores it there, not in
  189   -- config.toml). Falls back to the config token if present (for backward
  190   -- compat), but the vault token takes precedence.
  191   mVaultToken <- case mHandle of
  192     Nothing -> pure Nothing
  193     Just vh -> do
  194       r <- Vault.vhGet vh telegramVaultKey
  195       pure $ case r of
  196         Right bs -> Just (TE.decodeUtf8 bs)
  197         Left _   -> Nothing
  198   case resolveTelegramConfig (rcTelegram cfg) mVaultToken of
  199     Left err -> logIO logger ErrorS ("seal telegram: " <> ls err)
  200     Right resolved -> runTelegram chanDeps registry emptyChain resolved askReply

  202 -- | Open the vault if both recipient and identity are configured. Mirrors
  203 -- 'Seal.Tui.tryOpenVault'; duplicated here to keep this module standalone.
  204 tryOpenVault :: SealPaths -> SecurityConfig -> SealLogger -> IO (Maybe Vault.VaultHandle)
  205 tryOpenVault paths cfg logger =
  206   case (scVaultRecipient cfg, scVaultIdentity cfg) of
  207     (Just _, Just _) ->
  208       resolveEncryptor cfg >>= \case
  209         Left err -> do
  210           logIO logger WarningS ("vault not available: " <> ls (T.pack (show err)))
  211           pure Nothing
  212         Right enc -> do
  213           let vcfg = Vault.VaultConfig
  214                 { Vault.vcPath    = maybe (vaultFilePath paths) T.unpack (scVaultPath cfg)
  215                 , Vault.vcKeyType = fromMaybe "x25519" (scVaultKeyType cfg)
  216                 , Vault.vcUnlock  = parseUnlockMode (scVaultUnlock cfg)
  217                 }
  218           Just <$> Vault.openVault vcfg enc
  219     _ -> pure Nothing

  221 -- | Build a 'ChannelCaps' for the Telegram channel. When the 'AskPrompt'
  222 -- has non-empty options AND the handle has a last chat id, 'ccPrompt' sends
  223 -- the question + a numbered list of options (with descriptions) as the
  224 -- message body, with an inline keyboard (one button per option + an
  225 -- "Other" free-text button) attached. The message body carries the full
  226 -- option text + descriptions (Telegram button labels are too short for
  227 -- descriptions); the buttons carry just the short labels for tap
  228 -- convenience. 'answerCallbackQuery' shows a toast on tap; the loop also
  229 -- sends a persistent "✓ <label>" confirmation after delivery. When the
  230 -- options are empty OR no chat id is known, it falls back to the generic
  231 -- numbered-list rendering via 'chSend' + 'formatQuestionWithOptions'.
  232 mkTelegramHandleCaps :: TelegramTransport -> ChannelHandle -> AskReplyStore -> SessionId -> ChannelCaps
  233 mkTelegramHandleCaps transport h askReply sid = def
  234   { ccSend         = chSend h
  235   , ccPrompt       = \ap -> do
  236       let AskPrompt q opts = ap
  237       mChat <- chLastChatId h
  238       case (mChat, opts) of
  239         (Just chatId, _ : _) -> do
  240           outcome <- askHumanWithOptions askReply sid q opts (sendKeyboard chatId q opts)
  241           pure (fromRight "" outcome)
  242         _ -> do
  243           -- Empty options OR no chat id: fall back to the numbered list.
  244           chSend h (formatQuestionWithOptions q opts)
  245           outcome <- askHumanWithOptions askReply sid q opts (const (pure ()))
  246           pure (fromRight "" outcome)
  247   , ccPromptSecret = fmap (fromRight "") . chPromptSecret h
  248   , ccStreaming    = chStreaming h
  249   }
  250   where
  251     -- | The notify callback: sends the question + option descriptions as
  252     -- the message body, with the inline keyboard (short labels only)
  253     -- attached. The 8-hex prefix of the 'AskId' ties the buttons to this
  254     -- specific pending ask.
  255     sendKeyboard :: Text -> Text -> [QuestionOption] -> AskId -> IO ()
  256     sendKeyboard chatId q opts qid =
  257       let prefix = T.take 8 (askIdText qid)
  258           body = formatQuestionWithOptions q opts
  259       in tgSendWithKeyboard transport chatId body (buildKeyboard prefix opts)
  260 
  261 -- | Build the inline keyboard: one row per option (button label = the
  262 -- option's 'qoLabel', callback_data = @\"<prefix>:<idx>\"@). No "Other"
  263 -- free-text button (Telegram has no inline free-text input; the user can
  264 -- still type a free-text answer as a regular message, which the loop
  265 -- routes via 'deliverNextAnswerResolved' when no pending callback match
  266 -- is found). Pure. The callback_data is always ≤ 64 bytes (8 hex + 1
  267 -- colon + ≤ 1 char for the index = ≤ 10 bytes).
  268 buildKeyboard :: Text -> [QuestionOption] -> [[TelegramButton]]
  269 buildKeyboard prefix opts =
  270   [ [TelegramButton (qoLabel o) (prefix <> ":" <> T.pack (show i))]
  271   | (i, o) <- zip [0 :: Int ..] opts
  272   ]

  274 -- | The callback handler for Telegram: parses the @callback_data@
  275 -- (@\"<8hex>:<token>\"@), where @token@ is either a decimal index (the
  276 -- tapped option button) or @other@ (the Other free-text button). For an
  277 -- index, finds the pending ask by the 8-hex prefix, resolves the index to
  278 -- the option's label via the pending ask's 'pqiOptions', and delivers it
  279 -- by-id via 'deliverAnswer'. Sends a persistent "✓ <label>" confirmation
  280 -- to the chat via 'chSend' so the user sees their tap was registered (the
  281 -- LLM may take time to respond). A stale @:other@ token (from a legacy
  282 -- keyboard that had an "Other" button) falls through to 'False'. Returns
  283 -- 'True' if a callback was delivered; 'False' if not (fall through).
  284 --
  285 -- The 'AskReplyStore' is the first arg so the caller can partially apply
  286 -- it; the 'runChannelLoop' 'onCallback' hook supplies the 'ChannelHandle'.
  287 onTelegramCallback :: AskReplyStore -> ChannelHandle -> SessionId -> Text -> IO Bool
  288 onTelegramCallback store h sid body =
  289   case T.splitOn ":" body of
  290     [prefix, token]
  291       | T.length prefix == 8 && T.all isHexChar prefix ->
  292           case parseIndex token of
  293             Just idx -> resolveIndex h store sid prefix idx
  294             Nothing  -> pure False  -- non-numeric (e.g. legacy "other"): fall through
  295     _ -> pure False
  296   where
  297     isHexChar c = isDigit c || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F')
  298 
  299 -- | Parse a non-negative decimal index from the callback token. 'Nothing'
  300 -- if the token is not a clean decimal (e.g. empty, negative, or has
  301 -- trailing chars).
  302 parseIndex :: Text -> Maybe Int
  303 parseIndex t =
  304   case TR.decimal (T.strip t) of
  305     Right (n, rest) | T.null rest && n >= 0 -> Just n
  306     _ -> Nothing
  307 
  308 -- | Resolve a callback index to the option's label + deliver it by-id,
  309 -- then send a "✓ <label>" confirmation to the chat.
  310 resolveIndex :: ChannelHandle -> AskReplyStore -> SessionId -> Text -> Int -> IO Bool
  311 resolveIndex h store sid prefix idx = do
  312   mQid <- findByAskIdPrefix store sid prefix
  313   case mQid of
  314     Nothing -> pure False
  315     Just qid -> do
  316       ps <- pendingForSession store sid
  317       case filter (\p -> pqiId p == qid) ps of
  318         (p : _) ->
  319           case V.fromList (pqiOptions p) V.!? idx of
  320             Just o  -> do
  321               _accepted <- deliverAnswer store qid (AskReply ScopeOnce (qoLabel o))
  322               chSend h ("✓ " <> qoLabel o)
  323               pure True
  324             Nothing -> pure False  -- out of bounds
  325         [] -> pure False  -- stale (ask vanished between the two lookups)

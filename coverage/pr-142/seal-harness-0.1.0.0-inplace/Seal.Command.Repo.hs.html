    2 -- | The @/repo@ command group: list, add, remove, inspect, or test the
    3 -- source-control repos registered in @config\/repos.toml@ (design §4.6).
    4 -- Mirrors 'Seal.Command.Skill' (hsubparser with @list@/@info@ subcommands,
    5 -- 'InteractiveOnly' availability, plain-language renderers). Mutations
    6 -- (@add@\/@remove@) go through 'RepoRegistryHandle' (the service layer over
    7 -- @repos.toml@); the credential /value/ is NEVER read by this command (only
    8 -- the vault key /name/) — secret bytes live in the vault and are fetched at
    9 -- clone time by W3's 'Seal.SourceControl.Clone.resolveCloneTarget'.
   10 --
   11 -- /repo test runs @git ls-remote@ through the injected 'RepoTestSeam' (the
   12 -- real seam is 'lsRemoteRepo'; tests inject a stub) so the command is fully
   13 -- testable without running real git. The /repo test error-message mapping
   14 -- (AC6) reconciles the W3 'renderCloneError' wording for the slash surface:
   15 -- @git failed (exit N)@ becomes @git ls-remote failed (exit N)@ here, since
   16 -- the only git invocation /repo test performs is @git ls-remote@.

   46 -- Test seam
   47 ----------------------------------------------------------------------------

   49 -- | The /repo command depends on the live vault + @repoCloneStateDir@ for
   50 -- @/repo test@, and on the vault's key list for the @/repo info@ non-blocking
   51 -- advisory. To keep the command testable without running real git, the
   52 -- ls-remote seam is injected. Production wiring (in 'Seal.Command.Serve')
   53 -- passes a real seam built from 'Seal.SourceControl.Clone.lsRemoteRepo' and
   54 -- 'Seal.Security.Vault.vhList'; tests pass a stub.
   55 data RepoTestSeam = RepoTestSeam
   56   { rtsLsRemote  :: SourceRepo -> IO (Either CloneError Text)
   57     -- ^ The @git ls-remote@ seam (production: 'lsRemoteRepo').
   58   , rtsVaultList :: IO (Either VaultError [Text])
   59     -- ^ The vault key list (production: 'vhList') for the /repo info advisory.
   60   , rtsVaultPut :: Text -> ByteString -> IO (Either VaultError ())
   61     -- ^ The vault put seam (production: 'vhPut') for /repo add's PAT/MachineUser
   62     -- token ingestion (hidden stdin prompt → vault, never argv/disk).
   63   }

   69 -- | The @/repo@ command spec. Closes over the 'RepoRegistryHandle' (for
   70 -- list/add/remove/info) and a 'RepoTestSeam' (for /repo test + /repo info
   71 -- advisory). 'InteractiveOnly' (mirrors /skill) so it dispatches only on the
   72 -- channel the operator is typing on — not reachable by the agent's tool
   73 -- surface.
   74 repoCommandSpec :: RepoRegistryHandle -> RepoTestSeam -> CommandSpec
   75 repoCommandSpec regH seam = CommandSpec
   76   { csName         = CommandName "repo"
   77   , csAliases      = []
   78   , csGroup        = GroupRepos
   79   , csSynopsis     = "List, add, remove, inspect, or test source-control repos"
   80   , csParserInfo   = repoParserInfo regH seam
   81   , csAvailability = InteractiveOnly
   82   }
   83 
   84 repoParserInfo :: RepoRegistryHandle -> RepoTestSeam -> ParserInfo CommandAction
   85 repoParserInfo regH seam =
   86   info (repoParser regH seam <**> helper)
   87     (  progDesc "List, add, remove, inspect, or test source-control repos"
   88     <> header   "repo — manage the source-control repo registry"
   89     )
   90 
   91 repoParser :: RepoRegistryHandle -> RepoTestSeam -> Parser CommandAction
   92 repoParser regH seam = hsubparser
   93   (  command "list"
   94        (info (pure (listCmd regH))
   95              (progDesc "List all registered repos (id, url, credential kind)"))
   96   <> command "add"
   97        (info (addCmd regH seam <$> idArg <*> urlArg <*> vcsOpt <*> credOpt
   98                       <*> vaultKeyOpt <*> usernameOpt)
   99              (progDesc "Add or replace a source-control repo"))
  100   <> command "remove"
  101        (info (removeCmd regH <$> idArg)
  102              (progDesc "Remove a source-control repo by id"))
  103   <> command "info"
  104        (info (infoCmd regH seam <$> idArg)
  105              (progDesc "Show a repo's descriptor + a vault-key advisory"))
  106   <> command "test"
  107        (info (testCmd regH seam <$> idArg)
  108              (progDesc "Run git ls-remote to verify the credential resolves"))
  109   <> metavar "COMMAND"
  110   )

  175 listCmd regH = CommandAction $ \caps -> do

  183 -- add
  184 ----------------------------------------------------------------------------
  185 
  186 -- | @/repo add <id> <url> --vcs --cred --vault-key [--username]@. Validates
  187 -- the id, URL shape, host allow-list, VCS kind, and credential kind before
  188 -- mutating. The credential /value/ is NEVER read (only the vault key name).
  189 -- On a successful mutation, echoes @added repo \<id\>@.
  190 addCmd
  191   :: RepoRegistryHandle
  192   -> RepoTestSeam  -- ^ vault put seam (for PAT/MachineUser token ingestion)
  193   -> Text   -- ^ raw id
  194   -> Text   -- ^ url
  195   -> Text   -- ^ vcs kind (text)
  196   -> Text   -- ^ credential kind (text)
  197   -> Text   -- ^ vault key name
  198   -> Maybe Text  -- ^ optional username (machine_user only)
  199   -> CommandAction
  200 addCmd regH seam rawId url rawVcs rawCred vaultKey mUsername =
  201   CommandAction $ \caps -> do
  202     case validateAdd rawId url rawVcs rawCred vaultKey mUsername of
  203       Left err -> ccSend caps err
  204       Right repo -> case srCredential repo of
  205         CredDeployKey _      -> doUpsert caps repo
  206         CredPat _            -> doTokenPrompt caps repo
  207         CredMachineUser _ _ -> doTokenPrompt caps repo
  208   where
  209     doUpsert caps repo = do
  210       res <- rrhMutate regH (upsertRepo repo)
  211       case res of
  212         Left e  -> ccSend caps e
  213         Right _ -> ccSend caps ("added repo " <> repoIdText (srId repo))
  214     doTokenPrompt caps repo = do
  215       token <- ccPromptSecret caps ("PAT for " <> repoIdText (srId repo) <> ": ")
  216       ePut <- rtsVaultPut seam (cVaultKey (srCredential repo)) (TE.encodeUtf8 token)
  217       case ePut of
  218         Left VaultLocked -> ccSend caps "vault locked — run /vault unlock"
  219         Left _           -> ccSend caps "vault error — could not store token"
  220         Right _          -> doUpsert caps repo
  221 
  222 -- | Pure validation for @/repo add@. Runs every write-time check from the
  223 -- security table (§Security): id charset, URL shape, host allow-list, VCS
  224 -- kind, credential kind (with the machine_user username requirement).
  225 validateAdd
  226   :: Text -> Text -> Text -> Text -> Text -> Maybe Text
  227   -> Either Text SourceRepo
  228 validateAdd rawId url rawVcs rawCred vaultKey mUsername = do
  229   rid       <- mkRepoId rawId
  230   vcs       <- parseVcsKind rawVcs
  231   -- URL shape (defense in depth on top of host allow-list).
  232   if not (urlShapeValid url)
  233     then Left "URL is neither SSH (git@<host>:...) nor HTTPS (https://<host>/...)"
  234     else do
  235       host <- parseRepoHost url
  236       if not (hostAllowed host)
  237         then Left ("host " <> host <> " not supported (only github.com is supported in this pass)")
  238         else do
  239           cred <- parseCredentialKind rawCred vaultKey mUsername
  240           Right SourceRepo
  241             { srId               = rid
  242             , srUrl              = url
  243             , srVcsKind          = vcs
  244             , srCredential       = cred
  245             , srDeployKeyPublic  = Nothing
  246             , srKeyfilePath      = Nothing
  247             }

  253 -- | @/repo remove <id>@. Idempotent (204 semantics mirror the REST API): a
  254 -- missing id is still reported as @removed@.
  255 removeCmd :: RepoRegistryHandle -> Text -> CommandAction
  256 removeCmd regH raw = CommandAction $ \caps ->
  257   case mkRepoId raw of
  258     Left err -> ccSend caps err
  259     Right rid -> do
  260       res <- rrhMutate regH (removeRepo rid)
  261       case res of
  262         Left e  -> ccSend caps e
  263         Right _ -> ccSend caps ("removed repo " <> repoIdText rid)

  266 -- info
  267 ----------------------------------------------------------------------------
  268 
  269 -- | @/repo info <id>@. Renders the descriptor and appends a NON-BLOCKING
  270 -- vault-key advisory: if the vault lists its keys and the repo's
  271 -- @cVaultKey@ is not among them, the operator gets immediate feedback
  272 -- (@vault key \<name\> not found — clone will fail until it is added@) —
  273 -- without failing the command. Lazy-verify is preserved (the vault key is
  274 -- not actually fetched here; only its name is cross-checked against the
  275 -- vault's key list). A locked vault surfaces @vault locked — run /vault
  276 -- unlock to check@; any other vault error skips the advisory silently.
  277 infoCmd :: RepoRegistryHandle -> RepoTestSeam -> Text -> CommandAction
  278 infoCmd regH seam raw = CommandAction $ \caps ->
  279   case mkRepoId raw of
  280     Left err -> ccSend caps err
  281     Right rid -> do
  282       eRepos <- rrhList regH
  283       case eRepos of
  284         Left e -> ccSend caps e
  285         Right repos -> case findRepo rid repos of
  286           Nothing  -> ccSend caps ("repo not found: " <> repoIdText rid)
  287           Just r  -> do
  288             mapM_ (ccSend caps) (renderRepoInfo r)
  289             advisory <- vaultKeyAdvisory seam (srCredential r)
  290             for_ advisory (ccSend caps)
  291 
  292 -- | Find a repo by id in a list (helper to avoid a Map.fromList round-trip).
  293 findRepo :: RepoId -> [SourceRepo] -> Maybe SourceRepo
  294 findRepo rid = go
  295   where
  296     go []       = Nothing
  297     go (r : rs) = if srId r == rid then Just r else go rs
  298 
  299 -- | Compute the non-blocking vault-key advisory for /repo info. Returns
  300 -- 'Nothing' when no advisory applies (the vault key is present, or a
  301 -- non-lock vault error occurred — silent skip).
  302 vaultKeyAdvisory :: RepoTestSeam -> RepoCredential -> IO (Maybe Text)
  303 vaultKeyAdvisory seam cred = do
  304   eKeys <- rtsVaultList seam
  305   let vaultKey = cVaultKey cred
  306   pure $ case eKeys of
  307     Left VaultLocked -> Just "vault locked — run /vault unlock to check"
  308     Left _           -> Nothing  -- other vault error: skip the advisory
  309     Right keys ->
  310       if vaultKey `elem` keys
  311         then Nothing  -- key present: no advisory
  312         else Just ("vault key " <> vaultKey <> " not found — clone will fail until it is added")

  315 -- test

  318 -- | @/repo test <id>@. Resolves the repo, runs @git ls-remote@ via the
  319 -- injected seam, and echoes the head sha on success
  320 -- (@credential verified — \<sha\>@). On failure, renders the 'CloneError'
  321 -- via the /repo test-specific mapping (AC6): notably @CloneGitFailed n@
  322 -- becomes @git ls-remote failed (exit n)@ (NOT W3's @git failed (exit n)@),
  323 -- since the only git invocation /repo test performs is @git ls-remote@.
  324 testCmd :: RepoRegistryHandle -> RepoTestSeam -> Text -> CommandAction
  325 testCmd regH seam raw = CommandAction $ \caps ->
  326   case mkRepoId raw of
  327     Left err -> ccSend caps err
  328     Right rid -> do
  329       eRepos <- rrhList regH
  330       case eRepos of
  331         Left e -> ccSend caps e
  332         Right repos -> case findRepo rid repos of
  333           Nothing  -> ccSend caps ("repo not found: " <> repoIdText rid)
  334           Just r   -> do
  335             res <- rtsLsRemote seam r
  336             case res of
  337               Left e  -> ccSend caps (renderRepoTestError e)
  338               Right s -> ccSend caps ("credential verified — " <> s)
  339 
  340 -- | /repo test error-message mapping (AC6). Mirrors W3's 'renderCloneError'
  341 -- EXCEPT for @CloneGitFailed@, which is reconciled to @git ls-remote failed@
  342 -- per the W3 reviewer's non-blocking note (the slash surface invokes
  343 -- @git ls-remote@, not @git@; the W3 wording was generic and is sharpened
  344 -- here for the operator-facing message).
  345 renderRepoTestError :: CloneError -> Text
  346 renderRepoTestError = \case
  347   CloneVaultError VaultLocked ->
  348     "vault locked — run /vault unlock"
  349   CloneVaultError (VaultKeyNotFound k) ->
  350     "vault key " <> k <> " not found"
  351   CloneVaultError _ ->
  352     "vault error"
  353   CloneNoCredentialForUrl u ->
  354     "no credential resolvable for " <> u
  355   CloneUnsupportedVcs v ->
  356     "unsupported VCS: " <> vcsKindText v
  357   CloneHostNotSupported h ->
  358     "host " <> h <> " not supported (only github.com is supported in this pass)"
  359   CloneGitFailed n ->
  360     "git ls-remote failed (exit " <> T.pack (show n) <> ")"
  361   CloneAgentError msg ->
  362     "ssh-agent error: " <> msg

  368 -- | One line per repo for @/repo list@: @id  url  <human-credential-label>@.
  369 renderRepoLine :: SourceRepo -> Text
  370 renderRepoLine r =
  371   repoIdText (srId r) <> "  " <> srUrl r <> "  " <> credentialKindLabel (srCredential r)
  372 
  373 -- | Multi-line detail for @/repo info@. The @username@ line is emitted ONLY
  374 -- for 'CredMachineUser' (a public bot handle); PAT/DeployKey have no
  375 -- username.
  376 renderRepoInfo :: SourceRepo -> [Text]
  377 renderRepoInfo r =
  378   baseLines <> usernameLine
  379   where
  380     baseLines =
  381       [ "id:          " <> repoIdText (srId r)
  382       , "url:         " <> srUrl r
  383       , "vcs:         " <> vcsKindText (srVcsKind r)
  384       , "credential:  " <> credentialKindLabel (srCredential r)
  385       , "vault key:   " <> cVaultKey (srCredential r)
  386       ]
  387     usernameLine = case srCredential r of
  388       CredMachineUser _ u -> ["username:    " <> u]
  389       _                   -> []
  390 
  391 -- | Human-readable label for a 'RepoCredential' (the @credential_kind@ wire
  392 -- string is @pat@\/@deploy_key@\/@machine_user@; this is the long form for
  393 -- @/repo list@ + @/repo info@).
  394 credentialKindLabel :: RepoCredential -> Text
  395 credentialKindLabel = \case
  396   CredPat _         -> "Personal Access Token"
  397   CredDeployKey _   -> "SSH Deploy Key"
  398   CredMachineUser {} -> "Bot Account"

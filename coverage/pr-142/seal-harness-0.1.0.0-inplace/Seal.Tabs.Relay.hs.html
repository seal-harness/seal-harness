    1 {-# LANGUAGE DeriveGeneric #-}
    2 {-# LANGUAGE OverloadedStrings #-}
    3 -- | The streaming-aware per-conversation output relay. Focused
    4 -- conversations receive every 'StreamStart'/'ChunkOf'/'StreamEnd' verbatim;
    5 -- background 'ActivityDigest' conversations get at most one breadcrumb ping
    6 -- per burst; 'Firehose' forwards everything.
    7 module Seal.Tabs.Relay
    8   ( RelayEvent (..)
    9   , relayEvent
   10   ) where
   11 
   12 import Data.Text (Text)
   13 import Data.Text qualified as T
   14 import GHC.Generics (Generic)
   15 
   16 import Seal.Tabs.Types (RelayMode (..))
   17 
   18 -- | One streaming event to relay.
   19 data RelayEvent
   20   = StreamStart Text     -- ^ a stream began (carries a header/breadcrumb)
   21   | ChunkOf Text          -- ^ one chunk
   22   | StreamEnd             -- ^ the stream ended
   23   deriving stock (Eq, Show, Generic)
   24 
   25 -- | Relay one event to one conversation. Returns the lines to send to that
   26 -- conversation (0, 1, or many). Pure.
   27 --
   28 -- * 'FocusedOnly': 'StreamStart' → @[]@; 'ChunkOf' t → @[t]@;
   29 --   'StreamEnd' → @[]@. The focused conversation sees the chunks verbatim,
   30 --   no framing.
   31 -- * 'ActivityDigest': 'StreamStart' → @[]@; 'ChunkOf' t → @[]@ (suppress);
   32 --   'StreamEnd' → @[breadcrumb]@ (one ping per burst).
   33 -- * 'Firehose': every event → @[show event]@ (forwards everything, including
   34 --   framing — the firehose consumer wants the structure).
   35 relayEvent :: RelayMode -> RelayEvent -> [Text]
   36 relayEvent FocusedOnly (ChunkOf t) = [t]
   37 relayEvent FocusedOnly _           = []
   38 relayEvent ActivityDigest StreamEnd = ["[tab] activity"]
   39 relayEvent ActivityDigest _         = []
   40 relayEvent Firehose e                = [T.pack (show e)]

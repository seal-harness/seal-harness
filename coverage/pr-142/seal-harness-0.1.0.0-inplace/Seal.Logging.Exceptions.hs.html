    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Unified exception handling helper. Replaces the 6 duplicated @catch@
    3 -- patterns across 'Seal.Channels.Loop', 'Seal.Gateway.Send', and
    4 -- 'Seal.Channel.Cli' with a single function that:
    5 --
    6 -- 1. Catches /synchronous/ exceptions only. 'AsyncException' (including
    7 --    'ThreadKilled') is rethrown after logging at 'InfoS' — shutdown must
    8 --    propagate to the bracket, not be swallowed.
    9 -- 2. Logs the full exception to the operator log via 'logIO' at 'ErrorS',
   10 --    including the 'where' label and the logger's 'ChannelContext'.
   11 -- 3. Returns a /sanitized/ 'Text' for the user (via 'Left'): a generic
   12 --    message (@\"internal error (\<where\>) [ref: \<correlationId\>]"@)
   13 --    with a short correlation id so the operator can correlate the user's
   14 --    report to the log entry. 'TranscriptError' text passes through
   15 --    unchanged (it's already user-facing).
   16 -- 4. Calls the optional session-log path ('logTurnError') alongside the
   17 --    katip emission, so the per-session @seal.log@ audit trail is kept.
   18 -- 5. Escapes newlines in the exception text before logging (log-injection
   19 --    defense).
   20 module Seal.Logging.Exceptions
   21   ( withExceptionLogging
   22   ) where
   23 
   24 import Control.Exception
   25   ( catch, throwIO, fromException, SomeException, AsyncException (..) )
   26 import Data.Text (Text)
   27 import Data.Text qualified as T
   28 import System.Random (randomIO)
   29 
   30 import Seal.Logging.Logger (SealLogger, logIO, escapeNewlines)
   31 import Seal.Session.Log (logTurnError)
   32 import Seal.Handles.Transcript (TranscriptError (..))
   33 
   34 import Katip (Severity (..), ls)
   35 
   36 -- | Run an 'IO' action, catching synchronous exceptions. Logs the exception
   37 -- with the logger's context + the given 'where' label, calls the optional
   38 -- session-log fallback, and returns either a sanitized error text (for the
   39 -- caller to send to the user) or the original result.
   40 --
   41 -- 'AsyncException' (including 'ThreadKilled') is NOT caught — it is
   42 -- rethrown after logging at 'InfoS', so shutdown propagates to the
   43 -- bracket.
   44 withExceptionLogging
   45   :: SealLogger
   46   -> Maybe FilePath          -- ^ session log path (for seal.log fallback)
   47   -> Text                    -- ^ 'where' label (e.g. "slash command", "turn")
   48   -> IO a
   49   -> IO (Either Text a)
   50 withExceptionLogging logger mLogPath whereLabel action =
   51   (Right <$> action) `catch` \e -> do
   52     case (fromException e :: Maybe AsyncException) of
   53       Just asyncEx -> do
   54         logIO logger InfoS (escapeNewlines (ls $
   55           "shutdown signal in " <> whereLabel <> ": " <> showLs asyncEx))
   56         -- Also record the shutdown in the per-session log so an operator
   57         -- inspecting seal.log after a hung session can see WHY the turn
   58         -- died (e.g. ThreadKilled) rather than just a dangling [TURN]
   59         -- start with no matching end. Best-effort (logTurnError swallows
   60         -- IO errors) and rethrown after logging so the bracket cleanup
   61         -- (idle broadcast) still fires.
   62         logTurnError mLogPath ("shutdown signal in " <> whereLabel <> ": " <> showLs asyncEx)
   63         throwIO e
   64       Nothing -> do
   65         correlationId <- mkCorrelationId
   66         let (userMsg, fullDetail) = sanitizeException whereLabel e correlationId
   67         logIO logger ErrorS (escapeNewlines (ls fullDetail))
   68         logTurnError mLogPath fullDetail
   69         pure (Left userMsg)
   70 
   71 -- | Generate a short correlation id (8 hex chars from a random nonce).
   72 mkCorrelationId :: IO Text
   73 mkCorrelationId = do
   74   n <- abs <$> randomIO :: IO Int
   75   pure (T.justifyRight 8 '0' (T.pack (show n)))
   76 
   77 -- | Sanitize an exception for user-facing text. Returns:
   78 -- * (userMsg, fullDetail) where userMsg is safe to send to the user
   79 --   and fullDetail is the complete exception text for the operator log.
   80 sanitizeException :: Text -> SomeException -> Text -> (Text, Text)
   81 sanitizeException whereLabel e correlationId =
   82   case (fromException e :: Maybe TranscriptError) of
   83     Just (TranscriptError te) ->
   84       ("transcript error: " <> te, "transcript error: " <> te <> " (" <> whereLabel <> ")")
   85     Nothing ->
   86       ( "internal error (" <> whereLabel <> ") [ref: " <> correlationId <> "]"
   87       , whereLabel <> " failed: " <> T.pack (show e) <> " [ref: " <> correlationId <> "]"
   88       )
   89 
   90 -- | Helper to show something as LogStr-compatible Text.
   91 showLs :: Show a => a -> Text
   92 showLs = T.pack . show

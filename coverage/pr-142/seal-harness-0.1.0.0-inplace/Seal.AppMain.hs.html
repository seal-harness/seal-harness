   29 -- | Program information for 'runWithConfiguration'. Provides @--config-file@,
   30 -- @--print-config@, and @--help@ automatically.
   31 programInfo :: CUtils.ProgramInfo Config
   32 programInfo = CUtils.programInfo "seal — secure AI agent execution around the SealOp ISA"
   33   pConfig defaultConfig

   35 -- | Dispatch on the selected command, running it in 'App'. The 'SealLogger'
   36 -- is built once via 'withSealLogger' (a bracket that closes the scribe on
   37 -- exit) and threaded into 'Env' so 'App'-level 'logMsg' and 'IO'-level
   38 -- 'logIO' reach the same scribe.
   39 dispatch :: Config -> IO ()
   40 dispatch cfg =
   41   withSealLogger (view config_logLevel cfg) $ \logger -> do
   42     setGlobalLogger logger
   43     env <- mkEnv logger cfg
   44     runApp env $ case _config_command cfg of
   45       CommandNoOp   -> pure ()
   46       CommandTui autonomy -> liftIO (Seal.Tui.runTui autonomy logger)
   47       CommandSignal autonomy -> liftIO (Seal.Channels.Signal.Run.runSignalMain autonomy logger)
   48       CommandTelegram autonomy -> liftIO (Seal.Channels.Telegram.Run.runTelegramMain autonomy logger)
   49       CommandServe autonomy -> liftIO (Seal.Command.Serve.runServeMain autonomy logger)
   50       CommandGenOpenApi -> liftIO (BL.hPutStr stdout Seal.Gateway.OpenApi.encodeOpenApi)
   51 
   52 -- | Map the process arguments so that an empty argument list behaves as if
   53 -- @--help@ was passed. Running @seal@ with no arguments should print usage
   54 -- rather than silently doing nothing. Any non-empty argument list is passed
   55 -- through unchanged.
   56 withDefaultArgs :: [String] -> [String]
   57 withDefaultArgs [] = ["--help"]
   58 withDefaultArgs as = as
   59 
   60 -- | Raise the soft FD limit to the hard limit. The default macOS soft limit
   61 -- is 256, which the gateway's @buildListsSnapshot@ hot path blows through
   62 -- (hundreds of small files per request, historically via lazy IO that left
   63 -- handles open for the GC). Even with strict reads, raising the soft limit
   64 -- to the hard ceiling is cheap defense-in-depth against FD exhaustion. The
   65 -- raise is best-effort: a failure (e.g. a sandboxed runner that blocks
   66 -- @setrlimit@) is swallowed so the binary still starts.
   67 raiseFdLimit :: IO ()
   68 raiseFdLimit = do
   69   limits <- getResourceLimit ResourceOpenFiles
   70   case hardLimit limits of
   71     ResourceLimit _hard | softLimit limits /= hardLimit limits ->
   72       setResourceLimit ResourceOpenFiles limits { softLimit = hardLimit limits }
   73     _ -> pure ()
   74 
   75 -- | Entry point: parse defaults + config file + CLI flags, then dispatch.
   76 -- With no arguments, fall back to @--help@.
   77 appMain :: IO ()
   78 appMain = do
   79   raiseFdLimit
   80   args <- getArgs
   81   withArgs (withDefaultArgs args) (CUtils.runWithConfiguration programInfo dispatch)

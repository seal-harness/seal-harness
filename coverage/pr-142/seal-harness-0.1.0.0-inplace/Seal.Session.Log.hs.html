    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Best-effort per-session diagnostic log (@\<sessionDir\>\/seal.log@).
    3 --
    4 -- Records turn lifecycle events (start/end/duration) and failures (exceptions,
    5 -- provider errors) that are NOT already captured in @entries.jsonl@ or
    6 -- @conversation.jsonl@. The transcript files record what the model + opcodes
    7 -- did; this log records what the /harness/ did — where it started, where it
    8 -- stopped, and why it failed when it failed.
    9 --
   10 -- Design constraints:
   11 --
   12 -- 1. /Best-effort, never throws./ A write error (full disk, permissions,
   13 --    missing parent dir) is swallowed — the log must never break the agent
   14 --    loop. The worst case is a missing log line, not a crashed session.
   15 -- 2. /Append-only, one line per event./ Each line carries an ISO-8601
   16 --    timestamp prefix + a level tag + the message. Human-readable,
   17 --    grep-friendly.
   18 -- 3. /No duplication of transcript data./ Message content, tool-call inputs,
   19 --    and opcode results are already in @conversation.jsonl@ / @entries.jsonl@.
   20 --    This log records only lifecycle boundaries + failure diagnostics.
   21 module Seal.Session.Log
   22   ( appendSessionLog
   23   , logTurnStart
   24   , logTurnEnd
   25   , logTurnError
   26   , logProviderError
   27   , logMaxTurns
   28   , logTruncation
   29   , logTruncationGiveUp
   30   , maxLengthContinueRetries
   31   ) where
   32 
   33 import Control.Exception (catch, SomeException)
   34 import Data.Text (Text)
   35 import Data.Text qualified as T
   36 import Data.Time (UTCTime, getCurrentTime, formatTime, defaultTimeLocale)
   37 import System.IO (withFile, IOMode (AppendMode), hPutStr)
   38 
   39 -- | Maximum number of auto-continuation retries when the provider returns
   40 -- 'StopMaxTokens' with no tool calls (text was truncated mid-generation).
   41 -- After this many retries the loop surfaces a user-visible truncation
   42 -- notice and stops. Mirrors Hermes' @length_continue_retries < 3@ cap.
   43 maxLengthContinueRetries :: Int
   44 maxLengthContinueRetries = 3
   45 
   46 -- | Append a single line to the session log. Best-effort: any IO error is
   47 -- swallowed. The line is prefixed with an ISO-8601 timestamp and the given
   48 -- level tag. A 'Nothing' path is a no-op (logging disabled).
   49 appendSessionLog :: Maybe FilePath -> Text -> Text -> IO ()
   50 appendSessionLog Nothing       _     _ = pure ()
   51 appendSessionLog (Just path) level msg = do
   52   now <- getCurrentTime
   53   let line = T.unpack (formatLogLine now level msg)
   54   withFile path AppendMode (`hPutStr` (line <> "\n"))
   55     `catch` \(_ :: SomeException) -> pure ()
   56 
   57 -- | Format a log line: @\<ISO-8601 ts\> [LEVEL] message@
   58 formatLogLine :: UTCTime -> Text -> Text -> Text
   59 formatLogLine ts level msg =
   60   T.pack (formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%S%3QZ" ts)
   61   <> " [" <> level <> "] " <> msg
   62 
   63 -- | Log a turn start. @turnN@ is the 0-based remaining-turns counter from
   64 -- 'runTurn' (the first turn is @aeMaxTurns@).
   65 logTurnStart :: Maybe FilePath -> Int -> IO ()
   66 logTurnStart mPath turnN =
   67   appendSessionLog mPath "TURN" ("start (turns remaining: " <> T.pack (show turnN) <> ")")
   68 
   69 -- | Log a turn end with a duration in milliseconds.
   70 logTurnEnd :: Maybe FilePath -> Int -> Integer -> IO ()
   71 logTurnEnd mPath turnN ms =
   72   appendSessionLog mPath "TURN" ("end (turns remaining: " <> T.pack (show turnN)
   73     <> ", " <> T.pack (show ms) <> "ms)")
   74 
   75 -- | Log a turn failure (caught exception). The message is the exception text.
   76 logTurnError :: Maybe FilePath -> Text -> IO ()
   77 logTurnError mPath err =
   78   appendSessionLog mPath "ERROR" ("turn failed: " <> err)
   79 
   80 -- | Log a provider error (the 'Left' branch of 'providerComplete').
   81 logProviderError :: Maybe FilePath -> Text -> IO ()
   82 logProviderError mPath err =
   83   appendSessionLog mPath "ERROR" ("provider error: " <> err)
   84 
   85 -- | Log the max-turns stop (the loop hit @aeMaxTurns@ without a final text
   86 -- response).
   87 logMaxTurns :: Maybe FilePath -> IO ()
   88 logMaxTurns mPath =
   89   appendSessionLog mPath "WARN" "stopped: too many tool turns (hit aeMaxTurns)"
   90 
   91 -- | Log a @StopMaxTokens@ truncation that triggered an auto-continuation
   92 -- retry. @attempt@ is 1-based (1 = first retry requested). Mirrors Hermes'
   93 -- @↻ Requesting continuation (N/3)@ log line.
   94 logTruncation :: Maybe FilePath -> Int -> IO ()
   95 logTruncation mPath attempt =
   96   appendSessionLog mPath "WARN"
   97     ("response truncated (StopMaxTokens) - requesting continuation ("
   98       <> T.pack (show attempt) <> "/" <> T.pack (show maxLengthContinueRetries) <> ")")
   99 
  100 -- | Log a @StopMaxTokens@ truncation after all continuation retries are
  101 -- exhausted. The loop surfaces a user-visible truncation notice and stops.
  102 logTruncationGiveUp :: Maybe FilePath -> IO ()
  103 logTruncationGiveUp mPath =
  104   appendSessionLog mPath "WARN"
  105     ("stopped: response remained truncated after "
  106       <> T.pack (show maxLengthContinueRetries)
  107       <> " continuation attempts")

    4 -- | The validated 'TabIndex' (0..35) — the single index type reused
    5 -- everywhere (TabList slots, /N routing, /tab close <N>, /tab focus <N>).
    6 -- Smart-constructed so an out-of-range index fails to compile into any path.
    7 -- Plus 'TabKind', the closed enumeration of tab kinds.
    8 module Seal.Handles.Tab
    9   ( TabIndex (..)
   10   , mkTabIndex
   11   , tabIndexToInt
   12   , tabIndexToChar
   13   , tabIndexFromChar
   14   , maxTabIndex
   15   , TabKind (..)
   16   ) where
   17 
   18 import Data.Aeson (FromJSON, ToJSON)
   19 import Data.Char (isAsciiLower, isDigit, toLower)
   20 import Data.Text (Text)
   21 import Data.Text qualified as T
   22 import GHC.Generics (Generic)
   23 
   24 -- | The maximum tab index (0-based): 35 = 'z'. 36 slots total.
   25 maxTabIndex :: Int
   26 maxTabIndex = 35
   27 
   28 -- | A validated tab index: 0..35 (the terse grammar maps 0-9a-z to 0..35).
   29 -- Smart-constructed; the predicate rejects <0 and >35.
   30 newtype TabIndex = TabIndex Int
   31   deriving stock (Eq, Ord, Show, Generic)
   32   deriving anyclass (ToJSON, FromJSON)
   33 
   34 mkTabIndex :: Int -> Either Text TabIndex
   35 mkTabIndex n
   36   | n < 0           = Left ("tab index out of range: " <> T.pack (show n))
   37   | n > maxTabIndex = Left ("tab index out of range: " <> T.pack (show n))
   38   | otherwise       = Right (TabIndex n)
   39 
   40 tabIndexToInt :: TabIndex -> Int
   41 tabIndexToInt (TabIndex n) = n
   42 
   43 -- | 0->'0', 9->'9', 10->'a', 35->'z'.
   44 tabIndexToChar :: TabIndex -> Char
   45 tabIndexToChar (TabIndex n)
   46   | n < 10   = toEnum (fromEnum '0' + n)
   47   | otherwise = toEnum (fromEnum 'a' + (n - 10))
   48 
   49 -- | Inverse of 'tabIndexToChar', case-insensitive. 'Left' for non-[0-9a-z].
   50 tabIndexFromChar :: Char -> Either Text TabIndex
   51 tabIndexFromChar c =
   52   let lo = toLower c
   53   in case lo of
   54        _ | isDigit lo      -> mkTabIndex (fromEnum lo - fromEnum '0')
   55          | isAsciiLower lo -> mkTabIndex (fromEnum lo - fromEnum 'a' + 10)
   56          | otherwise       -> Left ("not a tab index char: " <> T.pack [c])
   57 
   58 -- | The closed enumeration of tab kinds.
   59 data TabKind
   60   = KindAi | KindProvider | KindHarness | KindShell | KindSsh | KindTmux
   61   deriving stock (Eq, Show, Enum, Bounded, Generic)
   62   deriving anyclass (ToJSON, FromJSON)

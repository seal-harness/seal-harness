   31 -- ---------------------------------------------------------------------------
   32 -- SignalAccount — the validated account label for the signal-cli argv
   33 -- ---------------------------------------------------------------------------
   34 
   35 -- | The signal-cli account label (a phone number or UUID). Smart-constructed:
   36 -- non-empty, no leading dash (option-injection defense — a leading-dash
   37 -- value would be interpreted as a flag by signal-cli), charset
   38 -- @[A-Za-z0-9+:-_@. The validated type that reaches the subprocess argv.
   39 newtype SignalAccount = SignalAccount Text
   40   deriving stock (Eq, Show)
   41 
   42 mkSignalAccount :: Text -> Either Text SignalAccount
   43 mkSignalAccount t
   44   | T.null t        = Left "signal account is empty"
   45   | T.head t == '-' = Left "signal account must not start with '-' (option injection)"
   46   | not (T.all validAccountChar t)
   47                     = Left "signal account has invalid characters"
   48   | otherwise       = Right (SignalAccount t)
   49 
   50 validAccountChar :: Char -> Bool
   51 validAccountChar c = c `Set.member` accountChars
   52   where
   53     accountChars :: Set Char
   54     accountChars = Set.fromList
   55       $ ['A'..'Z'] <> ['a'..'z'] <> ['0'..'9'] <> "+-:_"
   56 
   57 signalAccountText :: SignalAccount -> Text
   58 signalAccountText (SignalAccount t) = t
   59 
   60 -- ---------------------------------------------------------------------------
   61 -- SignalConfig — the [signal] config section
   62 -- ---------------------------------------------------------------------------
   63 
   64 -- | The @[signal]@ config section. All fields optional at the file level;
   65 -- @seal signal@ fails fast via 'resolveSignalConfig' if 'scAccount' is
   66 -- unset (no config account AND no vault-supplied account).
   67 data SignalConfig = SignalConfig
   68   { scAccount        :: Maybe Text       -- ^ phone number or UUID
   69   , scTextChunkLimit :: Maybe Int        -- ^ default 'defaultSignalChunkLimit'
   70   , scAllowFrom      :: AllowList Text   -- ^ sender allow-list (phone or UUID)
   71   } deriving stock (Eq, Show)
   72 
   73 -- | Signal's text-message character limit. Replies longer than this are
   74 -- chunked via 'Seal.Channels.Signal.Transport.chunkMessage'.
   75 defaultSignalChunkLimit :: Int
   76 defaultSignalChunkLimit = 1998
   77 
   78 defaultSignalConfig :: SignalConfig
   79 defaultSignalConfig = SignalConfig
   80   { scAccount = Nothing
   81   , scTextChunkLimit = Just defaultSignalChunkLimit
   82   , scAllowFrom = AllowAll
   83   }
   84 
   85 -- ---------------------------------------------------------------------------
   86 -- TOML codec (bidirectional, used by Seal.Config.File)
   87 -- ---------------------------------------------------------------------------
   88 
   89 -- | Bidirectional tomland codec for the @[signal]@ section. 'scAllowFrom'
   90 -- is a TOML array of strings (@allow_from = ["+1", "uuid:abc"]@); an absent
   91 -- array decodes as 'AllowAll'.
   92 signalConfigCodec :: Toml.TomlCodec SignalConfig
   93 signalConfigCodec = SignalConfig
   94   <$> Toml.dioptional (Toml.text "account")         .= scAccount
   95   <*> Toml.dioptional (Toml.int "text_chunk_limit") .= scTextChunkLimit
   96   <*> allowListCodec                                 .= scAllowFrom
   97 
   98 -- | Codec for the @allow_from@ array: 'AllowAll' when absent, 'AllowOnly'
   99 -- of the listed strings when present.
  100 allowListCodec :: Toml.TomlCodec (AllowList Text)
  101 allowListCodec = Toml.dimap toList fromList (Toml.dioptional (Toml.arrayOf Toml._Text "allow_from"))
  102   where
  103     toList :: AllowList Text -> Maybe [Text]
  104     toList AllowAll       = Nothing
  105     toList (AllowOnly xs) = Just (Set.toList xs)
  106     fromList :: Maybe [Text] -> AllowList Text
  107     fromList Nothing   = AllowAll
  108     fromList (Just xs) = AllowOnly (Set.fromList xs)
  109 
  110 -- ---------------------------------------------------------------------------
  111 -- Resolution — config section + vault account -> validated (account, limit, allow)
  112 -- ---------------------------------------------------------------------------
  113 
  114 -- | Resolve the @[signal]@ section + an optional vault-supplied account
  115 -- label into a validated 'SignalAccount', chunk limit, and
  116 -- 'AllowList UserId' for the channel's sender check. The vault account (if
  117 -- supplied) overrides the config account; either way it's smart-constructed
  118 -- here. A missing account (no section, no vault account) is rejected. The
  119 -- @allow_from@ 'AllowList Text' is mapped through 'mkUserId' to an
  120 -- 'AllowList UserId' — a malformed entry fails the whole resolution with a
  121 -- clear error.
  122 resolveSignalConfig
  123   :: Maybe SignalConfig -> Maybe Text
  124   -> Either Text (SignalAccount, Int, AllowList UserId)
  125 resolveSignalConfig mSection mVaultAccount =
  126   case mVaultAccount <|> (mSection >>= scAccount) of
  127     Nothing -> Left "signal: no account configured (set [signal].account or supply via vault)"
  128     Just acctRaw -> do
  129       acct <- mkSignalAccount acctRaw
  130       let sigCfg = fromMaybe defaultSignalConfig mSection
  131           limit  = fromMaybe defaultSignalChunkLimit (scTextChunkLimit sigCfg)
  132       allowUserId <- mapAllowUserId (scAllowFrom sigCfg)
  133       Right (acct, limit, allowUserId)
  134 
  135 -- | Map an 'AllowList Text' through 'mkUserId' to an 'AllowList UserId'.
  136 -- 'AllowAll' passes through; 'AllowOnly' maps each element (a malformed
  137 -- entry fails the whole resolution).
  138 mapAllowUserId :: AllowList Text -> Either Text (AllowList UserId)
  139 mapAllowUserId AllowAll = Right AllowAll
  140 mapAllowUserId (AllowOnly txts) = do
  141   uids <- traverse mkUserIdText (Set.toList txts)
  142   Right (AllowOnly (Set.fromList uids))
  143   where
  144     mkUserIdText t = case mkUserId t of
  145       Right u  -> Right u
  146       Left err -> Left ("signal allow_from has invalid user id " <> T.pack (show t) <> ": " <> err)

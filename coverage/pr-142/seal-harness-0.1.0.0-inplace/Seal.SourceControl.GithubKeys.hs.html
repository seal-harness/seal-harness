    3 -- | GitHub's published SSH host keys, pinned at compile time via
    4 -- 'file-embed' (tamper-resistant — the bytes are baked into the binary;
    5 -- a runtime attacker cannot swap them without rebuilding). Public data:
    6 -- <https://docs.github.com/en/authentication/keeping-your-account-and-data-secure/githubs-ssh-key-fingerprints>
    7 --
    8 -- Used as the @UserKnownHostsFile@ content for the no-disk clone seam (W2):
    9 -- @StrictHostKeyChecking=yes@ (NOT @accept-new@ / @/dev/null@ — §5.3: MITM
   10 -- defense) reads these pinned keys, so a GitHub clone verifies the server
   11 -- against a known-good host key without ever touching @~/.ssh/known_hosts@
   12 -- and without prompting. The file is written per-op to
   13 -- @<workdir>/.seal-known-hosts@ (public data, 0644, cleaned up after the op).
   14 module Seal.SourceControl.GithubKeys
   15   ( pinnedGithubKnownHosts
   16   ) where
   17 
   18 import Data.ByteString (ByteString)
   19 import Data.FileEmbed (embedFile)
   20 
   21 -- | The pinned GitHub host keys (RSA, ECDSA, Ed25519) in OpenSSH
   22 -- @known_hosts@ format. Embedded at compile time.
   23 pinnedGithubKnownHosts :: ByteString

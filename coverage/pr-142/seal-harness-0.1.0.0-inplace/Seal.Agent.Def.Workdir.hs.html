    2 -- | The workdir-scoped agent-definition discovery backend.
    3 --
    4 -- This module scans a session workdir (the workspace root, via the
    5 -- 'WorkdirFs' handle) for agent definitions shipped by cloned
    6 -- repositories. It is the single home for ALL workspace-derived agent-def
    7 -- discovery; the user store lives in "Seal.Agent.Def.Backend" (§3.9).
    8 --
    9 -- **No direct filesystem access.** This module deliberately does NOT
   10 -- import @System.Directory@ or @Data.Text.IO@ — every workspace read goes
   11 -- through the 'WorkdirFs' handle (single SafePath-confined chokepoint,
   12 -- §3.6). The compile-fail fixture
   13 -- @Seal.Agent.Def.BackendNoDirectFsFailSpec@ asserts this invariant by
   14 -- grepping this module's source for those import lines.
   15 --
   16 -- Two on-disk schemes are discovered (mirroring the user store):
   17 --
   18 -- 1. **FlatScheme** — one Markdown file per def at @agents\/\<id\>.md@,
   19 --    frontmatter + body = system prompt.
   20 -- 2. **DirScheme** (PureClaw-compatible) — a subdirectory per agent at
   21 --    @agents\/\<id\>\/@, with optional TOML frontmatter on @AGENTS.md@
   22 --    and the system prompt composed by reading bootstrap files
   23 --    (@SOUL.md@, @USER.md@, @AGENTS.md@ body, @MEMORY.md@,
   24 --    @IDENTITY.md@, @TOOLS.md@, @BOOTSTRAP.md@) in fixed order with
   25 --    @--- SOUL ---@-style section markers.
   26 --
   27 -- For @.agents\/@, the [.agents Protocol](https://dotagentsprotocol.com) is
   28 -- recognized: when @.agents\/@ contains @agents.md@ OR an @agents\/@
   29 -- subdirectory, it is treated as a **protocol root** — the project-level
   30 -- @.agents\/agents.md@ is loaded as an agent def (id @agents-md@), and each
   31 -- @.agents\/agents\/\<id\>\/agent.md@ sub-agent is loaded. Otherwise
   32 -- @.agents\/@ falls back to the legacy hybrid flat + DirScheme discovery.
   33 --
   34 -- @.seal\/agents@ and @agents@ always use the legacy discovery.
   35 --
   36 -- The re-exports from "Seal.Agent.Def.Backend" keep the public API
   37 -- stable ('workdirAgentDefBackend', 'composeDirSystemPrompt',
   38 -- 'encodeAgentDef', 'decodeAgentDef', etc.).

  166 -- | Optional TOML frontmatter on @AGENTS.md@ inside an agent directory.
  167 -- All fields are optional; unknown fields are ignored by the codec. A
  168 -- missing @AGENTS.md@ or unparseable frontmatter yields
  169 -- 'defaultDirAgentConfig'.
  170 data DirAgentConfig = DirAgentConfig
  171   { dacModel    :: Maybe Text
  172   , dacProvider :: Maybe Text
  173   , dacTools    :: Maybe [Text]   -- ^ TOML array of opcode names; absent -> 'AllowAll'
  174   } deriving stock (Eq, Show)

  176 -- | 'DirAgentConfig' with every field unset (the permissive fallback).
  177 defaultDirAgentConfig :: DirAgentConfig
  178 defaultDirAgentConfig = DirAgentConfig Nothing Nothing Nothing

  180 -- | Bidirectional tomland codec for 'DirAgentConfig'. Mirrors the
  181 -- @Toml.dioptional (Toml.text ...)@ pattern used in 'Seal.Config.File'.
  182 -- @tools@ uses @Toml.arrayOf Toml._Text@ to decode a TOML array of
  183 -- strings (e.g. @tools = ["FILE_READ", "ASK_HUMAN"]@).
  184 dirAgentConfigCodec :: Toml.TomlCodec DirAgentConfig
  185 dirAgentConfigCodec = DirAgentConfig
  186   <$> Toml.dioptional (Toml.text "model")              .= dacModel
  187   <*> Toml.dioptional (Toml.text "provider")           .= dacProvider
  188   <*> Toml.dioptional (Toml.arrayOf Toml._Text "tools") .= dacTools
  189 
  190 -- | Parse the TOML frontmatter off an @AGENTS.md@ document. A document
  191 -- with no frontmatter yields 'defaultDirAgentConfig'. Parse failures also
  192 -- yield 'defaultDirAgentConfig' (permissive — matches PureClaw's
  193 -- 'loadAgentConfig' fallback).
  194 parseDirAgentConfig :: Text -> DirAgentConfig
  195 parseDirAgentConfig input =
  196   case splitFrontmatterRaw input of
  197     (Nothing, _)   -> defaultDirAgentConfig
  198     (Just "", _)   -> defaultDirAgentConfig
  199     (Just toml, _) ->
  200       case Toml.decode dirAgentConfigCodec toml of
  201         Left _   -> defaultDirAgentConfig
  202         Right cfg -> cfg

  239 -- | Truncate a section body to @limit@ characters, appending the exact
  240 -- truncation marker. Strings at or under the limit are returned as-is.
  241 truncateSection :: Int -> Text -> Text
  242 truncateSection limit txt
  243   | T.length txt <= limit = txt
  244   | otherwise =
  245       T.take limit txt
  246         <> "\n[...truncated at " <> T.pack (show limit) <> " chars...]"

  252 -- | Encode an 'AgentDef' as a Markdown document (frontmatter + body, where
  253 -- the body is the system prompt).
  254 encodeAgentDef :: AgentDef -> Text
  255 encodeAgentDef d = encodeDoc fm body
  256   where
  257     ModelId modelName = adModel d
  258     body = fromMaybe "" (adSystem d)
  259     fm = Map.fromList $
  260       [ ("id", agentDefIdText (adId d))
  261       , ("name", adName d)
  262       , ("provider", adProvider d)
  263       , ("model", modelName)
  264       , ("tools", renderTools (adTools d))
  265       , ("created_at", isoTime (adCreatedAt d))
  266       , ("updated_at", isoTime (adUpdatedAt d))
  267       , ("session", sessionIdText (adSession d))
  268       ] ++ maybe [] (\g -> [("group", g)]) (adGroup d)

  270 -- | Decode a Markdown document into an 'AgentDef'. Returns 'Nothing' if the id
  271 -- field is missing or fails 'mkAgentDefId'.
  272 decodeAgentDef :: Text -> Maybe AgentDef
  273 decodeAgentDef content =
  274   case decodeDoc content of
  275     (fm, body) -> do
  276       aidT <- fmLookup "id" fm
  277       aid  <- either (const Nothing) Just (mkAgentDefId aidT)
  278       Just AgentDef
  279         { adId = aid
  280         , adName = fromMaybe "" (fmLookup "name" fm)
  281         , adProvider = fromMaybe "" (fmLookup "provider" fm)
  282         , adModel = ModelId (fromMaybe "" (fmLookup "model" fm))
  283         , adSystem = if T.null body then Nothing else Just body
  284         , adTools = decodeTools fm
  285         , adGroup = fmLookup "group" fm
  286         , adCreatedAt = parseTime (fmLookup "created_at" fm)
  287         , adUpdatedAt = parseTime (fmLookup "updated_at" fm)
  288         , adSession = fromRight (mkSystemSessionId "unknown") (mkSessionId (fromMaybe "unknown" (fmLookup "session" fm)))
  289         }

  295 -- | Decode the @tools@ frontmatter value (a 'Maybe [Text]' from the TOML
  296 -- codec): @Nothing@ -> 'AllowAll'; a list of opcode-name strings ->
  297 -- 'AllowOnly'. The string @"all"@ is not special here (the TOML codec
  298 -- produces a list, not a scalar); callers wanting @AllowAll@ simply omit
  299 -- the @tools@ key.
  300 decodeDirTools :: Maybe [Text] -> AllowList OpName
  301 decodeDirTools Nothing     = AllowAll
  302 decodeDirTools (Just [])   = AllowAll
  303 decodeDirTools (Just ts)   = AllowOnly (Set.fromList (map OpName ts))
  304 
  305 -- | Render an 'AllowList OpName' as @"all"@ or a JSON array string.
  306 renderTools :: AllowList OpName -> Text
  307 renderTools AllowAll       = "all"
  308 renderTools (AllowOnly xs) =
  309   TE.decodeUtf8 (BL.toStrict (encode (V.fromList [ String t | OpName t <- Set.toList xs ])))
  310 
  311 -- | Decode the @tools@ frontmatter field (flat scheme): @"all"@ ->
  312 -- 'AllowAll'; a JSON array of opcode-name strings -> 'AllowOnly';
  313 -- absent/other -> 'AllowAll'.
  314 decodeTools :: Map Text Text -> AllowList OpName
  315 decodeTools fm = case Map.lookup "tools" fm of
  316   Nothing    -> AllowAll
  317   Just "all" -> AllowAll
  318   Just _     -> case fmLookupList "tools" fm of
  319     Just ts -> AllowOnly (Set.fromList (map OpName ts))
  320     Nothing -> AllowAll
  321 
  322 -- | Render a 'UTCTime' as an ISO-8601 string (UTC, with @Z@ suffix).
  323 isoTime :: UTCTime -> Text
  324 isoTime = T.pack . formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%SZ"
  325 
  326 -- | Parse an ISO-8601 'UTCTime' from a frontmatter value. Defaults to epoch 0
  327 -- when absent or unparseable.
  328 parseTime :: Maybe Text -> UTCTime
  329 parseTime Nothing    = epochZero
  330 parseTime (Just raw) = fromMaybe epochZero (parseTimeM True defaultTimeLocale "%Y-%m-%dT%H:%M:%SZ" (T.unpack raw))
  331 
  332 -- | The epoch fallback for missing/unparseable timestamps.
  333 epochZero :: UTCTime
  334 epochZero = UTCTime (fromGregorian 1970 1 1) (secondsToDiffTime 0)

  340 -- | Read a single bootstrap section file, applying size/empty/truncation
  341 -- rules. Returns 'Nothing' when the file is missing, empty
  342 -- (including whitespace-only), or rejected as oversized. For @AGENTS.md@,
  343 -- only the body after the TOML frontmatter fence is injected (the
  344 -- frontmatter itself lives in 'DirAgentConfig').
  345 --
  346 -- Reads through the 'WorkdirFs' handle (single confinement chokepoint,
  347 -- §3.6); the section file is looked up as @\<sectionFileName\>@ directly
  348 -- under the anchor. The @limit@ is the per-section character cap; the
  349 -- byte-level size cap is applied inside 'wfsReadFile'.
  350 readSection :: WorkdirFs -> SectionKind -> Int -> IO (Maybe Text)
  351 readSection fs kind limit = do
  352   let rel = T.pack (sectionFileName kind)
  353   eTxt <- wfsReadFile fs =<< rpOrDie rel
  354   case eTxt of
  355     Left _ -> pure Nothing
  356     Right txt ->
  357       let contents = case kind of
  358             AgentsK -> case splitFrontmatterRaw txt of
  359               (_, body) -> body
  360             _ -> txt
  361           trimmed = T.dropWhileEnd Char.isSpace contents
  362       in if T.null (T.strip trimmed)
  363            then pure Nothing
  364            else pure (Just (truncateSection limit trimmed))
  365 
  366 -- | Compose a system prompt from an agent directory's bootstrap files.
  367 -- Files are read in the fixed injection order (SOUL, USER, AGENTS,
  368 -- MEMORY, IDENTITY, TOOLS, BOOTSTRAP), missing/empty/oversized files are
  369 -- skipped, and any section exceeding @limit@ characters is truncated with
  370 -- the exact marker @"\\n[...truncated at \<limit\> chars...]"@. Returns
  371 -- the empty string if every section is missing/empty.
  372 --
  373 -- The 'WorkdirFs' is anchored at the agent directory (the section files
  374 -- are looked up as @SOUL.md@\/@AGENTS.md@\/… directly under the anchor).
  375 composeDirSystemPrompt :: WorkdirFs -> Int -> IO Text
  376 composeDirSystemPrompt fs limit = do
  377   let kinds = [SoulK, UserK, AgentsK, MemoryK, IdentityK, ToolsK, BootstrapK]
  378   sections <- mapM (\k -> readSection fs k limit) kinds
  379   let rendered = [ sectionMarker k <> "\n" <> body
  380                  | (k, Just body) <- zip kinds sections
  381                  ]
  382   pure (T.intercalate "\n\n" rendered)

  384 -- | Load a DirScheme agent def from @agents\/\<id\>\/@. Composes the
  385 -- system prompt eagerly from the bootstrap files. Returns 'Nothing' if
  386 -- the directory does not exist or the dirname fails 'isValidAgentDefId'.
  387 --
  388 -- The 'WorkdirFs' is anchored at the agents directory (the per-agent
  389 -- subdirectory is @\<id\>@ under the anchor).
  390 loadDirAgentDef :: WorkdirFs -> AgentDefId -> IO (Maybe AgentDef)
  391 loadDirAgentDef fs aid = do
  392   let dirRp = agentDefIdText aid
  393   exists <- wfsDoesDirectoryExist fs =<< rpOrDie dirRp
  394   if not exists
  395     then pure Nothing
  396     else do
  397       let subFs = reanchorFs fs dirRp
  398       cfg  <- loadDirAgentConfig subFs
  399       body <- composeDirSystemPrompt subFs defaultSectionCharLimit
  400       mtime <- dirMTime subFs
  401       pure (Just AgentDef
  402         { adId = aid
  403         , adName = agentDefIdText aid
  404         , adProvider = fromMaybe "" (dacProvider cfg)
  405         , adModel = ModelId (fromMaybe "" (dacModel cfg))
  406         , adSystem = if T.null body then Nothing else Just body
  407         , adTools = decodeDirTools (dacTools cfg)
  408         , adGroup = Nothing
  409         , adCreatedAt = mtime
  410         , adUpdatedAt = mtime
  411         , adSession = mkSystemSessionId "manual"
  412         })

  414 -- | Read and parse the @AGENTS.md@ frontmatter inside an agent directory.
  415 -- Missing file or parse failure yields 'defaultDirAgentConfig'. The
  416 -- 'WorkdirFs' is anchored at the agent directory.
  417 loadDirAgentConfig :: WorkdirFs -> IO DirAgentConfig
  418 loadDirAgentConfig fs = do
  419   eTxt <- wfsReadFile fs =<< rpOrDie "AGENTS.md"
  420   case eTxt of
  421     Left _   -> pure defaultDirAgentConfig
  422     Right txt -> pure (parseDirAgentConfig txt)

  424 -- | Best-effort mtime for an agent directory: the mtime of @AGENTS.md@ if
  425 -- present, else the mtime of @SOUL.md@, else epoch zero. (Unstable across
  426 -- @git pull@ — see the design doc §3.2 timestamp note.) The 'WorkdirFs'
  427 -- is anchored at the agent directory.
  428 dirMTime :: WorkdirFs -> IO UTCTime
  429 dirMTime fs = go ["AGENTS.md", "SOUL.md"]
  430   where
  431     go [] = pure epochZero
  432     go (p:ps) = do
  433       eMt <- wfsModificationTime fs =<< rpOrDie (T.pack p)
  434       case eMt of
  435         Left _  -> go ps
  436         Right t -> pure t

  465 -- | Decode a protocol @agent.md@ frontmatter + body into an 'AgentDef'.
  466 -- The id is the subdir name, unless frontmatter @id@ is present and valid
  467 -- (then frontmatter wins). @enabled@ absent → enabled (appears); explicit
  468 -- @enabled: false@ → 'Nothing' (skipped per §3.3). Frontmatter
  469 -- @name@/@provider@/@model@ honored when present.
  470 decodeProtocolAgentMd :: Text -> Text -> Maybe AgentDef
  471 decodeProtocolAgentMd subDirName content =
  472   case decodeDoc content of
  473     (fm, body) ->
  474       case fmLookup "enabled" fm of
  475         Just v | T.strip v == "false" -> Nothing  -- disabled → skip
  476         _ ->
  477           let idText = fromMaybe subDirName (fmLookup "id" fm >>= validId)
  478           in case mkAgentDefId idText of
  479                Left _ -> Nothing
  480                Right aid -> Just AgentDef
  481                  { adId = aid
  482                  , adName = fromMaybe idText (fmLookup "name" fm)
  483                  , adProvider = fromMaybe "" (fmLookup "provider" fm)
  484                  , adModel = ModelId (fromMaybe "" (fmLookup "model" fm))
  485                  , adSystem = if T.null body then Nothing else Just body
  486                  , adTools = decodeTools fm
  487                  , adGroup = Nothing
  488                  , adCreatedAt = epochZero
  489                  , adUpdatedAt = epochZero
  490                  , adSession = mkSystemSessionId "manual"
  491                  }
  492   where
  493     validId t = if isValidAgentDefId t then Just t else Nothing

  519 -- | The detection predicate (§3.3): @.agents\/@ is a protocol root iff it
  520 -- contains @agents.md@ (a file) OR an @agents\/@ subdirectory. The
  521 -- 'WorkdirFs' is anchored at the @.agents\/@ directory.
  522 isProtocolRoot :: WorkdirFs -> IO Bool
  523 isProtocolRoot fs = do
  524   hasAgentsMd  <- wfsDoesFileExist fs =<< rpOrDie "agents.md"
  525   hasAgentsSub <- wfsDoesDirectoryExist fs =<< rpOrDie "agents"
  526   pure (hasAgentsMd || hasAgentsSub)

  528 -- | The protocol allow-list scan (§3.3). Scans only @agents\/\<id\>\/agent.md@
  529 -- (sub-agents) + @agents.md@ (project def). Skips everything else
  530 -- (@skills\/@, @tasks\/@, @memories\/@, @mcp.json@). The 'WorkdirFs' is
  531 -- anchored at the @.agents\/@ directory.
  532 listProtocolAgentDefs :: WorkdirFs -> IO [AgentDef]
  533 listProtocolAgentDefs fs = do
  534   -- The project-level agents.md (id agents-md).
  535   mProject <- loadProjectAgentDef fs
  536   -- Sub-agents under agents/<id>/agent.md.
  537   subExists <- wfsDoesDirectoryExist fs =<< rpOrDie "agents"
  538   subDefs <- if not subExists
  539     then pure []
  540     else do
  541       eEntries <- wfsListDirectory fs =<< rpOrDie "agents"
  542       let entries = fromRight [] eEntries
  543           candidate e = do
  544             isDir <- wfsDoesDirectoryExist fs =<< rpOrDie ("agents/" <> e)
  545             pure (if isDir && isValidAgentDefId e then Just e else Nothing)
  546       mEntries <- mapM candidate (sortOn id entries)
  547       catMaybes <$> mapM (loadProtocolSubAgent fs) (catMaybes mEntries)
  548   let allDefs = catMaybes [mProject] <> subDefs
  549   pure (sortOn (agentDefIdText . adId) allDefs)

  551 -- | Dispatch for the @.agents\/@ convention dir: protocol-root detection
  552 -- (§3.3). When @.agents\/@ contains @agents.md@ OR an @agents\/@
  553 -- subdirectory, scan the protocol allow-list; otherwise fall back to legacy
  554 -- 'listAgentDefsFs' (DirScheme with @SOUL.md@\/@AGENTS.md@ bootstrap
  555 -- files). The 'WorkdirFs' is anchored at the @.agents\/@ directory.
  556 listAgentsDotAgents :: WorkdirFs -> IO [AgentDef]
  557 listAgentsDotAgents fs = do
  558   isProto <- isProtocolRoot fs
  559   if isProto
  560     then listProtocolAgentDefs fs
  561     else listAgentDefsFs fs

  567 -- | A read-only 'AgentDefBackend' that scans a session workdir for agent
  568 -- definitions shipped by cloned repositories. For each top-level directory
  569 -- in the workdir (a cloned repo), it checks the conventional agent-def
  570 -- locations (@.agents\/@, @.seal\/agents\/@, @agents\/@).
  571 --
  572 -- This backend is /read-only/: 'adbUpdate'/'adbDelete' are no-ops (repo-local
  573 -- agent defs are immutable from the model's perspective). 'adbRead' scans
  574 -- on every call (workdirs are small). Within-workdir collisions (two repos
  575 -- ship a def with the same id): the alphabetically-first repo wins
  576 -- (deterministic; same as the skill backend's policy).
  577 --
  578 -- Every file open goes through the 'WorkdirFs' handle (symlink-escape
  579 -- confinement — §3.8; single chokepoint, §3.6) and is size-capped at
  580 -- 'maxScanBytes' + 'truncateSection'.
  581 workdirAgentDefBackend :: WorkdirFs -> IO AgentDefBackend
  582 workdirAgentDefBackend fs = pure AgentDefBackend
  583     { adbRead   = \aid -> do
  584         defs <- listWorkdirAgentDefs fs
  585         pure (Map.lookup aid (Map.fromList [(adId d, d) | d <- defs]))
  586     , adbUpdate = \_ -> pure ()
  587     , adbList   = listWorkdirAgentDefs fs
  588     , adbDelete = \_ -> pure ()
  589     }
  590 
  591 -- | Enumerate every agent def found under the conventional locations across
  592 -- all top-level directories (cloned repos) in the workdir anchored at the
  593 -- 'WorkdirFs'. The alphabetically-first repo wins on prefixed-id
  594 -- collisions (deterministic). Missing or empty workdirs yield @[]@.
  595 --
  596 -- Each def's id is prefixed with its repo's top-level directory name + @"--"@
  597 -- (e.g. @vtag--architect-agent@) and its @adName@ with @\<repo\>\/@ (e.g.
  598 -- @vtag\/Architect Agent@). This disambiguates repo-local agents from the
  599 -- user's own agents (and from other repos' agents) when the union backend
  600 -- merges workdir ⊕ user. The @--@ separator is charset-safe
  601 -- (@isValidAgentDefId@); the @/@ in the display name is for human
  602 -- readability only.
  603 --
  604 -- /Round-trip economy/: all structural questions (which repos exist, which
  605 -- convention dirs exist, which entries are dirs) are answered from ONE
  606 -- 'wfsSnapshot' call; only file CONTENTS are read through 'wfsReadFile'
  607 -- (the symlink-containment chokepoint). On @mode=remote@ this collapses a
  608 -- scan from one SSH round trip per probe to one round trip + two calls per
  609 -- discovered def file.
  610 listWorkdirAgentDefs :: WorkdirFs -> IO [AgentDef]
  611 listWorkdirAgentDefs fs = do
  612   eSnap <- wfsSnapshot fs
  613   case eSnap of
  614     -- Fail-soft-to-empty: a missing/stub/failing workdir yields no defs —
  615     -- the same terminal state as today's empty listings.
  616     Left _  -> pure []
  617     Right snap -> listWorkdirAgentDefsSnap snap fs
  618 
  619 -- | 'listWorkdirAgentDefs' over a caller-supplied snapshot — lets a
  620 -- consumer (the discovery cache) fetch ONE workspace snapshot and serve
  621 -- both scanners from it.
  622 listWorkdirAgentDefsSnap :: WorkspaceSnapshot -> WorkdirFs -> IO [AgentDef]
  623 listWorkdirAgentDefsSnap snap fs = do
  624   let repos = snapTopDirs snap
  625   perRepo <- forM repos $ \repo -> do
  626     defs <- concat <$> forM workdirAgentDefConventions (\conv -> do
  627       let convRp = repo <> "/" <> T.pack conv
  628       if not (snapIsDirectoryAt snap convRp)
  629         then pure []
  630         else
  631           -- Dispatch: .agents/ is protocol-aware; the rest use legacy.
  632           if conv == ".agents"
  633             then listAgentsDotAgentsSnap snap fs convRp
  634             else listAgentDefsFsSnap snap fs convRp)
  635     pure (mapMaybe (prefixWorkdirDef repo) defs)
  636   let merge m [] = m
  637       merge m (d:ds) = merge (Map.insertWith (\_new old -> old) (adId d) d m) ds
  638       merged = merge Map.empty (concat perRepo)
  639   pure (Map.elems merged)

  641 -- | Basename of a snapshot-relative path (@a\/b\/c@ → @c@). Inputs are
  642 -- snapshot-derived paths, never user input.
  643 snapBasename :: Text -> Text
  644 snapBasename = snd . T.breakOnEnd "/"

  646 -- | A read-only backend serving a PRE-COMPUTED def list (the discovery
  647 -- cache's point-in-time scan of the workdir). Mirrors
  648 -- 'workdirAgentDefBackend''s semantics: reads from memory, writes/deletes
  649 -- are no-ops (repo-local defs are immutable from the model's perspective).
  650 staticAgentDefBackend :: [AgentDef] -> IO AgentDefBackend
  651 staticAgentDefBackend defs = pure AgentDefBackend
  652   { adbRead   = \aid -> pure (Map.lookup aid (Map.fromList [(adId d, d) | d <- defs]))
  653   , adbUpdate = \_ -> pure ()
  654   , adbList   = pure defs
  655   , adbDelete = \_ -> pure ()
  656   }

  658 -- | Snapshot-driven dispatch for the @.agents\/@ convention dir: protocol-
  659 -- root detection (§3.3) answered from the snapshot, contents read through
  660 -- the anchored 'WorkdirFs'. Mirrors 'listAgentsDotAgents'.
  661 listAgentsDotAgentsSnap :: WorkspaceSnapshot -> WorkdirFs -> Text -> IO [AgentDef]
  662 listAgentsDotAgentsSnap snap fs anchor
  663   | snapIsFileAt snap (anchor <> "/agents.md")
  664       || snapIsDirectoryAt snap (anchor <> "/agents")
  665   = listProtocolAgentDefsSnap snap fs anchor
  666   | otherwise = listAgentDefsFsSnap snap fs anchor
  667 
  668 -- | Snapshot-driven protocol allow-list scan (§3.3): @agents.md@ (project
  669 -- def) + @agents\/\<id\>\/agent.md@ (sub-agents), mirroring
  670 -- 'listProtocolAgentDefs' with structure from the snapshot.
  671 listProtocolAgentDefsSnap :: WorkspaceSnapshot -> WorkdirFs -> Text -> IO [AgentDef]
  672 listProtocolAgentDefsSnap snap fs anchor = do
  673   let anchored = reanchorFs fs anchor
  674       agentsPath = anchor <> "/agents"
  675   mProject <- loadProjectAgentDef anchored
  676   subDefs <-
  677     if not (snapIsDirectoryAt snap agentsPath)
  678       then pure []
  679       else do
  680         let candidates =
  681               filter isValidAgentDefId (map snapBasename (snapChildDirsAt snap agentsPath))
  682         catMaybes <$> mapM (loadProtocolSubAgent anchored) candidates
  683   pure (sortOn (agentDefIdText . adId) (catMaybes [mProject] <> subDefs))

  685 -- | Snapshot-driven legacy enumerator (flat + DirScheme), mirroring
  686 -- 'listAgentDefsFs' with structure from the snapshot. Flat @\<id\>.md@
  687 -- files win on collision with same-id dirs.
  688 listAgentDefsFsSnap :: WorkspaceSnapshot -> WorkdirFs -> Text -> IO [AgentDef]
  689 listAgentDefsFsSnap snap fs anchor = do
  690   let mdFiles = [p | p <- snapChildFilesAt snap anchor, ".md" `T.isSuffixOf` p]
  691   flatDefs <- forM mdFiles $ \rel -> do
  692     eContent <- wfsReadFile fs =<< rpOrDie rel
  693     pure $ case eContent of
  694       Left _    -> Nothing
  695       Right txt -> decodeAgentDef txt
  696   let flatIds = Set.fromList (map (agentDefIdText . adId) (catMaybes flatDefs))
  697       dirNames =
  698         [ n | n <- map snapBasename (snapChildDirsAt snap anchor)
  699             , isValidAgentDefId n, not (Set.member n flatIds) ]
  700   dirDefs <- forM dirNames $ \n ->
  701     case mkAgentDefId n of
  702       Left _   -> pure Nothing
  703       Right aid -> loadDirAgentDef (reanchorFs fs anchor) aid
  704   pure (sortOn (agentDefIdText . adId) (catMaybes flatDefs <> catMaybes dirDefs))
  705 
  706 -- | Prefix a workdir-discovered def's id with @\<repo\>--\<id\>@ and its
  707 -- @adName@ with @\<repo\>\/\<name\>@. The id prefix uses @"--"@ (charset-safe
  708 -- per 'isValidAgentDefId'); the display name uses @"/"@ for readability. If
  709 -- the prefixed id fails validation (e.g. the repo dir has a char outside the
  710 -- charset), the def is dropped ('Nothing' — fail-closed).
  711 prefixWorkdirDef :: Text -> AgentDef -> Maybe AgentDef
  712 prefixWorkdirDef repo d =
  713   let prefixedIdText = repo <> "--" <> agentDefIdText (adId d)
  714   in case mkAgentDefId prefixedIdText of
  715        Left _ -> Nothing
  716        Right aid -> Just d
  717          { adId = aid
  718          , adName = repo <> "/" <> adName d
  719          }

  721 -- ---------------------------------------------------------------------------
  722 -- Workdir-local legacy enumerator (flat + dir, over WorkdirFs)
  723 -- ---------------------------------------------------------------------------
  724 
  725 -- | The workdir-local legacy enumerator (mirrors the user store's
  726 -- 'listAgentDefs' but over 'WorkdirFs'). Enumerates flat @\<id\>.md@ files
  727 -- + DirScheme @\<id\>\/@ subdirectories under the anchor, sorted by id.
  728 -- Malformed entries are skipped. On flat/dir collision for the same id,
  729 -- the flat file wins and the dir is dropped. The 'WorkdirFs' is anchored
  730 -- at the agents directory.
  731 listAgentDefsFs :: WorkdirFs -> IO [AgentDef]
  732 listAgentDefsFs fs = do
  733   eEntries <- wfsListDirectory fs =<< rpOrDie "."
  734   let entries = fromRight [] eEntries
  735   flatDefs <- collectFlatFs fs entries
  736   dirDefs  <- collectDirsFs fs entries (map adId flatDefs)
  737   pure (sortOn (agentDefIdText . adId) (flatDefs <> dirDefs))
  738 
  739 -- | Decode all flat @.md@ files under the anchor (workdir-local legacy).
  740 collectFlatFs :: WorkdirFs -> [Text] -> IO [AgentDef]
  741 collectFlatFs fs entries = do
  742   let mdFiles = [e | e <- entries, ".md" `T.isSuffixOf` e]
  743   defs <- forM mdFiles $ \e -> do
  744     eContent <- wfsReadFile fs =<< rpOrDie e
  745     pure $ case eContent of
  746       Left _    -> Nothing
  747       Right txt -> decodeAgentDef txt
  748   pure (catMaybes defs)
  749 
  750 -- | Load DirScheme defs from subdirectories under the anchor (workdir-local
  751 -- legacy), skipping ids already provided by the flat scheme (flat wins on
  752 -- conflict).
  753 collectDirsFs :: WorkdirFs -> [Text] -> [AgentDefId] -> IO [AgentDef]
  754 collectDirsFs fs entries flatIds = do
  755   let flatIdTexts = Set.fromList (map agentDefIdText flatIds)
  756       candidate e = do
  757         isDir <- wfsDoesDirectoryExist fs =<< rpOrDie e
  758         let validDir = isDir && isValidAgentDefId e
  759         pure (if validDir then Just e else Nothing)
  760   mEntries <- mapM candidate entries
  761   let dirNames = catMaybes mEntries
  762   defs <- forM dirNames $ \e -> do
  763     case mkAgentDefId e of
  764       Left _   -> pure Nothing
  765       Right aid ->
  766         if agentDefIdText aid `Set.member` flatIdTexts
  767           then pure Nothing  -- skip the dir def; flat wins
  768           else loadDirAgentDef fs aid
  769   pure (catMaybes defs)

  771 -- ---------------------------------------------------------------------------
  772 -- WorkdirFs re-anchoring + RemotePath helpers (internal)
  773 -- ---------------------------------------------------------------------------
  774 
  775 -- | Produce a 'WorkdirFs' "view" anchored at a sub-directory of the original
  776 -- anchor. Every 'wfs*' call on the re-anchored handle prepends @prefix@ to
  777 -- the supplied 'RemotePath'. Works for both the local and remote arms (no
  778 -- new constructor — pure record wrapper). The prefix is a relative path
  779 -- (e.g. @"my-repo\/.agents"@); the join uses @\/@ as the separator.
  780 reanchorFs :: WorkdirFs -> Text -> WorkdirFs
  781 reanchorFs fs prefix = WorkdirFs
  782   { wfsReadFile            = wfsReadFile fs           <=< joinRp prefix
  783   , wfsDoesFileExist       = wfsDoesFileExist fs      <=< joinRp prefix
  784   , wfsDoesDirectoryExist  = wfsDoesDirectoryExist fs <=< joinRp prefix
  785   , wfsListDirectory       = wfsListDirectory fs      <=< joinRp prefix
  786   , wfsFileSize            = wfsFileSize fs           <=< joinRp prefix
  787   , wfsModificationTime    = wfsModificationTime fs   <=< joinRp prefix
  788   , wfsSnapshot            = wfsSnapshot fs
  789     -- ^ Whole-workspace: re-anchoring changes where relative reads land,
  790     -- never the snapshot's root-relative frame.
  791   }
  792 
  793 -- | Join a prefix and a (possibly @"\.@) relative 'RemotePath' into a single
  794 -- 'RemotePath'. A @"\. "@ suffix collapses to the prefix (so re-anchoring at
  795 -- @"\. "@ is the identity). Crashes on invalid input — the prefix/suffix
  796 -- are internally generated from validated components, never user/LLM input.
  797 joinRp :: Text -> RemotePath -> IO RemotePath
  798 joinRp prefix rp' =
  799   let suffix = getRemotePath rp'
  800       joined
  801         | T.null prefix            = suffix
  802         | suffix == "."            = prefix
  803         | T.isPrefixOf "./" suffix = prefix <> "/" <> T.drop 2 suffix
  804         | otherwise                = prefix <> "/" <> suffix
  805   in case mkRemotePath joined of
  806        Right r  -> pure r
  807        Left err -> error ("joinRp: invalid remote path: " <> T.unpack err
  808                           <> ": " <> T.unpack joined)
  809 
  810 -- | Construct a 'RemotePath', crashing on invalid input. Used only for
  811 -- internally-generated, validated path components (never user/LLM input).
  812 rpOrDie :: Text -> IO RemotePath
  813 rpOrDie t = case mkRemotePath t of
  814   Right r  -> pure r
  815   Left err -> error ("rpOrDie: invalid remote path: " <> T.unpack err
  816                      <> ": " <> T.unpack t)

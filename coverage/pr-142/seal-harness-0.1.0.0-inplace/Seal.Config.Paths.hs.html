   38 -- | All paths derived from the seal home directory.
   39 --
   40 -- * 'spConfig' — version-controllable config tree; ordinary directory
   41 -- * 'spState'  — mutable runtime state; ordinary directory
   42 -- * 'spKeys'   — key material; created mode 0700, never version-controlled
   43 data SealPaths = SealPaths
   44   { spHome   :: FilePath   -- ^ @SEAL_HOME@ env var or @~\/.seal@
   45   , spConfig :: FilePath   -- ^ @\<home\>\/config@
   46   , spState  :: FilePath   -- ^ @\<home\>\/state@
   47   , spKeys   :: FilePath   -- ^ @\<home\>\/keys@
   48   , spCache  :: FilePath   -- ^ @\<home\>\/cache@ (per-session workdirs)
   49   } deriving stock (Eq, Show)
   50 
   51 -- | Resolve the seal home directory.
   52 --
   53 -- Returns the value of @SEAL_HOME@ when the variable is set; otherwise
   54 -- returns @~\/.seal@ via 'getHomeDirectory'.
   55 resolveSealHome :: IO FilePath
   56 resolveSealHome = do
   57   mEnv <- lookupEnv "SEAL_HOME"
   58   case mEnv of
   59     Just h  -> pure h
   60     Nothing -> do
   61       home <- getHomeDirectory
   62       pure (home </> ".seal")
   63 
   64 -- | Compute all sub-paths under the seal home directory without touching
   65 -- the filesystem.
   66 getSealPaths :: IO SealPaths
   67 getSealPaths = do
   68   home <- resolveSealHome
   69   pure SealPaths
   70     { spHome   = home
   71     , spConfig = home </> "config"
   72     , spState  = home </> "state"
   73     , spKeys   = home </> "keys"
   74     , spCache  = home </> "cache"
   75     }
   76 
   77 -- | Create the seal directory tree, setting restrictive permissions on the
   78 -- keys directory.
   79 --
   80 -- * @config\/@ and @state\/@ are created with default (umask-governed) mode.
   81 -- * @keys\/@ is created and then explicitly set to mode @0700@.
   82 --
   83 -- Calling this function when the directories already exist is safe
   84 -- ('createDirectoryIfMissing' is idempotent; 'setFileMode' is idempotent).
   85 ensureSealDirs :: SealPaths -> IO ()
   86 ensureSealDirs paths = do
   87   createDirectoryIfMissing True (spConfig paths)
   88   createDirectoryIfMissing True (spState  paths)
   89   createDirectoryIfMissing True (spCache  paths)
   90   createDirectoryIfMissing True (spKeys   paths)
   91   -- The vault lives in a subdirectory of config/ ('vaultFilePath'); create it
   92   -- so the atomic vault write has an existing parent for its .tmp file.
   93   createDirectoryIfMissing True (takeDirectory (vaultFilePath paths))
   94   -- The persistent ssh-agent registry lives in a subdirectory of state/repos;
   95   -- create it so the registry file has an existing parent (#88).
   96   createDirectoryIfMissing True (sshAgentsDir paths)
   97   setFileMode (spKeys paths) 0o700

  108 -- | Absolute path to the security config file: @\<home\>\/security.toml@.
  109 -- This is a SIBLING of @config\/@ (NOT inside it) so the git-versioned config
  110 -- repo cannot resurrect a lower-security version (design V8). The file holds
  111 -- the boot-only, agent-immutable 'SecurityConfig' (untrusted_execution +
  112 -- vault settings). Mode 0600, owned by the harness user.
  113 securityFilePath :: SealPaths -> FilePath
  114 securityFilePath paths = spHome paths </> "security.toml"
  115 
  116 -- | Absolute path to the source-control repo registry TOML file:
  117 -- @\<config\>\/repos.toml@.
  118 reposFilePath :: SealPaths -> FilePath
  119 reposFilePath paths = spConfig paths </> "repos.toml"

  121 -- | Private state dir for clone-time temp files (GIT_ASKPASS helper scripts,
  122 -- deploy-key keyfiles, managed known_hosts). Security invariants: created
  123 -- mode @0700@, never under @\/tmp@ (so secrets never touch a world-writable
  124 -- temp dir), never version-controlled (so secret bytes never land in git).
  125 repoCloneStateDir :: SealPaths -> FilePath
  126 repoCloneStateDir paths = spState paths </> "repos"
  127 
  128 -- | The harness-private dir holding the encrypted deploy-key keyfiles (rev 3
  129 -- of the design: the encrypted keyfile stays on the harness disk as
  130 -- ciphertext; the passphrase lives in the vault; @ssh-add@ decrypts the
  131 -- keyfile into agent memory at git-op time using the passphrase piped to its
  132 -- stdin). Path: @\<state\>\/repos\/keys@. Created mode @0700@ by the caller
  133 -- (the @ssh-keygen@ invocation in W5 creates it on first key generation).
  134 -- The encrypted keyfile is the per-repo artifact named by 'repoIdText'.
  135 repoKeysDir :: SealPaths -> FilePath
  136 repoKeysDir paths = repoCloneStateDir paths </> "keys"
  137 
  138 -- | The directory holding the persistent ssh-agent registry file:
  139 -- @\<state\>\/repos\/ssh-agents@. The registry (@registry.json@) records the
  140 -- @SSH_AUTH_SOCK@ \/ @SSH_AGENT_PID@ of each repo's agent so they survive
  141 -- @seal serve@ restarts (#88). Created mode @0700@ by 'ensureSealDirs'
  142 -- (the socket paths are not secrets but shouldn't be world-readable).
  143 sshAgentsDir :: SealPaths -> FilePath
  144 sshAgentsDir paths = repoCloneStateDir paths </> "ssh-agents"

  183 -- | The session's diagnostic log: @\<sessionDir\>\/seal.log@. Records turn
  184 -- lifecycle events (start/end/duration) and failures (exceptions, provider

  192 -- | The persisted tab list: @\<state\>\/tabs.json@. Mirrors 'ui_state.json'
  193 -- (the frontend-owned recall file). Written atomically (0600) by
  194 -- 'Seal.Tabs.Persist.saveTabList' on every tab mutation; loaded at boot by
  195 -- 'loadTabList' so the tab list survives a @seal serve@ restart.
  196 tabListPath :: SealPaths -> FilePath
  197 tabListPath paths = spState paths </> "tabs.json"
  198 
  199 -- | The persisted cursor map: @\<state\>\/cursors.json@. Maps each
  200 -- conversation ('ConversationKey') to the 'TabRef' it's focused on.
  201 -- Written atomically (0600) by 'Seal.Channels.Cursor.Persist.saveCursorMap'
  202 -- on every cursor mutation; loaded at boot by 'loadCursorMap' so a
  203 -- conversation re-resolves to its prior session (carrying the user's
  204 -- @\/model use@ choice) after a @seal serve@ restart. Without this, the
  205 -- in-memory cursor store is lost on restart and the loop mints a fresh
  206 -- default-model session for every existing conversation.
  207 cursorMapPath :: SealPaths -> FilePath
  208 cursorMapPath paths = spState paths </> "cursors.json"

  210 -- | Directory for a sub-agent's transcript, nested under its parent session:
  211 -- @\<state\>\/sessions\/\<parent-id\>\/agents\/\<child-id\>@. Each forked agent
  212 -- instance gets its own two-file transcript here so the parent's
  213 -- @conversation.jsonl@ \/ @entries.jsonl@ stay uncontaminated (the two-file
  214 -- format's @erConvLen@ and envelope-delta fold are per-session; mixing a
  215 -- sub-agent's entries into the parent's files would corrupt reconstruction).
  216 -- The Audited log is shared (cross-session and canonical).
  217 agentSessionDir :: SealPaths -> SessionId -> SessionId -> FilePath
  218 agentSessionDir paths parentSid childSid =
  219   sessionDir paths parentSid </> "agents" </> T.unpack (sessionIdText childSid)
  220 
  221 -- | Root for per-session working directories: @\<cache\>\/workdirs@.
  222 workdirsRoot :: SealPaths -> FilePath
  223 workdirsRoot paths = spCache paths </> "workdirs"
  224 
  225 -- | One session's working directory: @\<cache\>\/workdirs\/\<session-id\>@.
  226 -- Asserts 'isValidSessionId' as defense-in-depth (the constructor is
  227 -- locked down, but this catches any future raw-constructor leak before
  228 -- the path reaches mkdir or rm -rf).
  229 sessionWorkdir :: SealPaths -> SessionId -> FilePath
  230 sessionWorkdir paths sid
  231   | isValidSessionId (sessionIdText sid) =
  232       workdirsRoot paths </> T.unpack (sessionIdText sid)
  233   | otherwise =
  234       error ("sessionWorkdir: invalid session id: " <> T.unpack (sessionIdText sid))

   46 -- | The error type for workdir lifecycle operations.
   47 data WorkdirError
   48   = WdMkdirFailed FilePath Text      -- ^ local mkdir failed (path, reason)
   49   | WdRemoteMkdirFailed Text          -- ^ remote SSH mkdir failed (reason)
   50   | WdInvalidSessionId Text           -- ^ SessionId failed validation
   51   | WdNotUnderWorkdirsRoot FilePath    -- ^ cleanup path escaped workdirsRoot
   52   deriving stock (Eq, Show)
   53 
   54 -- | Create the per-session workdir at @<cache>/workdirs/<sid>@.
   55 -- Idempotent: if the workdir already exists, it is reused (NOT cleared —
   56 -- the operator may want to resume or inspect). Returns the workdir path
   57 -- on success. Fails closed on permission errors (does NOT fall back to
   58 -- the cwd — that would reintroduce the cross-session clobber bug).
   59 ensureSessionWorkdir :: SealPaths -> SessionId -> IO (Either WorkdirError FilePath)
   60 ensureSessionWorkdir paths sid = do
   61   let sidText = sessionIdText sid
   62   if not (isValidSessionId sidText)
   63     then pure (Left (WdInvalidSessionId sidText))
   64     else do
   65       let wdPath = sessionWorkdir paths sid
   66       eResult <- try (createDirectoryIfMissing True wdPath) :: IO (Either IOException ())
   67       pure $ case eResult of
   68         Left ioErr -> Left (WdMkdirFailed wdPath (T.pack (show ioErr)))
   69         Right _   -> Right wdPath
   70 
   71 -- | Remove the per-session workdir. Asserts the path is under
   72 -- 'workdirsRoot' (canonicalize + prefix check — defeats symlink swap).
   73 -- Idempotent: no error if the workdir is already gone. Returns
   74 -- 'Either WorkdirError ()' so cleanup failures are not silently
   75 -- swallowed.
   76 cleanupSessionWorkdir :: SealPaths -> SessionId -> IO (Either WorkdirError ())
   77 cleanupSessionWorkdir paths sid = do
   78   let sidText = sessionIdText sid
   79   if not (isValidSessionId sidText)
   80     then pure (Left (WdInvalidSessionId sidText))
   81     else do
   82       let wdPath = sessionWorkdir paths sid
   83           wdRoot = workdirsRoot paths
   84       exists <- doesDirectoryExist wdPath
   85       if not exists
   86         then pure (Right ())  -- already gone — idempotent
   87         else do
   88           -- Defense-in-depth: canonicalize and verify the path is under
   89           -- workdirsRoot before rm -rf (defeats symlink swap).
   90           canonWd <- canonicalizePath wdPath
   91           canonRoot <- canonicalizePath wdRoot
   92           if not (splitDirectories canonRoot `isPrefixOf` splitDirectories canonWd)
   93             then pure (Left (WdNotUnderWorkdirsRoot canonWd))
   94             else do
   95               eResult <- try (removeDirectoryRecursive canonWd) :: IO (Either IOException ())
   96               pure $ case eResult of
   97                 Left ioErr -> Left (WdMkdirFailed canonWd (T.pack (show ioErr)))
   98                 Right _   -> Right ()

  104 -- | Compute the remote per-session workdir path:
  105 -- @<scWorkspace>/workdirs/<sid>@. Pure — no IO, no SSH.
  106 remoteSessionWorkdirPath :: SshConfig -> SessionId -> Text
  107 remoteSessionWorkdirPath sshCfg sid =
  108   getRemotePath (scWorkspace sshCfg) <> "/workdirs/" <> sessionIdText sid
  109 
  110 -- | Create the per-session workdir on the REMOTE machine via SSH
  111 -- @mkdir -p@. Validates the 'SessionId' before the SSH call. Idempotent
  112 -- (mkdir -p is a no-op if the dir exists). Returns the remote workdir
  113 -- path (as 'Text') on success. Fails closed on SSH errors (does NOT
  114 -- fall back to the shared scWorkspace — that would reintroduce the
  115 -- cross-session clobber bug).
  116 ensureRemoteSessionWorkdir
  117   :: SshConfig -> RemoteRunner -> SessionId
  118   -> IO (Either WorkdirError Text)
  119 ensureRemoteSessionWorkdir sshCfg runner sid = do
  120   let sidText = sessionIdText sid
  121   if not (isValidSessionId sidText)
  122     then pure (Left (WdInvalidSessionId sidText))
  123     else do
  124       let remoteWd = remoteSessionWorkdirPath sshCfg sid
  125           cmdText = "mkdir -p '" <> T.unpack remoteWd <> "'"
  126       case mkShellCommand (T.pack cmdText) of
  127         Left _err -> pure (Left (WdRemoteMkdirFailed "invalid mkdir command"))
  128         Right cmd -> do
  129           res <- runRemoteShell runner sshCfg cmd
  130           pure $ case res of
  131             Left e   -> Left (WdRemoteMkdirFailed (T.pack (show e)))
  132             Right _  -> Right remoteWd

  138 -- | The fail-closed 'WorkspaceRoot' returned when workdir creation fails.
  139 -- Matches the convention used at the wiring sites (e.g. @Cli.hs@).
  140 failClosedRoot :: WorkspaceRoot
  141 failClosedRoot = WorkspaceRoot "/nonexistent-workdir-fail-closed"

  143 -- | The per-session execution bundle: the 'UIOEnv' (carrying 'UntrustedIO'
  144 -- + 'CloneDeps'), the 'WorkdirFs' for discovery, and the shared
  145 -- 'WorkspaceRoot' used by UIO + WorkdirFs + the ISA registry. Constructed
  146 -- by 'mkSessionExec' from the 'SecurityConfig'; the single 'RemoteRunner'
  147 -- is shared between 'seUIOEnv' (UIO's 'UntrustedIO') and 'seWorkdirFs'
  148 -- (via 'runUIOWithEnv') — one SSH connection, not two. On ANY workdir-
  149 -- creation failure, ALL three fields are stubs (fail-closed) — never a
  150 -- mix of real + stub.
  151 data SessionExec = SessionExec
  152   { seUIOEnv        :: UIOEnv
  153   , seWorkdirFs     :: WorkdirFs
  154   , seWorkspaceRoot :: WorkspaceRoot
  155   }
  156 
  157 -- | Construct the per-session 'SessionExec' from the 'SecurityConfig'.
  158 -- Mirrors 'mkSessionUntrustedIO' with the same three cases (local,
  159 -- remote + configured, remote + absent/incomplete) but additionally
  160 -- builds the 'WorkdirFs' and the shared 'WorkspaceRoot'. The
  161 -- 'RemoteRunner' is passed explicitly so tests can inject
  162 -- 'mkFakeRemoteRunnerRecording' (no real SSH); production wiring passes
  163 -- 'mkRealRemoteRunner'. The single runner is shared between 'seUIOEnv'
  164 -- and 'seWorkdirFs'. On ANY workdir-creation failure, returns a
  165 -- 'SessionExec' with both handles as stubs and 'seWorkspaceRoot' as
  166 -- 'failClosedRoot' — never mixed.
  167 mkSessionExec
  168   :: SealPaths -> SecurityConfig -> SessionId
  169   -> CloneDeps -> RemoteRunner -> IO SessionExec
  170 mkSessionExec paths secCfg sid cloneDeps runner =
  171   case untrustedExecConfigFromSecurity secCfg of
  172     Nothing -> do
  173       eWd <- ensureSessionWorkdir paths sid
  174       case eWd of
  175         Right wd ->
  176           let wsRoot  = WorkspaceRoot wd
  177               uio     = mkLocalUntrustedIO wsRoot
  178               uioEnv  = mkTestUIOEnv uio cloneDeps
  179               wfs     = mkLocalWorkdirFs wsRoot maxScanBytes
  180           in pure SessionExec
  181                { seUIOEnv        = uioEnv
  182                , seWorkdirFs     = wfs
  183                , seWorkspaceRoot = wsRoot
  184                }
  185         Left _err -> pure (failClosedSessionExec cloneDeps)
  186 
  187     Just uec ->
  188       case uecRemote uec of
  189         Nothing -> pure (failClosedSessionExec cloneDeps)
  190 
  191         Just sshCfg -> do
  192           eRemoteWd <- ensureRemoteSessionWorkdir sshCfg runner sid
  193           case eRemoteWd of
  194             Left _err -> pure (failClosedSessionExec cloneDeps)
  195             Right remoteWdText ->
  196               case mkRemotePath remoteWdText of
  197                 Left _err -> pure (failClosedSessionExec cloneDeps)
  198                 Right remotePath ->
  199                   let sshCfg' = sshCfg { scWorkspace = remotePath }
  200                       wsRoot  = WorkspaceRoot (T.unpack (getRemotePath remotePath))
  201                       uio     = mkRemoteUntrustedIO sshCfg' runner
  202                       uioEnv  = mkTestUIOEnv uio cloneDeps
  203                       wfs     = mkRemoteWorkdirFs uioEnv sshCfg' wsRoot maxScanBytes
  204                   in pure SessionExec
  205                        { seUIOEnv        = uioEnv
  206                        , seWorkdirFs     = wfs
  207                        , seWorkspaceRoot = wsRoot
  208                        }

  210 -- | The fail-closed 'SessionExec': both handles are stubs, the root is
  211 -- 'failClosedRoot'. Used on ANY workdir-creation failure.
  212 failClosedSessionExec :: CloneDeps -> SessionExec
  213 failClosedSessionExec cloneDeps = SessionExec
  214   { seUIOEnv        = mkTestUIOEnv mkRemoteUntrustedIOStub cloneDeps
  215   , seWorkdirFs     = mkWorkdirFsStub
  216   , seWorkspaceRoot = failClosedRoot
  217   }

  219 -- | Did this exec fail closed? (The workdir bootstrap failed and every
  220 -- handle is a stub.) The exec cache uses this to avoid memoizing failures.
  221 isFailClosedSessionExec :: SessionExec -> Bool
  222 isFailClosedSessionExec e = seWorkspaceRoot e == failClosedRoot

  224 -- | Construct the per-session 'UntrustedIO' from the 'SecurityConfig':
  225 --
  226 --   * @mode=local@ (or absent): create the local workdir via
  227 --     'ensureSessionWorkdir' and construct 'mkLocalUntrustedIO' with it
  228 --     as the 'WorkspaceRoot'.
  229 --
  230 --   * @mode=remote@ + remote configured: create the remote workdir via
  231 --     'ensureRemoteSessionWorkdir' (SSH @mkdir -p@), clone the
  232 --     'SshConfig' with @scWorkspace@ = the per-session remote workdir,
  233 --     and construct 'mkRemoteUntrustedIO' with the cloned config.
  234 --
  235 --   * @mode=remote@ + remote absent/incomplete: return the fail-closed
  236 --     stub ('mkRemoteUntrustedIOStub').
  237 --
  238 -- This is the single entry point the wiring sites call — it handles
  239 -- both local and remote, creates the workdir, and returns the handle.
  240 -- On workdir creation failure, returns the fail-closed stub (the session
  241 -- should surface the error and NOT proceed — the wiring site checks the
  242 -- 'WorkdirError' via 'ensureSessionWorkdir' separately if it needs to
  243 -- surface the error to the user).
  244 --
  245 -- Back-compat thin wrapper over 'mkSessionExec' (threads
  246 -- 'mkRealRemoteRunner', preserving EXACT current semantics).
  247 mkSessionUntrustedIO
  248   :: SealPaths -> SecurityConfig -> SessionId -> IO UntrustedIO
  249 mkSessionUntrustedIO paths secCfg sid =
  250   uieUntrustedIO . seUIOEnv <$> mkSessionExec paths secCfg sid stubCloneDeps mkRealRemoteRunner

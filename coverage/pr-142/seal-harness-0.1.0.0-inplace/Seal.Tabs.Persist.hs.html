    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Tab list persistence — the 'TabList' survives a @seal serve@ restart.
    3 -- Written atomically (0600) to @\<state\>\/tabs.json@ on every tab mutation;
    4 -- loaded at boot and re-resolved to the tab refs (ids validated; unparseable
    5 -- refs skipped — defense-in-depth against a tampered file). Writes are
    6 -- serialized via a module-level MVar so concurrent mutations cannot
    7 -- interleave file writes.
    8 module Seal.Tabs.Persist
    9   ( saveTabList
   10   , loadTabList
   11   ) where
   12 
   13 import Control.Concurrent.MVar (MVar, modifyMVar_, newMVar)
   14 import Data.Aeson qualified as A
   15 import Data.ByteString.Lazy qualified as BL
   16 import System.Directory (doesFileExist)
   17 import System.IO.Unsafe (unsafePerformIO)
   18 
   19 import Katip (Severity (..))
   20 import Seal.Core.Types (mkSessionId, sessionIdText)
   21 import Seal.Harness.Id (parseHarnessId, harnessIdToText)
   22 import Seal.Logging.Global (globalLogIO)
   23 import Seal.Tabs.Types (Tab (..), TabList (..), TabRef (..))
   24 import Seal.Util.AtomicJson (saveJsonAtomic)
   25 
   26 -- | Module-level write lock — serializes concurrent 'saveTabList' calls so
   27 -- two tab mutations cannot interleave file writes (a bare forkIO TVar
   28 -- listener would race; the explicit MVar guarantees atomicity). One lock
   29 -- per process, shared across all paths (a single @tabs.json@ per state dir).
   30 --
   31 -- NOTE: 'Seal.Util.AtomicJson.saveJsonAtomic' has its own module-level
   32 -- lock, so the MVar here is redundant for the write itself. It is retained
   33 -- because 'loadTabList' / 'filterValidTabs' are pure and do not take the
   34 -- lock; keeping the local lock documents the serialization invariant. A
   35 -- future cleanup can drop it once all callers route through
   36 -- 'saveJsonAtomic' (which they now do).
   37 writeLock :: MVar ()
   38 writeLock = unsafePerformIO (newMVar ())
   39 {-# NOINLINE writeLock #-}
   40 
   41 -- | Save a 'TabList' to @path@ atomically via 'saveJsonAtomic' (0600,
   42 -- MVar-serialized). A thrown IO error propagates to the caller
   43 -- ('Seal.Tabs.persistIf') which catches, logs a warning, and swallows —
   44 -- the in-memory handle stays authoritative within the session. Takes the
   45 -- 'TabList' directly (not a 'TabsHandle') so this module does NOT import
   46 -- 'Seal.Tabs' (avoids a cycle).
   47 saveTabList :: FilePath -> TabList -> IO ()
   48 saveTabList path tl = modifyMVar_ writeLock $ \_ ->
   49   saveJsonAtomic path (A.encode tl)
   50 
   51 -- | Load the tab list from @path@. Missing file -> 'Nothing' (fresh empty
   52 -- list). Corrupt JSON -> 'Nothing' + a stderr warning (ids + error type
   53 -- only — no session content). Every 'TabRef' id is re-validated via
   54 -- 'mkSessionId' / 'parseHarnessId'; tabs carrying an unparseable id are
   55 -- skipped (defense-in-depth against a tampered @tabs.json@).
   56 loadTabList :: FilePath -> IO (Maybe TabList)
   57 loadTabList path = do
   58   exists <- doesFileExist path
   59   if not exists
   60     then pure Nothing
   61     else do
   62       bs <- BL.readFile path
   63       case A.decode bs :: Maybe TabList of
   64         Nothing -> do
   65           globalLogIO WarningS "could not parse tabs.json; using empty tab list"
   66           pure Nothing
   67         Just tl -> pure (Just (filterValidTabs tl))
   68 
   69 -- | Drop tabs whose 'TabRef' id fails to re-parse. The on-disk ids were
   70 -- validated when written, so a parse failure here means the file was
   71 -- tampered or written by a future/incompatible version — skip rather than
   72 -- error so a corrupt file self-heals on the next save.
   73 filterValidTabs :: TabList -> TabList
   74 filterValidTabs tl = tl { tlTabs = filter (validRef . tRef) (tlTabs tl) }
   75   where
   76     validRef (BoundSession sid) = case mkSessionId (sessionIdText sid) of Right _ -> True; Left _ -> False
   77     validRef (BoundHarness hid) = case parseHarnessId (harnessIdToText hid) of Right _ -> True; Left _ -> False

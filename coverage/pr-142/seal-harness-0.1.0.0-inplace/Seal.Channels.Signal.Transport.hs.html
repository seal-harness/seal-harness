   39 -- | The testability seam over signal-cli.
   40 data SignalTransport = SignalTransport
   41   { stReceive :: IO (Either Text Value)
   42     -- ^ Next inbound JSON value (one line from signal-cli's stdout).
   43   , stSend    :: Text -> Text -> IO ()
   44     -- ^ Send a message: recipient, body. Writes a JSON-RPC @send@ frame.
   45   , stClose   :: IO ()
   46   }
   47 
   48 -- | A mock transport backed by a 'TQueue' of inbound 'Value's and an 'IORef'
   49 -- of captured sends. 'stReceive' pops the next inbound (or returns
   50 -- @Left "inbox empty"@); 'stSend' appends @(recipient, body)@ to the
   51 -- capture; 'stClose' is a no-op (idempotent).
   52 mkMockSignalTransport :: [Value] -> IO (SignalTransport, IO [(Text, Text)])
   53 mkMockSignalTransport scripted = do
   54   q <- newTQueueIO
   55   mapM_ (atomically . writeTQueue q) scripted
   56   capRef <- newIORef []
   57   let transport = SignalTransport
   58         { stReceive = do
   59             m <- atomically (tryReadTQueue q)
   60             case m of
   61               Just v  -> pure (Right v)
   62               Nothing -> pure (Left "signal inbox empty")
   63         , stSend = \r b -> modifyIORef' capRef ((r, b) :)
   64         , stClose = pure ()
   65         }
   66       getCaptured = reverse <$> readIORef capRef
   67   pure (transport, getCaptured)

   69 -- | Spawn @signal-cli --output=json --trust-new-identities=always
   70 -- -u <account> jsonRpc@ as a child process via "System.Process", line-
   71 -- buffered JSON-RPC over stdin/stdout. 'stReceive' reads one line from
   72 -- stdout and decodes it; 'stSend' writes a JSON-RPC @send@ frame to stdin;
   73 -- 'stClose' terminates the child (with a 5s graceful wait). A preflight
   74 -- @signal-cli --version@ probe fails fast if the binary is absent.
   75 --
   76 -- The '<account>' is the smart-constructed validated 'SignalAccount' from
   77 -- "Seal.Signal.Config" — option injection fails to compile. This module
   78 -- takes it as 'Text' here to avoid a cycle (Signal.Config imports nothing
   79 -- from this module; the constructor is enforced at the call site).
   80 mkRealSignalTransport :: Text -> IO (Either Text SignalTransport)
   81 mkRealSignalTransport account = do
   82   versionOk <- probeSignalCli
   83   if not versionOk
   84     then pure (Left "signal-cli not installed or on PATH")
   85     else do
   86       eStarted <- try @IOException $
   87         createProcess
   88           ( (proc "signal-cli"
   89               [ "--output=json"
   90               , "--trust-new-identities=always"
   91               , "-u", T.unpack account
   92               , "jsonRpc"
   93               ])
   94               { std_in = CreatePipe, std_out = CreatePipe, std_err = Inherit }
   95           )
   96       case eStarted of
   97         Left e -> pure (Left ("signal-cli launch failed: " <> T.pack (show e)))
   98         Right (mIn, mOut, _err, ph) -> do
   99           (hIn, hOut) <- case (mIn, mOut) of
  100             (Just a, Just b) -> pure (a, b)
  101             _ -> error "mkRealSignalTransport: pipe creation failed (unreachable)"
  102           hSetBuffering hIn (BlockBuffering Nothing)
  103           hSetBuffering hOut LineBuffering
  104           pure (Right SignalTransport
  105             { stReceive = do
  106                 line <- hGetLine hOut
  107                 pure $ case A.decode (BL.fromStrict (TE.encodeUtf8 (T.pack line))) of
  108                   Just v  -> Right v
  109                   Nothing -> Left ("signal-cli: malformed JSON line: " <> T.pack line)
  110             , stSend = \recipient body -> do
  111                 let frame = A.object
  112                       [ "jsonrpc" A..= ("2.0" :: Text)
  113                       , "method"  A..= ("send" :: Text)
  114                       , "params"  A..= A.object
  115                           [ "recipient" A..= [recipient]
  116                           , "message"   A..= body
  117                           ]
  118                       ]
  119                 BL.hPutStr hIn (A.encode frame)
  120                 TIO.hPutStrLn hIn ""
  121                 hFlush hIn
  122             , stClose = do
  123                 _ <- try @IOException (hClose hIn)
  124                 terminateProcess ph
  125                 _ <- timeout 5000000 (waitForProcess ph)
  126                 pure ()
  127             })

  129 -- | Preflight @signal-cli --version@. Returns 'True' if the binary is on
  130 -- PATH and exits successfully. Mirrors 'Seal.Security.Vault.Age's
  131 -- @age --version@ preflight.
  132 probeSignalCli :: IO Bool
  133 probeSignalCli = do
  134   r <- try @IOException $
  135         withCreateProcess
  136           ( (proc "signal-cli" ["--version"])
  137               { std_out = NoStream, std_err = NoStream }
  138           ) $ \_ _ _ ph -> waitForProcess ph
  139   pure $ case r of
  140     Right ExitSuccess -> True
  141     _                 -> False

  144 -- chunkMessage
  145 -- ---------------------------------------------------------------------------

  147 -- | Split a message into chunks of at most 'limit' characters, preferring
  148 -- paragraph boundaries (@\\n\\n@), then line boundaries (@\\n@), hard-cut
  149 -- as a last resort. **Chunks carry their trailing separator** (except the
  150 -- last chunk, which has none), so 'T.concat' of the chunks is identity:
  151 --
  152 -- > T.concat (chunkMessage limit t) == t
  153 --
  154 -- This makes the chunks the literal bytes to send, in order — the receiver
  155 -- concatenates and sees the original text. Pure.
  156 chunkMessage :: Int -> Text -> [Text]
  157 chunkMessage limit t
  158   | limit < 1 = error "chunkMessage: limit must be >= 1"
  159   | T.null t  = []
  160   | otherwise = go t
  161   where
  162     go s
  163       | T.null s       = []
  164       | T.length s <= limit = [s]
  165       | otherwise =
  166           let (chunk, rest) = nextChunk limit s
  167           in chunk : if T.null rest then [] else go rest
  168 
  169 -- | Extract the next chunk (with its trailing separator) and the remaining
  170 -- text. Prefers a paragraph boundary (@\\n\\n@) within the limit, then a
  171 -- line boundary (@\\n@), then a hard cut at the limit.
  172 nextChunk :: Int -> Text -> (Text, Text)
  173 nextChunk limit s =
  174   case findParagraphBreak limit s of
  175     Just n -> T.splitAt n s
  176     Nothing -> case findLineBreak limit s of
  177       Just n -> T.splitAt n s
  178       Nothing -> T.splitAt limit s
  179 
  180 -- | The 1-based cut position (inclusive of the @\\n\\n@ separator) of the
  181 -- last @\\n\\n@ within the first 'limit' characters, or 'Nothing' if there
  182 -- is none. The chunk will then be @take n s@ (which ends in @\\n\\n@).
  183 findParagraphBreak :: Int -> Text -> Maybe Int
  184 findParagraphBreak limit s =
  185   -- Search the first `limit` characters for the rightmost "\n\n".
  186   let window = T.take limit s
  187   in case T.breakOnEnd "\n\n" window of
  188        (pre, _post) | not (T.null pre) -> Just (T.length pre)
  189        _ -> Nothing
  190 
  191 -- | The 1-based cut position (inclusive of the @\\n@ separator) of the last
  192 -- @\\n@ within the first 'limit' characters, or 'Nothing' if there is none.
  193 findLineBreak :: Int -> Text -> Maybe Int
  194 findLineBreak limit s =
  195   let window = T.take limit s
  196   in case T.breakOnEnd "\n" window of
  197        (pre, _post) | not (T.null pre) -> Just (T.length pre)
  198        _ -> Nothing

  201 -- SignalEnvelope — parsed inbound envelope + server-derived conversation id
  202 -- ---------------------------------------------------------------------------

  204 -- | A parsed inbound signal-cli envelope: the peer-derived conversation id,
  205 -- the sender's user id, and the message body. The conversation id is
  206 -- **server-derived from authenticated transport metadata** (the peer's
  207 -- phone number + UUID), never read from the message body — enforced
  208 -- structurally by 'conversationIdForSignal' taking only the peer fields.
  209 data SignalEnvelope = SignalEnvelope
  210   { seConversationId :: ConversationId
  211   , seSender         :: UserId
  212   , seBody           :: Text
  213   } deriving stock (Eq, Show)
  214 
  215 -- | Derive the server-side 'ConversationId' from the peer's authenticated
  216 -- transport metadata. signal-cli envelopes carry both a phone number
  217 -- (@source@) and a UUID (@sourceUuid@); the conversation id is
  218 -- @sig:<source>:<sourceUuid>@ (or @sig:<source>@ when the UUID is absent).
  219 -- Never reads the message body. A missing or empty @source@ is rejected
  220 -- (a peer phone number is required).
  221 conversationIdForSignal :: Maybe Text -> Maybe Text -> Either Text ConversationId
  222 conversationIdForSignal mSource mUuid =
  223   case mSource of
  224     Nothing        -> Left "signal envelope missing source (peer phone number)"
  225     Just src
  226       | T.null src -> Left "signal envelope source is empty"
  227       | otherwise  -> mkConversationId full
  228       where
  229         full = case mUuid of
  230           Nothing   -> "sig:" <> src
  231           Just uuid -> "sig:" <> src <> ":" <> uuid
  232 
  233 -- | Parse a raw signal-cli JSON value into a 'SignalEnvelope'. Handles both
  234 -- raw envelopes (@{"envelope": {...}}@) and JSON-RPC-wrapped messages
  235 -- (@{"jsonrpc":"2.0","method":"receive","params":{"envelope":{...}}}@).
  236 -- Returns 'Left' on a malformed value, a missing peer field, or a
  237 -- conversation-id smart-constructor failure. The body is taken from the
  238 -- envelope's @dataMessage.message@; a @conversationId@ key in the body is
  239 -- IGNORED (the conversation id is always server-derived via
  240 -- 'conversationIdForSignal').
  241 parseSignalEnvelope :: Value -> Either Text SignalEnvelope
  242 parseSignalEnvelope v = do
  243   env <- unwrapEnvelope v
  244   source <- fieldText "source" env
  245   let mUuid = fieldTextMaybe "sourceUuid" env
  246   cid <- conversationIdForSignal (Just source) mUuid
  247   uid <- case mkUserId source of
  248     Right u  -> Right u
  249     Left err -> Left ("signal sender not a valid UserId: " <> err)
  250   let body = extractBody env
  251   Right SignalEnvelope
  252     { seConversationId = cid
  253     , seSender         = uid
  254     , seBody           = body
  255     }
  256 
  257 -- | Unwrap a raw envelope (@{"envelope": {...}}@) or a JSON-RPC-wrapped
  258 -- message (@{"params": {"envelope": {...}}}@) to the inner envelope object.
  259 -- Returns 'Left' if neither shape is present or the value is not an object.
  260 unwrapEnvelope :: Value -> Either Text Value
  261 unwrapEnvelope v =
  262   case v of
  263     A.Object o ->
  264       case KeyMap.lookup (Key.fromString "envelope") o of
  265         Just env -> Right env
  266         Nothing -> case KeyMap.lookup (Key.fromString "params") o of
  267           Just (A.Object p) -> case KeyMap.lookup (Key.fromString "envelope") p of
  268             Just env -> Right env
  269             Nothing  -> Left "signal envelope: no envelope in params"
  270           _ -> Left "signal envelope: no envelope and no params"
  271     _ -> Left "signal envelope: not an object"
  272 
  273 -- | Extract a required text field from an object, failing on absence or
  274 -- non-text value.
  275 fieldText :: Text -> Value -> Either Text Text
  276 fieldText key v =
  277   case v of
  278     A.Object o -> case KeyMap.lookup (Key.fromString (T.unpack key)) o of
  279       Just (A.String t) -> Right t
  280       _ -> Left ("signal envelope missing or non-text field: " <> key)
  281     _ -> Left ("signal envelope field " <> key <> ": not an object")
  282 
  283 -- | Extract an optional text field from an object.
  284 fieldTextMaybe :: Text -> Value -> Maybe Text
  285 fieldTextMaybe key v = do
  286   inner <- fieldValueMaybe key v
  287   case inner of
  288     A.String t -> Just t
  289     _ -> Nothing
  290 
  291 -- | Extract an optional sub-value from an object.
  292 fieldValueMaybe :: Text -> Value -> Maybe Value
  293 fieldValueMaybe key v =
  294   case v of
  295     A.Object o -> KeyMap.lookup (Key.fromString (T.unpack key)) o
  296     _ -> Nothing

  298 -- | Extract the message body from @dataMessage.message@. Empty when absent
  299 -- (a non-data envelope, e.g. a receipt — the caller drops these).
  300 extractBody :: Value -> Text
  301 extractBody v = fromMaybe "" $ do
  302   dm <- fieldValueMaybe "dataMessage" v
  303   fieldTextMaybe "message" dm

   50 -- | Which on-disk source produced the entries. Surfaced in 'TranscriptTimings'
   51 -- so the @Server-Timing@ header can name the path the request took.
   52 data TranscriptSource
   53   = TSMissing      -- ^ no transcript file exists
   54   | TSLegacy       -- ^ legacy @transcript.jsonl@
   55   | TSConvEntries  -- @conversation.jsonl@ + @entries.jsonl@ (reconstructed)
   56   | TSConvOnly     -- @conversation.jsonl@ only (no entries sidecar)
   57   deriving (Eq, Show)

   59 -- | Per-phase timings for a single 'readTranscriptEntries' call, in
   60 -- milliseconds. Phase durations are measured with 'getCurrentTime' (wall
   61 -- clock); for the small file sizes typical of a transcript, GHC's wall
   62 -- clock has enough resolution. Phases are exclusive of each other EXCEPT
   63 -- @ttTotal@, which spans the entire call. A phase that didn't run (e.g.
   64 -- @ttReconstruct@ when the legacy path was taken) is 0.
   65 data TranscriptTimings = TranscriptTimings
   66   { ttSource       :: TranscriptSource
   67   , ttEntryCount   :: Int           -- ^ number of frontend entries returned
   68   , ttFileReadMs   :: Integer       -- ^ time spent in 'TIO.readFile' (conv + entries files combined)
   69   , ttParseMs      :: Integer       -- ^ 'A.decode' of every line (both files combined)
   70   , ttReconstructMs :: Integer      -- @'reconstruct' (two-file path only)
   71   , ttRewriteMs    :: Integer        -- 'teLineToFrontend' / 'reconEntryToFrontend' / 'convLineToFrontend'
   72   , ttEncodeMs     :: Integer        -- ^ 'A.encode' of the frontend [Value] (set by 'handleTranscript', 0 for the bare reader)
   73   , ttTotalMs      :: Integer       -- ^ wall-clock total of the whole call
   74   }

   76 -- | Format a 'TranscriptTimings' as a @Server-Timing@ header value per
   77 -- <https://www.w3.org/TR/server-timing/>. Each phase is a separate
   78 -- comma-separated token with a short name, a numeric @dur@ (milliseconds,
   79 -- 3-decimal precision in the spec but we emit integers), and a @desc@
   80 -- naming the phase. The frontend parses these via @performance.getEntriesByName@
   81 -- or by reading the @Server-Timing@ response header directly.
   82 --
   83 -- Example output:
   84 --   @tt;dur=42;desc="total", fr;dur=3;desc="file-read", pr;dur=18;desc="parse", rc;dur=12;desc="reconstruct", rw;dur=4;desc="rewrite", en;dur=8;desc="encode", src;desc="conv+entries", n;desc="127"@
   85 renderServerTiming :: TranscriptTimings -> BC.ByteString
   86 renderServerTiming t =
   87   let phase name ms desc =
   88         BC.pack name <> ";dur=" <> BC.pack (show ms) <> ";desc=\"" <> BC.pack desc <> "\""
   89       tag name desc = BC.pack name <> ";desc=\"" <> BC.pack desc <> "\""
   90       srcDesc = case ttSource t of
   91         TSMissing      -> "missing"
   92         TSLegacy       -> "legacy"
   93         TSConvEntries  -> "conv+entries"
   94         TSConvOnly     -> "conv-only"
   95       parts =
   96         [ phase "tt" (ttTotalMs t)       "total"
   97         , phase "fr" (ttFileReadMs t)    "file-read"
   98         , phase "pr" (ttParseMs t)        "parse"
   99         , phase "rc" (ttReconstructMs t) "reconstruct"
  100         , phase "rw" (ttRewriteMs t)      "rewrite"
  101         , phase "en" (ttEncodeMs t)       "encode"
  102         , tag  "src" srcDesc
  103         , tag  "n"   (show (ttEntryCount t))
  104         ]
  105   in BC.intercalate ", " parts

  107 -- | Set the encode-phase duration + bump the total to span read+encode.
  108 -- 'handleTranscript' uses this after timing 'A.encode' so the @en@ token
  109 -- reflects the JSON-serialization cost (which dominates for transcripts
  110 -- carrying large web_fetch / shell-output tool_result blocks) and @tt@
  111 -- reflects the full read+encode cost.
  112 setEncodeMs :: Integer -> Integer -> TranscriptTimings -> TranscriptTimings
  113 setEncodeMs encodeMs totalMs tt = tt { ttEncodeMs = encodeMs, ttTotalMs = totalMs }

  115 -- | 'zipWith' + 'mapMaybe': apply a partial function across two lists
  116 -- in lockstep, dropping the elements for which the function returns
  117 -- 'Nothing'.
  118 zipWithMaybe :: (a -> b -> Maybe c) -> [a] -> [b] -> [c]
  119 zipWithMaybe _ [] _ = []
  120 zipWithMaybe _ _ [] = []
  121 zipWithMaybe f (a : as) (b : bs) = case f a b of
  122   Just c  -> c : zipWithMaybe f as bs
  123   Nothing -> zipWithMaybe f as bs

  125 -- | Read a session's transcript as the frontend's TranscriptEntry JSON shape.
  126 -- Returns @[]@ for a missing/invalid session. Discards timing information;

  128 readTranscriptEntries

  133 -- | Instrumented variant of 'readTranscriptEntries': returns the frontend
  134 -- values alongside per-phase wall-clock timings. Used by the @/transcript@
  135 -- HTTP handler to emit a @Server-Timing@ header; other callers (broadcast,
  136 -- snippet helpers) can use the plain 'readTranscriptEntries' and ignore the
  137 -- overhead of timing capture.
  138 readTranscriptEntriesTimed
  139   :: SealPaths -> Text -> String -> SessionId -> IO ([Value], TranscriptTimings)
  140 readTranscriptEntriesTimed paths model fallbackTs sid = do
  141   tStart <- getCurrentTime
  142   let legacyPath = sessionTranscriptPath paths sid
  143       convPath   = sessionConversationPath paths sid
  144   legacyExists <- doesFileExist legacyPath
  145   convExists   <- doesFileExist convPath
  146   if legacyExists
  147     then do
  148       tFr0 <- getCurrentTime
  149       raw <- readFileTextStrict legacyPath
  150       tFr1 <- getCurrentTime
  151       tPr0 <- getCurrentTime
  152       let vals = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  153                           (filter (not . T.null) (T.lines raw))
  154       tPr1 <- getCurrentTime
  155       tRw0 <- getCurrentTime
  156       let frontend = map teLineToFrontend vals
  157       tRw1 <- getCurrentTime
  158       tEnd <- getCurrentTime
  159       let tt = TranscriptTimings
  160             { ttSource        = TSLegacy
  161             , ttEntryCount    = length frontend
  162             , ttFileReadMs    = ms tFr0 tFr1
  163             , ttParseMs       = ms tPr0 tPr1
  164             , ttReconstructMs = 0
  165             , ttRewriteMs     = ms tRw0 tRw1
  166             , ttEncodeMs      = 0
  167             , ttTotalMs       = ms tStart tEnd
  168             }
  169       pure (frontend, tt)
  170     else if convExists
  171       then do
  172         tFr0 <- getCurrentTime
  173         raw <- readFileTextStrict convPath
  174         tFr1 <- getCurrentTime
  175         tPr0 <- getCurrentTime
  176         let msgVals = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  177                                (filter (not . T.null) (T.lines raw)) :: [A.Value]
  178             msgs    = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  179                                (filter (not . T.null) (T.lines raw)) :: [Message]
  180         tPr1 <- getCurrentTime
  181         entriesExist <- doesFileExist (sessionEntriesPath paths sid)
  182         if entriesExist
  183           then do
  184             tFr2 <- getCurrentTime
  185             eraw <- readFileTextStrict (sessionEntriesPath paths sid)
  186             tFr3 <- getCurrentTime
  187             tPr2 <- getCurrentTime
  188             let evs = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  189                                (filter (not . T.null) (T.lines eraw)) :: [EntryRecord]
  190             tPr3 <- getCurrentTime
  191             tRc0 <- getCurrentTime
  192             let reconstructed = reconstruct msgs evs
  193                 frontend = zipWithMaybe reconEntryToFrontend [0..] reconstructed
  194             tRc1 <- getCurrentTime
  195             tEnd <- getCurrentTime
  196             let tt = TranscriptTimings
  197                   { ttSource        = TSConvEntries
  198                   , ttEntryCount    = length frontend
  199                   , ttFileReadMs    = ms tFr0 tFr1 + ms tFr2 tFr3
  200                   , ttParseMs       = ms tPr0 tPr1 + ms tPr2 tPr3
  201                   , ttReconstructMs = ms tRc0 tRc1
  202                   , ttRewriteMs     = 0  -- reconstruction includes rewrite
  203                   , ttEncodeMs      = 0
  204                   , ttTotalMs       = ms tStart tEnd
  205                   }
  206             pure (frontend, tt)
  207           else do
  208             tRw0 <- getCurrentTime
  209             let frontend = zipWith (convLineToFrontend model [] fallbackTs) [0..] msgVals
  210             tRw1 <- getCurrentTime
  211             tEnd <- getCurrentTime
  212             let tt = TranscriptTimings
  213                   { ttSource        = TSConvOnly
  214                   , ttEntryCount    = length frontend
  215                   , ttFileReadMs    = ms tFr0 tFr1
  216                   , ttParseMs       = ms tPr0 tPr1
  217                   , ttReconstructMs = 0
  218                   , ttRewriteMs     = ms tRw0 tRw1
  219                   , ttEncodeMs      = 0
  220                   , ttTotalMs       = ms tStart tEnd
  221                   }
  222             pure (frontend, tt)
  223       else do
  224         tEnd <- getCurrentTime
  225         let tt = TranscriptTimings
  226               { ttSource        = TSMissing
  227               , ttEntryCount    = 0
  228               , ttFileReadMs    = 0
  229               , ttParseMs       = 0
  230               , ttReconstructMs = 0
  231               , ttRewriteMs     = 0
  232               , ttEncodeMs      = 0
  233               , ttTotalMs       = ms tStart tEnd
  234               }
  235         pure ([], tt)
  236   where
  237     ms a b = round (realToFrac (b `diffUTCTime` a) * 1000 :: Double)

  239 -- | Extract a short snippet of the first user message in a session's
  240 -- transcript, for use as the default session title when the user has not
  241 -- set an explicit description. Reads the two-file format
  242 -- (@conversation.jsonl@) first, falling back to the legacy
  243 -- @transcript.jsonl@. Returns 'Nothing' when the session has no transcript
  244 -- or no user message with text content. The snippet is truncated to 80
  245 -- characters (on a codepoint boundary) to keep the sessions list compact.
  246 firstUserMessageSnippet :: SealPaths -> SessionId -> IO (Maybe Text)
  247 firstUserMessageSnippet paths sid = do
  248   let legacyPath = sessionTranscriptPath paths sid
  249       convPath   = sessionConversationPath paths sid
  250   legacyExists <- doesFileExist legacyPath
  251   convExists   <- doesFileExist convPath
  252   if convExists
  253     then do
  254       raw <- readFileTextStrict convPath
  255       let msgs = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  256                           (filter (not . T.null) (T.lines raw)) :: [Message]
  257       pure (snippetFromMessages msgs)
  258     else if legacyExists
  259       then do
  260         raw <- readFileTextStrict legacyPath
  261         let entries = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  262                                (filter (not . T.null) (T.lines raw)) :: [TranscriptEntry]
  263         pure (snippetFromMessages (legacyRequestMessages entries))
  264       else pure Nothing

  266 -- | Extract the first user message's text from a list of messages, truncated
  267 -- to 80 characters. Tool-use and tool-result blocks are skipped (only the
  268 -- first 'CbText' block of the first 'User' message is used).
  269 snippetFromMessages :: [Message] -> Maybe Text
  270 snippetFromMessages msgs =
  271   case filter (\m -> msgRole m == User) msgs of
  272     (m : _) -> case [t | CbText t <- msgContent m] of
  273       (t : _) -> Just (truncateSnippet 80 t)
  274       []      -> Nothing
  275     [] -> Nothing

  277 -- | Truncate a snippet to at most @n@ characters, appending an ellipsis when
  278 -- truncation occurs. 'T.take' is codepoint-safe for UTF-8 'Text'.
  279 truncateSnippet :: Int -> Text -> Text
  280 truncateSnippet n t
  281   | T.length t <= n = t
  282   | otherwise       = T.take n t <> "…"

  284 -- | Extract the request messages from legacy @transcript.jsonl@ entries.
  285 -- Only 'Request'-direction entries carry a payload with a @messages@ array;
  286 -- we collect messages from every request entry (the first user message is
  287 -- always in the first request).
  288 legacyRequestMessages :: [TranscriptEntry] -> [Message]
  289 legacyRequestMessages = concatMap go
  290   where
  291     go te
  292       | teDirection te /= Request = []
  293       | otherwise = case tePayload te of
  294           A.Object o -> case KeyMap.lookup (Key.fromText "messages") o of
  295             Just (A.Array arr) -> mapMaybe decodeMsg (V.toList arr)
  296             _ -> []
  297           _ -> []
  298     decodeMsg v = case A.fromJSON v :: A.Result Message of
  299       A.Success m -> Just m
  300       A.Error _   -> Nothing

  302 -- | Map a legacy on-disk transcript line (Haskell TranscriptEntry JSON
  303 -- with @te*@-prefixed fields) to the frontend's TranscriptEntry shape.
  304 teLineToFrontend :: A.Value -> A.Value
  305 teLineToFrontend rawLine =
  306   let o = case rawLine of
  307         A.Object m -> m
  308         _          -> KeyMap.empty
  309       k = Key.fromText
  310       lookupT key = case KeyMap.lookup (k key) o of
  311         Just (A.String t) -> Just t
  312         _                 -> Nothing
  313       -- The on-disk `payload` field is a JSON string; decode it to a Value
  314       -- so we can send it as a JSON object (not a string) in the wire
  315       -- response. Falls back to the raw string when it's not valid JSON
  316       -- (defensive against a malformed on-disk line).
  317       payloadVal = case KeyMap.lookup (k "payload") o of
  318         Just (A.String s) -> case A.decode (BL.fromStrict (TE.encodeUtf8 s)) of
  319           Just v  -> v
  320           Nothing -> A.String s
  321         other -> fromMaybe A.Null other
  322       -- The legacy transcript.jsonl entry's `meta` field carries the
  323       -- channel label (stamped by runTurn's requestMeta). Extract it so
  324       -- the frontend can attribute user messages to their channel on the
  325       -- legacy path too. Null when absent (response entries, old entries).
  326       mChannel = case KeyMap.lookup (k "meta") o of
  327         Just (A.Object mo) -> case KeyMap.lookup (k "channel") mo of
  328           Just (A.String ch) -> Just ch
  329           _                  -> Nothing
  330         _ -> Nothing
  331   in object
  332      [ "id"        .= lookupT "id"
  333      , "timestamp" .= lookupT "timestamp"
  334      , "direction" .= lookupT "direction"
  335      , "payload"   .= payloadVal
  336      , "harness"   .= lookupT "correlation"
  337      , "model"     .= lookupT "model"
  338      , "channel"   .= mChannel
  339      , "raw"       .= TE.decodeUtf8 (BL.toStrict (A.encode rawLine))
  340      ]

  342 -- | Synthesize a frontend TranscriptEntry from a conversation.jsonl line
  343 -- (@role@/@content@). User → request; Assistant → response. The
  344 -- line index (0-based) is used as the entry id — conversation.jsonl
  345 -- carries no per-entry id, and the line index is stable across reads of
  346 -- the same file, so the frontend's dedup-by-id works.
  347 convLineToFrontend :: Text -> [String] -> String -> Int -> A.Value -> A.Value
  348 convLineToFrontend model entryTimestamps fallbackTs idx rawLine =
  349   let o = case rawLine of
  350         A.Object m -> m
  351         _          -> KeyMap.empty
  352       k = Key.fromText
  353       role = case KeyMap.lookup (k "role") o of
  354         Just (A.String t) -> t
  355         _                  -> "user"
  356       rawContent = case KeyMap.lookup (k "content") o of
  357         Just (A.Array arr) -> arr
  358         _                  -> mempty
  359       contentBlocks = map cbToFrontend (V.toList rawContent)
  360       content = A.Array (V.fromList contentBlocks)
  361       direction :: Text
  362       direction = if T.toCaseFold role == "user" then "request" else "response"
  363       payloadJson = if direction == "request"
  364         then A.object ["messages" A..= [A.object ["role" A..= ("user" :: Text), "content" A..= content]]]
  365         else A.object ["content" A..= content]
  366       entryId = T.pack (show idx)
  367       ts = fromMaybe fallbackTs (lookup idx (zip [0..] entryTimestamps))
  368     in object
  369        [ "id"        .= entryId
  370        , "timestamp" .= T.pack ts
  371        , "direction" .= direction
  372        , "payload"   .= payloadJson
  373        , "harness"   .= (Nothing :: Maybe Text)
  374        , "model"     .= model
  375        , "channel"   .= (Nothing :: Maybe Text)
  376       , "raw"       .= TE.decodeUtf8 (BL.toStrict (A.encode rawLine))
  377       ]

  379 -- | Rewrite one on-disk 'ContentBlock' (GHC-Generics 'TaggedObject' shape)
  380 -- into the Anthropic-style block the frontend parses.
  381 cbToFrontend :: A.Value -> A.Value
  382 cbToFrontend blk =
  383   let bo = case blk of
  384         A.Object m -> m
  385         _          -> KeyMap.empty
  386       k = Key.fromText
  387       tag = case KeyMap.lookup (k "tag") bo of
  388         Just (A.String t) -> Just t
  389         _                  -> Nothing
  390       contents = KeyMap.lookup (k "contents") bo
  391       lookupT key m = case KeyMap.lookup (k key) m of
  392         Just (A.String t) -> Just t
  393         _                 -> Nothing
  394       lookupB key m = case KeyMap.lookup (k key) m of
  395         Just (A.Bool b) -> Just b
  396         _               -> Nothing
  397   in case tag of
  398        Just "CbText" -> case contents of
  399          Just (A.String t) -> object ["type" .= ("text" :: Text), "text" .= t]
  400          _                 -> fallback
  401        Just "CbToolUse" -> object
  402          [ "type"  .= ("tool_use" :: Text)
  403          , "id"    .= fromMaybe "" (lookupT "id" bo)
  404          , "name"  .= fromMaybe "" (lookupT "name" bo)
  405          , "input" .= fromMaybe A.Null (KeyMap.lookup (k "input") bo)
  406          ]
  407        Just "CbToolResult" -> object
  408          [ "type"        .= ("tool_result" :: Text)
  409          , "tool_use_id" .= fromMaybe "" (lookupT "forId" bo)
  410          , "content"     .= toolResultPartsToFrontend (KeyMap.lookup (k "parts") bo)
  411          , "is_error"    .= fromMaybe False (lookupB "isError" bo)
  412          ]
  413        _ -> fallback
  414   where
  415      fallback = object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]

  417 -- | Rewrite the on-disk 'parts' value into the Anthropic-style array
  418 -- the frontend parses (@[{type:"text", text:"..."}]@). 'ToolResultPart'
  419 -- is a @newtype TrpText Text@, so aeson's derived 'ToJSON' serializes it
  420 -- as a bare JSON string (not a 'TaggedObject'). The on-disk 'parts' is
  421 -- thus @["text"]@, not @[{tag:"TrpText", contents:"text"}]@. Handles both
  422 -- the bare-string shape and the object shape (defensive fallback).
  423 toolResultPartsToFrontend :: Maybe A.Value -> A.Value
  424 toolResultPartsToFrontend mparts =
  425   case mparts of
  426     Just (A.Array arr) ->
  427       A.Array (V.fromList (map trpToFrontend (V.toList arr)))
  428     other -> fromMaybe A.Null other

  430 trpToFrontend :: A.Value -> A.Value
  431 trpToFrontend blk =
  432   case blk of
  433     A.String t -> object ["type" .= ("text" :: Text), "text" .= t]
  434     A.Object m ->
  435       let tag = case KeyMap.lookup (Key.fromString "tag") m of
  436             Just (A.String t) -> Just t
  437             _                 -> Nothing
  438           contents = KeyMap.lookup (Key.fromString "contents") m
  439       in case tag of
  440            Just "TrpText" -> case contents of
  441              Just (A.String t) -> object ["type" .= ("text" :: Text), "text" .= t]
  442              _                 -> object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]
  443            _ -> object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]
  444     _ -> object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]

  446 -- | The set of opcode names whose 'EKHarness' entries should surface to
  447 -- the web frontend SPA as distinct harness entries (not be dropped by
  448 -- 'reconEntryToFrontend'). v1: @SKILL_LOAD@ so /skill load invocations
  449 -- appear in the transcript as identifiable skill-load events (per the
  450 -- user's "properly identified as a skill load operation" requirement);
  451 -- @SETUP_REPO@ so the clone/no-op/conflict/failure outcome of a
  452 -- repo-attached session creation is visible in the chat (the frontend's
  453 -- 'transcriptToMessages' renders it as a tool-call box; without this
  454 -- whitelist entry, the SETUP_REPO result recorded by
  455 -- 'recordSetupRepoResult' is dropped before reaching either the WS
  456 -- broadcast or the HTTP seed, so the user sees nothing for the repo
  457 -- setup).
  458 -- Approval-bearing entries always surface regardless of this whitelist.
  459 -- A 'Set' rather than a list to make the shared-state surface
  460 -- discoverable (see the design doc's §8 risk 2).
  461 userSurfacingOps :: Set.Set Text
  462 userSurfacingOps = Set.fromList ["SKILL_LOAD", "SETUP_REPO"]
  463 
  464 -- | Predicate: does a harness payload's @op.name@ fall in
  465 -- 'userSurfacingOps'? Returns 'False' for payloads with no @op@ key, a
  466 -- non-object @op@, or no @name@ field — those are dropped by the filter
  467 -- (matching the pre-v1 behavior for non-approval harness entries).
  468 isUserSurfacingOp :: KeyMap.KeyMap Value -> Bool
  469 isUserSurfacingOp o =
  470   case KeyMap.lookup (Key.fromText "op") o of
  471     Just (Object opObj) ->
  472       case KeyMap.lookup (Key.fromText "name") opObj of
  473         Just (String n) -> n `Set.member` userSurfacingOps
  474         _               -> False
  475     _ -> False

  477 -- | Map a reconstructed 'TranscriptEntry' (from 'reconstruct') to the
  478 -- frontend's TranscriptEntry JSON shape.
  479 reconEntryToFrontend :: Int -> TranscriptEntry -> Maybe A.Value
  480 reconEntryToFrontend idx te =
  481   case tePayload te of
  482     A.Null -> Nothing
  483     -- Drop harness entries (opcode invocations from the dispatcher) that
  484     -- carry a "harness" key in the payload — UNLESS they also carry an
  485     -- "approval" key (which marks them as approval-evidence entries that
  486     -- should surface so the user sees the confirmation decision in the
  487     -- transcript), OR their op.name is in 'userSurfacingOps' (which marks
  488     -- them as user-surfacing opcode invocations like /skill load that
  489     -- should appear as distinct harness entries, not be folded into the
  490     -- surrounding user/assistant bubbles).
  491     A.Object o
  492       | KeyMap.member (Key.fromText "harness") o,
  493         not (KeyMap.member (Key.fromText "approval") o),
  494         not (isUserSurfacingOp o) -> Nothing
  495     payloadVal -> Just $
  496       let dirStr = case teDirection te of
  497             Request  -> "request" :: Text
  498             Response -> "response"
  499           payloadJson = rewritePayload False payloadVal (teDirection te)
  500           -- The `raw` view preserves full tool input_schemas so the
  501           -- "View raw JSON" modal shows the verbatim schemas the LLM was
  502           -- sent (the `payload` field strips them to keep the wire small).
  503           rawJson = rewritePayload True payloadVal (teDirection te)
  504           entryId = let raw = teId te in if T.null raw then T.pack (show idx) else raw
  505           -- The originating channel label (e.g. "telegram", "web", "cli"),
  506           -- stamped into the request entry's erMeta by runTurn's
  507           -- requestMeta (and into SKILL_LOAD entries by
  508           -- recordSkillLoadResult). Surfaced as a top-level `channel` field
  509           -- so the frontend can attribute user messages — and skill loads —
  510           -- to the channel they came from. Null when the entry carries no
  511           -- channel (e.g. response entries, CLI TUI turns with no
  512           -- MessageSource).
  513           mChannel = case Map.lookup "channel" (teMeta te) of
  514             Just (A.String ch) -> Just ch
  515             _                  -> Nothing
  516       in object
  517          [ "id"        .= entryId
  518          , "timestamp" .= T.pack (showIso (teTimestamp te))
  519          , "direction" .= dirStr
  520          -- `payload` is a JSON OBJECT (not a string) so the frontend can
  521          -- access its fields directly without a second JSON.parse pass.
  522          -- The old format encoded payload as a JSON string
  523          -- (TE.decodeUtf8 (A.encode payloadJson)), which forced the browser
  524          -- to parse the outer JSON (un-escaping every `\"` in the payload
  525          -- string), then call JSON.parse(payload) AGAIN to parse the inner
  526          -- JSON — two full parse passes with O(n) escape processing in
  527          -- between. For a 279KB body this took ~889ms; as an object it's a
  528          -- single pass and should be <10ms.
  529          , "payload"   .= payloadJson
  530          , "harness"   .= (Nothing :: Maybe Text)
  531          , "model"     .= (Nothing :: Maybe Text)
  532          , "channel"   .= mChannel
  533          -- The `raw` field carries the full rewritten payload with tool
  534          -- input_schemas preserved, so the frontend's "View raw JSON"
  535          -- modal shows the verbatim schemas the LLM was sent. The legacy
  536          -- path (teLineToFrontend) still ships the verbatim on-disk line.
  537          , "raw"       .= TE.decodeUtf8 (BL.toStrict (A.encode rawJson))
  538          ]
  539 
  540 -- | Rewrite a reconstructed payload 'Value' from GHC-Generics
  541 -- 'TaggedObject' encoding to the Anthropic-style JSON the frontend parses.
  542 -- When @preserveSchemas@ is 'True', the @tools@ array keeps its full
  543 -- @input_schema@/parameters fields — used to build the @raw@ view so the
  544 -- "View raw JSON" modal shows the verbatim schemas the LLM was sent. When
  545 -- 'False', 'toolsWithDescriptions' strips them to keep the wire payload
  546 -- small (the frontend's collapsed Tools row only needs names + descriptions).
  547 rewritePayload :: Bool -> A.Value -> Direction -> A.Value
  548 rewritePayload preserveSchemas val dir =
  549   case val of
  550     A.Object o ->
  551       let k = Key.fromText
  552           rewriteMsgs = case KeyMap.lookup (k "messages") o of
  553             Just (A.Array arr) ->
  554               let msgs = map rewriteMessage (V.toList arr)
  555               in ["messages" .= A.Array (V.fromList msgs)]
  556             _ -> []
  557           rewriteContent = case KeyMap.lookup (k "content") o of
  558             Just (A.Array arr) ->
  559               let blocks = map cbToFrontend (V.toList arr)
  560               in ["content" .= A.Array (V.fromList blocks)]
  561             _ -> []
  562           passthrough key = case KeyMap.lookup key o of
  563             Just v  -> [key .= v]
  564             Nothing -> []
  565           usageFields = case KeyMap.lookup (k "usage") o of
  566             Just (A.Object uo) ->
  567               let uIn  = case KeyMap.lookup (k "input")  uo of
  568                     Just n -> Just n; _ -> Nothing
  569                   uOut = case KeyMap.lookup (k "output") uo of
  570                     Just n -> Just n; _ -> Nothing
  571               in case (uIn, uOut) of
  572                    (Just _, Just _) ->
  573                      [ "usage" .= object
  574                        [ "input_tokens"  .= uIn
  575                        , "output_tokens" .= uOut
  576                        ]
  577                      ]
  578                    _ -> []
  579             _ -> []
  580           toolsField = if preserveSchemas
  581             then passthrough (k "tools")
  582             else toolsWithDescriptions o
  583           fields = case dir of
  584             Request ->
  585               passthrough (k "system")
  586                <> passthrough (k "model")
  587                -- Replace the full tool definitions with name + description,
  588                -- stripping input_schema/parameters to keep the wire payload
  589                -- small. The full definitions (with input_schema) are the
  590                -- same across every request turn and can be 11KB+ each;
  591                -- shipping them in all 73 request entries of a 146-turn
  592                -- session wastes ~840KB. The frontend's "Tools" collapsed
  593                -- row shows the names + descriptions, and the expanded
  594                -- view's JSON carries the name + description array (useful
  595                -- for seeing WHAT tools were available + what they do, just
  596                -- not the full schemas). The frontend already deduplicates
  597                -- tools via `seenTools` so only the first occurrence renders.
  598                -- The @raw@ view (built with preserveSchemas=True) ships the
  599                -- full tools array so the "View raw JSON" modal shows the
  600                -- verbatim schemas the LLM was sent.
  601                <> toolsField
  602                <> passthrough (k "toolChoice")
  603                <> passthrough (k "maxTokens")
  604                <> passthrough (k "approval")
  605                <> passthrough (k "op")
  606                -- Harness/opcode-invocation entries (EKHarness) carry the
  607                -- opcode's input + result under these keys (see
  608                -- Seal.Transcript.Reconstruct.harnessPayload). The SKILL_LOAD
  609                -- result entry is a Request-direction harness entry; without
  610                -- passing @input@ + @result@ through, the frontend's
  611                -- @transcriptToMessages@ sees @op.name = "SKILL_LOAD"@ but no
  612                -- @result@, so the skill-load tool-call box never renders.
  613                 <> passthrough (k "input")
  614                 <> passthrough (k "result")
  615                 <> passthrough (k "channel")
  616                 <> rewriteMsgs
  617             Response ->
  618               passthrough (k "model")
  619                <> rewriteContent
  620                <> usageFields
  621                <> passthrough (k "stop")
  622                <> passthrough (k "durationMs")
  623       in A.object fields
  624     _ -> val
  625 
  626 -- | Extract name + description from a tools array, dropping input_schema /
  627 -- parameters to keep the wire payload small. Returns
  628 -- @["tools" .= [...]]@ where each element is
  629 -- @{"name": "<toolName>", "description": "<toolDesc>"}@ — the shape the
  630 -- frontend's 'extractToolDefNames' parses. Returns @[]@ when the tools field
  631 -- is absent or not an array.
  632 toolsWithDescriptions :: KeyMap.KeyMap Value -> [(Key, Value)]
  633 toolsWithDescriptions o =
  634   case KeyMap.lookup (Key.fromText "tools") o of
  635     Just (A.Array arr) ->
  636       let defs = map toolNameDesc (V.toList arr)
  637       in ["tools" .= A.Array (V.fromList defs)]
  638     _ -> []
  639   where
  640     -- Extract {name, description} from a tool definition Value, dropping
  641     -- input_schema / parameters. Handles both the Anthropic shape
  642     -- ({name, description, input_schema}) and the Ollama shape
  643     -- ({type:"function", function:{name, description, parameters}}).
  644     toolNameDesc v = case v of
  645       A.Object td ->
  646         case ( KeyMap.lookup (Key.fromText "name") td
  647              , KeyMap.lookup (Key.fromText "description") td ) of
  648           (Just n, Just d) -> A.object ["name" .= n, "description" .= d]
  649           (Just n, Nothing) -> A.object ["name" .= n]
  650           (Nothing, _) -> case KeyMap.lookup (Key.fromText "function") td of
  651             Just (A.Object fn) -> case KeyMap.lookup (Key.fromText "name") fn of
  652               Just n -> case KeyMap.lookup (Key.fromText "description") fn of
  653                 Just d -> A.object ["name" .= n, "description" .= d]
  654                 Nothing -> A.object ["name" .= n]
  655               Nothing -> A.object ["name" .= A.Null]
  656             _ -> A.object ["name" .= A.Null]
  657       _ -> A.object ["name" .= A.Null]
  658 
  659 -- | Rewrite one 'Message' (@{role, content: [...]}@, strip-prefix camelCase
  660 -- JSON) to the Anthropic-style shape (@{role, content: [...]}@) the frontend
  661 -- parses, with content blocks rewritten by 'cbToFrontend'. The role is
  662 -- lowercased because GHC-Generics encodes 'User'/'Assistant' as
  663 -- @"User"@/@"Assistant"@ but the frontend checks @msg.role === "user"@.
  664 rewriteMessage :: A.Value -> A.Value
  665 rewriteMessage msg =
  666   case msg of
  667     A.Object mo ->
  668       let k = Key.fromText
  669           role = case KeyMap.lookup (k "role") mo of
  670             Just (A.String t) -> T.toLower t
  671             _                 -> "user" :: Text
  672           rawContent = case KeyMap.lookup (k "content") mo of
  673             Just (A.Array arr) -> arr
  674             _                  -> mempty
  675           blocks = map cbToFrontend (V.toList rawContent)
  676       in A.object
  677          [ "role"    .= role
  678          , "content" .= A.Array (V.fromList blocks)
  679          ]
  680     _ -> msg

  682 -- | ISO-8601 with milliseconds + Z (the frontend's expected timestamp shape).
  683 showIso :: UTCTime -> String
  684 showIso = formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%S%3QZ"

  686 -- | The ISO timestamp of the most recent @request@-direction entry in a
  687 -- session's transcript (the last time the USER sent a message), or 'Nothing'
  688 -- when the session has no transcript or no user message. Used by the sidebar
  689 -- to sort tabs by last-user-message time (oldest first within each status
  690 -- bucket). Reads the same transcript paths as 'firstUserMessageSnippet';
  691 -- returns the timestamp as an ISO string (matching the frontend's
  692 -- TranscriptEntry.timestamp shape).
  693 lastUserMessageAt :: SealPaths -> SessionId -> IO (Maybe Text)
  694 lastUserMessageAt paths sid = do
  695   -- model + fallbackTs are only used for synthesized conversation.jsonl
  696   -- entries that lack an entries.jsonl sidecar; the empty model + a
  697   -- neutral fallback keep that path consistent without affecting the
  698   -- entries.jsonl path (which carries real timestamps).
  699   entries <- readTranscriptEntries paths "" "" sid
  700   pure (lastRequestTimestamp entries)
  701   where
  702     -- Walk the frontend-shaped entries in reverse, return the first
  703     -- (i.e. most recent) @request@-direction entry's timestamp.
  704     lastRequestTimestamp :: [A.Value] -> Maybe Text
  705     lastRequestTimestamp vals =
  706       case reverse vals of
  707         [] -> Nothing
  708         xs -> go xs
  709       where
  710         k = Key.fromText
  711         isBlank (A.String t) = T.null t
  712         isBlank _ = True
  713         go [] = Nothing
  714         go (v : rest) =
  715           case v of
  716             A.Object o ->
  717               case ( KeyMap.lookup (k "direction") o
  718                    , KeyMap.lookup (k "timestamp") o ) of
  719                 (Just (A.String d), Just (A.String t))
  720                   | d == "request" && not (isBlank (A.String t)) -> Just t
  721                 _ -> go rest
  722             _ -> go rest

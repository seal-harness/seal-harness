    2 -- | The remote SSH executor (the 'EbRemote' arm of 'ExecBackend'). Shells
    3 -- out to the system @ssh@ binary via fixed argv (no shell interpreter)
    4 -- with **mandatory host-key pinning**: @StrictHostKeyChecking=yes@ and a
    5 -- pinned @UserKnownHostsFile@. A host-key mismatch is a hard security
    6 -- failure ('ExecHostKeyMismatch'), never bypassed. Batch mode
    7 -- (@BatchMode=yes@) prevents interactive prompts (a headless run cannot
    8 -- confirm a new host key, so adoption fail-closes).
    9 --
   10 -- == Connection multiplexing
   11 --
   12 -- Every invocation joins one of TWO DISJOINT @ControlMaster=auto@ pools,
   13 -- keyed by the 'ControlPath' suffix:
   14 --
   15 --   * @m-%C@ — plain sessions (never @-A@): discovery snapshots, file
   16 --     reads/writes, shell execs.
   17 --   * @a-%C@ — agent-forwarding sessions only (the git-credential ops that
   18 --     pass @-A@).
   19 --
   20 -- The split preserves the §5.6 opt-in invariant: agent forwarding over a
   21 -- multiplexed connection requires the /master/ process to have been started
   22 -- with @-A@, so a plain op can never ride an agent-forwarding master (and a
   23 -- forwarding op never upgrades the shared plain pool). The first op to each
   24 -- pool pays the TCP + key-exchange + auth handshake; every subsequent op for
   25 -- ~10 minutes ('ControlPersist') rides the persistent master socket — turning
   26 -- N serialized round trips from N handshakes into 1 + N fast channel opens.
   27 --
   28 -- Security posture is unchanged: the master itself was established under the
   29 -- same pinning options (same argv shape ⇒ same config), @BatchMode=yes@
   30 -- still forbids interactive prompts, and a stale socket (master killed) is
   31 -- detected by @ControlMaster=auto@, which falls back to a fresh direct
   32 -- connection. The socket directory is private (mode 0700) under @~/.seal/@.

   68 -- | Build the fixed argv for an SSH exec. The argv is:
   69 --
   70 -- @ssh -o StrictHostKeyChecking=yes -o BatchMode=yes -o
   71 -- UserKnownHostsFile=\<pinned\> [-p \<port\>] [-i \<identity\>] user\@host
   72 -- -- \<command\>@
   73 --
   74 -- The @--@ separator guards against option injection from the command
   75 -- (the validated 'ShellCommand' is passed as a single arg, not
   76 -- interpreted as ssh flags). No @-c@ shell interpreter — the command runs
   77 -- as the remote user's default shell, but the command string is a single
   78 -- argv element (ssh itself passes it to the remote shell, but the
   79 -- *local* argv is fixed, no shell wrapping).
   80 --
   81 -- Agent forwarding is OFF (no @-A@) — this is the default for all
   82 -- non-credential ops (SHELL_EXEC, file writes, process ops). Git-credential
   83 -- ops (clone/fetch/pull/push with a deploy key) use 'sshExecArgvForwarding'
   84 -- to opt in to @-A@ so the forwarded @SSH_AUTH_SOCK@ reaches the remote
   85 -- @git@. A blanket @-A@ would widen the signing-oracle surface; opt-in
   86 -- per git-op is the security-critical invariant (design §5.6, W2 DoD).
   87 sshExecArgv :: SshConfig -> Text -> [String]
   88 sshExecArgv cfg cmd =
   89   [ "ssh"
   90   , "-o", "StrictHostKeyChecking=yes"
   91   , "-o", "BatchMode=yes"
   92   , "-o", "UserKnownHostsFile=" <> scKnownHosts cfg
   93   ]
   94   <> muxArgs "m"
   95   <> portArg
   96   <> identityArg
   97   <> [ userAtHost
   98      , "--"
   99      , T.unpack cmd
  100      ]
  101   where
  102     userAtHost = T.unpack (getSshUser (scUser cfg)) <> "@"
  103                  <> T.unpack (getSshHost (scHost cfg))
  104     portArg = case scPort cfg of
  105       22 -> []
  106       p  -> ["-p", show p]
  107     identityArg = case scIdentity cfg of
  108       Nothing -> []
  109       Just f  -> ["-i", f]
  110 
  111 -- | Variant of 'sshExecArgv' that adds @-A@ (forward the agent). Used by
  112 -- git-credential ops (clone/fetch/pull/push with a deploy key) so the
  113 -- remote @git@ can sign via the forwarded @SSH_AUTH_SOCK@. The @-A@ is
  114 -- opt-in per git-op — non-credential remote ops use 'sshExecArgv' (no
  115 -- @-A@) to keep the signing-oracle surface minimal (design §5.6).
  116 --
  117 -- The remote arm of 'uioShellExecGitEnv' (UntrustedIO.hs) uses this via
  118 -- 'runRemoteShellForwarding' for deploy-key ops. The @SSH_AUTH_SOCK@ /
  119 -- @SSH_AGENT_PID@ are stripped from the env prefix (the forwarded socket
  120 -- replaces them) and the @known_hosts@ content is written to a remote
  121 -- temp file so @GIT_SSH_COMMAND@'s @UserKnownHostsFile=@ resolves on the
  122 -- remote machine. PAT ops work on both executors (token in argv, no
  123 -- agent).
  124 sshExecArgvForwarding :: SshConfig -> Text -> [String]
  125 sshExecArgvForwarding cfg cmd =
  126   [ "ssh", "-A"
  127   , "-o", "StrictHostKeyChecking=yes"
  128   , "-o", "BatchMode=yes"
  129   , "-o", "UserKnownHostsFile=" <> scKnownHosts cfg
  130   ]
  131   <> muxArgs "a"
  132   <> portArg
  133   <> identityArg
  134   <> [ userAtHost
  135      , "--"
  136      , T.unpack cmd
  137      ]
  138   where
  139     userAtHost = T.unpack (getSshUser (scUser cfg)) <> "@"
  140                  <> T.unpack (getSshHost (scHost cfg))
  141     portArg = case scPort cfg of
  142       22 -> []
  143       p  -> ["-p", show p]
  144     identityArg = case scIdentity cfg of
  145       Nothing -> []
  146       Just f  -> ["-i", f]

  148 -- | The multiplexing options for one master pool. @pool@ is @\"m\"@ (plain)
  149 -- or @\"a\"@ (agent-forwarding); the ControlPath suffix keeps the pools
  150 -- disjoint. @%C@ is ssh's hash of |local host|local port|remote host|remote
  151 -- port, so each target gets its own master even within one pool.
  152 muxArgs :: String -> [String]
  153 muxArgs pool =
  154   [ "-o", "ControlMaster=auto"
  155   , "-o", "ControlPath=" <> sshMuxBaseDir </> (pool <> "-%C")
  156   , "-o", "ControlPersist=600"
  157   ]
  158 
  159 -- | The absolute mux socket directory, @~/.seal/ssh-mux@ (mode 0700),
  160 -- created once at first use. An absolute path (not a @~@-form) so the argv
  161 -- builder stays pure without relying on ssh's tilde expansion of @-o@
  162 -- values; 'ensureMuxDir' re-ensures it before every spawn.
  163 {-# NOINLINE sshMuxBaseDir #-}
  164 sshMuxBaseDir :: FilePath
  165 sshMuxBaseDir = unsafePerformIO $ do
  166   home <- getHomeDirectory
  167   let d = home </> ".seal" </> "ssh-mux"
  168   ensureMuxDirAt d
  169   pure d
  170 
  171 -- | Idempotently create the mux socket directory with private permissions.
  172 ensureMuxDirAt :: FilePath -> IO ()
  173 ensureMuxDirAt d = do
  174   createDirectoryIfMissing True d
  175   setFileMode d 0o700

  177 -- | The SSH runner seam — a record of two IO actions:
  178 --
  179 --   * 'runRemote' runs an argv with no stdin (the existing path, used by
  180 --     'runRemoteShell' and the command opcodes).
  181 --   * 'runRemoteStdin' runs an argv with a stdin payload (used by the
  182 --     remote 'UntrustedIO' arm for file writes — the content is piped
  183 --     over the SSH channel, never interpolated into the command string).
  184 --
  185 -- The real implementation ('mkRealRemoteRunner') uses 'System.Process';
  186 -- tests supply a fake (via 'mkFakeRemoteRunner' or
  187 -- 'mkFakeRemoteRunnerRecording') that simulates unreachable /
  188 -- host-key-mismatch / canned-stdout, and records the argv + stdin for
  189 -- assertions.
  190 data RemoteRunner = RemoteRunner
  191   { runRemote      :: [String] -> IO (Either ExecError Text)
  192   , runRemoteStdin :: [String] -> ByteString -> IO (Either ExecError Text)
  193   , runRemoteEnv   :: [(String, String)] -> [String] -> IO (Either ExecError Text)
  194     -- ^ Run an argv with local env overrides MERGED over the inherited
  195     -- environment. Used by the deploy-key remote clone path: the local
  196     -- @ssh -A@ process needs the SEAL agent's @SSH_AUTH_SOCK@ (not the
  197     -- ambient one) so the forwarded agent has the deploy key loaded.
  198   }

  200 -- | The real SSH runner via 'System.Process'. Fail-closed: any IOError or
  201 -- non-zero exit becomes a structured 'ExecError'. Exit code 255 with a
  202 -- "Host key verification failed" stderr → 'ExecHostKeyMismatch' (hard
  203 -- failure, never bypassed).
  204 mkRealRemoteRunner :: RemoteRunner
  205 mkRealRemoteRunner = RemoteRunner
  206   { runRemote      = runReal Nothing []
  207   , runRemoteStdin = \argv stdinBytes -> runReal (Just stdinBytes) [] argv
  208   , runRemoteEnv   = runReal Nothing
  209   }
  210   where
  211     runReal :: Maybe ByteString -> [(String, String)] -> [String] -> IO (Either ExecError Text)
  212     runReal mStdin envExtras argv = do
  213       -- Re-ensure the mux socket dir before every spawn (guards against a
  214       -- mid-run deletion; ssh fatals if ControlPath is uncreatable).
  215       ensureMuxDirAt sshMuxBaseDir
  216       let (program, args) = case argv of
  217             (p : as) -> (p, as)
  218             []       -> error "runReal: empty argv (unreachable)"
  219           cp0 = (proc program args)
  220                   { std_out = CreatePipe, std_err = CreatePipe
  221                   , create_group = True
  222                   }
  223           -- ^ @create_group = True@ puts the ssh child in its own POSIX
  224           -- process group so 'withManagedProcess' can kill the whole group
  225           -- on cleanup (SIGTERM → grace → SIGKILL). Killing the local ssh
  226           -- process closes the SSH channel, which terminates the remote
  227           -- command (design Risks §2 — a backgrounded remote child via
  228           -- @setsid ... &@ survives, accepted v1 limitation).
  229           cp1 = case mStdin of
  230             Nothing -> cp0 { std_in = NoStream }
  231             Just _  -> cp0 { std_in = CreatePipe }
  232       mEnv <- case envExtras of
  233         [] -> pure Nothing
  234         xs -> Just <$> mergeEnvReal xs
  235       let cp = cp1 { env = mEnv }
  236           maxOutput = 50_000  -- bounded output cap (Task 4)
  237       res <- try @IOError
  238              (withManagedProcess cp $ \ph mIn hOut hErr -> do
  239                 -- Pipe the stdin payload (if any) to the process and close
  240                 -- the write end so the remote @cat@/@tee@ sees EOF.
  241                 case (mIn, mStdin) of
  242                   (Just hIn, Just bs) -> do
  243                     BS.hPut hIn bs
  244                     hClose hIn
  245                   _ -> pure ()
  246                 out <- readBounded hOut maxOutput
  247                 err <- readBounded hErr maxOutput
  248                 ec  <- waitForProcess ph
  249                 pure (ec, out, err))
  250       case res of
  251         Left _ioErr -> pure (Left ExecRemoteUnreachable)  -- launch fail = unreachable
  252         Right (ExitSuccess, out, _) -> pure (Right out)
  253         Right (ExitFailure 255, _, err)
  254           | "Host key verification failed" `T.isInfixOf` err
  255           -> pure (Left ExecHostKeyMismatch)
  256           | otherwise
  257           -> pure (Left ExecRemoteUnreachable)
  258         Right (ExitFailure 127, _, _)  -> pure (Left ExecRemoteUnreachable)  -- ssh not on PATH
  259         Right (ExitFailure _n, _, _err) -> pure (Left ExecRemoteUnreachable)

  261 -- | Run a shell command via the remote SSH executor. The command is a
  262 -- validated 'ShellCommand' (NUL rejected). Returns the stdout or a
  263 -- structured 'ExecError'. Agent forwarding is OFF (no @-A@) — use
  264 -- 'runRemoteShellForwarding' for git-credential ops.
  265 runRemoteShell :: RemoteRunner -> SshConfig -> ShellCommand -> IO (Either ExecError Text)
  266 runRemoteShell runner cfg cmd =
  267   let argv = sshExecArgv cfg (textShellCommand cmd)
  268   in runRemote runner argv

  270 -- | Variant of 'runRemoteShell' that forwards the agent (@-A@). Used by
  271 -- git-credential ops (clone/fetch/pull/push with a deploy key).
  272 runRemoteShellForwarding
  273   :: RemoteRunner -> SshConfig -> ShellCommand -> IO (Either ExecError Text)
  274 runRemoteShellForwarding runner cfg cmd =
  275   let argv = sshExecArgvForwarding cfg (textShellCommand cmd)
  276   in runRemote runner argv

  278 -- | Run a shell command via the remote SSH executor WITH a stdin payload.
  279 -- The bytes are piped to the remote process's stdin — used by the remote
  280 -- 'UntrustedIO' arm for file writes (the file content goes over the SSH
  281 -- channel, never interpolated into the command string, so content with
  282 -- quotes/backticks/@$()@ is safe). The argv is the same fixed
  283 -- 'sshExecArgv' shape; the stdin is a runtime parameter. Agent forwarding
  284 -- is OFF — use 'runRemoteWithStdinForwarding' for git-credential ops.
  285 runRemoteWithStdin
  286   :: RemoteRunner -> SshConfig -> ShellCommand -> ByteString
  287   -> IO (Either ExecError Text)
  288 runRemoteWithStdin runner cfg cmd stdinBytes =
  289   let argv = sshExecArgv cfg (textShellCommand cmd)
  290   in runRemoteStdin runner argv stdinBytes

  295 runRemoteWithStdinForwarding
  296   :: RemoteRunner -> SshConfig -> ShellCommand -> ByteString

  298 runRemoteWithStdinForwarding runner cfg cmd stdinBytes =

  302 -- | A fake 'RemoteRunner' that always returns a canned result (the
  303 -- 'Either' is supplied by the caller). The argv + stdin are ignored.
  304 -- Useful for the unreachable / host-key-mismatch tests.
  305 mkFakeRemoteRunner :: Either ExecError Text -> RemoteRunner
  306 mkFakeRemoteRunner canned = RemoteRunner
  307   { runRemote      = \_argv -> pure canned
  308   , runRemoteStdin = \_argv _stdin -> pure canned
  309   , runRemoteEnv   = \_env _argv -> pure canned
  310   }

  312 -- | A fake 'RemoteRunner' that records every call's argv + stdin into an
  313 -- 'IORef' (oldest-last) and returns a canned result. Used by the remote
  314 -- 'UntrustedIO' tests to assert the SSH argv + piped content are correct
  315 -- (file ops go through SSH, not the local FS). The recorded entry is
  316 -- @(argv, mStdin)@ — 'Nothing' for the no-stdin path, @'Just' bytes@ for
  317 -- the stdin path.
  318 mkFakeRemoteRunnerRecording
  319   :: IORef [([String], Maybe ByteString)]
  320   -> Either ExecError Text
  321   -> RemoteRunner
  322 mkFakeRemoteRunnerRecording ref canned = RemoteRunner
  323   { runRemote      = \argv -> do
  324       modifyIORef' ref (++ [(argv, Nothing)])
  325       pure canned
  326   , runRemoteStdin = \argv stdin -> do
  327       modifyIORef' ref (++ [(argv, Just stdin)])
  328       pure canned
  329   , runRemoteEnv = \_env argv -> do
  330       modifyIORef' ref (++ [(argv, Nothing)])
  331       pure canned
  332   }

  334 -- | Merge env overrides over the inherited environment (overrides win).
  335 -- Mirrors 'mergeEnv' in 'UntrustedIO.hs' but kept local to avoid a
  336 -- circular import.
  337 mergeEnvReal :: [(String, String)] -> IO [(String, String)]
  338 mergeEnvReal overrides =
  339   Map.toList . Map.union (Map.fromList overrides) . Map.fromList <$> getEnvironment

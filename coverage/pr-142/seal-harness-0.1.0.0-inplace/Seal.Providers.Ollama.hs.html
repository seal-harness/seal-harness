   54 -- | Response timeout for Ollama @/api/chat@ requests, in microseconds.
   55 -- The 'newTlsManager' default is 90 seconds, which is too short for
   56 -- non-streaming requests to remote models proxied through the local Ollama
   57 -- daemon (e.g. @glm-5.2:cloud@): with a large context (49k+ input tokens)
   58 -- the model can take 74+ seconds to generate the full response before
   59 -- returning it in one shot. A 5-minute timeout gives generous headroom for
   60 -- large-context non-streaming generation. This is a per-request override
   61 -- (the manager's default is not changed, so other HTTP clients in the
   62 -- process keep their own timeouts).
   63 ollamaResponseTimeoutMicro :: Int
   64 ollamaResponseTimeoutMicro = 300_000_000  -- 5 minutes

  116 encodeRequest :: CompletionRequest -> Value
  117 encodeRequest cr = object $
  118   [ "model"    .= crModel cr
  119   , "stream"   .= False
  120   , "messages" .= (systemMsgs <> concatMap encMsg (crMessages cr))
  121   , "options"  .= object ["num_predict" .= crMaxTokens cr]
  122   ]
  123   <> ["tools" .= map encTool (crTools cr) | not (null (crTools cr))]
  124   where
  125     systemMsgs =
  126       maybe []
  127         (\s -> [object ["role" .= ("system" :: Text), "content" .= s]])
  128         (crSystem cr)
  129 
  130 -- | Like 'encodeRequest' but with @"stream" .= True@. Used by 'streamComplete'
  131 -- for the streaming HTTP request.
  132 encodeStreamRequest :: CompletionRequest -> Value
  133 encodeStreamRequest cr =
  134   case encodeRequest cr of
  135     Object o -> Object (KeyMap.insert "stream" (Bool True) o)
  136     other    -> other

  138 -- | Flatten one provider-agnostic message into zero or more Ollama messages.
  139 -- A User message becomes a "user" message (its text, if any) followed by one
  140 -- "tool" message per tool-result block. An Assistant message becomes one
  141 -- "assistant" message carrying its text and any tool_calls.
  142 encMsg :: Message -> [Value]
  143 encMsg (Message User blocks) =
  144   let texts = [t | CbText t <- blocks]
  145       userMsg =
  146         [ object ["role" .= ("user" :: Text), "content" .= T.intercalate "\n" texts]
  147         | not (null texts) ]
  148       toolMsgs =
  149         [ object ["role" .= ("tool" :: Text), "content" .= renderToolContent isErr parts]
  150         | CbToolResult _ parts isErr <- blocks ]
  151   in userMsg <> toolMsgs
  152 encMsg (Message Assistant blocks) =
  153   let content = T.intercalate "\n" [t | CbText t <- blocks]
  154       toolCalls =
  155         [ object ["function" .= object ["name" .= n, "arguments" .= inp]]
  156         | CbToolUse _ (OpName n) inp <- blocks ]
  157       tc = ["tool_calls" .= toolCalls | not (null toolCalls)]
  158   in [object (["role" .= ("assistant" :: Text), "content" .= content] <> tc)]
  159 
  160 -- | Ollama's tool role carries no structured error flag, so an errored result
  161 -- is marked in-band: its text is prefixed so the model can see the call failed.
  162 renderToolContent :: Bool -> [ToolResultPart] -> Text
  163 renderToolContent isErr parts =
  164   let body = T.intercalate "\n" [t | TrpText t <- parts]
  165   in if isErr then "[tool error] " <> body else body

  167 encTool :: ToolDefinition -> Value
  168 encTool (ToolDefinition (OpName n) d sch) =
  169   -- Omit parameters entirely when it's the on-demand stub (OpenAI/Ollama's
  170   -- parameters field is optional). Keeps the stub tools at zero schema-token
  171   -- cost; the model retrieves a tool's real schema via OPCODE_DESCRIBE.
  172   if sch == stubSchema
  173     then object
  174            [ "type" .= ("function" :: Text)
  175            , "function" .= object ["name" .= n, "description" .= d]
  176            ]
  177     else object
  178            [ "type" .= ("function" :: Text)
  179            , "function" .= object ["name" .= n, "description" .= d, "parameters" .= sch]
  180            ]

  187 -- | Like 'decodeResponse' but starts tool-call ids at the given index so
  188 -- multiple responses in the same conversation get globally unique ids
  189 -- (Ollama tool calls carry no id).
  190 decodeResponseFrom :: Int -> Value -> Either Text CompletionResponse
  191 decodeResponseFrom start = mapLeft T.pack . parseEither (parseRespFrom start)
  192   where mapLeft f = either (Left . f) Right
  193 
  194 parseRespFrom :: Int -> Value -> Parser CompletionResponse
  195 parseRespFrom start = withObject "ollama response" $ \o -> do
  196   msg        <- o .: "message"
  197   content    <- msg .:? "content" .!= ""
  198   rawCalls   <- msg .:? "tool_calls" .!= ([] :: [Value])
  199   toolBlocks <- traverse parseToolCall (zip [start ..] rawCalls)
  200   doneReason <- o .:? "done_reason"
  201   promptTok  <- o .:? "prompt_eval_count" .!= 0
  202   evalTok    <- o .:? "eval_count" .!= 0
  203   let textBlocks = [CbText content | not (T.null content)]
  204       blocks     = textBlocks <> toolBlocks
  205       stop       = if not (null toolBlocks) then StopToolUse else stopFromDone doneReason
  206   pure (CompletionResponse blocks stop (Usage promptTok evalTok))
  207 
  208 -- | Ollama tool calls carry no id; synthesize a stable "call_<i>" per index.
  209 parseToolCall :: (Int, Value) -> Parser ContentBlock
  210 parseToolCall (i, v) = flip (withObject "tool_call") v $ \o -> do
  211   fn   <- o .: "function"
  212   name <- fn .: "name"
  213   args <- fn .:? "arguments" .!= object []
  214   pure (CbToolUse (ToolCallId ("call_" <> T.pack (show i))) (OpName name) args)
  215 
  216 stopFromDone :: Maybe Text -> StopReason
  217 stopFromDone (Just "length") = StopMaxTokens
  218 stopFromDone (Just "stop")   = StopEnd
  219 stopFromDone Nothing         = StopEnd
  220 stopFromDone (Just other)    = StopOther other

  222 -- Streaming chunk decoding --------------------------------------------------

  224 -- | Accumulator state for decoding Ollama streaming NDJSON chunks into
  225 -- 'StreamEvent's. Ollama streams one JSON object per line; each carries
  226 -- incremental @message.content@ (text deltas) and, when the model emits tool
  227 -- calls, a complete @tool_calls@ array in a single chunk (Ollama does not
  228 -- stream tool-call arguments incrementally — the whole array appears at
  229 -- once). The state tracks which tool ids have already been emitted so we
  230 -- only fire 'StreamToolStart'/'StreamToolEnd' once per tool.
  231 data StreamChunkState = StreamChunkState
  232   { scsToolIndex :: Int
  233     -- ^ The next tool-call index to assign (Ollama tool calls carry no id;
  234     -- we synthesize @call_\<i\>@ sequentially, matching the non-streaming path).
  235   , scsSeenToolIds :: [ToolCallId]
  236     -- ^ Tool ids already emitted via 'StreamToolStart'/'StreamToolEnd', so
  237     -- we don't re-emit them if the same chunk is processed twice or if
  238     -- subsequent chunks repeat the tool_calls array.
  239   , scsSawTools :: Bool
  240     -- ^ True once any tool call was emitted during this stream. Ollama's
  241     -- @done_reason@ for a tool-call response is still @"stop"@ (which maps
  242     -- to 'StopEnd'), but the non-streaming path overrides the stop reason
  243     -- to 'StopToolUse' when tool blocks are present (see 'parseRespFrom').
  244     -- This flag lets the streaming path do the same override in the
  245     -- 'StreamDone' chunk so the transcript's stop reason is accurate.
  246   }
  247 
  248 -- | Initial streaming state (no tools seen yet).
  249 initialStreamChunkState :: StreamChunkState
  250 initialStreamChunkState = StreamChunkState 0 [] False
  251 
  252 -- | Decode one Ollama streaming NDJSON chunk into zero or more 'StreamEvent's,
  253 -- threading the accumulator state. A single chunk can carry:
  254 --
  255 --   * @message.content@ (a text delta) → 'StreamTextChunk'
  256 --   * @message.tool_calls@ (complete tool calls) → 'StreamToolStart' +
  257 --     'StreamToolEnd' per tool (Ollama emits these as complete objects, not
  258 --     incremental deltas)
  259 --   * @done == true@ → 'StreamDone' with stop reason + usage
  260 --
  261 -- Returns the updated state alongside the events.
  262 decodeStreamChunk
  263   :: StreamChunkState -> Value -> Either Text (StreamChunkState, [StreamEvent])
  264 decodeStreamChunk st = mapLeft T.pack . parseEither (parseStreamChunk st)
  265   where mapLeft f = either (Left . f) Right
  266 
  267 parseStreamChunk :: StreamChunkState -> Value -> Parser (StreamChunkState, [StreamEvent])
  268 parseStreamChunk st = withObject "ollama stream chunk" $ \o -> do
  269   done <- o .:? "done" .!= False
  270   if done
  271     then do
  272       doneReason <- o .:? "done_reason"
  273       promptTok  <- o .:? "prompt_eval_count" .!= 0
  274       evalTok    <- o .:? "eval_count" .!= 0
  275       let rawStop = stopFromDone doneReason
  276           -- Override StopEnd → StopToolUse when tools were emitted during
  277           -- this stream (Ollama reports done_reason="stop" even for
  278           -- tool-call responses; the non-streaming path does the same
  279           -- override in parseRespFrom).
  280           stop = if scsSawTools st && rawStop == StopEnd then StopToolUse else rawStop
  281       pure (st, [StreamDone stop (Usage promptTok evalTok)])
  282     else do
  283       msgVal <- o .:? "message" .!= object []
  284       (content, rawCalls) <- parseMsgFields msgVal
  285       let textEvents = [StreamTextChunk content | not (T.null content)]
  286           (st', toolEvents) = foldl processTool (st, []) rawCalls
  287       pure (st', textEvents <> toolEvents)
  288   where
  289     processTool (s, evs) rawCall = case parseEither parseToolCallRaw rawCall of
  290       Left _ -> (s, evs)
  291       Right (mTcid, name, args) ->
  292         let tcid = fromMaybe (ToolCallId ("call_" <> T.pack (show (scsToolIndex s)))) mTcid
  293         in if tcid `elem` scsSeenToolIds s
  294           then (s, evs)  -- already emitted; skip duplicate
  295           else
  296             let s' = s { scsToolIndex = scsToolIndex s + 1
  297                        , scsSeenToolIds = tcid : scsSeenToolIds s
  298                        , scsSawTools = True
  299                        }
  300             in (s', evs <> [StreamToolStart tcid name, StreamToolEnd tcid name args])
  301 
  302 parseMsgFields :: Value -> Parser (Text, [Value])
  303 parseMsgFields = withObject "stream message" $ \msg -> do
  304   content <- msg .:? "content" .!= ""
  305   rawCalls <- msg .:? "tool_calls" .!= ([] :: [Value])
  306   pure (content, rawCalls)
  307 
  308 parseToolCallRaw :: Value -> Parser (Maybe ToolCallId, OpName, Value)
  309 parseToolCallRaw = withObject "tool_call" $ \o -> do
  310   fn   <- o .: "function"
  311   name <- fn .: "name"
  312   args <- fn .:? "arguments" .!= object []
  313   mId  <- o .:? "id" .!= ("" :: Text)
  314   let mTcid = if T.null mId then Nothing else Just (ToolCallId mId)
  315   pure (mTcid, OpName name, args)

  333 -- | Render an 'HttpException' as a user-facing error. A response timeout
  334 -- (the model took longer than 'ollamaResponseTimeoutMicro' to generate)
  335 -- gets a distinct message so the user knows the daemon is running but the
  336 -- model was too slow — not that the daemon is down. Other transport errors
  337 -- (connection refused, etc.) use the generic 'unreachableMsg'.
  338 ollamaHttpExceptionMsg :: Text -> HttpException -> Text
  339 ollamaHttpExceptionMsg base = \case
  340   HttpExceptionRequest _ ResponseTimeout ->
  341     "Ollama response timed out after "
  342       <> T.pack (show (ollamaResponseTimeoutMicro `div` 1_000_000))
  343       <> "s — the model (likely a remote/cloud model with a large context) \
  344          \took too long to respond. Try reducing the conversation context, \
  345          \switching to a faster/local model, or send \"continue\" to retry."
  346   _ -> unreachableMsg base

  350 -- | POST {base}/api/chat with the given headers; decode, or return a key-safe
  351 -- transport / HTTP-status error.
  352 sendChat
  353   :: Ollama -> RequestHeaders -> CompletionRequest
  354   -> IO (Either Text CompletionResponse)
  355 sendChat o hdrs cr = do
  356   result <- try $ do
  357     initReq <- parseRequest (T.unpack ("POST " <> chatUrl (olBaseUrl o)))
  358     let req = initReq
  359           { requestBody     = RequestBodyLBS (encode (encodeRequest cr))
  360           , requestHeaders  = hdrs
  361           , responseTimeout  = responseTimeoutMicro ollamaResponseTimeoutMicro
  362           }
  363     httpLbs req (olManager o)
  364   case result of
  365     Left (e :: HttpException) -> pure (Left (ollamaHttpExceptionMsg (olBaseUrl o) e))
  366     Right resp -> do
  367       let code = statusCode (responseStatus resp)
  368       if code >= 200 && code <= 299
  369         then case eitherDecode (responseBody resp) of
  370           Left e  -> pure (Left (T.pack e))
  371           Right v -> do
  372             startIdx <- claimToolCallIds (olCallCounter o) (countToolCalls v)
  373             pure (decodeResponseFrom startIdx v)
  374         else pure $ Left $ ollamaErrorText code
  375           (TE.decodeUtf8With TEE.lenientDecode (BL.toStrict (responseBody resp)))
  376 
  377 -- | Count how many tool_calls appear in the response (to advance the counter).
  378 countToolCalls :: Value -> Int
  379 countToolCalls v = case v of
  380   Object o -> case KeyMap.lookup (Key.fromString "message") o of
  381     Just (Object msg) -> case KeyMap.lookup (Key.fromString "tool_calls") msg of
  382       Just (Array arr) -> V.length arr
  383       _ -> 0
  384     _ -> 0
  385   _ -> 0
  386 
  387 -- | Atomically claim @n@ contiguous tool-call ids, returning the inclusive
  388 -- start of the claimed range. 'atomicModifyIORef'' advances the counter by
  389 -- @n@ and returns the previous value in one CAS, so concurrent callers can
  390 -- never overlap (the read-then-advance pattern raced here: two threads both
  391 -- read 0 before either advanced, both emitted "call_0"). Claiming 0 is a
  392 -- no-op read. Exposed for the concurrency test in 'OllamaSpec'.
  393 claimToolCallIds :: IORef Int -> Int -> IO Int
  394 claimToolCallIds counter n = atomicModifyIORef' counter (\i -> (i + n, i))

  430 -- | POST {base}/api/chat with @stream:true@ and stream NDJSON chunks to the
  431 -- callback. Uses 'withResponse' (incremental body reading) instead of
  432 -- 'httpLbs' (whole body), so the first chunk arrives as soon as the model
  433 -- starts generating — avoiding the 90s response timeout that plagued
  434 -- non-streaming requests to remote/cloud models.
  435 sendChatStream
  436   :: Ollama -> RequestHeaders -> CompletionRequest
  437   -> (StreamEvent -> IO Bool)
  438   -> IO (Either Text StreamOutcome)
  439 sendChatStream o hdrs cr k = do
  440   result <- try $ do
  441     initReq <- parseRequest (T.unpack ("POST " <> chatUrl (olBaseUrl o)))
  442     let req = initReq
  443           { requestBody     = RequestBodyLBS (encode (encodeStreamRequest cr))
  444           , requestHeaders  = hdrs
  445           , responseTimeout  = responseTimeoutMicro ollamaResponseTimeoutMicro
  446           }
  447     withResponse req (olManager o) $ \resp -> do
  448       let code = statusCode (responseStatus resp)
  449       if code >= 200 && code <= 299
  450         then streamBody (responseBody resp)
  451         else do
  452           -- Non-2xx: read the body for the error message and return it
  453           -- directly (not via an exception).
  454           body <- brReadSome (responseBody resp) 4096
  455           let errTxt = ollamaErrorText code
  456                 (TE.decodeUtf8With TEE.lenientDecode (BL.toStrict body))
  457           pure (Left errTxt)
  458   case result of
  459     Left (e :: HttpException) -> pure (Left (ollamaHttpExceptionMsg (olBaseUrl o) e))
  460     Right inner -> pure inner
  461   where
  462     -- Read the response body chunk by chunk, decode each line as a JSON
  463     -- chunk, emit StreamEvents to the callback. Returns False immediately if
  464     -- the callback returns False (abort). Accumulates the final stop+usage
  465     -- from the StreamDone event.
  466     streamBody body = go BS.empty initialStreamChunkState
  467       where
  468         go buf st' = do
  469           chunk <- brRead body
  470           if BS.null chunk
  471             then -- Stream ended without a done chunk; flush any remaining
  472                  -- buffer as a text chunk and treat as a clean stop.
  473                  do unless (BS.null buf) $
  474                       do (_, events) <- processBuffer buf st'
  475                          void (emitEvents events)
  476                     pure (Right (StreamOutcome StopEnd (Usage 0 0)))
  477             else do
  478               let combined = buf <> chunk
  479                   ls = BS.split 10 combined  -- split on '\n'
  480               if BS.null (last ls)
  481                 then do
  482                   -- trailing newline → all complete lines
  483                   let lines' = filter (not . BS.null) (init ls)
  484                   mOut <- processLines st' lines'
  485                   case mOut of
  486                     Just outcome -> pure (Right outcome)
  487                     Nothing -> go BS.empty st'
  488                 else do
  489                   -- no trailing newline → last element is a partial line
  490                   let lines' = filter (not . BS.null) (init ls)
  491                       partial = last ls
  492                   mOut <- processLines st' lines'
  493                   case mOut of
  494                     Just outcome -> pure (Right outcome)
  495                     Nothing -> go partial st'
  496     -- Decode a batch of complete lines, threading state. Returns Just
  497     -- outcome if a StreamDone was encountered (or the callback aborted),
  498     -- Nothing to continue reading.
  499     processLines _ [] = pure Nothing
  500     processLines st' (l:ls) =
  501       case eitherDecodeStrict l of
  502         Left _ -> processLines st' ls  -- skip malformed line
  503         Right v -> case decodeStreamChunk st' v of
  504           Left _ -> processLines st' ls
  505           Right (st'', events) -> do
  506             mOut <- emitEvents events
  507             case mOut of
  508               Just outcome -> pure (Just outcome)
  509               Nothing -> processLines st'' ls
  510     -- Emit events to the callback; return Just outcome if StreamDone or
  511     -- callback returned False (abort → treat as clean stop).
  512     emitEvents [] = pure Nothing
  513     emitEvents (e:es) = do
  514       continue <- k e
  515       if not continue
  516         then pure (Just (StreamOutcome StopEnd (Usage 0 0)))
  517         else case e of
  518           StreamDone stop usage -> pure (Just (StreamOutcome stop usage))
  519           _ -> emitEvents es
  520     -- Process a raw buffer (used for the final flush when the stream ends
  521     -- without a newline). Returns (state, events).
  522     processBuffer buf st' =
  523       case eitherDecodeStrict buf of
  524         Left _ -> pure (st', [])
  525         Right v -> case decodeStreamChunk st' v of
  526           Left _ -> pure (st', [])
  527           Right r -> pure r

  529 -- | Run @k@ with request headers built from the optional key; the key bytes
  530 -- live only inside the 'withApiKey' continuation.
  531 withHeaders :: Ollama -> (RequestHeaders -> IO r) -> IO r
  532 withHeaders o k = case olApiKey o of
  533   Nothing  -> k (ollamaHeaders Nothing)
  534   Just key -> withApiKey key (k . ollamaHeaders . Just)

   10 -- | The subcommand selected on the command line. 'CommandNoOp' is the
   11 -- harmless placeholder carried by 'defaultConfig'; it is not exposed as a
   12 -- subcommand (the subcommands are @tui@, @signal@, and @serve@) and is
   13 -- excluded from the config-file 'FromJSON'/'ToJSON' instances.
   14 -- Each subcommand carries the autonomy level selected via @--yolo@
   15 -- ('Full' — bypass the human-confirmation gate) vs the default
   16 -- ('Supervised' — prompt before running any Untrusted opcode).
   17 data Command
   18   = CommandNoOp
   19   | CommandTui AutonomyLevel
   20   | CommandSignal AutonomyLevel
   21   | CommandTelegram AutonomyLevel
   22   | CommandServe AutonomyLevel
   23   | CommandGenOpenApi
   24   deriving (Eq, Show)
   25 
   26 pCommand :: Parser Command
   27 pCommand = hsubparser
   28   $  command "tui" (info (CommandTui <$> pAutonomy)
   29                          (progDesc "Start the interactive terminal UI (TUI)"))
   30   <> command "signal" (info (CommandSignal <$> pAutonomy)
   31                              (progDesc "Run the agent over the Signal channel"))
   32   <> command "telegram" (info (CommandTelegram <$> pAutonomy)
   33                                (progDesc "Run the agent over the Telegram channel"))
   34   <> command "serve" (info (CommandServe <$> pAutonomy)
   35                             (progDesc "Run the web gateway server"))
   36   <> command "gen-openapi" (info (pure CommandGenOpenApi)
   37                                   (progDesc "Print the OpenAPI 3.1 spec (JSON) to stdout"))
   38 
   39 -- | @--yolo@ sets 'Full' autonomy (bypass the untrusted-opcode approval
   40 -- gate); absent defaults to 'Supervised'.
   41 pAutonomy :: Parser AutonomyLevel
   42 pAutonomy = flag Supervised Full
   43   ( long "yolo"
   44   <> help "Bypass the approval gate so untrusted opcodes run without prompting (ACK audit still recorded)"
   45   )

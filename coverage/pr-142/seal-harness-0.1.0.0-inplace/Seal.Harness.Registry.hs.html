    4 -- | The harness registry — the ground truth a tab's 'BoundHarness'
    5 -- references. STM-backed, keyed by 'HarnessId', with race-safe CRUD and a
    6 -- 'mergeReconcile' that merges observed harnesses into entries **by key
    7 -- inside one STM transaction** so concurrent inserts are never clobbered
    8 -- (the lost-update-safe path).
    9 module Seal.Harness.Registry
   10   ( HarnessOrigin (..)
   11   , Liveness (..)
   12   , HarnessEntry (..)
   13   , HarnessRegistry (..)
   14   , ObservedHarness (..)
   15   , newHarnessRegistry
   16   , insert
   17   , lookupById
   18   , lookupByLabel
   19   , modify
   20   , delete
   21   , snapshot
   22   , mergeReconcile
   23   ) where

   25 import Control.Applicative ((<|>))

   36 -- | How a harness came to be in the registry.
   37 data HarnessOrigin = HoSpawned | HoDiscovered | HoAdopted
   38   deriving stock (Eq, Show, Generic)
   39   deriving anyclass (ToJSON, FromJSON)
   40 
   41 -- | The liveness states a harness can be in (mirrors 'HarnessStatus' but
   42 -- as a pure registry-cache value, not the live IO query).
   43 data Liveness = LvIdle | LvThinking | LvAwaitingInput | LvExited | LvOrphaned
   44   deriving stock (Eq, Show, Generic)
   45   deriving anyclass (ToJSON, FromJSON)
   46 
   47 -- | One registry entry: the durable identity + the reconciled
   48 -- coordinate/health cache.
   49 data HarnessEntry = HarnessEntry
   50   { heId          :: HarnessId
   51   , heLabel       :: Text              -- ^ human-facing label (mutable)
   52   , heOrigin      :: HarnessOrigin
   53   , heLiveness    :: Liveness
   54   , heTmuxCoord   :: Maybe Text        -- ^ "session:window.pane" when backed by tmux
   55   , heFlavour     :: Maybe Text        -- ^ the harness flavour (claude-code, codex, …)
   56   , heOrphanTicks :: Int               -- ^ consecutive Orphaned ticks (for grace eviction)
   57   } deriving stock (Eq, Show, Generic)
   58   deriving anyclass (ToJSON, FromJSON)
   59 
   60 -- | The observed-harness shape the reconcile loop produces (pure data; the
   61 -- observer fills it from a screen capture + the tmux markers).
   62 data ObservedHarness = ObservedHarness
   63   { ohId        :: HarnessId
   64   , ohLiveness  :: Liveness
   65   , ohTmuxCoord :: Maybe Text
   66   , ohFlavour   :: Maybe Text
   67   } deriving stock (Eq, Show)
   68 
   69 -- | The STM-backed registry. The 'TVar' holds a map keyed by 'HarnessId'.
   70 newtype HarnessRegistry = HarnessRegistry (TVar (Map HarnessId HarnessEntry))
   71 
   72 newHarnessRegistry :: IO HarnessRegistry
   73 newHarnessRegistry = HarnessRegistry <$> newTVarIO Map.empty
   74 
   75 insert :: HarnessRegistry -> HarnessEntry -> STM ()
   76 insert (HarnessRegistry tv) e = writeTVar tv . Map.insert (heId e) e =<< readTVar tv
   77 
   78 lookupById :: HarnessRegistry -> HarnessId -> STM (Maybe HarnessEntry)
   79 lookupById (HarnessRegistry tv) hid = Map.lookup hid <$> readTVar tv
   80 
   81 lookupByLabel :: HarnessRegistry -> Text -> STM (Maybe HarnessEntry)
   82 lookupByLabel (HarnessRegistry tv) label = do
   83   es <- Map.elems <$> readTVar tv
   84   pure (es ? label)
   85   where
   86     es ? lbl = case filter (\e -> heLabel e == lbl) es of
   87       (e:_) -> Just e
   88       []    -> Nothing
   89 
   90 modify :: HarnessRegistry -> HarnessId -> (HarnessEntry -> HarnessEntry) -> STM ()
   91 modify (HarnessRegistry tv) hid f = do
   92   m <- readTVar tv
   93   case Map.lookup hid m of
   94     Just e  -> writeTVar tv (Map.insert hid (f e) m)
   95     Nothing -> pure ()  -- no-op when absent
   96 
   97 delete :: HarnessRegistry -> HarnessId -> STM ()
   98 delete (HarnessRegistry tv) hid = writeTVar tv . Map.delete hid =<< readTVar tv
   99 
  100 -- | All entries, sorted by id (deterministic snapshot order).
  101 snapshot :: HarnessRegistry -> IO [HarnessEntry]
  102 snapshot (HarnessRegistry tv) =
  103   sortOn heId . Map.elems <$> readTVarIO tv
  104 
  105 -- | Merge a list of observed harnesses (from the reconcile sweep) into the
  106 -- registry **inside one STM transaction**, so concurrent inserts never
  107 -- clobber. Observed entries are merged by key: an existing entry keeps its
  108 -- origin + label (a 'HoSpawned' stays 'HoSpawned', not overwritten to
  109 -- 'HoDiscovered'), 'heLiveness' is updated, 'heOrphanTicks' is reset to 0
  110 -- on a non-Orphan observation or incremented on an Orphan one. A new
  111 -- observed id is inserted as 'HoDiscovered' with 'heOrphanTicks = 0'.
  112 -- Returns the new entries (the merged + inserted ones).
  113 mergeReconcile :: HarnessRegistry -> [ObservedHarness] -> STM [HarnessEntry]
  114 mergeReconcile (HarnessRegistry tv) observed = do
  115   m0 <- readTVar tv
  116   let m1 = foldl applyObs m0 observed
  117   writeTVar tv m1
  118   pure [ e | e <- Map.elems m1, heId e `elem` map ohId observed ]
  119   where
  120     applyObs m oh =
  121       let hid = ohId oh
  122       in case Map.lookup hid m of
  123            Just e ->
  124              let ticks = if ohLiveness oh == LvOrphaned
  125                            then heOrphanTicks e + 1
  126                            else 0
  127                  e' = e { heLiveness    = ohLiveness oh
  128                         , heTmuxCoord   = ohTmuxCoord oh
  129                         , heFlavour     = ohFlavour oh <|> heFlavour e
  130                         , heOrphanTicks = ticks
  131                         }
  132              in Map.insert hid e' m
  133            Nothing ->
  134              Map.insert hid HarnessEntry
  135                { heId          = hid
  136                , heLabel       = "discovered"  -- placeholder; the caller may rename
  137                , heOrigin      = HoDiscovered
  138                , heLiveness    = ohLiveness oh
  139                , heTmuxCoord   = ohTmuxCoord oh
  140                , heFlavour     = ohFlavour oh
  141                , heOrphanTicks = 0
  142                } m

   33 -- | The live tab-list handle. Backed by a 'TVar' so concurrent tab commands
   34 -- are race-safe (each operation is one STM transaction). The optional
   35 -- @thSave@ action is invoked after a successful mutation so the tab list
   36 -- persists to disk (W5); it's 'Nothing' for a plain 'newTabsHandle' and
   37 -- @Just (saveTabList path h)@ for 'newPersistingTabsHandle'.
   38 data TabsHandle = TabsHandle
   39   { thVar  :: TVar TabList
   40   , thSave :: Maybe (IO ())
   41   }
   42 
   43 newTabsHandle :: IO TabsHandle
   44 newTabsHandle = TabsHandle <$> newTVarIO emptyTabList <*> pure Nothing
   45 
   46 -- | Create a 'TabsHandle' that persists every mutation to @path@ (atomic
   47 -- write, 0600, MVar-serialized — see 'Seal.Tabs.Persist.saveTabList'). Used
   48 -- by 'Seal.Command.Serve.runServeMain' so the tab list survives a restart.
   49 newPersistingTabsHandle :: FilePath -> IO TabsHandle
   50 newPersistingTabsHandle path = do
   51   tv <- newTVarIO emptyTabList
   52   let h = TabsHandle { thVar = tv, thSave = Just (snapshotTabs h >>= saveTabList path) }
   53   pure h
   54 
   55 snapshotTabs :: TabsHandle -> IO TabList
   56 snapshotTabs h = readTVarIO (thVar h)
   57 
   58 -- | Replace the handle's 'TabList' in one STM transaction. Used at boot to
   59 -- seed the in-memory handle from the persisted @tabs.json@ (after the
   60 -- caller has dropped stale tabs + renumbered). Does NOT persist (the caller
   61 -- is loading FROM disk — writing back would be a redundant no-op).
   62 seedTabsHandle :: TabsHandle -> TabList -> IO ()
   63 seedTabsHandle h tl = atomically (writeTVar (thVar h) tl)
   64 
   65 -- | Insert a tab at the lowest free slot. 'Right' the new index; 'Left' on
   66 -- I2 (duplicate ref) or full. Persists via 'thSave' after a successful STM
   67 -- commit (W5).
   68 insertTabH :: TabsHandle -> TabRef -> TabKind -> Maybe Text -> IO (Either Text TabIndex)
   69 insertTabH h ref kind label = do
   70   r <- atomically $ do
   71     let tv = thVar h
   72     tl <- readTVar tv
   73     case insertTab ref kind label tl of
   74       Left e        -> pure (Left e)
   75       Right tl' -> do
   76         writeTVar tv tl'
   77         case tlTabs tl' of
   78           []     -> pure (Left "insert succeeded but list is empty (unreachable)")
   79           (t:_)  -> pure (Right (tIndex t))
   80   persistIf h r
   81 
   82 -- | Remove a tab (compacts; I1). 'Left' if out of range. Persists on success.
   83 removeTabH :: TabsHandle -> TabIndex -> IO (Either Text ())
   84 removeTabH h idx = do
   85   r <- atomically $ do
   86     let tv = thVar h
   87     tl <- readTVar tv
   88     case removeTab tl idx of
   89       Left e       -> pure (Left e)
   90       Right tl' -> writeTVar tv tl' >> pure (Right ())
   91   persistIf h r
   92 
   93 -- | Rename a tab. 'Left' if the index is out of range. Persists on success.
   94 renameTabH :: TabsHandle -> TabIndex -> Text -> IO (Either Text ())
   95 renameTabH h idx name = do
   96   r <- atomically $ do
   97     let tv = thVar h
   98     tl <- readTVar tv
   99     case renameTab tl idx name of
  100       Left e       -> pure (Left e)
  101       Right tl' -> writeTVar tv tl' >> pure (Right ())
  102   persistIf h r
  103 
  104 -- | Rebind a tab's 'TabRef' in place (preserves index/kind/label/status;
  105 -- used by @\/new@). 'Left' on out-of-range index or I2 violation (the new
  106 -- ref is already bound to a different tab). Rebind-to-same-ref is a no-op
  107 -- 'Right'. Single STM transaction — race-safe vs concurrent tab ops.
  108 -- Persists on success (W5).
  109 rebindTabH :: TabsHandle -> TabIndex -> TabRef -> IO (Either Text ())
  110 rebindTabH h idx newRef = do
  111   r <- atomically $ do
  112     let tv = thVar h
  113     tl <- readTVar tv
  114     case rebindTab idx newRef tl of
  115       Left e       -> pure (Left e)
  116       Right tl' -> writeTVar tv tl' >> pure (Right ())
  117   persistIf h r
  118 
  119 -- | Focus a tab: validates the index is in range. 'Right ()' on success;
  120 -- 'Left' if out of range. (The actual cursor state is per-conversation and
  121 -- tracked by the relay/wiring; this just validates the index.) No persist
  122 -- (focus is a pure validation — no state change).
  123 focusTabH :: TabsHandle -> TabIndex -> IO (Either Text ())
  124 focusTabH h idx = atomically $ do
  125   let tv = thVar h
  126   tl <- readTVar tv
  127   if tabCount tl > 0 && idx `elem` map tIndex (tlTabs tl)
  128     then pure (Right ())
  129     else pure (Left "tab index out of range")
  130 
  131 -- | Run the persist action (if any) after a successful mutation, returning
  132 -- the original result. A 'Left' outcome (mutation rejected) is a no-op for
  133 -- the save. A save failure is logged to stderr (ids + error only — no
  134 -- session content) and swallowed — the in-memory handle stays authoritative
  135 -- within the session; the next successful mutation will retry the save
  136 -- (writing the full current list, so a missed save self-heals).
  137 persistIf :: TabsHandle -> Either a b -> IO (Either a b)
  138 persistIf h r = do
  139   case r of
  140     Left _  -> pure ()
  141     Right _ -> case thSave h of
  142       Nothing  -> pure ()
  143       Just act -> act `catch` \e -> globalLogIO WarningS ("[persist] tabs.json save failed: " <> ls (T.pack (show (e :: SomeException))))
  144   pure r

  146 -- | Idempotent: if no tab binds @sid@, insert a @'BoundSession' sid@ tab of
  147 -- the given kind at the lowest free slot. Sources the 'SessionId' only from
  148 -- server-validated contexts (the caller passes @smId meta@ from a
  149 -- 'SessionMeta' loaded by 'loadSessionMeta' / minted by 'newSession' — never
  150 -- a raw client string). Failure (@Left "tab list full (36 slots)"@ or
  151 -- @Left "tab ref already bound"@ from a concurrent insert) is logged to
  152 -- stderr (ids + error only — no session content) and does NOT propagate;
  153 -- the tab is a UI affordance, not a correctness requirement. Shared by the
  154 -- web 'handleSend' (W2) and the channel/CLI 'plainTurn' paths (W3).
  155 ensureTabForSession :: TabsHandle -> TabKind -> SessionId -> IO ()
  156 ensureTabForSession tabsH kind sid = do
  157   tl <- snapshotTabs tabsH
  158   let alreadyBound = any (\t -> tRef t == BoundSession sid) (tlTabs tl)
  159   if alreadyBound
  160     then pure ()
  161     else do
  162       r <- insertTabH tabsH (BoundSession sid) kind Nothing
  163       case r of
  164         Left e -> globalLogIO WarningS ("[auto-tab] could not insert tab for " <> ls (sessionIdText sid) <> ": " <> ls e)
  165         Right _ -> pure ()

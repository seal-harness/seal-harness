    2 -- | Validated argv newtypes (Invariant 2 — type-guaranteed argument
    3 -- sanitization). Every value derived from user/LLM input that reaches a
    4 -- subprocess argv must be carried by one of these validated,
    5 -- smart-constructed newtypes — never raw 'Text'/'String'. The exec
    6 -- wrappers accept ONLY these types in their signatures, so passing
    7 -- unsanitized input fails to compile.
    8 --
    9 -- Defense against option injection is built into every smart constructor:
   10 -- reject leading-dash values and always pass a leading @--@ separator
   11 -- before user-derived arguments at the call site. Reject NUL, newlines
   12 -- (where they would break argv token boundaries), and control chars.

   21   , mkBinName

   35 -- | A single argv token (a program argument). Rejects leading-dash (option
   36 -- injection), NUL, newlines, and control chars.
   37 newtype ShellArg = ShellArg Text
   38   deriving stock (Eq, Show)
   39 
   40 mkShellArg :: Text -> Either Text ShellArg
   41 mkShellArg t
   42   | T.null t            = Left "shell arg is empty"
   43   | T.head t == '-'     = Left "shell arg must not start with '-' (option injection)"
   44   | T.any isArgBad t    = Left "shell arg contains NUL, newline, or control char"
   45   | otherwise           = Right (ShellArg t)
   46   where
   47     isArgBad c = c == '\0' || c == '\n' || isControl c
   48 
   49 textShellArg :: ShellArg -> Text
   50 textShellArg (ShellArg t) = t

   52 -- | A shell command string — the single argument passed to @\/bin\/sh -c@.
   53 -- Rejects NUL (which would terminate the C-string early). Does NOT reject
   54 -- leading dash (a command may begin with @-@; the executor passes it as a
   55 -- single arg to @-c@, not as a flag). Does NOT reject newlines (a shell
   56 -- command legitimately spans lines).
   57 newtype ShellCommand = ShellCommand Text
   58   deriving stock (Eq, Show)
   59 
   60 mkShellCommand :: Text -> Either Text ShellCommand
   61 mkShellCommand t
   62   | T.null t            = Left "shell command is empty"
   63   | T.any (== '\0') t   = Left "shell command contains NUL"
   64   | otherwise           = Right (ShellCommand t)
   65 
   66 textShellCommand :: ShellCommand -> Text
   67 textShellCommand (ShellCommand t) = t

   69 -- | A binary name (e.g. @python3@, @node@, @ls@). The executor resolves
   70 -- it on PATH via 'System.Process.proc' (RawCommand — no shell), so the
   71 -- value never passes through a shell. Minimal validation: reject empty
   72 -- and NUL (which would truncate the C-string). Leading dashes are
   73 -- permitted (a binary *may* legitimately begin with @-@), and the
   74 -- argv-passing model means leading-dash args are passed as argv tokens,
   75 -- not flags to a shell. Path separators are permitted (the binary may be
   76 -- an absolute or relative path). Control chars other than NUL are
   77 -- permitted (the executor passes them verbatim through argv).
   78 newtype BinName = BinName Text
   79   deriving stock (Eq, Show)
   80 
   81 mkBinName :: Text -> Either Text BinName
   82 mkBinName t
   83   | T.null t          = Left "binary name is empty"
   84   | T.any (== '\0') t = Left "binary name contains NUL"
   85   | otherwise         = Right (BinName t)
   86 
   87 textBinName :: BinName -> Text
   88 textBinName (BinName t) = t

   90 -- | A single argv token passed to a binary. The executor uses
   91 -- 'System.Process.proc' (RawCommand), so the value is never interpreted
   92 -- by a shell — it is passed verbatim as one argv entry. Minimal
   93 -- validation: reject empty and NUL. Leading dashes are permitted (a
   94 -- leading-dash arg is a flag for the binary, not option injection — the
   95 -- caller explicitly chose to pass it).
   96 newtype BinArg = BinArg Text
   97   deriving stock (Eq, Show)
   98 
   99 mkBinArg :: Text -> Either Text BinArg
  100 mkBinArg t
  101   | T.null t          = Left "bin arg is empty"
  102   | T.any (== '\0') t = Left "bin arg contains NUL"
  103   | otherwise         = Right (BinArg t)
  104 
  105 textBinArg :: BinArg -> Text
  106 textBinArg (BinArg t) = t

  112 -- 'UntrustedIO' search methods accept; the caller never passes raw 'Text'.

  122   where
  123     isArgBad c = c == '\0' || c == '\n' || isControl c

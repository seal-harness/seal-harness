    2 -- | Pure reconstruction from the new two-file format
    3 -- (@conversation.jsonl@ + @entries.jsonl@) back to the old
    4 -- 'TranscriptEntry' stream. The reconstructed entries are byte-identical to
    5 -- what the old @transcript.jsonl@ format would have stored, modulo the
    6 -- intentional uncapping improvement (the new format stores full messages
    7 -- where the old format may have size-capped payloads).
    8 --
    9 -- A 'Request' entry @i@ is reconstructed with:
   10 --
   11 -- * @messages = conversation[0 .. convLen_i)@ (the prefix of conversation lines
   12 --   in effect at that turn);
   13 -- * the left-folded effective envelope at @i@ (model / system / tools /
   14 --   toolChoice / maxTokens), wrapped into a 'CompletionRequest'-shaped
   15 --   'Value';
   16 -- * the same direction + meta as the stored entry.
   17 --
   18 -- A 'Response' entry is reconstructed by re-inserting the assistant content
   19 -- blocks from @conversation.jsonl@ (the lines between the prior request's
   20 -- @convLen@ and this response's @convLen@) into the stored response envelope.
   21 --
   22 -- 'EKHarness' and 'EKCompaction' events carry no envelope; they round-trip
   23 -- their metadata + a 'Request'/'Response' direction derived from their kind
   24 -- (a harness turn is a 'Request' from the user side; a compaction is a
   25 -- boundary marker with 'Request' direction).
   26 module Seal.Transcript.Reconstruct
   27   ( reconstruct
   28   ) where
   29

   40 -- | Reconstruct the old 'TranscriptEntry' stream from the two new files.
   41 --
   42 -- The input is the full conversation (the @conversation.jsonl@ contents decoded
   43 -- into a @[Message]@) and the full entry stream (the @entries.jsonl@ contents
   44 -- decoded into an @[EntryRecord]@). The output is one 'TranscriptEntry' per
   45 -- 'EntryRecord', in order, with payloads rebuilt from the conversation prefix
   46 -- and the effective envelope.
   47 --
   48 -- The first request's envelope is the initial envelope passed in (there is no
   49 -- prior to delta against, so the first request's full envelope is the
   50 -- baseline). Subsequent requests fold deltas on top.
   51 --
   52 -- Both 'EKRequest' and 'EKResponse' entries carry ONLY the NEW messages added
   53 -- since the prior turn (@conv[start:end]@), NOT the cumulative conversation
   54 -- prefix. This is the whole point of the two-file delta format: the on-disk
   55 -- @conversation.jsonl@ already stores each message exactly once, and
   56 -- re-embedding the full history into every request entry would be O(N²) in
   57 -- the conversation length (a 146-turn session would ship ~5,000 redundant
   58 -- message copies, turning 280KB on disk into ~18MB on the wire). The frontend
   59 -- extracts only the last message of each request for display, so the chat
   60 -- view is unaffected; the "View raw JSON" modal shows the delta payload (the
   61 -- new messages + the effective envelope), which is what the user actually
   62 -- wants to inspect when debugging a single turn.
   63 reconstruct :: [Message] -> [EntryRecord] -> [TranscriptEntry]
   64 reconstruct conv = go 0 Nothing
   65   where
   66     -- @start@ is the conversation-line index where the current turn's new
   67     -- messages begin (i.e. the prior response's convLen, or 0 for the first
   68     -- turn). @mEnv@ is the effective envelope at the most recent request (used
   69     -- to reconstruct response payloads, which carry the request's model).
   70     go :: Int -> Maybe Envelope -> [EntryRecord] -> [TranscriptEntry]
   71     go _ _       [] = []
   72     go start mEnv (e : es) =
   73       case erKind e of
   74         EKRequest ->
   75           let env = effectiveAt e mEnv
   76               end = erConvLen e
   77               -- Delta-only: the NEW messages added since the prior turn
   78               -- (conv[start:end]), NOT the cumulative prefix.
   79               msgs = take (end - start) (drop start conv)
   80               -- Only include envelope fields when they CHANGED from the
   81               -- prior effective envelope. The frontend deduplicates System
   82               -- Prompt and Tools rows independently, so shipping the same
   83               -- 36KB system prompt + 14KB tools array in all 73 request
   84               -- entries of a 146-turn session wastes ~3.6MB. Omitting
   85               -- unchanged fields lets the frontend skip the rows for those
   86               -- entries (it only renders the first occurrence of each
   87               -- unique system prompt / tools pair).
   88               sys = if envSystem env /= (envSystem =<< mEnv)
   89                       then envSystem env
   90                       else Nothing
   91               tools = if envTools env /= maybe [] envTools mEnv
   92                         then Just (envTools env)
   93                         else Nothing
   94               payload = requestPayload env sys tools msgs
   95               entry = toEntry e Request payload
   96           in entry : go end (Just env) es
   97         EKResponse ->
   98           let end = erConvLen e
   99               msgs = take (end - start) (drop start conv)
  100               payload = responsePayload mEnv msgs e
  101               entry = toEntry e Response payload
  102           in entry : go end mEnv es
  103         EKHarness ->
  104           -- A harness entry (opcode invocation) adds no conversation lines
  105           -- (erConvLen = 0), so the conversation cursor is preserved — the
  106           -- next response entry must slice from the same @start@, not from 0.
  107           -- Resetting the cursor here would make the next response's
  108           -- @take end (drop 0 conv)@ return the ENTIRE conversation, which
  109           -- the frontend renders as one giant duplicate response row.
  110           let payload = harnessPayload (take (erConvLen e) (drop start conv)) e
  111               entry = toEntry e Request payload
  112           in entry : go start mEnv es
  113         EKCompaction ->
  114           let entry = toEntry e Request Null
  115           in entry : go (erConvLen e) mEnv es
  116 
  117     -- The effective envelope at a request entry. The first request has no
  118     -- prior envelope, so its delta is folded against a default baseline; in
  119     -- practice the writer always emits a full envelope on the first request,
  120     -- so the delta carries every field. We fall back to a sensible default
  121     -- only if the delta is absent (a malformed entry).
  122     effectiveAt :: EntryRecord -> Maybe Envelope -> Envelope
  123     effectiveAt e mEnv =
  124       let baseline = fromMaybe defaultEnv mEnv
  125       in case erEnvelope e of
  126            Nothing -> baseline
  127            Just d  -> applyDelta baseline d
  128 
  129     defaultEnv :: Envelope
  130     defaultEnv = Envelope (ModelId "") Nothing [] ToolAuto 0

  132 -- | Build the old 'Request' payload: a 'CompletionRequest'-shaped JSON object
  133 -- carrying the model, system, tools, toolChoice, maxTokens, and the message
  134 -- prefix in effect at this turn. The @sys@ and @tools@ arguments are the
  135 -- system prompt and tools to include — 'Nothing' means "omit the field"
  136 -- (used when the value is unchanged from the prior request, so the frontend
  137 -- skips the redundant row). Mirrors the old 'ToJSON CompletionRequest'
  138 -- encoding so a "view raw" reproduces the bytes the old format stored.
  139 requestPayload :: Envelope -> Maybe Text -> Maybe [ToolDefinition] -> [Message] -> Value
  140 requestPayload env sys mTools msgs = object $
  141   [ "model"      .= envModel env
  142   , "messages"   .= msgs
  143   ] <> toolsField
  144     <> [ "toolChoice" .= envToolChoice env
  145        , "maxTokens"  .= envMaxTokens env
  146        ] <> systemField
  147   where
  148     systemField = case sys of
  149       Nothing -> []  -- unchanged from prior; omit so frontend skips
  150       Just s  -> ["system" .= s]
  151     toolsField = case mTools of
  152       Nothing -> []  -- unchanged from prior; omit so frontend skips
  153       Just ts -> ["tools" .= ts]
  154 
  155 -- | Build the old 'Response' payload: the assistant content blocks (drawn
  156 -- from the conversation lines added since the prior turn) plus the usage /
  157 -- stop / duration metadata stored on the entry.
  158 responsePayload :: Maybe Envelope -> [Message] -> EntryRecord -> Value
  159 responsePayload mEnv msgs e = object
  160   [ "model"    .= (envModel <$> mEnv)
  161   , "content"  .= concatMap msgContent msgs
  162   , "usage"    .= erUsage e
  163   , "stop"     .= erStop e
  164   , "durationMs" .= erDurationMs e
  165   ]

  167 -- | A harness turn's payload: the conversation lines added since the prior
  168 -- turn (the harness's input/output as messages), plus the harness name and
  169 -- the opcode metadata. The @op@ field is always included in the base
  170 -- payload (v1: so the frontend filter 'Seal.Gateway.Transcript.reconEntryToFrontend'
  171 -- can whitelist @op.name == "SKILL_LOAD"@ and surface /skill load
  172 -- invocations as distinct harness entries). The @input@, @result@, and
  173 -- @channel@ fields are included when present in @erMeta@ (the 'SKILL_LOAD'
  174 -- result entry recorded by 'Seal.ISA.Dispatch.recordSkillLoadResult' carries
  175 -- input + result + the originating channel label, so the frontend can render
  176 -- the skill body in a collapsible tool-call box AND surface channel
  177 -- provenance in the source label). When the entry carries approval metadata
  178 -- (erMeta has an "approval" key), the payload also includes the approval
  179 -- info so the frontend can render the confirmation decision.
  180 harnessPayload :: [Message] -> EntryRecord -> Value
  181 harnessPayload msgs e =
  182   let base = object
  183         [ "messages" .= msgs
  184         , "harness"  .= erHarness e
  185         , "op"       .= (Map.lookup "op" (erMeta e) :: Maybe Value)
  186         , "input"    .= (Map.lookup "input" (erMeta e) :: Maybe Value)
  187         , "result"   .= (Map.lookup "result" (erMeta e) :: Maybe Value)
  188         , "channel"  .= (Map.lookup "channel" (erMeta e) :: Maybe Value)
  189         ]
  190   in case Map.lookup "approval" (erMeta e) of
  191        Nothing -> base
  192        Just approvalVal -> object
  193          [ "messages" .= msgs
  194          , "harness"  .= erHarness e
  195          , "op"       .= (Map.lookup "op" (erMeta e) :: Maybe Value)
  196          , "input"    .= (Map.lookup "input" (erMeta e) :: Maybe Value)
  197          , "result"   .= (Map.lookup "result" (erMeta e) :: Maybe Value)
  198          , "channel"  .= (Map.lookup "channel" (erMeta e) :: Maybe Value)
  199          , "approval" .= approvalVal
  200          ]

  202 -- | Lift an 'EntryRecord' into a 'TranscriptEntry' with the given direction
  203 -- and payload. The id, timestamp, duration, correlation, and meta are
  204 -- carried over; the model is 'Nothing' (the dispatcher's opcode-invocation
  205 -- entries set teModel = Nothing; the loop's provider entries set it from the
  206 -- envelope — but the reconstructed old-format entry keeps teModel = Nothing
  207 -- to match the existing dispatch path; the response payload carries the
  208 -- model separately).
  209 toEntry :: EntryRecord -> Direction -> Value -> TranscriptEntry
  210 toEntry e dir payload = TranscriptEntry
  211   { teId = erId e
  212   , teTimestamp = erTimestamp e
  213   , teModel = Nothing
  214   , teDirection = dir
  215   , tePayload = payload
  216   , teDurationMs = erDurationMs e
  217   , teCorrelation = erCorrelation e
  218   , teMeta = erMeta e
  219   }

    2 -- | Per-session concurrency control + reply fan-out.
    3 --
    4 -- Two concerns are handled here:
    5 --
    6 -- 1. **Write lock**: a per-session 'MVar' serializes turns on the same
    7 --    session. Both the channel 'plainTurn' and the web 'plainTurn' acquire
    8 --    the session's lock before calling 'withTwoFileTranscript', preventing
    9 --    concurrent writes from corrupting the two-file transcript format.
   10 --    Different sessions run concurrently; only turns on the SAME session
   11 --    queue.
   12 --
   13 -- 2. **Reply fan-out**: a registry of 'ChannelHandle's subscribed to a
   14 --    session. When a turn completes (from any source — web, Telegram,
   15 --    Signal), the reply is sent to every registered channel handle for
   16 --    that session via 'chSend'. The web frontend already receives entries
   17 --    via the WS broker ('BeEntryRecorded'); the fan-out handles chat
   18 --    channels, which need an explicit 'chSend' to deliver the reply to
   19 --    the peer.
   20 module Seal.Session.Lock
   21   ( SessionLocks
   22   , newSessionLocks
   23   , withSessionLock
   24   , sessionTurnInFlight
   25   , ReplyRegistry
   26   , newReplyRegistry
   27   , replySubscribe
   28   , replyUnsubscribe
   29   , replyFanout
   30   , replyFanoutMessage
   31   , replySubscriberCount
   32   , replyMigrateAll
   33   ) where

   52 -- ---------------------------------------------------------------------------
   53 -- Write locks
   54 -- ---------------------------------------------------------------------------
   55 
   56 -- | The per-session lock map. Each session gets its own 'MVar' on first
   57 -- use; subsequent turns on the same session acquire that 'MVar'.
   58 newtype SessionLocks = SessionLocks (TVar (Map SessionId (MVar ())))
   59 
   60 -- | Build a new empty lock map.
   61 newSessionLocks :: IO SessionLocks
   62 newSessionLocks = SessionLocks <$> newTVarIO Map.empty
   63 
   64 -- | Acquire the session's lock, run the action, release. Creates the
   65 -- 'MVar' on first use. Different sessions run concurrently; only turns on
   66 -- the SAME session serialize.
   67 withSessionLock :: SessionLocks -> SessionId -> IO a -> IO a
   68 withSessionLock (SessionLocks tv) sid action = do
   69   lock <- do
   70     m <- readTVarIO tv
   71     case Map.lookup sid m of
   72       Just l  -> pure l
   73       Nothing -> do
   74         l <- newMVar ()
   75         atomically $ do
   76           m' <- readTVar tv
   77           case Map.lookup sid m' of
   78             Just existing -> pure existing
   79             Nothing -> do
   80               writeTVar tv (Map.insert sid l m')
   81               pure l
   82   withMVar lock (const action)
   83 
   84 -- | Non-blocking check: is a turn currently in flight for this session?
   85 -- Returns 'True' if the session's lock MVar is empty (held by a running
   86 -- turn via 'withSessionLock'); 'False' if full (no turn in flight) or the
   87 -- session has no lock yet (never started a turn). Used by the web
   88 -- @POST /api/sessions/:id/stop@ endpoint to report @pending@ (design
   89 -- Blocker Resolution #2 + Designer round-2 question). Uses
   90 -- 'tryReadMVar' (non-blocking): 'Just ()' = full = not held = not in
   91 -- flight; 'Nothing' = empty = held = in flight.
   92 sessionTurnInFlight :: SessionLocks -> SessionId -> IO Bool
   93 sessionTurnInFlight (SessionLocks tv) sid = do
   94   m <- readTVarIO tv
   95   case Map.lookup sid m of
   96     Nothing -> pure False  -- no lock yet → never started a turn → not in flight
   97     Just lock -> do
   98       mFilled <- tryReadMVar lock
   99       pure (isNothing mFilled)  -- Nothing = empty = held = in flight

  101 -- ---------------------------------------------------------------------------
  102 -- Reply fan-out
  103 -- ---------------------------------------------------------------------------
  104 
  105 -- | A reply sink wraps a 'ChannelHandle' and an 'IORef' guard so the
  106 -- same handle can be subscribed to at most one session at a time. The
  107 -- guard carries the 'SessionId' the handle is currently subscribed to;
  108 -- 'replyUnsubscribe' only removes it if the guard matches (preventing a
  109 -- race where the handle already re-subscribed to a different session).
  110 data ReplySink = ReplySink
  111   { rsHandle :: ChannelHandle
  112   , rsGuard  :: IORef (Maybe SessionId)
  113   , rsLabel  :: Text
  114   }
  115 
  116 -- | The per-session reply registry. Maps each 'SessionId' to the set of
  117 -- 'ReplySink's that should receive replies for that session. STM-backed
  118 -- so subscribe/unsubscribe/fan-out are race-safe. Multiple channels of
  119 -- DIFFERENT kinds can subscribe to the same session (e.g. a Telegram
  120 -- handle and a Signal handle both following the same tab); re-subscribing
  121 -- a channel of the same kind replaces the old handle for that kind.
  122 newtype ReplyRegistry = ReplyRegistry (TVar (Map SessionId [ReplySink]))
  123 
  124 -- | Build a new empty reply registry.
  125 newReplyRegistry :: IO ReplyRegistry
  126 newReplyRegistry = ReplyRegistry <$> newTVarIO Map.empty
  127 
  128 -- | Subscribe a 'ChannelHandle' to a session's replies. Subscribers
  129 -- accumulate per channel kind: a new subscription for the same
  130 -- @chLabel@ replaces the old handle for that kind (so re-subscribing the
  131 -- same Telegram conversation updates its handle without creating a
  132 -- duplicate), but a DIFFERENT channel kind (e.g. Signal) is preserved.
  133 -- This lets a session be followed by one handle per append-only channel
  134 -- simultaneously, enabling cross-channel message mirroring. The handle is
  135 -- associated with an 'IORef' guard (returned to the caller) so it can be
  136 -- later unsubscribed.
  137 replySubscribe
  138   :: ReplyRegistry -> ChannelHandle -> SessionId
  139   -> IO (IORef (Maybe SessionId))
  140 replySubscribe (ReplyRegistry tv) h sid = do
  141   guard <- newIORef (Just sid)
  142   let sink = ReplySink h guard (chLabel h)
  143   atomically $ do
  144     m <- readTVar tv
  145     let sinks = Map.findWithDefault [] sid m
  146         -- Drop any existing sink with the same channel label (replace
  147         -- same-kind), then prepend the new one.
  148         sinks' = sink : filter (\s -> rsLabel s /= rsLabel sink) sinks
  149     writeTVar tv (Map.insert sid sinks' m)
  150   pure guard
  151 
  152 -- | Unsubscribe a handle from a session. The 'IORef' guard must match
  153 -- the session id; if it doesn't (the handle already re-subscribed to a
  154 -- different session), this is a no-op. This prevents a stale unsubscribe
  155 -- from removing a handle that's now listening to a different session.
  156 replyUnsubscribe :: ReplyRegistry -> IORef (Maybe SessionId) -> SessionId -> IO ()
  157 replyUnsubscribe (ReplyRegistry tv) guard sid = do
  158   g <- readIORef guard
  159   case g of
  160     Just s | s == sid -> do
  161       atomically $ do
  162         m <- readTVar tv
  163         let m' = Map.adjust (filter (\sink -> rsGuard sink /= guard)) sid m
  164         writeTVar tv (Map.filter (not . null) m')
  165       writeIORef guard Nothing
  166     _ -> pure ()
  167 
  168 -- | Fan out a reply to every 'ChannelHandle' subscribed to a session.
  169 -- The reply text is sent via 'chSend' to each handle. The web frontend
  170 -- already receives entries via the WS broker, so it does not appear in
  171 -- this registry (only chat channels do). Errors are logged to stderr and
  172 -- swallowed (a dead channel handle should not prevent other handles from
  173 -- receiving the reply).
  174 replyFanout :: ReplyRegistry -> SessionId -> Text -> IO ()
  175 replyFanout (ReplyRegistry tv) sid text = do
  176   m <- readTVarIO tv
  177   for_ (Map.lookup sid m) $ \sinks ->
  178     mapM_ (\sink -> safeSend (rsHandle sink) text) sinks
  179   where
  180     safeSend h t = chSend h t `catch` \e ->
  181       globalLogIO WarningS ("reply fanout: send failed: " <> ls (T.pack (show (e :: IOException))))
  182 
  183 -- | Fan out a user message to every 'ChannelHandle' subscribed to a
  184 -- session EXCEPT the channel that sent the message (identified by its
  185 -- @excludeLabel@). Each receiving channel gets the message prefixed with
  186 -- the sender's channel label in brackets (e.g. @"[telegram] hi"@), so
  187 -- the user can see which channel the message originated from. This is
  188 -- the cross-channel message-mirroring primitive: every user message sent
  189 -- on any append-only channel is mirrored to all OTHER subscribed
  190 -- append-only channels. The web frontend is NOT in the registry (it
  191 -- sees the message directly via the transcript), so it is excluded by
  192 -- construction. Errors are logged and swallowed.
  193 replyFanoutMessage
  194   :: ReplyRegistry -> SessionId -> Text -> Text -> IO ()
  195 replyFanoutMessage (ReplyRegistry tv) sid senderLabel text = do
  196   m <- readTVarIO tv
  197   for_ (Map.lookup sid m) $ \sinks ->
  198     mapM_ (\sink ->
  199       if rsLabel sink == senderLabel
  200         then pure ()  -- don't mirror back to the sender
  201         else safeSend (rsHandle sink) ("[" <> senderLabel <> "] " <> text))
  202       sinks
  203   where
  204     safeSend h t = chSend h t `catch` \e ->
  205       globalLogIO WarningS ("message fanout: send failed: " <> ls (T.pack (show (e :: IOException))))

  207 -- | The number of channels currently subscribed to a session's replies.
  208 -- Used by the web send path to decide whether the last assistant reply
  209 -- counts as "seen" (delivered to ≥1 chat channel) for the sidebar's
  210 -- tab-status indicator.
  211 replySubscriberCount :: ReplyRegistry -> SessionId -> IO Int
  212 replySubscriberCount (ReplyRegistry tv) sid =
  213   maybe 0 length . Map.lookup sid <$> readTVarIO tv

  215 -- | Migrate every 'ReplySink' subscribed to @oldSid@ to @newSid@. Used by
  216 -- @\/new@ when a tab is rebound to a fresh session: the channel handles
  217 -- subscribed to the old session should follow the rebind so future
  218 -- fan-outs (including tab-close notifications) reach them under the new
  219 -- session id. Single STM transaction — race-safe vs concurrent
  220 -- subscribe/unsubscribe/fan-out. Returns the count of migrated sinks.
  221 replyMigrateAll :: ReplyRegistry -> SessionId -> SessionId -> IO Int
  222 replyMigrateAll (ReplyRegistry tv) oldSid newSid = atomically $ do
  223   m <- readTVar tv
  224   let (matched, rest) = Map.partitionWithKey (\k _ -> k == oldSid) m
  225       sinks = concat (Map.elems matched)
  226       m' = if null sinks
  227              then rest
  228              else Map.insertWith (++) newSid sinks (Map.delete oldSid rest)
  229   writeTVar tv m'
  230   pure (length sinks)

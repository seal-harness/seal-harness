    2 -- | Human-interaction opcodes (Trusted): SHOW_HUMAN emits a line to the user;
    3 -- ASK_HUMAN prompts and returns the reply. Both go through the channel handle —
    4 -- no shell, no provider.
    5 module Seal.ISA.Ops.Human
    6   ( showHumanOp
    7   , askHumanOp
    8   ) where

   10 import Control.Monad.IO.Class (liftIO)
   11 import Data.Aeson (Value (..), object, withObject, (.:), (.:?), (.=))
   12 import Data.Aeson.Key (Key, fromText)
   13 import Data.Aeson.Types (parseMaybe)
   14 import Data.Maybe (fromMaybe)
   15 import Data.Text (Text)

   17 import Seal.Channel.Caps

   23 -- | Extract a Text value from a JSON object by key.
   24 -- Uses 'Key' (aeson 2.x) so string literals work directly via OverloadedStrings.
   25 strField :: Key -> Value -> Maybe Text
   26 strField k = parseMaybe (withObject "in" (.: k))
   27 
   28 -- | Parse the optional @options@ array from an ASK_HUMAN input into a list
   29 -- of 'QuestionOption's. 'Nothing' when the field is absent (open-ended);
   30 -- 'Just opts' when present (including 'Just []', which 'validateOptions'
   31 -- rejects).
   32 parseOptions :: Value -> Maybe (Maybe [QuestionOption])
   33 parseOptions = parseMaybe $ withObject "in" $ \o -> do
   34   mOpts <- o .:? fromText "options"
   35   case mOpts of
   36     Nothing -> pure Nothing
   37     Just arr -> Just <$> traverse parseOption arr
   38   where
   39     parseOption = withObject "option" $ \oo ->
   40       QuestionOption <$> oo .: fromText "label"
   41                      <*> (fromMaybe "" <$> (oo .:? fromText "description"))
   42 
   43 -- | Build a JSON-Schema object with a single required string property.
   44 singleStringSchema :: Text -> Text -> Value
   45 singleStringSchema fieldName fieldDesc =
   46   object
   47     [ "type" .= ("object" :: Text)
   48     , "properties" .= object
   49         [fromText fieldName .= object
   50            [ "type" .= ("string" :: Text)
   51            , "description" .= fieldDesc
   52            ]]
   53     , "required" .= ([fieldName] :: [Text])
   54     ]
   55 
   56 -- | The JSON-Schema for ASK_HUMAN's input: a required @question@ string plus
   57 -- an optional @options@ array (max 8 items, each @{label, description}@).
   58 askHumanSchema :: Value
   59 askHumanSchema = object
   60   [ "type" .= ("object" :: Text)
   61   , "properties" .= object
   62       [ fromText "question" .= object
   63           [ "type" .= ("string" :: Text)
   64           , "description" .= ("The question to present to the human operator." :: Text)
   65           ]
   66       , fromText "options" .= object
   67           [ "type" .= ("array" :: Text)
   68           , "description" .= ("Optional discrete choices. When present, channels render one button per choice plus an 'Other' free-text option. When absent, the question is open-ended." :: Text)
   69           , "maxItems" .= (8 :: Int)
   70           , "items" .= object
   71               [ "type" .= ("object" :: Text)
   72               , "properties" .= object
   73                   [ fromText "label" .= object
   74                       [ "type" .= ("string" :: Text)
   75                       , "description" .= ("The value returned to the agent when picked (1-5 words, concise)." :: Text)
   76                       ]
   77                   , fromText "description" .= object
   78                       [ "type" .= ("string" :: Text)
   79                       , "description" .= ("One-line explanation of the choice." :: Text)
   80                       ]
   81                   ]
   82               , "required" .= (["label"] :: [Text])
   83               ]
   84           ]
   85       ]
   86   , "required" .= (["question"] :: [Text])
   87   ]

   89 -- | SHOW_HUMAN: emit @message@ to the human via the channel.
   90 -- Returns an empty, non-error result; the channel itself is the side-effect.
   91 showHumanOp :: ChannelCaps -> Opcode
   92 showHumanOp caps = TrustedOpcode
   93   { toName = OpName "SHOW_HUMAN"
   94   , toTrust = Trusted
   95   , toDesc = "Display a message to the human operator."
   96   , toInSchema = singleStringSchema "message" "The message to display to the human operator."
   97   , toOutSchema = object []
   98   , toAuthorize =
   99       maybe (Left "SHOW_HUMAN requires {message:string}") (const (Right ())) . strField "message"
  100   , toRun = \_ v -> do
  101       let msg = fromMaybe "" (strField "message" v)
  102       liftIO (ccSend caps msg)
  103       pure (OpResult [] False Null)
  104   }
  105 
  106 -- | ASK_HUMAN: send @question@ to the human and return their typed reply.
  107 -- When the optional @options@ array is present, the channel renders one
  108 -- button per choice plus an "Other" free-text option; the human's chosen
  109 -- text (a label or a typed reply) is returned.
  110 askHumanOp :: ChannelCaps -> Opcode
  111 askHumanOp caps = TrustedOpcode
  112   { toName = OpName "ASK_HUMAN"
  113   , toTrust = Trusted
  114   , toDesc = "Ask the human operator a question and return their reply."
  115   , toInSchema = askHumanSchema
  116   , toOutSchema = object []
  117   , toAuthorize = \v ->
  118       case strField "question" v of
  119         Nothing -> Left "ASK_HUMAN requires {question:string}"
  120         Just _ -> case parseOptions v of
  121           -- Malformed options field (not an array of objects) → reject.
  122           Nothing -> Left "ASK_HUMAN: invalid 'options' field"
  123           -- Absent options field (Nothing inner) → open-ended (valid).
  124           Just Nothing -> Right ()
  125           -- Present options → validate.
  126           Just (Just opts) -> case validateOptions opts of
  127             Left e -> Left ("ASK_HUMAN: " <> e)
  128             Right _ -> Right ()
  129   , toRun = \_ v -> do
  130       let q = fromMaybe "" (strField "question" v)
  131           opts = case parseOptions v of
  132             Just (Just os) -> os
  133             _ -> []
  134       ans <- liftIO (ccPrompt caps (AskPrompt q opts))
  135       pure (OpResult [TrpText ans] False Null)
  136   }

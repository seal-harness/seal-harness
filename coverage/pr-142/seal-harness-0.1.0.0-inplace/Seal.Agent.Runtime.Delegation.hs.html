    2 -- | The delegation pipeline — the Seal analog of Hermes' @delegate_task@.
    3 --
    4 -- 'runDelegate' spawns one or more child agents with isolated context, runs each
    5 -- against a goal to completion (synchronously), and returns a structured
    6 -- result per child. The parent blocks until all children finish (or time out).
    7 --
    8 -- Each child gets:
    9 --
   10 --   * a fresh 'SessionId' and its own two-file transcript nested under
   11 --     @\<parent-session\>\/agents\/\<child-id\>@ (the worker-builder opens it);
   12 --   * a focused system prompt built from the def's @adSystem@ + the goal +
   13 --     optional context (the worker-builder assembles it);
   14 --   * a narrowed tool set: the def's @adTools@ allow-list with the delegation
   15 --     blocklist always stripped (no recursive spawning, no def mutation, no
   16 --     lifecycle introspection) — the worker-builder is responsible for
   17 --     applying the blocklist when it constructs the child ISA registry.
   18 --
   19 -- Features ported from @delegate_task@:
   20 --
   21 --   * batch mode ('tasks' array) with a concurrency cap
   22 --     (@delegation.max_concurrent_children@);
   23 --   * depth cap (@delegation.max_spawn_depth@, default 1, clamped to [1,3])
   24 --     + @orchestrator_enabled@ kill switch (the role field is accepted but the
   25 --     wiring layer enforces the effective role when it builds the child's
   26 --     ISA registry);
   27 --   * timeout (@delegation.child_timeout_seconds@, default 600s, floor 30s)
   28 --     via 'System.Timeout.timeout';
   29 --   * a heartbeat thread that touches a parent-activity cell every 30s so a
   30 --     gateway inactivity timeout doesn't fire while the child works;
   31 --   * a spawn-pause flag so an operator can freeze new fan-out without
   32 --     interrupting running children;
   33 --   * per-child provider\/model override via @[delegation] provider/model/base_url@
   34 --     (the worker-builder reads the override and resolves the child provider
   35 --     from it instead of inheriting the parent's);
   36 --   * rich result aggregation: summary, status, exit_reason, tokens,
   37 --     tool_trace, files read/written (the worker reports via the
   38 --     'ChildRunHooks' accumulators).
   39 --
   40 -- Differences from @delegate_task@ (by design):
   41 --
   42 --   * Children are /transcripted/ — each child's conversation + entries land
   43 --     on disk under the parent's @agents\/\<child-id\>@ dir, surviving restart.
   44 --   * The child run is the existing 'Seal.Agent.Loop.runTurn' with the goal as
   45 --     the first user message — there is no separate child AIAgent class.

  106 -- | The @[delegation]@ section of @config.toml@. Every field is optional; a
  107 -- missing key decodes as 'Nothing' and the resolver falls back to a default.
  108 data DelegationConfig = DelegationConfig
  109   { dcMaxConcurrentChildren :: Maybe Int
  110     -- ^ Cap on parallel children per 'runDelegate' batch. Default 3, floor 1.
  111   , dcChildTimeoutSeconds :: Maybe Double
  112     -- ^ Hard per-child timeout in seconds. Default 600, floor 30.
  113   , dcMaxSpawnDepth :: Maybe Int
  114     -- ^ Max delegation tree depth. Default 1 (flat), clamped to [1,3].
  115   , dcOrchestratorEnabled :: Maybe Bool
  116     -- ^ Kill switch for the orchestrator role. Default True.
  117   , dcProvider :: Maybe Text
  118     -- ^ Per-child provider override (e.g. route subagents to a cheaper model).
  119   , dcModel :: Maybe Text
  120     -- ^ Per-child model override.
  121   , dcBaseUrl :: Maybe Text
  122     -- ^ Per-child base URL override (OpenAI-compatible direct endpoint).
  123   , dcApiKey :: Maybe Text
  124     -- ^ Per-child API key override (when @dcBaseUrl@ is set).
  125   , dcApiMode :: Maybe Text
  126     -- ^ Per-child API mode override (@chat_completions@ / @anthropic_messages@).
  127   , dcSubagentAutoApprove :: Maybe Bool
  128     -- ^ Whether subagent dangerous-command approvals auto-approve. Default
  129     -- False (auto-deny). Not yet wired (no approval callback in the child
  130     -- worker); reserved for future use.
  131   } deriving stock (Eq, Show)

  167 -- | Resolve a loaded 'DelegationConfig' to its effective numeric knobs, with
  168 -- the documented clamps applied. Provider/model/base_url/api_key/api_mode
  169 -- overrides are read directly from the config (they're 'Maybe'-typed).
  170 resolveDelegationConfig :: DelegationConfig
  171                        -> (Int, Double, Int, Bool)
  172 resolveDelegationConfig cfg =
  173   ( maxConc
  174   , childTimeout
  175   , maxDepth
  176   , orchEnabled
  177   )
  178   where
  179     maxConc = max 1 $ fromMaybe defaultMaxConcurrentChildren (dcMaxConcurrentChildren cfg)
  180     childTimeout = max minChildTimeoutSeconds $
  181                      fromMaybe defaultChildTimeoutSeconds (dcChildTimeoutSeconds cfg)
  182     rawDepth = fromMaybe defaultMaxSpawnDepth (dcMaxSpawnDepth cfg)
  183     maxDepth = max minSpawnDepth (min maxSpawnDepthCap rawDepth)
  184     orchEnabled = fromMaybe True (dcOrchestratorEnabled cfg)
  185

  207 -- | Opaque identifier for one live subagent. Smart-constructed from the def
  208 -- id + a short random suffix so multiple concurrent children of the same def
  209 -- don't collide.
  210 newtype SubagentId = SubagentId { unSubagentId :: Text }
  211   deriving stock (Eq, Ord, Show)
  212 
  213 -- | Mint a fresh 'SubagentId' from a def-id text + 8 hex chars. The def id is
  214 -- embedded for debuggability (the TUI can show which def a subagent ran);
  215 -- the random suffix disambiguates concurrent spawns.
  216 mkSubagentId :: Text -> IO SubagentId
  217 mkSubagentId defIdText = do
  218   suffix <- randomRIO (0 :: Int, 0xFFFFFFFF)
  219   pure (SubagentId (defIdText <> "-" <> showHexI32 suffix))
  220 
  221 subagentIdText :: SubagentId -> Text
  222 subagentIdText (SubagentId t) = t
  223 
  224 -- | Show an 'Int' as 8-char zero-padded lowercase hex.
  225 showHexI32 :: Int -> Text
  226 showHexI32 n = T.justifyRight 8 '0' (T.pack (go n))
  227   where
  228     go x
  229       | x <= 0    = "0"
  230       | otherwise = loop x ""
  231     loop y acc
  232       | y == 0    = acc
  233       | otherwise = let (q, r) = y `divMod` 16
  234                         c = if r < 10 then toEnum (r + fromEnum '0')
  235                                       else toEnum (r - 10 + fromEnum 'a')
  236                     in loop q (c : acc)

  242 -- | One task to delegate. The def id resolves the provider/model/system/tools
  243 -- template; the goal is the first user message; the context is optional
  244 -- background appended to the system prompt.
  245 data ChildTask = ChildTask
  246   { ctDefId   :: !Text
  247     -- ^ The agent def id to spawn from (must already exist via AGENT_DEF_WRITE).
  248   , ctGoal    :: !Text
  249     -- ^ The task goal — becomes the first user message in the child's turn.
  250   , ctContext :: !(Maybe Text)
  251     -- ^ Optional background context appended to the system prompt.
  252   , ctRole    :: !(Maybe Text)
  253     -- ^ @\"leaf\"@ (default) or @\"orchestrator\"@. Per-task role beats the
  254     -- top-level one. Orchestrators may spawn their own subagents, bounded by
  255     -- @max_spawn_depth@.
  256   } deriving stock (Eq, Show)

  258 -- | Why a child stopped.

  345 -- | A worker-builder: given a def + fresh session id + the task + the run
  346 -- hooks, run the child turn to completion and return the outcome.
  347 --
  348 -- The wiring layer (Cli.hs, Channels.Loop.hs, Gateway.Send.hs) closes over
  349 -- the per-turn 'AgentEnv' deps (caps, execBackend, appEnv, eCfg, parent
  350 -- session id, parent depth) and produces the IO action. Keeping this as a
  351 -- parameter decouples the opcode from 'Seal.Agent.Loop.runTurn' / provider
  352 -- resolution so the opcode is unit-testable with a fake worker.
  353 type AgentWorkerBuilder
  354   =  AgentDef
  355   -> SessionId
  356   -> ChildTask
  357   -> ChildRunHooks
  358   -> IO ChildWorkerOutcome
  359 
  360 -- | Thread-safe accumulators the worker-builder writes to during the child
  361 -- run. 'runDelegate' reads them after the worker returns to populate the
  362 -- 'ChildResult'. The worker appends to these via 'atomicModifyIORef'' (the
  363 -- IORefs are single-writer-per-child, but atomicModify is cheap and safe).
  364 data ChildRunHooks = ChildRunHooks
  365   { crhToolTrace    :: IORef [ToolTraceEntry]
  366     -- ^ Append-only tool-trace accumulator.
  367   , crhFilesRead    :: IORef [Text]
  368     -- ^ Append-only file-read accumulator.
  369   , crhFilesWritten :: IORef [Text]
  370     -- ^ Append-only file-written accumulator.
  371   , crhInterrupted  :: IORef Bool
  372     -- ^ Polled by the worker between iterations; when 'True' the worker
  373     -- should stop at its next turn boundary and return 'CwoInterrupted'.
  374   }
  375 
  376 -- | What the worker-builder reports back to 'runDelegate'. The builder does
  377 -- NOT fill @crTaskIndex@ / @crSubagentId@ / @crDurationSeconds@ — those are
  378 -- filled in by the orchestrator. It DOES fill in the summary, tokens, exit
  379 -- reason, and child session id it observed.
  380 data ChildWorkerOutcome = ChildWorkerOutcome
  381   { cwoSummary      :: !(Maybe Text)
  382     -- ^ The final text response from the child (or an error message).
  383   , cwoExitReason   :: !ChildExitReason
  384     -- ^ How the child's run ended (from the worker's perspective).
  385   , cwoTokensInput  :: !Int
  386   , cwoTokensOutput :: !Int
  387   , cwoChildSession :: !(Maybe SessionId)
  388     -- ^ The 'SessionId' the worker ran under (so the parent can look up the
  389     -- child's transcript later).
  390   }

  396 -- | A process-global flag that, when paused, makes 'runDelegate' reject new
  397 -- spawns with a structured error. Running children keep running; only NEW
  398 -- 'runDelegate' calls fail fast until 'setSpawnPaused' clears the flag.
  399 newtype SpawnPauseFlag = SpawnPauseFlag (TVar Bool)
  400 
  401 newSpawnPauseFlag :: IO SpawnPauseFlag
  402 newSpawnPauseFlag = SpawnPauseFlag <$> newTVarIO False
  403 
  404 setSpawnPaused :: SpawnPauseFlag -> Bool -> IO Bool
  405 setSpawnPaused (SpawnPauseFlag tv) paused = do
  406   atomically (writeTVar tv paused)
  407   pure paused
  408 
  409 isSpawnPaused :: SpawnPauseFlag -> IO Bool
  410 isSpawnPaused (SpawnPauseFlag tv) = readTVarIO tv

  412 -- ---------------------------------------------------------------------------
  413 -- Parent activity heartbeat
  414 -- ---------------------------------------------------------------------------
  415 
  416 -- | A mutable cell the parent's gateway/loop touches to record \"the parent
  417 -- is alive and working\". 'runDelegate' starts a heartbeat thread that
  418 -- periodically touches this while a child runs, so the parent's gateway
  419 -- inactivity timeout doesn't fire during a long delegation.
  420 newtype ParentActivity = ParentActivity (MVar ())
  421 
  422 newParentActivity :: IO ParentActivity
  423 newParentActivity = ParentActivity <$> newMVar ()
  424 
  425 -- | Touch the parent-activity cell (idempotent, never blocks).
  426 touchParentActivity :: ParentActivity -> IO ()
  427 touchParentActivity (ParentActivity mv) =
  428   void (tryPutMVar mv () >> tryTakeMVar mv)

  430 -- ---------------------------------------------------------------------------
  431 -- Run
  432 -- ---------------------------------------------------------------------------
  433 
  434 -- | The input to 'runDelegate' — one task to run (single mode) or many
  435 -- (batch mode). The caller (the AGENT_START opcode) normalizes the model's
  436 -- input to this before calling.
  437 data DelegateInput
  438   = DiSingle !ChildTask
  439   | DiBatch ![ChildTask]
  440 
  441 -- | The top-level delegation runner. Spawns one or more child agents, runs
  442 -- each against its goal to completion (synchronously), and returns a
  443 -- 'ChildResult' per task. The parent blocks until all children finish (or
  444 -- time out). See the module header for the feature list.
  445 runDelegate
  446   :: DelegationConfig
  447   -> SpawnPauseFlag
  448   -> Maybe ParentActivity
  449   -> Int
  450      -- ^ the parent's current delegation depth (0 = top-level parent)
  451   -> DelegateInput
  452   -> (ChildTask -> IO (Either Text (AgentDef, AgentWorkerBuilder, SessionId)))
  453      -- ^ resolver: look up the def, build the worker, mint a fresh child
  454      -- 'SessionId'. 'Left' = def not found / invalid / depth would be
  455      -- exceeded. Per-task so the caller can cache or re-resolve.
  456   -> IO (Either Text [ChildResult])
  457 runDelegate cfg pauseFlag mParentActivity parentDepth input resolveTask = do
  458   paused <- isSpawnPaused pauseFlag
  459   if paused
  460     then pure (Left "Delegation spawning is paused. Clear the pause via the TUI or the AGENT_INTERRUPT RPC before retrying.")
  461     else do
  462       let (maxConc, childTimeout, maxDepth, _orchEnabled) = resolveDelegationConfig cfg
  463       if parentDepth >= maxDepth
  464         then pure (Left ("Delegation depth limit reached (depth=" <> T.pack (show parentDepth)
  465                           <> ", max_spawn_depth=" <> T.pack (show maxDepth)
  466                           <> "). Raise delegation.max_spawn_depth in config.toml if deeper nesting is required (cap: "
  467                           <> T.pack (show maxSpawnDepthCap) <> ")."))
  468         else do
  469           tasks <- case input of
  470             DiSingle t -> pure [t]
  471             DiBatch ts -> pure ts
  472           if null tasks
  473             then pure (Left "No tasks provided.")
  474             else do
  475               when (length tasks > maxConc) $
  476                 -- We don't reject oversize batches; the semaphore caps
  477                 -- parallelism. (delegate_task rejects; we cap.)
  478                 pure ()
  479               runBatch maxConc childTimeout tasks
  480   where
  481     -- Run all tasks with a concurrency cap. Single-task short-cut avoids the
  482     -- thread-pool overhead.
  483     runBatch :: Int -> Double -> [ChildTask] -> IO (Either Text [ChildResult])
  484     runBatch maxConc childTimeout tasks =
  485       case tasks of
  486         [t] -> do
  487           r <- runOne 0 t childTimeout
  488           pure (Right [r])
  489         _   -> do
  490           sem <- newMVar maxConc
  491           mvars <- forM (zip [0 ..] tasks) $ \(idx, task) -> do
  492             resMVar <- newEmptyMVar
  493             void (forkIO $
  494               bracketSem sem $
  495                 runOne idx task childTimeout >>= putMVar resMVar)
  496             pure resMVar
  497           results <- forM mvars readMVar
  498           pure (Right results)
  499 
  500     -- Acquire / release the semaphore MVar around an IO action.
  501     bracketSem :: MVar Int -> IO a -> IO a
  502     bracketSem sem act =
  503       takeMVar sem *> act <* putMVar sem (maxBound :: Int)
  504       -- The putMVar value doesn't matter; we just need to release the slot.
  505       -- Using maxBound is a no-op marker (the sem is a counting semaphore via
  506       -- takeMVar/putMVar).
  507 
  508     -- Run a single task to completion with a hard timeout and a heartbeat
  509     -- thread that touches the parent-activity cell.
  510     runOne :: Int -> ChildTask -> Double -> IO ChildResult
  511     runOne idx task childTimeout = do
  512       subagentId <- mkSubagentId (ctDefId task)
  513       let micros = round (childTimeout * 1000000) :: Int
  514       start <- getCurrentTime
  515       -- Thread-safe accumulators the worker writes to.
  516       traceRef     <- newIORef []
  517       readRef     <- newIORef []
  518       writtenRef  <- newIORef []
  519       interruptedRef <- newIORef False
  520       let hooks = ChildRunHooks traceRef readRef writtenRef interruptedRef
  521       mRes <- resolveTask task >>= \case
  522         Left err -> pure (Left err)
  523         Right (def, worker, childSid) -> do
  524           -- Start the heartbeat thread.
  525           hbStop <- newEmptyMVar
  526           hbThreadId <- forkIO (heartbeatLoop mParentActivity hbStop)
  527           let runWithCatch =
  528                 worker def childSid task hooks
  529                   `catch` \e -> pure (ChildWorkerOutcome
  530                                        (Just (T.pack (show (e :: SomeException))))
  531                                        CerError 0 0 (Just childSid))
  532           mOutcome <- timeout micros runWithCatch
  533           -- Stop the heartbeat.
  534           void (tryPutMVar hbStop ())
  535           killThread hbThreadId
  536           pure (maybe (Left "timeout") Right mOutcome)
  537       end <- getCurrentTime
  538       let dur = realToFrac (end `diffUTCTime` start) :: Double
  539       case mRes of
  540         Left err ->
  541           pure (ChildResult
  542                   { crTaskIndex = idx
  543                   , crStatus = if err == "timeout" then CsTimeout else CsError
  544                   , crSummary = Nothing
  545                   , crExitReason = if err == "timeout" then CerTimeout else CerError
  546                   , crDurationSeconds = dur
  547                   , crSubagentId = subagentId
  548                   , crTokensInput = 0
  549                   , crTokensOutput = 0
  550                   , crToolTrace = []
  551                   , crError = Just (if err == "timeout"
  552                                       then "Subagent timed out after " <> T.pack (show childTimeout) <> "s."
  553                                       else err)
  554                   , crFilesRead = []
  555                   , crFilesWritten = []
  556                   , crChildSession = Nothing
  557                   })
  558         Right outcome -> do
  559           traceList <- readIORef traceRef
  560           filesRead <- readIORef readRef
  561           filesWritten <- readIORef writtenRef
  562           pure (ChildResult
  563                   { crTaskIndex = idx
  564                   , crStatus = childStatusFor (cwoExitReason outcome)
  565                   , crSummary = cwoSummary outcome
  566                   , crExitReason = cwoExitReason outcome
  567                   , crDurationSeconds = dur
  568                   , crSubagentId = subagentId
  569                   , crTokensInput = cwoTokensInput outcome
  570                   , crTokensOutput = cwoTokensOutput outcome
  571                   , crToolTrace = reverse traceList
  572                   , crError = Nothing
  573                   , crFilesRead = reverse filesRead
  574                   , crFilesWritten = reverse filesWritten
  575                   , crChildSession = cwoChildSession outcome
  576                   })

  578 -- | Map a worker-reported exit reason to the aggregate status.
  579 childStatusFor :: ChildExitReason -> ChildStatus
  580 childStatusFor = \case
  581   CerCompleted    -> CsCompleted
  582   CerMaxIterations -> CsFailed
  583   CerTimeout      -> CsTimeout
  584   CerInterrupted  -> CsInterrupted
  585   CerError        -> CsError

  587 -- | A 30s heartbeat that touches the parent-activity cell while the child
  588 -- runs. Stops when the stop-MVar is filled.
  589 heartbeatLoop :: Maybe ParentActivity -> MVar () -> IO ()
  590 heartbeatLoop mParentActivity stopRef = loop
  591   where
  592     loop = do
  593       stop <- tryTakeMVar stopRef
  594       case stop of
  595         Just ()  -> pure ()
  596         Nothing  -> do
  597           for_ mParentActivity touchParentActivity
  598           threadDelay 30000000  -- 30s
  599           loop

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The @/skill@ command group: list defined skills, show one skill's
    3 -- full body, and load a skill into the current session. Skills are stored
    4 -- as structured records materialized from the Audited log into the
    5 -- in-memory 'SkillBackend'; @/skill list@ and @/skill info@ read that
    6 -- backend directly (no filesystem discovery, no audit-trail entry).
    7 -- @/skill load@ dispatches the 'SKILL_LOAD' opcode via the channel-supplied
    8 -- 'CallDispatcher' (the same closure @/call@ uses), which records an
    9 -- 'EKHarness' entry to the session transcript with
   10 -- @erMeta.op.name = "SKILL_LOAD"@ and @erMeta.input.id = <id>@ — the
   11 -- audit-trail attribution that distinguishes a skill load from a user
   12 -- pasting the body.
   13 module Seal.Command.Skill
   14   ( skillCommandSpec
   15   , renderSkillLine
   16   , renderSkillInfo
   17   ) where

   35 -- | The @/skill@ command spec. Closes over the 'SkillBackend' (for
   36 -- @/skill list@ \/ @/skill info@, which read the materialized view of the
   37 -- Audited log directly) and the channel-supplied 'CallDispatcher' (for
   38 -- @/skill load@, which dispatches the 'SKILL_LOAD' opcode against the
   39 -- active session's ISA registry + transcript — the same dispatcher
   40 -- @/call@ uses, so the audit trail records both paths uniformly).
   41 skillCommandSpec :: SkillBackend -> CallDispatcher -> CommandSpec
   42 skillCommandSpec backend dispatcher = CommandSpec
   43   { csName         = CommandName "skill"
   44   , csAliases      = []
   45   , csGroup        = GroupSkills
   46   , csSynopsis     = "List defined skills, show one skill's body, or load one into the session"
   47   , csParserInfo   = skillParserInfo backend dispatcher
   48   , csAvailability = InteractiveOnly
   49   }
   50 
   51 skillParserInfo :: SkillBackend -> CallDispatcher -> ParserInfo CommandAction
   52 skillParserInfo backend dispatcher =
   53   info (skillParser backend dispatcher <**> helper)
   54     (  progDesc "Inspect or load agent skills"
   55     <> header   "skill — list, show, or load a skill"
   56     )
   57 
   58 skillParser :: SkillBackend -> CallDispatcher -> Parser CommandAction
   59 skillParser backend dispatcher = hsubparser
   60   (  command "list"
   61        (info (pure (listCmd backend)) (progDesc "List all defined skills (id + description)"))
   62   <> command "info"
   63        (info (infoCmd backend <$> skillArg)
   64              (progDesc "Show one skill's full body"))
   65   <> command "load"
   66        (info (loadCmd dispatcher <$> skillArg <*> messageArg)
   67              (progDesc "Load one skill into the current session, with an optional trailing message"))
   68   <> metavar "COMMAND"
   69   )
   70 
   71 -- | Required skill-id argument.
   72 skillArg :: Parser Text
   73 skillArg = T.pack <$> strArgument (metavar "SKILL" <> help "Skill id (e.g. greet)")
   74 
   75 -- | Optional trailing message (everything after the skill id). Joined with
   76 -- spaces so @/skill load foo do something@ yields the message @do something@.
   77 -- Empty when no trailing text is supplied (just a skill load, no message).
   78 messageArg :: Parser Text
   79 messageArg = T.intercalate " " . map T.pack <$> many (strArgument (metavar "MESSAGE..." <> help "Optional message appended after the skill body"))
   80 
   81 listCmd :: SkillBackend -> CommandAction
   82 listCmd backend = CommandAction $ \caps -> do
   83   skills <- sbList backend
   84   if null skills
   85     then ccSend caps "no skills defined"
   86     else mapM_ (ccSend caps . renderSkillLine) skills
   87 
   88 infoCmd :: SkillBackend -> Text -> CommandAction
   89 infoCmd backend raw = CommandAction $ \caps ->
   90   case mkSkillId raw of
   91     Left err -> ccSend caps err
   92     Right sid -> do
   93       mSkill <- sbRead backend sid
   94       case mSkill of
   95         Nothing -> ccSend caps ("skill not found: " <> skillIdText sid)
   96         Just s  -> mapM_ (ccSend caps) (renderSkillInfo s)
   97 
   98 -- | @/skill load <id> [message...]@ — dispatch the 'SKILL_LOAD' opcode with
   99 -- @{"id": <id>, "message": <message>}@ via the channel-supplied
  100 -- 'CallDispatcher'. Mirrors @/call@'s pattern: echo a header line first (so
  101 -- the "Command output" bubble is self-contained).
  102 --
  103 -- The optional trailing @message@ is forwarded to the dispatcher as a
  104 -- @message@ field in the opcode input. 'recordSkillLoadResult' appends it to
  105 -- @conversation.jsonl@ as a separate 'User' message AFTER the skill body, so
  106 -- the model sees the skill followed by the user's request on the next turn
  107 -- (e.g. @/skill load start #123@ loads the @start@ skill then sends
  108 -- @#123@ as the user's message). When no message is supplied, the input
  109 -- omits the @message@ key and the load behaves as before (skill body only).
  110 --
  111 -- On success, the skill body is NOT rendered via 'ccSend' — the dispatcher
  112 -- records a second 'EKHarness' entry to the transcript carrying the
  113 -- @orRecorded@ value (which includes the body), and the frontend renders
  114 -- that entry as a collapsible tool-call box. This keeps the slash bubble
  115 -- to just the echo line and avoids duplicating the body in the "command
  116 -- output — not saved" transient bubble.
  117 --
  118 -- On error (dispatch 'Left' or 'Right' with the error flag set), the error
  119 -- text IS rendered via 'ccSend' so the user sees it in the slash bubble —
  120 -- error paths produce no transcript body entry, so the slash bubble is
  121 -- the only surface for the error message.
  122 loadCmd :: CallDispatcher -> Text -> Text -> CommandAction
  123 loadCmd dispatcher raw message = CommandAction $ \caps -> do
  124   ccSend caps ("$ /skill load " <> raw)
  125   case mkSkillId raw of
  126     Left err -> ccSend caps err
  127     Right sid -> do
  128       let input = object
  129             [ "id" .= skillIdText sid
  130             , "message" .= message
  131             ]
  132       res <- dispatcher (OpName "SKILL_LOAD") input
  133       case res of
  134         Left e  -> ccSend caps (renderDispatchError e)
  135         Right r
  136           | orIsError r -> mapM_ (ccSend caps) (renderOpResult r)
  137           | otherwise   -> pure ()
  138 
  139 -- | One line per skill for @/skill list@.
  140 renderSkillLine :: Skill -> Text
  141 renderSkillLine s = skillIdText (skId s) <> "  " <> skDescription s
  142 
  143 -- | Multi-line detail for @/skill info@.
  144 renderSkillInfo :: Skill -> [Text]
  145 renderSkillInfo s =
  146   [ "id:          " <> skillIdText (skId s)
  147   , "description: " <> skDescription s
  148   , "updated:     " <> T.pack (show (skUpdatedAt s))
  149   , "session:     " <> T.pack (show (skSession s))
  150   , ""
  151   , skBody s

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Pure Markdown codec for 'Skill' — frontmatter + body. Split out of
    3 -- 'Seal.Skills.Backend' so the embedded built-in skills
    4 -- ('Seal.Skills.Builtins') can decode their compile-time sources without
    5 -- importing the IO-shaped backend (which would create a module cycle:
    6 -- Backend imports Builtins for the union layer, Builtins imports the
    7 -- codec).
    8 module Seal.Skills.Codec
    9   ( encodeSkill
   10   , decodeSkill
   11   ) where
   12 
   13 import Data.Either (fromRight)
   14 import Data.Map.Strict qualified as Map
   15 import Data.Maybe (fromMaybe)
   16 import Data.Text (Text)
   17 import Data.Text qualified as T
   18 import Data.Time (UTCTime (..))
   19 import Data.Time.Calendar (fromGregorian)
   20 import Data.Time.Clock (secondsToDiffTime)
   21 import Data.Time.Format (defaultTimeLocale, formatTime, parseTimeM)
   22 
   23 import Seal.Core.Types (mkSessionId, mkSystemSessionId, sessionIdText)
   24 import Seal.Skills.Types (Skill (..), mkSkillId, skillIdText)
   25 import Seal.Store.Markdown (decodeDoc, encodeDoc, fmLookup)
   26 
   27 -- | Encode a 'Skill' as a Markdown document (frontmatter + body).
   28 encodeSkill :: Skill -> Text
   29 encodeSkill s = encodeDoc fm (skBody s)
   30   where
   31     fm = Map.fromList $
   32       [ ("id", skillIdText (skId s))
   33       , ("description", skDescription s)
   34       , ("created_at", isoTime (skCreatedAt s))
   35       , ("updated_at", isoTime (skUpdatedAt s))
   36       , ("session", sessionIdText (skSession s))
   37       ] ++ maybe [] (\g -> [("group", g)]) (skGroup s)
   38 
   39 -- | Decode a Markdown document into a 'Skill'. Returns 'Nothing' if the id
   40 -- field is missing or fails 'mkSkillId'. Timestamps default to epoch 0 when
   41 -- absent or unparseable (the file was hand-edited without them).
   42 decodeSkill :: Text -> Maybe Skill
   43 decodeSkill content =
   44   case decodeDoc content of
   45     (fm, body) -> do
   46       sidT <- fmLookup "id" fm
   47       sid  <- either (const Nothing) Just (mkSkillId sidT)
   48       Just Skill
   49         { skId = sid
   50         , skDescription = fromMaybe "" (fmLookup "description" fm)
   51         , skBody = body
   52         , skGroup = fmLookup "group" fm
   53         , skCreatedAt = parseTime (fmLookup "created_at" fm)
   54         , skUpdatedAt = parseTime (fmLookup "updated_at" fm)
   55         , skSession = fromRight (mkSystemSessionId "unknown") (mkSessionId (fromMaybe "unknown" (fmLookup "session" fm)))
   56         }
   57 
   58 -- | Render a 'UTCTime' as an ISO-8601 string (UTC, with @Z@ suffix).
   59 isoTime :: UTCTime -> Text
   60 isoTime = T.pack . formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%SZ"
   61 
   62 -- | Parse an ISO-8601 'UTCTime' from a frontmatter value. Defaults to epoch 0
   63 -- when absent or unparseable (hand-edited files).
   64 parseTime :: Maybe Text -> UTCTime
   65 parseTime Nothing    = epochZero
   66 parseTime (Just raw) = fromMaybe epochZero (parseTimeM True defaultTimeLocale "%Y-%m-%dT%H:%M:%SZ" (T.unpack raw))
   67 
   68 -- | The epoch fallback for missing/unparseable timestamps.
   69 epochZero :: UTCTime
   70 epochZero = UTCTime (fromGregorian 1970 1 1) (secondsToDiffTime 0)

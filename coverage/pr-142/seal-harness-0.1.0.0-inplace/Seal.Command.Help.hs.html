   31 -- ---------------------------------------------------------------------------
   32 -- Help index
   33 -- ---------------------------------------------------------------------------
   34 
   35 -- | Render the grouped \/help index.
   36 --
   37 -- Output format:
   38 -- @
   39 -- Available commands:
   40 --
   41 --   /help [command]   Show this help, or detailed help for a command
   42 --
   43 -- General
   44 --   /ping             Check connectivity
   45 --
   46 -- Vault
   47 --   /vault            Manage the encrypted secret vault
   48 -- @
   49 --
   50 -- The synthetic @\/help@ entry is always present at the top even though
   51 -- @\/help@ is not a registered 'CommandSpec' (to avoid a registry-knot).
   52 renderHelpIndex :: Registry -> Text
   53 renderHelpIndex reg =
   54   T.unlines $
   55     [ "Available commands:"
   56     , ""
   57     , syntheticHelpLine
   58     , ""
   59     ]
   60     ++ concatMap renderGroup (Map.toAscList grouped)
   61   where
   62     specs   = registrySpecs reg
   63     -- Group specs preserving original insertion order within each group.
   64     grouped :: Map CommandGroup [CommandSpec]
   65     grouped =
   66       Map.fromListWith (flip (++))
   67         [(csGroup s, [s]) | s <- specs]
   68 
   69     syntheticHelpLine :: Text
   70     syntheticHelpLine =
   71       "  " <> T.justifyLeft colWidth ' ' "/help [command]"
   72            <> "Show this help, or detailed help for a command"
   73 
   74     renderGroup :: (CommandGroup, [CommandSpec]) -> [Text]
   75     renderGroup (grp, grpSpecs) =
   76       [ groupHeader grp ]
   77       ++ map renderSpec grpSpecs
   78       ++ [ "" ]
   79 
   80     groupHeader :: CommandGroup -> Text
   81     groupHeader GroupGeneral  = "General"
   82     groupHeader GroupProvider = "Providers"
   83     groupHeader GroupSession  = "Sessions"
   84     groupHeader GroupModel    = "Model"
   85     groupHeader GroupVault    = "Vault"
   86     groupHeader GroupSkills   = "Skills"
   87     groupHeader GroupAgent    = "Agents"
   88     groupHeader GroupTools    = "Tools"
   89     groupHeader GroupRepos    = "Repos"
   90 
   91     renderSpec :: CommandSpec -> Text
   92     renderSpec s =
   93       let CommandName n = csName s
   94           label         = "/" <> n
   95       in "  " <> T.justifyLeft colWidth ' ' label <> csSynopsis s
   96 
   97     -- Column width for the command label column (includes the leading slash).
   98     colWidth :: Int
   99     colWidth = 18

  101 -- ---------------------------------------------------------------------------
  102 -- Per-command help
  103 -- ---------------------------------------------------------------------------
  104 
  105 -- | Render a specific command's full optparse help by running its
  106 -- 'ParserInfo' with @["--help"]@ via 'execParserPure'.
  107 --
  108 -- @execParserPure defaultPrefs info ["--help"]@ always returns
  109 -- @Failure (ParserFailure ParserHelp)@ because @--help@ is optparse's
  110 -- built-in action. 'renderFailure' converts that to @(String, ExitCode)@;
  111 -- we take the 'String', pack it to 'Text', and discard the exit code
  112 -- ('ExitSuccess' for @--help@).
  113 --
  114 -- This means the rendered text is 100% derived from the optparse parser
  115 -- and cannot drift from the actual flags the command accepts.
  116 renderHelpFor :: Registry -> CommandName -> Text
  117 renderHelpFor reg name@(CommandName n) =
  118   case lookupSpec reg name of
  119     Nothing   -> "No such command: " <> n <> "\n"
  120     Just spec ->
  121       case execParserPure defaultPrefs (csParserInfo spec) ["--help"] of
  122         Failure f ->
  123           -- progName argument to renderFailure is used as the program name
  124           -- in the rendered usage line; we use the slash-command name.
  125           let (msg, _exitCode) = renderFailure f ("/" <> T.unpack n)
  126           in T.pack msg
  127         -- The following two branches are unreachable when the input is
  128         -- ["--help"] against a well-formed ParserInfo, but we must be
  129         -- total to satisfy -Wall -Werror.
  130         Success _           -> ""
  131         CompletionInvoked _ -> ""

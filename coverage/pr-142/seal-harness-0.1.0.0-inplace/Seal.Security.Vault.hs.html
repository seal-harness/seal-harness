    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | An encrypted secret vault. Values are stored as a base64-encoded JSON map,
    3 -- encrypted as a whole by the 'VaultEncryptor'. Writes are atomic
    4 -- (tmp → chmod 0600 → rename). Three unlock modes trade memory residency for
    5 -- convenience. All mutations are serialised by an 'MVar'; the decrypted map is
    6 -- cached in a 'TVar' for the startup/on-demand modes.
    7 module Seal.Security.Vault
    8   ( UnlockMode (..)
    9   , VaultConfig (..)
   10   , VaultStatus (..)
   11   , VaultHandle (..)
   12   , openVault
   13   ) where

   51 data VaultHandle = VaultHandle
   52   { vhInit :: IO (Either VaultError ())
   53   , vhUnlock :: IO (Either VaultError ())
   54   , vhLock :: IO ()
   55   , vhGet :: Text -> IO (Either VaultError ByteString)
   56   , vhPut :: Text -> ByteString -> IO (Either VaultError ())
   57   , vhDelete :: Text -> IO (Either VaultError ())
   58   , vhList :: IO (Either VaultError [Text])
   59   , vhStatus :: IO VaultStatus
   60   , vhRekey :: VaultEncryptor -> Text -> (Text -> IO Bool) -> IO (Either VaultError ())
   61   }

   63 -- Internal mutable state (not exported).
   64 data VaultState = VaultState
   65   { stConfig :: VaultConfig
   66   , stEncryptor :: IORef VaultEncryptor
   67   , stKeyType :: IORef Text
   68   , stCache :: TVar (Maybe (Map Text ByteString))
   69   , stWriteLock :: MVar ()
   70   }
   71 
   72 openVault :: VaultConfig -> VaultEncryptor -> IO VaultHandle
   73 openVault cfg enc = do
   74   st <- VaultState cfg
   75           <$> newIORef enc
   76           <*> newIORef (vcKeyType cfg)
   77           <*> newTVarIO Nothing
   78           <*> newMVar ()
   79   pure VaultHandle
   80     { vhInit   = vaultInit st
   81     , vhUnlock = vaultUnlock st
   82     , vhLock   = atomically (writeTVar (stCache st) Nothing)
   83     , vhGet    = withCurrentMap st . lookupKey
   84     , vhPut    = \k v -> mutate st (Right . Map.insert k v)
   85     , vhDelete = \k -> mutate st $ \m ->
   86         if Map.member k m then Right (Map.delete k m) else Left (VaultKeyNotFound k)
   87     , vhList   = withCurrentMap st (Right . Map.keys)
   88     , vhStatus = vaultStatus st
   89     , vhRekey  = vaultRekey st
   90     }

   96 vaultInit :: VaultState -> IO (Either VaultError ())
   97 vaultInit st = withMVar (stWriteLock st) $ \_ -> do
   98   exists <- doesFileExist (vcPath (stConfig st))
   99   if exists
  100     then pure (Left VaultAlreadyExists)
  101     else persistToDisk st Map.empty
  102 
  103 -- | Decrypt the on-disk vault into the cache (for the cached modes).
  104 vaultUnlock :: VaultState -> IO (Either VaultError ())
  105 vaultUnlock st = do
  106   res <- readMap st
  107   case res of
  108     Left e  -> pure (Left e)
  109     Right m -> do
  110       atomically (writeTVar (stCache st) (Just m))
  111       pure (Right ())
  112 
  113 vaultStatus :: VaultState -> IO VaultStatus
  114 vaultStatus st = do
  115   cache   <- readTVarIO (stCache st)
  116   keyType <- readIORef (stKeyType st)
  117   pure VaultStatus
  118     { vsLocked      = isNothing cache
  119     , vsSecretCount = maybe 0 Map.size cache
  120     , vsKeyType     = keyType
  121     }
  122 
  123 -- | Obtain the current map per unlock mode, then apply a pure observation.
  124 -- 'UnlockOnDemand' transparently unlocks first.
  125 withCurrentMap
  126   :: VaultState
  127   -> (Map Text ByteString -> Either VaultError a)
  128   -> IO (Either VaultError a)
  129 withCurrentMap st f = do
  130   prepareAccess st
  131   em <- currentMap st
  132   pure (em >>= f)
  133 
  134 -- | Read-modify-write a mutation under the write lock.
  135 mutate
  136   :: VaultState
  137   -> (Map Text ByteString -> Either VaultError (Map Text ByteString))
  138   -> IO (Either VaultError ())
  139 mutate st f = do
  140   prepareAccess st
  141   withMVar (stWriteLock st) $ \_ -> do
  142     em <- currentMap st
  143     case em >>= f of
  144       Left e   -> pure (Left e)
  145       Right m' -> writeMap st m'
  146 
  147 -- | For 'UnlockOnDemand', ensure the cache is populated before we take the
  148 -- write lock (so we never deadlock by unlocking inside it).
  149 prepareAccess :: VaultState -> IO ()
  150 prepareAccess st = case vcUnlock (stConfig st) of
  151   UnlockOnDemand -> do
  152     cache <- readTVarIO (stCache st)
  153     case cache of
  154       Just _  -> pure ()
  155       Nothing -> void (vaultUnlock st)
  156   _ -> pure ()
  157 
  158 -- | The current decrypted map according to unlock mode. Per-access always
  159 -- reads disk; the cached modes read the 'TVar' and are 'VaultLocked' if empty.
  160 currentMap :: VaultState -> IO (Either VaultError (Map Text ByteString))
  161 currentMap st = case vcUnlock (stConfig st) of
  162   UnlockPerAccess -> readMap st
  163   _ -> maybe (Left VaultLocked) Right <$> readTVarIO (stCache st)
  164 
  165 -- | Write a map to disk without updating the in-memory cache.
  166 persistToDisk :: VaultState -> Map Text ByteString -> IO (Either VaultError ())
  167 persistToDisk st m = do
  168   enc <- readIORef (stEncryptor st)
  169   let payload = BL.toStrict (Aeson.encode (encodeValues m))
  170   res <- veEncrypt enc payload
  171   case res of
  172     Left e           -> pure (Left e)
  173     Right ciphertext -> do
  174       atomicWrite (vcPath (stConfig st)) ciphertext
  175       pure (Right ())
  176 
  177 -- | Persist a map to disk and, for the cached modes, refresh the cache.
  178 writeMap :: VaultState -> Map Text ByteString -> IO (Either VaultError ())
  179 writeMap st m = do
  180   res <- persistToDisk st m
  181   case res of
  182     Left e  -> pure (Left e)
  183     Right () -> do
  184       case vcUnlock (stConfig st) of
  185         UnlockPerAccess -> pure ()
  186         _ -> atomically (writeTVar (stCache st) (Just m))
  187       pure (Right ())
  188 
  189 -- | Read and decrypt the on-disk vault into a map.
  190 readMap :: VaultState -> IO (Either VaultError (Map Text ByteString))
  191 readMap st = do
  192   enc        <- readIORef (stEncryptor st)
  193   fileResult <- try @IOException (BS.readFile (vcPath (stConfig st)))
  194   case fileResult of
  195     Left _       -> pure (Left VaultNotFound)
  196     Right fileBs -> do
  197       plain <- veDecrypt enc fileBs
  198       pure (plain >>= decodePayload)

  200 -- ---------------------------------------------------------------------------
  201 -- Rekey: write to .new, verify byte-for-byte, confirm, atomic replace
  202 -- ---------------------------------------------------------------------------
  203 
  204 vaultRekey
  205   :: VaultState -> VaultEncryptor -> Text -> (Text -> IO Bool)
  206   -> IO (Either VaultError ())
  207 vaultRekey st newEnc newKeyType confirm = withMVar (stWriteLock st) $ \_ -> do
  208   let path    = vcPath (stConfig st)
  209       newPath = path <> ".new"
  210   cur <- readMap st
  211   case cur of
  212     Left e         -> pure (Left e)
  213     Right plainMap -> do
  214       let payload = BL.toStrict (Aeson.encode (encodeValues plainMap))
  215       enc <- veEncrypt newEnc payload
  216       case enc of
  217         Left e           -> pure (Left e)
  218         Right ciphertext -> do
  219           atomicWrite newPath ciphertext
  220           verified <- verifyRekey newEnc newPath plainMap
  221           if not verified
  222             then cleanup newPath >> pure (Left (VaultBackendError "rekey verification failed"))
  223             else do
  224               oldKeyType <- readIORef (stKeyType st)
  225               ok <- confirm (rekeyPrompt oldKeyType newKeyType (Map.size plainMap))
  226               if not ok
  227                 then cleanup newPath >> pure (Left (VaultBackendError "rekey cancelled"))
  228                 else do
  229                   renameFile newPath path
  230                   writeIORef (stEncryptor st) newEnc
  231                   writeIORef (stKeyType st) newKeyType
  232                   case vcUnlock (stConfig st) of
  233                     UnlockPerAccess -> atomically (writeTVar (stCache st) Nothing)
  234                     _ -> atomically (writeTVar (stCache st) (Just plainMap))
  235                   pure (Right ())
  236 
  237 verifyRekey :: VaultEncryptor -> FilePath -> Map Text ByteString -> IO Bool
  238 verifyRekey newEnc newPath expected = do
  239   readBack <- try @IOException (BS.readFile newPath)
  240   case readBack of
  241     Left _   -> pure False
  242     Right bs -> do
  243       plain <- veDecrypt newEnc bs
  244       pure $ case plain >>= decodePayload of
  245         Right m -> m == expected
  246         Left _  -> False
  247 
  248 rekeyPrompt :: Text -> Text -> Int -> Text
  249 rekeyPrompt oldKt newKt n =
  250   "Replace vault? Old: " <> oldKt <> ", New: " <> newKt
  251     <> ", " <> T.pack (show n) <> " secrets verified identical"
  252 
  253 cleanup :: FilePath -> IO ()
  254 cleanup path = do
  255   exists <- doesFileExist path
  256   when exists (removeFile path)

  262 lookupKey :: Text -> Map Text ByteString -> Either VaultError ByteString
  263 lookupKey k = maybe (Left (VaultKeyNotFound k)) Right . Map.lookup k
  264 
  265 -- | Atomic write: tmp file, chmod 0600, rename over the target.
  266 atomicWrite :: FilePath -> ByteString -> IO ()
  267 atomicWrite path bs = do
  268   let tmp = path <> ".tmp"
  269   BS.writeFile tmp bs
  270   setFileMode tmp 0o600
  271   renameFile tmp path

  273 -- Values are base64 so binary secrets survive the JSON round-trip.
  274 encodeValues :: Map Text ByteString -> Map Text Text
  275 encodeValues = Map.map (TE.decodeUtf8 . B64.encode)
  276 
  277 decodePayload :: ByteString -> Either VaultError (Map Text ByteString)
  278 decodePayload plain =
  279   case Aeson.decodeStrict plain of
  280     Nothing      -> Left (VaultBackendError "invalid JSON")
  281     Just encoded -> maybe (Left (VaultBackendError "invalid base64")) Right
  282                           (traverse decodeValue encoded)
  283   where
  284     decodeValue t = either (const Nothing) Just (B64.decode (TE.encodeUtf8 t))

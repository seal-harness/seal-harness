   24 data AgentEnv = AgentEnv
   25   { aeProvider :: SomeProvider
   26     -- | The provider's label (e.g. @\"ollama\"@), used only for display —
   27     -- 'aeProvider' is existential and carries no name of its own.
   28   , aeProviderLabel :: Text
   29   , aeModel :: ModelId
   30   , aeSystem :: Maybe Text
   31     -- ^ The system prompt injected at the start of every turn. For the
   32     -- main session this comes from the bound default agent's 'adSystem';
   33     -- for a forked sub-agent it comes from the def's 'adSystem'.
   34   , aeRegistry :: Registry
   35   , aeTranscript :: TwoFileHandle
   36   , aeBackend :: BackendExec
   37   , aeUIOEnv :: UIOEnv
   38     -- ^ The untrusted-execution environment (carrying the 'UntrustedIO'
   39     -- capability handle + the Git 'CloneDeps' surface), threaded to
   40     -- 'Seal.ISA.Dispatch.dispatch' for Untrusted opcodes. Backend (local
   41     -- FS vs SSH transport) is selected once at wiring time via
   42     -- 'mkSessionExec' (which builds the 'UIOEnv' from the 'SecurityConfig');
   43     -- the opcode never sees the backend, only the capability. Trusted/Audited
   44     -- opcodes ignore it (the GADT 'Opcode' has no 'UIOEnv' field for them —
   45     -- type-level capability scoping, spec §4/§8). The 'CloneDeps' lives
   46     -- inside the 'UIOEnv' (via 'uieCloneDeps'), so Git opcodes source it
   47     -- from here rather than a separate field.
   48   , aeAbortFlag :: AbortFlag
   49     -- ^ The session-scoped abort flag. Set by the channel layer (Signal
   50     -- @\/stop@, CLI Ctrl+C, Web @POST /api/sessions/:id/stop@) to cancel
   51     -- in-flight tool calls. Polled by the dispatch wrapper
   52     -- ('Seal.Tools.Exec.Timeout.runWithTimeoutAbortRetry') during the
   53     -- three-way race. Cleared once at 'runTurn' entry (before any tool
   54     -- call); a mid-turn abort keeps the flag set until the next turn.
   55     -- Looked up from a 'Seal.Tools.Exec.Abort.SessionAbortRegistry' by
   56     -- 'SessionId' (the web gateway has no 'AgentEnv' at the @/stop@ call
   57     -- site — the registry is the vehicle, design Blocker Resolution #2).
   58   , aeToolTimeout :: ToolTimeoutConfig
   59     -- ^ The per-call timeout/retry config (default 120s, max 600s, 3
   60     -- retries, 2s base delay, 2.0 backoff, 50KB output cap). Loaded from
   61     -- @config.toml@ @[tool_timeout]@ at startup via
   62     -- 'Seal.Config.File.toolTimeoutConfig'.
   63   , aeCaps :: ChannelCaps
   64   , aeSession :: SessionId
   65   , aeMaxTurns :: Int
   66   , aeChannel :: Text
   67     -- ^ The channel this turn's message arrived on (e.g. @\"telegram\"@,
   68     -- @\"web\"@, @\"cli\"@). This is the /arrival/ channel, NOT the
   69     -- session's channel-of-origin — a web message sent to a
   70     -- Telegram-created session carries @\"web\"@, not @\"telegram\"@. Each
   71     -- wiring site passes the channel the message actually came in on:
   72     -- the web gateway passes @"web"@, the CLI passes @"cli"@, and the
   73     -- inbox-channel loop passes @smChannel@ (correct because an inbox
   74     -- session is always created on the channel its messages arrive on).
   75     -- 'runTurn' stamps this into every request entry's @erMeta@ @channel@
   76     -- field so the frontend can attribute every user message to the
   77     -- channel it came from. This is the single source of truth for channel
   78     -- provenance — 'MessageSource' carries the finer-grained conversation
   79     -- id + user id, but the channel label itself always comes from here.
   80   , aeMessageSource :: Maybe MessageSource
   81     -- ^ The authenticated-transport identity of the inbound message this
   82     -- turn is answering. 'Nothing' for the CLI TUI and the web (which
   83     -- bypass 'MessageSource'); @'Just' ms@ for channels that carry one
   84     -- (Signal, Telegram). 'runTurn' folds the 'msConversationId' into the
   85     -- request 'EntryRecord's @erMeta@ @conversationId@ field. The
   86     -- @channel@ field comes from 'aeChannel' (above), NOT from here, so
   87     -- every channel — including those with no 'MessageSource' — is
   88     -- attributed.
   89   , aeAutonomy :: AutonomyLevel
   90     -- ^ The operator-selected autonomy level. 'Full' (@--yolo@) bypasses the
   91     -- human-confirmation gate for Untrusted opcodes (they run immediately after
   92     -- the ACK-before-execute audit). 'Supervised' (the default) prompts the
   93     -- human via 'ccPrompt' before executing any Untrusted opcode; a non-"yes"
   94     -- reply cancels the call (the model sees a denied result). 'Deny' is
   95     -- enforced by the opcode's own authorize gate (rejected before the
   96     -- dispatcher runs).
   97   , aeApprovals :: ApprovalCache
   98     -- ^ The approval cache for Untrusted opcodes under 'Supervised' autonomy.
   99     -- Records "for this session" and "always" approvals so subsequent calls
  100     -- to the same opcode skip the prompt. 'ScopeRejected' entries short-
  101     -- circuit to denied. Threaded from the channel wiring (web, CLI, Signal).
  102   , aeDebugRequestsPath :: Maybe FilePath
  103     -- ^ When 'Just', every 'CompletionRequest' sent to the LLM is appended
  104     -- (redundantly, in full) to this file as one JSONL line per request.
  105     -- The contract: each line is the complete 'CompletionRequest' exactly as
  106     -- passed to the provider — including the full 'crMessages' history — so
  107     -- we can debug whether the two-file storage format is correctly feeding
  108     -- the session history to the LLM. 'Nothing' (the default) means no
  109     -- debug file is written.
  110   , aeOnEntry :: IO ()
  111     -- ^ A hook called by the loop after each transcript entry is recorded
  112     -- (response entries, tool-result entries, approval-evidence entries).
  113     -- The web channel wires this to 'broadcastNewEntries' so the frontend
  114     -- sees new entries live — including tool calls that are pending
  115     -- confirmation — rather than only at the end of the turn. The CLI and
  116     -- Signal channels set this to @pure ()@ (no live broadcast needed).
  117   , aeOnUserMessage :: Maybe (IO ())
  118     -- ^ When 'Just action', the loop records the initial user message with
  119     -- 'tfwRecordAndAck' (synchronously fsync'd) and then runs @action@ —
  120     -- guaranteeing the user message is durable on disk before @action@
  121     -- runs. Used by the @/bg@ channel path to broadcast a @lists@ snapshot
  122     -- whose snippet (the first user message) is populated, so the web
  123     -- sidebar shows the session name immediately rather than after the
  124     -- first LLM response. 'Nothing' (the default) keeps the async
  125     -- 'tfwRecordAsync' write (no fsync latency at turn start).
  126   , aeOnStop :: Maybe (Text -> IO ())
  127     -- ^ When 'Just fanout', the loop calls @fanout text@ with the final
  128     -- user-visible text at every stop branch (final answer, truncation
  129     -- give-up, max-turns stop, provider error). This is the chat-channel
  130     -- notification hook: wiring sites bind it to 'replyFanout' against the
  131     -- session's 'ReplyRegistry' so every subscribed chat channel (Telegram,
  132     -- Signal, …) attached to the tab receives the stop notice — not just
  133     -- the arrival channel (which 'ccSend' covers). 'Nothing' (the default
  134     -- for tests and the standalone CLI) means no fan-out; the arrival
  135     -- channel alone is notified via 'ccSend'.
  136   , aeOnDemandSchemas :: Bool
  137     -- ^ When 'True', the loop emits stub @input_schema@s in the @tools@
  138     -- field (via 'Seal.ISA.Registry.registryToolDefs'') to save tokens,
  139     -- and the registry is expected to include the @OPCODE_DESCRIBE@ /
  140     -- @OPCODE_LIST@ opcodes so the model can fetch full schemas on demand.
  141     -- 'False' (the default) sends full schemas inline, matching the
  142     -- pre-flag behavior.
  143   , aeLogPath :: Maybe FilePath
  144     -- ^ When 'Just', turn lifecycle events (start/end/duration) and failures
  145     -- (exceptions, provider errors) are appended to this per-session log file
  146     -- (@\<sessionDir\>\/seal.log@). Best-effort: a write error is swallowed.
  147     -- Records ONLY events not already in @entries.jsonl@ / @conversation.jsonl@.
  148     -- 'Nothing' disables per-session logging (the default for tests / fake
  149     -- transcripts).
  150   }

  152 -- | A parameter object bundling the per-turn inputs to 'mkSessionAgentEnv'.
  153 -- The fields are collected into one record so call sites construct it with
  154 -- named-field syntax (no positional-counting mistakes) and future additions
  155 -- are a one-field change. Lives here (next to 'AgentEnv') so the unified
  156 -- turn engine ('Seal.Core.TurnEngine') can import it without a cycle with
  157 -- 'Seal.Channel.Cli'.
  158 data TurnEnv = TurnEnv
  159   { teCaps          :: ChannelCaps
  160   , teProvider      :: SomeProvider
  161   , teProviderLabel :: Text
  162   , teModel         :: ModelId
  163   , teSession       :: SessionId
  164   , teSystem        :: Maybe Text
  165   , teRegistry      :: Registry
  166   , teTranscript    :: TwoFileHandle
  167   , teUioEnv        :: UIOEnv
  168   , teDebugReqPath  :: Maybe FilePath
  169   , teAutonomy      :: AutonomyLevel
  170   , teApprovals     :: ApprovalCache
  171   , teOnEntry       :: IO ()
  172   , teOnDemand      :: Bool
  173   , teLogPath       :: Maybe FilePath
  174   , teMaxTurns      :: Int
  175   , teOnUserMessage :: Maybe (IO ())
  176   , teChannel       :: Text
  177   , teOnStop        :: Maybe (Text -> IO ())
  178   , teAbortFlag     :: AbortFlag
  179   , teToolTimeout   :: ToolTimeoutConfig
  180   }
  181 
  182 -- | Build the per-turn 'AgentEnv' from a 'TurnEnv'. The 'aeBackend' is
  183 -- always 'localBackend' (Trusted/Audited opcodes run in-process; Untrusted
  184 -- opcodes source their backend from 'aeUIOEnv', not from here). The
  185 -- 'aeMessageSource' is left 'Nothing' here — the unified turn engine sets
  186 -- it via a record update after construction (it's an inbound-property, not
  187 -- a per-turn-env field).
  188 mkSessionAgentEnv :: TurnEnv -> AgentEnv
  189 mkSessionAgentEnv te = AgentEnv
  190   { aeProvider   = teProvider te
  191   , aeProviderLabel = teProviderLabel te
  192   , aeModel      = teModel te
  193   , aeSystem     = teSystem te
  194   , aeRegistry   = teRegistry te
  195   , aeTranscript = teTranscript te
  196   , aeBackend    = localBackend
  197   , aeUIOEnv     = teUioEnv te
  198   , aeCaps       = teCaps te
  199   , aeSession    = teSession te
  200   , aeMaxTurns   = teMaxTurns te
  201   , aeChannel    = teChannel te
  202   , aeMessageSource = Nothing
  203   , aeAutonomy   = teAutonomy te
  204   , aeApprovals  = teApprovals te
  205   , aeDebugRequestsPath = teDebugReqPath te
  206   , aeOnEntry    = teOnEntry te
  207   , aeOnUserMessage = teOnUserMessage te
  208   , aeOnStop     = teOnStop te
  209   , aeOnDemandSchemas = teOnDemand te
  210   , aeLogPath    = teLogPath te
  211   , aeAbortFlag  = teAbortFlag te
  212   , aeToolTimeout = teToolTimeout te
  213   }

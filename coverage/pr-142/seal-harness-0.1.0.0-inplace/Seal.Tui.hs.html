   52 -- | Open the vault if both recipient and identity are configured.
   53 -- Failures print a warning and return 'Nothing' so the TUI still starts;
   54 -- vault commands will direct the user to run @\/vault setup@.
   55 tryOpenVault :: SealPaths -> SecurityConfig -> IO (Maybe VaultHandle)
   56 tryOpenVault paths cfg =
   57   case (scVaultRecipient cfg, scVaultIdentity cfg) of
   58     (Just _, Just _) ->
   59       resolveEncryptor cfg >>= \case
   60         Left err -> do
   61           putStrLn ("Warning: vault not available: " <> show err)
   62           pure Nothing
   63         Right enc -> do
   64           -- Honor vault_path from config if set; fall back to the default.
   65           -- setupCmd and rekeyExisting in Seal.Vault.Commands replicate this
   66           -- same expression so all three stay in sync.
   67           let vcfg = VaultConfig
   68                 { vcPath    = maybe (vaultFilePath paths) T.unpack
   69                                     (scVaultPath cfg)
   70                 , vcKeyType = fromMaybe "x25519" (scVaultKeyType cfg)
   71                 , vcUnlock  = parseUnlockMode (scVaultUnlock cfg)
   72                 }
   73           Just <$> openVault vcfg enc
   74     _ -> pure Nothing

   76 -- | Full TUI wiring. The autonomy level threads through to 'mkSessionAgentEnv'
   77 -- so 'Supervised' (the default) prompts before running Untrusted opcodes.
   78 runTui :: AutonomyLevel -> SealLogger -> IO ()
   79 runTui autonomy logger = do
   80   paths <- getSealPaths
   81   ensureSealDirs paths
   82   migrateSecurityConfig paths
   83   let cfgPath = configFilePath paths
   84   cfg <- loadRuntimeConfig cfgPath >>= \case
   85     Left err -> do
   86       putStrLn ("Warning: could not load config: " <> T.unpack err)
   87       pure defaultRuntimeConfig
   88     Right c  -> pure c
   89   secCfg <- loadSecurityConfig (securityFilePath paths) >>= \case
   90     Left err -> do
   91       putStrLn ("Warning: could not load security config: " <> T.unpack err)
   92       pure defaultSecurityConfig
   93     Right c  -> pure c
   94   mHandle <- tryOpenVault paths secCfg
   95   ref     <- newIORef mHandle
   96   let rt = VaultRuntime
   97             { vrPaths      = paths
   98             , vrConfigPath = cfgPath
   99             , vrHandleRef  = ref
  100             }
  101   -- A dedicated manager for the /provider test round-trip. (M2 consolidates
  102   -- this with the chat provider's manager when the startup hardcode is removed.)
  103   mgr <- newTlsManager
  104   callCounter <- newIORef 0
  105   let pr = ProviderRuntime
  106             { prConfigPath  = cfgPath
  107             , prVault       = rt
  108             , prManager     = mgr
  109             , prCallCounter = callCounter
  110             }
  111   -- The config directory is a git repo (versioning + audit for the
  112   -- evolutionary stores: skills, agent-defs, memory live as Markdown files
  113   -- under config/skills, config/agents, config/memory). ensureConfigRepo
  114   -- runs `git init` + an initial empty commit if needed; idempotent.
  115   let cfgRoot = spConfig paths
  116   ensureConfigRepo cfgRoot
  117   let repo = openConfigRepo cfgRoot
  118   -- The evolutionary-store backends are disk-backed (Markdown + git), created
  119   -- once and shared between the @\/skill@ \/ @\/agent@ command specs
  120   -- (read-only) and the ISA opcodes (mutate, auto-commit). Disk is canonical.
  121   -- Built before initSession so the default agent can be resolved from disk.
  122   backends <- newBackends cfgRoot repo
  123   tabsH   <- newTabsHandle
  124   cli <- mkRealSignalCli
  125   tgApi <- mkRealTelegramBotApi
  126   vaultStore <- mkRealVaultStore mHandle
  127   let channelRt = ChannelRuntime { crConfigPath = cfgPath, crSignalCli = cli
  128                                  , crTelegramBotApi = tgApi
  129                                  , crVaultStore = vaultStore }
  130   -- Every launch starts a fresh session (resume is a follow-on milestone).
  131   -- The default agent (if set in config) is bound here: its id persists in
  132   -- smAgent and its non-empty provider/model override the config defaults.
  133   sessionMeta <- initSession paths cfg (bAgentDefs backends)
  134   activeRef   <- newIORef sessionMeta
  135   let sr = SessionRuntime
  136              { srPaths      = paths
  137              , srConfigPath = cfgPath
  138              , srActive     = activeRef
  139              }
  140   -- The /bg command is wired inside runCliTui (it needs the ISA registry +
  141   -- transcript handle that are constructed there). The AskReplyStore is the
  142   -- async bridge for /bg confirmation prompts: a forked /bg turn's ccPrompt
  143   -- routes through askHuman, and the CLI loop delivers the next input line
  144   -- as the answer via deliverNextAnswerAny. 0 = block indefinitely.
  145   askReply <- newAskReplyStore 0
  146   repoReg <- mkRepoRegistryHandle (reposFilePath paths)
  147   abortReg <- newSessionAbortRegistry
  148   -- The /new command: mints a fresh session and inserts a new tab into
  149   -- the TabsHandle. The ndInsertTab closure reads the old sid from
  150   -- srActive BEFORE swapping, inserts a new tab bound to the new session,
  151   -- then writes the new meta to srActive, and returns the old sid so the
  152   -- confirmation line can name it. No repo setup in the standalone CLI
  153   -- (no dispatcher wired). The repo registry is wired so @-r@ can resolve
  154   -- registered repo IDs to URLs.
  155   let newDeps = NewDeps
  156         { ndPaths = paths
  157         , ndCfg = pure cfg
  158         , ndAgentDefs = backends
  159         , ndChannelLabel = "cli"
  160         , ndOldMeta = readIORef activeRef
  161         , ndInsertTab = \_caps newMeta -> do
  162             oldMeta <- readIORef activeRef
  163             let oldSid = smId oldMeta
  164             _ <- insertTabH tabsH (BoundSession (smId newMeta)) KindProvider Nothing
  165             writeIORef activeRef newMeta
  166             pure oldSid
  167         , ndSetupRepo = Nothing
  168         , ndRepoReg = Just repoReg
  169         }
  170   let coreDeps = CoreCommandDeps
  171         { ccdVault       = rt
  172         , ccdProvider    = pr
  173         , ccdSession     = sr
  174         , ccdAgentDefs   = bAgentDefs backends
  175         , ccdCfgPath     = cfgPath
  176         , ccdPaths       = paths
  177         , ccdTabs        = tabsH
  178         , ccdTabCloseNotifier = noTabCloseNotifier
  179         , ccdAbortReg    = abortReg
  180         , ccdStopWriter  = mkStopTranscriptWriter paths Nothing
  181         , ccdModelWriter = mkModelTranscriptWriter paths Nothing
  182         , ccdRepoReg     = repoReg
  183         , ccdRepoSeam    = Nothing
  184         }
  185       registry = mkRegistry
  186         ( coreCommandSpecs coreDeps
  187           <> [ channelCommandSpec channelRt
  188              , newCommandSpec newDeps
  189              ]
  190         )
  191   harnessReg <- newHarnessRegistry
  192   tmuxRunner <- mkRealTmuxRunner
  193   runCliTui paths rt repoReg pr sr registry emptyChain backends tabsH autonomy askReply logger harnessReg tmuxRunner abortReg

    2 -- | The memory store backend. Disk is canonical: memory entries live as
    3 -- Markdown files under @config\/memory\/\<id\>.md@ (frontmatter + body, where
    4 -- the body is the memory content and tags/created/updated/session live in
    5 -- frontmatter). 'markdownMemoryBackend' reads by enumerating the directory
    6 -- and writes by atomic file replace + auto-commit; delete removes the file and
    7 -- commits. 'noneBackend' (in-memory) is kept for tests.
    8 --
    9 -- The git repo is the versioning + audit layer; model-authored writes
   10 -- (@MEMORY_STORE@ \/ @MEMORY_UPDATE@ \/ @MEMORY_DELETE@, which are Trusted
   11 -- file writes) auto-commit.

   13   ( MemoryBackend (..)
   14   , noneBackend
   15   , markdownMemoryBackend
   16   , encodeMemory
   17   , decodeMemory
   18   ) where

   47 -- | The memory store capability. Each operation is IO (the Markdown backend
   48 -- writes to disk + git); 'mbList' returns all memories sorted by id.
   49 data MemoryBackend = MemoryBackend
   50   { mbStore  :: MemoryEntry -> IO ()
   51   , mbRecall :: MemoryId -> IO (Maybe MemoryEntry)
   52   , mbList   :: IO [MemoryEntry]
   53   , mbUpdate :: MemoryEntry -> IO ()
   54   , mbDelete :: MemoryId -> IO ()
   55   }
   56 
   57 -- | The in-memory backend: a single 'IORef' over a 'Map'. Used by tests.
   58 noneBackend :: IO MemoryBackend
   59 noneBackend = do
   60   ref <- newIORef (Map.empty :: Map MemoryId MemoryEntry)
   61   pure MemoryBackend
   62     { mbStore  = \e -> modifyIORef' ref (Map.insert (meId e) e)
   63     , mbRecall = \mid -> Map.lookup mid <$> readIORef ref
   64     , mbList   = Map.elems <$> readIORef ref
   65     , mbUpdate = \e -> modifyIORef' ref (Map.insert (meId e) e)
   66     , mbDelete = modifyIORef' ref . Map.delete
   67     }
   68 
   69 -- | The Markdown backend. One file per memory under @dir@ (the
   70 -- @config/memory@ directory); writes are atomic (tmp → chmod 0600 → rename)
   71 -- and auto-committed to the config git repo; delete removes the file and
   72 -- commits. Reads enumerate the directory. Malformed files are skipped.
   73 markdownMemoryBackend :: FilePath -> ConfigRepo -> IO MemoryBackend
   74 markdownMemoryBackend dir repo = pure MemoryBackend
   75   { mbStore  = writeMemory dir repo
   76   , mbRecall = readMemory dir
   77   , mbList   = listMemories dir
   78   , mbUpdate = writeMemory dir repo
   79   , mbDelete = deleteMemory dir repo
   80   }
   81 
   82 -- | The filename for a memory: @\<id\>.md@.
   83 memoryFile :: FilePath -> MemoryId -> FilePath
   84 memoryFile dir mid = dir </> T.unpack (memoryIdText mid) <.> "md"
   85 
   86 -- | Encode a 'MemoryEntry' as a Markdown document (frontmatter + body, where
   87 -- the body is the memory content).
   88 encodeMemory :: MemoryEntry -> Text
   89 encodeMemory e = encodeDoc fm (meContent e)
   90   where
   91     fm = Map.fromList
   92       [ ("id", memoryIdText (meId e))
   93       , ("tags", renderTags (meTags e))
   94       , ("created_at", isoTime (meCreatedAt e))
   95       , ("updated_at", isoTime (meUpdatedAt e))
   96       , ("session", sessionIdText (meSession e))
   97       ]
   98 
   99 -- | Decode a Markdown document into a 'MemoryEntry'. Returns 'Nothing' if the
  100 -- id field is missing or fails 'mkMemoryId'.
  101 decodeMemory :: Text -> Maybe MemoryEntry
  102 decodeMemory content =
  103   case decodeDoc content of
  104     (fm, body) -> do
  105       midT <- fmLookup "id" fm
  106       mid  <- either (const Nothing) Just (mkMemoryId midT)
  107       Just MemoryEntry
  108         { meId = mid
  109         , meContent = body
  110         , meTags = fromMaybe [] (fmLookupList "tags" fm)
  111         , meCreatedAt = parseTime (fmLookup "created_at" fm)
  112         , meUpdatedAt = parseTime (fmLookup "updated_at" fm)
  113         , meSession = fromRight (mkSystemSessionId "unknown") (mkSessionId (fromMaybe "unknown" (fmLookup "session" fm)))
  114         }
  115 
  116 -- | Write one memory to disk (atomic) and auto-commit.
  117 writeMemory :: FilePath -> ConfigRepo -> MemoryEntry -> IO ()
  118 writeMemory dir repo e = do
  119   let path = memoryFile dir (meId e)
  120       tmp  = path <.> "tmp"
  121   TIO.writeFile tmp (encodeMemory e)
  122   setFileMode tmp 0o600
  123   renameFile tmp path
  124   let rel = "memory" </> (T.unpack (memoryIdText (meId e)) <.> "md")
  125   _ <- gitCommitAll repo rel ("seal: MEMORY write " <> memoryIdText (meId e))
  126   pure ()
  127 
  128 -- | Read one memory by id. Returns 'Nothing' if the file is absent or malformed.
  129 readMemory :: FilePath -> MemoryId -> IO (Maybe MemoryEntry)
  130 readMemory dir mid = do
  131   let path = memoryFile dir mid
  132   exists <- doesFileExist path
  133   if not exists
  134     then pure Nothing
  135     else do
  136       content <- TIO.readFile path
  137       pure (decodeMemory content)
  138 
  139 -- | Enumerate all memories in the directory, sorted by id. Malformed files
  140 -- are skipped.
  141 listMemories :: FilePath -> IO [MemoryEntry]
  142 listMemories dir = do
  143   entries <- listDirectory dir
  144   let mdFiles = [e | e <- entries, ".md" `T.isSuffixOf` T.pack e]
  145   mems <- forM mdFiles $ \e -> do
  146     content <- TIO.readFile (dir </> e)
  147     pure (decodeMemory content)
  148   pure (sortOn (memoryIdText . meId) (catMaybes mems))
  149 
  150 -- | Delete one memory file and auto-commit. Idempotent.
  151 deleteMemory :: FilePath -> ConfigRepo -> MemoryId -> IO ()
  152 deleteMemory dir repo mid = do
  153   let path = memoryFile dir mid
  154   exists <- doesFileExist path
  155   if not exists
  156     then pure ()
  157     else do
  158       removeFile path
  159       let rel = "memory" </> (T.unpack (memoryIdText mid) <.> "md")
  160       _ <- gitCommitAll repo rel ("seal: MEMORY delete " <> memoryIdText mid)
  161       pure ()
  162 
  163 -- | Render the tags list as a JSON array string for the frontmatter line.
  164 renderTags :: [Text] -> Text
  165 renderTags ts = TE.decodeUtf8 (BL.toStrict (encode (V.fromList (map String ts))))
  166 
  167 -- | Render a 'UTCTime' as an ISO-8601 string (UTC, with @Z@ suffix).
  168 isoTime :: UTCTime -> Text
  169 isoTime = T.pack . formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%SZ"
  170 
  171 -- | Parse an ISO-8601 'UTCTime' from a frontmatter value. Defaults to epoch 0
  172 -- when absent or unparseable.
  173 parseTime :: Maybe Text -> UTCTime
  174 parseTime Nothing    = epochZero
  175 parseTime (Just raw) = fromMaybe epochZero (parseTimeM True defaultTimeLocale "%Y-%m-%dT%H:%M:%SZ" (T.unpack raw))
  176 
  177 -- | The epoch fallback for missing/unparseable timestamps.
  178 epochZero :: UTCTime
  179 epochZero = UTCTime (fromGregorian 1970 1 1) (secondsToDiffTime 0)

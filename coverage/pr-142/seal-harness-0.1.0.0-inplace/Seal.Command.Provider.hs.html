   65 -- | Render the outcome of @/provider test@ for a provider labelled @label@.
   66 formatTestResult :: Text -> Either Text CompletionResponse -> Text
   67 formatTestResult label = \case
   68   Left e  -> label <> " test FAILED: " <> e
   69   Right r ->
   70     label <> " OK — model responded ("
   71       <> T.pack (show (uOutput (rsUsage r))) <> " output tokens, stop="
   72       <> T.pack (show (rsStop r)) <> ")"

   74 -- | Everything the @/provider@ handlers need: where config lives, the vault
   75 -- (for credentials), and an HTTP manager (for the live @test@ round-trip).
   76 -- The @prCallCounter@ is a process-lifetime monotonic counter shared across
   77 -- every Ollama provider instance built by 'Seal.Providers.Registry.resolveProvider'.
   78 -- Ollama tool-call responses carry no id, so Seal synthesizes @call_<i>@;
   79 -- the counter must outlive any single 'Ollama' value (which is rebuilt each
   80 -- turn) so ids stay unique across a whole session.
   81 data ProviderRuntime = ProviderRuntime
   82   { prConfigPath  :: FilePath
   83   , prVault       :: VaultRuntime
   84   , prManager     :: Manager
   85   , prCallCounter :: IORef Int
   86   }

   98 providerParserInfo :: ProviderRuntime -> ParserInfo CommandAction
   99 providerParserInfo pr =
  100   info (providerParser pr <**> helper)

  102     <> header   "provider — configure and test model providers"

  105 providerParser :: ProviderRuntime -> Parser CommandAction
  106 providerParser pr = hsubparser
  107   (  command "add"
  108        (info (addCmd pr <$> provArg)
  109              (progDesc "Store a provider API key (hidden prompt) in the vault"))
  110   <> command "login"
  111        (info (loginCmd pr <$> provArg)
  112              (progDesc "Log in via OAuth (Claude subscription) and store tokens"))
  113   <> command "list"
  114        (info (pure (listCmd pr))
  115              (progDesc "List known providers and their credential status"))
  116   <> command "test"
  117        (info (testCmd pr <$> provArg)
  118              (progDesc "Run a live round-trip to verify a provider works"))
  119   <> command "default"
  120        (info (defaultCmd pr <$> provArg)
  121              (progDesc "Set the default provider for new sessions"))
  122   <> command "remove"
  123        (info (removeCmd pr <$> provArg)
  124              (progDesc "Remove a provider's stored credential"))
  125   <> metavar "COMMAND"
  126   )

  133 withVaultHandle :: ProviderRuntime -> ChannelCaps -> (VaultHandle -> IO ()) -> IO ()
  134 withVaultHandle pr caps k = do
  135   mh <- readIORef (vrHandleRef (prVault pr))
  136   maybe (ccSend caps "vault not configured — run /vault setup") k mh

  138 withProvider :: ChannelCaps -> Text -> (KnownProvider -> IO ()) -> IO ()
  139 withProvider caps lbl k =
  140   maybe (ccSend caps (unknownProviderMsg lbl)) k (parseProvider lbl)
  141 
  142 unknownProviderMsg :: Text -> Text
  143 unknownProviderMsg lbl =
  144   "unknown provider: " <> lbl <> " (known: "
  145     <> T.intercalate ", " (map providerLabel knownProviders) <> ")"

  149 -- | Seed the default provider (if unset) and, only if that provider's
  150 -- section has no default model yet, seed its @[providers.<label>]@
  151 -- @default_model@. Never touches the global @default_model@ key, so a
  152 -- later @/model default@ isn't shadowed by a stale global value.
  153 seedProviderDefaults :: KnownProvider -> RuntimeConfig -> RuntimeConfig
  154 seedProviderDefaults kp fc0 =
  155   let fc1 = fc0 { rcDefaultProvider = rcDefaultProvider fc0 <|> Just (providerLabel kp) }
  156       lbl = providerLabel kp
  157   in if isJust (providerDefaultModel fc1 lbl)
  158        then fc1
  159        else upsertProvider lbl (\p -> p { pcDefaultModel = Just (modelText (defaultModelFor kp)) }) fc1

  161 addCmd :: ProviderRuntime -> Text -> CommandAction
  162 addCmd pr lbl = CommandAction $ \caps ->
  163   withProvider caps lbl $ \kp ->
  164     withVaultHandle pr caps $ \vh ->
  165       if providerLabel kp == "ollama"
  166         then addOllama pr caps vh kp
  167         else do
  168           val <- ccPromptSecret caps ("API key for " <> providerLabel kp <> ": ")
  169           res <- vhPut vh (vaultKeyName kp) (TE.encodeUtf8 val)
  170           case res of
  171             Left e   -> ccSend caps (vaultErrText e)
  172             Right () -> do
  173               _ <- updateRuntimeConfig (prConfigPath pr) (seedProviderDefaults kp)
  174               ccSend caps ("Stored API key for " <> providerLabel kp <> ".")
  175 
  176 -- | Ollama onboarding: prompt for the base URL (blank keeps the default),
  177 -- persist it to config, then prompt for an optional API key (blank => local,
  178 -- nothing stored). Seeds provider/model defaults like the generic path.
  179 addOllama :: ProviderRuntime -> ChannelCaps -> VaultHandle -> KnownProvider -> IO ()
  180 addOllama pr caps vh kp = do
  181   urlIn <- ccPrompt caps
  182              (AskPrompt ("Ollama base URL [" <> defaultOllamaBaseUrl <> "] (blank = default): ") [])
  183   let mUrl = if T.null (T.strip urlIn) then Nothing else Just (T.strip urlIn)
  184   keyIn <- ccPromptSecret caps "Ollama API key (blank for local): "
  185   keyRes <-
  186     if T.null keyIn
  187       then pure (Right ())
  188       else vhPut vh (vaultKeyName kp) (TE.encodeUtf8 keyIn)
  189   case keyRes of
  190     Left e   -> ccSend caps (vaultErrText e)
  191     Right () -> do
  192       _ <- updateRuntimeConfig (prConfigPath pr) (seedAll mUrl)
  193       ccSend caps "Configured ollama."
  194   where
  195     seedAll mUrl fc =
  196       let fc' = seedProviderDefaults kp fc
  197       in case mUrl of
  198            Nothing -> fc'
  199            Just u  -> upsertProvider (providerLabel kp) (\p -> p { pcBaseUrl = Just u }) fc'

  201 -- | Set the default provider for new sessions (writes @default_provider@).
  202 -- Only a valid provider label is accepted; no credential or vault is required,
  203 -- so a keyless local provider can be made the default before any key exists.
  204 -- Unlike the @<|>@ seeding on @/provider add@, this reassigns an existing
  205 -- default. The confirmation echoes the provider's resolved default model so it
  206 -- is clear what new sessions will start with.
  207 defaultCmd :: ProviderRuntime -> Text -> CommandAction
  208 defaultCmd pr lbl = CommandAction $ \caps ->
  209   withProvider caps lbl $ \kp -> do
  210     let plabel = providerLabel kp
  211     res <- updateRuntimeConfig (prConfigPath pr)
  212              (\fc -> fc { rcDefaultProvider = Just plabel })
  213     case res of
  214       Left e   -> ccSend caps e
  215       Right () -> do
  216         eCfg <- loadRuntimeConfig (prConfigPath pr)
  217         let ModelId m = resolveDefaultModel
  218                           (either (const Nothing) (`providerDefaultModel` plabel) eCfg)
  219                           plabel
  220         ccSend caps
  221           ("default provider set to " <> plabel <> " (model: " <> m <> ")")
  222 
  223 listCmd :: ProviderRuntime -> CommandAction
  224 listCmd pr = CommandAction $ \caps ->
  225   withVaultHandle pr caps $ \vh -> do
  226     eCfg <- loadRuntimeConfig (prConfigPath pr)
  227     let cfg = fromRight defaultRuntimeConfig eCfg
  228         def = rcDefaultProvider cfg
  229     configured <- configuredProviders (Just vh) cfg
  230     if null configured
  231       then ccSend caps "no providers configured — run /provider add <provider>"
  232       else mapM_ (reportOne caps vh def) configured
  233   where
  234     reportOne caps vh def kp = do
  235       eKey <- vhGet vh (vaultKeyName kp)
  236       auth <- authLabel vh kp eKey
  237       let mark = if Just (providerLabel kp) == def then " (default)" else ""
  238       ccSend caps (providerLabel kp <> mark <> " — " <> auth)
  239 
  240     authLabel vh kp eKey
  241       | providerLabel kp == "ollama" =
  242           pure $ case eKey of
  243             Right _          -> "auth: api-key"
  244             Left VaultLocked -> "auth: (vault locked)"
  245             _                -> "auth: none (local)"
  246       | otherwise = do
  247           eOAuth <- vhGet vh anthropicOAuthKey
  248           pure $ case (eOAuth, eKey) of
  249             (Right _, _)          -> "auth: oauth"
  250             (_, Right _)          -> "auth: api-key"
  251             (Left VaultLocked, _) -> "auth: (vault locked)"
  252             _                     -> "auth: none"

  254 -- | OAuth login flow. Anthropic-only for this milestone. Prints the authorize
  255 -- URL, best-effort opens the browser, reads the pasted CODE#STATE, exchanges
  256 -- it for tokens, and stores them in the vault. Tokens are never echoed.
  257 loginCmd :: ProviderRuntime -> Text -> CommandAction
  258 loginCmd pr lbl = CommandAction $ \caps ->
  259   withProvider caps lbl $ \kp ->
  260     if providerLabel kp /= "anthropic"
  261       then ccSend caps
  262              ("OAuth login is only supported for anthropic (got: " <> providerLabel kp <> ")")
  263       else withVaultHandle pr caps $ \vh -> do
  264         pkce <- newPkce
  265         let url = buildAuthorizeUrl pkce
  266         ccSend caps "Open this URL, approve access, then paste the code shown:"
  267         ccSend caps url
  268         openBrowser url
  269         pasted <- ccPrompt caps (AskPrompt "code: " [])
  270         let (code, state) = parsePastedCode pasted
  271         eTokens <- exchangeCode (prManager pr) pkce code state
  272         case eTokens of
  273           Left e       -> ccSend caps ("login failed: " <> e)
  274           Right tokens -> do
  275             res <- vhPut vh anthropicOAuthKey (serializeTokens tokens)
  276             case res of
  277               Left e   -> ccSend caps (vaultErrText e)
  278               Right () -> do
  279                 _ <- updateRuntimeConfig (prConfigPath pr) (seedProviderDefaults kp)
  280                 ccSend caps ("Logged in to " <> providerLabel kp <> " via OAuth.")
  281 
  282 -- | Best-effort browser open; failure is silently ignored (headless-friendly).
  283 openBrowser :: Text -> IO ()
  284 openBrowser url = do
  285   let opener = if os == "darwin" then "open" else "xdg-open"
  286   _ <- try @SomeException (callProcess opener [T.unpack url])
  287   pure ()

  289 testCmd :: ProviderRuntime -> Text -> CommandAction
  290 testCmd pr lbl = CommandAction $ \caps ->
  291   withProvider caps lbl $ \kp ->
  292     withVaultHandle pr caps $ \vh -> do
  293       eCfg <- loadRuntimeConfig (prConfigPath pr)
  294       let canonLbl = providerLabel kp
  295           model = case eCfg of
  296             Right c -> resolveDefaultModel (providerDefaultModel c canonLbl) canonLbl
  297             Left _  -> defaultModelFor kp
  298           baseUrl = fromMaybe defaultOllamaBaseUrl
  299                       (either (const Nothing) (`providerBaseUrl` "ollama") eCfg)
  300       eProv <- resolveProvider (Just vh) (prManager pr) baseUrl kp model (prCallCounter pr)
  301       case eProv of
  302         Left e   -> ccSend caps (formatTestResult (providerLabel kp) (Left e))
  303         Right sp -> do
  304           r <- completeSome sp (pingRequest model) :: IO (Either Text CompletionResponse)
  305           ccSend caps (formatTestResult (providerLabel kp) r)
  306 
  307 removeCmd :: ProviderRuntime -> Text -> CommandAction
  308 removeCmd pr lbl = CommandAction $ \caps ->
  309   withProvider caps lbl $ \kp ->
  310     withVaultHandle pr caps $ \vh -> do
  311       r1 <- deleteIfPresent vh (vaultKeyName kp)
  312       r2 <- deleteIfPresent vh anthropicOAuthKey
  313       case (r1, r2) of
  314         (Left e, _) -> ccSend caps (vaultErrText e)
  315         (_, Left e) -> ccSend caps (vaultErrText e)
  316         _           -> do
  317           _ <- updateRuntimeConfig (prConfigPath pr) (clearDefault kp)
  318           ccSend caps ("Removed credentials for " <> providerLabel kp <> ".")
  319   where
  320     -- | Delete a key; a missing key is not an error.
  321     deleteIfPresent vh k = do
  322       res <- vhDelete vh k
  323       pure $ case res of
  324         Left (VaultKeyNotFound _) -> Right ()
  325         other                     -> other
  326     clearDefault kp fc
  327       | rcDefaultProvider fc == Just (providerLabel kp) =
  328           fc { rcDefaultProvider = Nothing, rcDefaultModel = Nothing }
  329       | otherwise = fc

  334 -- | Resolve the active session's provider from the vault, or explain why
  335 -- not. Key bytes never surface: 'resolveProvider' returns an opaque
  336 -- 'SomeProvider'. Reads the session's 'smProvider'/'smModel' and the
  337 -- configured Ollama base URL (falling back to 'defaultOllamaBaseUrl').
  338 resolveSessionProvider
  339   :: ProviderRuntime -> SessionMeta -> IO (Either Text (SomeProvider, ModelId))
  340 resolveSessionProvider pr meta =
  341   case parseProvider (smProvider meta) of
  342     Nothing -> pure (Left ("unknown provider in session: " <> smProvider meta))
  343     Just kp -> do
  344       eCfg <- loadRuntimeConfig (prConfigPath pr)
  345       let baseUrl = fromMaybe defaultOllamaBaseUrl (either (const Nothing) (`providerBaseUrl` "ollama") eCfg)
  346           model   = ModelId (smModel meta)
  347       mh <- readIORef (vrHandleRef (prVault pr))
  348       fmap (fmap (, model)) (resolveProvider mh (prManager pr) baseUrl kp model (prCallCounter pr))

  350 -- | Resolve a provider+model from explicit labels (for AGENT_START, which
  351 -- builds a fresh AgentEnv from a def rather than the active session).
  352 resolveDefProvider
  353   :: ProviderRuntime -> Text -> ModelId -> IO (Either Text (SomeProvider, ModelId))
  354 resolveDefProvider pr providerLabel' model =
  355   case parseProvider providerLabel' of
  356     Nothing -> pure (Left ("unknown provider in agent def: " <> providerLabel'))
  357     Just kp -> do
  358       eCfg <- loadRuntimeConfig (prConfigPath pr)
  359       let baseUrl = fromMaybe defaultOllamaBaseUrl (either (const Nothing) (`providerBaseUrl` "ollama") eCfg)
  360       mh <- readIORef (vrHandleRef (prVault pr))
  361       fmap (fmap (, model)) (resolveProvider mh (prManager pr) baseUrl kp model (prCallCounter pr))

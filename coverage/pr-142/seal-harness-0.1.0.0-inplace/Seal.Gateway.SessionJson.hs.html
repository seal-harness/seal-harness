    2 -- | The JSON encoders for tabs and sessions, shared between
    3 -- 'Seal.Gateway.API' (the REST endpoints) and 'Seal.Gateway.ListsSnapshot'
    4 -- (the WS @lists@ frame + the @GET /api/lists@ body). Lives one level below
    5 -- 'Seal.Gateway.API' so 'Seal.Gateway.ListsSnapshot' can import these
    6 -- without creating a source-level module cycle with 'Seal.Gateway.API'
    7 -- (which imports 'Seal.Gateway.ListsSnapshot' for the /api/lists route).

   30 -- | One tab as JSON (the widened TabInfoWire shape the frontend expects).
   31 tabToJson :: Tab -> Value
   32 tabToJson t = object
   33   [ "index" .= tabIndexToInt (tIndex t)
   34   , "kind" .= tabKindWire (tKind t)
   35   , "label" .= tLabel t
   36   , "status" .= statusWire (tKind t) (tStatus t)
   37   , "session_id" .= sessionWire (tRef t)
   38   , "ext_modified" .= False
   39   , "stale" .= False
   40   , "origin" .= (Nothing :: Maybe Text)
   41   , "attach_command" .= (Nothing :: Maybe Text)
   42   ]
   43 
   44 -- | Map a 'TabKind' to the frontend's wire vocab.
   45 tabKindWire :: TabKind -> Text
   46 tabKindWire k = case k of
   47   KindHarness  -> "harness"
   48   KindProvider -> "session:provider"
   49   KindAi       -> "session:ai"
   50   KindShell    -> "shell"
   51   KindSsh      -> "shell:ssh"
   52   KindTmux     -> "tmux"
   53 
   54 -- | Map ('TabKind', 'TabStatus') to the frontend's status vocab.
   55 statusWire :: TabKind -> TabStatus -> Text
   56 statusWire _ Dead = "exited"
   57 statusWire KindHarness Live = "running"
   58 statusWire _ Live = "idle"
   59 
   60 -- | Derive the @session_id@ wire field from a 'TabRef'.
   61 sessionWire :: TabRef -> Maybe Text
   62 sessionWire (BoundSession sid) = Just (sessionIdText sid)
   63 sessionWire (BoundHarness _)   = Nothing

   65 -- | Map a 'SessionMeta' to the frontend's 'SessionInfo' JSON shape
   66 -- (camelCase). The on-disk 'SessionMeta' uses snake_case; the gateway maps
   67 -- to the frontend's shape without changing 'SessionMeta's instance.
   68 -- @description@ is the user-set title (persisted via @PUT
   69 -- /api/sessions/:id/description@); @autoSummary@ and @channelUserId@ are
   70 -- fields the backend doesn't track yet and are returned as @null@.
   71 -- @firstMessageSnippet@ is derived from the session's transcript (the
   72 -- first user message), so a session has a readable title before the user
   73 -- sets an explicit description.
   74 --
   75 -- @agent@ is the display label for the session's active agent (used by
   76 -- the sidebar / chat header). It prefers 'smAgentName' (set whenever an
   77 -- agent is effective — either a bound 'smAgent' or a one-off uploaded
   78 -- file's frontmatter id), falling back to 'smAgent'\'s id for
   79 -- backwards-compat with old session.json files that predate smAgentName.
   80 sessionInfoJson :: Maybe Text -> Maybe Text -> SessionMeta -> Value
   81 sessionInfoJson mSnippet mLastUserMessageAt m = object
   82   [ "id" .= sessionIdText (smId m)
   83   , "agent" .= ( smAgentName m <|> (agentDefIdText <$> smAgent m) )
   84   , "runtime" .= ("session:" <> smProvider m)
   85   , "model" .= smModel m
   86   , "lastActive" .= smLastActive m
   87   , "createdAt" .= smCreatedAt m
   88   , "description" .= smDescription m
   89   , "autoSummary" .= (Nothing :: Maybe Text)
   90   , "firstMessageSnippet" .= mSnippet
   91   , "channel" .= smChannel m
   92   , "channelUserId" .= (Nothing :: Maybe Text)
   93   , "lastUserMessageAt" .= mLastUserMessageAt
   94   ]
   95 
   96 -- | Build the 'SessionInfo' JSON for a session, reading the first user
   97 -- message snippet and the last user-message timestamp from the transcript
   98 -- so the session has a default title before the user sets an explicit
   99 -- description and the sidebar can sort tabs by last-user-message time.
  100 sessionInfoJsonWithSnippet :: SealPaths -> SessionMeta -> IO Value
  101 sessionInfoJsonWithSnippet paths m = do
  102   mSnippet <- firstUserMessageSnippet paths (smId m)
  103   mLastUserMessageAt <- lastUserMessageAt paths (smId m)
  104   pure (sessionInfoJson mSnippet mLastUserMessageAt m)

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The per-conversation tab cursor store. Each conversation (a Telegram
    3 -- chat, a Signal conversation, a TUI session) has a cursor pointing at the
    4 -- tab it's currently focused on. The tab list is shared (one
    5 -- 'TabsHandle' in the gateway); the cursor is per-conversation so
    6 -- @/tab focus N@ on Telegram only affects that Telegram conversation,
    7 -- not other conversations or the TUI.
    8 --
    9 -- The store is a 'TVar' backed 'Map' from 'ConversationKey' to 'TabRef'.
   10 -- 'ConversationKey' is 'ChannelKind' × 'ConversationId' — the
   11 -- server-derived conversation identity (never user-supplied), so a sender
   12 -- cannot forge a cursor key to hijack another conversation's tab.
   13 --
   14 -- == Persistence
   15 --
   16 -- The cursor store survives a @seal serve@ (or standalone
   17 -- @seal telegram@ / @seal signal@) restart via the optional 'csSave' hook
   18 -- (mirroring 'Seal.Tabs.TabsHandle.thSave'). A non-persisting store
   19 -- ('newCursorStore', 'csSave' = 'Nothing') is used by tests; a persisting
   20 -- store ('newPersistingCursorStore') writes the full current map to
   21 -- @\<state\>\/cursors.json@ atomically (0600) after every mutation, and is
   22 -- loaded + seeded at boot so an existing conversation re-resolves to its
   23 -- prior session (carrying the user's @\/model use@ choice) instead of a
   24 -- fresh default session. Without persistence, a restart loses every
   25 -- conversation → tab binding and the loop mints a brand-new default-model
   26 -- session on the next message — the "model change lost on restart" bug.
   27 module Seal.Channels.Cursor
   28   ( CursorStore
   29   , newCursorStore
   30   , newPersistingCursorStore
   31   , seedCursorStore
   32   , cursorLookup
   33   , cursorSet
   34   , cursorClear
   35   , cursorClearAll
   36   , cursorMigrateAll
   37   , snapshotCursor
   38   ) where
   39

   52 -- | The live cursor store. Backed by a 'TVar' so concurrent reads/writes
   53 -- are race-safe (each STM operation is one transaction). The optional
   54 -- 'csSave' action is invoked after a successful mutation so the map
   55 -- persists to disk; 'Nothing' for a plain 'newCursorStore' (tests,
   56 -- non-persisting modes) and @Just (... saveCursorMap path ...)@ for
   57 -- 'newPersistingCursorStore'. Mirrors 'Seal.Tabs.TabsHandle'.
   58 data CursorStore = CursorStore
   59   { csVar  :: TVar (Map ConversationKey TabRef)
   60   , csSave :: Maybe (IO ())
   61   }
   62 
   63 -- | A conversation identity: 'ChannelKind' × 'ConversationId'.
   64 -- Re-exported here for convenience; the key type is 'ConversationKey'.
   65 type ConversationKey = (Text, Text)
   66   -- ^ (channel-kind-text, conversation-id-text). We use the text forms
   67   -- rather than the structured types so the store doesn't depend on
   68   -- 'ChannelKind' or 'ConversationId' directly (keeping the module
   69   -- lightweight). Callers mint the key from 'MessageSource'.
   70 
   71 -- | Build a new empty non-persisting cursor store. Used by tests and any
   72 -- mode that does not need cross-restart persistence.
   73 newCursorStore :: IO CursorStore
   74 newCursorStore = CursorStore <$> newTVarIO Map.empty <*> pure Nothing
   75 
   76 -- | Create a 'CursorStore' that persists every mutation to @path@ (atomic
   77 -- write, 0600, MVar-serialized via 'Seal.Util.AtomicJson.saveJsonAtomic').
   78 -- Used by 'Seal.Command.Serve.runServeMain' and the standalone
   79 -- @seal telegram@ / @seal signal@ entry points so the cursor map survives
   80 -- a restart. The save action snapshots the current map inside itself
   81 -- (so the last writer wins with a consistent view, even if mutations
   82 -- interleave the save).
   83 newPersistingCursorStore :: FilePath -> IO CursorStore
   84 newPersistingCursorStore path = do
   85   tv <- newTVarIO Map.empty
   86   let store = CursorStore { csVar = tv, csSave = Just (saveAction store) }
   87       saveAction s = saveCursorMap path =<< snapshotCursor s
   88   pure store
   89 
   90 -- | Replace the handle's map in one STM transaction. Used at boot to
   91 -- seed the in-memory store from the persisted @cursors.json@ (after the
   92 -- caller has dropped stale entries, if desired). Does NOT persist (the
   93 -- caller is loading FROM disk — writing back would be a redundant no-op).
   94 seedCursorStore :: CursorStore -> Map ConversationKey TabRef -> IO ()
   95 seedCursorStore s m = atomically (writeTVar (csVar s) m)
   96 
   97 -- | Snapshot the current map. Used by the save action and tests.
   98 snapshotCursor :: CursorStore -> IO (Map ConversationKey TabRef)
   99 snapshotCursor s = readTVarIO (csVar s)
  100 
  101 -- | Look up the tab a conversation is focused on. 'Nothing' when the
  102 -- conversation has no cursor yet (first message — the caller should create
  103 -- a new tab and set the cursor).
  104 cursorLookup :: CursorStore -> ConversationKey -> IO (Maybe TabRef)
  105 cursorLookup s key =
  106   Map.lookup key <$> readTVarIO (csVar s)
  107 
  108 -- | Set (or replace) a conversation's focused tab. Called when the
  109 -- conversation sends its first message (new tab created) or when the user
  110 -- runs @/tab focus N@. Persists via 'csSave' after the STM commit.
  111 cursorSet :: CursorStore -> ConversationKey -> TabRef -> IO ()
  112 cursorSet s key ref = do
  113   atomically $ do
  114     m <- readTVar (csVar s)
  115     writeTVar (csVar s) (Map.insert key ref m)
  116   persistCursor s
  117 
  118 -- | Clear a conversation's cursor (e.g. when the channel disconnects).
  119 -- Harmless if the key was never set. Persists via 'csSave'.
  120 cursorClear :: CursorStore -> ConversationKey -> IO ()
  121 cursorClear s key = do
  122   atomically $ do
  123     m <- readTVar (csVar s)
  124     writeTVar (csVar s) (Map.delete key m)
  125   persistCursor s
  126 
  127 -- | Clear every conversation whose cursor points at @ref@. Used when a tab
  128 -- is closed: the closed tab's 'TabRef' is stale, so any conversation still
  129 -- focused on it should drop the cursor (the next message will create a
  130 -- fresh tab). Single STM transaction — race-safe vs concurrent
  131 -- cursorLookup/cursorSet. Persists via 'csSave'.
  132 cursorClearAll :: CursorStore -> TabRef -> IO ()
  133 cursorClearAll s ref = do
  134   atomically $ do
  135     m <- readTVar (csVar s)
  136     writeTVar (csVar s) (Map.filter (/= ref) m)
  137   persistCursor s
  138 
  139 -- | Migrate every conversation whose cursor equals @oldRef@ to @newRef@.
  140 -- Used by @\/new@ on inbox channels: when a tab is rebound to a fresh
  141 -- session, every conversation focused on that tab follows the rebind (per
  142 -- the user's model: a tab has one session at a time; all channels focused
  143 -- on the tab follow it to the new session). Returns the count of migrated
  144 -- cursors (for the confirmation line / observability). Single STM
  145 -- transaction — race-safe vs concurrent cursorLookup/cursorSet. Persists
  146 -- via 'csSave'.
  147 cursorMigrateAll :: CursorStore -> TabRef -> TabRef -> IO Int
  148 cursorMigrateAll s oldRef newRef = do
  149   count <- atomically $ do
  150     m <- readTVar (csVar s)
  151     let (matched, rest) = Map.partition (== oldRef) m
  152         m' = Map.map (const newRef) matched <> rest
  153     writeTVar (csVar s) m'
  154     pure (Map.size matched)
  155   persistCursor s
  156   pure count
  157 
  158 -- | Run the persist action (if any) after a successful mutation. A save
  159 -- failure is logged to stderr (ids + error only — no conversation content)
  160 -- and swallowed — the in-memory store stays authoritative within the
  161 -- session; the next successful mutation will retry the save (writing the
  162 -- full current map, so a missed save self-heals). Mirrors
  163 -- 'Seal.Tabs.persistIf'.
  164 persistCursor :: CursorStore -> IO ()
  165 persistCursor s =
  166   case csSave s of
  167     Nothing  -> pure ()
  168     Just act -> act `catch` \e ->
  169       globalLogIO WarningS ("[persist] cursors.json save failed: " <> ls (T.pack (show (e :: SomeException))))

   21 -- | A nested sub-record that stands in for "large amounts of configuration."
   22 -- Demonstrates nested config parsing via '%.:' (config file) and '%::' (CLI).
   23 data ServerConfig = ServerConfig
   24   { _serverConfig_host :: !Text
   25   , _serverConfig_port :: !Int
   26   } deriving (Eq, Show)
   27 
   28 makeLenses ''ServerConfig
   29 
   30 defaultServerConfig :: ServerConfig
   31 defaultServerConfig = ServerConfig
   32   { _serverConfig_host = "127.0.0.1"
   33   , _serverConfig_port = 8080
   34   }
   35 
   36 instance FromJSON (ServerConfig -> ServerConfig) where
   37   parseJSON = withObject "ServerConfig" $ \o -> id
   38     <$< serverConfig_host ..: "host" % o
   39     <*< serverConfig_port ..: "port" % o
   40 
   41 instance ToJSON ServerConfig where
   42   toJSON s = object
   43     [ "host" .= _serverConfig_host s
   44     , "port" .= _serverConfig_port s
   45     ]
   46 
   47 pServerConfig :: MParser ServerConfig
   48 pServerConfig = id
   49   <$< serverConfig_host .:: strOption
   50       ( long "server-host"
   51       <> help "Server host" )
   52   <*< serverConfig_port .:: option auto
   53       ( long "server-port"
   54       <> metavar "PORT"
   55       <> help "Server port" )

   57 -- | The resolved program configuration. The '_config_command' field is
   58 -- CLI-only (it is not part of the 'FromJSON'/'ToJSON' instances) and holds
   59 -- the selected subcommand plus its per-command arguments.
   60 data Config = Config
   61   { _config_command :: !Command
   62   , _config_logLevel :: !Text
   63   , _config_server :: !ServerConfig
   64   } deriving (Eq, Show)
   65 
   66 makeLenses ''Config
   67 
   68 defaultConfig :: Config
   69 defaultConfig = Config
   70   { _config_command = CommandNoOp
   71   , _config_logLevel = "Info"
   72   , _config_server = defaultServerConfig
   73   }
   74 
   75 -- | Config-file 'FromJSON' instance (update-function style). The '_config_command'
   76 -- field is intentionally excluded: commands are not config-file data.
   77 instance FromJSON (Config -> Config) where
   78   parseJSON = withObject "Config" $ \o -> id
   79     <$< config_logLevel ..: "log-level" % o
   80     <*< config_server %.: "server" % o
   81 
   82 -- | Hand-written 'ToJSON' for '--print-config'. Excludes '_config_command'.
   83 instance ToJSON Config where
   84   toJSON c = object
   85     [ "log-level" .= _config_logLevel c
   86     , "server" .= _config_server c
   87     ]
   88 
   89 -- | CLI parser. The command is a flat field parsed with '.::' from an
   90 -- 'hsubparser'; everything else mirrors the config-file structure.
   91 pConfig :: MParser Config
   92 pConfig = id
   93   <$< config_command .:: pCommand
   94   <*< config_logLevel .:: strOption
   95       ( long "log-level"
   96       <> help "Minimum log severity (Debug|Info|Notice|Warning|Error)" )
   97   <*< config_server %:: pServerConfig

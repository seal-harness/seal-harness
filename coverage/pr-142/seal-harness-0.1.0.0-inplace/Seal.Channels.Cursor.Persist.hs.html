    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Cursor-map persistence — the 'CursorStore''s @ConversationKey → TabRef@
    3 -- map survives a @seal serve@ (or standalone @seal telegram@ / @seal
    4 -- signal@) restart. Written atomically (0600) to @\<state\>\/cursors.json@
    5 -- on every cursor mutation via 'Seal.Util.AtomicJson.saveJsonAtomic';
    6 -- loaded at boot by 'loadCursorMap' so a conversation re-resolves to its
    7 -- prior session (carrying the user's @\/model use@ choice) instead of a
    8 -- fresh default session.
    9 --
   10 -- The map is encoded as a JSON array of @{"key": [...], "ref": {...}}@
   11 -- pairs (JSON object keys must be strings, and 'ConversationKey' is a
   12 -- '(Text, Text)' tuple, so it cannot key a JSON object directly). Every
   13 -- 'TabRef' id is re-validated on load ('mkSessionId' / 'parseHarnessId')
   14 -- and entries whose id fails to re-parse are dropped (defense-in-depth
   15 -- against a tampered file), mirroring 'Seal.Tabs.Persist.filterValidTabs'.
   16 module Seal.Channels.Cursor.Persist
   17   ( saveCursorMap
   18   , loadCursorMap
   19   ) where
   20 
   21 import Data.Aeson qualified as A
   22 import Data.Map.Strict (Map)
   23 import Data.Map.Strict qualified as Map
   24 import Data.Text (Text)
   25 import System.Directory (doesFileExist)
   26 
   27 import Katip (Severity (..))
   28 import Seal.Core.Types (mkSessionId, sessionIdText)
   29 import Seal.Harness.Id (parseHarnessId, harnessIdToText)
   30 import Seal.Logging.Global (globalLogIO)
   31 import Seal.Tabs.Types (TabRef (..))
   32 import Seal.Util.AtomicJson (saveJsonAtomic)
   33 import Seal.Util.StrictIO (decodeFileStrict)

   35 -- | The on-disk wire type: a list of key/ref pairs. The map is converted
   36 -- to/from this list on save/load. 'ConversationKey' is a '(Text, Text)'
   37 -- tuple; aeson encodes a tuple as a JSON array, so 'cwKey' serializes as
   38 -- @["telegram","12345"]@. 'TabRef' already has 'ToJSON'/''FromJSON'
   39 -- instances (used by @tabs.json@); reused here.
   40 data CursorWire = CursorWire
   41   { cwKey :: (Text, Text)
   42   , cwRef :: TabRef
   43   }
   44 
   45 -- Aeson derives the tuple and TabRef codecs via their existing instances.
   46 -- We hand-write the 'CursorWire' codec (not derive) to control field
   47 -- names and keep the on-disk shape explicit.
   48 
   49 instance A.ToJSON CursorWire where
   50   toJSON cw = A.object
   51     [ "key" A..= cwKey cw
   52     , "ref" A..= cwRef cw
   53     ]
   54 
   55 instance A.FromJSON CursorWire where
   56   parseJSON = A.withObject "CursorWire" $ \o -> CursorWire
   57     <$> o A..: "key"
   58     <*> o A..: "ref"
   59 
   60 -- | Save a cursor map to @path@ atomically (0600, MVar-serialized via
   61 -- 'saveJsonAtomic'). Never throws; a write failure is the caller's
   62 -- responsibility to 'catch' (the persisting cursor store does so and logs
   63 -- a warning, matching the tab-list pattern).
   64 saveCursorMap :: FilePath -> Map (Text, Text) TabRef -> IO ()
   65 saveCursorMap path m =
   66   saveJsonAtomic path (A.encode (map (uncurry CursorWire) (Map.toList m)))
   67 
   68 -- | Load the cursor map from @path@. Missing file -> 'Nothing' (fresh
   69 -- empty store). Corrupt JSON -> 'Nothing' + a stderr warning (ids + error
   70 -- type only — no conversation content). Every 'TabRef' id is re-validated;
   71 -- entries with an unparseable id are dropped (defense-in-depth against a
   72 -- tampered @cursors.json@).
   73 loadCursorMap :: FilePath -> IO (Maybe (Map (Text, Text) TabRef))
   74 loadCursorMap path = do
   75   exists <- doesFileExist path
   76   if not exists
   77     then pure Nothing
   78     else do
   79       mWires <- decodeFileStrict path :: IO (Maybe [CursorWire])
   80       case mWires of
   81         Nothing -> do
   82           globalLogIO WarningS "could not parse cursors.json; using empty cursor store"
   83           pure Nothing
   84         Just ws -> pure (Just (filterValid (Map.fromList [ (cwKey w, cwRef w) | w <- ws ])))
   85 
   86 -- | Drop entries whose 'TabRef' id fails to re-parse. The on-disk ids were
   87 -- validated when written, so a parse failure here means the file was
   88 -- tampered or written by a future/incompatible version — skip rather than
   89 -- error so a corrupt file self-heals on the next save. Mirrors
   90 -- 'Seal.Tabs.Persist.filterValidTabs'.
   91 filterValid :: Map (Text, Text) TabRef -> Map (Text, Text) TabRef
   92 filterValid = Map.filter validRef
   93   where
   94     validRef (BoundSession sid) = case mkSessionId (sessionIdText sid) of Right _ -> True; Left _ -> False
   95     validRef (BoundHarness hid) = case parseHarnessId (harnessIdToText hid) of Right _ -> True; Left _ -> False

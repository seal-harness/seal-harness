    1 -- | Opaque secret values. Constructors are intentionally NOT exported, there
    2 -- are no JSON/serialization instances, and 'Show' is redacted. The only way
    3 -- to observe the payload is the CPS accessor, which scopes the secret to a
    4 -- single continuation so it cannot leak into a longer-lived binding.
    5 module Seal.Security.Secrets
    6   ( ApiKey, BearerToken, PairingCode, RefreshToken, SecretKey
    7   , mkApiKey, mkBearerToken, mkPairingCode, mkRefreshToken, mkSecretKey
    8   , withApiKey, withBearerToken, withPairingCode, withRefreshToken, withSecretKey
    9   ) where
   10 
   11 import Data.ByteString (ByteString)
   12 import Data.Text (Text)
   13 
   14 newtype ApiKey      = ApiKey      ByteString
   15 newtype BearerToken = BearerToken ByteString
   16 newtype RefreshToken = RefreshToken ByteString
   17 newtype SecretKey   = SecretKey   ByteString
   18 newtype PairingCode = PairingCode Text
   19 
   20 instance Show ApiKey      where show _ = "ApiKey <redacted>"
   21 instance Show BearerToken where show _ = "BearerToken <redacted>"
   22 instance Show RefreshToken where show _ = "RefreshToken <redacted>"
   23 instance Show SecretKey   where show _ = "SecretKey <redacted>"
   24 instance Show PairingCode where show _ = "PairingCode <redacted>"
   25 
   26 mkApiKey :: ByteString -> ApiKey
   27 mkApiKey = ApiKey
   28 
   29 mkBearerToken :: ByteString -> BearerToken
   30 mkBearerToken = BearerToken
   31 
   32 mkRefreshToken :: ByteString -> RefreshToken
   33 mkRefreshToken = RefreshToken
   34 
   35 mkSecretKey :: ByteString -> SecretKey
   36 mkSecretKey = SecretKey
   37 
   38 mkPairingCode :: Text -> PairingCode
   39 mkPairingCode = PairingCode
   40 
   41 withApiKey :: ApiKey -> (ByteString -> r) -> r
   42 withApiKey (ApiKey b) f = f b
   43 
   44 withBearerToken :: BearerToken -> (ByteString -> r) -> r
   45 withBearerToken (BearerToken b) f = f b
   46 
   47 withRefreshToken :: RefreshToken -> (ByteString -> r) -> r
   48 withRefreshToken (RefreshToken b) f = f b
   49 
   50 withSecretKey :: SecretKey -> (ByteString -> r) -> r
   51 withSecretKey (SecretKey b) f = f b
   52 
   53 withPairingCode :: PairingCode -> (Text -> r) -> r
   54 withPairingCode (PairingCode t) f = f t

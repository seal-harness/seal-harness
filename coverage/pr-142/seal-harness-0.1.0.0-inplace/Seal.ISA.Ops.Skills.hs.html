    2 -- | The Skills opcode group: SKILL_WRITE, SKILL_LOAD, SKILL_LIST, SKILL_DELETE.
    3 -- All Audited — the dispatcher writes both the session transcript and the
    4 -- Audited log; the opcodes mutate the in-memory/Markdown backend (the
    5 -- materialized view). 'orRecorded' carries the secret-free 'SkillId' + op name
    6 -- + @was_new@ flag (so the audit log distinguishes create vs update); the
    7 -- skill DESCRIPTION and BODY are agent-visible data (not a vault secret) and
    8 -- are recorded in full in both logs.
    9 --
   10 -- 'SKILL_WRITE' is an upsert: if the skill already exists, its description
   11 -- and/or body are updated (the original 'skSession' provenance and
   12 -- 'skCreatedAt' are preserved; only 'skUpdatedAt' is bumped); if not, a
   13 -- fresh skill is created. This merges the former SKILL_CREATE + SKILL_UPDATE
   14 -- into a single opcode, eliminating one failure path.
   15 module Seal.ISA.Ops.Skills
   16   ( skillWriteOp
   17   , skillLoadOp
   18   , skillListOp
   19   , skillDeleteOp
   20   ) where

   38 -- | Build a JSON-Schema object with a single required string property.
   39 singleStringSchema :: Text -> Text -> Value
   40 singleStringSchema fieldName fieldDesc =
   41   object
   42     [ "type" .= ("object" :: Text)
   43     , "properties" .= object
   44         [fromText fieldName .= object
   45            [ "type" .= ("string" :: Text)
   46            , "description" .= fieldDesc
   47            ]]
   48     , "required" .= ([fieldName] :: [Text])
   49     ]
   50 
   51 -- | Extract the @id@ string field from a JSON object.
   52 idField :: Value -> Maybe Text
   53 idField = parseMaybe (withObject "in" (.: "id"))
   54 
   55 -- | Extract the @description@ string field (defaults to empty when absent).
   56 descriptionField :: Value -> Text
   57 descriptionField v = fromMaybe "" (parseMaybe (withObject "in" (.: "description")) v)
   58 
   59 -- | Extract the @body@ string field (defaults to empty when absent).
   60 bodyField :: Value -> Text
   61 bodyField v = fromMaybe "" (parseMaybe (withObject "in" (.: "body")) v)
   62 
   63 -- | Extract the optional @group@ string field. 'Nothing' when absent or
   64 -- empty (after stripping). A skill's group controls its on-disk location
   65 -- (@config\/skills\/\<group\>\/\<id\>.md@) and its display grouping in the
   66 -- @\<available_skills\>@ catalog.
   67 groupField :: Value -> Maybe Text
   68 groupField v = do
   69   raw <- parseMaybe (withObject "in" (.:? "group")) v
   70   t <- raw
   71   let t' = T.strip t
   72   if T.null t' then Nothing else Just t'

   74 -- | SKILL_WRITE: upsert a skill by id. If the skill already exists, its
   75 -- description, body, and group are updated (the original 'skSession'
   76 -- provenance and 'skCreatedAt' are preserved; only 'skUpdatedAt' is
   77 -- bumped); if not, a fresh skill is created with the current session as
   78 -- provenance. The description and body are recorded in full (agent-visible
   79 -- data); 'orRecorded' carries the id + op name + description + body +
   80 -- @was_new@ (secret-free).
   81 skillWriteOp :: SkillBackend -> SessionId -> Opcode
   82 skillWriteOp backend session = TrustedOpcode
   83   { toName = OpName "SKILL_WRITE"
   84   , toTrust = Trusted
   85   , toDesc = "Create or update an agent skill by id (upsert; preserves provenance on update)."
   86   , toInSchema = object
   87       [ "type" .= ("object" :: Text)
   88       , "properties" .= object
   89           [ fromText "id" .= object
   90               [ "type" .= ("string" :: Text)
   91               , "description" .= ("Skill id ([A-Za-z0-9_-]+)." :: Text)
   92               ]
   93           , fromText "description" .= object
   94               [ "type" .= ("string" :: Text)
   95               , "description" .= ("Short human-readable description of the skill." :: Text)
   96               ]
   97           , fromText "body" .= object
   98               [ "type" .= ("string" :: Text)
   99               , "description" .= ("The skill body (Markdown). Agent-visible." :: Text)
  100               ]
  101           , fromText "group" .= object
  102               [ "type" .= ("string" :: Text)
  103               , "description" .= ("Optional category for grouping in the available-skills catalog. When set, the skill is stored under config/skills/<group>/<id>.md; when omitted, the skill keeps its existing group (or is ungrouped on create)." :: Text)
  104               ]
  105           ]
  106       , "required" .= (["id", "description", "body"] :: [Text])
  107       ]
  108   , toOutSchema = object []
  109   , toAuthorize = maybe (Left "SKILL_WRITE requires {id:string}") checkId . idField
  110   , toRun = \_ v -> do
  111       let mId = idField v >>= either (const Nothing) Just . mkSkillId
  112           mNewGroup = groupField v
  113       case mId of
  114         Nothing -> pure (OpResult [TrpText "invalid skill id"] True (object []))
  115         Just sid -> do
  116           mExisting <- liftIO (sbRead backend sid)
  117           now <- liftIO getCurrentTime
  118           let (skill, wasNew) = case mExisting of
  119                 Just existing ->
  120                   ( existing
  121                       { skDescription = descriptionField v
  122                       , skBody = bodyField v
  123                       , skGroup = case mNewGroup of
  124                           Just g  -> Just g
  125                           Nothing -> skGroup existing
  126                       , skUpdatedAt = now
  127                       }
  128                   , False
  129                   )
  130                 Nothing ->
  131                   ( Skill
  132                       { skId = sid
  133                       , skDescription = descriptionField v
  134                       , skBody = bodyField v
  135                       , skGroup = mNewGroup
  136                       , skCreatedAt = now
  137                       , skUpdatedAt = now
  138                       , skSession = session
  139                       }
  140                   , True
  141                   )
  142           liftIO (sbCreate backend skill)
  143           let recorded = object
  144                 [ "id" .= skillIdText sid
  145                 , "description" .= skDescription skill
  146                 , "body" .= skBody skill
  147                 , "group" .= skGroup skill
  148                 , "created_at" .= skCreatedAt skill
  149                 , "updated_at" .= skUpdatedAt skill
  150                 , "session" .= skSession skill
  151                 , "was_new" .= wasNew
  152                 ]
  153           pure (OpResult [TrpText (if wasNew then "created" else "updated")] False recorded)
  154   }
  155   where
  156     checkId t = either (Left . ("invalid skill id: " <>)) (const (Right ())) (mkSkillId t)

  158 -- | SKILL_LOAD: return one skill by id. Audited — the agent reading its own
  159 -- skills is an evolutionary event worth logging, with secret-free metadata.
  160 -- The skill body is returned to the model (agent-visible) and recorded in full.
  161 skillLoadOp :: SkillBackend -> Opcode
  162 skillLoadOp backend = TrustedOpcode
  163   { toName = OpName "SKILL_LOAD"
  164   , toTrust = Trusted
  165   , toDesc = "Load one agent skill by id into the current session."
  166   , toInSchema = singleStringSchema "id" "The skill id to load."
  167   , toOutSchema = object []
  168   , toAuthorize = maybe (Left "SKILL_LOAD requires {id:string}") checkId . idField
  169   , toRun = \_ v -> do
  170       let mId = idField v >>= either (const Nothing) Just . mkSkillId
  171       case mId of
  172         Nothing -> pure (OpResult [TrpText "invalid skill id"] True (object []))
  173         Just sid -> do
  174           mSkill <- liftIO (sbRead backend sid)
  175           case mSkill of
  176             Nothing -> pure (OpResult [TrpText "skill not found"] True (object ["id" .= skillIdText sid]))
  177             Just s  -> do
  178               let rendered = "# " <> skillIdText (skId s) <> "\n\n"
  179                       <> skDescription s <> "\n\n---\n\n" <> skBody s
  180                   recorded = object
  181                     [ "id" .= skillIdText sid
  182                     , "description" .= skDescription s
  183                     , "body" .= skBody s
  184                     , "group" .= skGroup s
  185                     , "updated_at" .= skUpdatedAt s
  186                     , "session" .= skSession s
  187                     ]
  188               pure (OpResult [TrpText rendered] False recorded)
  189   }
  190   where
  191     checkId t = either (Left . ("invalid skill id: " <>)) (const (Right ())) (mkSkillId t)

  193 -- | SKILL_DELETE: remove a skill by id. Idempotent (deleting a missing id is
  194 -- a success with a "not present" message, not an error). Mirrors
  195 -- 'Seal.ISA.Ops.Memory.memoryDeleteOp'.
  196 skillDeleteOp :: SkillBackend -> Opcode
  197 skillDeleteOp backend = TrustedOpcode
  198   { toName = OpName "SKILL_DELETE"
  199   , toTrust = Trusted
  200   , toDesc = "Delete an agent skill by id (idempotent)."
  201   , toInSchema = singleStringSchema "id" "The skill id to delete."
  202   , toOutSchema = object []
  203   , toAuthorize = maybe (Left "SKILL_DELETE requires {id:string}") checkId . idField
  204   , toRun = \_ v -> do
  205       let mId = idField v >>= either (const Nothing) Just . mkSkillId
  206       case mId of
  207         Nothing -> pure (OpResult [TrpText "invalid skill id"] True (object []))
  208         Just sid -> do
  209           mExisting <- liftIO (sbRead backend sid)
  210           liftIO (sbDelete backend sid)
  211           let msg = case mExisting of
  212                 Nothing -> "deleted (was not present)"
  213                 Just _  -> "deleted"
  214               recorded = object ["id" .= skillIdText sid]
  215           pure (OpResult [TrpText msg] False recorded)
  216   }
  217   where
  218     checkId t = either (Left . ("invalid skill id: " <>)) (const (Right ())) (mkSkillId t)

  220 -- | SKILL_LIST: enumerate all defined skills (id + description). Audited —
  221 -- listing is an evolutionary event worth logging, with secret-free metadata
  222 -- (no skill bodies in the recorded payload; the model sees only the
  223 -- id+description summary).
  224 skillListOp :: SkillBackend -> Opcode
  225 skillListOp backend = TrustedOpcode
  226   { toName = OpName "SKILL_LIST"
  227   , toTrust = Trusted
  228   , toDesc = "List all defined agent skills (id + description)."
  229   , toInSchema = object
  230       [ "type" .= ("object" :: Text)
  231       , "properties" .= object []
  232       ]
  233   , toOutSchema = object []
  234   , toAuthorize = const (Right ())
  235   , toRun = \_ _ -> do
  236       allSkills <- liftIO (sbList backend)
  237       let rendered = case allSkills of
  238             [] -> "(no skills defined)"
  239             _  -> T.intercalate "\n"
  240                     [ skillIdText (skId s)
  241                         <> maybe "" (\g -> " [" <> g <> "]") (skGroup s)
  242                         <> ": " <> skDescription s
  243                     | s <- allSkills ]
  244           recorded = object
  245             [ "count" .= length allSkills
  246             , "ids" .= fmap (skillIdText . skId) allSkills
  247             ]
  248       pure (OpResult [TrpText rendered] False recorded)
  249   }

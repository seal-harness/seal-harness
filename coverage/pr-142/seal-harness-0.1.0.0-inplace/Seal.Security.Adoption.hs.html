    1 {-# LANGUAGE DeriveGeneric #-}
    2 -- | Consent-gated adoption of an external tmux window. A discovered window
    3 -- requires @consent_confirmed@ from an interactive channel; a headless run
    4 -- cannot confirm, so adoption fail-closes (the user must confirm via an
    5 -- interactive channel).
    6 module Seal.Security.Adoption
    7   ( ConsentChannel (..)
    8   , AdoptError (..)
    9   , AdoptedHarness (..)
   10   , authorizeAdoption
   11   ) where
   12 
   13 import Data.Text (Text)
   14 import GHC.Generics (Generic)
   15 
   16 import Seal.Harness.Id (HarnessId)
   17 
   18 -- | The interactive channels that can confirm an adoption.
   19 data ConsentChannel = CcCli | CcSignal | CcWeb
   20   deriving stock (Eq, Show)
   21 
   22 -- | The adoption error type. Matched on to drive control flow.
   23 data AdoptError
   24   = AeHeadlessNoConsent       -- ^ cannot confirm consent in a headless run
   25   | AeConsentMissing          -- ^ the channel is interactive but consent was not given
   26   | AeAlreadyManaged HarnessId  -- ^ the window already has a seal_id
   27   deriving stock (Eq, Show)
   28 
   29 -- | An adopted harness: the discovered window's coord + the minted id.
   30 data AdoptedHarness = AdoptedHarness
   31   { ahCoord :: Text
   32   , ahId    :: HarnessId
   33   } deriving stock (Eq, Show, Generic)
   34 
   35 -- | Authorize an adoption: requires @consent_confirmed@ from an
   36 -- interactive 'ConsentChannel'. A headless run (@Nothing@ channel)
   37 -- fail-closes with 'AeHeadlessNoConsent'; an interactive channel with
   38 -- @consent_confirmed = False@ fail-closes with 'AeConsentMissing'.
   39 authorizeAdoption :: Maybe ConsentChannel -> Bool -> Either AdoptError ()
   40 authorizeAdoption Nothing _ = Left AeHeadlessNoConsent
   41 authorizeAdoption (Just _) False = Left AeConsentMissing
   42 authorizeAdoption (Just _) True = Right ()

   40 data PathError
   41   = PathEscapesWorkspace FilePath
   42   | PathIsBlocked Text
   43   | PathDoesNotExist FilePath
   44   | PathInsecureMode FilePath
   45   deriving stock (Eq, Show)
   46 
   47 blockedNames :: [FilePath]
   48 blockedNames = [".env", ".ssh", ".gnupg", ".netrc", ".seal"]
   49 
   50 -- | Lexically collapse @.@ and @..@ segments without touching the filesystem.
   51 -- A @..@ removes the preceding ordinary segment; if there is none (or the
   52 -- predecessor is itself an un-collapsible @..@) the @..@ is retained, so an
   53 -- attempt to climb above the root survives into the containment check and is
   54 -- rejected there rather than being silently normalised away.
   55 lexicalCollapse :: [FilePath] -> [FilePath]
   56 lexicalCollapse = reverse . foldl' step []
   57   where
   58     step acc "." = acc
   59     step (x : xs) ".." | x /= ".." = xs
   60     step acc seg = seg : acc
   61 
   62 -- | Validate that a requested path is safely confined within the workspace.
   63 --
   64 -- Steps (in order):
   65 --   1. Reject if any segment of the requested path is a blocked name.
   66 --   2. Lexically collapse @..@/@.@ in the joined path (no filesystem access)
   67 --      and reject with 'PathEscapesWorkspace' if it leaves the root. This is a
   68 --      purely lexical determination, so a @..@ escape is caught BEFORE any
   69 --      filesystem call — independent of whether the target exists.
   70 --   3. Canonicalize the joined path (follows symlinks); a missing final
   71 --      component makes @canonicalizePath@ throw, caught as 'PathDoesNotExist'.
   72 --   4. Re-run the containment check on the canonical path to catch symlinks
   73 --      that resolve outside the workspace.
   74 --   5. Confirm the path exists; return 'SafePath' on success.
   75 --
   76 -- Containment check approach: component-wise prefix test via
   77 -- @splitDirectories canonRoot \`isPrefixOf\` splitDirectories candidate@,
   78 -- which correctly rejects @\/ws\/barbaz@ as not inside @\/ws\/bar@.
   79 mkSafePath :: WorkspaceRoot -> FilePath -> IO (Either PathError SafePath)
   80 mkSafePath (WorkspaceRoot root) requested = do
   81   canonRoot <- canonicalizePath root
   82   -- 1. Blocked-name check (pure, on the segments of the requested path).
   83   if any (`elem` blockedNames) (splitDirectories requested)
   84     then pure $ Left $ PathIsBlocked $ T.pack $ "path touches a blocked location: " <> requested
   85     else do
   86       let joined = if isAbsolute requested then requested else canonRoot </> requested
   87           rootDirs = splitDirectories canonRoot
   88           -- 2. Lexically resolve `..`/`.` and check containment, no filesystem access.
   89           lexicalDirs = lexicalCollapse (splitDirectories joined)
   90       if not (rootDirs `isPrefixOf` lexicalDirs)
   91         then pure $ Left $ PathEscapesWorkspace (joinPath lexicalDirs)
   92         else resolveAndCheck rootDirs joined (joinPath lexicalDirs)
   93 
   94   where
   95     resolveAndCheck rootDirs joined lexical = do
   96       -- 3. Canonicalize (follows symlinks). Throws IOException if path is missing.
   97       canonResult <- try (canonicalizePath joined) :: IO (Either IOException FilePath)
   98       case canonResult of
   99         Left _ -> pure $ Left $ PathDoesNotExist lexical
  100         Right canon ->
  101           -- 4. Post-canonicalization containment check: catches symlink escapes.
  102           if not (rootDirs `isPrefixOf` splitDirectories canon)
  103             then pure $ Left $ PathEscapesWorkspace canon
  104             else do
  105               exists <- doesPathExist canon
  106               if not exists
  107                 then pure $ Left $ PathDoesNotExist canon
  108                 else pure $ Right $ SafePath canon
  109 
  110 -- | Like 'mkSafePath' but allows the final path component to NOT exist
  111 -- (for FILE_WRITE: the file may be created fresh). Steps 1-2 (blocked-name
  112 -- + lexical collapse + containment) are identical. Step 3 canonicalizes
  113 -- the PARENT (which must exist), then re-joins the final component. Step 4
  114 -- confirms the parent is contained; the final component's existence is
  115 -- irrelevant (it may or may not exist).
  116 mkSafePathForWrite :: WorkspaceRoot -> FilePath -> IO (Either PathError SafePath)
  117 mkSafePathForWrite (WorkspaceRoot root) requested = do
  118   canonRoot <- canonicalizePath root
  119   if any (`elem` blockedNames) (splitDirectories requested)
  120     then pure $ Left $ PathIsBlocked $ T.pack $ "path touches a blocked location: " <> requested
  121     else do
  122       let joined = if isAbsolute requested then requested else canonRoot </> requested
  123           rootDirs = splitDirectories canonRoot
  124           lexicalDirs = lexicalCollapse (splitDirectories joined)
  125       if not (rootDirs `isPrefixOf` lexicalDirs)
  126         then pure $ Left $ PathEscapesWorkspace (joinPath lexicalDirs)
  127         else do
  128           -- Canonicalize the parent (must exist), then re-join the final
  129           -- component. If the parent doesn't exist, the canonicalization
  130           -- fails → PathDoesNotExist (the parent must exist for a write).
  131           let lexicalPath = joinPath lexicalDirs
  132           case splitDirectories lexicalPath of
  133             [] -> pure $ Left $ PathEscapesWorkspace lexicalPath
  134             [_] ->
  135               -- The path is a single component (the root itself); writing
  136               -- the root is not meaningful, but the root exists.
  137               resolveParentAndCheck rootDirs lexicalPath canonRoot
  138             segments ->
  139               let parentLex = joinPath (init segments)
  140               in resolveParentAndCheck rootDirs lexicalPath
  141                    =<< canonicalizePath parentLex
  142   where
  143     resolveParentAndCheck rootDirs lexicalPath canonParent = do
  144       -- Step 4: confirm the canonical parent is contained.
  145       if not (rootDirs `isPrefixOf` splitDirectories canonParent)
  146         then pure $ Left $ PathEscapesWorkspace canonParent
  147         else do
  148           parentExists <- doesPathExist canonParent
  149           if not parentExists
  150             then pure $ Left $ PathDoesNotExist canonParent
  151             else pure $ Right $ SafePath lexicalPath

  157 -- | Lexically validate a workspace-relative path for the REMOTE untrusted
  158 -- executor. This is the remote-safe counterpart of 'mkSafePath'/'mkSafePathForWrite':
  159 -- it performs ONLY the lexical confinement steps (blocked-name check +
  160 -- '..'/@.@ collapse + component-wise containment under the root) and skips
  161 -- the local-FS 'canonicalizePath' (the file lives on the remote machine;
  162 -- the harness cannot canonicalize it locally). The validated 'SafePath' is
  163 -- the lexically-resolved path anchored under the (remote) workspace root.
  164 --
  165 -- Use this for all remote-arm 'UntrustedIO' file methods. The returned
  166 -- 'SafePath' is what gets passed to the SSH call as a single argv element
  167 -- after @--@, so no shell on the remote ever interprets it as a flag.
  168 --
  169 -- Pure: no IO, no symlink resolution. Symlink confinement on the remote is
  170 -- the remote OS's responsibility (the workspace root is the agent's
  171 -- confined directory; the SSH user has no access outside it by policy).
  172 mkSafePathRemote :: WorkspaceRoot -> FilePath -> Either PathError SafePath
  173 mkSafePathRemote (WorkspaceRoot root) requested =
  174   if any (`elem` blockedNames) (splitDirectories requested)
  175     then Left $ PathIsBlocked $ T.pack $ "path touches a blocked location: " <> requested
  176     else
  177       let joined      = if isAbsolute requested then requested else root </> requested
  178           rootDirs    = splitDirectories root
  179           lexicalDirs = lexicalCollapse (splitDirectories joined)
  180       in if not (rootDirs `isPrefixOf` lexicalDirs)
  181            then Left $ PathEscapesWorkspace (joinPath lexicalDirs)
  182            else Right $ SafePath (joinPath lexicalDirs)

  191 -- | Create (mkdir -p) and harden (chmod 0700) a keys directory, returning a
  192 -- typed 'KeysRoot'. Idempotent — safe to call on an already-existing directory.
  193 ensureKeysRoot :: FilePath -> IO KeysRoot
  194 ensureKeysRoot dir = do
  195   createDirectoryIfMissing True dir
  196   setFileMode dir 0o700
  197   pure (KeysRoot dir)

  213 -- | Validate that a requested path is safely confined within the 'KeysRoot'.
  214 --
  215 -- Steps (in order):
  216 --   1. Lexically collapse @..@\/@.@ and check containment under the root —
  217 --      identical to 'mkSafePath' (reuses 'lexicalCollapse' + component-wise
  218 --      @splitDirectories@ prefix check).
  219 --   2. Canonicalize (follows symlinks); re-run the containment check to catch
  220 --      symlink escapes.
  221 --   3. If the target does not yet exist (key to be written later): return
  222 --      @Right@ with the lexically-resolved path.
  223 --   4. If the target exists: require @fileOwner == getEffectiveUserID@ and
  224 --      @fileMode & 0o777 ∈ {0o600, 0o400}@; otherwise return
  225 --      @Left (PathInsecureMode path)@.
  226 mkSafeKeyPath :: KeysRoot -> FilePath -> IO (Either PathError SafeKeyPath)
  227 mkSafeKeyPath (KeysRoot root) requested = do
  228   canonRoot <- canonicalizePath root
  229   let joined = if isAbsolute requested then requested else canonRoot </> requested
  230       rootDirs = splitDirectories canonRoot
  231       lexicalDirs = lexicalCollapse (splitDirectories joined)
  232   if not (rootDirs `isPrefixOf` lexicalDirs)
  233     then pure $ Left $ PathEscapesWorkspace (joinPath lexicalDirs)
  234     else do
  235       canonResult <- try (canonicalizePath joined) :: IO (Either IOException FilePath)
  236       case canonResult of
  237         Left _ ->
  238           -- Path does not exist yet — allowed; the caller will create it.
  239           pure $ Right $ SafeKeyPath (joinPath lexicalDirs)
  240         Right canon ->
  241           if not (rootDirs `isPrefixOf` splitDirectories canon)
  242             then pure $ Left $ PathEscapesWorkspace canon
  243             else do
  244               exists <- doesPathExist canon
  245               if not exists
  246                 -- canonicalizePath resolved the parent but the file is absent —
  247                 -- still allowed; the caller will create it.
  248                 then pure $ Right $ SafeKeyPath canon
  249                 else checkSecurity canon
  250   where
  251     checkSecurity canon = do
  252       status <- getFileStatus canon
  253       euid <- getEffectiveUserID
  254       let owner = fileOwner status
  255           mode = fileMode status `intersectFileModes` 0o777
  256       if (owner /= euid) || (mode `notElem` [0o600, 0o400])
  257         then pure $ Left $ PathInsecureMode canon
  258         else pure $ Right $ SafeKeyPath canon

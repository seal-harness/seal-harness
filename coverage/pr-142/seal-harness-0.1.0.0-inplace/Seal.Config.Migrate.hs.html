    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | One-time, idempotent migration of security-critical fields (vault
    3 -- settings + @[untrusted_execution]@) from @config\/config.toml@ to
    4 -- @~\/.seal\/security.toml@ (design §4 Approach B).
    5 --
    6 -- The migration runs at boot, before 'loadSecurityConfig' / 'loadRuntimeConfig'.
    7 -- It is idempotent and fail-open (a write failure logs an error and boot
    8 -- proceeds — the legacy values are read directly for that session, re-tried
    9 -- next boot).
   10 module Seal.Config.Migrate
   11   ( migrateSecurityConfig
   12   ) where

   32 -- | One-time, idempotent migration. Examines both @config.toml@ and
   33 -- @security.toml@ and moves any legacy security fields from @config.toml@ to
   34 -- @security.toml@. Cases (design §4 B):
   35 --
   36 --   1. @security.toml@ absent, @config.toml@ has legacy fields → migrate.
   37 --   2. @security.toml@ exists, @config.toml@ has stale legacy fields →
   38 --      @security.toml@ wins (precedence); clean stale fields from @config.toml@.
   39 --   3. @security.toml@ exists, @config.toml@ clean → no-op (idempotent).
   40 --   4. Neither exists → no-op (defaults apply).
   41 --   5. Write fails (permissions) → log a loud error and proceed (fail-open).
   42 migrateSecurityConfig :: SealPaths -> IO ()
   43 migrateSecurityConfig paths = do
   44   let cfgPath = configFilePath paths
   45       secPath = securityFilePath paths
   46   secExists <- doesFileExist secPath
   47   cfgExists <- doesFileExist cfgPath
   48   case (secExists, cfgExists) of
   49     -- Case 4: neither exists → no-op.
   50     (False, False) -> pure ()
   51     -- Case 3: security.toml exists, config.toml absent → no-op.
   52     (True, False)  -> pure ()
   53     -- Case 1 & 2: config.toml exists — check for legacy fields.
   54     (secExists', True) -> do
   55       contents <- TIO.readFile cfgPath
   56       let hasLegacy = any (`T.isInfixOf` contents) legacyMarkers
   57       if not hasLegacy
   58         then pure ()  -- Case 3 (clean): no-op.
   59         else do
   60           if secExists'
   61             then do
   62               -- Case 2: security.toml exists → clean stale fields from
   63               -- config.toml (security.toml wins).
   64               cleanConfigToml cfgPath
   65               hPutStrLn stderr $ "[seal] removed stale [untrusted_execution]/vault_* from "
   66                                  <> cfgPath
   67                                  <> " (security.toml takes precedence)"
   68             else do
   69               -- Case 1: security.toml absent → migrate.
   70               migrateFields cfgPath secPath contents
   71 
   72 -- | Lines/tokens that indicate legacy security fields are present in config.toml.
   73 legacyMarkers :: [Text]
   74 legacyMarkers =
   75   [ "vault_path"
   76   , "vault_recipient"
   77   , "vault_identity"
   78   , "vault_unlock"
   79   , "vault_key_type"
   80   , "[untrusted_execution]"
   81   ]
   82 
   83 -- | Migrate legacy fields from config.toml to security.toml (Case 1).
   84 migrateFields :: FilePath -> FilePath -> Text -> IO ()
   85 migrateFields cfgPath secPath contents = do
   86   -- Parse the legacy fields from config.toml using the security codec.
   87   case Toml.parse contents of
   88     Left _ -> pure ()  -- malformed; skip migration.
   89     Right toml ->
   90       case Toml.runTomlCodec securityConfigCodec toml of
   91         Success secCfg -> do
   92           writeResult <- tryWriteSecurityConfig secPath secCfg
   93           case writeResult of
   94             Right () -> do
   95               -- Clean config.toml (re-save RuntimeConfig, which drops legacy fields).
   96               cleanConfigToml cfgPath
   97               hPutStrLn stderr $ "[seal] migrated [untrusted_execution]/vault_* from "
   98                                  <> cfgPath <> " to " <> secPath
   99                                  <> " (one-time). Edit " <> secPath
  100                                  <> " for future changes."
  101             Left err ->
  102               -- Case 5: write failed → fail-open.
  103               hPutStrLn stderr $ "[seal] WARNING: could not write " <> secPath
  104                                  <> " — " <> T.unpack err
  105                                  <> ". Reading legacy fields from config.toml for this session."
  106                                  <> " Fix permissions and restart."
  107         Failure _ ->
  108           -- Could not parse legacy fields; skip migration.
  109           pure ()
  110 
  111 -- | Clean config.toml: re-save the loaded RuntimeConfig, which drops the
  112 -- legacy fields (the RuntimeConfig codec no longer parses them).
  113 cleanConfigToml :: FilePath -> IO ()
  114 cleanConfigToml cfgPath = do
  115   eCfg <- loadRuntimeConfig cfgPath
  116   case eCfg of
  117     Right cfg -> saveRuntimeConfig cfgPath cfg
  118     Left _    -> pure ()
  119 
  120 -- | Try to write the security config (fail-open on error).
  121 tryWriteSecurityConfig :: FilePath -> SecurityConfig -> IO (Either Text ())
  122 tryWriteSecurityConfig path cfg = do
  123   result <- try @SomeException (saveSecurityConfig path cfg)
  124   pure $ case result of
  125     Left err  -> Left (T.pack (show err))
  126     Right ()  -> Right ()

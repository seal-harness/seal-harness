    2 -- | The @[gateway]@ config section: port, bind host, static dir, allowed
    3 -- origins. The server binds loopback by default; a non-loopback host emits
    4 -- a runtime warning (the full slash-command surface is reachable by
    5 -- anything that can reach the address).

   19 -- | The gateway config section.
   20 data GatewayConfig = GatewayConfig
   21   { gcPort           :: Int        -- ^ default 8080 (REST + static)
   22   , gcWsPort         :: Int        -- ^ default 8081 (WS stream)
   23   , gcHost           :: Text       -- ^ default "127.0.0.1"
   24   , gcStaticDir      :: Maybe Text  -- ^ the frontend dist dir (Nothing = no static serving)
   25   , gcAllowedOrigins :: [Text]     -- ^ the WS Origin allowlist (default localhost:8080)
   26   } deriving stock (Eq, Show)
   27 
   28 -- | The compiled-in default. Loopback + port 8080 + the local origin.
   29 defaultGatewayConfig :: GatewayConfig
   30 defaultGatewayConfig = GatewayConfig
   31   { gcPort = 8080
   32   , gcWsPort = 8081
   33   , gcHost = "127.0.0.1"
   34   , gcStaticDir = Nothing
   35   , gcAllowedOrigins = ["http://localhost:8080"]
   36   }

   38 -- | Bidirectional tomland codec for the @[gateway]@ section. Every field is
   39 -- optional: a missing key decodes as 'Nothing', then 'withGatewayDefaults'
   40 -- (called by 'Seal.Command.Serve.runServeMain') fills in the
   41 -- 'defaultGatewayConfig' values. This lets a user set just @host@ or just
   42 -- @port@ without specifying the rest.
   43 gatewayConfigCodec :: Toml.TomlCodec PartialGatewayConfig
   44 gatewayConfigCodec = PartialGatewayConfig
   45   <$> Toml.dimap (fmap fromIntegral) (fmap fromIntegral) (Toml.dioptional (Toml.integer "port")) .= pgcPort
   46   <*> Toml.dimap (fmap fromIntegral) (fmap fromIntegral) (Toml.dioptional (Toml.integer "ws_port")) .= pgcWsPort
   47   <*> Toml.dioptional (Toml.text "host") .= pgcHost
   48   <*> Toml.dioptional (Toml.text "static_dir") .= pgcStaticDir
   49   <*> originsCodec .= pgcAllowedOrigins
   50 
   51 -- | A partial gateway config where every field is optional. 'withGatewayDefaults'
   52 -- merges it with 'defaultGatewayConfig' to produce a complete 'GatewayConfig'.
   53 data PartialGatewayConfig = PartialGatewayConfig
   54   { pgcPort           :: Maybe Int
   55   , pgcWsPort         :: Maybe Int
   56   , pgcHost           :: Maybe Text
   57   , pgcStaticDir      :: Maybe Text
   58   , pgcAllowedOrigins :: [Text]
   59   } deriving stock (Eq, Show)
   60 
   61 -- | Merge a partial gateway config with the compiled-in defaults, producing
   62 -- a complete 'GatewayConfig'. Each 'Nothing' field falls back to
   63 -- 'defaultGatewayConfig'; an empty 'pgcAllowedOrigins' also falls back (so
   64 -- omitting @allowed_origins@ keeps the safe loopback default rather than
   65 -- opening to all origins).
   66 withGatewayDefaults :: PartialGatewayConfig -> GatewayConfig
   67 withGatewayDefaults p = GatewayConfig
   68   { gcPort           = fromMaybe (gcPort defaultGatewayConfig) (pgcPort p)
   69   , gcWsPort         = fromMaybe (gcWsPort defaultGatewayConfig) (pgcWsPort p)
   70   , gcHost           = fromMaybe (gcHost defaultGatewayConfig) (pgcHost p)
   71   , gcStaticDir      = pgcStaticDir p  -- Nothing = no static serving (the default)
   72   , gcAllowedOrigins = if null (pgcAllowedOrigins p)
   73                          then gcAllowedOrigins defaultGatewayConfig
   74                          else pgcAllowedOrigins p
   75   }
   76 
   77 -- | Codec for the @allowed_origins@ array (a TOML array of strings).
   78 originsCodec :: Toml.TomlCodec [Text]
   79 originsCodec = Toml.dimap Just maybeToList (Toml.dioptional (Toml.arrayOf Toml._Text "allowed_origins"))
   80   where
   81     maybeToList Nothing   = []
   82     maybeToList (Just xs) = xs

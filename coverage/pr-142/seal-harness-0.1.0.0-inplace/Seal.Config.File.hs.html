   71 -- | The agent/operator-tunable runtime configuration persisted in
   72 -- @config\/config.toml@. Every field is optional; a missing key decodes as
   73 -- 'Nothing'. Security-critical boot-only fields (vault, untrusted_execution)
   74 -- live in 'Seal.Config.Security.SecurityConfig' (loaded from
   75 -- @security.toml@) and are NOT present here.
   76 data RuntimeConfig = RuntimeConfig
   77   { rcDefaultProvider :: Maybe Text
   78     -- ^ Provider id used for new sessions (e.g. @\"anthropic\"@).
   79   , rcDefaultModel :: Maybe Text
   80     -- ^ Model id used for new sessions (e.g. @\"claude-opus-4-8\"@).
   81   , rcDefaultAgent :: Maybe Text
   82     -- ^ Agent def id used by default for agent-driven flows (set via
   83     -- @\/agent default@). 'Nothing' means no default agent is selected.
   84   , rcProviders :: Map Text ProviderConfig
   85     -- ^ Per-provider config sections (@[providers.<label>]@).
   86   , rcRetrieval :: Maybe RetrievalConfig
   87     -- ^ Optional @[retrieval]@ section (Dynamic Retrieval tuning). Absent
   88     -- means 'defaultRetrievalConfig' applies at resolution time.
   89   , rcSignal :: Maybe SignalConfig
   90     -- ^ Optional @[signal]@ section (Signal channel config). Absent means
   91     -- the Signal channel is not configured.
   92   , rcTelegram :: Maybe TelegramConfig
   93     -- ^ Optional @[telegram]@ section (Telegram channel config). Absent
   94     -- means the Telegram channel is not configured.
   95   , rcGateway :: Maybe PartialGatewayConfig
   96     -- ^ Optional @[gateway]@ section (web gateway config). Absent means the
   97     -- gateway is not configured. Each field inside is optional too; the
   98     -- call site merges with 'Seal.Gateway.Config.withGatewayDefaults'.
   99   , rcDebugSessionTranscript :: Maybe Bool
  100     -- ^ Optional @debug_session_transcript@ flag. When @true@, every
  101     -- 'CompletionRequest' sent to the LLM is also appended (redundantly,
  102     -- in full) to a @requests.jsonl@ file alongside the session's
  103     -- @conversation.jsonl@ / @entries.jsonl@. The contract: each line is
  104     -- the complete JSON-encoded 'CompletionRequest' exactly as passed to
  105     -- the provider, including the full @crMessages@ history. Used to
  106     -- debug whether the two-file storage format is correctly feeding the
  107     -- session history to the LLM. Absent (the default) means the file is
  108     -- not written.
  109   , rcOnDemandSchemas :: Maybe Bool
  110     -- ^ Optional @on_demand_schemas@ flag. When @true@, the full
  111     -- @input_schema@ JSON for every opcode is replaced with a minimal
  112     -- stub (@{\"type\":\"object\"}@) in the @tools@ field of each
  113     -- 'CompletionRequest' (saving tokens on every turn), and a
  114     -- read-only @OPCODE_DESCRIBE@ / @OPCODE_LIST@ pair is registered so
  115     -- the model can retrieve an opcode's full input/output schema
  116     -- on-demand before calling it. Absent (the default) preserves the
  117     -- existing behavior: full schemas are sent inline and the describe
  118     -- opcodes are not registered.
  119   , rcDelegation :: Maybe DelegationFileConfig
  120     -- ^ Optional @[delegation]@ section (subagent spawning tunables).
  121     -- Absent means 'defaultDelegationConfig' applies at resolution time.
  122   , rcWeb :: Maybe WebConfig
  123     -- ^ Optional @[web]@ section (web tool config: search endpoint +
  124     -- fetch/search allow-lists + max fetch bytes). Absent means the web
  125     -- tools are fail-closed (no endpoint configured).
  126   , rcWorkdir :: Maybe WorkdirConfig
  127     -- ^ Optional @[workdir]@ section (per-session workdir lifecycle).
  128     -- Absent means defaults apply (persist, no chroot — chroot is
  129     -- deferred). Per-session workdirs are always on regardless of this
  130     -- section; this config only controls cleanup-on-exit.
  131   , rcSkills :: Maybe SkillsConfig
  132     -- ^ Optional @[skills]@ section (skill auto-injection at session
  133     -- start). Absent means the built-in default applies: the
  134     -- @seal-usage@ skill is auto-injected into the system prompt. Set
  135     -- @[skills] autoload = ""@ to disable auto-injection explicitly.
  136   , rcAgent :: Maybe AgentConfig
  137     -- ^ Optional @[agent]@ section (static behavioral guidance blocks
  138     -- injected into the system prompt). Absent means all three blocks
  139     -- are injected (the defaults); individual flags disable each block.
  140   , rcMaxTurns :: Maybe Int
  141     -- ^ Optional top-level @max_turns@ key: the maximum number of
  142     -- tool-use iterations per turn before the loop stops. Absent →
  143     -- 'defaultMaxTurns' (90). Values < 1 are clamped to 1 at resolution
  144     -- time. Mirrors Hermes' default.
  145   , rcToolTimeout :: Maybe ToolTimeoutFileConfig
  146     -- ^ Optional @[tool_timeout]@ section (per-call timeout/abort/retry
  147     -- behavior for opcode dispatch). Absent means
  148     -- 'Seal.Tools.Timeout.defaultToolTimeoutConfig' applies at resolution
  149     -- time. Each field inside is optional too; the resolver fills defaults.
  150   } deriving stock (Eq, Show)

  195 -- | The @[tool_timeout]@ section: per-call timeout/abort/retry behavior for
  196 -- opcode dispatch. Every field is optional at the TOML layer; a missing key
  197 -- decodes as 'Nothing' and the resolver ('toolTimeoutConfig') fills the
  198 -- compiled-in default at resolution time. Mirrors the @delegation.*@ pattern
  199 -- ('DelegationFileConfig').
  200 data ToolTimeoutFileConfig = ToolTimeoutFileConfig
  201   { ttfcDefaultSeconds  :: Maybe Int
  202     -- ^ Default per-call timeout in seconds (default 120).
  203   , ttfcMaxSeconds      :: Maybe Int
  204     -- ^ Hard cap in seconds (default 600).
  205   , ttfcRetryMax        :: Maybe Int
  206     -- ^ Max retry attempts (default 3).
  207   , ttfcRetryBaseMicros :: Maybe Int
  208     -- ^ Base delay between retries in microseconds (default 2_000_000).
  209   , ttfcRetryFactor     :: Maybe Double
  210     -- ^ Backoff multiplier (default 2.0).
  211   , ttfcKillGraceMicros :: Maybe Int
  212     -- ^ SIGTERM→SIGKILL grace period in microseconds (default 5_000_000).
  213   , ttfcMaxOutputBytes  :: Maybe Int
  214     -- ^ Bounded output cap in bytes (default 50_000).
  215   , ttfcAbortPollMicros :: Maybe Int
  216     -- ^ Abort-poll interval in microseconds (default 100_000).
  217   } deriving stock (Eq, Show)

  363 -- | Resolve the per-turn max-iterations bound from the loaded config. Absent
  364 -- or non-positive values fall back to 'defaultMaxTurns'; values below 1 are
  365 -- clamped to 1 (one tool call before the stop).
  366 maxTurnsConfig :: RuntimeConfig -> Int
  367 maxTurnsConfig cfg = case rcMaxTurns cfg of
  368   Nothing      -> defaultMaxTurns
  369   Just n | n < 1 -> 1
  370   Just n       -> n

  372 -- | Resolve the effective 'ToolTimeoutConfig': the config-file section if
  373 -- present (with defaults filling absent fields), else
  374 -- 'defaultToolTimeoutConfig'.
  375 toolTimeoutConfig :: RuntimeConfig -> ToolTimeoutConfig
  376 toolTimeoutConfig cfg = case rcToolTimeout cfg of
  377   Nothing    -> defaultToolTimeoutConfig
  378   Just ttfc  -> ToolTimeoutConfig
  379     { ttcDefaultSeconds  = fromMaybe (ttcDefaultSeconds  defaultToolTimeoutConfig) (ttfcDefaultSeconds  ttfc)
  380     , ttcMaxSeconds      = fromMaybe (ttcMaxSeconds      defaultToolTimeoutConfig) (ttfcMaxSeconds      ttfc)
  381     , ttcRetryMax        = fromMaybe (ttcRetryMax        defaultToolTimeoutConfig) (ttfcRetryMax        ttfc)
  382     , ttcRetryBaseMicros = fromMaybe (ttcRetryBaseMicros defaultToolTimeoutConfig) (ttfcRetryBaseMicros ttfc)
  383     , ttcRetryFactor     = fromMaybe (ttcRetryFactor     defaultToolTimeoutConfig) (ttfcRetryFactor     ttfc)
  384     , ttcKillGraceMicros = fromMaybe (ttcKillGraceMicros defaultToolTimeoutConfig) (ttfcKillGraceMicros ttfc)
  385     , ttcMaxOutputBytes  = fromMaybe (ttcMaxOutputBytes  defaultToolTimeoutConfig) (ttfcMaxOutputBytes  ttfc)
  386     , ttcAbortPollMicros = fromMaybe (ttcAbortPollMicros defaultToolTimeoutConfig) (ttfcAbortPollMicros ttfc)
  387     }

  393 -- | Bidirectional tomland codec for 'RuntimeConfig'.

  398   <$> Toml.dioptional (Toml.text "default_provider") .= rcDefaultProvider
  399   <*> Toml.dioptional (Toml.text "default_model")    .= rcDefaultModel
  400   <*> Toml.dioptional (Toml.text "default_agent")    .= rcDefaultAgent
  401   <*> Toml.tableMap Toml._KeyText (Toml.table providerConfigCodec) "providers" .= rcProviders
  402   <*> Toml.dioptional (Toml.table retrievalConfigCodec "retrieval") .= rcRetrieval
  403   <*> Toml.dioptional (Toml.table signalConfigCodec "signal")       .= rcSignal
  404   <*> Toml.dioptional (Toml.table telegramConfigCodec "telegram")   .= rcTelegram
  405   <*> Toml.dioptional (Toml.table gatewayConfigCodec "gateway")    .= rcGateway
  406   <*> Toml.dioptional (Toml.bool "debug_session_transcript") .= rcDebugSessionTranscript
  407   <*> Toml.dioptional (Toml.bool "on_demand_schemas") .= rcOnDemandSchemas
  408   <*> Toml.dioptional (Toml.table delegationConfigCodec "delegation") .= rcDelegation
  409   <*> Toml.dioptional (Toml.table webConfigCodec "web") .= rcWeb
  410   <*> Toml.dioptional (Toml.table workdirConfigCodec "workdir") .= rcWorkdir
  411   <*> Toml.dioptional (Toml.table skillsConfigCodec "skills")   .= rcSkills
  412   <*> Toml.dioptional (Toml.table agentConfigCodec "agent")     .= rcAgent

  495 -- ---------------------------------------------------------------------------
  496 -- @providers@ table normalization
  497 -- ---------------------------------------------------------------------------
  498 
  499 -- | tomland's 'Toml.tableMap' \/ 'Toml.table' combinators look up the
  500 -- @providers@ node by requiring it to carry an explicit value in the parsed
  501 -- AST. A TOML file that declares only @[providers.\<label\>]@ sub-tables
  502 -- (the idiomatic style, and the one every hand-written config uses) never
  503 -- writes a bare @[providers]@ header, so that node is /implicit/ in
  504 -- tomland's prefix tree and the lookup silently returns an empty map
  505 -- instead of the sub-tables' contents.
  506 --
  507 -- This walks the parsed AST once before decoding and makes that node
  508 -- explicit — without disturbing anything else — so both styles decode
  509 -- correctly. Round-tripped files (written by 'saveRuntimeConfig') already have
  510 -- an explicit node and pass through unchanged.
  511 normalizeProvidersTable :: Toml.TOML -> Toml.TOML
  512 normalizeProvidersTable t =
  513   t { Toml.tomlTables = HashMap.adjust explicitNode "providers" (Toml.tomlTables t) }
  514   where
  515     -- 'Toml.tableMap' only ever looks at the 'Just' value of a matching
  516     -- node, never its 'Toml.Branch' children — so an implicit @providers@
  517     -- 'Toml.Branch' must be collapsed into a single 'Toml.Leaf' whose value
  518     -- embeds those children as its own 'Toml.tomlTables', mirroring exactly
  519     -- what a bare @[providers]@ header would have produced.
  520     explicitNode :: Toml.PrefixTree Toml.TOML -> Toml.PrefixTree Toml.TOML
  521     explicitNode tree = case tree of
  522       Toml.Branch pref Nothing children ->
  523         Toml.Leaf pref (mempty { Toml.tomlTables = children })
  524       Toml.Branch _ (Just _) _ -> tree
  525       Toml.Leaf (_ :|| [])  _  -> tree
  526       Toml.Leaf (_ :|| (p : ps)) v ->
  527         Toml.Leaf ("providers" :|| []) (mempty { Toml.tomlTables = Toml.single (p :|| ps) v })

  533 -- | Load the config file at @path@.
  534 --
  535 -- * File absent  → @Right 'defaultRuntimeConfig'@
  536 -- * Parse error  → @Left@ with the rendered tomland diagnostics
  537 loadRuntimeConfig :: FilePath -> IO (Either Text RuntimeConfig)
  538 loadRuntimeConfig path = do
  539   exists <- doesFileExist path
  540   if not exists
  541     then pure (Right defaultRuntimeConfig)
  542     else do
  543       contents <- TIO.readFile path
  544       pure $ case Toml.parse contents of
  545         Left err   -> Left (Toml.unTomlParseError err)
  546         Right toml -> case Toml.runTomlCodec runtimeConfigCodec (normalizeProvidersTable toml) of
  547           Success cfg  -> Right cfg
  548           Failure errs -> Left (Toml.prettyTomlDecodeErrors errs)
  549 
  550 -- | Save @cfg@ to @path@ atomically: write @path.tmp@, rename over @path@.
  551 -- The file is not chmod-restricted (config.toml is not secret material;
  552 -- unlike vault.age / security.toml which are handled with 0600).
  553 saveRuntimeConfig :: FilePath -> RuntimeConfig -> IO ()
  554 saveRuntimeConfig path cfg = do
  555   let encoded = Toml.encode runtimeConfigCodec cfg
  556       tmp     = path <> ".tmp"
  557   TIO.writeFile tmp encoded
  558   renameFile tmp path
  559 
  560 -- | Process-wide lock serializing config writes to prevent lost-update
  561 -- races (design V7). Multiple concurrent 'updateRuntimeConfig' callers
  562 -- (e.g. Gateway PUT + a future CONFIG_UPDATE, or two sessions) can silently
  563 -- clobber each other without this lock. The MVar is initialized once via
  564 -- 'unsafePerformIO' — idiomatic for a process-wide lock (mirrors
  565 -- 'Seal.Security.Vault.stWriteLock').
  566 {-# NOINLINE configWriteLock #-}
  567 configWriteLock :: MVar ()
  568 configWriteLock = unsafePerformIO (newMVar ())
  569 
  570 -- | Load the config at @path@, apply @f@, save. Propagates any load
  571 -- error as @Left Text@ without writing. The update function operates on
  572 -- 'RuntimeConfig' only — it physically cannot touch security-critical
  573 -- fields (vault, untrusted_execution) because they live in
  574 -- 'Seal.Config.Security.SecurityConfig', a different type (design §4 E).
  575 -- The load-modify-save is serialized behind 'configWriteLock' to prevent
  576 -- lost-update races (design V7).
  577 updateRuntimeConfig :: FilePath -> (RuntimeConfig -> RuntimeConfig) -> IO (Either Text ())
  578 updateRuntimeConfig path f = withMVar configWriteLock $ \_ -> do
  579   result <- loadRuntimeConfig path
  580   case result of
  581     Left err  -> pure (Left err)
  582     Right cfg -> saveRuntimeConfig path (f cfg) >> pure (Right ())

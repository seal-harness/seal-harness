   32 -- | The configuration for WEB_FETCH.
   33 data WebFetchConfig = WebFetchConfig
   34   { wfcManager    :: Maybe Manager  -- ^ HTTP manager (Nothing = fail-closed)
   35   , wfcAllowList  :: [Text]         -- ^ allowed domains (empty = all allowed)
   36   , wfcMaxBytes   :: Int            -- ^ operator-configured byte ceiling
   37   , wfcAuthKey    :: Maybe Text    -- ^ vault key reference (NOT inline auth)
   38   }
   39 
   40 -- | WEB_FETCH opcode. Input: @{ url: Text }@. Fetches the URL via
   41 -- @http-client@, truncates the response body to 'wfcMaxBytes', and enforces
   42 -- the domain allow-list at the gate. 'orRecorded' carries the URL + status +
   43 -- byte count (secret-free metadata; the body is NOT recorded).
   44 webFetchOp :: WebFetchConfig -> Opcode
   45 webFetchOp cfg = UntrustedOpcode
   46   { uoName = OpName "WEB_FETCH"
   47   , uoDesc = "Fetch a URL (bounded bytes, allow-listed, auth-redacted)."
   48   , uoInSchema = webFetchSchema
   49   , uoOutSchema = object []
   50   , uoAuthorize = \v ->
   51       case urlField v of
   52         Nothing -> Left "WEB_FETCH requires {url:string}"
   53         Just u
   54           | T.null u -> Left "WEB_FETCH: url is empty"
   55           | not (domainAllowed u (wfcAllowList cfg)) ->
   56               Left ("WEB_FETCH: domain not in allow-list: " <> hostOf u)
   57           | otherwise -> Right ()
   58   , uoRun = \v -> do
   59       let u = fromMaybe "" (urlField v)
   60       case wfcManager cfg of
   61         Nothing -> pure (OpResult
   62           [TrpText "WEB_FETCH: no HTTP manager configured"]
   63           True (object ["url" .= u, "status" .= (0 :: Int), "bytes" .= (0 :: Int)]))
   64         Just mgr -> uioLiftIO (doFetch mgr (wfcMaxBytes cfg) u)
   65   }
   66 
   67 -- | Perform the HTTP fetch: SSRF check → parse the URL → execute the
   68 -- request → truncate the response body to the byte ceiling → return the
   69 -- decoded body + recorded metadata. Errors (SSRF, transport, non-2xx)
   70 -- surface as structured 'OpResult's with @isError=True@.
   71 doFetch :: Manager -> Int -> Text -> IO OpResult
   72 doFetch mgr maxBytes u = do
   73   -- SSRF defense: block private/internal/metadata IPs BEFORE the HTTP call.
   74   eSafe <- isSafeUrl u
   75   case eSafe of
   76     Left err -> pure (OpResult
   77       [TrpText ("WEB_FETCH: blocked by SSRF protection: " <> renderUrlSafetyError err)]
   78       True recordedErr)
   79     Right () -> doFetchHttp mgr maxBytes u
   80   where
   81     recordedErr = object ["url" .= u, "status" .= (0 :: Int), "bytes" .= (0 :: Int)]
   82 
   83 -- | The HTTP fetch proper (SSRF check already passed).
   84 doFetchHttp :: Manager -> Int -> Text -> IO OpResult
   85 doFetchHttp mgr maxBytes u = do
   86   eReq <- try (parseRequest (T.unpack u))
   87   case eReq of
   88     Left (_ :: HttpException) ->
   89       pure (OpResult [TrpText ("WEB_FETCH: invalid URL: " <> u)] True recordedErr)
   90     Right req -> do
   91       eResp <- try (httpLbs req mgr)
   92       case eResp of
   93         Left (_ :: HttpException) ->
   94           pure (OpResult [TrpText "WEB_FETCH: HTTP request failed (connection or transport error)"]
   95                            True recordedErr)
   96         Right resp -> do
   97           let code = statusCode (responseStatus resp)
   98               body = responseBody resp
   99               truncated = truncateBytes body maxBytes
  100               bodyText = TE.decodeUtf8With TEE.lenientDecode (BL.toStrict truncated)
  101               byteCount = BL.length truncated
  102               recorded = object
  103                 [ "url" .= u
  104                 , "status" .= code
  105                 , "bytes" .= byteCount
  106                 ]
  107           if code >= 200 && code <= 299
  108             then pure (OpResult [TrpText bodyText] False recorded)
  109             else pure (OpResult
  110               [TrpText ("WEB_FETCH: HTTP " <> T.pack (show code) <> ": " <> bodyText)]
  111               True recorded)
  112   where
  113     recordedErr = object ["url" .= u, "status" .= (0 :: Int), "bytes" .= (0 :: Int)]
  114 
  115 -- | Truncate a lazy 'ByteString' to at most @n@ bytes.
  116 truncateBytes :: BL.ByteString -> Int -> BL.ByteString
  117 truncateBytes bs n
  118   | n <= 0    = BL.empty
  119   | otherwise = BL.take (fromIntegral n) bs

  121 -- | Check whether the URL's host is in the allow-list. An empty allow-list
  122 -- means all domains are permitted.
  123 domainAllowed :: Text -> [Text] -> Bool
  124 domainAllowed _ [] = True
  125 domainAllowed url allowList =
  126   case hostOfMaybe url of
  127     Nothing  -> False
  128     Just host -> any (`T.isSuffixOf` host) allowList
  129 
  130 -- | Extract the host from a URL (the authority component). Returns the
  131 -- empty string when the URL has no parseable host.
  132 hostOf :: Text -> Text
  133 hostOf url = fromMaybe "" (hostOfMaybe url)
  134 
  135 -- | Extract the host from a URL, returning 'Nothing' when absent.
  136 -- Handles @scheme://host/path@ and @scheme://host:port/path@.
  137 hostOfMaybe :: Text -> Maybe Text
  138 hostOfMaybe url =
  139   case T.breakOn "://" url of
  140     (_, after) | T.null after -> Nothing  -- no scheme separator
  141     (_, after) ->
  142       let noScheme = T.drop 3 after  -- drop the "://" (3 chars)
  143           authority = T.takeWhile (\c -> c /= '/' && c /= '?' && c /= '#') noScheme
  144           host = T.takeWhile (/= ':') authority  -- strip port
  145       in if T.null host then Nothing else Just host

   51 -- | Opcodes that a child agent must NEVER have access to. Mirrors Hermes'
   52 -- @DELEGATE_BLOCKED_TOOLS@. Stripping these means a child cannot:
   53 --
   54 --   * recursively spawn its own subagents (@AGENT_START@) — that would
   55 --     bypass the depth cap;
   56 --   * mutate agent definitions (@AGENT_DEF_WRITE@ / @AGENT_DEF_DELETE@) —
   57 --     only the parent should evolve the def store;
   58 --   * introspect or control live instances (@AGENT_INSTANCES@ /
   59 --     @AGENT_STATUS@ / @AGENT_STOP@ / @AGENT_INTERRUPT@) — those are
   60 --     parent/operator controls, not child tools.
   61 delegationBlocklist :: Set.Set OpName
   62 delegationBlocklist = Set.fromList
   63   [ OpName "AGENT_START"
   64   , OpName "AGENT_DEF_WRITE"
   65   , OpName "AGENT_DEF_DELETE"
   66   , OpName "AGENT_INSTANCES"
   67   , OpName "AGENT_STATUS"
   68   , OpName "AGENT_STOP"
   69   , OpName "AGENT_INTERRUPT"
   70   ]
   71 
   72 -- | Apply the 'delegationBlocklist' to a child's tool allow-list. Only
   73 -- narrows 'AllowOnly' (set-difference with the blocklist); 'AllowAll' is
   74 -- returned unchanged because the blocklist is enforced at registry-build
   75 -- time by omitting blocklisted opcodes from the ops list (we can't enumerate
   76 -- the universe of opcode names to form a complement here).
   77 narrowAllowList :: AllowList OpName -> AllowList OpName
   78 narrowAllowList AllowAll       = AllowAll
   79 narrowAllowList (AllowOnly xs) = AllowOnly (Set.difference xs delegationBlocklist)

   81 -- | Filter a list of opcodes to remove any whose name is in the
   82 -- 'delegationBlocklist'. The wiring layer calls this on its base ops list
   83 -- before passing to 'Seal.ISA.Registry.mkRegistry' to build the child's
   84 -- narrowed registry. This is the primary blocklist enforcement — it works
   85 -- regardless of whether the def's @adTools@ is 'AllowAll' or 'AllowOnly'.
   86 filterBlocklisted :: [opcode] -> (opcode -> OpName) -> [opcode]
   87 filterBlocklisted ops getName =
   88   [ o | o <- ops, not (getName o `Set.member` delegationBlocklist) ]

   90 -- | The per-channel deps the worker-builder closes over. The wiring layer
   91 -- (Cli.hs, Channels.Loop.hs, Gateway.Send.hs) builds this from its own
   92 -- per-turn closure and passes it to 'mkDelegateWorker'.
   93 data DelegationWorkerDeps = DelegationWorkerDeps
   94   { dwdPaths        :: SealPaths
   95   , dwdParentSid    :: SessionId
   96     -- ^ The parent's session id — the child's transcript nests under it.
   97   , dwdAppEnv       :: Env
   98     -- ^ The top-level app env (katip logging, config) — re-used for the
   99     -- child's 'runApp'.
  100   , dwdMkUIOEnv :: SessionId -> IO UIOEnv
  101     -- ^ Construct the child's 'UIOEnv' (carrying the 'UntrustedIO'
  102     -- capability handle + Git 'CloneDeps') from the child's session id.
  103     -- The wiring layer creates the child's workdir (per-session isolation)
  104     -- and resolves the security config into the env. Called at child-start
  105     -- time (after the child's sid is minted).
  106   , dwdAutonomy     :: AutonomyLevel
  107   , dwdApprovals    :: ApprovalCache
  108   , dwdOnDemand     :: Bool
  109   , dwdParentDepth  :: Int
  110     -- ^ The parent's delegation depth; the child's depth is this + 1.
  111   , dwdResolveProvider :: AgentDef -> IO (Either Text (SomeProvider, ModelId))
  112     -- ^ Resolve the child's provider+model from the def, applying any
  113     -- delegation.provider/model/base_url override (the wiring layer reads
  114     -- the override from the RuntimeConfig and threads it here).
  115   , dwdChildRegistry
  116       :: AgentDef -> SessionId -> ChannelCaps -> IO Registry
  117     -- ^ Build the child's narrowed ISA registry. The wiring layer is
  118     -- responsible for applying 'delegationBlocklist' to the def's
  119     -- @adTools@ allow-list and constructing the registry. The caps + sid
  120     -- are passed in so the registry can close over them (ASK_HUMAN etc.).
  121   , dwdChildSystemPrompt :: AgentDef -> ChildTask -> IO (Maybe Text)
  122     -- ^ Build the child's system prompt from the def's @adSystem@ + the
  123     -- task's @ctContext@. Runs in 'IO' so the wiring layer can load the
  124     -- auto-injected skill (default @seal-usage@) from the skill backend and
  125     -- append it. 'Nothing' means no system prompt.
  126   , dwdOnEntry :: IO ()
  127     -- ^ The on-entry hook for the child's transcript (live broadcast).
  128     -- 'pure ()' for the CLI; 'broadcastNewEntries' for web/channels.
  129   , dwdChannel :: Text
  130     -- ^ The parent session's channel label (e.g. @\"telegram\"@, @\"web\"@,
  131     -- @\"cli\"@), inherited by the child so the child's request entries are
  132     -- attributed to the same channel the parent turn ran on. Stamped into
  133     -- the child's 'aeChannel' so 'runTurn' attributes the child's user
  134     -- messages (the task goals) to the originating channel.
  135   , dwdAbortFlag :: SessionId -> IO AbortFlag
  136     -- ^ Construct the child's abort flag from the child's session id
  137     -- (looked up from the 'Seal.Tools.Exec.Abort.SessionAbortRegistry').
  138     -- The child's abort is independent of the parent's in v1 (a parent
  139     -- abort doesn't auto-abort the child, and vice versa).
  140   , dwdToolTimeout :: ToolTimeoutConfig
  141     -- ^ The per-call timeout/retry config, inherited from the parent
  142     -- (loaded once from @config.toml@ @[tool_timeout]@ at startup).
  143   }

  145 -- | Build the 'AgentWorkerBuilder' the AGENT_START opcode closes over. This
  146 -- is the shared delegation worker: open the child transcript, build the
  147 -- child env, run 'runTurn' with the goal as the first user message, capture
  148 -- the final text response as the summary, and report the outcome.
  149 --
  150 -- The summary is captured via a 'ChannelCaps' whose 'ccSend' writes to an
  151 -- IORef; 'runTurn' calls 'ccSend' with the final text response, so we read
  152 -- the IORef after the run. The child's @ccPrompt@ is a no-op (children don't
  153 -- prompt the human — that would deadlock the parent).
  154 mkDelegateWorker :: DelegationWorkerDeps -> AgentWorkerBuilder
  155 mkDelegateWorker deps agentDef childSid task _hooks = do
  156   let childDir = agentSessionDir (dwdPaths deps) (dwdParentSid deps) childSid
  157   createDirectoryIfMissing True childDir
  158   eProv <- dwdResolveProvider deps agentDef
  159   case eProv of
  160     Left err -> pure (ChildWorkerOutcome
  161                        (Just ("agent start failed: " <> err))
  162                        CerError 0 0 (Just childSid))
  163     Right (prov, model) ->
  164       withTwoFileTranscript childDir $ \childTHandle -> do
  165         summaryRef <- newIORef (Nothing :: Maybe Text)
  166         let capturingCaps = def
  167               { ccSend = \t -> atomicModifyIORef' summaryRef (const (Just t, ()))
  168               , ccStreaming    = False  -- children: capture final summary, no per-delta sends
  169               }
  170         childReg <- dwdChildRegistry deps agentDef childSid capturingCaps
  171         childUioEnv <- dwdMkUIOEnv deps childSid
  172         childSystem <- dwdChildSystemPrompt deps agentDef task
  173         childAbortFlag <- dwdAbortFlag deps childSid
  174         let env = AgentEnv
  175               { aeProvider   = prov
  176               , aeProviderLabel = providerLabel agentDef
  177               , aeModel      = model
  178               , aeSystem     = childSystem
  179               , aeRegistry   = childReg
  180               , aeTranscript = childTHandle
  181               , aeBackend    = localBackend
  182               , aeUIOEnv     = childUioEnv
  183               , aeCaps       = capturingCaps
  184               , aeSession    = childSid
  185               , aeMaxTurns   = 90
  186               , aeChannel    = dwdChannel deps
  187               , aeMessageSource = Nothing
  188               , aeAutonomy   = dwdAutonomy deps
  189               , aeApprovals  = dwdApprovals deps
  190               , aeDebugRequestsPath = Nothing
  191               , aeOnEntry    = dwdOnEntry deps
  192               , aeOnUserMessage = Nothing
  193               , aeOnStop     = Nothing
  194               , aeOnDemandSchemas = dwdOnDemand deps
  195               , aeLogPath    = Nothing
  196               , aeAbortFlag  = childAbortFlag
  197               , aeToolTimeout = dwdToolTimeout deps
  198               }
  199         runApp (dwdAppEnv deps) (runTurn env (ctGoal task))
  200           `catch` \e -> writeIORef summaryRef
  201                        (Just ("child runTurn raised: " <> T.pack (show (e :: SomeException))))
  202         summary <- readIORef summaryRef
  203         let exitReason = case summary of
  204               Just _  -> CerCompleted
  205               Nothing -> CerMaxIterations
  206         -- Token counts: the child's per-turn usage is recorded in the
  207         -- transcript entries; we don't aggregate them here (would require
  208         -- reading entries.jsonl). Report 0 for now — a follow-up can sum
  209         -- the child's EKResponse entries.
  210         pure (ChildWorkerOutcome summary exitReason 0 0 (Just childSid))
  211   where
  212     providerLabel d = if T.null (adProvider d) then "ollama" else adProvider d

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | A name-indexed opcode set; derives the provider tool-definition list the
    3 -- agent is offered each turn. Tool definitions preserve registration order
    4 -- (not alphabetical) so the wiring layer controls which tools the model sees
    5 -- first — many LLMs bias toward the first tool whose description matches,
    6 -- so the order is a lever for steering tool selection (e.g. WEB_FETCH before
    7 -- BROWSER_OPEN for simple page fetches).
    8 module Seal.ISA.Registry
    9   ( Registry
   10   , mkRegistry
   11   , lookupOp
   12   , registryToolDefs
   13   , registryToolDefs'
   14   , stubSchema
   15   , secretOpNames
   16   ) where
   17 
   18 import Data.Map.Strict (Map)

   20 import Data.Set (Set)
   21 import Data.Set qualified as Set
   22

   27 -- | The registry carries both a name-indexed 'Map' (O(log n) dispatch
   28 -- lookup) and the registration-ordered opcode list (for tool-definition
   29 -- emission in a stable, wiring-controlled order).
   30 data Registry = Registry (Map OpName Opcode) [Opcode]
   31 
   32 mkRegistry :: [Opcode] -> Registry
   33 mkRegistry ops = Registry (Map.fromList [(opName o, o) | o <- ops]) ops
   34 
   35 lookupOp :: Registry -> OpName -> Maybe Opcode
   36 lookupOp (Registry m _) n = Map.lookup n m
   37 
   38 -- | The tool definitions the provider is offered each turn, in registration
   39 -- order (the order 'mkRegistry' received). NOT alphabetical — the wiring
   40 -- layer controls the order so it can steer which tools the model tries first.
   41 registryToolDefs :: Registry -> [ToolDefinition]
   42 registryToolDefs = registryToolDefs' False
   43 
   44 -- | Like 'registryToolDefs' but with a flag to swap full @input_schema@s for
   45 -- a minimal stub. When @useStub@ is 'True', each tool definition carries
   46 -- 'stubSchema' instead of the opcode's real 'opInSchema' — the name and
   47 -- description are preserved so the model still knows what tools exist, but
   48 -- the per-opcode schema JSON (the bulk of the tokens) is omitted. The model
   49 -- is expected to call the @OPCODE_DESCRIBE@ opcode to retrieve a tool's full
   50 -- schema before calling it. When @useStub@ is 'False' this is identical to
   51 -- 'registryToolDefs'. The provider encoders OMIT the @input_schema@ field
   52 -- entirely when it equals 'stubSchema' (see 'Seal.Providers.Class'), so the
   53 -- stub costs zero tokens on the wire rather than the few a real stub object
   54 -- would cost.
   55 registryToolDefs' :: Bool -> Registry -> [ToolDefinition]
   56 registryToolDefs' useStub (Registry _ order) =
   57   [ ToolDefinition (opName o) (opDesc o) (if useStub then stubSchema else opInSchema o)
   58   | o <- order
   59   ]
   60 
   61 -- | The set of opcode names whose tool results may carry secrets and must be
   62 -- redacted from the on-disk @conversation.jsonl@. Only opcodes that return a
   63 -- vault secret value in 'orParts' belong here — currently just 'SECRET_GET'.
   64 -- Other opcodes (MEMORY_RECALL, FILE_READ, SHELL_EXEC, etc.) return
   65 -- agent-visible data that is safe to persist verbatim and display in the
   66 -- frontend. Using trust level as a proxy was wrong: MEMORY_RECALL is Trusted
   67 -- but returns memory content (not vault secrets), so redacting it hid
   68 -- harmless output behind @<redacted:secret>@.
   69 secretOpNames :: Registry -> Set OpName
   70 secretOpNames (Registry m _) =
   71   Set.fromList [ opName o | o <- Map.elems m, opName o `Set.member` secretOpcodes ]
   72 
   73 -- | The static set of opcode names that return secret values in 'orParts'.
   74 secretOpcodes :: Set OpName
   75 secretOpcodes = Set.fromList [ OpName "SECRET_GET" ]

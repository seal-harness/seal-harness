    2 -- | The Agent opcode group: three Audited definition opcodes
    3 -- (@AGENT_DEF_WRITE@, @AGENT_DEF_READ@, @AGENT_DEF_LIST@, @AGENT_DEF_DELETE@)
    4 -- and four Trusted lifecycle opcodes (@AGENT_INSTANCES@, @AGENT_START@,
    5 -- @AGENT_STATUS@, @AGENT_STOP@, @AGENT_INTERRUPT@). Only the *definition*
    6 -- mutations are Audited — running an instance is harness-internal, not an
    7 -- evolutionary mutation, so the lifecycle ops are Trusted.
    8 --
    9 -- @AGENT_DEF_WRITE@ is an upsert: if the def already exists, its name/system/
   10 -- tools are updated (the original 'adSession' provenance and 'adCreatedAt' are
   11 -- preserved; only 'adUpdatedAt' is bumped); if not, a fresh def is created.
   12 -- This merges the former AGENT_DEF_CREATE + AGENT_DEF_UPDATE into a single
   13 -- opcode. @orRecorded@ carries a @was_new@ flag so the audit log still
   14 -- distinguishes create vs update.
   15 --
   16 -- @AGENT_INSTANCES@ (renamed from @AGENT_LIST@) snapshots the in-process agent
   17 -- runtime (running instances), NOT the definitions. @AGENT_DEF_LIST@ lists the
   18 -- definitions. The rename stops the confusion between "list definitions" and
   19 -- "list running instances".
   20 --
   21 -- @AGENT_START@ is the Seal analog of Hermes' @delegate_task@: it spawns one
   22 -- or more child agents with isolated context, runs each against a goal to
   23 -- completion (synchronously), and returns a structured JSON result per child.
   24 -- The parent blocks until all children finish (or time out). See
   25 -- 'Seal.Agent.Runtime.Delegation' for the full feature list. The opcode is a
   26 -- thin shim over 'runDelegate' — it normalizes the model's input, resolves the
   27 -- def, delegates the worker construction to the wiring layer's
   28 -- 'AgentWorkerBuilder', and serializes the 'ChildResult' list to JSON.

   90 -- | Extract the @id@ string field from a JSON object.
   91 idField :: Value -> Maybe Text
   92 idField = parseMaybe (withObject "in" (.: "id"))
   93 
   94 -- | Extract a required text field (defaults to empty when absent).
   95 textField :: Text -> Value -> Text
   96 textField name v = fromMaybe "" (parseMaybe (withObject "in" (.: fromText name)) v)
   97 
   98 -- | Extract an optional text field (defaults to 'Nothing' when absent).
   99 textFieldMaybe :: Text -> Value -> Maybe Text
  100 textFieldMaybe name v =
  101   case parseMaybe (withObject "in" (.:? fromText name)) v :: Maybe (Maybe Text) of
  102     Just (Just t) -> Just t
  103     _             -> Nothing

  105 -- | Decode the @tools@ field: @\"all\"@ (or absent) -> 'AllowAll'; an array of
  106 -- opcode-name strings -> 'AllowOnly'. Malformed -> 'AllowAll' (permissive).
  107 toolsField :: Value -> AllowList OpName
  108 toolsField v =
  109   case parseMaybe (withObject "in" (.:? "tools")) v :: Maybe (Maybe Value) of
  110     Just (Just (String "all")) -> AllowAll
  111     Just (Just (Array xs))     -> AllowOnly (Set.fromList [ OpName t | String t <- V.toList xs ])
  112     _                          -> AllowAll

  114 -- | Resolve the @group@ field from the opcode input, preserving the existing
  115 -- def's group when the input omits it (mirrors the gateway @stampAgentDef@
  116 -- and the skill @stampSkill@ back-compat policy). An explicit empty string
  117 -- clears the group (sets 'Nothing'); a non-empty stripped value sets it.
  118 groupField :: Value -> Maybe AgentDef -> Maybe Text
  119 groupField v mExisting =
  120   case textFieldMaybe "group" v of
  121     Just g  -> let g' = T.strip g in if T.null g' then Nothing else Just g'
  122     Nothing -> adGroup =<< mExisting

  128 -- | AGENT_DEF_WRITE: upsert an agent definition by id. If the def already
  129 -- exists, its name/system/tools are updated (the original 'adSession'
  130 -- provenance and 'adCreatedAt' are preserved; only 'adUpdatedAt' is bumped);
  131 -- if not, a fresh def is created. The name, provider, model, system prompt,
  132 -- and tool list are recorded in full (agent-visible data); 'orRecorded'
  133 -- carries the id + op name + fields + @was_new@.
  134 agentDefWriteOp :: AgentDefBackend -> SessionId -> Opcode
  135 agentDefWriteOp backend session = TrustedOpcode
  136   { toName = OpName "AGENT_DEF_WRITE"
  137   , toTrust = Trusted
  138   , toDesc = "Create or update an agent definition by id (upsert; preserves provenance on update)."
  139   , toInSchema = object
  140       [ "type" .= ("object" :: Text)
  141       , "properties" .= object
  142           [ fromText "id" .= object
  143               [ "type" .= ("string" :: Text)
  144               , "description" .= ("Agent def id ([A-Za-z0-9_-]+)." :: Text)
  145               ]
  146           , fromText "name" .= object
  147               [ "type" .= ("string" :: Text)
  148               , "description" .= ("Human-readable agent name." :: Text)
  149               ]
  150           , fromText "provider" .= object
  151               [ "type" .= ("string" :: Text)
  152               , "description" .= ("Provider label, e.g. \"ollama\"." :: Text)
  153               ]
  154           , fromText "model" .= object
  155               [ "type" .= ("string" :: Text)
  156               , "description" .= ("Model id, e.g. \"llama3\"." :: Text)
  157               ]
  158           , fromText "system" .= object
  159               [ "type" .= ("string" :: Text)
  160               , "description" .= ("Optional system prompt." :: Text)
  161               ]
  162           , fromText "tools" .= object
  163               [ "type" .= ("array" :: Text)
  164               , "description" .= ("Allowed opcode names, or \"all\"." :: Text)
  165               ]
  166           , fromText "group" .= object
  167               [ "type" .= ("string" :: Text)
  168               , "description" .= ("Optional display group (e.g. \"core\"). Omit for the default (ungrouped) section." :: Text)
  169               ]
  170           ]
  171       , "required" .= (["id", "name", "provider", "model"] :: [Text])
  172       ]
  173   , toOutSchema = object []
  174   , toAuthorize = maybe (Left "AGENT_DEF_WRITE requires {id:string}") checkId . idField
  175   , toRun = \_ v -> do
  176       let mId = idField v >>= either (const Nothing) Just . mkAgentDefId
  177       case mId of
  178         Nothing -> pure (OpResult [TrpText "invalid agent def id"] True (object []))
  179         Just aid -> do
  180           mExisting <- liftIO (adbRead backend aid)
  181           now <- liftIO getCurrentTime
  182           let (def, wasNew) = case mExisting of
  183                 Just existing ->
  184                   ( existing
  185                       { adName = textField "name" v
  186                       , adSystem = textFieldMaybe "system" v
  187                       , adTools = toolsField v
  188                       , adGroup = groupField v (Just existing)
  189                       , adUpdatedAt = now
  190                       }
  191                   , False
  192                   )
  193                 Nothing ->
  194                   ( AgentDef
  195                       { adId = aid
  196                       , adName = textField "name" v
  197                       , adProvider = textField "provider" v
  198                       , adModel = ModelId (textField "model" v)
  199                       , adSystem = textFieldMaybe "system" v
  200                       , adTools = toolsField v
  201                       , adGroup = groupField v Nothing
  202                       , adCreatedAt = now
  203                       , adUpdatedAt = now
  204                       , adSession = session
  205                       }
  206                   , True
  207                   )
  208           liftIO (adbUpdate backend def)
  209           let recorded = encodeDefRecorded def wasNew
  210           pure (OpResult [TrpText (if wasNew then "defined" else "updated")] False recorded)
  211   }
  212   where
  213     checkId t = either (Left . ("invalid agent def id: " <>)) (const (Right ())) (mkAgentDefId t)

  333 -- ---------------------------------------------------------------------------
  334 -- AGENT_START (synchronous goal-driven delegation)
  335 -- ---------------------------------------------------------------------------
  336 
  337 -- | The wiring-layer bundle the AGENT_START opcode closes over. The
  338 -- 'AgentWorkerBuilder' resolves the def's provider+model, opens a fresh
  339 -- two-file transcript under @\<parent-session\>\/agents\/\<child-id\>@, builds
  340 -- a fresh 'AgentEnv' bound to the new session + child transcript, runs the
  341 -- turn with the goal as the first user message, and reports the outcome via
  342 -- 'ChildWorkerOutcome'. The 'DelegationConfig' / 'SpawnPauseFlag' /
  343 -- 'ParentActivity' are process-global (or per-channel) and threaded in
  344 -- here so the opcode doesn't read config.
  345 data AgentStartWiring = AgentStartWiring
  346   { aswDefBackend   :: AgentDefBackend
  347   , aswRuntime      :: AgentRuntime
  348   , aswConfig       :: IO DelegationConfig
  349     -- ^ Re-read the [delegation] config per AGENT_START call (so config
  350     -- changes take effect without a restart). The IO action reads
  351     -- @config.toml@ and returns the resolved 'DelegationConfig'.
  352   , aswPauseFlag    :: SpawnPauseFlag
  353   , aswParentActivity :: Maybe ParentActivity
  354   , aswMintSession  :: IO SessionId
  355     -- ^ Mint a fresh 'SessionId' for a child.
  356   , aswParentDepth  :: Int
  357     -- ^ The parent's delegation depth (0 for a top-level turn).
  358   , aswWorker       :: AgentWorkerBuilder
  359     -- ^ The worker-builder (closes over per-turn 'AgentEnv' deps).
  360   }
  361 
  362 -- | AGENT_START: spawn one or more child agents, run each against a goal to
  363 -- completion, return a JSON result per child. Input is either
  364 -- @{id, goal, context?, role?}@ (single mode) or @{tasks: [{id, goal, context?, role?}, ...]}@
  365 -- (batch mode). Returns JSON with a @results@ array, one entry per task.
  366 agentStartOp :: AgentStartWiring -> Opcode
  367 agentStartOp wiring = TrustedOpcode
  368   { toName = OpName "AGENT_START"
  369   , toTrust = Trusted
  370   , toDesc = "Spawn one or more child agents, run each against a goal to completion, return a JSON result per child. Single mode: {id, goal, context?, role?}. Batch mode: {tasks: [{id, goal, context?, role?}, ...]}."
  371   , toInSchema = object
  372       [ "type" .= ("object" :: Text)
  373       , "properties" .= object
  374           [ fromText "id" .= object
  375               [ "type" .= ("string" :: Text)
  376               , "description" .= ("Agent def id (single-task mode)." :: Text)
  377               ]
  378           , fromText "goal" .= object
  379               [ "type" .= ("string" :: Text)
  380               , "description" .= ("The task goal — becomes the child's first user message (single-task mode)." :: Text)
  381               ]
  382           , fromText "context" .= object
  383               [ "type" .= ("string" :: Text)
  384               , "description" .= ("Optional background context appended to the child's system prompt." :: Text)
  385               ]
  386           , fromText "role" .= object
  387               [ "type" .= ("string" :: Text)
  388               , "description" .= ("\"leaf\" (default) or \"orchestrator\". Orchestrators may spawn their own subagents, bounded by max_spawn_depth." :: Text)
  389               ]
  390           , fromText "tasks" .= object
  391               [ "type" .= ("array" :: Text)
  392               , "description" .= ("Batch mode: array of {id, goal, context?, role?}. Cap on parallelism is delegation.max_concurrent_children." :: Text)
  393               ]
  394           ]
  395       , "required" .= (["goal"] :: [Text])
  396       ]
  397   , toOutSchema = object []
  398   , toAuthorize = \v ->
  399       -- Require either a top-level goal (single) or a tasks array (batch).
  400       let hasGoal = case textFieldMaybe "goal" v of { Just _ -> True; Nothing -> False }
  401           hasTasks = case parseMaybe (withObject "in" (.:? "tasks")) v :: Maybe (Maybe Value) of { Just (Just _) -> True; _ -> False }
  402       in if hasGoal || hasTasks
  403            then Right ()
  404            else Left "AGENT_START requires {goal:string} (single) or {tasks:array} (batch)."
  405   , toRun = \_ v -> do
  406       input <- liftIO (parseInput v)
  407       case input of
  408         Left err -> pure (OpResult [TrpText err] True (object []))
  409         Right di -> do
  410           cfg <- liftIO (aswConfig wiring)
  411           eResults <- liftIO (runDelegate
  412                                 cfg
  413                                 (aswPauseFlag wiring)
  414                                 (aswParentActivity wiring)
  415                                 (aswParentDepth wiring)
  416                                 di
  417                                 (resolveTask (aswDefBackend wiring)
  418                                              (aswRuntime wiring)
  419                                              (aswMintSession wiring)
  420                                              (aswParentDepth wiring)
  421                                              (aswWorker wiring)))
  422           case eResults of
  423             Left err -> pure (OpResult [TrpText err] True (object []))
  424             Right results -> do
  425               -- Register each finished child in the runtime registry so
  426               -- AGENT_INSTANCES / AGENT_STATUS / AGENT_STOP can observe
  427               -- it after the synchronous run completes. The child is
  428               -- recorded with status Stopped (the synchronous worker has
  429               -- already finished by the time we reach here). Zipped with
  430               -- the input tasks to recover the AgentDefId (the ChildResult
  431               -- carries the SubagentId but not the def id).
  432               let tasks = diTasks di
  433               zipWithM_ (\t r -> liftIO (registerChild (aswRuntime wiring) t r)) tasks results
  434               let rendered = encodeResultsJson results
  435               pure (OpResult [TrpText rendered] False (object ["results" .= results]))
  436    }

  438 -- | Parse the model's input into a 'DelegateInput'. Single mode requires
  439 -- @id@ + @goal@; batch mode requires a @tasks@ array of @{id, goal, ...}@.
  440 parseInput :: Value -> IO (Either Text DelegateInput)
  441 parseInput v =
  442   case parseMaybe (withObject "in" (.:? "tasks")) v :: Maybe (Maybe Value) of
  443     Just (Just (Array arr)) | not (V.null arr) -> do
  444       tasks <- mapM parseTask (V.toList arr)
  445       case sequence tasks of
  446         Left err     -> pure (Left err)
  447         Right tsList -> pure (Right (DiBatch tsList))
  448     _ -> do
  449       let mDefId = idField v
  450           mGoal  = textFieldMaybe "goal" v
  451       case (mDefId, mGoal) of
  452         (Just defId, Just goal) | not (T.null goal) ->
  453           pure (Right (DiSingle (ChildTask defId goal (textFieldMaybe "context" v) (textFieldMaybe "role" v))))
  454         (Just _, Just _) -> pure (Left "AGENT_START requires a non-empty 'goal'.")
  455         (Just _, Nothing) -> pure (Left "AGENT_START single-task mode requires a 'goal'.")
  456         (Nothing, _) -> pure (Left "AGENT_START requires an 'id' (agent def id) in single-task mode, or a 'tasks' array in batch mode.")
  457 
  458 -- | Parse one element of the @tasks@ array into a 'ChildTask'.
  459 parseTask :: Value -> IO (Either Text ChildTask)
  460 parseTask v =
  461   case idField v of
  462     Nothing -> pure (Left "Each task requires an 'id' (agent def id).")
  463     Just defId ->
  464       case textFieldMaybe "goal" v of
  465         Nothing -> pure (Left "Each task requires a 'goal'.")
  466         Just goal | T.null goal -> pure (Left "Each task requires a non-empty 'goal'.")
  467                  | otherwise -> pure (Right (ChildTask defId goal (textFieldMaybe "context" v) (textFieldMaybe "role" v)))
  468 
  469 -- | Resolve a task to its def + worker + fresh session id. Returns Left if
  470 -- the def id is invalid or the def doesn't exist.
  471 resolveTask
  472   :: AgentDefBackend
  473   -> AgentRuntime
  474   -> IO SessionId
  475   -> Int
  476   -> AgentWorkerBuilder
  477   -> ChildTask
  478   -> IO (Either Text (AgentDef, AgentWorkerBuilder, SessionId))
  479 resolveTask defBackend _runtime mintSession _parentDepth worker task = do
  480   case mkAgentDefId (ctDefId task) of
  481     Left err -> pure (Left err)
  482     Right aid -> do
  483       mDef <- adbRead defBackend aid
  484       case mDef of
  485         Nothing  -> pure (Left ("agent def not found: " <> ctDefId task))
  486         Just def -> do
  487           sid <- mintSession
  488           pure (Right (def, worker, sid))
  489 
  490 -- | Register a finished child in the runtime registry (post-hoc; the worker
  491 -- ran synchronously to completion). Records the instance with status
  492 -- 'Stopped' (the synchronous child has already finished by the time this is
  493 -- called), so AGENT_INSTANCES / AGENT_STATUS / AGENT_STOP can observe it.
  494 -- Recovers the 'AgentDefId' from the task's @ctDefId@ (the ChildResult
  495 -- carries the 'SubagentId' but not the def id).
  496 registerChild :: AgentRuntime -> ChildTask -> ChildResult -> IO ()
  497 registerChild runtime task result =
  498   case mkAgentDefId (ctDefId task) of
  499     Left _ -> pure ()  -- malformed def id; skip registration
  500     Right aid ->
  501       case crChildSession result of
  502         Nothing -> pure ()  -- no child session; skip (error/timeout case)
  503         Just session -> do
  504           let depth = 0  -- parent depth; the child's is depth+1 (not tracked here)
  505           registerCompletedAgent runtime aid (crSubagentId result) session depth
  506 
  507 -- | Extract the task list from a 'DelegateInput' (in order).
  508 diTasks :: DelegateInput -> [ChildTask]
  509 diTasks (DiSingle t) = [t]
  510 diTasks (DiBatch ts) = ts

  530           case mStatus of
  531             Nothing -> pure (OpResult [TrpText "not running"] False (object ["subagent_id" .= subagentIdText sid, "status" .= ("stopped" :: Text)]))
  532             Just s  -> pure (OpResult [TrpText (renderStatus s)] False (object ["subagent_id" .= subagentIdText sid, "status" .= renderStatus s]))
  533   }
  534   where
  535     checkSubagentId _ = Right ()
  536 
  537 -- ---------------------------------------------------------------------------
  538 -- AGENT_STOP
  539 -- ---------------------------------------------------------------------------
  540 
  541 -- | AGENT_STOP: stop a running agent instance (kill the thread, deregister).
  542 -- Idempotent. Now keyed by @subagent_id@ (the new AGENT_START returns
  543 -- subagent ids, not def ids).
  544 agentStopOp :: AgentRuntime -> Opcode
  545 agentStopOp runtime = TrustedOpcode
  546   { toName = OpName "AGENT_STOP"
  547   , toTrust = Trusted
  548   , toDesc = "Stop a running agent instance by subagent_id (idempotent)."
  549   , toInSchema = singleStringSchema "subagent_id" "The subagent id to stop."
  550   , toOutSchema = object []
  551   , toAuthorize = maybe (Left "AGENT_STOP requires {subagent_id:string}") (const (Right ())) . subagentIdField
  552   , toRun = \_ v -> do
  553       let mSid = subagentIdField v
  554       case mSid of
  555         Nothing -> pure (OpResult [TrpText "invalid subagent id"] True (object []))
  556         Just sid -> do
  557           _ <- liftIO (stopAgent runtime sid)
  558           pure (OpResult [TrpText "stopped"] False (object ["subagent_id" .= subagentIdText sid]))
  559   }
  560 
  561 -- ---------------------------------------------------------------------------
  562 -- AGENT_INTERRUPT
  563 -- ---------------------------------------------------------------------------
  564 
  565 -- | AGENT_INTERRUPT: request that a single running subagent stop at its next
  566 -- iteration boundary. Unlike AGENT_STOP (which hard-kills the thread),
  567 -- AGENT_INTERRUPT sets a flag the worker polls between turns, letting it
  568 -- exit cleanly. Trusted.
  569 agentInterruptOp :: AgentRuntime -> Opcode
  570 agentInterruptOp runtime = TrustedOpcode
  571   { toName = OpName "AGENT_INTERRUPT"
  572   , toTrust = Trusted
  573   , toDesc = "Request that a running subagent stop at its next iteration boundary (cooperative; the worker polls an interrupt flag between turns)."
  574   , toInSchema = singleStringSchema "subagent_id" "The subagent id to interrupt."
  575   , toOutSchema = object []
  576   , toAuthorize = maybe (Left "AGENT_INTERRUPT requires {subagent_id:string}") (const (Right ())) . subagentIdField
  577   , toRun = \_ v -> do
  578       let mSid = subagentIdField v
  579       case mSid of
  580         Nothing -> pure (OpResult [TrpText "invalid subagent id"] True (object []))
  581         Just sid -> do
  582           found <- liftIO (interruptAgent runtime sid)
  583           let msg = if found then "interrupt requested" else "subagent not running"
  584           pure (OpResult [TrpText msg] False (object ["subagent_id" .= subagentIdText sid, "found" .= found]))
  585   }

  587 -- ---------------------------------------------------------------------------
  588 -- JSON encoding of ChildResult
  589 -- ---------------------------------------------------------------------------
  590 
  591 -- | Render the results list as a JSON string for the model (the @orParts@
  592 -- text the model sees). One line per result:
  593 -- @subagent_id | status | summary-or-error@. The full structured JSON goes
  594 -- into 'orRecorded' via the 'ToJSON ChildResult' instance defined in
  595 -- 'Seal.Agent.Runtime.Delegation'.
  596 encodeResultsJson :: [ChildResult] -> Text
  597 encodeResultsJson [] = "(no results)"
  598 encodeResultsJson rs = T.intercalate "\n" (map renderOne rs)
  599   where
  600     renderOne r =
  601       subagentIdText (crSubagentId r) <> " | " <>
  602       T.pack (show (crStatus r)) <> " | " <>
  603       maybe "(no summary)" (T.take 200) (summaryOrError r)
  604     summaryOrError r = case crSummary r of
  605       Just s  -> Just s
  606       Nothing -> crError r
  607 
  608 renderStatus :: AgentStatus -> Text
  609 renderStatus = \case
  610   Starting     -> "starting"
  611   Running      -> "running"
  612   Idle         -> "idle"
  613   Stopped      -> "stopped"
  614   Interrupted  -> "interrupted"
  615   Crashed m    -> "crashed: " <> m
  616 
  617 -- | Extract the @subagent_id@ string field from a JSON object.
  618 subagentIdField :: Value -> Maybe SubagentId
  619 subagentIdField v = do
  620   t <- parseMaybe (withObject "in" (.: "subagent_id")) v
  621   if T.null t then Nothing else Just (SubagentId t)
  622 
  623 -- ---------------------------------------------------------------------------
  624 -- Helpers (def rendering / encoding)
  625 -- ---------------------------------------------------------------------------
  626 
  627 -- | Build a JSON-Schema object with a single required string property.
  628 singleStringSchema :: Text -> Text -> Value
  629 singleStringSchema fieldName fieldDesc =
  630   object
  631     [ "type" .= ("object" :: Text)
  632     , "properties" .= object
  633         [fromText fieldName .= object
  634            [ "type" .= ("string" :: Text)
  635            , "description" .= fieldDesc
  636            ]]
  637     , "required" .= ([fieldName] :: [Text])
  638     ]
  639 
  640 -- | Encode the secret-free 'AgentDef' fields into the 'orRecorded' payload.
  641 -- The @was_new@ flag distinguishes create vs update in the audit log.
  642 encodeDefRecorded :: AgentDef -> Bool -> Value
  643 encodeDefRecorded d wasNew = object
  644   [ "id"         .= agentDefIdText (adId d)
  645   , "name"       .= adName d
  646   , "provider"   .= adProvider d
  647   , "model"      .= adModel d
  648   , "system"     .= adSystem d
  649   , "tools"      .= encodeTools (adTools d)
  650   , "group"      .= adGroup d
  651   , "created_at" .= adCreatedAt d
  652   , "updated_at" .= adUpdatedAt d
  653   , "session"    .= adSession d
  654   , "was_new"    .= wasNew
  655   ]
  656 
  657 -- | Encode an 'AllowList OpName' for the recorded payload: @\"all\"@ for
  658 -- 'AllowAll', or a JSON array of opcode-name strings for 'AllowOnly'.
  659 encodeTools :: AllowList OpName -> Value
  660 encodeTools AllowAll       = String "all"
  661 encodeTools (AllowOnly xs) = Array (V.fromList [ String t | OpName t <- Set.toList xs ])
  662 
  663 -- | Render an 'AgentDef' as a Markdown-ish text block for the model.
  664 renderDef :: AgentDef -> Text
  665 renderDef d =
  666   "# " <> adName d <> " (" <> agentDefIdText (adId d) <> ")\n\n"
  667   <> "provider: " <> adProvider d <> "\n"
  668   <> "model: " <> modelName <> "\n"
  669   <> "system: " <> fromMaybe "(none)" (adSystem d) <> "\n"
  670   <> "tools: " <> renderTools (adTools d)
  671   where
  672     ModelId modelName = adModel d
  673 
  674 -- | Render an 'AllowList OpName' as a comma-separated list, or @\"all\"@.
  675 renderTools :: AllowList OpName -> Text
  676 renderTools AllowAll       = "all"
  677 renderTools (AllowOnly xs) = T.intercalate ", " [ t | OpName t <- Set.toList xs ]

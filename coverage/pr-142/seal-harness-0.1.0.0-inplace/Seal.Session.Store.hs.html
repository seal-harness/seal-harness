   66 -- | @YYYYMMDD-HHMMSS-mmm@. Timestamp-leading so a lexicographic sort of ids is
   67 -- chronological; only digits and dashes, so it is a valid 'SessionId'.
   68 formatSessionId :: UTCTime -> Text
   69 formatSessionId t =
   70   let base   = formatTime defaultTimeLocale "%Y%m%d-%H%M%S" t
   71       millis = (diffTimeToPicoseconds (utctDayTime t) `div` 1000000000) `mod` 1000
   72       s      = show millis
   73       mmm    = replicate (3 - length s) '0' <> s
   74   in T.pack (base <> "-" <> mmm)

   76 -- | Mint a fresh 'SessionMeta' (id + timestamps) for the given selection,
   77 -- without writing anything to disk. Used by channels that want an
   78 -- in-memory active session without polluting the sessions list — the
   79 -- session is only persisted when the user actually sends a message.
   80 newSessionMeta :: SealPaths -> Text -> Text -> Text -> Maybe AgentDefId -> IO SessionMeta
   81 newSessionMeta _paths provider model channel mAgent = do
   82   now <- getCurrentTime
   83   sid <- case mkSessionId (formatSessionId now) of
   84     Right s -> pure s
   85     Left e  -> ioError (userError ("session id generation failed: " <> T.unpack e))
   86   pure SessionMeta
   87     { smId = sid, smProvider = provider, smModel = model
   88     , smChannel = channel, smAgent = mAgent
   89     , smSystemOverride = Nothing, smAgentName = Nothing
   90     , smDescription = Nothing
   91     , smCreatedAt = now, smLastActive = now }
   92 
   93 -- | Create a fresh session directory + session.json for the given selection.
   94 newSession :: SealPaths -> Text -> Text -> Text -> Maybe AgentDefId -> IO SessionMeta
   95 newSession paths provider model channel mAgent = do
   96   meta <- newSessionMeta paths provider model channel mAgent
   97   saveSessionMeta paths meta
   98   pure meta

  100 -- | Persist @session.json@ atomically: dir 0700, write a unique temp file,
  101 -- chmod 0600, rename. The temp file is unique per call (via 'openBinaryTempFile')
  102 -- so concurrent saves from multiple threads (channel loop + gateway API +
  103 -- /model command) don't collide on a shared @.tmp@ path ("resource busy"
  104 -- / "does not exist" rename race). The rename is atomic on POSIX.
  105 saveSessionMeta :: SealPaths -> SessionMeta -> IO ()
  106 saveSessionMeta paths meta = do
  107   let dir  = sessionDir paths (smId meta)
  108       path = sessionMetaPath paths (smId meta)
  109   createDirectoryIfMissing True dir
  110   setFileMode dir 0o700
  111   (tmp, h) <- openBinaryTempFile dir (takeBaseName path <> ".tmp")
  112   BL.hPut h (encode meta)
  113   hClose h
  114   setFileMode tmp 0o600
  115   renameFile tmp path

  117 -- | All non-archived sessions, newest 'smLastActive' first. Corrupt/undecodable
  118 -- session.json files are silently skipped (a partial write never breaks the
  119 -- list). A session is archived when an @archived@ marker file exists in its
  120 -- session directory (see 'sessionArchivedMarkerPath'); archived sessions are
  121 -- returned by 'listArchivedSessions' instead.
  122 listSessions :: SealPaths -> IO [SessionMeta]
  123 listSessions paths = do
  124   let root = sessionsRoot paths
  125   exists <- doesDirectoryExist root
  126   if not exists
  127     then pure []
  128     else do
  129       entries <- listDirectory root
  130       dirs    <- filterM (doesDirectoryExist . (root </>)) entries
  131       metas   <- forM dirs $ \e -> do
  132         let mp = root </> e </> "session.json"
  133             marker = root </> e </> "archived"
  134         ok <- doesFileExist mp
  135         archived <- doesFileExist marker
  136         if not ok || archived then pure Nothing else decodeFileStrict mp
  137       pure (sortOn (Down . smLastActive) (catMaybes metas))
  138 
  139 -- | All archived sessions (those with an @archived@ marker file), newest
  140 -- 'smLastActive' first. Corrupt/undecodable session.json files are silently
  141 -- skipped. The marker file's presence is the source of truth; a missing
  142 -- session.json with a marker is also skipped (defensive against a stale
  143 -- marker left by an out-of-band delete).
  144 listArchivedSessions :: SealPaths -> IO [SessionMeta]
  145 listArchivedSessions paths = do
  146   let root = sessionsRoot paths
  147   exists <- doesDirectoryExist root
  148   if not exists
  149     then pure []
  150     else do
  151       entries <- listDirectory root
  152       dirs    <- filterM (doesDirectoryExist . (root </>)) entries
  153       metas   <- forM dirs $ \e -> do
  154         let mp = root </> e </> "session.json"
  155             marker = root </> e </> "archived"
  156         ok <- doesFileExist mp
  157         archived <- doesFileExist marker
  158         if not ok || not archived then pure Nothing else decodeFileStrict mp
  159       pure (sortOn (Down . smLastActive) (catMaybes metas))

  161 -- | Set or clear the archived flag on a session by creating/removing the
  162 -- @archived@ marker file. Returns 'False' when the session's @session.json@
  163 -- can't be found (the caller surfaces a 404); 'True' on success. The
  164 -- transcript and session.json are never touched — archiving is a pure UI
  165 -- hint. Idempotent: archiving an already-archived session (or unarchiving a
  166 -- non-archived one) is a no-op success.
  167 updateSessionArchived :: SealPaths -> SessionId -> Bool -> IO Bool
  168 updateSessionArchived paths sid archived = do
  169   let mp = sessionMetaPath paths sid
  170       marker = sessionArchivedMarkerPath paths sid
  171   exists <- doesFileExist mp
  172   if not exists
  173     then pure False
  174     else do
  175       markerExists <- doesFileExist marker
  176       if archived
  177         then unless markerExists (writeFile marker "")
  178         else when markerExists (removeFile marker)
  179       pure True

  181 -- | The provider label + model a new session should start with: the configured
  182 -- defaults, falling back to the configured provider's own default model (or
  183 -- Anthropic when no provider is configured).
  184 defaultSessionSelection :: RuntimeConfig -> (Text, Text)
  185 defaultSessionSelection cfg =
  186   ( provLabel
  187   , fromMaybe fallbackModel (rcDefaultModel cfg) )
  188   where
  189     provLabel = fromMaybe "anthropic" (rcDefaultProvider cfg)
  190     ModelId fallbackModel = resolveDefaultModel (providerDefaultModel cfg provLabel) provLabel

  192 -- | Resolve the default agent for a new session. Returns 'Nothing' if
  193 -- @default_agent@ is unset, the id fails validation, or the def is not on
  194 -- disk (a warning is emitted in the latter case). Otherwise returns the
  195 -- bound 'AgentDefId' and the def's provider/model overrides (non-empty
  196 -- values that should replace the config defaults).
  197 resolveDefaultAgent
  198   :: AgentDefBackend -> RuntimeConfig -> IO (Maybe AgentDefId, Maybe Text, Maybe Text)
  199 resolveDefaultAgent backend cfg =
  200   case rcDefaultAgent cfg of
  201     Nothing -> pure (Nothing, Nothing, Nothing)
  202     Just raw ->
  203       case mkAgentDefId raw of
  204         Left _ -> do
  205           putStrLn ("warning: default_agent " <> T.unpack raw <> " is not a valid id; proceeding without one")
  206           pure (Nothing, Nothing, Nothing)
  207         Right aid -> do
  208           mDef <- adbRead backend aid
  209           case mDef of
  210             Nothing -> do
  211               putStrLn ("warning: default agent " <> T.unpack raw <> " not found; proceeding without one")
  212               pure (Nothing, Nothing, Nothing)
  213             Just d  -> pure (Just aid, override (adProvider d), overrideModel (adModel d))
  214   where
  215     override t = if T.null t then Nothing else Just t
  216     overrideModel (ModelId m) = if T.null m then Nothing else Just m

  218 -- | Create a new session from the config defaults, on the @cli@ channel.
  219 -- If @default_agent@ is set and the def exists, its id is persisted in
  220 -- 'smAgent' and its non-empty provider/model override the config defaults
  221 -- (PureClaw @resolveOverride@ precedence: frontmatter > config > default).
  222 initSession :: SealPaths -> RuntimeConfig -> AgentDefBackend -> IO SessionMeta
  223 initSession paths cfg backend = do
  224   (mAgent, mProv, mModel) <- resolveDefaultAgent backend cfg
  225   let (cfgProv, cfgModel) = defaultSessionSelection cfg
  226       provider = fromMaybe cfgProv mProv
  227       model    = fromMaybe cfgModel mModel
  228   newSession paths provider model "cli" mAgent

  230 -- | Build an in-memory 'SessionMeta' from the config defaults, on the @web@
  231 -- channel, WITHOUT persisting to disk. The web gateway uses this so the
  232 -- active-session ref has a valid meta (provider/model fallbacks) without
  233 -- polluting the sessions list. The session is persisted only when the user
  234 -- actually sends the first message (which writes the transcript).
  235 initSessionMeta :: SealPaths -> RuntimeConfig -> AgentDefBackend -> IO SessionMeta
  236 initSessionMeta paths cfg backend = do
  237   (mAgent, mProv, mModel) <- resolveDefaultAgent backend cfg
  238   let (cfgProv, cfgModel) = defaultSessionSelection cfg
  239       provider = fromMaybe cfgProv mProv
  240       model    = fromMaybe cfgModel mModel
  241   newSessionMeta paths provider model "web" mAgent

  243 -- | Update the bound agent for an existing session. Loads the current
  244 -- 'SessionMeta' from disk, replaces 'smAgent' (Nothing when the supplied id
  245 -- is empty/invalid), and persists atomically. Returns 'False' when the
  246 -- session's @session.json@ can't be found or parsed (the caller surfaces a
  247 -- 404/400); 'True' on a successful write.
  248 --
  249 -- /Mutual exclusion/: binding an agent clears any 'smSystemOverride'
  250 -- (one-off uploaded file) — the two are alternative sources of the system
  251 -- prompt, and the backend enforces they never coexist so the UI never
  252 -- shows a stale \"agent X\" label while an uploaded file is actually
  253 -- driving the prompt. The frontend only needs to fire this single PUT.
  254 -- 'smAgentName' is set to the agent's id so the sidebar has a stable
  255 -- display label.
  256 updateSessionAgent :: SealPaths -> SessionId -> Maybe AgentDefId -> IO Bool
  257 updateSessionAgent paths sid mAgent = do
  258   let mp = sessionMetaPath paths sid
  259   exists <- doesFileExist mp
  260   if not exists
  261     then pure False
  262     else do
  263       mMeta <- decodeFileStrict mp :: IO (Maybe SessionMeta)
  264       case mMeta of
  265         Nothing  -> pure False
  266         Just meta -> do
  267           -- Clear the override + set the display name when binding an
  268           -- agent; clear both when clearing the agent.
  269           let next = case mAgent of
  270                 Just aid -> meta { smAgent = mAgent
  271                                  , smSystemOverride = Nothing
  272                                  , smAgentName = Just (agentDefIdText aid) }
  273                 Nothing  -> meta { smAgent = Nothing, smAgentName = Nothing }
  274           saveSessionMeta paths next
  275           pure True

  277 -- | Update (or clear) the ad-hoc system prompt override for a session.
  278 -- 'Nothing' clears the override (fall back to the bound agent's
  279 -- 'adSystem'); 'Just t' sets it so 'plainTurn' uses @t@ verbatim as the
  280 -- system prompt. Returns 'False' when the session's @session.json@ can't
  281 -- be found or parsed; 'True' on a successful write.
  282 --
  283 -- The display label is resolved in this order:
  284 -- 1. The @id@ field in the file's TOML frontmatter (when parseable).
  285 -- 2. The caller-supplied @mFallbackName@ (e.g. the uploaded filename).
  286 -- 3. 'Nothing' — no label.
  287 --
  288 -- /Mutual exclusion/: setting an override clears 'smAgent' so the session
  289 -- doesn't display a stale agent label in the UI while an uploaded file is
  290 -- actually driving the prompt. Clearing the override leaves 'smAgent'
  291 -- untouched (the caller can re-bind an agent in a follow-up PUT if
  292 -- desired). The frontend only needs to fire this single PUT.
  293 updateSessionSystemOverride
  294   :: SealPaths
  295   -> SessionId
  296   -> Maybe Text          -- ^ the file content (Nothing = clear)
  297   -> Maybe Text          -- ^ fallback display label (e.g. filename) when no frontmatter id
  298   -> IO Bool
  299 updateSessionSystemOverride paths sid mOverride mFallbackName = do
  300   let mp = sessionMetaPath paths sid
  301   exists <- doesFileExist mp
  302   if not exists
  303     then pure False
  304     else do
  305       mMeta <- decodeFileStrict mp :: IO (Maybe SessionMeta)
  306       case mMeta of
  307         Nothing  -> pure False
  308         Just meta -> do
  309           let next = case mOverride of
  310                 Just content ->
  311                   let label = parseAgentFileId content <|> mFallbackName
  312                   in meta { smSystemOverride = Just content
  313                           , smAgent = Nothing
  314                           , smAgentName = label }
  315                 Nothing ->
  316                   -- Clearing: leave smAgentName untouched (the follow-up
  317                   -- updateSessionAgent from the Remove button will reset
  318                   -- it to the default agent's id).
  319                   meta { smSystemOverride = Nothing }
  320           saveSessionMeta paths next
  321           pure True

  323 -- | Update (or clear) the user-set display description for a session.
  324 -- 'Nothing' clears it (the UI then falls back to the auto-summary /
  325 -- first-message snippet / agent name); 'Just t' sets it (a blank or
  326 -- all-whitespace string is normalized to 'Nothing' so the caller can pass
  327 -- the raw input). Returns 'False' when the session's @session.json@ can't
  328 -- be found or parsed; 'True' on a successful write. Only touches
  329 -- 'smDescription' — all other fields (agent binding, override, etc.) are
  330 -- preserved.
  331 updateSessionDescription :: SealPaths -> SessionId -> Maybe Text -> IO Bool
  332 updateSessionDescription paths sid mDesc = do
  333   let mp = sessionMetaPath paths sid
  334   exists <- doesFileExist mp
  335   if not exists
  336     then pure False
  337     else do
  338       mMeta <- decodeFileStrict mp :: IO (Maybe SessionMeta)
  339       case mMeta of
  340         Nothing  -> pure False
  341         Just meta -> do
  342           let next = meta { smDescription = normalized }
  343           saveSessionMeta paths next
  344           pure True
  345   where
  346     normalized = case mDesc of
  347       Just t
  348         | not (T.null (T.strip t)) -> Just t
  349         | otherwise                -> Nothing
  350       Nothing                      -> Nothing

  352 -- | Parse the @id@ field from an uploaded agent file's TOML frontmatter.
  353 -- Returns 'Nothing' when the file has no frontmatter or no @id@ key.
  354 parseAgentFileId :: Text -> Maybe Text
  355 parseAgentFileId content =
  356   let (fm, _body) = decodeDoc content
  357   in case fmLookup "id" fm of
  358        Just t | not (T.null (T.strip t)) -> Just (T.strip t)
  359        _                                 -> Nothing

  361 -- | After a successful SETUP_REPO, rebind the session's 'smAgent' to the
  362 -- cloned repo's project-level @.agents\/agents.md@ def (id
  363 -- @\<repo\>--agents-md@) when the repo ships one. This makes the repo's
  364 -- project instructions the effective system prompt on the next turn,
  365 -- superseding the @default_agent@ stamped at session creation.
  366 --
  367 -- The discovery runs over the 'WorkdirFs' handle (the single
  368 -- 'mkSessionExec'-constructed chokepoint), so the behavior is identical
  369 -- for @mode=local@ and @mode=remote@ — the workdir is scanned through the
  370 -- same handle in both arms, never via direct filesystem access.
  371 --
  372 -- /Idempotence / non-clobber/: the bind is skipped when the session's
  373 -- current 'smAgent' is already a repo-prefixed def (its id text contains
  374 -- the @"--"@ repo-prefix separator). So a user who has explicitly picked a
  375 -- repo agent (or another repo's agent) via the dropdown is left alone on a
  376 -- re-clone / subsequent SETUP_REPO. A session still on its creation-time
  377 -- @default_agent@ (e.g. @zoe@, which has no @"--"@) is rebound to the
  378 -- repo's @agents-md@; a session with no bound agent ('Nothing') is also
  379 -- rebound. Missing @session.json@, no @.agents\/agents.md@ in the workdir,
  380 -- or a discovery error → no-op (fail-closed, never throws).
  381 autoBindRepoAgent :: WorkdirFs -> SealPaths -> SessionId -> IO ()
  382 autoBindRepoAgent wfs paths sid = do
  383   backend <- workdirAgentDefBackend wfs
  384   defs <- adbList backend
  385   autoBindRepoAgentWith defs paths sid
  386 
  387 -- | 'autoBindRepoAgent' over PRE-COMPUTED defs — the turn engine calls this
  388 -- with the discovery cache's scan result so no extra scan is issued.
  389 autoBindRepoAgentWith :: [AgentDef] -> SealPaths -> SessionId -> IO ()
  390 autoBindRepoAgentWith defs paths sid =
  391   case agentsMdDef defs of
  392     Nothing      -> pure ()
  393     Just agentsMdAid -> do
  394       mMeta <- decodeFileStrict (sessionMetaPath paths sid) :: IO (Maybe SessionMeta)
  395       case mMeta of
  396         Nothing -> pure ()
  397         Just meta
  398           | alreadyRepoBound (smAgent meta) -> pure ()
  399           | otherwise -> void (updateSessionAgent paths sid (Just agentsMdAid))
  400   where
  401     -- The repo default is the prefixed agents-md def (e.g.
  402     -- @my-repo--agents-md@). The prefix varies per repo, so match by the
  403     -- @"--agents-md"@ suffix (same rule as 'handleSessionAgents' in the
  404     -- gateway). Returns the first match (alphabetical repo order, matching
  405     -- 'listWorkdirAgentDefs' determinism).
  406     agentsMdDef ds =
  407       case [ adId d | d <- ds, "--agents-md" `T.isSuffixOf` agentDefIdText (adId d) ] of
  408         (aid:_) -> Just aid
  409         []      -> Nothing
  410     -- A def is repo-prefixed when its id text contains the @"--"@ separator
  411     -- ('prefixWorkdirDef' uses @"--"@ between the repo dir name and the
  412     -- def id). User-store ids never contain @"--"@ (they're single-segment
  413     -- ids). This is a deliberate superset: any repo-prefixed binding counts
  414     -- as "user already chose a repo agent" so we don't clobber it.
  415     alreadyRepoBound Nothing = False
  416     alreadyRepoBound (Just aid) = "--" `T.isInfixOf` agentDefIdText aid

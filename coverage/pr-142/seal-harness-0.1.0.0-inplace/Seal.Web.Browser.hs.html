    2 -- | BROWSER_* opcodes (Untrusted): a thin abstraction over a pluggable
    3 -- browser driver (Playwright/headless-Chrome in a future phase). The
    4 -- default driver is fail-closed (@noBrowserDriver@) — the opcodes
    5 -- surface a structured error when no driver is configured.

   25 -- | The browser-driver interface — a record of IO actions. A real
   26 -- Playwright/headless-Chrome driver lands in a future phase; the default
   27 -- is 'noBrowserDriver' (fail-closed).
   28 data BrowserDriver = BrowserDriver
   29   { bdOpen  :: Text -> IO (Either Text Text)       -- ^ open a URL → page id
   30   , bdClick :: Text -> IO (Either Text Text)       -- ^ click a selector → ok
   31   , bdRead  :: Text -> IO (Either Text Text)       -- ^ read page text → content
   32   }
   33 
   34 -- | The fail-closed default driver.
   35 noBrowserDriver :: BrowserDriver
   36 noBrowserDriver = BrowserDriver
   37   { bdOpen  = \_ -> pure (Left "BROWSER_OPEN: no browser driver configured")
   38   , bdClick = \_ -> pure (Left "BROWSER_CLICK: no browser driver configured")
   39   , bdRead  = \_ -> pure (Left "BROWSER_READ: no browser driver configured")
   40   }

   42 -- | BROWSER_OPEN opcode. Input: @{ url: Text }@.
   43 browserOpenOp :: BrowserDriver -> Opcode
   44 browserOpenOp _drv = UntrustedOpcode
   45   { uoName = OpName "BROWSER_OPEN"
   46   , uoDesc = "Open a URL in a browser (driver-pluggable, fail-closed default)."
   47   , uoInSchema = browserSchema "url" "The URL to open."
   48   , uoOutSchema = object []
   49   , uoAuthorize = \v ->
   50       case urlField v of
   51         Nothing -> Left "BROWSER_OPEN requires {url:string}"
   52         Just u | T.null u -> Left "BROWSER_OPEN: url is empty"
   53                | otherwise -> Right ()
   54   , uoRun = \v -> do
   55       let u = fromMaybe "" (urlField v)
   56           recorded = object [ "url" .= u ]
   57       pure (OpResult [TrpText "BROWSER_OPEN: no browser driver configured"] True recorded)
   58   }
   59 
   60 -- | BROWSER_CLICK opcode. Input: @{ selector: Text }@.
   61 browserClickOp :: BrowserDriver -> Opcode
   62 browserClickOp _drv = UntrustedOpcode

   64   , uoDesc = "Click an element in a browser (driver-pluggable, fail-closed default)."

   78 -- | BROWSER_READ opcode. Input: @{ selector: Text }@.
   79 browserReadOp :: BrowserDriver -> Opcode
   80 browserReadOp _drv = UntrustedOpcode
   81   { uoName = OpName "BROWSER_READ"
   82   , uoDesc = "Read text from a browser page (driver-pluggable, fail-closed default)."
   83   , uoInSchema = browserSchema "selector" "The CSS selector to read (or empty for the whole page)."
   84   , uoOutSchema = object []
   85   , uoAuthorize = \_v -> Right ()  -- selector is optional (empty = whole page)
   86   , uoRun = \v -> do
   87       let s = fromMaybe "" (selectorField v)
   88           recorded = object [ "selector" .= s ]
   89       pure (OpResult [TrpText "BROWSER_READ: no browser driver configured"] True recorded)
   90   }

   92 browserSchema :: Text -> Text -> Value
   93 browserSchema fieldName desc =
   94   object
   95     [ "type" .= ("object" :: Text)
   96     , "properties" .= object
   97         [ fromText fieldName .= object
   98             [ "type" .= ("string" :: Text)
   99             , "description" .= desc
  100             ]
  101         ]
  102     , "required" .= [fieldName]
  103     ]
  104 
  105 urlField :: Value -> Maybe Text
  106 urlField = parseMaybe (withObject "in" (.: "url"))
  107 
  108 selectorField :: Value -> Maybe Text
  109 selectorField = parseMaybe (withObject "in" (.: "selector"))

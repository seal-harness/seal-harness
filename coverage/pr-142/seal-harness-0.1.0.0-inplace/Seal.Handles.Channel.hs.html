    1 -- | The widened channel capability record — the target the Signal channel
    2 -- (Phase 2b) and the web gateway (Phase 7) implement. Coexists with the
    3 -- existing 'Seal.Channel.Caps.ChannelCaps' (kept for the CLI TUI, which is
    4 -- not unified into this handle in Phase 2a). House style: no type class;
    5 -- callers receive the handle and call fields directly.
    6 module Seal.Handles.Channel
    7   ( ChannelHandle (..)
    8   , Deferral (..)
    9   ) where
   10 
   11 import Data.Text (Text)
   12 
   13 import Seal.Core.MessageSource (MessageSource)
   14 
   15 -- | A structured deferral for interactive ops on request/response channels
   16 -- (Signal, the future unified CLI). The web channel is async-only and never
   17 -- returns 'Deferred' — it returns 'AsyncQueued'. Kept simple in 2a: the
   18 -- payload is just a marker the caller can match; 2b fills in the real shape
   19 -- (a continuation id + a timeout) when a real consumer demands it.
   20 data Deferral = Deferred | AsyncQueued
   21   deriving stock (Eq, Show)
   22 
   23 -- | The widened channel capability record. Every field is an IO action so the
   24 -- type is uniform between real and fake variants.
   25 data ChannelHandle = ChannelHandle
   26   { chLabel       :: Text
   27   -- ^ The channel's label (e.g. @"telegram"@, @"signal"@, @"cli"@, @"web"@).
   28   -- Used by the reply registry to dedup subscribers: at most one handle per
   29   -- channel kind is kept per session, so re-subscribing the same channel
   30   -- replaces the old handle but a different channel's handle is preserved.
   31   -- The web frontend is NOT in the registry (it receives entries via the WS
   32   -- broker), so only append-only chat channels appear here.
   33   , chSend        :: Text -> IO ()
   34   -- ^ Emit one line to the user.
   35   , chSendError   :: Text -> IO ()
   36   -- ^ Emit an error line (may be formatted differently on some channels).
   37   , chSendChunk   :: Text -> IO ()
   38   -- ^ Emit one streaming chunk (for tool output / long replies). Channels
   39   -- that do not stream may batch and call 'chSend' once.
   40   , chPrompt      :: Text -> IO (Either Deferral Text)
   41   -- ^ Visible prompt; returns 'Right' the typed line on interactive channels,
   42   -- 'Left Deferred' on channels that cannot answer inline (the caller must
   43   -- wait for a follow-on message), 'Left AsyncQueued' on async channels.
   44   , chPromptSecret :: Text -> IO (Either Deferral Text)
   45   -- ^ Hidden (no-echo) prompt; same return shape as 'chPrompt'.
   46   , chStreaming   :: Bool
   47   -- ^ Whether this channel benefits from streaming (web: yes; Signal: yes,
   48   -- chunked; CLI TUI: yes, line-by-line).
   49   , chReadSecret  :: IO (Maybe Text)
   50   -- ^ Pull a secret the channel itself holds (e.g. a pairing token).
   51   -- 'Nothing' on channels with no channel-held secret. NOT a vault accessor
   52   -- — the vault is reached via the vault handle, not the channel.
   53   , chReceive     :: IO (Maybe MessageSource, Text)
   54   -- ^ Pull the next inbound message from the channel's inbox, with its
   55   -- authenticated 'MessageSource'. Returns @(Nothing, "")@ when the inbox
   56   -- is empty (the caller may block or poll depending on the channel).
   57   , chLastChatId  :: IO (Maybe Text)
   58   -- ^ The last chat id the channel addressed a reply to (for transports
   59   -- that need it, e.g. Telegram's @tgSendWithKeyboard@ which addresses a
   60   -- chat by id). @Nothing@ for channels that address by user id (Signal)
   61   -- or have no chat-id concept (CLI, web, fakes).
   62   }

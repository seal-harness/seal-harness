    2 -- | The agent-definition store backend. Disk is canonical. Two on-disk
    3 -- schemes are discovered:
    4 --
    5 -- 1. **FlatScheme** — one Markdown file per def at @agents\/\<id\>.md@,
    6 --    frontmatter (id\/name\/provider\/model\/tools\/timestamps\/session) +
    7 --    body = system prompt. 'AGENT_DEF_CREATE' \/ 'AGENT_DEF_UPDATE' write
    8 --    this form. This is the model-authored channel.
    9 --
   10 -- 2. **DirScheme** (PureClaw-compatible) — a subdirectory per agent at
   11 --    @agents\/\<id\>\/@, with optional TOML frontmatter on @AGENTS.md@
   12 --    (model\/provider\/tools) and the system prompt composed by reading
   13 --    bootstrap files (@SOUL.md@, @USER.md@, @AGENTS.md@ body,
   14 --    @MEMORY.md@, @IDENTITY.md@, @TOOLS.md@, @BOOTSTRAP.md@) in fixed
   15 --    order with @--- SOUL ---@-style section markers. This is the
   16 --    human-authored channel — drop a directory under @agents\/@ and it
   17 --    is discovered.
   18 --
   19 -- **Conflict policy**: if both @agents\/\<id\>.md@ and @agents\/\<id\>\/@
   20 -- exist, the flat file wins (it carries provenance and is the
   21 -- model-authored form). The backend emits a warning on collision rather
   22 -- than silently deduplicating.
   23 --
   24 -- **Directories are a one-time import path**: the first
   25 -- 'AGENT_DEF_CREATE' \/ 'AGENT_DEF_UPDATE' for a DirScheme agent writes
   26 -- @agents\/\<id\>.md@ (taking the composed prompt as the flat body),
   27 -- after which the flat file takes precedence. The user can delete the
   28 -- original directory at their leisure.
   29 --
   30 -- 'markdownAgentDefBackend' reads by enumerating the directory (both
   31 -- schemes) and writes by atomic file replace + auto-commit.
   32 -- 'noneBackend' (in-memory) is kept for tests.
   33 --
   34 -- The git repo is the versioning + audit layer; model-authored writes
   35 -- (@AGENT_DEF_CREATE@ \/ @AGENT_DEF_UPDATE@, which are Trusted file writes)
   36 -- auto-commit.
   37 --
   38 -- The /workdir-scoped/ discovery backend (for @.agents\/@ discovery from
   39 -- cloned repos) lives in "Seal.Agent.Def.Workdir" and is re-exported here.
   40 -- That module has NO direct @System.Directory@\/@Data.Text.IO@ access —
   41 -- every workspace read goes through the 'WorkdirFs' handle (§3.6). This
   42 -- module (the user store) is local-FS by design (§3.9).

   92 -- | The agent-definition store capability. Each operation is IO; 'adbList'
   93 -- returns all defs sorted by id. (Defined in "Seal.Agent.Def.Workdir" to
   94 -- avoid an import cycle; re-exported here.)

   96 -- | The in-memory backend: a single 'IORef' over a 'Map'. Used by tests.
   97 noneBackend :: IO AgentDefBackend
   98 noneBackend = do
   99   ref <- newIORef (Map.empty :: Map AgentDefId AgentDef)
  100   pure AgentDefBackend
  101     { adbRead   = \aid -> Map.lookup aid <$> readIORef ref
  102     , adbUpdate = \d -> modifyIORef' ref (Map.insert (adId d) d)
  103     , adbList   = Map.elems <$> readIORef ref
  104     , adbDelete = modifyIORef' ref . Map.delete
  105     }

  107 -- | The Markdown backend. Discovers both flat @agents\/\<id\>.md@ files and
  108 -- PureClaw-style @agents\/\<id\>\/@ subdirectories. Writes are atomic
  109 -- (tmp → chmod 0600 → rename) and auto-committed to the config git repo.
  110 -- Malformed files / dirs are skipped.
  111 markdownAgentDefBackend :: FilePath -> ConfigRepo -> IO AgentDefBackend
  112 markdownAgentDefBackend dir repo = pure AgentDefBackend
  113   { adbRead   = readAgentDef dir
  114   , adbUpdate = writeAgentDef dir repo
  115   , adbList   = listAgentDefs dir
  116   , adbDelete = deleteAgentDef dir repo
  117   }

  123 -- | The filename for a def in the flat layout: @\<id\>.md@.
  124 defFile :: FilePath -> AgentDefId -> FilePath
  125 defFile dir aid = dir </> T.unpack (agentDefIdText aid) <.> "md"
  126 
  127 -- | The directory for a def: @agents\/\<id\>@.
  128 defDir :: FilePath -> AgentDefId -> FilePath
  129 defDir dir aid = dir </> T.unpack (agentDefIdText aid)
  130 
  131 -- | The full path for a def write, honoring its group. A grouped def lives
  132 -- at @\<root\>\/\<group\>\/\<id\>.md@; an ungrouped def (or one whose group
  133 -- is empty) lives at @\<root\>\/\<id\>.md@ (back-compat with the flat
  134 -- layout). The group directory is created on write. Mirrors
  135 -- 'Seal.Skills.Backend.skillPath'.
  136 defPath :: FilePath -> AgentDef -> FilePath
  137 defPath root d =
  138   case adGroup d of
  139     Just g | not (T.null (T.strip g)) ->
  140       root </> T.unpack (T.strip g) </> T.unpack (agentDefIdText (adId d)) <.> "md"
  141     _ -> defFile root (adId d)
  142 
  143 -- | Write one def to disk (atomic) and auto-commit. Honors 'adGroup': a
  144 -- grouped def is written under @\<root\>\/\<group\>\/\<id\>.md@ (the group
  145 -- directory is created if absent); an ungrouped def is written flat at
  146 -- @\<root\>\/\<id\>.md@. The git commit path is relative to the config repo
  147 -- root and encodes the group when present.
  148 writeAgentDef :: FilePath -> ConfigRepo -> AgentDef -> IO ()
  149 writeAgentDef dir repo d = do
  150   let path = defPath dir d
  151       tmp  = path <.> "tmp"
  152       parent = takeDirectory path
  153   createDirectoryIfMissing True parent
  154   TIO.writeFile tmp (encodeAgentDef d)
  155   setFileMode tmp 0o600
  156   renameFile tmp path
  157   let rel = "agents" </> case adGroup d of
  158         Just g | not (T.null (T.strip g)) ->
  159           T.unpack (T.strip g) </> (T.unpack (agentDefIdText (adId d)) <.> "md")
  160         _ -> T.unpack (agentDefIdText (adId d)) <.> "md"
  161   _ <- gitCommitAll repo rel ("seal: AGENT_DEF write " <> agentDefIdText (adId d))
  162   pure ()
  163 
  164 -- | Delete one def file and auto-commit. Idempotent (no-op if the file is
  165 -- absent). Searches both the grouped and flat layouts: a def written under
  166 -- a group is deleted from its group directory; a flat-layout def is deleted
  167 -- from the root. If a file exists in both, both are removed. Does NOT remove
  168 -- PureClaw-style subdirectories (a directory is a human-authored import
  169 -- path; the model does not delete those).
  170 deleteAgentDef :: FilePath -> ConfigRepo -> AgentDefId -> IO ()
  171 deleteAgentDef dir repo aid = do
  172   let base = T.unpack (agentDefIdText aid) <.> "md"
  173   groups <- listGroupSubdirs dir
  174   let groupedPaths = [ dir </> T.unpack g </> base | g <- groups ]
  175       flatPath     = defFile dir aid
  176   candidates <- filterM doesFileExist (groupedPaths ++ [flatPath])
  177   forM_ candidates $ \path -> do
  178     removeFile path
  179     let rel = "agents" </> fromMaybe base (stripRoot dir path)
  180     _ <- gitCommitAll repo rel ("seal: AGENT_DEF delete " <> agentDefIdText aid)
  181     pure ()
  182   where
  183     stripRoot r p =
  184       let r' = dropTrailingSlash r
  185           p' = dropTrailingSlash p
  186       in if r' `isPrefixPath` p'
  187            then Just (drop (length r' + 1) p')
  188            else Nothing
  189     dropTrailingSlash = reverse . dropWhile (== '/') . reverse
  190     isPrefixPath pre s = take (length pre) s == pre

  196 -- | The local 'WorkdirFs' anchored at the user store's agents directory.
  197 -- The DirScheme reads ('loadDirAgentDef' et al.) live in
  198 -- "Seal.Agent.Def.Workdir" and operate over 'WorkdirFs'; the user store
  199 -- constructs a local-arm 'WorkdirFs' at its @dir@ so the DirScheme reads
  200 -- share the single confined code path (§3.6) while remaining local-FS
  201 -- (§3.9).
  202 userDirFs :: FilePath -> WorkdirFs
  203 userDirFs dir = mkLocalWorkdirFs (WorkspaceRoot dir) maxScanBytes

  205 -- | Read one def by id. Probes, in order: the native grouped layout
  206 -- (@\<group\>\/\<id\>.md@) for each group subdirectory, then the flat
  207 -- layout (@\<id\>.md@). On the flat path, the DirScheme
  208 -- (@agents\/\<id\>\/@) is the fallback when the flat file is absent. The
  209 -- returned def's 'adGroup' is filled from the parent directory when the
  210 -- frontmatter omitted it (the common case), so a grouped file reads back
  211 -- as grouped regardless of whether its frontmatter redundantly declares a
  212 -- group. Returns 'Nothing' if no candidate exists. Mirrors
  213 -- 'Seal.Skills.Backend.readSkill'.
  214 readAgentDef :: FilePath -> AgentDefId -> IO (Maybe AgentDef)
  215 readAgentDef dir aid = do
  216   let flatPath = defFile dir aid
  217   groups <- listGroupSubdirs dir
  218   let base = T.unpack (agentDefIdText aid) <.> "md"
  219       groupedPaths = [ (g, dir </> T.unpack g </> base) | g <- groups ]
  220   mGrouped <- firstMatchGrouped groupedPaths
  221   case mGrouped of
  222     Just d  -> pure (Just d)
  223     Nothing -> do
  224       flatExists <- doesFileExist flatPath
  225       if flatExists
  226         then do
  227           -- Conflict check: warn if a directory also exists.
  228           let dirPath = defDir dir aid
  229           dirExists <- doesDirectoryExist dirPath
  230           when dirExists $
  231             putStrLn ("warning: agent def " <> T.unpack (agentDefIdText aid)
  232                       <> " has both a flat file and a directory; flat file takes precedence")
  233           content <- TIO.readFile flatPath
  234           pure (decodeAgentDef content)
  235         else loadDirAgentDef (userDirFs dir) aid
  236 
  237 -- | Enumerate all defs in the directory (flat + grouped + DirScheme),
  238 -- sorted by id. Malformed flat files and malformed dirs are skipped. On
  239 -- flat/dir collision for the same id, the flat file wins and the dir is
  240 -- dropped. Grouped @.md@ files one directory down are stamped with their
  241 -- group. A subdirectory is classified as a *group* when it contains at
  242 -- least one @.md@ file that decodes as an 'AgentDef' (has an @id@
  243 -- frontmatter); otherwise it is treated as a DirScheme
  244 -- (@agents\/\<id\>\/@ with bootstrap files). Mirrors
  245 -- 'Seal.Skills.Backend.listSkills'.
  246 listAgentDefs :: FilePath -> IO [AgentDef]
  247 listAgentDefs dir = do
  248   exists <- doesDirectoryExist dir
  249   if not exists
  250     then pure []
  251     else do
  252       entries <- listDirectory dir
  253       flatDefs <- collectFlat dir entries
  254       (groupDirs, dirSchemeDirs) <- classifySubdirs dir entries
  255       groupedDefs <- collectGrouped dir groupDirs
  256       let groupSet = Set.fromList groupDirs
  257       dirDefs  <- collectDirs dir dirSchemeDirs (map adId (flatDefs ++ groupedDefs)) groupSet
  258       pure (sortOn (agentDefIdText . adId) (flatDefs ++ groupedDefs ++ dirDefs))
  259 
  260 -- | Classify immediate subdirectories of @dir@ into group dirs (contain at
  261 -- least one decodable flat def @.md@) and DirScheme dirs (everything else
  262 -- with a valid agent-def id). Hidden dirs are skipped.
  263 classifySubdirs :: FilePath -> [FilePath] -> IO ([Text], [FilePath])
  264 classifySubdirs dir entries = do
  265   let visible = [e | e <- entries, not (T.isPrefixOf "." (T.pack e))]
  266   isDirBools <- mapM (\e -> doesDirectoryExist (dir </> e)) visible
  267   let dirNames = [e | (e, b) <- zip visible isDirBools, b]
  268   go dirNames ([], [])
  269   where
  270     go [] acc = pure acc
  271     go (e:es) (gs, ds) = do
  272       let gDir = dir </> e
  273       subEntries <- listDirectory gDir
  274       let mdFiles = [f | f <- subEntries, ".md" `T.isSuffixOf` T.pack f]
  275       decoded <- mapM (\f -> decodeAgentDef <$> TIO.readFile (gDir </> f)) mdFiles
  276       if any isJust decoded
  277         then go es (T.pack e : gs, ds)
  278         else go es (gs, e : ds)
  279 
  280 -- | Decode all flat @.md@ files in the directory (top-level only).
  281 collectFlat :: FilePath -> [FilePath] -> IO [AgentDef]
  282 collectFlat dir entries = do
  283   let mdFiles = [e | e <- entries, ".md" `T.isSuffixOf` T.pack e]
  284   defs <- forM mdFiles $ \e -> do
  285     content <- TIO.readFile (dir </> e)
  286     pure (decodeAgentDef content)
  287   pure (catMaybes defs)
  288 
  289 -- | Decode all @.md@ files one directory down (grouped layout), stamping
  290 -- 'adGroup' from the parent directory when the frontmatter omitted it.
  291 collectGrouped :: FilePath -> [Text] -> IO [AgentDef]
  292 collectGrouped dir groups = do
  293   results <- forM groups $ \g -> do
  294     let gDir = dir </> T.unpack g
  295     entries <- listDirectory gDir
  296     let mdFiles = [e | e <- entries, ".md" `T.isSuffixOf` T.pack e]
  297     forM mdFiles $ \e -> do
  298       content <- TIO.readFile (gDir </> e)
  299       pure (stampGroup g (decodeAgentDef content))
  300   pure (concatMap catMaybes results)
  301 
  302 -- | Load DirScheme defs from subdirectories, skipping ids already
  303 -- provided by the flat or grouped scheme (flat/grouped wins on conflict).
  304 -- Emits a warning per collision. @dirSchemeDirs@ is the pre-classified list
  305 -- of DirScheme candidate subdirectory names.
  306 collectDirs :: FilePath -> [FilePath] -> [AgentDefId] -> Set.Set Text -> IO [AgentDef]
  307 collectDirs dir dirSchemeDirs flatIds _groupSet = do
  308   let flatIdTexts = Set.fromList (map agentDefIdText flatIds)
  309   defs <- forM dirSchemeDirs $ \e -> do
  310     case mkAgentDefId (T.pack e) of
  311       Left _   -> pure Nothing
  312       Right aid -> do
  313         if agentDefIdText aid `Set.member` flatIdTexts
  314           then do
  315             putStrLn ("warning: agent def " <> e
  316                       <> " has both a flat file and a directory; flat file takes precedence")
  317             pure Nothing  -- skip the dir def; flat wins
  318           else loadDirAgentDef (userDirFs dir) aid
  319   pure (catMaybes defs)
  320 
  321 -- | Enumerate the immediate sub-directories of @dir@ (non-recursive, no
  322 -- hidden dirs), sorted for deterministic output. Returns the visible
  323 -- sub-directory names as 'Text'. Missing @dir@ yields @[]@. Mirrors
  324 -- 'Seal.Skills.Backend.listSubdirs' but on local 'FilePath'.
  325 listGroupSubdirs :: FilePath -> IO [Text]
  326 listGroupSubdirs dir = do
  327   exists <- doesDirectoryExist dir
  328   if not exists
  329     then pure []
  330     else do
  331       entries <- listDirectory dir
  332       let visible = [e | e <- entries, not (T.isPrefixOf "." (T.pack e))]
  333       ds <- filterM (\e -> doesDirectoryExist (dir </> e)) (sortOn id visible)
  334       pure (map T.pack ds)
  335 
  336 -- | Stamp 'adGroup' from the parent directory when the frontmatter omitted
  337 -- it. Mirrors 'Seal.Skills.Backend.readAndStampGroup' / 'stampGroup'.
  338 stampGroup :: Text -> Maybe AgentDef -> Maybe AgentDef
  339 stampGroup g (Just d) = case adGroup d of
  340   Just _  -> Just d
  341   Nothing -> Just d { adGroup = Just g }
  342 stampGroup _ Nothing = Nothing
  343 
  344 -- | Read the first grouped @.md@ file that decodes, stamping its group.
  345 -- Short-circuits on the first match.
  346 firstMatchGrouped :: [(Text, FilePath)] -> IO (Maybe AgentDef)
  347 firstMatchGrouped []       = pure Nothing
  348 firstMatchGrouped ((g, p):ps) = do
  349   exists <- doesFileExist p
  350   if not exists
  351     then firstMatchGrouped ps
  352     else do
  353       content <- TIO.readFile p
  354       case stampGroup g (decodeAgentDef content) of
  355         Just d  -> pure (Just d)
  356         Nothing -> firstMatchGrouped ps

  358 -- | A union of a workdir 'AgentDefBackend' (repo-local agent defs) and a
  359 -- user 'AgentDefBackend' (the on-disk @~/.seal/config/agents/@ store).
  360 -- Workdir-wins on id collisions: workdir shadows user. Reads check
  361 -- workdir first, then user. Listing merges both with workdir winning on
  362 -- collision. Writes go to the /user/ backend only (repo-local defs are
  363 -- immutable from the model's perspective).
  364 unionAgentDefBackend :: AgentDefBackend -> AgentDefBackend -> AgentDefBackend
  365 unionAgentDefBackend workdir user = AgentDefBackend
  366     { adbRead   = \aid -> do
  367         mWd <- adbRead workdir aid
  368         case mWd of
  369           Just d  -> pure (Just d)
  370           Nothing -> adbRead user aid
  371     , adbUpdate = adbUpdate user
  372     , adbList   = do
  373         wdDefs  <- adbList workdir
  374         userDefs <- adbList user
  375         let wdMap   = Map.fromList [(adId d, d) | d <- wdDefs]
  376             userMap = Map.fromList [(adId d, d) | d <- userDefs]
  377             merged  = Map.union wdMap userMap
  378         pure (Map.elems merged)
  379     , adbDelete = adbDelete user
  380     }

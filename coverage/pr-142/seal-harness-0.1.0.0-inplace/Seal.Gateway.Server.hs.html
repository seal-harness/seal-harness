    2 -- | The assembled Warp application: the REST API routes + static file
    3 -- serving with SPA fallback + the non-loopback bind warning. The WS stream
    4 -- server runs separately (on @gcWsPort@) via 'Seal.Gateway.Stream'.

   11 import Data.Text qualified as T

   15 import Network.Wai.Handler.Warp (run)

   23 -- | The assembled WAI application: the REST API routes, then static file
   24 -- serving with SPA fallback (if the static dir is configured).
   25 gatewayApp :: ApiDeps -> Maybe FilePath -> Application
   26 gatewayApp deps mStaticDir req respond =
   27   case pathInfo req of
   28     ("api" : _) -> apiApp deps req respond
   29     _ -> case mStaticDir of
   30       Nothing -> respond (responseLBS status404 [("Content-Type", "application/json")] "{\"error\":\"not found\"}")
   31       Just staticDir -> serveStatic staticDir req respond
   32 
   33 -- | Serve a static file with SPA fallback (serve @index.html@ if the path
   34 -- doesn't match a file).
   35 serveStatic :: FilePath -> Application
   36 serveStatic staticDir req respond = do
   37   let pathParts = map T.unpack (pathInfo req)
   38       filePath = staticDir </> case pathParts of
   39         [] -> "index.html"
   40         _ -> case xs of
   41           [p] -> p
   42           _ -> T.unpack (T.intercalate "/" (map T.pack xs))
   43       xs = pathParts
   44   exists <- doesFileExist filePath
   45   if exists
   46     then respond (responseFile status200 [("Content-Type", contentTypeFor (takeExtension filePath))] filePath Nothing)
   47     else respond (responseFile status200 [("Content-Type", "text/html")] (staticDir </> "index.html") Nothing)
   48 
   49 -- | The content-type for a file extension.
   50 contentTypeFor :: String -> BC.ByteString
   51 contentTypeFor ext = case ext of
   52   ".html" -> "text/html"
   53   ".js"   -> "application/javascript"
   54   ".css"  -> "text/css"
   55   ".json" -> "application/json"
   56   ".svg"  -> "image/svg+xml"
   57   ".png"  -> "image/png"
   58   ".ico"  -> "image/x-icon"
   59   _       -> "application/octet-stream"
   60 
   61 -- | Run the gateway. Prints a startup banner with the bind address + the
   62 -- served static dir. When the host is non-loopback:
   63 --   - If @failClosedOnNonLoopback@ is True (mode=remote), REFUSES to start
   64 --     (design V6: a non-loopback gateway is a config-tamper surface — the
   65 --     unauthenticated @updateRuntimeConfig@ caller is reachable by anything
   66 --     that can reach the address).
   67 --   - Otherwise, warns (the full slash-command surface is reachable).
   68 runGateway :: GatewayConfig -> Bool -> ApiDeps -> IO ()
   69 runGateway cfg failClosedOnNonLoopback deps = do
   70   let host = gcHost cfg
   71       port = gcPort cfg
   72   hPutStrLn stderr "Seal Harness gateway"
   73   hPutStrLn stderr ("  URL:      http://" <> T.unpack host <> ":" <> show port)
   74   if host /= "127.0.0.1" && host /= "::1" && host /= "localhost"
   75     then
   76       if failClosedOnNonLoopback
   77         then do
   78           hPutStrLn stderr ("ERROR: refusing to start gateway on non-loopback address " <> T.unpack host
   79                             <> " in remote-only mode — bind 127.0.0.1 or disable remote-only")
   80           pure ()  -- do NOT call run; exit without serving
   81         else do
   82           hPutStrLn stderr ("Warning: binding to " <> T.unpack host
   83                             <> " — the full slash-command surface is reachable by anything that can reach this address")
   84           let mStaticDir = fmap T.unpack (gcStaticDir cfg)
   85               app = gatewayApp deps mStaticDir
   86           run port app
   87     else do
   88       let mStaticDir = fmap T.unpack (gcStaticDir cfg)
   89           app = gatewayApp deps mStaticDir
   90       run port app

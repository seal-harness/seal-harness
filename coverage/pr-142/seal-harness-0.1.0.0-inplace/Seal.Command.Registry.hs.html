    1 -- | The unified slash-command registry builder. All channels (web, CLI,
    2 -- Signal, Telegram) share the same core set of @\/@-commands, defined once
    3 -- here. Only channel-specific specs (the terse @\/N@ tab-grammar, the
    4 -- @\/channel@ channel-management command, the CLI's @\/bg@, and the
    5 -- per-request @\/call@\/@\/skill@\/@\/stop@ rebuilds on the multi-session
    6 -- web) stay at their wiring sites.
    7 --
    8 -- The carve-out for the web: the @\/N@ terse grammar (@\/1@ focus, @\/1
    9 -- payload@ inject) is a tab-level operation the web frontend handles via
   10 -- the sidebar, not via the send endpoint — 'Seal.Gateway.Send.handleSend'
   11 -- short-circuits 'Focus'/'Inject'/'TabCommand' routes with a deferral
   12 -- message before they reach the registry. The @terseGrammarSpec@ is still
   13 -- included in the core list so @\/help@ displays the grammar; only its
   14 -- execution is web-rejected.
   15 module Seal.Command.Registry
   16   ( CoreCommandDeps (..)
   17   , coreCommandSpecs
   18   ) where

   36 -- | The shared dependencies for the core slash-command specs. Built once
   37 -- at startup by each wiring site (web 'Seal.Command.Serve', CLI
   38 -- 'Seal.Tui', Signal/Telegram 'Seal.Channels.Signal.Run' /
   39 -- 'Seal.Channels.Telegram.Run') and passed to 'coreCommandSpecs' to
   40 -- produce the channel-agnostic slice of the 'Registry'.
   41 --
   42 -- The web's multi-session @\/stop@ (which must target the request's
   43 -- session, not the process-global @srActive@) is handled by
   44 -- 'Seal.Gateway.Send.replaceCallSkillSpecs' rebuilding the stop spec
   45 -- per-request via 'Seal.Command.Stop.stopCommandSpecForSession'; the
   46 -- @srActive@-reading spec produced here is the fallback for the
   47 -- single-session channels.
   48 data CoreCommandDeps = CoreCommandDeps
   49   { ccdVault       :: VaultRuntime
   50     -- ^ For @\/vault@ (status/unlock/lock/rekey).
   51   , ccdProvider    :: ProviderRuntime
   52     -- ^ For @\/provider@ (list/switch) + @\/model@.
   53   , ccdSession     :: SessionRuntime
   54     -- ^ For @\/session@, @\/model@, and the single-session @\/stop@.
   55   , ccdAgentDefs   :: AgentDefBackend
   56     -- ^ For @\/agent@ (list/edit/bind).
   57   , ccdCfgPath     :: FilePath
   58     -- ^ The config.toml path (for @\/agent@'s file edits).
   59   , ccdPaths       :: SealPaths
   60     -- ^ For @\/tab@ (session-meta lookups).
   61   , ccdTabs        :: TabsHandle
   62     -- ^ For @\/tab@ (list/new/close/focus/resume/rename).
   63   , ccdTabCloseNotifier :: TabCloseNotifier
   64     -- ^ Invoked after @\/tab close@ so attached channels are notified.
   65   , ccdAbortReg    :: SessionAbortRegistry
   66     -- ^ For @\/stop@ (the single-session variant; the web rebuilds
   67     -- per-request).
   68   , ccdStopWriter  :: StopTranscriptWriter
   69     -- ^ Writes the stop message to the session's transcript + broadcasts
   70     -- it so the stop appears cross-channel. Built at the wiring site from
   71     -- 'SealPaths' + the broker.
   72   , ccdModelWriter :: ModelTranscriptWriter
   73     -- ^ Writes the /model use confirmation to the session's transcript +
   74     -- broadcasts it (twin of 'ccdStopWriter'). Built at the wiring site
   75     -- from 'SealPaths' + the broker.
   76   , ccdRepoReg     :: RepoRegistryHandle
   77     -- ^ For @\/repo@ (list/add/remove/info/test).
   78   , ccdRepoSeam    :: Maybe RepoTestSeam
   79     -- ^ The @git ls-remote@ + vault-list seam for @\/repo test@ / @\/repo
   80     -- info@. 'Nothing' on channels that don't wire the seam yet
   81     -- (Signal/Telegram/TUI standalone); @\/repo@ is omitted from the
   82     -- registry in that case.
   83   }
   84 
   85 -- | Build the core (channel-agnostic) slash-command specs. Each wiring
   86 -- site passes its 'CoreCommandDeps' + any channel-specific specs to
   87 -- 'mkRegistry':
   88 --
   89 -- @
   90 -- 'mkRegistry' ('coreCommandSpecs' coreDeps <> [channelSpecificSpec, ...])
   91 -- @
   92 --
   93 -- The web additionally rebuilds @\/call@, @\/skill@, and @\/stop@
   94 -- per-request via 'Seal.Gateway.Send.replaceCallSkillSpecs' so those
   95 -- commands target the request's explicit 'SessionId' (the web is
   96 -- multi-session; @srActive@ is not authoritative there).
   97 --
   98 -- @\/new@ is deliberately NOT in the core list: on inbox channels
   99 -- (Signal, Telegram) @\/new@ is handled at the loop level
  100 -- ('Seal.Channels.Loop.handleNewSession') because the conversation key +
  101 -- cursor aren't available to a registry 'CommandAction'. The CLI and web
  102 -- add @newCommandSpec@ to their registries directly (closing over their
  103 -- channel-specific @ndInsertTab@).
  104 coreCommandSpecs :: CoreCommandDeps -> [CommandSpec]
  105 coreCommandSpecs d =
  106   [ vaultCommandSpec (ccdVault d)
  107   , providerCommandSpec (ccdProvider d)
  108   , sessionCommandSpec (ccdSession d)
  109   , modelCommandSpec (ccdProvider d) (ccdSession d) (ccdModelWriter d)
  110   , agentCommandSpec (ccdAgentDefs d) (ccdCfgPath d)
  111   , tabCommandSpec (ccdPaths d) (ccdTabs d) (ccdTabCloseNotifier d)
  112   , stopCommandSpec (ccdAbortReg d) (ccdSession d) (ccdStopWriter d)
  113   , terseGrammarSpec
  114   ]
  115   <> [ repoCommandSpec (ccdRepoReg d) seam | Just seam <- [ccdRepoSeam d] ]

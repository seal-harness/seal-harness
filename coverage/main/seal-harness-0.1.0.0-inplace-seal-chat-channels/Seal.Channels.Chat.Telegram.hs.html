   80 -- | Chunk a message to the given character limit. Telegram's hard limit
   81 -- is 4096; we leave headroom.
   82 chunkMessage :: Int -> Text -> [Text]
   83 chunkMessage limit msg
   84   | T.length msg <= limit = [msg]
   85   | otherwise = go msg
   86   where
   87     go t
   88       | T.null t = []
   89       | T.length t <= limit = [t]
   90       | otherwise = T.take limit t : go (T.drop limit t)

   92 -- | A mock transport backed by a 'TQueue' of inbound messages and an
   93 -- 'IORef' of captured sends. For testing.
   94 mkMockTelegramChatTransport :: [ReceivedMessage] -> IO (TelegramChatTransport, IO [Text])
   95 mkMockTelegramChatTransport scripted = do
   96   q <- newTQueueIO
   97   mapM_ (atomically . writeTQueue q) scripted
   98   capRef <- newIORef []
   99   msgIdRef <- newIORef (0 :: Int)
  100   let transport = TelegramChatTransport
  101         { tctReceive = do
  102             m <- atomically (tryReadTQueue q)
  103             case m of
  104               Just msg -> pure (Right msg)
  105               Nothing -> pure (Left "inbox empty")
  106         , tctSend = \_c b -> modifyIORef' capRef (b :)
  107         , tctSendWithId = \_c _b -> do
  108             n <- readIORef msgIdRef
  109             writeIORef msgIdRef (n + 1)

  116         , tctEditReplyMarkup = \_c _mid -> pure ()
  117         , tctEditMessage = \_c _mid _content -> pure True
  118         , tctClose = pure ()
  119         }
  120       getCaptured = reverse <$> readIORef capRef

  131 -- | Spawn the real Telegram transport: long-polls @getUpdates@ and sends
  132 -- via @sendMessage@. The 'Manager' is the shared TLS-configured HTTP
  133 -- manager from the server. The token is the validated 'TelegramToken'
  134 -- text. Returns 'Left' with a diagnostic if the initial @getUpdates@
  135 -- call fails (the bot still works on retry — this is not fatal).
  136 mkRealTelegramChatTransport :: Text -> Network.HTTP.Client.Manager -> IO TelegramChatTransport
  137 mkRealTelegramChatTransport token mgr = do
  138   buffer <- newTQueueIO
  139   offsetRef <- newIORef (0 :: Int)
  140   pure TelegramChatTransport
  141     { tctReceive = fillAndReceive buffer offsetRef
  142     , tctSend = sendViaApi mgr token
  143     , tctSendWithId = tgSendWithIdViaApi mgr token
  144     , tctSendWithKeyboard = tgSendWithKeyboardViaApi mgr token
  145     , tctAnswerCallback = answerCallbackQueryViaApi mgr token
  146     , tctEditReplyMarkup = editReplyMarkupViaApi mgr token
  147     , tctEditMessage = tgEditMessageViaApi mgr token
  148     , tctClose = pure ()
  149     }
  150   where
  151     fillAndReceive buffer offsetRef = do
  152       m <- atomically (tryReadTQueue buffer)
  153       case m of
  154         Just msg -> pure (Right msg)
  155         Nothing -> do
  156           offset <- readIORef offsetRef
  157           eResult <- getUpdates mgr token offset
  158           case eResult of
  159             Left err -> pure (Left err)
  160             Right (updates, rawIds) ->
  161               if null updates
  162                 then
  163                   if null rawIds
  164                     then fillAndReceive buffer offsetRef
  165                     else do
  166                       let lastId = maximum rawIds
  167                       modifyIORef' offsetRef (const (lastId + 1))
  168                       fillAndReceive buffer offsetRef
  169                 else do
  170                   let lastId = maximum (map fst updates)
  171                   modifyIORef' offsetRef (const (lastId + 1))
  172                   mapM_ ((atomically . writeTQueue buffer) . snd) updates
  173                   fillAndReceive buffer offsetRef

  175 -- | The live Telegram chat channel state.
  176 data TelegramChatChannel = TelegramChatChannel
  177   { tccAllowList  :: AllowList UserId
  178   , tccChunkLimit :: Int
  179   , tccInbox      :: TQueue InboundMessage
  180   , tccTransport  :: TelegramChatTransport
  181   , tccLastChat   :: IORef (Maybe Text)
  182   , tccReaderAlive :: IORef Bool
  183   }

  185 instance ChatChannel TelegramChatChannel where
  186   ccReceive ch = do
  187     mMsg <- atomically (tryReadTQueue (tccInbox ch))
  188     case mMsg of
  189       Just msg -> pure (Just msg)
  190       Nothing -> do
  191         alive <- readIORef (tccReaderAlive ch)
  192         if alive
  193           then threadDelay 1000 >> ccReceive ch
  194           else pure Nothing
  195 
  196   ccSend ch t =
  197     mapM_ (sendRaw ch) (chunkMessage (tccChunkLimit ch) t)
  198 
  199   ccSendWithId ch t = do
  200     mChat <- readIORef (tccLastChat ch)
  201     case mChat of
  202       Nothing -> pure Nothing
  203       Just chatId -> do
  204         mId <- tctSendWithId (tccTransport ch) chatId t
  205         pure (ChatMessageId <$> mId)
  206 
  207   ccEditMessage ch (ChatMessageId msgId) content = do
  208     mChat <- readIORef (tccLastChat ch)
  209     case mChat of
  210       Nothing -> pure False
  211       Just chatId -> tctEditMessage (tccTransport ch) chatId msgId content
  212 
  213   ccLabel _ = "telegram"
  214 
  215   ccSendWithOptions ch body opts askIdPrefix = do
  216     mChat <- readIORef (tccLastChat ch)
  217     case mChat of
  218       Nothing -> pure Nothing
  219       Just chatId -> do
  220         let keyboard = buildKeyboard askIdPrefix opts
  221         mId <- tctSendWithKeyboard (tccTransport ch) chatId body keyboard
  222         pure (ChatMessageId <$> mId)
  223 
  224   ccAnswerCallback ch = tctAnswerCallback (tccTransport ch)
  225 
  226   ccEditReplyMarkup ch (ChatMessageId msgId) = do
  227     mChat <- readIORef (tccLastChat ch)
  228     case mChat of
  229       Nothing -> pure ()
  230       Just chatId -> tctEditReplyMarkup (tccTransport ch) chatId msgId
  231 
  232   ccLastChatId ch = readIORef (tccLastChat ch)

  234 -- | Send one chunk verbatim to the last chat.
  235 sendRaw :: TelegramChatChannel -> Text -> IO ()
  236 sendRaw ch t = do
  237   mChat <- readIORef (tccLastChat ch)
  238   case mChat of
  239     Nothing -> pure ()
  240     Just chatId -> tctSend (tccTransport ch) chatId t

  242 -- | Build the inline keyboard: one row per option (button label = the
  243 -- option's 'qoLabel', callback_data = @\"<prefix>:<idx>\"@). Pure. The
  244 -- callback_data is always ≤ 64 bytes (8 hex + 1 colon + ≤ 1 char = ≤ 10
  245 -- bytes). Mirrors the old 'Seal.Channels.Telegram.Run.buildKeyboard'.
  246 buildKeyboard :: Text -> [QuestionOption] -> [[TelegramButton]]
  247 buildKeyboard prefix opts =
  248   [ [TelegramButton (qoLabel o) (prefix <> ":" <> T.pack (show i))]
  249   | (i, o) <- zip [0 :: Int ..] opts
  250   ]

  252 -- | Run the reader thread with cleanup. Spawns a background thread that
  253 -- loops 'tctReceive', allow-lists the sender, and pushes 'InboundMessage's
  254 -- to 'tccInbox'. On transport close or exception, the thread exits.
  255 withTelegramChatChannel
  256   :: AllowList UserId
  257   -> Int
  258   -> TelegramChatTransport
  259   -> (TelegramChatChannel -> IO a)
  260   -> IO a
  261 withTelegramChatChannel allow chunkLimit transport action =
  262   bracket before after (action . snd)
  263   where
  264     before = do
  265       inbox <- newTQueueIO
  266       lastChat <- newIORef (Nothing :: Maybe Text)
  267       alive <- newIORef True
  268       let ch = TelegramChatChannel
  269             { tccAllowList = allow
  270             , tccChunkLimit = chunkLimit
  271             , tccInbox = inbox
  272             , tccTransport = transport
  273             , tccLastChat = lastChat
  274             , tccReaderAlive = alive
  275             }
  276       tid <- forkIO (readerLoop ch)
  277       pure (tid, ch)
  278     after (tid, _) = do
  279       killThread tid
  280       tctClose transport

  282 -- | The background reader: loop 'tctReceive', construct a 'MessageSource'
  283 -- from the received conversation id + sender, update the last chat id,
  284 -- and push to the inbox. Exits when 'tctReceive' returns 'Left'.
  285 readerLoop :: TelegramChatChannel -> IO ()
  286 readerLoop ch = go
  287   where
  288     go = do
  289       eVal <- try (tctReceive (tccTransport ch)) :: IO (Either SomeException (Either Text ReceivedMessage))
  290       case eVal of
  291         Left _ -> writeIORef (tccReaderAlive ch) False
  292         Right (Left _) -> writeIORef (tccReaderAlive ch) False
  293         Right (Right (ReceivedMessage cid mSender replyTo body mCbData _mCbId _mCbMsgId))
  294           | not (T.null body) -> do
  295               case mkMessageSource cid Telegram (mkSender <$> mSender) mempty of
  296                 Right ms -> do
  297                   writeIORef (tccLastChat ch) (Just replyTo)
  298                   atomically (writeTQueue (tccInbox ch) (InboundMessage ms body mCbData))
  299                 Left _ -> pure ()
  300               go
  301           | otherwise -> go

  313 -- | Call @getUpdates@ with long-polling (30s timeout). Returns the parsed
  314 -- updates as @(update_id, ReceivedMessage)@ pairs + all raw update ids
  315 -- (so the caller can advance the offset even when some updates fail to
  316 -- parse).
  317 getUpdates :: Manager -> Text -> Int -> IO (Either Text ([(Int, ReceivedMessage)], [Int]))
  318 getUpdates mgr token offset = do
  319   let url = T.unpack (telegramApiBase <> token <> "/getUpdates")
  320              <> "?offset=" <> show offset <> "&timeout=30&allowed_updates=%5B%22message%22%2C%22callback_query%22%5D"
  321   eReq <- try @SomeException (parseRequest url)
  322   case eReq of
  323     Left ex -> pure (Left ("getUpdates request error: " <> T.pack (show ex)))
  324     Right req0 -> do
  325       let req = req0 { responseTimeout = responseTimeoutMicro 60000000 }
  326       eResp <- try @SomeException (httpLbs req mgr)
  327       case eResp of
  328         Left ex -> pure (Left ("getUpdates network error: " <> T.pack (show ex)))
  329         Right resp ->
  330           let code = statusCode (responseStatus resp)
  331               body = responseBody resp
  332           in if code == 200
  333                then pure (parseGetUpdatesResponse body)
  334                else pure (Left ("getUpdates returned HTTP " <> T.pack (show code)
  335                               <> " — " <> T.pack (show body)))
  336 
  337 -- | Parse the JSON response from @getUpdates@.
  338 parseGetUpdatesResponse :: BL.ByteString -> Either Text ([(Int, ReceivedMessage)], [Int])
  339 parseGetUpdatesResponse body =
  340   case A.decode body of
  341     Nothing -> Left "getUpdates: malformed JSON response"
  342     Just (A.Object o) -> case KeyMap.lookup (Key.fromString "result") o of
  343       Just (A.Array arr) ->
  344         let raws = V.toList arr
  345             allIds = [ uid | v <- raws, Just uid <- [rawUpdateId v] ]
  346             parsed = [ parseOneUpdate v | v <- raws ]
  347             okUpdates = [ (uid, u) | Right (uid, u) <- parsed ]
  348         in Right (okUpdates, allIds)
  349       _ -> Right ([], [])
  350     Just _ -> Left "getUpdates: response not an object"
  351   where
  352     parseOneUpdate v = do
  353       uid <- updateId v
  354       case parseTelegramUpdate v of
  355         Left err -> Left err
  356         Right u  -> Right (uid, u)
  357 
  358 -- | Extract the raw @update_id@ from an update object.
  359 rawUpdateId :: Value -> Maybe Int
  360 rawUpdateId v =
  361   case v of
  362     A.Object o -> case KeyMap.lookup (Key.fromString "update_id") o of
  363       Just (A.Number n) -> Just (round n)
  364       _ -> Nothing
  365     _ -> Nothing
  366 
  367 -- | Extract the numeric @update_id@.
  368 updateId :: Value -> Either Text Int
  369 updateId v =
  370   case v of
  371     A.Object o -> case KeyMap.lookup (Key.fromString "update_id") o of
  372       Just (A.Number n) -> Right (round n)
  373       _ -> Left "update missing update_id"
  374     _ -> Left "update not an object"

  376 -- | Parse a Telegram update into a 'ReceivedMessage'. Handles @message@
  377 -- updates (text messages). Skips non-message updates with a 'Left'.
  378 parseTelegramUpdate :: Value -> Either Text ReceivedMessage
  379 parseTelegramUpdate v =
  380   case v of
  381     A.Object o ->
  382       case KeyMap.lookup (Key.fromString "callback_query") o of
  383         Just cq -> parseCallbackQuery cq
  384         Nothing -> case KeyMap.lookup (Key.fromString "message") o of
  385           Just m  -> parseMessage m
  386           Nothing -> Left "update has no message or callback_query field"
  387     _ -> Left "update not an object"
  388   where
  389     parseMessage msg =
  390       case msg of
  391         A.Object mo -> do
  392           chatId <- requireChatId mo
  393           cid <- case mkConversationId ("tg:" <> chatId) of
  394             Right c -> Right c
  395             Left err -> Left ("conversation id construction failed: " <> err)
  396           let mSender = extractSenderId mo
  397               body = extractText mo
  398           Right ReceivedMessage
  399             { rmConversationId = cid
  400             , rmSender = mSender
  401             , rmReplyTo = chatId
  402             , rmBody = body
  403             , rmCallbackData = Nothing
  404             , rmCallbackId = Nothing
  405             , rmCallbackMessageId = Nothing
  406             }
  407         _ -> Left "message not an object"
  408     parseCallbackQuery cq =
  409       case cq of
  410         A.Object cqo -> do
  411           callbackId <- case KeyMap.lookup (Key.fromString "id") cqo of
  412             Just (A.String t) -> Right t
  413             Just (A.Number n) -> Right (T.pack (show (round n :: Int)))
  414             _ -> Left "callback_query.id missing"
  415           callbackData <- case KeyMap.lookup (Key.fromString "data") cqo of
  416             Just (A.String t) -> Right t
  417             _ -> Left "callback_query.data missing"
  418           msg <- case KeyMap.lookup (Key.fromString "message") cqo of
  419             Just m -> Right m
  420             Nothing -> Left "callback_query has no message field"
  421           case msg of
  422             A.Object mo -> do
  423               chatId <- requireChatId mo
  424               cid <- case mkConversationId ("tg:" <> chatId) of
  425                 Right c -> Right c
  426                 Left err -> Left ("conversation id construction failed: " <> err)
  427               let mSender = extractSenderId cqo
  428                   mMsgId = case KeyMap.lookup (Key.fromString "message_id") mo of
  429                     Just (A.Number n) -> Just (T.pack (show (round n :: Int)))
  430                     _ -> Nothing
  431               Right ReceivedMessage
  432                 { rmConversationId = cid
  433                 , rmSender = mSender
  434                 , rmReplyTo = chatId
  435                 , rmBody = callbackData
  436                 , rmCallbackData = Just callbackData
  437                 , rmCallbackId = Just callbackId
  438                 , rmCallbackMessageId = mMsgId
  439                 }
  440             _ -> Left "callback_query.message not an object"
  441         _ -> Left "callback_query not an object"
  442 
  443 -- | Extract @chat.id@ from a message object.
  444 requireChatId :: A.Object -> Either Text Text
  445 requireChatId mo =
  446   case KeyMap.lookup (Key.fromString "chat") mo of
  447     Just (A.Object co) -> case KeyMap.lookup (Key.fromString "id") co of
  448       Just (A.Number n) -> Right (T.pack (show (round n :: Int)))
  449       _ -> Left "chat.id missing"
  450     _ -> Left "chat field missing"

  455   case KeyMap.lookup (Key.fromString "from") mo of
  456     Just (A.Object fo) -> case KeyMap.lookup (Key.fromString "id") fo of
  457       Just (A.Number n) -> Just (T.pack (show (round n :: Int)))
  458       _ -> Nothing

  461 -- | Extract @text@ from a message object.

  468 -- | Send a message via the Bot API @sendMessage@.
  469 sendViaApi :: Manager -> Text -> Text -> Text -> IO ()
  470 sendViaApi mgr token chatId body = do
  471   eReq <- try @SomeException
  472     (parseRequest (T.unpack (telegramApiBase <> token <> "/sendMessage")))
  473   case eReq of
  474     Left _ -> pure ()
  475     Right req0 -> do
  476       let payload = A.object [ "chat_id" .= chatId, "text" .= body ]
  477           req = req0 { method = methodPost
  478                      , requestBody = RequestBodyLBS (A.encode payload)
  479                      , requestHeaders = [("Content-Type", "application/json")]
  480                      }
  481       _ <- try @SomeException (httpLbs req mgr)
  482       pure ()

  488     (parseRequest (T.unpack (telegramApiBase <> token <> "/sendMessage")))
  489   case eReq of
  490     Left _ -> pure Nothing
  491     Right req0 -> do
  492       let payload = A.object [ "chat_id" .= chatId, "text" .= body ]
  493           req = req0 { method = methodPost
  494                      , requestBody = RequestBodyLBS (A.encode payload)

  497       eResp <- try @SomeException (httpLbs req mgr)

  505 -- | Edit a previously sent message via @editMessageText@.
  506 tgEditMessageViaApi :: Manager -> Text -> Text -> Text -> Text -> IO Bool
  507 tgEditMessageViaApi mgr token chatId messageId content = do
  508   eReq <- try @SomeException
  509     (parseRequest (T.unpack (telegramApiBase <> token <> "/editMessageText")))
  510   case eReq of
  511     Left _ -> pure False
  512     Right req0 -> do
  513       let payload = A.object
  514             [ "chat_id" .= chatId
  515             , "message_id" .= messageId
  516             , "text" .= content
  517             ]
  518           req = req0 { method = methodPost
  519                      , requestBody = RequestBodyLBS (A.encode payload)
  520                      , requestHeaders = [("Content-Type", "application/json")]
  521                      }
  522       eResp <- try @SomeException (httpLbs req mgr)
  523       case eResp of
  524         Left _ -> pure False
  525         Right resp ->
  526           if statusCode (responseStatus resp) == 200
  527             then pure (parseOk (responseBody resp))
  528             else
  529               let errText = TE.decodeUtf8 (BL.toStrict (responseBody resp))
  530               in if "not modified" `T.isInfixOf` errText
  531                    then pure True
  532                    else pure False

  558 -- | Send a message with an inline keyboard via @sendMessage@. The payload
  559 -- includes @reply_markup: {inline_keyboard: ...}@. MUST NOT set
  560 -- @parse_mode@ (plain text — no markdown/HTML interpretation). Returns the

  562 tgSendWithKeyboardViaApi :: Manager -> Text -> Text -> Text -> [[TelegramButton]] -> IO (Maybe Text)

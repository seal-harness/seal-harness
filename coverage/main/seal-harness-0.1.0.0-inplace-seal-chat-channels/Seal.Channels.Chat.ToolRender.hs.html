    2 -- | Tool-call rendering helpers for chat channels. Ports the per-opcode
    3 -- emoji mapping and 'formatToolLine' from
    4 -- 'Seal.Channels.StreamProgress' so the chat-channel package can render
    5 -- tool-progress lines with the same emoji prefixes the old channel
    6 -- implementation used, without depending on server internals.
    7 module Seal.Channels.Chat.ToolRender
    8   ( opEmoji
    9   , formatToolLine
   10   ) where
   11 
   12 import Data.Set (Set)
   13 import Data.Set qualified as Set
   14 import Data.Text (Text)
   15 import Data.Text qualified as T

   17 -- | A local opaque type standing in for 'OpName' — the chat-channel
   18 -- package does not depend on 'Seal.Core.Types', so we use 'Text' wrapped
   19 -- in a newtype to keep the signature self-consistent. The loop passes
   20 -- the raw opcode name text.
   21 newtype OpName = OpName Text
   22   deriving stock (Eq, Ord, Show)
   23 
   24 -- | Format a tool-call progress line: the per-opcode emoji, the opcode
   25 -- name, and a truncated input. For secret-bearing opcodes (in the
   26 -- 'secretOps' set), the input is replaced with @"<redacted>"@. Pure.
   27 formatToolLine :: Set OpName -> Text -> Text -> Text
   28 formatToolLine secretOps name input =
   29   opEmoji name <> " " <> name <> " " <> inputDisplay
   30   where
   31     inputDisplay
   32       | OpName name `Set.member` secretOps = "<redacted>"
   33       | otherwise = truncateInput input
   34     truncateInput t =
   35       if T.length t > 120
   36         then T.take 120 t <> "..."
   37         else t
   38

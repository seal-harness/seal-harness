    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The Telegram bot command menu, built from the gateway's slash-command
    3 -- registry.
    4 --
    5 -- The menu is what the Telegram client shows after the user types @\/@. It
    6 -- is registered with @setMyCommands@ at startup (see
    7 -- 'Seal.Command.Serve.forkNewTelegramChatChannel'), not derived by the
    8 -- platform — Telegram remembers the last registration server-side, so a
    9 -- stale menu can look right while nothing re-registers it. That is why the
   10 -- request body is what the tests pin.
   11 --
   12 -- Everything here is pure except 'registerWithRetry'; the HTTP call lives in
   13 -- 'Seal.Channels.Chat.Telegram.tgSetCommandsViaApi'.
   14 module Seal.Channels.Chat.TelegramCommands
   15   ( BotCommand (..)
   16   , sanitizeTelegramName
   17   , telegramMenuCommands
   18   , commandsPayload
   19   , registerWithRetry
   20     -- * Limits (Telegram's, as constants)
   21   , maxMenuCommands
   22   , maxCommandNameLength
   23   , maxDescriptionLength
   24   ) where
   25

   38 -- | One @setMyCommands@ entry: the command name (no leading slash) and the
   39 -- one-line description shown beside it.
   40 data BotCommand = BotCommand
   41   { bcName        :: !Text
   42   , bcDescription :: !Text
   43   } deriving stock (Eq, Show)
   44 
   45 -- | Telegram's cap on the number of menu entries.
   46 maxMenuCommands :: Int
   47 maxMenuCommands = 100
   48 
   49 -- | Telegram's cap on a command name's length, in characters.
   50 maxCommandNameLength :: Int
   51 maxCommandNameLength = 32
   52 
   53 -- | Telegram's cap on a command description's length, in characters.
   54 maxDescriptionLength :: Int
   55 maxDescriptionLength = 256
   56 
   57 -- | Reduce an arbitrary command name to something Telegram accepts:
   58 -- lowercase, @-@ becomes @_@, anything outside @[a-z0-9_]@ is dropped, runs of
   59 -- underscores are collapsed, leading\/trailing underscores are trimmed, and
   60 -- the result is clamped to 'maxCommandNameLength'.
   61 --
   62 -- A name that reduces to nothing (e.g. @\"!!!\"@) yields @\"\"@; the caller
   63 -- drops it.
   64 sanitizeTelegramName :: Text -> Text
   65 sanitizeTelegramName =
   66     T.take maxCommandNameLength
   67       . trimUnderscores
   68       . collapseUnderscores
   69       . T.filter keep
   70       . T.map dashToUnderscore
   71       . T.toCaseFold
   72   where
   73     dashToUnderscore '-' = '_'
   74     dashToUnderscore c   = c
   75     keep c = isAsciiLower c || isDigit c || c == '_'
   76 
   77     collapseUnderscores = T.pack . go . T.unpack
   78       where
   79         go (a : b : rest)
   80           | a == '_' && b == '_' = go (b : rest)
   81         go (c : rest) = c : go rest
   82         go []         = []
   83 
   84     trimUnderscores = T.dropWhile (== '_') . T.dropWhileEnd (== '_')
   85 
   86 -- | Build the menu: @help@, then the registry (in registry order), then
   87 -- @bg@ and @watch@.
   88 --
   89 -- Registry names are sanitised ('sanitizeTelegramName'); entries that
   90 -- sanitise to nothing, or to a single character, are dropped. A one-character
   91 -- entry would shadow the terse @\/0@–@\/z@ tab grammar (the @\/N@ focus\/inject
   92 -- commands), so it must never be offered in the menu. Duplicates are dropped
   93 -- first-wins, and the list is capped at 'maxMenuCommands' by truncating the
   94 -- tail.
   95 --
   96 -- @telegramMenuCommands []@ is @[help, bg, watch]@: the built-in entries are
   97 -- part of the menu, never appended by 'commandsPayload'.
   98 telegramMenuCommands :: [CommandJson] -> [BotCommand]
   99 telegramMenuCommands registry =
  100     take maxMenuCommands (dedupe candidates)
  101   where
  102     candidates =
  103       [ BotCommand "help" "Show available commands" ]
  104         <> fromRegistry
  105         <> [ BotCommand "bg" "Run a prompt in a background session"
  106            , BotCommand "watch" "Toggle tab notifications" ]
  107 
  108     fromRegistry =
  109       [ BotCommand (sanitizeTelegramName (cjName cj))
  110                    (T.take maxDescriptionLength (cjSynopsis cj))
  111       | cj <- registry
  112       ]
  113 
  114     -- Drop the empty and the one-character names (see the doc comment).
  115     keepCommand b =
  116       let n = bcName b in not (T.null n) && T.length n > 1
  117 
  118     dedupe = reverse . snd . foldl' step (Set.empty, [])
  119 
  120     step (seen, acc) b
  121       | not (keepCommand b)        = (seen, acc)
  122       | Set.member (bcName b) seen = (seen, acc)
  123       | otherwise = (Set.insert (bcName b) seen, b : acc)
  124 
  125 -- | The @setMyCommands@ request body. Pure and exported so the wire shape is
  126 -- pinned by a test (the built-in entries are added by
  127 -- 'telegramMenuCommands', not here).
  128 commandsPayload :: [BotCommand] -> Value
  129 commandsPayload cmds = A.object
  130   [ "commands" .= [ A.object
  131       [ "command"     .= bcName c
  132       , "description" .= bcDescription c
  133       ] | c <- cmds ]
  134   ]

  136 -- | Fetch the registry and register the menu, retrying a bounded number of
  137 -- times with a fixed delay between attempts. Returns the last error when
  138 -- every attempt failed.
  139 --
  140 -- The retry exists for startup: the channel listeners are forked before the
  141 -- gateway binds its HTTP port, so an immediate fetch always races the bind.
  142 registerWithRetry
  143   :: Int
  144      -- ^ attempts (clamped to at least 1)
  145   -> Int
  146      -- ^ delay between attempts, in microseconds
  147   -> IO (Either Text [CommandJson])
  148      -- ^ fetch the registry (@GET \/api\/commands@)
  149   -> ([BotCommand] -> IO (Either Text ()))
  150      -- ^ register the menu (@setMyCommands@)
  151   -> IO (Either Text ())
  152 registerWithRetry attempts delayUs fetch register = go (max 1 attempts)
  153   where
  154     go :: Int -> IO (Either Text ())
  155     go remaining = do
  156       eCmds <- fetch
  157       eResult <- case eCmds of
  158         Left e   -> pure (Left e)
  159         Right cs -> register (telegramMenuCommands cs)
  160       case eResult of
  161         Right () -> pure (Right ())
  162         Left e
  163           | remaining <= 1 -> pure (Left e)
  164           | otherwise -> do
  165               when (delayUs > 0) (threadDelay delayUs)
  166               go (remaining - 1)

    2 -- | The pure rendering core for the single-edited-bubble progress style
    3 -- (@PsSingleBubble@), plus the ask renderer and the word-aware message
    4 -- chunker. No IO: everything here is directly testable.
    5 --
    6 -- == Why offset-splice
    7 --
    8 -- The gateway's streaming accumulator is /turn-scoped/: 'webAskCaps' creates
    9 -- its accumulator once per turn and appends every delta
   10 -- ("Seal.Gateway.Send"), so each @entry-update@ frame carries the WHOLE turn's
   11 -- text so far — not just the current response segment. That makes the frame
   12 -- text monotone within a turn, and it means any design that both commits
   13 -- per-response text into blocks and renders frames will emit text twice.
   14 --
   15 -- 'bubbleText' therefore treats the frame text as the sole body and remembers
   16 -- each tool line with the absolute text offset at which it fired. Rendering
   17 -- splices tool lines into that body at their offsets, which keeps the body
   18 -- monotone (no duplication) while preserving the order in which tools actually
   19 -- ran — rather than dumping every tool line below all of the text.
   20 module Seal.Channels.Chat.Bubble
   21   ( Splice
   22   , bubbleText
   23   , formatAsk
   24   , chunkMessage
   25   , chunkLen
   26   ) where
   27

   38 -- | A tool-progress line together with the length of the streamed text at the
   39 -- moment it fired. The offset is absolute within the turn's cumulative text,
   40 -- so it stays valid as later frames extend that text.
   41 type Splice = (Int, Text)
   42 
   43 -- | Render the bubble body from the turn's cumulative text, the tool lines
   44 -- spliced into it, and the trailing cursor (empty for a final edit).
   45 --
   46 -- Offsets are clamped to the text length: a tool-call activity can arrive
   47 -- before the frame that grew the text past it. The body always begins with the
   48 -- text received before the first tool line, so no part of the turn's text is
   49 -- reordered or repeated. Empty tool lines are dropped rather than rendered as a
   50 -- blank block. Pure.
   51 bubbleText :: [Splice] -> Text -> Text -> Text
   52 bubbleText splices text cursor =
   53   T.intercalate "\n\n" (filter (not . T.null) blocks) <> cursor
   54   where
   55     blocks = go 0 (sortOn fst [ s | s@(_, line) <- splices, not (T.null line) ])
   56 
   57     -- Walk the splices in offset order. @consumed@ is how much of @text@ has
   58     -- already been emitted, so offsets stay absolute as the walk advances.
   59     go consumed [] = [T.drop consumed text]
   60     go consumed ((rawOffset, line) : more) =
   61       let offset = max consumed (min rawOffset (T.length text))
   62           before = T.take (offset - consumed) (T.drop consumed text)
   63       in before : line : go offset more
   64 
   65 -- | Render an @ASK_HUMAN@ question with a hint telling the human how to reply.
   66 -- An open-ended question gets a bare reply hint; one with options gets a
   67 -- numbered list plus the numeric hint (the form a chat channel without inline
   68 -- keyboards can answer). Pure.
   69 formatAsk :: Text -> [QuestionOption] -> Text
   70 formatAsk question [] = question <> "\n\nReply with your answer."
   71 formatAsk question opts =
   72   question
   73   <> "\n\n"
   74   <> T.intercalate "\n" (zipWith formatOption [1 :: Int ..] opts)
   75   <> "\n\nReply with a number or type your own answer."
   76   where
   77     formatOption n (QuestionOption lbl desc)
   78       | T.null desc = T.pack (show n) <> ") " <> lbl
   79       | otherwise   = T.pack (show n) <> ") " <> lbl <> " \8212 " <> desc

   85 -- | How many @unit@s @t@ occupies.
   86 chunkLen :: ChunkUnit -> Text -> Int
   87 chunkLen ChunkUtf8Bytes  = BS.length . TE.encodeUtf8
   88 chunkLen ChunkUtf16Units = T.foldl' (\n c -> n + codeUnits c) 0
   89   where
   90     -- A code point above the BMP (an emoji, many CJK extensions) needs a
   91     -- surrogate pair, i.e. two UTF-16 code units.
   92     codeUnits c = if fromEnum c > 0xFFFF then 2 else 1
   93 
   94 -- | Split a message into chunks of at most @limit@ @unit@s, breaking at the
   95 -- most natural boundary that fits.
   96 --
   97 -- Platforms inline only the first message and demote the rest to an
   98 -- attachment (or split it themselves at an arbitrary point), so where the
   99 -- split falls is user-visible. Breaking mid-sentence reads as a glitch;
  100 -- breaking at a paragraph break reads as two messages that were always meant
  101 -- to be two. Candidate boundaries, best first:
  102 --
  103 -- 1. paragraph break (@\n\n@)
  104 -- 2. line break (@\n@)
  105 -- 3. sentence end (@.@ @?@ @!@ followed by whitespace, not an abbreviation)
  106 -- 4. clause break (@;@ @:@ @,@ followed by a space)
  107 -- 5. word break (a space)
  108 -- 6. hard split — only when a single token is longer than the whole budget
  109 --
  110 -- Within a class the LATEST boundary that fits wins, so chunks are as full as
  111 -- the class allows rather than as short as possible.
  112 --
  113 -- The separator stays at the END of the chunk it precedes — it is neither
  114 -- dropped nor moved to the front of the next chunk. Keeping it there is what
  115 -- stops a blank line from becoming its own message: the "\n\n" between two
  116 -- paragraphs is consumed by the chunk before it, so the next chunk begins with
  117 -- the following paragraph's text.
  118 --
  119 -- Invariant: @T.concat (chunkMessage u l t)@ equals @t@ modulo TRAILING
  120 -- whitespace. Every non-space character survives, in order, with no separator
  121 -- lost from the middle; only a final all-whitespace remainder is discarded
  122 -- (see 'go').
  123 --
  124 -- Pure. A non-positive @limit@ is treated as "cannot chunk" and returns the
  125 -- text unchanged rather than looping.
  126 chunkMessage :: ChunkUnit -> Int -> Text -> [Text]
  127 chunkMessage unit limit msg
  128   | limit <= 0 = [msg]
  129   | chunkLen unit msg <= limit = [msg]
  130   | otherwise = go msg
  131   where
  132     go t
  133       | T.null t = []
  134       -- A remainder holding only separators is dropped rather than sent as a
  135       -- message of its own: a reply ending in a newline would otherwise
  136       -- deliver a final blank message, which reads as a glitch. Nothing is
  137       -- lost -- it holds no content, since every non-space character has
  138       -- already been emitted. The invariant preserved is "no content lost",
  139       -- not "every byte preserved".
  140       --
  141       -- Folding it into the previous chunk instead is not an option: the
  142       -- remainder is whatever did not fit, so appending it can push that chunk
  143       -- past the platform limit -- exactly what chunking exists to prevent.
  144       -- Only a TRAILING all-space remainder is dropped; the
  145       -- separator-before-content rule in 'naturalCut' already prevents an
  146       -- INTERIOR blank chunk.
  147       | T.null (T.strip t) = []
  148       | chunkLen unit t <= limit = [t]
  149       | otherwise =
  150           let (chunk, rest) = splitOne t
  151           in chunk : go rest
  152 
  153     -- Cut @t@ into (chunk, remainder). The chunk is never empty, so 'go'
  154     -- always makes progress.
  155     splitOne t =
  156       let fitting = takeFitting unit limit t
  157           cut
  158             | T.null fitting = 1   -- one character exceeds the whole budget
  159             | otherwise      = naturalCut fitting
  160       in (T.take cut t, T.drop cut t)
  161 
  162 -- | The longest prefix of @t@ whose unit count fits @limit@, counting whole
  163 -- characters only. A byte-level cut could split a codepoint and produce
  164 -- invalid UTF-8, which the platform would reject or mangle.
  165 takeFitting :: ChunkUnit -> Int -> Text -> Text
  166 takeFitting unit limit t = T.take (fittingChars 0 0) t
  167   where
  168     fittingChars n used =
  169       case T.uncons (T.drop n t) of
  170         Nothing -> n
  171         Just (c, _) ->
  172           let w = chunkLen unit (T.singleton c)
  173           in if used + w <= limit then fittingChars (n + 1) (used + w) else n
  174 
  175 -- | Choose how many characters of @fitting@ to keep, preferring the most
  176 -- natural boundary. The latest boundary in the highest-priority class that
  177 -- has one wins; with no separator at all, the whole fitted prefix is used.
  178 naturalCut :: Text -> Int
  179 naturalCut fitting =
  180   case catMaybes [paragraphCut, lineCut, sentenceCut, clauseCut, wordCut] of
  181     (n : _) -> n
  182     []      -> T.length fitting
  183   where
  184     -- A candidate cut is usable only if the chunk it produces holds some
  185     -- actual content. Without this, a separator at the very start of the
  186     -- remaining text (the "\n\n" between two paragraphs, which is where the
  187     -- previous chunk's trim left off) becomes the only candidate and we emit a
  188     -- message consisting of nothing but a blank line.
  189     usable n = T.any (not . isSpaceChar) (T.take n fitting)
  190 
  191     -- The latest position just past a run of separators, if any is > 0.
  192     latestAfter pat = case [ i + T.length pat
  193                            | i <- [0 .. T.length fitting - T.length pat]
  194                            , pat `T.isPrefixOf` T.drop i fitting
  195                            , i + T.length pat > 0
  196                            , usable (i + T.length pat) ] of
  197                         [] -> Nothing
  198                         ns -> Just (maximum ns)
  199     paragraphCut = latestAfter "\n\n"
  200     lineCut      = latestAfter "\n"
  201     wordCut      = latestAfter " "
  202     clauseCut    = latestBefore ' '
  203     sentenceCut  = latestSentence
  204 
  205     -- Break after a clause separator that is followed by a space.
  206     latestBefore ch = case [ i + 1
  207                            | i <- [1 .. T.length fitting - 1]
  208                            , T.index fitting i == ch
  209                            , T.index fitting (i - 1) `elem` (";:," :: String)
  210                            , usable (i + 1) ] of
  211                         [] -> Nothing
  212                         ns -> Just (maximum ns)
  213 
  214     -- Break after sentence-ending punctuation followed by whitespace.
  215     latestSentence = case [ i + 1
  216                           | i <- [0 .. T.length fitting - 1]
  217                           , T.index fitting i `elem` (".?!" :: String)
  218                           , endsSentence fitting i
  219                           , usable (i + 1) ] of
  220                        [] -> Nothing
  221                        ns -> Just (maximum ns)
  222 
  223 -- | Does the terminator at offset @i@ plausibly end a sentence? Requires the
  224 -- next character to be whitespace or absent, and rejects abbreviations and
  225 -- single-letter initials.
  226 --
  227 -- Deliberately conservative: a MISSED sentence break falls through to a
  228 -- clause or word break (a cosmetic loss), whereas a WRONG one — splitting
  229 -- "Dr. Smith" after "Dr." — is visible. Note "e.g." is caught by the
  230 -- single-letter-initial rule only if the lookbehind is applied per token, so
  231 -- the abbreviation list carries the common dotted forms explicitly.
  232 endsSentence :: Text -> Int -> Bool
  233 endsSentence fitting i =
  234   let after = T.drop (i + 1) fitting
  235       nextOk = case T.uncons after of
  236         Nothing     -> True
  237         Just (c, _) -> c == ' ' || c == '\n' || c == '\t'
  238       -- The word the terminator is attached to. Taken back to the previous
  239       -- whitespace, so "e.g." yields "e.g" here and the abbreviation list
  240       -- matches on the token WITHOUT its final period.
  241       prevToken = T.dropWhileEnd (== '.') (T.takeWhileEnd (/= ' ') (T.take i fitting))
  242       lowerTok  = T.toLower prevToken
  243       isAbbrev  = lowerTok `elem`
  244         [ "dr", "mr", "mrs", "ms", "prof", "sr", "jr", "st", "vs", "etc"
  245         , "e.g", "i.e", "fig", "no", "vol", "al", "cf", "approx", "inc", "ltd" ]
  246       isInitial = T.length prevToken == 1 && T.all isAsciiUpper prevToken
  247   in nextOk && not isAbbrev && not isInitial

  250 -- | Whitespace, spelled locally so this module needs no @Data.Char@ import for

  252 isSpaceChar :: Char -> Bool
  253 isSpaceChar c = c == ' ' || c == '\n' || c == '\t' || c == '\r'

    2 -- | The pure rendering core for the single-edited-bubble progress style
    3 -- (@PsSingleBubble@), plus the ask renderer and the word-aware message
    4 -- chunker. No IO: everything here is directly testable.
    5 --
    6 -- == Why offset-splice
    7 --
    8 -- The gateway's streaming accumulator is /turn-scoped/: 'webAskCaps' creates
    9 -- its accumulator once per turn and appends every delta
   10 -- ("Seal.Gateway.Send"), so each @entry-update@ frame carries the WHOLE turn's
   11 -- text so far — not just the current response segment. That makes the frame
   12 -- text monotone within a turn, and it means any design that both commits
   13 -- per-response text into blocks and renders frames will emit text twice.
   14 --
   15 -- 'bubbleText' therefore treats the frame text as the sole body and remembers
   16 -- each tool line with the absolute text offset at which it fired. Rendering
   17 -- splices tool lines into that body at their offsets, which keeps the body
   18 -- monotone (no duplication) while preserving the order in which tools actually
   19 -- ran — rather than dumping every tool line below all of the text.
   20 module Seal.Channels.Chat.Bubble
   21   ( Splice
   22   , bubbleText
   23   , formatAsk
   24   , formatQuestionBody
   25   , liveAskHint
   26   , chunkMessage
   27   , chunkLen
   28   ) where

   40 -- | A tool-progress line together with the length of the streamed text at the
   41 -- moment it fired. The offset is absolute within the turn's cumulative text,
   42 -- so it stays valid as later frames extend that text.
   43 type Splice = (Int, Text)
   44 
   45 -- | Render the bubble body from the turn's cumulative text, the tool lines
   46 -- spliced into it, and the trailing cursor (empty for a final edit).
   47 --
   48 -- Offsets are clamped to the text length: a tool-call activity can arrive
   49 -- before the frame that grew the text past it. The body always begins with the
   50 -- text received before the first tool line, so no part of the turn's text is
   51 -- reordered or repeated. Empty tool lines are dropped rather than rendered as a
   52 -- blank block. Pure.
   53 bubbleText :: [Splice] -> Text -> Text -> Text
   54 bubbleText splices text cursor =
   55   T.intercalate "\n\n" (filter (not . T.null) blocks) <> cursor
   56   where
   57     blocks = go 0 (sortOn fst [ s | s@(_, line) <- splices, not (T.null line) ])
   58 
   59     -- Walk the splices in offset order. @consumed@ is how much of @text@ has
   60     -- already been emitted, so offsets stay absolute as the walk advances.
   61     go consumed [] = [T.drop consumed text]
   62     go consumed ((rawOffset, line) : more) =
   63       let offset = max consumed (min rawOffset (T.length text))
   64           before = T.take (offset - consumed) (T.drop consumed text)
   65       in before : line : go offset more

   67 -- | Render an @ASK_HUMAN@ question with a hint telling the human how to reply.
   68 -- An open-ended question gets a bare reply hint; one with options gets a
   69 -- numbered list plus the numeric hint (the form a chat channel without inline
   70 -- keyboards can answer). Pure.
   71 formatAsk :: Text -> [QuestionOption] -> Text
   72 formatAsk question opts = formatQuestionBody question opts <> liveAskHint opts
   73 
   74 -- | The question and its numbered options, with **no** reply hint.
   75 --
   76 -- Split out from 'formatAsk' because the hint is only correct when a plain
   77 -- reply actually reaches the question. On the @\\/last@ path it does not (a
   78 -- plain reply goes to the focused tab, not the tab being viewed), so @\\/last@
   79 -- supplies its own answer hint. Emitting "Reply with a number" there would
   80 -- tell the user to do the one thing that misroutes their answer.
   81 --
   82 -- Joining 'formatQuestionBody' with 'liveAskHint' reproduces 'formatAsk'
   83 -- exactly, for both the open-ended and the optioned shape. Pure.
   84 formatQuestionBody :: Text -> [QuestionOption] -> Text
   85 formatQuestionBody question [] = question
   86 formatQuestionBody question opts =
   87   question
   88   <> "\n\n"
   89   <> T.intercalate "\n" (zipWith formatOption [1 :: Int ..] opts)
   90   where
   91     formatOption n (QuestionOption lbl desc)
   92       | T.null desc = T.pack (show n) <> ") " <> lbl
   93       | otherwise   = T.pack (show n) <> ") " <> lbl <> " \8212 " <> desc
   94 
   95 -- | The trailing hint 'formatAsk' appends. Carries its own leading blank line,
   96 -- so @'formatQuestionBody' q os <> 'liveAskHint' os@ is byte-identical to
   97 -- 'formatAsk' @q os@. Pure.
   98 liveAskHint :: [QuestionOption] -> Text
   99 liveAskHint []   = "\n\nReply with your answer."
  100 liveAskHint _    = "\n\nReply with a number or type your own answer."

  106 -- | How many @unit@s @t@ occupies.
  107 chunkLen :: ChunkUnit -> Text -> Int
  108 chunkLen ChunkUtf8Bytes  = BS.length . TE.encodeUtf8
  109 chunkLen ChunkUtf16Units = T.foldl' (\n c -> n + codeUnits c) 0
  110   where
  111     -- A code point above the BMP (an emoji, many CJK extensions) needs a
  112     -- surrogate pair, i.e. two UTF-16 code units.
  113     codeUnits c = if fromEnum c > 0xFFFF then 2 else 1
  114 
  115 -- | Split a message into chunks of at most @limit@ @unit@s, breaking at the
  116 -- most natural boundary that fits.
  117 --
  118 -- Platforms inline only the first message and demote the rest to an
  119 -- attachment (or split it themselves at an arbitrary point), so where the
  120 -- split falls is user-visible. Breaking mid-sentence reads as a glitch;
  121 -- breaking at a paragraph break reads as two messages that were always meant
  122 -- to be two. Candidate boundaries, best first:
  123 --
  124 -- 1. paragraph break (@\n\n@)
  125 -- 2. line break (@\n@)
  126 -- 3. sentence end (@.@ @?@ @!@ followed by whitespace, not an abbreviation)
  127 -- 4. clause break (@;@ @:@ @,@ followed by a space)
  128 -- 5. word break (a space)
  129 -- 6. hard split — only when a single token is longer than the whole budget
  130 --
  131 -- Within a class the LATEST boundary that fits wins, so chunks are as full as
  132 -- the class allows rather than as short as possible.
  133 --
  134 -- The separator stays at the END of the chunk it precedes — it is neither
  135 -- dropped nor moved to the front of the next chunk. Keeping it there is what
  136 -- stops a blank line from becoming its own message: the "\n\n" between two
  137 -- paragraphs is consumed by the chunk before it, so the next chunk begins with
  138 -- the following paragraph's text.
  139 --
  140 -- Invariant: @T.concat (chunkMessage u l t)@ equals @t@ modulo TRAILING
  141 -- whitespace. Every non-space character survives, in order, with no separator
  142 -- lost from the middle; only a final all-whitespace remainder is discarded
  143 -- (see 'go').
  144 --
  145 -- Pure. A non-positive @limit@ is treated as "cannot chunk" and returns the
  146 -- text unchanged rather than looping.
  147 chunkMessage :: ChunkUnit -> Int -> Text -> [Text]
  148 chunkMessage unit limit msg
  149   | limit <= 0 = [msg]
  150   | chunkLen unit msg <= limit = [msg]
  151   | otherwise = go msg
  152   where
  153     go t
  154       | T.null t = []
  155       -- A remainder holding only separators is dropped rather than sent as a
  156       -- message of its own: a reply ending in a newline would otherwise
  157       -- deliver a final blank message, which reads as a glitch. Nothing is
  158       -- lost -- it holds no content, since every non-space character has
  159       -- already been emitted. The invariant preserved is "no content lost",
  160       -- not "every byte preserved".
  161       --
  162       -- Folding it into the previous chunk instead is not an option: the
  163       -- remainder is whatever did not fit, so appending it can push that chunk
  164       -- past the platform limit -- exactly what chunking exists to prevent.
  165       -- Only a TRAILING all-space remainder is dropped; the
  166       -- separator-before-content rule in 'naturalCut' already prevents an
  167       -- INTERIOR blank chunk.
  168       | T.null (T.strip t) = []
  169       | chunkLen unit t <= limit = [t]
  170       | otherwise =
  171           let (chunk, rest) = splitOne t
  172           in chunk : go rest
  173 
  174     -- Cut @t@ into (chunk, remainder). The chunk is never empty, so 'go'
  175     -- always makes progress.
  176     splitOne t =
  177       let fitting = takeFitting unit limit t
  178           cut
  179             | T.null fitting = 1   -- one character exceeds the whole budget
  180             | otherwise      = naturalCut fitting
  181       in (T.take cut t, T.drop cut t)
  182 
  183 -- | The longest prefix of @t@ whose unit count fits @limit@, counting whole
  184 -- characters only. A byte-level cut could split a codepoint and produce
  185 -- invalid UTF-8, which the platform would reject or mangle.
  186 takeFitting :: ChunkUnit -> Int -> Text -> Text
  187 takeFitting unit limit t = T.take (fittingChars 0 0) t
  188   where
  189     fittingChars n used =
  190       case T.uncons (T.drop n t) of
  191         Nothing -> n
  192         Just (c, _) ->
  193           let w = chunkLen unit (T.singleton c)
  194           in if used + w <= limit then fittingChars (n + 1) (used + w) else n
  195 
  196 -- | Choose how many characters of @fitting@ to keep, preferring the most
  197 -- natural boundary. The latest boundary in the highest-priority class that
  198 -- has one wins; with no separator at all, the whole fitted prefix is used.
  199 naturalCut :: Text -> Int
  200 naturalCut fitting =
  201   case catMaybes [paragraphCut, lineCut, sentenceCut, clauseCut, wordCut] of
  202     (n : _) -> n
  203     []      -> T.length fitting
  204   where
  205     -- A candidate cut is usable only if the chunk it produces holds some
  206     -- actual content. Without this, a separator at the very start of the
  207     -- remaining text (the "\n\n" between two paragraphs, which is where the
  208     -- previous chunk's trim left off) becomes the only candidate and we emit a
  209     -- message consisting of nothing but a blank line.
  210     usable n = T.any (not . isSpaceChar) (T.take n fitting)
  211 
  212     -- The latest position just past a run of separators, if any is > 0.
  213     latestAfter pat = case [ i + T.length pat
  214                            | i <- [0 .. T.length fitting - T.length pat]
  215                            , pat `T.isPrefixOf` T.drop i fitting
  216                            , i + T.length pat > 0
  217                            , usable (i + T.length pat) ] of
  218                         [] -> Nothing
  219                         ns -> Just (maximum ns)
  220     paragraphCut = latestAfter "\n\n"
  221     lineCut      = latestAfter "\n"
  222     wordCut      = latestAfter " "
  223     clauseCut    = latestBefore ' '
  224     sentenceCut  = latestSentence
  225 
  226     -- Break after a clause separator that is followed by a space.
  227     latestBefore ch = case [ i + 1
  228                            | i <- [1 .. T.length fitting - 1]
  229                            , T.index fitting i == ch
  230                            , T.index fitting (i - 1) `elem` (";:," :: String)
  231                            , usable (i + 1) ] of
  232                         [] -> Nothing
  233                         ns -> Just (maximum ns)
  234 
  235     -- Break after sentence-ending punctuation followed by whitespace.
  236     latestSentence = case [ i + 1
  237                           | i <- [0 .. T.length fitting - 1]
  238                           , T.index fitting i `elem` (".?!" :: String)
  239                           , endsSentence fitting i
  240                           , usable (i + 1) ] of
  241                        [] -> Nothing
  242                        ns -> Just (maximum ns)
  243 
  244 -- | Does the terminator at offset @i@ plausibly end a sentence? Requires the
  245 -- next character to be whitespace or absent, and rejects abbreviations and
  246 -- single-letter initials.
  247 --
  248 -- Deliberately conservative: a MISSED sentence break falls through to a
  249 -- clause or word break (a cosmetic loss), whereas a WRONG one — splitting
  250 -- "Dr. Smith" after "Dr." — is visible. Note "e.g." is caught by the
  251 -- single-letter-initial rule only if the lookbehind is applied per token, so
  252 -- the abbreviation list carries the common dotted forms explicitly.
  253 endsSentence :: Text -> Int -> Bool
  254 endsSentence fitting i =
  255   let after = T.drop (i + 1) fitting
  256       nextOk = case T.uncons after of
  257         Nothing     -> True
  258         Just (c, _) -> c == ' ' || c == '\n' || c == '\t'
  259       -- The word the terminator is attached to. Taken back to the previous
  260       -- whitespace, so "e.g." yields "e.g" here and the abbreviation list
  261       -- matches on the token WITHOUT its final period.
  262       prevToken = T.dropWhileEnd (== '.') (T.takeWhileEnd (/= ' ') (T.take i fitting))
  263       lowerTok  = T.toLower prevToken
  264       isAbbrev  = lowerTok `elem`
  265         [ "dr", "mr", "mrs", "ms", "prof", "sr", "jr", "st", "vs", "etc"
  266         , "e.g", "i.e", "fig", "no", "vol", "al", "cf", "approx", "inc", "ltd" ]
  267       isInitial = T.length prevToken == 1 && T.all isAsciiUpper prevToken
  268   in nextOk && not isAbbrev && not isInitial

  271 -- | Whitespace, spelled locally so this module needs no @Data.Char@ import for
  272 -- one predicate.
  273 isSpaceChar :: Char -> Bool
  274 isSpaceChar c = c == ' ' || c == '\n' || c == '\t' || c == '\r'

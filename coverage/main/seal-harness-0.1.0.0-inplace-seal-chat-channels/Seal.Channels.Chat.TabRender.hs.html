    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Pure rendering for the chat channel's @\/tab list@.
    3 --
    4 -- The registry's @\/tab list@ (@Seal.Command.Tab.listCmd@) cannot serve this
    5 -- surface: it has no access to the broker's thinking set, so it cannot show
    6 -- whether a tab is busy, and its format carries no repo. Chat renders from
    7 -- @GET \/api\/lists@ instead, which carries both.
    8 --
    9 -- Row format (D5):
   10 --
   11 -- > <tab-id> <activity> <repo>: <name>     -- when the session has a repo
   12 -- > <tab-id> <activity>: <name>            -- when it does not
   13 --
   14 -- @<tab-id>@ is the index as @\/tab close N@ accepts it (0-9, a-z);
   15 -- @<activity>@ is @busy@ or @idle@.
   16 module Seal.Channels.Chat.TabRender
   17   ( TabListRow (..)
   18   , renderTabListRow
   19   , tabListRows
   20   ) where
   21

   33 -- | One resolved row of @\/tab list@.
   34 data TabListRow = TabListRow
   35   { tlrIndex :: !Int
   36   , tlrBusy  :: !Bool
   37   , tlrRepo  :: Maybe Text
   38   , tlrName  :: !Text
   39   } deriving stock (Eq, Show)
   40 
   41 -- | Render one row. Pure, and the single source of the wire format.
   42 renderTabListRow :: TabListRow -> Text
   43 renderTabListRow r =
   44   renderIndex (tlrIndex r) <> " " <> activity <> repoPart <> ": " <> tlrName r
   45   where
   46     activity = if tlrBusy r then "busy" else "idle"
   47     repoPart = maybe "" (" " <>) (tlrRepo r)
   48 
   49 -- | Render a tab index the way the terse grammar spells it (0-9, then a-z).
   50 -- An out-of-range index (which the server never emits) falls back to decimals.
   51 renderIndex :: Int -> Text
   52 renderIndex n = case mkTabIndex n of
   53   Right i -> T.singleton (tabIndexToChar i)
   54   Left _  -> T.pack (show n)
   55 
   56 -- | Resolve every tab into a row.
   57 tabListRows :: ListsJson -> [TabListRow]
   58 tabListRows lj = map row (ljTabs lj)
   59   where
   60     thinking = ljThinkingSessionIds lj
   61 
   62     row t = TabListRow
   63       { tlrIndex = tjIndex t
   64       , tlrBusy  = isBusy t
   65       , tlrRepo  = mSession t >>= lookupText "repoUrl" >>= repoFromUrl
   66       , tlrName  = fromMaybe "harness" (nameOf t)
   67       }
   68 
   69     -- A tab is busy when its session is thinking. A harness tab has no
   70     -- session, so it falls back to the harness's own liveness.
   71     isBusy t = case tjSessionId t of
   72       Just sid -> sid `elem` thinking
   73       Nothing  -> tjStatus t == Just "running"
   74 
   75     -- The session info object behind a tab, when there is one.
   76     mSession t = tjSessionId t >>= \sid ->
   77       case filter (\s -> lookupText "id" s == Just sid) (ljTabSessions lj) of
   78         (s : _) -> Just s
   79         []      -> Nothing
   80 
   81     -- Label, then the frontend's title cascade, then a stable fallback.
   82     nameOf t = firstNonEmpty
   83       [ tjLabel t
   84       , mSession t >>= lookupText "description"
   85       , mSession t >>= lookupText "autoSummary"
   86       , mSession t >>= lookupText "firstMessageSnippet"
   87       , mSession t >>= lookupText "agent"
   88       , Just $ case tjSessionId t of
   89           Just sid -> "session:" <> T.take 12 sid
   90           Nothing  -> "harness"
   91       ]
   92 
   93 -- | The last non-empty path segment of a repo URL, with a trailing @.git@ and
   94 -- any trailing slash removed. @Nothing@ for an empty or segment-less URL.
   95 repoFromUrl :: Text -> Maybe Text
   96 repoFromUrl raw =
   97   case lastNonEmpty (T.splitOn "/" (T.dropWhileEnd (== '/') (T.strip raw))) of
   98     Nothing  -> Nothing
   99     Just seg -> Just (fromMaybe seg (T.stripSuffix ".git" seg))
  100 
  101 lastNonEmpty :: [Text] -> Maybe Text
  102 lastNonEmpty = foldl' (\acc t -> if T.null (T.strip t) then acc else Just t) Nothing
  103 
  104 -- | The first non-empty 'Just' in the list.
  105 firstNonEmpty :: [Maybe Text] -> Maybe Text
  106 firstNonEmpty = foldr pick Nothing
  107   where
  108     pick (Just t) _ | not (T.null (T.strip t)) = Just t
  109     pick _        acc                          = acc
  110 
  111 -- | Look up a string field in a JSON object.
  112 lookupText :: Text -> Value -> Maybe Text
  113 lookupText k val = case val of
  114   A.Object o -> case KeyMap.lookup (Key.fromText k) o of
  115     Just (A.String t) -> Just t
  116     _                 -> Nothing
  117   _ -> Nothing

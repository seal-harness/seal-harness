   26 -- | A routing error (invalid tab index char).
   27 newtype RouteError = RouteError Text
   28   deriving stock (Eq, Show)
   29 
   30 -- | The Layer-1 routing decision for chat channels. Mirrors
   31 -- 'Seal.Routing.Route.RoutingDecision' but lives in the chat-channels
   32 -- package so it has no dependency on server internals.
   33 data ChatRoute
   34   = ChatFocus TabIndex                 -- ^ /N
   35   | ChatInject TabIndex Text           -- ^ /N payload
   36   | ChatPlain Text                     -- ^ plain text to the focused session
   37   | ChatTabCommand TabSlashCommand     -- ^ /tab <subcommand> …
   38   | ChatCurrentTab                     -- ^ bare /tab — show the current tab
   39   | ChatNewSession Text                -- ^ /new [args] — the raw arg string
   40   | ChatSlash Text                     -- ^ other /commands (deferred to the HTTP API)
   41   deriving stock (Eq, Show)
   42 
   43 -- | Route one inbound line. The Layer-1 terse grammar:
   44 --
   45 -- * @\/N@          -> 'ChatFocus' N (N is a single char 0-9a-z, at end-of-string
   46 --                   or followed by a space)
   47 -- * @\/N payload@  -> 'ChatInject' N payload
   48 -- * @\/tab@        -> 'ChatCurrentTab' (show the current tab)
   49 -- * @\/new [args]@ -> 'ChatNewSession' (the raw text after @/new @)
   50 -- * @\/<other>…@   -> 'ChatSlash' (deferred to the HTTP API — the gateway
   51 --                   server routes slash commands for the web frontend)
   52 -- * anything else  -> 'ChatPlain'
   53 route :: Text -> Either RouteError ChatRoute
   54 route t
   55   | T.null t             = Right (ChatPlain t)
   56   | T.head t /= '/'      = Right (ChatPlain t)
   57   | otherwise            =
   58       let rest = T.drop 1 t  -- drop the leading '/'
   59       in case T.uncons rest of
   60            Nothing -> Right (ChatPlain "/")  -- a bare "/" — treat as plain
   61            Just (c, after)
   62               | isTabChar c && (T.null after || T.head after == ' ') ->
   63                   -- single-char /N or /N payload (the tab grammar)
   64                   case tabIndexFromChar c of
   65                     Left e -> Left (RouteError e)
   66                     Right idx -> Right (focusOrInject idx after)
   67                | rest == "tab" ->
   68                   Right ChatCurrentTab
   69                | rest == "new" || T.isPrefixOf "new " rest ->
   70                   Right (ChatNewSession (stripNewPrefix rest))
   71                | otherwise ->
   72                   Right (ChatSlash rest)
   73   where
   74     isTabChar c = isDigit c || isAsciiLower c
   75 
   76 -- | Extract the argument text from the @new@ prefix. @\"new\"@ yields
   77 -- @\"\"@; @\"new -p anthropic\"@ yields @\"-p anthropic\"@.
   78 stripNewPrefix :: Text -> Text
   79 stripNewPrefix rest =
   80   if rest == "new"
   81     then ""
   82     else T.drop 1 (snd (T.breakOn " " rest))  -- everything after the first space
   83 
   84 -- | Given a valid tab index + the text after it: if the rest is empty (or
   85 -- whitespace-only), it's a Focus; otherwise it's an Inject (the payload is
   86 -- the text after the first space, preserving internal spaces verbatim).
   87 focusOrInject :: TabIndex -> Text -> ChatRoute
   88 focusOrInject idx after
   89   | T.null (T.strip after) = ChatFocus idx
   90   | otherwise               =
   91       let payload = T.drop 1 (snd (T.breakOn " " after))  -- everything after the first space
   92       in ChatInject idx payload
   93 
   94 -- | Parse @\/tab focus \<N\>@ from the inbound body. Returns 'Just' the
   95 -- 'TabIndex' if the body is @\/tab focus \<N\>@ (case-insensitive, N is a
   96 -- single tab-index char 0-9a-z), 'Nothing' otherwise. Used by the loop to
   97 -- intercept @\/tab focus@ locally (send a 'FocusOp' over WS, not an HTTP
   98 -- send).
   99 parseTabFocus :: Text -> Maybe TabIndex
  100 parseTabFocus body =
  101   case route body of
  102     Right (ChatSlash rest) ->
  103       let parts = T.words (T.toCaseFold rest)
  104       in case parts of
  105            ["tab", "focus", idxStr] -> case T.uncons idxStr of
  106              Just (c, _) -> either (const Nothing) Just (tabIndexFromChar c)
  107              Nothing     -> Nothing
  108            _ -> Nothing
  109     _ -> Nothing
  110 
  111 -- | The terse-grammar synopsis (for /help). One line.
  112 terseSynopsis :: Text
  113 terseSynopsis = "/N [payload]  Switch to tab N (0-9a-z), or inject payload into it"

  147 -- | Per-conversation watch-all-tabs toggle state. When 'True' for a
  148 -- conversation, the channel receives a notification every time any
  149 -- non-focused tab finishes thinking (harness-status transitions from
  150 -- @thinking@ to @idle@). Thread-safe via 'TVar'.
  151 data WatchState = WatchState
  152   { wsVar  :: TVar (Map ConversationKey Bool)
  153   , wsSave :: Maybe (IO ())
  154   -- ^ The save action (snapshots the TVar and writes to disk). Called
  155   -- inside the 'wsLock' critical section so the snapshot is consistent
  156   -- with the mutation.
  157   , wsLock :: MVar ()
  158   -- ^ Serializes the mutation+persist sequence so a stale snapshot from
  159   -- one thread cannot overwrite a newer on-disk write from another.
  160   -- Without this lock, concurrent toggles from Signal + Telegram (which
  161   -- share one 'WatchState') can lose state after a restart.
  162   }
  163 
  164 -- | Create a new empty watch-state map (watch mode off for all
  165 -- conversations).
  166 newWatchState :: IO WatchState
  167 newWatchState = do
  168   tv <- newTVarIO Map.empty
  169   lock <- newMVar ()
  170   pure (WatchState tv Nothing lock)
  171 
  172 -- | Create a persisting watch-state store. The supplied save function is
  173 -- called after every mutation (snapshotting the full current map, so the
  174 -- last writer wins with a consistent view). Mirrors
  175 -- 'Seal.Channels.Cursor.newPersistingCursorStore'.
  176 newPersistingWatchState :: (Map ConversationKey Bool -> IO ()) -> IO WatchState
  177 newPersistingWatchState saveFn = do
  178   tv <- newTVarIO Map.empty
  179   lock <- newMVar ()
  180   let ws = WatchState tv (Just (saveAction ws)) lock
  181       saveAction s = saveFn =<< snapshotWatch s
  182   pure ws
  183 
  184 -- | Replace the store's map in one STM transaction. Used at boot to seed
  185 -- the in-memory store from the persisted @watch_state.json@. Does NOT
  186 -- persist (the caller is loading FROM disk).
  187 seedWatchState :: WatchState -> Map ConversationKey Bool -> IO ()
  188 seedWatchState ws m = atomically (writeTVar (wsVar ws) m)
  189 
  190 -- | Snapshot the current map. Used by the save action.
  191 snapshotWatch :: WatchState -> IO (Map ConversationKey Bool)
  192 snapshotWatch ws = readTVarIO (wsVar ws)
  193 
  194 -- | Look up whether watch mode is enabled for a conversation. 'False'
  195 -- when the conversation has no entry (the default).
  196 lookupWatch :: WatchState -> ConversationKey -> IO Bool
  197 lookupWatch ws key = fromMaybe False . Map.lookup key <$> readTVarIO (wsVar ws)
  198 
  199 -- | Toggle watch mode for a conversation. Returns the new state.
  200 toggleWatch :: WatchState -> ConversationKey -> IO Bool
  201 toggleWatch ws key = do
  202   -- Hold the lock through mutation+persist so no concurrent thread can
  203   -- mutate the TVar between our snapshot and our save (which would let a
  204   -- stale snapshot overwrite a newer on-disk write).
  205   withMVar (wsLock ws) $ \_ -> do
  206     newVal <- atomically $ do
  207       m <- readTVar (wsVar ws)
  208       let v = not (fromMaybe False (Map.lookup key m))
  209       writeTVar (wsVar ws) (Map.insert key v m)
  210       pure v
  211     persistWatch ws
  212     pure newVal
  213 
  214 -- | Run the persist action (if any) after a successful mutation. A save
  215 -- failure is logged to stderr and swallowed — the in-memory store stays
  216 -- authoritative within the session; the next successful mutation will
  217 -- retry the save (writing the full current map, so a missed save
  218 -- self-heals). Mirrors 'Seal.Channels.Cursor.persistCursor'.
  219 persistWatch :: WatchState -> IO ()
  220 persistWatch ws =
  221   case wsSave ws of
  222     Nothing  -> pure ()
  223     Just act -> act `catch` \e ->
  224       dbg ("[watch] watch_state.json save failed: " <> T.pack (show (e :: SomeException)))

  250 -- | The per-conversation background-session store. Each conversation's
  251 -- list is in creation order (oldest first) so the per-conversation cap can
  252 -- evict the oldest session when a new one is started. Thread-safe via
  253 -- 'TVar'.
  254 --
  255 -- The store also remembers the session ids this conversation has owned for
  256 -- a bounded window after they are retired ('bsKnown'). That memory is what
  257 -- keeps a bg turn out of the watch-all-tabs path: 'BeActivity' is broadcast
  258 -- to every subscriber, so the conversation's own connection receives
  259 -- @harness-status idle@ / @tool-call@ for the bg session too -- possibly
  260 -- after the bg connection's reader has already retired it. Without the
  261 -- memory those late events would be classified as "not a bg session" and
  262 -- produce a bogus "Tab ?" watch notification plus a duplicate reply.
  263 data BgSessions = BgSessions
  264   { bsLive  :: TVar (Map ConversationKey [BgSession])
  265   , bsKnown :: TVar (Map ConversationKey [SessionId])
  266   }
  267 
  268 -- | How many retired bg session ids a conversation remembers for the watch
  269 -- exclusion. Bounded so a long-lived conversation cannot grow without
  270 -- limit; the window only has to outlive one turn's activity broadcasts.
  271 maxKnownBgSids :: Int
  272 maxKnownBgSids = 32
  273 
  274 -- | Create a new empty background-session store.
  275 newBgSessions :: IO BgSessions
  276 newBgSessions = BgSessions <$> newTVarIO Map.empty <*> newTVarIO Map.empty

  278 -- | The conversation's live background sessions, oldest first.
  279 bgSessionsFor :: BgSessions -> ConversationKey -> IO [BgSession]
  280 bgSessionsFor bgs key =
  281   fromMaybe [] . Map.lookup key <$> readTVarIO (bsLive bgs)

  283 -- | Does @sid@ belong to one of this conversation's background sessions
  284 -- (live, or retired within the remembered window)? Used to keep bg turns
  285 -- out of the watch-all-tabs path.
  286 bgOwnsSession :: BgSessions -> ConversationKey -> SessionId -> IO Bool
  287 bgOwnsSession bgs key sid =
  288   elem sid . fromMaybe [] . Map.lookup key <$> readTVarIO (bsKnown bgs)
  289 
  290 -- | Register a background session at the end of its conversation's list
  291 -- (creation order) and remember its id for the watch exclusion.
  292 bgInsert :: BgSessions -> ConversationKey -> BgSession -> IO ()
  293 bgInsert bgs key sess = do
  294   atomically (modifyTVar' (bsLive bgs)
  295     (Map.insertWith (flip (++)) key [sess]))
  296   atomically (modifyTVar' (bsKnown bgs)
  297     (Map.insertWith (\new old -> take maxKnownBgSids (new ++ filter (`notElem` new) old))
  298        key [bgSid sess]))
  299 
  300 -- | Untrack a background session. Returns the session when it was tracked,
  301 -- 'Nothing' when it was already gone (so retirements are idempotent).
  302 -- Retiring does NOT forget the session id (see 'bsKnown'): late activity
  303 -- events for it still must stay out of the watch path.
  304 bgRemove :: BgSessions -> ConversationKey -> SessionId -> IO (Maybe BgSession)
  305 bgRemove bgs key sid = atomically $ do
  306   let tv = bsLive bgs
  307   m <- readTVar tv
  308   let sessions = fromMaybe [] (Map.lookup key m)
  309       (dropped, kept) = partition ((== sid) . bgSid) sessions
  310       m' = case kept of
  311              [] -> Map.delete key m
  312              _  -> Map.insert key kept m
  313   writeTVar tv m'
  314   pure (case dropped of
  315           (s : _) -> Just s
  316           []      -> Nothing)

  318 -- | Drop every pending ask registered for a session. A retired bg session
  319 -- can no longer be answered, so its asks must not accumulate (or be
  320 -- resolvable by a later button tap).
  321 dropPendingAsks :: PendingAsks -> SessionId -> IO ()
  322 dropPendingAsks pendingAsks sid =
  323   atomically (modifyTVar' (paStore pendingAsks) (Map.delete sid))

  406 -- | Run the generic chat-channel loop. Blocks until the channel's
  407 -- 'ccReceive' returns EOF. Each conversation gets its own WS connection
  408 -- for streaming.
  409 runChatChannel :: ChatChannel c => ChatChannelConfig -> c -> WatchState -> IO ()
  410 runChatChannel cfg chan watchState = do
  411   sessions <- newSessionMap
  412   -- Map of conversation keys to their WS client + streaming state.
  413   wsConns <- newTVarIO Map.empty :: IO (TVar (Map ConversationKey WsConn))
  414   pendingAsks <- newPendingAsks
  415   tabTracker <- newTVarIO Map.empty :: IO (TVar (Map ConversationKey (SessionId, [TabJson])))
  416   thinkingTabs <- newThinkingTabs
  417   bgSessions <- newBgSessions
  418   loop sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs
  419   where
  420     loop sessions wsConns pendingAsks bgSessions tabTracker ws tt = do
  421       mMsg <- ccReceive chan
  422       case mMsg of
  423         Nothing -> pure ()  -- EOF
  424         Just (InboundMessage src body mCbData) -> do
  425           dbg ("received: " <> body)
  426           let key = convKeyFromSource src
  427           case mCbData of
  428             Just cbData -> do
  429               -- Callback query (button tap): resolve to a pending ask,
  430               -- answer via HTTP, acknowledge, and remove keyboard.
  431               handleCallback cfg chan sessions pendingAsks bgSessions key src cbData
  432             Nothing ->
  433               -- Regular text message: route normally.
  434               handleInboundWithSource cfg chan sessions wsConns pendingAsks bgSessions tabTracker ws tt key (Just src) body
  435           loop sessions wsConns pendingAsks bgSessions tabTracker ws tt

  437 -- | 'handleInbound' with no channel provenance (for the tests and any caller
  438 -- that has no 'MessageSource'). The live loop calls
  439 -- 'handleInboundWithSource' with the inbound message's source.
  440 handleInbound
  441   :: ChatChannel c
  442   => ChatChannelConfig -> c -> SessionMap
  443   -> TVar (Map ConversationKey WsConn)

  449   -> ConversationKey -> Text

  451 handleInbound cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key =

  454 -- | Handle one inbound message with its authenticated channel provenance:
  455 -- resolve the session, route, and dispatch. The source (when present) rides
  456 -- along to the HTTP send so the gateway stamps the channel + user id into the
  457 -- turn's transcript metadata.
  458 handleInboundWithSource
  459   :: ChatChannel c
  460   => ChatChannelConfig -> c -> SessionMap
  461   -> TVar (Map ConversationKey WsConn)
  462   -> PendingAsks
  463   -> BgSessions
  464   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  465   -> WatchState
  466   -> ThinkingTabs
  467   -> ConversationKey -> Maybe MessageSource -> Text
  468   -> IO ()
  469 handleInboundWithSource cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key mSrc body = do
  470   let apiBase = gcApiBase (cccGateway cfg)
  471       mgr = cccHttpManager cfg
  472   -- Intercept /watch (and /watch on|off) before normal routing — it
  473   -- toggles per-conversation loop state, not a gateway command.
  474   if isWatchCommand body
  475     then do
  476       dbg ("[watch] /watch command received, key=" <> ckConv key)
  477       handleWatchToggle chan watchState key body
  478       -- When watch mode is turned ON, ensure a WS connection exists so
  479       -- the channel receives BeActivity events for all tabs. Without
  480       -- this, /watch as the first message would enable watch mode but
  481       -- never receive any events (no WS connection = no event source).
  482       watchOn <- lookupWatch watchState key
  483       dbg ("[watch] watchOn=" <> (if watchOn then "true" else "false") <> ", ensuring WS conn")
  484       when watchOn $
  485         void (resolveSession cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key)
  486     else case parseTabFocus body of
  487     Just idx -> do
  488       handleFocus cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key idx
  489     Nothing -> case route body of
  490       Right (ChatFocus idx) ->
  491         handleFocus cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key idx
  492       Right (ChatInject idx payload) -> do
  493         -- Send the payload to the target tab's session WITHOUT changing
  494         -- the focused tab (no FocusOp, no "focused tab N" confirmation).
  495         handleInject cfg chan mSrc idx payload
  496       Right ChatCurrentTab -> do
  497         -- Send the current tab info via HTTP (the gateway routes /tab).
  498         mSid <- sessionLookup sessions key
  499         case mSid of
  500           Just sid -> do
  501             eResult <- cccSendMessage cfg mgr apiBase sid Nothing "/tab"
  502             case eResult of
  503               Right sr -> ccSend chan (srResponse sr)
  504               Left e -> ccSend chan ("error: " <> e)
  505           Nothing -> ccSend chan "no current tab"
  506       Right (ChatNewSession args) -> do
  507         -- Create a new session via HTTP, update the session map.
  508         handleNewSession cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key args
  509       Right (ChatBackground prompt) -> do
  510         -- Run the prompt in a freshly minted headless session. The
  511         -- conversation's focused session and tab list are untouched.
  512         handleBackground cfg chan pendingAsks bgSessions key mSrc prompt
  513       Right (ChatSlash _cmd) ->
  514         sendSlash cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key mSrc body
  515       Right (ChatTabCommand _) ->
  516         -- Tab commands go through the HTTP API (the gateway routes them).
  517         sendSlash cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key mSrc body
  518       Right ChatTabList -> handleTabList cfg chan
  519       Right (ChatPlain text) -> do
  520         -- A plain message answers a pending ask from one of this

  644 -- | Handle @/bg <prompt>@: mint a headless session (no tab), open a
  645 -- dedicated WS connection focused on it, register it, and run the prompt on
  646 -- it. The reply is posted back to this conversation by
  647 -- 'handleBgServerEvent'; the conversation's focused session, session map,
  648 -- and tab tracking are never touched.
  649 --
  650 -- A blank prompt answers 'bgUsage' and creates nothing; @-h@ / @--help@
  651 -- answers 'bgHelp' and creates nothing. Every failure (mint, connect, send)
  652 -- closes whatever it opened, untracks the session, drops its pending asks,
  653 -- and reports on the channel.
  654 handleBackground
  655   :: ChatChannel c
  656   => ChatChannelConfig -> c -> PendingAsks -> BgSessions
  657   -> ConversationKey -> Maybe MessageSource -> Text
  658   -> IO ()
  659 handleBackground cfg chan pendingAsks bgSessions key mSrc prompt
  660   | T.null (T.strip prompt) = ccSend chan bgUsage
  661   | isBgHelpArg prompt      = ccSend chan bgHelp
  662   | otherwise               = do
  663       let apiBase = gcApiBase (cccGateway cfg)
  664           mgr = cccHttpManager cfg
  665           gwCfg = cccGateway cfg
  666       -- (a) mint a headless session, (b) validate its id.
  667       eMinted <- cccNewSession cfg mgr apiBase (A.object ["headless" .= A.Bool True])
  668       case eMinted >>= mkSessionId of
  669         Left e -> ccSend chan ("bg session failed: " <> e)
  670         Right bgSid -> do
  671           -- (c) per-session streaming state + delivery flag.
  672           ss <- newStreamingState
  673           delivered <- newIORef False
  674           let callback =
  675                 handleBgServerEvent cfg chan key pendingAsks bgSessions bgSid ss delivered
  676           -- (d) the dedicated connection for the bg session.
  677           eWs <- cccStartWs cfg (gcHost gwCfg) (gcWsPort gwCfg) callback
  678           case eWs of
  679             Left e -> ccSend chan ("bg session failed: " <> e)
  680             Right ws -> do
  681               -- (e) focus before registering or sending, so no turn event
  682               -- can arrive for an untracked session (which would leak the
  683               -- connection and skip the watch exclusion).
  684               wcFocus ws bgSid
  685               evictOldestBgSession pendingAsks bgSessions key
  686               bgInsert bgSessions key BgSession { bgSid = bgSid, bgWs = ws }
  687               -- (f) run the prompt on the bg session (never on the
  688               -- conversation's session) and confirm.
  689               eSent <- cccSendMessage cfg mgr apiBase bgSid mSrc prompt
  690               case eSent of
  691                 Right sr | srKind sr /= "error" ->
  692                   ccSend chan ("bg session " <> shortSessionId bgSid <> " started")
  693                 failure -> do
  694                   _ <- bgRemove bgSessions key bgSid
  695                   wcClose ws
  696                   dropPendingAsks pendingAsks bgSid
  697                   ccSend chan ("bg session failed: " <> bgSendFailure failure)
  698   where
  699     bgSendFailure (Left e)   = e
  700     bgSendFailure (Right sr) = fromMaybe "send failed" (srError sr)

  702 -- | Enforce the per-conversation cap: when the conversation already holds
  703 -- 'maxBgSessions' live bg sessions, retire the oldest one so a further
  704 -- session can start.
  705 evictOldestBgSession :: PendingAsks -> BgSessions -> ConversationKey -> IO ()
  706 evictOldestBgSession pendingAsks bgSessions key = do
  707   existing <- bgSessionsFor bgSessions key
  708   when (length existing >= maxBgSessions) $
  709     case existing of
  710       (oldest : _) -> void (retireBgSession pendingAsks bgSessions key (bgSid oldest))

  713 -- | Retire one background session: close its WS connection, untrack it, and
  714 -- drop its pending asks. Returns 'True' when it was still tracked, so a
  715 -- repeated @idle@ is a no-op.
  716 retireBgSession
  717   :: PendingAsks -> BgSessions -> ConversationKey -> SessionId -> IO Bool
  718 retireBgSession pendingAsks bgSessions key sid = do
  719   mSess <- bgRemove bgSessions key sid
  720   for_ mSess (wcClose . bgWs)
  721   dropPendingAsks pendingAsks sid
  722   pure (isJust mSess)

  724 -- | The WS event handler for a background session's dedicated connection.
  725 -- Acts only on events for 'bgSid':
  726 --
  727 -- * a @response@ 'SeEntry' — post the recorded reply to the conversation and
  728 --   mark it delivered. Token deltas are deliberately not streamed for bg
  729 --   turns (pre-regression parity: the reply arrives on the recorded entry).
  730 -- * 'SeAsk' — render the question on the conversation (and register it as
  731 --   pending, so a button tap can answer it).
  732 -- * @tool-call@ activity — post the tool-progress line.
  733 -- * @harness-status@ @idle@ — retire the session (close, untrack, drop its
  734 --   asks) and report a turn that produced no reply.
  735 -- * everything else (entry-updates, list snapshots, replay markers, other
  736 --   sessions) — ignored.
  737 handleBgServerEvent
  738   :: ChatChannel c
  739   => ChatChannelConfig -> c -> ConversationKey -> PendingAsks -> BgSessions
  740   -> SessionId -> StreamingState -> IORef Bool -> ServerEvent -> IO ()
  741 handleBgServerEvent cfg chan key pendingAsks bgSessions bgSid ss delivered ev =
  742   case ev of
  743     SeEntry sid val
  744       | sid == bgSid
  745       , extractDirection val == "response"
  746       , not (T.null (extractEntryText val)) -> do
  747           handleEntry cfg chan ss val
  748           writeIORef delivered True
  749     SeAsk sid val
  750       | sid == bgSid -> handleAsk cfg chan key pendingAsks sid val
  751     SeAskResolved _ val -> forgetAskByPrefix pendingAsks (extractAskId val)
  752     SeActivity sid val
  753       | sid == bgSid -> case extractActivityKind val of
  754           "tool-call" -> handleToolCallActivity cfg chan ss val
  755           "harness-status"
  756             | extractActivityStatus val == "idle" -> do
  757                 -- The close trigger is harness-status idle only (not
  758                 -- reply-delivered, which also fires at every turn end when
  759                 -- sub-agent children keep the session thinking).
  760                 wasTracked <- retireBgSession pendingAsks bgSessions key bgSid
  761                 when wasTracked $ do
  762                   wasDelivered <- readIORef delivered
  763                   if wasDelivered
  764                     then do
  765                       -- The reply was accumulated by 'handleEntry' but bg
  766                       -- idle retires the session WITHOUT going through
  767                       -- 'handleHarnessStatus', so nothing else would deliver
  768                       -- it: the single-bubble style never writes the answer
  769                       -- into the bubble (it must arrive as its own notifying
  770                       -- message). Send it here.
  771                       answer <- stripCursorText (cccStreamCfg cfg)
  772                                   <$> readIORef (ssAccumulated ss)
  773                       unless (T.null (T.strip answer)) $ ccSend chan answer
  774                     else ccSend chan ("bg session " <> shortSessionId bgSid
  775                                       <> " produced no reply")
  776           _ -> pure ()
  777     _ -> pure ()

  779 -- | Ensure a WS connection exists for the conversation. If one exists,
  780 -- send a FocusOp to change focus. If not, start a new WS connection with
  781 -- the streaming event handler and send a FocusOp.
  782 --
  783 -- A connection is only reusable for the session it was built for.
  784 -- 'handleServerEvent' closes over the focused session **by value**, so a
  785 -- FocusOp cannot re-target the filter: an existing connection focused on
  786 -- session A keeps dropping session B's entries and asks. Re-targeting
  787 -- therefore closes the old connection and builds a new one whose callback
  788 -- is bound to the new session.
  789 ensureWsConn
  790   :: ChatChannel c
  791   => ChatChannelConfig -> c
  792   -> TVar (Map ConversationKey WsConn)
  793   -> PendingAsks
  794   -> BgSessions
  795   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  796   -> WatchState
  797   -> ThinkingTabs
  798   -> ConversationKey -> SessionId
  799   -> IO ()
  800 ensureWsConn cfg chan wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key sid = do
  801   conns <- readTVarIO wsConns
  802   dbg ("[watch] ensureWsConn key=" <> ckConv key <> " hasConn=" <> (case Map.lookup key conns of Just _ -> "true"; Nothing -> "false"))
  803   case Map.lookup key conns of
  804     Just (boundSid, ws, _)
  805       | boundSid == sid -> wcFocus ws sid  -- already connected; just change focus
  806       -- Re-target: the callback closed over 'boundSid' by value, so a
  807       -- FocusOp alone would leave this conversation deaf to 'sid'.
  808       | otherwise -> do
  809           atomically (modifyTVar' wsConns (Map.delete key))
  810           wcClose ws
  811           dbg ("[watch] ensureWsConn re-target " <> sessionIdText boundSid
  812                <> " -> " <> sessionIdText sid)
  813           connectWs
  814     Nothing -> do
  815       connectWs
  816   where
  817     -- Start a new WS connection whose event handler is bound to 'sid'.
  818     connectWs = do
  819       let gwCfg = cccGateway cfg
  820           callback = handleServerEvent cfg chan key wsConns pendingAsks
  821             bgSessions tabTracker watchState thinkingTabs sid
  822       eWs <- cccStartWs cfg (gcHost gwCfg) (gcWsPort gwCfg) callback
  823       case eWs of
  824         Left e -> dbg ("[watch] WS connect FAILED: " <> e)
  825         Right ws -> do
  826           ss <- newStreamingState
  827           atomically (modifyTVar' wsConns (Map.insert key (sid, ws, ss)))
  828           wcFocus ws sid

  830 -- | The WS event handler: processes streaming events for one conversation.
  831 -- Creates/edits/finalizes platform messages from entry-update/entry/activity
  832 -- events.
  833 handleServerEvent
  834   :: ChatChannel c
  835   => ChatChannelConfig -> c -> ConversationKey
  836   -> TVar (Map ConversationKey WsConn)
  837   -> PendingAsks
  838   -> BgSessions
  839   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  840   -> WatchState
  841   -> ThinkingTabs
  842   -> SessionId -> ServerEvent -> IO ()
  843 handleServerEvent cfg chan key wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs focusedSid ev = do
  844   -- Staleness guard: when the WS connection for this conversation has been
  845   -- re-targeted to a different session (the user changed focus via /N or
  846   -- sent a message that created a new session), this callback is stale —
  847   -- its 'focusedSid' no longer matches the 'boundSid' in 'wsConns'. A
  848   -- stale callback must not process events: the old streaming state has
  849   -- been replaced, and processing events here would send messages from
  850   -- the wrong session's accumulated text (e.g. a partial reply fragment
  851   -- arriving before the watch notification from the new connection).
  852   conns <- readTVarIO wsConns
  853   let isCurrent = case Map.lookup key conns of
  854         Just (boundSid, _, _) -> boundSid == focusedSid
  855         Nothing -> False
  856   when isCurrent $ case ev of
  857     SeEntryUpdate sid val
  858       | sid == focusedSid ->
  859           withStreamingState wsConns key (\ss -> handleEntryUpdate cfg chan ss val)
  860     SeEntry sid val
  861       | sid == focusedSid ->
  862           withStreamingState wsConns key (\ss -> handleEntry cfg chan ss val)
  863     SeActivity sid val
  864       | sid == focusedSid -> do
  865           dbg ("[watch] SeActivity focused sid=" <> sessionIdText sid <> " kind=" <> extractActivityKind val <> " status=" <> extractActivityStatus val)
  866           withStreamingState wsConns key (\ss -> handleActivity cfg chan ss val)
  867       | otherwise         -> do
  868           -- A background session's activity arrives on that session's own
  869           -- WS connection, but it can also be broadcast to this one. It
  870           -- must never reach the watch-all-tabs path: a bg turn is tracked
  871           -- by 'handleBgServerEvent', and a watch notification would emit
  872           -- a bogus "Tab ?" line plus a duplicate reply for every bg turn.
  873           isBg <- bgOwnsSession bgSessions key sid
  874           dbg ("[watch] SeActivity non-focused sid=" <> sessionIdText sid <> " focusedSid=" <> sessionIdText focusedSid <> " kind=" <> extractActivityKind val <> " status=" <> extractActivityStatus val <> " bg=" <> (if isBg then "true" else "false"))
  875           unless isBg $
  876             handleWatchActivity cfg chan key pendingAsks watchState thinkingTabs focusedSid sid val
  877     -- Tool-call events are BeActivity with kind="tool-call", broadcast
  878     -- by the server's aeOnToolCall hook. They arrive as SeActivity.
  879     -- handleActivity dispatches on kind internally.
  880     SeAsk sid val
  881       | sid == focusedSid -> handleAsk cfg chan key pendingAsks sid val
  882     -- A question answered/cancelled/timed out elsewhere (the web UI) is no
  883     -- longer answerable from this channel.
  884     SeAskResolved _ val -> forgetAskByPrefix pendingAsks (extractAskId val)
  885     SeLists val -> handleLists cfg chan key tabTracker watchState thinkingTabs val
  886     _ -> pure ()  -- ignore events for other sessions or irrelevant types

 1013 -- | A non-focused watched session just went idle. That single activity event
 1014 -- covers two very different situations, and the activity itself cannot tell
 1015 -- them apart:
 1016 --
 1017 -- * the turn finished normally, or
 1018 -- * the turn is blocked waiting for an ASK_HUMAN answer.
 1019 --
 1020 -- The gateway broadcasts @idle@ /before/ it blocks on the prompt
 1021 -- ('wrapCapsForAskStatus'), so a question may also be registered a moment
 1022 -- after the event. Ask the gateway for the session's pending questions, with a
 1023 -- short bounded re-check to cover that ordering, then send the one
 1024 -- notification that matches.
 1025 --
 1026 -- The pending-ask store doubles as the dedupe: a question already registered
 1027 -- for this session has been announced, so a later idle must not announce it
 1028 -- again. It is cleared by the normal resolve paths
 1029 -- ('SeAskResolved' -> 'forgetAskByPrefix', 'answerAsk' -> 'forgetAsk').
 1030 notifyWatchCompletion
 1031   :: ChatChannel c
 1032   => ChatChannelConfig -> c -> PendingAsks -> SessionId -> IO ()
 1033 notifyWatchCompletion cfg chan pendingAsks sid = do
 1034   mQ <- pollForQuestion cfg sid 5
 1035   case mQ of
 1036     Just q -> do

 1201 -- | Handle an @entry-update@ event: create or edit the streaming bubble.
 1202 handleEntryUpdate
 1203   :: ChatChannel c
 1204   => ChatChannelConfig -> c -> StreamingState -> Value -> IO ()
 1205 handleEntryUpdate cfg chan ss val = do
 1206   let streamCfg = cccStreamCfg cfg
 1207       text = extractEntryText val
 1208   unless (T.null text) $ do
 1209     -- Post-finalize guard (issue #198 follow-up): a late
 1210     -- entry-update can arrive AFTER the turn's recorded entry
 1211     -- (the server's streaming path and the post-turn broadcast
 1212     -- are unsynchronized). Editing now would overwrite the
 1213     -- finalized text and re-add the cursor. Ignore updates
 1214     -- entirely while the turn is finalized.
 1215     finalized <- readIORef (ssFinalized ss)
 1216     unless finalized $ do
 1217       -- The separate-messages style treats the accumulator as the live text
 1218       -- of the bubble it is growing, so it tracks the newest frame. The
 1219       -- single-bubble style must NOT: its accumulator holds the recorded
 1220       -- answers (the text of the NOTIFYING final message), and a frame
 1221       -- previews the same response, so writing it here duplicates the answer.
 1222       when (ccProgressStyle chan == PsSeparateMessages) $
 1223         writeIORef (ssAccumulated ss) text
 1224       -- Single-edited-bubble style: one message per turn, edited in place.
 1225       -- The frame text is the whole turn so far (the gateway's accumulator
 1226       -- is turn-scoped), so it is the sole body; tool lines are spliced into
 1227       -- it at their recorded offsets.
 1228       -- Live-edit only an existing progress bubble. Creating one here would
 1229       -- put the answer text in a bubble that is then edited rather than
 1230       -- sent, which does not notify - and a plain answer must notify. A
 1231       -- bubble exists only once a tool has run this turn.
 1232       --
 1233       -- The frame is also NOT folded into the answer accumulator: an
 1234       -- entry-update previews the same response its recorded entry will
 1235       -- carry, so accumulating both would duplicate the answer.
 1236       when (ccProgressStyle chan == PsSingleBubble) $ do
 1237         hasBubble <- isJust <$> readIORef (ssMsgId ss)
 1238         when hasBubble $ do
 1239           open <- editGateOpen cfg chan ss
 1240           when open $ do
 1241             edited <- updateBubble cfg chan ss
 1242               (spliceContent ss (Just (if T.null text then "" else addCursor (cccStreamCfg cfg) text)))
 1243             -- This branch only runs when a bubble exists, so it is always an
 1244             -- edit, never a create.
 1245             when edited (markBubbleEdited cfg ss)
 1246       -- Separate-messages style: stream a growing bubble per segment.
 1247       when (ccProgressStyle chan == PsSeparateMessages && ccSupportsStreaming chan) $ do
 1248         now <- cccNow cfg
 1249         mLastEdit <- readIORef (ssLastEdit ss)
 1250         mMsgId <- readIORef (ssMsgId ss)
 1251         -- Gate on NEW text since the last edit, not the total length
 1252         -- (issue #198, part 3). The accumulator only grows within a
 1253         -- response, so a total-length threshold degenerates to an edit
 1254         -- on EVERY frame once the text passes spcBufferThreshold —
 1255         -- one outbound platform edit per arriving WS frame (the
 1256         -- 'lots of small updates' flood). Measuring the delta since
 1257         -- the last edit keeps the cadence the config intends: an edit
 1258         -- per interval, or per 80 NEW codepoints, whichever first.
 1259         lastLen <- readIORef (ssLastLen ss)
 1260         when (shouldEdit streamCfg now mLastEdit (T.length text - lastLen)) $ do
 1261           let content = addCursor streamCfg text
 1262           case mMsgId of
 1263             Nothing -> do
 1264               mId <- ccSendWithId chan content
 1265               case mId of
 1266                 Just id' -> do
 1267                   writeIORef (ssMsgId ss) (Just id')
 1268                   writeIORef (ssLastEdit ss) (Just now)
 1269                   writeIORef (ssLastLen ss) (T.length text)
 1270                 Nothing -> pure ()
 1271             Just id' -> do
 1272               ok <- ccEditMessage chan id' content
 1273               when ok $ do
 1274                 writeIORef (ssLastEdit ss) (Just now)
 1275                 writeIORef (ssLastLen ss) (T.length text)

 1277 -- | Render the single-edited-bubble body for the state's committed data.
 1278 -- The text argument is what to use as the body when the state has no
 1279 -- accumulated frame text (the recorded entry, for a turn that never
 1280 -- streamed); otherwise the accumulated frame text wins, because it is the
 1281 -- whole turn so far and therefore the most current view.
 1282 spliceContent :: StreamingState -> Maybe Text -> IO Text
 1283 spliceContent ss mPreview = do
 1284   splices <- readIORef (ssSplices ss)
 1285   accum <- readIORef (ssAccumulated ss)

 1288 -- | Create the turn's bubble on first use, then edit it. Returns 'True' when
 1289 -- the content is on the platform (created or edited).
 1290 --
 1291 -- A failed EDIT deliberately does NOT fall back to sending a fresh message:
 1292 -- the answer is delivered separately by 'finalizeSingleBubble', the body here
 1293 -- is tool progress only, and a fresh send would both notify (the thing this
 1294 -- style exists to avoid) and repeat on every later event. The next frame,
 1295 -- tool call or idle re-renders the whole body from 'ssSplices' /
 1296 -- 'ssAccumulated', which an edit never modifies - so nothing is lost.
 1297 -- | The single-bubble edit gate. One shared bucket per conversation, so the
 1298 -- frame and tool sources together cannot exceed one edit per interval.
 1299 -- Interval-only: the separate-messages path's 'shouldEdit' is an @||@ whose
 1300 -- 80-codepoint branch fires on every frame, which is unbounded.
 1301 editGateOpen
 1302   :: ChatChannel c
 1303   => ChatChannelConfig -> c -> StreamingState -> IO Bool
 1304 editGateOpen cfg chan ss = do
 1305   now <- cccNow cfg
 1306   mLast <- readIORef (ssLastBubbleEdit ss)
 1307   let intervalMs = ccEditIntervalMs chan
 1308       elapsedMs = case mLast of
 1309         Nothing  -> intervalMs          -- no edit yet this turn: edit now
 1310         Just last' ->
 1311           round (realToFrac (diffUTCTime now last') * (1000 :: Double))
 1312   pure (elapsedMs >= intervalMs)

 1321 updateBubble
 1322   :: ChatChannel c
 1323   => ChatChannelConfig -> c -> StreamingState -> IO Text -> IO Bool
 1324 updateBubble cfg chan ss mkContent = do
 1325   content <- mkContent
 1326   over <- shouldFallBackToChunks cfg chan content
 1327   if over then pure False else do
 1328       mMsgId <- readIORef (ssMsgId ss)
 1329       case mMsgId of
 1330         Nothing -> do
 1331           -- Prefer a SILENT create: a turn's progress should be visible
 1332           -- without pinging the human, so a tool turn notifies exactly once
 1333           -- (the answer) rather than once for the bubble too. A channel that
 1334           -- cannot send silently returns 'Nothing' and we fall back to the
 1335           -- notifying send, so progress is never lost.
 1336           mSilent <- ccSendWithIdSilently chan content
 1337           mId <- case mSilent of
 1338             Just i  -> pure (Just i)
 1339             Nothing -> ccSendWithId chan content
 1340           for_ mId (writeIORef (ssMsgId ss) . Just)
 1341           pure (isJust mId)
 1342         Just id' -> do
 1343           ok <- ccEditMessage chan id' content
 1344           unless ok $ do
 1345             -- Do NOT send a fresh message. Three reasons, all load-bearing:
 1346             --
 1347             -- * the answer is delivered separately by 'finalizeSingleBubble',
 1348             --   and this body is tool progress only, so nothing is lost;
 1349             -- * a fresh send NOTIFIES, which is the very thing this style
 1350             --   exists to avoid;
 1351             -- * the returned id used to be discarded, so 'ssMsgId' kept
 1352             --   naming the dead message and every later tool call re-sent the
 1353             --   entire body - an unbounded notifying loop, not one duplicate.
 1354             --
 1355             -- The next frame, tool call or idle re-renders the whole body from
 1356             -- 'ssSplices'/'ssAccumulated', which an edit never modifies.
 1357             dbg "[bubble] edit failed; leaving the bubble for the next event"
 1358           pure ok

 1360 -- | Merge one recorded response into the turn's answer.
 1361 --
 1362 -- Two things can happen, and they are easy to confuse:
 1363 --
 1364 -- * the same response arrives again / later text extends the previous one
 1365 --   (the gateway's accumulator is turn-scoped, so a second response's text can
 1366 --   already contain the first). The previous contribution is REPLACED, so the
 1367 --   overlap is not concatenated twice;
 1368 -- * a genuinely new segment arrives (a @StopMaxTokens@ continuation, which
 1369 --   does not contain the earlier text). It is APPENDED.
 1370 --
 1371 -- @prev@ is the text of the previous entry, already folded into @accum@; it is
 1372 -- what must be dropped before adding @next@.
 1373 mergeAnswer :: Text -> Text -> Text -> Text
 1374 mergeAnswer accum prev next
 1375   | T.null next = accum
 1376   | T.null accum = next
 1377   -- The new text extends / repeats the previous entry: swap the old
 1378   -- contribution for the new one.
 1379   | not (T.null prev), prev `T.isPrefixOf` next = dropSuffix prev accum <> next
 1380   -- A fresh segment.
 1381   | otherwise = accum <> "\n\n" <> next
 1382   where
 1383     dropSuffix suf t = case T.stripSuffix suf t of
 1384       Just before -> T.dropWhileEnd (== '\n') before
 1385       Nothing     -> t

 1393 -- | Deliver the turn's final body. Normally that is one more edit of the
 1394 -- bubble. When the body has outgrown a single platform message there is no
 1395 -- message left to edit, so the text is sent once through the channel's
 1396 -- chunking path (the same path every ordinary long reply takes).
 1397 --
 1398 -- Deliberately called only at turn end: sending from the overflow branch of
 1399 -- 'updateBubble' would re-send the whole body on every frame, entry and
 1400 -- idle - one full-text delivery per event instead of one per turn.
 1401 finalizeSingleBubble
 1402   :: ChatChannel c
 1403   => ChatChannelConfig -> c -> StreamingState -> IO ()
 1404 finalizeSingleBubble cfg chan ss = do
 1405   -- 1. Close the progress bubble: strip the cursor so a half-typed preview
 1406   --    does not linger. The bubble holds tool progress only.
 1407   mBubble <- readIORef (ssMsgId ss)
 1408   let hadBubble = isJust mBubble
 1409   when (isJust mBubble) $ do
 1410     body <- spliceContent ss (Just "")
 1411     over <- shouldFallBackToChunks cfg chan body
 1412     if over
 1413       then do
 1414         -- The body outgrew one message. Abandoning the bubble mid-turn used
 1415         -- to leave a truncated prefix frozen on screen WITH a streaming
 1416         -- cursor, which reads as "still working" forever. Replace it with one
 1417         -- deliberate closing line instead; the full text arrives below via
 1418         -- the chunked send path.
 1419         dbg ("[bubble] progress body over limit ("
 1420              <> T.pack (show (T.length body)) <> " chars); closing it")
 1421         case mBubble of
 1422           Just id' -> void (ccEditMessage chan id' overLimitCloseText)
 1423           Nothing  -> pure ()
 1424         writeIORef (ssMsgId ss) Nothing
 1425       else void (updateBubble cfg chan ss (pure body))
 1426   -- 2. Deliver the answer as its own message. This is what notifies the
 1427   --    human: an edit never does.
 1428   answer <- stripCursorText (cccStreamCfg cfg) <$> readIORef (ssAccumulated ss)
 1429   dbg ("[bubble] delivering final answer as a new message ("
 1430        <> T.pack (show (T.length answer)) <> " chars)")
 1431   if T.null (T.strip answer)
 1432     then
 1433       -- A tools-only turn has no answer to deliver. Without this it would
 1434       -- notify ZERO times - the bubble is silent and edits are silent - which
 1435       -- would break the rule that the last message before idle notifies.
 1436       -- Send a short close notice so every turn notifies exactly once.
 1437       when hadBubble $ ccSend chan closeNoticeText
 1438     else ccSend chan answer

 1458 -- without the cursor (or, if the edit fails, send the full text as a new
 1459 -- message), then reset the bubble state so the next entry-update starts

 1472     Nothing -> isJust <$> ccSendWithId chan finalText

 1489   let direction = extractDirection val
 1490   when (direction == "response") $ do

 1531 -- | Handle a @harness-status@ activity: on @thinking@ (turn start),
 1532 -- clear the finalized flag so the new turn streams. On @idle@ (turn
 1533 -- end), finalize any in-progress streaming bubble with the accumulated
 1534 -- text (a safety net — the recorded entry is the normal finalize path)
 1535 -- and reset all streaming state.
 1536 handleHarnessStatus
 1537   :: ChatChannel c
 1538   => ChatChannelConfig -> c -> StreamingState -> Value -> IO ()
 1539 handleHarnessStatus cfg chan ss val =
 1540   case extractActivityStatus val of
 1541     "thinking" ->
 1542       -- Turn start: clear the finalized flag so the new turn streams.
 1543       writeIORef (ssFinalized ss) False
 1544     "idle" -> do
 1545       case ccProgressStyle chan of
 1546         -- Single bubble: a final edit with the cursor stripped. The recorded
 1547         -- entry already committed the exact text, so this only clears the
 1548         -- cursor and closes the turn; a withheld idle (running background
 1549         -- children) leaves complete, un-cursored text behind.
 1550         PsSingleBubble -> do
 1551           alreadyFinal <- readIORef (ssFinalized ss)
 1552           unless alreadyFinal $
 1553             finalizeSingleBubble cfg chan ss
 1554         PsSeparateMessages -> do
 1555           mMsgId <- readIORef (ssMsgId ss)
 1556           accum <- readIORef (ssAccumulated ss)
 1557           -- For non-streaming channels the accumulated text is the full
 1558           -- response. Send it as a new message (mMsgId is Nothing because no
 1559           -- streaming bubble was created). For streaming channels, only
 1560           -- finalize if there's an active streaming bubble (mMsgId is Just).
 1561           when (not (T.null accum) &&
 1562                 (not (ccSupportsStreaming chan) || isJust mMsgId)) $
 1563             finalizeBubble cfg chan ss accum False
 1564       resetStreamingState ss
 1565       writeIORef (ssFinalized ss) True
 1566     _ -> pure ()

 1568 -- | Handle a @tool-call@ activity: finalize the current text bubble
 1569 -- (segment break — the pre-tool text stays its own message, and the
 1570 -- next entry-update starts a NEW bubble below the tool line), then send
 1571 -- the tool-progress line as its own platform message.
 1572 handleToolCallActivity
 1573   :: ChatChannel c
 1574   => ChatChannelConfig -> c -> StreamingState -> Value -> IO ()
 1575 handleToolCallActivity cfg chan ss val = do
 1576   -- Segment break (issue #198 follow-up, symptom 1): the pre-tool
 1577   -- streamed text and the post-tool streamed text must be SEPARATE
 1578   -- platform messages, with the tool line between them. Without this,
 1579   -- the next entry-update edits the pre-tool bubble (ssMsgId is still
 1580   -- set) and appends the post-tool text onto the pre-tool text.
 1581   case ccProgressStyle chan of
 1582     PsSingleBubble -> case extractToolName val of
 1583       Nothing -> pure ()
 1584       Just toolName -> do
 1585         -- Record the tool line with the offset it fired at, then re-render
 1586         -- the one bubble. This keeps the line in its true place in the
 1587         -- turn's text stream instead of appending it at the end, and it
 1588         -- produces no new notification.
 1589         accum <- readIORef (ssAccumulated ss)
 1590         let mInput = extractToolInput val
 1591             line = formatToolLine mempty toolName (fromMaybe "" mInput)
 1592         modifyIORef' (ssSplices ss) (<> [(T.length accum, line)])
 1593         -- Gated on the SAME bucket as the frame path. The first tool call of a
 1594         -- turn is the bubble's create ('hasBubble' is false, so this branch's
 1595         -- edit creates it via 'updateBubble'); a create does not consume the
 1596         -- bucket, so the first real edit after it lands immediately.
 1597         hadBubble <- isJust <$> readIORef (ssMsgId ss)
 1598         open <- editGateOpen cfg chan ss
 1599         when open $ do
 1600           edited <- updateBubble cfg chan ss (spliceContent ss (Just ""))
 1601           -- A CREATE is not an edit: it does not consume the bucket, so the
 1602           -- next event can edit immediately and the progress becomes visible
 1603           -- straight away.
 1604           when (edited && hadBubble) (markBubbleEdited cfg ss)
 1605     PsSeparateMessages -> do
 1606       mMsgId <- readIORef (ssMsgId ss)
 1607       accum <- readIORef (ssAccumulated ss)
 1608       -- For non-streaming channels, finalize when there's accumulated text
 1609       -- even without a streaming bubble (mMsgId is Nothing). For streaming
 1610       -- channels, only finalize when there's an in-progress bubble.
 1611       when (not (T.null accum) &&
 1612             (not (ccSupportsStreaming chan) || isJust mMsgId)) $
 1613         finalizeBubble cfg chan ss accum False
 1614       case extractToolName val of
 1615         Nothing -> pure ()
 1616         Just toolName -> do
 1617           let mInput = extractToolInput val
 1618               line = formatToolLine mempty toolName (fromMaybe "" mInput)
 1619           ccSend chan line

 1621 -- | Handle an @ask@ event: render the question on the platform. When the
 1622 -- ask has options and the channel supports inline keyboards, sends the
 1623 -- question + an inline keyboard (one button per option). Otherwise falls
 1624 -- back to the numbered-list text rendering. Registers the pending ask so
 1625 -- the callback handler can resolve a button tap to the option label.
 1626 -- The ask is registered even when it offers no options (an open-ended
 1627 -- 'ASK_HUMAN', or the confirmation gate's bare prompt), so it can still be
 1628 -- resolved and dequeued instead of leaking into the pending map.
 1629 handleAsk
 1630   :: ChatChannel c
 1631   => ChatChannelConfig -> c -> ConversationKey -> PendingAsks -> SessionId -> Value
 1632   -> IO ()
 1633 handleAsk _cfg chan _key pendingAsks sid val = do
 1634   let question = extractAskQuestion val
 1635       askId = extractAskId val
 1636       opts = extractAskOptions val
 1637   -- One renderer for both shapes, so an open-ended ask carries a
 1638   -- "Reply with your answer." hint instead of arriving as a bare question
 1639   -- the human cannot tell is waiting on them. A channel with inline

 1642   if null opts
 1643     then ccSend chan (formatAsk question [])
 1644     else do
 1645       let prefix = T.take 8 askId

 1647       case mMsgId of
 1648         Just _  -> pure ()

 1657       pure n

 1700 -- | Handle /new: mint a fresh session and make it the conversation's
 1701 -- focused session.
 1702 --
 1703 -- Exactly ONE mint happens per @/new@:
 1704 --
 1705 --   * the conversation already has a session ⇒ @POST \/api\/sessions\/:id\/new@
 1706 --     ('cccRebindNew'), which mints AND rebinds the tab in one call;
 1707 --   * no prior session ⇒ @POST \/api\/sessions\/new@ ('cccNewSession').
 1708 --
 1709 -- Minting on both paths would create two sessions and reproduce the exact
 1710 -- orphan-session bug this work exists to fix.
 1711 --
 1712 -- The arguments are the CLI's (@-p\/--provider@, @-m\/--model@,
 1713 -- @-r\/--repo@, @-b\/--branch@), parsed by the shared 'parseNewArgs' so chat
 1714 -- and CLI accept the same flags. @-b@ only qualifies @-r@: without a repo
 1715 -- there is nothing to check out, so no setup is dispatched. Message order:
 1716 -- rebind warning, then repo failure, then the confirmation.
 1717 handleNewSession
 1718   :: ChatChannel c
 1719   => ChatChannelConfig -> c -> SessionMap -> TVar (Map ConversationKey WsConn)
 1720   -> PendingAsks -> BgSessions -> TVar (Map ConversationKey (SessionId, [TabJson]))
 1721   -> WatchState -> ThinkingTabs
 1722   -> ConversationKey -> Text
 1723   -> IO ()
 1724 handleNewSession cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key args = do
 1725   let apiBase = gcApiBase (cccGateway cfg)
 1726       mgr = cccHttpManager cfg
 1727       na = parseNewArgs args
 1728   mPrior <- sessionLookup sessions key
 1729   eOutcome <- case mPrior of
 1730     Just oldSid -> do
 1731       -- Absent provider/model inherit the old session's (the gateway applies
 1732       -- that); the channel is this channel's label, so the new session stays
 1733       -- attributable to the medium it was created from.
 1734       eRes <- cccRebindNew cfg mgr apiBase oldSid
 1735         (rebindBody (naProvider na) (naModel na) (Just (ccLabel chan)))
 1736       pure $ fmap (\res -> NewOutcome
 1737         { noSid         = nsrSessionId res
 1738         , noRebindError = nsrRebindError res
 1739         , noOldSid      = Just (sessionIdText oldSid)
 1740         , noProvider    = Just (nsrProvider res)
 1741         , noModel       = Just (nsrModel res)
 1742         }) eRes
 1743     Nothing -> do
 1744       eSid <- cccNewSession cfg mgr apiBase (mintBody (naProvider na) (naModel na) (Just (ccLabel chan)))
 1745       pure $ fmap (\sidText -> NewOutcome
 1746         { noSid         = sidText
 1747         , noRebindError = Nothing
 1748         , noOldSid      = Nothing
 1749         , noProvider    = Nothing
 1750         , noModel       = Nothing
 1751         }) eSid
 1752   case eOutcome of
 1753     Left e -> ccSend chan ("new session failed: " <> e)
 1754     Right out -> case mkSessionId (noSid out) of
 1755       Left _ -> ccSend chan "new session failed: invalid session id"
 1756       Right sid -> do
 1757         sessionInsert sessions key sid
 1758         -- Re-target the conversation's WS connection on BOTH paths: without
 1759         -- this the new session's replies and asks are filtered out (WU-5).
 1760         ensureWsConn cfg chan wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key sid
 1761         -- A rebind failure is a warning, not a failure: the session exists.
 1762         for_ (noRebindError out) $ \e ->
 1763           ccSend chan ("warning: /new tab rebind failed: " <> e)
 1764         mRepoErr <- setupRepoFor cfg mgr apiBase sid (naRepo na) (naBranch na)
 1765         for_ mRepoErr $ \e -> ccSend chan ("repo setup failed: " <> e)
 1766         ccSend chan (newConfirmation out)

 1850 -- | Handle a callback query (button tap) from the platform. The callback_data
 1851 -- is @"<8hex>:<index>"@. Resolves the 8-hex prefix to a pending ask, resolves
 1852 -- the index to the option label, answers via the HTTP API, acknowledges the
 1853 -- callback (dismiss the spinner), removes the keyboard, and sends a
 1854 -- @✓ <label>@ confirmation. If no pending ask matches, the callback is
 1855 -- silently dropped (stale button tap).
 1856 handleCallback
 1857   :: ChatChannel c
 1858   => ChatChannelConfig -> c -> SessionMap -> PendingAsks -> BgSessions
 1859   -> ConversationKey -> MessageSource -> Text
 1860   -> IO ()
 1861 handleCallback cfg chan sessions pendingAsks bgSessions key _src cbData = do
 1862   case parseCallbackData cbData of
 1863     Nothing -> pure ()  -- not a valid callback_data format
 1864     Just (prefix, idx) -> do
 1865       -- Only asks owned by this conversation are answerable: its focused
 1866       -- session, or one of its own background sessions. A tap in another
 1867       -- conversation can never resolve them.
 1868       sids <- answerableSids sessions bgSessions key
 1869       asksMap <- readTVarIO (paStore pendingAsks)
 1870       let matches =
 1871             [ (sid, ask)
 1872             | sid <- sids
 1873             , Just asks <- [Map.lookup sid asksMap]
 1874             , ask <- asks
 1875             , T.isPrefixOf prefix (paAskId ask)
 1876             ]
 1877       case matches of
 1878         [] -> pure ()
 1879         ((sid, ask) : _) ->
 1880           case atIndex (paOptions ask) idx of
 1881             Nothing -> pure ()
 1882             Just opt -> answerAsk cfg chan pendingAsks sid (paAskId ask) (qoLabel opt)

 1941 -- | Answer a pending ask belonging to one of this conversation's background
 1942 -- sessions with an inbound plain-text message. Returns 'True' when the
 1943 -- message was consumed as an answer (so it must NOT be routed to the
 1944 -- focused session).
 1945 --
 1946 -- Background sessions only, deliberately. The conversation's OWN (focused)
 1947 -- session is not consulted here: a plain message for it is routed to the
 1948 -- gateway, whose @POST \/api\/sessions\/:id\/send@ offers the text to that
 1949 -- session's pending ask before starting a turn. That keeps the answer path
 1950 -- working when the loop never saw the @ask@ event (a reconnect or a channel
 1951 -- restart) and guarantees the reply never queues behind the session lock the
 1952 -- blocked turn holds.
 1953 --
 1954 -- A numeric body (1-based) selects the corresponding option label; any other
 1955 -- body is delivered verbatim -- which is what the confirmation gate needs
 1956 -- (@y@/@yes@ approve once, anything else rejects). The oldest pending ask
 1957 -- wins: bg sessions are scanned in creation order, and each session's list
 1958 -- is newest-first ('handleAsk' prepends), so the oldest ask for a session is
 1959 -- its last element.
 1960 answerPendingBgAsk
 1961   :: ChatChannel c
 1962   => ChatChannelConfig -> c -> PendingAsks -> BgSessions -> ConversationKey -> Text
 1963   -> IO Bool
 1964 answerPendingBgAsk cfg chan pendingAsks bgSessions key body = do
 1965   bg <- bgSessionsFor bgSessions key
 1966   asksMap <- readTVarIO (paStore pendingAsks)
 1967   let bgIds = map bgSid bg
 1968       -- A question raised on a non-focused tab never reaches 'handleAsk' (the
 1969       -- broker focus-filters asks), so the watch path registers it with
 1970       -- 'paWatched' set. Those are answerable here too — otherwise the
 1971       -- notification could be read but not replied to.
 1972       watched =
 1973         [ (sess, ask)
 1974         | (sess, asks) <- Map.toList asksMap
 1975         , sess `notElem` bgIds
 1976         , Just ask <- [earliestRegistered (filter paWatched asks)]
 1977         ]
 1978   -- Per session the earliest-registered ask is the list tail ('handleAsk'
 1979   -- prepends); across sessions 'paSeq' orders them, which session order
 1980   -- cannot.
 1981   let candidates =
 1982         [ (bgSid sess, ask)
 1983         | sess <- bg
 1984         , Just asks <- [Map.lookup (bgSid sess) asksMap]
 1985         , Just ask <- [earliestRegistered asks]
 1986         ] <> watched
 1987   case oldestAcross candidates of
 1988     Nothing -> pure False
 1989     Just (sid, ask) -> do
 1990       answerAsk cfg chan pendingAsks sid (paAskId ask) (resolveAnswer (paOptions ask) body)
 1991       pure True
 1992 
 1993 -- | The earliest-registered ask of all candidates (ties keep list order).
 1994 oldestAcross :: [(SessionId, PendingAsk)] -> Maybe (SessionId, PendingAsk)
 1995 oldestAcross = foldl' pick Nothing
 1996   where
 1997     pick Nothing x = Just x
 1998     pick (Just best) x
 1999       | paSeq (snd x) < paSeq (snd best) = Just x
 2000       | otherwise                        = Just best

 2002 -- | The earliest-registered ask in one session's list (ties keep list order).
 2003 earliestRegistered :: [PendingAsk] -> Maybe PendingAsk
 2004 earliestRegistered = foldl' pick Nothing
 2005   where
 2006     pick Nothing ask = Just ask
 2007     pick (Just best) ask
 2008       | paSeq ask < paSeq best = Just ask
 2009       | otherwise              = Just best

 2028 -- | Handle a @lists@ WS event: compare the new tab list with the last known
 2029 -- one for this conversation. If the focused session's tab was removed (closed),
 2030 -- send the "tab closed" notification and clear the tracking state so the next
 2031 -- message creates a fresh tab. Also seeds the 'ThinkingTabs' set from the
 2032 -- snapshot's @thinkingSessionIds@ so tabs already thinking before the WS
 2033 -- connection was established still trigger a finished notification.
 2034 handleLists
 2035   :: ChatChannel c
 2036   => ChatChannelConfig -> c -> ConversationKey
 2037   -> TVar (Map ConversationKey (SessionId, [TabJson]))
 2038   -> WatchState -> ThinkingTabs -> Value -> IO ()
 2039 handleLists cfg chan key tabTracker watchState thinkingTabs val = do
 2040   -- Seed the thinking-tabs set from the snapshot's thinkingSessionIds.
 2041   -- This is the core fix for the "missed thinking event" problem: when a
 2042   -- WS connection is established (or a lists snapshot is broadcast on tab
 2043   -- changes), sessions that were already thinking before the connection
 2044   -- existed are included in the snapshot. Without this seeding, the
 2045   -- ThinkingTabs set starts empty and the thinking→idle transition is
 2046   -- never detected for those sessions (the idle event arrives but
 2047   -- wasThinking is false → no notification).
 2048   seedThinkingTabsFromLists watchState thinkingTabs key val
 2049   -- Fetch the current tabs from the gateway to get an accurate list.
 2050   let apiBase = gcApiBase (cccGateway cfg)
 2051       mgr = cccHttpManager cfg
 2052   eTabs <- httpGetTabs mgr apiBase
 2053   case eTabs of
 2054     Left _ -> pure ()
 2055     Right newTabs -> do
 2056       tracker <- readTVarIO tabTracker
 2057       case Map.lookup key tracker of
 2058         Nothing -> pure ()  -- no tracked session for this conversation
 2059         Just (focusedSid, _oldTabs) -> do
 2060           -- Check if the focused session's tab was removed.
 2061           let newSessionIds = mapMaybe tjSessionId newTabs

 2065               atomically (modifyTVar' tabTracker (Map.insert key (focusedSid, newTabs)))

 2068               ccSend chan ("tab closed (session " <> sessionIdText focusedSid
 2069                         <> "); a new tab will be created on your next message")
 2070               -- Clear the tracking state so the next message creates a fresh tab.
 2071               atomically (modifyTVar' tabTracker (Map.delete key))
 2072   where
 2073     mkSessionId' t = case mkSessionId t of Right s -> Just s; Left _ -> Nothing

 2075 -- | Seed the 'ThinkingTabs' set for a conversation from the
 2076 -- @thinkingSessionIds@ field in a @lists@ WS event payload. This ensures
 2077 -- sessions that were already thinking before the WS connection was
 2078 -- established are tracked, so their eventual @idle@ transition triggers a
 2079 -- watch notification. Replaces the per-conversation thinking set (rather

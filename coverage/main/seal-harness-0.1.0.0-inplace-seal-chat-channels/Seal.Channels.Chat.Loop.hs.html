  166 -- | Per-conversation watch-all-tabs toggle state. When 'True' for a
  167 -- conversation, the channel receives a notification every time any
  168 -- non-focused tab finishes thinking (harness-status transitions from
  169 -- @thinking@ to @idle@). Thread-safe via 'TVar'.
  170 data WatchState = WatchState
  171   { wsVar  :: TVar (Map ConversationKey Bool)
  172   , wsSave :: Maybe (IO ())
  173   -- ^ The save action (snapshots the TVar and writes to disk). Called
  174   -- inside the 'wsLock' critical section so the snapshot is consistent
  175   -- with the mutation.
  176   , wsLock :: MVar ()
  177   -- ^ Serializes the mutation+persist sequence so a stale snapshot from
  178   -- one thread cannot overwrite a newer on-disk write from another.
  179   -- Without this lock, concurrent toggles from Signal + Telegram (which
  180   -- share one 'WatchState') can lose state after a restart.
  181   }
  182 
  183 -- | Create a new empty watch-state map (watch mode off for all
  184 -- conversations).
  185 newWatchState :: IO WatchState
  186 newWatchState = do
  187   tv <- newTVarIO Map.empty
  188   lock <- newMVar ()
  189   pure (WatchState tv Nothing lock)
  190 
  191 -- | Create a persisting watch-state store. The supplied save function is
  192 -- called after every mutation (snapshotting the full current map, so the
  193 -- last writer wins with a consistent view). Mirrors
  194 -- 'Seal.Channels.Cursor.newPersistingCursorStore'.
  195 newPersistingWatchState :: (Map ConversationKey Bool -> IO ()) -> IO WatchState
  196 newPersistingWatchState saveFn = do
  197   tv <- newTVarIO Map.empty
  198   lock <- newMVar ()
  199   let ws = WatchState tv (Just (saveAction ws)) lock
  200       saveAction s = saveFn =<< snapshotWatch s
  201   pure ws
  202 
  203 -- | Replace the store's map in one STM transaction. Used at boot to seed
  204 -- the in-memory store from the persisted @watch_state.json@. Does NOT
  205 -- persist (the caller is loading FROM disk).
  206 seedWatchState :: WatchState -> Map ConversationKey Bool -> IO ()
  207 seedWatchState ws m = atomically (writeTVar (wsVar ws) m)
  208 
  209 -- | Snapshot the current map. Used by the save action.
  210 snapshotWatch :: WatchState -> IO (Map ConversationKey Bool)
  211 snapshotWatch ws = readTVarIO (wsVar ws)

  213 -- | Look up whether watch mode is enabled for a conversation. 'False'
  214 -- when the conversation has no entry (the default).
  215 lookupWatch :: WatchState -> ConversationKey -> IO Bool
  216 lookupWatch ws key = fromMaybe False . Map.lookup key <$> readTVarIO (wsVar ws)
  217 
  218 -- | Toggle watch mode for a conversation. Returns the new state.
  219 toggleWatch :: WatchState -> ConversationKey -> IO Bool
  220 toggleWatch ws key = do
  221   -- Hold the lock through mutation+persist so no concurrent thread can
  222   -- mutate the TVar between our snapshot and our save (which would let a
  223   -- stale snapshot overwrite a newer on-disk write).
  224   withMVar (wsLock ws) $ \_ -> do
  225     newVal <- atomically $ do
  226       m <- readTVar (wsVar ws)
  227       let v = not (fromMaybe False (Map.lookup key m))
  228       writeTVar (wsVar ws) (Map.insert key v m)
  229       pure v
  230     persistWatch ws
  231     pure newVal
  232 
  233 -- | Run the persist action (if any) after a successful mutation. A save
  234 -- failure is logged to stderr and swallowed — the in-memory store stays
  235 -- authoritative within the session; the next successful mutation will
  236 -- retry the save (writing the full current map, so a missed save
  237 -- self-heals). Mirrors 'Seal.Channels.Cursor.persistCursor'.
  238 persistWatch :: WatchState -> IO ()
  239 persistWatch ws =
  240   case wsSave ws of
  241     Nothing  -> pure ()
  242     Just act -> act `catch` \e ->
  243       dbg ("[watch] watch_state.json save failed: " <> T.pack (show (e :: SomeException)))

  269 -- | The per-conversation background-session store. Each conversation's
  270 -- list is in creation order (oldest first) so the per-conversation cap can
  271 -- evict the oldest session when a new one is started. Thread-safe via
  272 -- 'TVar'.
  273 --
  274 -- The store also remembers the session ids this conversation has owned for
  275 -- a bounded window after they are retired ('bsKnown'). That memory is what
  276 -- keeps a bg turn out of the watch-all-tabs path: 'BeActivity' is broadcast
  277 -- to every subscriber, so the conversation's own connection receives
  278 -- @harness-status idle@ / @tool-call@ for the bg session too -- possibly
  279 -- after the bg connection's reader has already retired it. Without the
  280 -- memory those late events would be classified as "not a bg session" and
  281 -- produce a bogus "Tab ?" watch notification plus a duplicate reply.
  282 data BgSessions = BgSessions
  283   { bsLive  :: TVar (Map ConversationKey [BgSession])
  284   , bsKnown :: TVar (Map ConversationKey [SessionId])
  285   }
  286 
  287 -- | How many retired bg session ids a conversation remembers for the watch
  288 -- exclusion. Bounded so a long-lived conversation cannot grow without
  289 -- limit; the window only has to outlive one turn's activity broadcasts.
  290 maxKnownBgSids :: Int
  291 maxKnownBgSids = 32
  292 
  293 -- | Create a new empty background-session store.
  294 newBgSessions :: IO BgSessions
  295 newBgSessions = BgSessions <$> newTVarIO Map.empty <*> newTVarIO Map.empty

  297 -- | The conversation's live background sessions, oldest first.
  298 bgSessionsFor :: BgSessions -> ConversationKey -> IO [BgSession]
  299 bgSessionsFor bgs key =
  300   fromMaybe [] . Map.lookup key <$> readTVarIO (bsLive bgs)

  302 -- | Does @sid@ belong to one of this conversation's background sessions
  303 -- (live, or retired within the remembered window)? Used to keep bg turns
  304 -- out of the watch-all-tabs path.
  305 bgOwnsSession :: BgSessions -> ConversationKey -> SessionId -> IO Bool
  306 bgOwnsSession bgs key sid =
  307   elem sid . fromMaybe [] . Map.lookup key <$> readTVarIO (bsKnown bgs)
  308 
  309 -- | Register a background session at the end of its conversation's list
  310 -- (creation order) and remember its id for the watch exclusion.
  311 bgInsert :: BgSessions -> ConversationKey -> BgSession -> IO ()
  312 bgInsert bgs key sess = do
  313   atomically (modifyTVar' (bsLive bgs)
  314     (Map.insertWith (flip (++)) key [sess]))
  315   atomically (modifyTVar' (bsKnown bgs)
  316     (Map.insertWith (\new old -> take maxKnownBgSids (new ++ filter (`notElem` new) old))
  317        key [bgSid sess]))
  318 
  319 -- | Untrack a background session. Returns the session when it was tracked,
  320 -- 'Nothing' when it was already gone (so retirements are idempotent).
  321 -- Retiring does NOT forget the session id (see 'bsKnown'): late activity
  322 -- events for it still must stay out of the watch path.
  323 bgRemove :: BgSessions -> ConversationKey -> SessionId -> IO (Maybe BgSession)
  324 bgRemove bgs key sid = atomically $ do
  325   let tv = bsLive bgs
  326   m <- readTVar tv
  327   let sessions = fromMaybe [] (Map.lookup key m)
  328       (dropped, kept) = partition ((== sid) . bgSid) sessions
  329       m' = case kept of
  330              [] -> Map.delete key m
  331              _  -> Map.insert key kept m
  332   writeTVar tv m'
  333   pure (case dropped of
  334           (s : _) -> Just s
  335           []      -> Nothing)

  337 -- | Drop every pending ask registered for a session. A retired bg session
  338 -- can no longer be answered, so its asks must not accumulate (or be
  339 -- resolvable by a later button tap).
  340 dropPendingAsks :: PendingAsks -> SessionId -> IO ()
  341 dropPendingAsks pendingAsks sid =
  342   atomically (modifyTVar' (paStore pendingAsks) (Map.delete sid))

  431 -- | Run the generic chat-channel loop. Blocks until the channel's
  432 -- 'ccReceive' returns EOF. Each conversation gets its own WS connection
  433 -- for streaming.
  434 runChatChannel :: ChatChannel c => ChatChannelConfig -> c -> WatchState -> IO ()
  435 runChatChannel cfg chan watchState = do
  436   sessions <- newSessionMap
  437   -- Map of conversation keys to their WS client + streaming state.
  438   wsConns <- newTVarIO Map.empty :: IO (TVar (Map ConversationKey WsConn))
  439   pendingAsks <- newPendingAsks
  440   tabTracker <- newTVarIO Map.empty :: IO (TVar (Map ConversationKey (SessionId, [TabJson])))
  441   thinkingTabs <- newThinkingTabs
  442   bgSessions <- newBgSessions
  443   loop sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs
  444   where
  445     loop sessions wsConns pendingAsks bgSessions tabTracker ws tt = do
  446       mMsg <- ccReceive chan
  447       case mMsg of
  448         Nothing -> pure ()  -- EOF
  449         Just (InboundMessage src body mCbData) -> do
  450           dbg ("received: " <> body)
  451           let key = convKeyFromSource src
  452           case mCbData of
  453             Just cbData -> do
  454               -- Callback query (button tap): resolve to a pending ask,
  455               -- answer via HTTP, acknowledge, and remove keyboard.
  456               handleCallback cfg chan sessions pendingAsks bgSessions key src cbData
  457             Nothing ->
  458               -- Regular text message: route normally.
  459               handleInboundWithSource cfg chan sessions wsConns pendingAsks bgSessions tabTracker ws tt key (Just src) body
  460           loop sessions wsConns pendingAsks bgSessions tabTracker ws tt

  462 -- | 'handleInbound' with no channel provenance (for the tests and any caller
  463 -- that has no 'MessageSource'). The live loop calls
  464 -- 'handleInboundWithSource' with the inbound message's source.
  465 handleInbound
  466   :: ChatChannel c
  467   => ChatChannelConfig -> c -> SessionMap

  474   -> ConversationKey -> Text

  476 handleInbound cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key =

  479 -- | Handle one inbound message with its authenticated channel provenance:
  480 -- resolve the session, route, and dispatch. The source (when present) rides
  481 -- along to the HTTP send so the gateway stamps the channel + user id into the
  482 -- turn's transcript metadata.
  483 handleInboundWithSource
  484   :: ChatChannel c
  485   => ChatChannelConfig -> c -> SessionMap
  486   -> TVar (Map ConversationKey WsConn)
  487   -> PendingAsks
  488   -> BgSessions
  489   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  490   -> WatchState
  491   -> ThinkingTabs
  492   -> ConversationKey -> Maybe MessageSource -> Text
  493   -> IO ()
  494 handleInboundWithSource cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key mSrc body = do
  495   let apiBase = gcApiBase (cccGateway cfg)
  496       mgr = cccHttpManager cfg
  497   -- Intercept /watch (and /watch on|off) before normal routing — it
  498   -- toggles per-conversation loop state, not a gateway command.
  499   if isWatchCommand body
  500     then do
  501       dbg ("[watch] /watch command received, key=" <> ckConv key)
  502       handleWatchToggle chan watchState key body
  503       -- When watch mode is turned ON, ensure a WS connection exists so
  504       -- the channel receives BeActivity events for all tabs. Without
  505       -- this, /watch as the first message would enable watch mode but
  506       -- never receive any events (no WS connection = no event source).
  507       watchOn <- lookupWatch watchState key
  508       dbg ("[watch] watchOn=" <> (if watchOn then "true" else "false") <> ", ensuring WS conn")
  509       when watchOn $
  510         void (resolveSession cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key)
  511     else case parseTabFocus body of
  512     Just idx -> do
  513       handleFocus cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key idx
  514     Nothing -> case route body of
  515       Right (ChatFocus idx) ->
  516         handleFocus cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key idx
  517       Right (ChatInject idx payload) -> do
  518         -- Send the payload to the target tab's session WITHOUT changing
  519         -- the focused tab (no FocusOp, no "focused tab N" confirmation).
  520         handleInject cfg chan mSrc idx payload
  521       Right ChatCurrentTab -> do
  522         -- Send the current tab info via HTTP (the gateway routes /tab).
  523         mSid <- sessionLookup sessions key
  524         case mSid of
  525           Just sid -> do
  526             eResult <- cccSendMessage cfg mgr apiBase sid Nothing "/tab"
  527             case eResult of
  528               Right sr -> ccSend chan (srResponse sr)
  529               Left e -> ccSend chan ("error: " <> e)
  530           Nothing -> ccSend chan "no current tab"
  531       Right (ChatNewSession args) -> do
  532         -- Create a new session via HTTP, update the session map.
  533         handleNewSession cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key args
  534       Right (ChatBackground prompt) -> do
  535         -- Run the prompt in a freshly minted headless session. The
  536         -- conversation's focused session and tab list are untouched.
  537         handleBackground cfg chan pendingAsks bgSessions key mSrc prompt
  538       Right (ChatSlash _cmd) ->
  539         sendSlash cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key mSrc body
  540       Right (ChatTabCommand _) ->
  541         -- Tab commands go through the HTTP API (the gateway routes them).
  542         sendSlash cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key mSrc body
  543       Right ChatTabList -> handleTabList cfg chan
  544       Right (ChatLast args) -> do
  545         mSid <- sessionLookup sessions key
  546         handleLast cfg chan mSid key args
  547       Right (ChatPlain text) -> do
  548         -- A plain message answers a pending ask from one of this
  549         -- conversation's background sessions, if there is one. Checked
  550         -- here (not before routing) so a /command is never consumed as an
  551         -- answer.
  552         -- (A plain message for the conversation's own session is forwarded
  553         -- to the gateway, which answers that session's pending ask itself —
  554         -- see 'answerPendingBgAsk'.)
  555         consumed <- answerPendingBgAsk cfg chan pendingAsks bgSessions key text
  556         unless consumed $
  557           sendPlain cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key mSrc text
  558       Left _ -> ccSend chan "error: invalid command"

  560 -- | Handle a focus command: resolve the tab index to a session id via
  561 -- @GET /api/tabs@, send a @FocusOp@ over WS, send "focused tab N" to the
  562 -- platform, and optionally send the last assistant reply.
  563 handleFocus
  564   :: ChatChannel c
  565   => ChatChannelConfig -> c -> SessionMap
  566   -> TVar (Map ConversationKey WsConn)
  567   -> PendingAsks
  568   -> BgSessions
  569   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  570   -> WatchState
  571   -> ThinkingTabs
  572   -> ConversationKey -> TabIndex
  573   -> IO ()
  574 handleFocus cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key idx = do
  575   let apiBase = gcApiBase (cccGateway cfg)
  576       mgr = cccHttpManager cfg
  577   eTabs <- httpGetTabs mgr apiBase
  578   case eTabs of
  579     Left e -> ccSend chan ("focus failed: " <> e)
  580     Right tabs ->
  581       case [ t | t <- tabs, tjIndex t == tabIndexToInt idx ] of
  582         [] -> ccSend chan "focus failed: tab index out of range"
  583         (tab : _) ->
  584           case tjSessionId tab of
  585             Nothing -> ccSend chan "focus failed: tab has no session"
  586             Just sidText -> case mkSessionId sidText of
  587               Left _ -> ccSend chan "focus failed: invalid session id"
  588               Right sid -> do
  589                 -- Update the session map so subsequent messages route here.
  590                 sessionInsert sessions key sid
  591                 -- Track the focused session + current tab list for tab-close detection.
  592                 atomically (modifyTVar' tabTracker (Map.insert key (sid, tabs)))
  593                 -- Ensure a WS connection exists for this conversation.
  594                 ensureWsConn cfg chan wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key sid
  595                 -- Send "focused tab N" confirmation.
  596                 ccSend chan ("focused tab " <> T.singleton (tabIndexToChar idx))
  597                 -- Then the same block /last would give for this tab, so the
  598                 -- user lands knowing what state it is in. LastEmpty and a
  599                 -- fetch failure stay silent: the focus succeeded, and a
  600                 -- trailing "last failed:" would read as a failed focus.
  601                 sendFocusBlock cfg chan idx sid

  603 -- | Handle an inject command (@\/N payload@): resolve the tab index to a
  604 -- session id via @GET /api/tabs@, then send the payload to that session
  605 -- via @POST /api/sessions/:id/send@. Unlike 'handleFocus', this does NOT
  606 -- change the focused tab, send a FocusOp over WS, update the session map,
  607 -- send a "focused tab N" confirmation, or send the last assistant reply.
  608 -- The payload is delivered to the target session; any response will be
  609 -- visible when the user later focuses on that tab.
  610 handleInject
  611   :: ChatChannel c
  612   => ChatChannelConfig -> c -> Maybe MessageSource -> TabIndex -> Text
  613   -> IO ()
  614 handleInject cfg chan mSrc idx payload = do
  615   let apiBase = gcApiBase (cccGateway cfg)
  616       mgr = cccHttpManager cfg
  617   eTabs <- httpGetTabs mgr apiBase
  618   case eTabs of
  619     Left e -> ccSend chan ("inject failed: " <> e)
  620     Right tabs ->
  621       case [ t | t <- tabs, tjIndex t == tabIndexToInt idx ] of
  622         [] -> ccSend chan "inject failed: tab index out of range"
  623         (tab : _) ->
  624           case tjSessionId tab of
  625             Nothing -> ccSend chan "inject failed: tab has no session"
  626             Just sidText -> case mkSessionId sidText of
  627               Left _ -> ccSend chan "inject failed: invalid session id"
  628               Right sid -> do
  629                 -- Send the payload to the target session via HTTP.
  630                 -- Do NOT update the session map, send a FocusOp, or
  631                 -- send a focus confirmation — the current focused tab
  632                 -- is unchanged.
  633                 eResult <- cccSendMessage cfg mgr apiBase sid mSrc payload
  634                 case eResult of
  635                   Right sr | srKind sr == "error" -> ccSend chan (fromMaybe "error" (srError sr))
  636                   Right sr | not (T.null (srResponse sr)) -> ccSend chan (srResponse sr)
  637                   -- An answer-shaped payload that produced an empty reply
  638                   -- means no question was waiting: the text was taken as a
  639                   -- fresh prompt instead. Staying silent here (the old
  640                   -- behaviour) leaves the user believing they answered a
  641                   -- question that has since expired.
  642                   Right sr | srKind sr == "assistant"
  643                           , isAnswerShaped payload ->
  644                       ccSend chan ("/last " <> T.singleton (tabIndexToChar idx)
  645                                    <> " has no question waiting \8212 that message started a new turn")
  646                   Right _ -> pure ()  -- assistant response comes via WS
  647                   Left e -> ccSend chan ("error: " <> e)

  649 -- | Could this injected payload plausibly be an answer to a question?
  650 --
  651 -- Narrow on purpose: a short, whitespace-free token (a number, @yes@, @no@). A
  652 -- real prompt with no reply yet is also an empty @assistant@ response, and must
  653 -- stay silent as it always has.
  654 isAnswerShaped :: Text -> Bool
  655 isAnswerShaped t =
  656   let s = T.strip t

  694 -- | Handle @/bg <prompt>@: mint a headless session (no tab), open a
  695 -- dedicated WS connection focused on it, register it, and run the prompt on
  696 -- it. The reply is posted back to this conversation by
  697 -- 'handleBgServerEvent'; the conversation's focused session, session map,
  698 -- and tab tracking are never touched.
  699 --
  700 -- A blank prompt answers 'bgUsage' and creates nothing; @-h@ / @--help@
  701 -- answers 'bgHelp' and creates nothing. Every failure (mint, connect, send)
  702 -- closes whatever it opened, untracks the session, drops its pending asks,
  703 -- and reports on the channel.
  704 handleBackground
  705   :: ChatChannel c
  706   => ChatChannelConfig -> c -> PendingAsks -> BgSessions
  707   -> ConversationKey -> Maybe MessageSource -> Text
  708   -> IO ()
  709 handleBackground cfg chan pendingAsks bgSessions key mSrc prompt
  710   | T.null (T.strip prompt) = ccSend chan bgUsage
  711   | isBgHelpArg prompt      = ccSend chan bgHelp
  712   | otherwise               = do
  713       let apiBase = gcApiBase (cccGateway cfg)
  714           mgr = cccHttpManager cfg
  715           gwCfg = cccGateway cfg
  716       -- (a) mint a headless session, (b) validate its id.
  717       eMinted <- cccNewSession cfg mgr apiBase (A.object ["headless" .= A.Bool True])
  718       case eMinted >>= mkSessionId of
  719         Left e -> ccSend chan ("bg session failed: " <> e)
  720         Right bgSid -> do
  721           -- (c) per-session streaming state + delivery flag.
  722           ss <- newStreamingState
  723           delivered <- newIORef False
  724           let callback =
  725                 handleBgServerEvent cfg chan key pendingAsks bgSessions bgSid ss delivered
  726           -- (d) the dedicated connection for the bg session.
  727           eWs <- cccStartWs cfg (gcHost gwCfg) (gcWsPort gwCfg) callback
  728           case eWs of
  729             Left e -> ccSend chan ("bg session failed: " <> e)
  730             Right ws -> do
  731               -- (e) focus before registering or sending, so no turn event
  732               -- can arrive for an untracked session (which would leak the
  733               -- connection and skip the watch exclusion).
  734               wcFocus ws bgSid
  735               evictOldestBgSession pendingAsks bgSessions key
  736               bgInsert bgSessions key BgSession { bgSid = bgSid, bgWs = ws }
  737               -- (f) run the prompt on the bg session (never on the
  738               -- conversation's session) and confirm.
  739               eSent <- cccSendMessage cfg mgr apiBase bgSid mSrc prompt
  740               case eSent of
  741                 Right sr | srKind sr /= "error" ->
  742                   ccSend chan ("bg session " <> shortSessionId bgSid <> " started")
  743                 failure -> do
  744                   _ <- bgRemove bgSessions key bgSid
  745                   wcClose ws
  746                   dropPendingAsks pendingAsks bgSid
  747                   ccSend chan ("bg session failed: " <> bgSendFailure failure)
  748   where
  749     bgSendFailure (Left e)   = e
  750     bgSendFailure (Right sr) = fromMaybe "send failed" (srError sr)

  752 -- | Enforce the per-conversation cap: when the conversation already holds
  753 -- 'maxBgSessions' live bg sessions, retire the oldest one so a further
  754 -- session can start.
  755 evictOldestBgSession :: PendingAsks -> BgSessions -> ConversationKey -> IO ()
  756 evictOldestBgSession pendingAsks bgSessions key = do
  757   existing <- bgSessionsFor bgSessions key
  758   when (length existing >= maxBgSessions) $
  759     case existing of
  760       (oldest : _) -> void (retireBgSession pendingAsks bgSessions key (bgSid oldest))
  761       []           -> pure ()

  763 -- | Retire one background session: close its WS connection, untrack it, and
  764 -- drop its pending asks. Returns 'True' when it was still tracked, so a
  765 -- repeated @idle@ is a no-op.
  766 retireBgSession
  767   :: PendingAsks -> BgSessions -> ConversationKey -> SessionId -> IO Bool
  768 retireBgSession pendingAsks bgSessions key sid = do
  769   mSess <- bgRemove bgSessions key sid
  770   for_ mSess (wcClose . bgWs)
  771   dropPendingAsks pendingAsks sid
  772   pure (isJust mSess)

  774 -- | The WS event handler for a background session's dedicated connection.
  775 -- Acts only on events for 'bgSid':
  776 --
  777 -- * a @response@ 'SeEntry' — post the recorded reply to the conversation and
  778 --   mark it delivered. Token deltas are deliberately not streamed for bg
  779 --   turns (pre-regression parity: the reply arrives on the recorded entry).
  780 -- * 'SeAsk' — render the question on the conversation (and register it as
  781 --   pending, so a button tap can answer it).
  782 -- * @tool-call@ activity — post the tool-progress line.
  783 -- * @harness-status@ @idle@ — retire the session (close, untrack, drop its
  784 --   asks) and report a turn that produced no reply.
  785 -- * everything else (entry-updates, list snapshots, replay markers, other
  786 --   sessions) — ignored.
  787 handleBgServerEvent
  788   :: ChatChannel c
  789   => ChatChannelConfig -> c -> ConversationKey -> PendingAsks -> BgSessions
  790   -> SessionId -> StreamingState -> IORef Bool -> ServerEvent -> IO ()
  791 handleBgServerEvent cfg chan key pendingAsks bgSessions bgSid ss delivered ev =
  792   case ev of
  793     SeEntry sid val
  794       | sid == bgSid
  795       , extractDirection val == "response"
  796       , not (T.null (extractEntryText val)) -> do
  797           handleEntry cfg chan ss val
  798           writeIORef delivered True
  799     SeAsk sid val
  800       | sid == bgSid -> handleAsk cfg chan key pendingAsks sid val
  801     SeAskResolved _ val -> forgetAskByPrefix pendingAsks (extractAskId val)
  802     SeActivity sid val
  803       | sid == bgSid -> case extractActivityKind val of
  804           "tool-call" -> handleToolCallActivity cfg chan ss val
  805           "harness-status"
  806             | extractActivityStatus val == "idle" -> do
  807                 -- The close trigger is harness-status idle only (not
  808                 -- reply-delivered, which also fires at every turn end when
  809                 -- sub-agent children keep the session thinking).
  810                 wasTracked <- retireBgSession pendingAsks bgSessions key bgSid
  811                 when wasTracked $ do
  812                   wasDelivered <- readIORef delivered
  813                   if wasDelivered
  814                     then do
  815                       -- The reply was accumulated by 'handleEntry' but bg
  816                       -- idle retires the session WITHOUT going through
  817                       -- 'handleHarnessStatus', so nothing else would deliver
  818                       -- it: the single-bubble style never writes the answer
  819                       -- into the bubble (it must arrive as its own notifying
  820                       -- message). Send it here.
  821                       answer <- stripCursorText (cccStreamCfg cfg)
  822                                   <$> readIORef (ssAccumulated ss)
  823                       unless (T.null (T.strip answer)) $ ccSend chan answer
  824                     else ccSend chan ("bg session " <> shortSessionId bgSid
  825                                       <> " produced no reply")
  826           _ -> pure ()
  827     _ -> pure ()

  829 -- | Ensure a WS connection exists for the conversation. If one exists,
  830 -- send a FocusOp to change focus. If not, start a new WS connection with
  831 -- the streaming event handler and send a FocusOp.
  832 --
  833 -- A connection is only reusable for the session it was built for.
  834 -- 'handleServerEvent' closes over the focused session **by value**, so a
  835 -- FocusOp cannot re-target the filter: an existing connection focused on
  836 -- session A keeps dropping session B's entries and asks. Re-targeting
  837 -- therefore closes the old connection and builds a new one whose callback
  838 -- is bound to the new session.
  839 ensureWsConn
  840   :: ChatChannel c
  841   => ChatChannelConfig -> c
  842   -> TVar (Map ConversationKey WsConn)
  843   -> PendingAsks
  844   -> BgSessions
  845   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  846   -> WatchState
  847   -> ThinkingTabs
  848   -> ConversationKey -> SessionId
  849   -> IO ()
  850 ensureWsConn cfg chan wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key sid = do
  851   conns <- readTVarIO wsConns
  852   dbg ("[watch] ensureWsConn key=" <> ckConv key <> " hasConn=" <> (case Map.lookup key conns of Just _ -> "true"; Nothing -> "false"))
  853   case Map.lookup key conns of
  854     Just (boundSid, ws, _)
  855       | boundSid == sid -> wcFocus ws sid  -- already connected; just change focus
  856       -- Re-target: the callback closed over 'boundSid' by value, so a
  857       -- FocusOp alone would leave this conversation deaf to 'sid'.
  858       | otherwise -> do
  859           atomically (modifyTVar' wsConns (Map.delete key))
  860           wcClose ws
  861           dbg ("[watch] ensureWsConn re-target " <> sessionIdText boundSid
  862                <> " -> " <> sessionIdText sid)
  863           connectWs
  864     Nothing -> do
  865       connectWs
  866   where
  867     -- Start a new WS connection whose event handler is bound to 'sid'.
  868     connectWs = do
  869       let gwCfg = cccGateway cfg
  870           callback = handleServerEvent cfg chan key wsConns pendingAsks
  871             bgSessions tabTracker watchState thinkingTabs sid
  872       eWs <- cccStartWs cfg (gcHost gwCfg) (gcWsPort gwCfg) callback
  873       case eWs of
  874         Left e -> dbg ("[watch] WS connect FAILED: " <> e)
  875         Right ws -> do
  876           ss <- newStreamingState
  877           atomically (modifyTVar' wsConns (Map.insert key (sid, ws, ss)))
  878           wcFocus ws sid

  880 -- | The WS event handler: processes streaming events for one conversation.
  881 -- Creates/edits/finalizes platform messages from entry-update/entry/activity
  882 -- events.
  883 handleServerEvent
  884   :: ChatChannel c
  885   => ChatChannelConfig -> c -> ConversationKey
  886   -> TVar (Map ConversationKey WsConn)
  887   -> PendingAsks
  888   -> BgSessions
  889   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  890   -> WatchState
  891   -> ThinkingTabs
  892   -> SessionId -> ServerEvent -> IO ()
  893 handleServerEvent cfg chan key wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs focusedSid ev = do
  894   -- Staleness guard: when the WS connection for this conversation has been
  895   -- re-targeted to a different session (the user changed focus via /N or
  896   -- sent a message that created a new session), this callback is stale —
  897   -- its 'focusedSid' no longer matches the 'boundSid' in 'wsConns'. A
  898   -- stale callback must not process events: the old streaming state has
  899   -- been replaced, and processing events here would send messages from
  900   -- the wrong session's accumulated text (e.g. a partial reply fragment
  901   -- arriving before the watch notification from the new connection).
  902   conns <- readTVarIO wsConns
  903   let isCurrent = case Map.lookup key conns of
  904         Just (boundSid, _, _) -> boundSid == focusedSid
  905         Nothing -> False
  906   when isCurrent $ case ev of
  907     SeEntryUpdate sid val
  908       | sid == focusedSid ->
  909           withStreamingState wsConns key (\ss -> handleEntryUpdate cfg chan ss val)
  910     SeEntry sid val
  911       | sid == focusedSid ->
  912           withStreamingState wsConns key (\ss -> handleEntry cfg chan ss val)
  913     SeActivity sid val
  914       | sid == focusedSid -> do
  915           dbg ("[watch] SeActivity focused sid=" <> sessionIdText sid <> " kind=" <> extractActivityKind val <> " status=" <> extractActivityStatus val)
  916           withStreamingState wsConns key (\ss -> handleActivity cfg chan ss val)
  917       | otherwise         -> do
  918           -- A background session's activity arrives on that session's own
  919           -- WS connection, but it can also be broadcast to this one. It
  920           -- must never reach the watch-all-tabs path: a bg turn is tracked
  921           -- by 'handleBgServerEvent', and a watch notification would emit
  922           -- a bogus "Tab ?" line plus a duplicate reply for every bg turn.
  923           isBg <- bgOwnsSession bgSessions key sid
  924           dbg ("[watch] SeActivity non-focused sid=" <> sessionIdText sid <> " focusedSid=" <> sessionIdText focusedSid <> " kind=" <> extractActivityKind val <> " status=" <> extractActivityStatus val <> " bg=" <> (if isBg then "true" else "false"))
  925           unless isBg $
  926             handleWatchActivity cfg chan key pendingAsks watchState thinkingTabs focusedSid sid val
  927     -- Tool-call events are BeActivity with kind="tool-call", broadcast
  928     -- by the server's aeOnToolCall hook. They arrive as SeActivity.
  929     -- handleActivity dispatches on kind internally.
  930     SeAsk sid val
  931       | sid == focusedSid -> handleAsk cfg chan key pendingAsks sid val
  932     -- A question answered/cancelled/timed out elsewhere (the web UI) is no
  933     -- longer answerable from this channel.
  934     SeAskResolved _ val -> forgetAskByPrefix pendingAsks (extractAskId val)
  935     SeLists val -> handleLists cfg chan key tabTracker watchState thinkingTabs val
  936     _ -> pure ()  -- ignore events for other sessions or irrelevant types

  940 withStreamingState

  968   case T.words (T.toLower body) of
  969     ["/watch", "on"]  -> setWatch True

  973       ccSend chan (watchStatusMsg cur)

  976       cur <- lookupWatch watchState key

  982       -- Hold the lock through mutation+persist (same reason as
  983       -- 'toggleWatch').
  984       withMVar (wsLock watchState) $ \_ -> do
  985         atomically (modifyTVar' (wsVar watchState) (Map.insert key v))
  986         persistWatch watchState

 1063 -- | A non-focused watched session just went idle. That single activity event
 1064 -- covers two very different situations, and the activity itself cannot tell
 1065 -- them apart:
 1066 --
 1067 -- * the turn finished normally, or
 1068 -- * the turn is blocked waiting for an ASK_HUMAN answer.
 1069 --
 1070 -- The gateway broadcasts @idle@ /before/ it blocks on the prompt
 1071 -- ('wrapCapsForAskStatus'), so a question may also be registered a moment
 1072 -- after the event. Ask the gateway for the session's pending questions, with a
 1073 -- short bounded re-check to cover that ordering, then send the one
 1074 -- notification that matches.

 1076 -- The pending-ask store doubles as the dedupe: a question already registered
 1077 -- for this session has been announced, so a later idle must not announce it

 1081   :: ChatChannel c

 1259 -- | Handle an @entry-update@ event: create or edit the streaming bubble.
 1260 handleEntryUpdate
 1261   :: ChatChannel c
 1262   => ChatChannelConfig -> c -> StreamingState -> Value -> IO ()
 1263 handleEntryUpdate cfg chan ss val = do
 1264   let streamCfg = cccStreamCfg cfg
 1265       text = extractEntryText val
 1266   unless (T.null text) $ do
 1267     -- Post-finalize guard (issue #198 follow-up): a late
 1268     -- entry-update can arrive AFTER the turn's recorded entry
 1269     -- (the server's streaming path and the post-turn broadcast
 1270     -- are unsynchronized). Editing now would overwrite the
 1271     -- finalized text and re-add the cursor. Ignore updates
 1272     -- entirely while the turn is finalized.
 1273     finalized <- readIORef (ssFinalized ss)
 1274     unless finalized $ do
 1275       -- The separate-messages style treats the accumulator as the live text
 1276       -- of the bubble it is growing, so it tracks the newest frame. The
 1277       -- single-bubble style must NOT: its accumulator holds the recorded
 1278       -- answers (the text of the NOTIFYING final message), and a frame
 1279       -- previews the same response, so writing it here duplicates the answer.
 1280       when (ccProgressStyle chan == PsSeparateMessages) $
 1281         writeIORef (ssAccumulated ss) text
 1282       -- Single-edited-bubble style: one message per turn, edited in place.
 1283       -- The frame text is the whole turn so far (the gateway's accumulator
 1284       -- is turn-scoped), so it is the sole body; tool lines are spliced into
 1285       -- it at their recorded offsets.
 1286       -- Live-edit only an existing progress bubble. Creating one here would
 1287       -- put the answer text in a bubble that is then edited rather than
 1288       -- sent, which does not notify - and a plain answer must notify. A
 1289       -- bubble exists only once a tool has run this turn.
 1290       --
 1291       -- The frame is also NOT folded into the answer accumulator: an
 1292       -- entry-update previews the same response its recorded entry will
 1293       -- carry, so accumulating both would duplicate the answer.
 1294       when (ccProgressStyle chan == PsSingleBubble) $ do
 1295         hasBubble <- isJust <$> readIORef (ssMsgId ss)
 1296         when hasBubble $ do
 1297           open <- editGateOpen cfg chan ss
 1298           when open $ do
 1299             edited <- updateBubble cfg chan ss
 1300               (spliceContent ss (Just (if T.null text then "" else addCursor (cccStreamCfg cfg) text)))
 1301             -- This branch only runs when a bubble exists, so it is always an
 1302             -- edit, never a create.
 1303             when edited (markBubbleEdited cfg ss)
 1304       -- Separate-messages style: stream a growing bubble per segment.
 1305       when (ccProgressStyle chan == PsSeparateMessages && ccSupportsStreaming chan) $ do
 1306         now <- cccNow cfg
 1307         mLastEdit <- readIORef (ssLastEdit ss)
 1308         mMsgId <- readIORef (ssMsgId ss)
 1309         -- Gate on NEW text since the last edit, not the total length
 1310         -- (issue #198, part 3). The accumulator only grows within a
 1311         -- response, so a total-length threshold degenerates to an edit
 1312         -- on EVERY frame once the text passes spcBufferThreshold —
 1313         -- one outbound platform edit per arriving WS frame (the
 1314         -- 'lots of small updates' flood). Measuring the delta since
 1315         -- the last edit keeps the cadence the config intends: an edit
 1316         -- per interval, or per 80 NEW codepoints, whichever first.
 1317         lastLen <- readIORef (ssLastLen ss)
 1318         when (shouldEdit streamCfg now mLastEdit (T.length text - lastLen)) $ do
 1319           let content = addCursor streamCfg text
 1320           case mMsgId of
 1321             Nothing -> do
 1322               mId <- ccSendWithId chan content
 1323               case mId of
 1324                 Just id' -> do
 1325                   writeIORef (ssMsgId ss) (Just id')
 1326                   writeIORef (ssLastEdit ss) (Just now)
 1327                   writeIORef (ssLastLen ss) (T.length text)
 1328                 Nothing -> pure ()
 1329             Just id' -> do
 1330               ok <- ccEditMessage chan id' content
 1331               when ok $ do
 1332                 writeIORef (ssLastEdit ss) (Just now)
 1333                 writeIORef (ssLastLen ss) (T.length text)

 1346 -- | Create the turn's bubble on first use, then edit it. Returns 'True' when
 1347 -- the content is on the platform (created or edited).
 1348 --
 1349 -- A failed EDIT deliberately does NOT fall back to sending a fresh message:
 1350 -- the answer is delivered separately by 'finalizeSingleBubble', the body here
 1351 -- is tool progress only, and a fresh send would both notify (the thing this
 1352 -- style exists to avoid) and repeat on every later event. The next frame,
 1353 -- tool call or idle re-renders the whole body from 'ssSplices' /
 1354 -- 'ssAccumulated', which an edit never modifies - so nothing is lost.
 1355 -- | The single-bubble edit gate. One shared bucket per conversation, so the
 1356 -- frame and tool sources together cannot exceed one edit per interval.
 1357 -- Interval-only: the separate-messages path's 'shouldEdit' is an @||@ whose

 1380   :: ChatChannel c

 1382 updateBubble cfg chan ss mkContent = do
 1383   content <- mkContent

 1385   if over then pure False else do
 1386       mMsgId <- readIORef (ssMsgId ss)
 1387       case mMsgId of
 1388         Nothing -> do

 1418 -- | Merge one recorded response into the turn's answer.
 1419 --
 1420 -- Two things can happen, and they are easy to confuse:
 1421 --
 1422 -- * the same response arrives again / later text extends the previous one
 1423 --   (the gateway's accumulator is turn-scoped, so a second response's text can
 1424 --   already contain the first). The previous contribution is REPLACED, so the
 1425 --   overlap is not concatenated twice;
 1426 -- * a genuinely new segment arrives (a @StopMaxTokens@ continuation, which
 1427 --   does not contain the earlier text). It is APPENDED.
 1428 --
 1429 -- @prev@ is the text of the previous entry, already folded into @accum@; it is
 1430 -- what must be dropped before adding @next@.
 1431 mergeAnswer :: Text -> Text -> Text -> Text
 1432 mergeAnswer accum prev next
 1433   | T.null next = accum
 1434   | T.null accum = next
 1435   -- The new text extends / repeats the previous entry: swap the old
 1436   -- contribution for the new one.
 1437   | not (T.null prev), prev `T.isPrefixOf` next = dropSuffix prev accum <> next
 1438   -- A fresh segment.
 1439   | otherwise = accum <> "\n\n" <> next
 1440   where
 1441     dropSuffix suf t = case T.stripSuffix suf t of
 1442       Just before -> T.dropWhileEnd (== '\n') before
 1443       Nothing     -> t

 1451 -- | Deliver the turn's final body. Normally that is one more edit of the
 1452 -- bubble. When the body has outgrown a single platform message there is no
 1453 -- message left to edit, so the text is sent once through the channel's
 1454 -- chunking path (the same path every ordinary long reply takes).
 1455 --
 1456 -- Deliberately called only at turn end: sending from the overflow branch of
 1457 -- 'updateBubble' would re-send the whole body on every frame, entry and
 1458 -- idle - one full-text delivery per event instead of one per turn.
 1459 finalizeSingleBubble
 1460   :: ChatChannel c
 1461   => ChatChannelConfig -> c -> StreamingState -> IO ()
 1462 finalizeSingleBubble cfg chan ss = do
 1463   -- 1. Close the progress bubble: strip the cursor so a half-typed preview
 1464   --    does not linger. The bubble holds tool progress only.
 1465   mBubble <- readIORef (ssMsgId ss)
 1466   let hadBubble = isJust mBubble
 1467   when (isJust mBubble) $ do
 1468     body <- spliceContent ss (Just "")
 1469     over <- shouldFallBackToChunks cfg chan body
 1470     if over
 1471       then do
 1472         -- The body outgrew one message. Abandoning the bubble mid-turn used
 1473         -- to leave a truncated prefix frozen on screen WITH a streaming
 1474         -- cursor, which reads as "still working" forever. Replace it with one
 1475         -- deliberate closing line instead; the full text arrives below via
 1476         -- the chunked send path.
 1477         dbg ("[bubble] progress body over limit ("
 1478              <> T.pack (show (T.length body)) <> " chars); closing it")
 1479         case mBubble of
 1480           Just id' -> void (ccEditMessage chan id' overLimitCloseText)
 1481           Nothing  -> pure ()
 1482         writeIORef (ssMsgId ss) Nothing
 1483       else void (updateBubble cfg chan ss (pure body))
 1484   -- 2. Deliver the answer as its own message. This is what notifies the
 1485   --    human: an edit never does.
 1486   answer <- stripCursorText (cccStreamCfg cfg) <$> readIORef (ssAccumulated ss)
 1487   dbg ("[bubble] delivering final answer as a new message ("
 1488        <> T.pack (show (T.length answer)) <> " chars)")
 1489   if T.null (T.strip answer)
 1490     then
 1491       -- A tools-only turn has no answer to deliver. Without this it would
 1492       -- notify ZERO times - the bubble is silent and edits are silent - which
 1493       -- would break the rule that the last message before idle notifies.
 1494       -- Send a short close notice so every turn notifies exactly once.
 1495       when hadBubble $ ccSend chan closeNoticeText
 1496     else ccSend chan answer

 1554       -- response — send it even if we already sent the accumulated text
 1555       -- via idle (the recorded entry is the canonical source). Skip if
 1556       -- we already delivered via finalizeBubble (mMsgId was set and
 1557       -- finalized).
 1558       alreadyDelivered <- readIORef (ssFinalized ss)
 1559       unless alreadyDelivered $
 1560         case ccProgressStyle chan of
 1561           -- Single bubble: the recorded entry is the authoritative text. It
 1562           -- arrives before the tool dispatch, so committing it here means the
 1563           -- bubble always shows exact text rather than a throttled prefix.
 1564           -- No finalization latch: later segment frames must still be able to
 1565           -- extend the bubble.
 1566           -- The answer is accumulated, not written into the progress
 1567           -- bubble: the bubble is a separate message that gets edited (no
 1568           -- notification), while the answer must arrive as a NEW message so
 1569           -- the human is notified. Response entries are appended because a
 1570           -- turn can record several (a StopMaxTokens continuation).
 1571           PsSingleBubble -> do
 1572             accum <- readIORef (ssAccumulated ss)
 1573             prev <- readIORef (ssLastEntry ss)
 1574             writeIORef (ssLastEntry ss) text
 1575             writeIORef (ssAccumulated ss) (mergeAnswer accum prev text)
 1576           PsSeparateMessages -> finalizeBubble cfg chan ss text True

 1583 handleActivity cfg chan ss val =

 1589 -- | Handle a @harness-status@ activity: on @thinking@ (turn start),
 1590 -- clear the finalized flag so the new turn streams. On @idle@ (turn
 1591 -- end), finalize any in-progress streaming bubble with the accumulated
 1592 -- text (a safety net — the recorded entry is the normal finalize path)
 1593 -- and reset all streaming state.
 1594 handleHarnessStatus
 1595   :: ChatChannel c
 1596   => ChatChannelConfig -> c -> StreamingState -> Value -> IO ()
 1597 handleHarnessStatus cfg chan ss val =
 1598   case extractActivityStatus val of
 1599     "thinking" ->
 1600       -- Turn start: clear the finalized flag so the new turn streams.
 1601       writeIORef (ssFinalized ss) False
 1602     "idle" -> do
 1603       case ccProgressStyle chan of
 1604         -- Single bubble: a final edit with the cursor stripped. The recorded
 1605         -- entry already committed the exact text, so this only clears the
 1606         -- cursor and closes the turn; a withheld idle (running background
 1607         -- children) leaves complete, un-cursored text behind.
 1608         PsSingleBubble -> do
 1609           alreadyFinal <- readIORef (ssFinalized ss)
 1610           unless alreadyFinal $
 1611             finalizeSingleBubble cfg chan ss
 1612         PsSeparateMessages -> do
 1613           mMsgId <- readIORef (ssMsgId ss)
 1614           accum <- readIORef (ssAccumulated ss)
 1615           -- For non-streaming channels the accumulated text is the full
 1616           -- response. Send it as a new message (mMsgId is Nothing because no
 1617           -- streaming bubble was created). For streaming channels, only
 1618           -- finalize if there's an active streaming bubble (mMsgId is Just).
 1619           when (not (T.null accum) &&
 1620                 (not (ccSupportsStreaming chan) || isJust mMsgId)) $
 1621             finalizeBubble cfg chan ss accum False
 1622       resetStreamingState ss
 1623       writeIORef (ssFinalized ss) True
 1624     _ -> pure ()

 1626 -- | Handle a @tool-call@ activity: finalize the current text bubble
 1627 -- (segment break — the pre-tool text stays its own message, and the
 1628 -- next entry-update starts a NEW bubble below the tool line), then send
 1629 -- the tool-progress line as its own platform message.
 1630 handleToolCallActivity
 1631   :: ChatChannel c
 1632   => ChatChannelConfig -> c -> StreamingState -> Value -> IO ()
 1633 handleToolCallActivity cfg chan ss val = do
 1634   -- Segment break (issue #198 follow-up, symptom 1): the pre-tool
 1635   -- streamed text and the post-tool streamed text must be SEPARATE
 1636   -- platform messages, with the tool line between them. Without this,
 1637   -- the next entry-update edits the pre-tool bubble (ssMsgId is still
 1638   -- set) and appends the post-tool text onto the pre-tool text.
 1639   case ccProgressStyle chan of
 1640     PsSingleBubble -> case extractToolName val of
 1641       Nothing -> pure ()
 1642       Just toolName -> do
 1643         -- Record the tool line with the offset it fired at, then re-render
 1644         -- the one bubble. This keeps the line in its true place in the
 1645         -- turn's text stream instead of appending it at the end, and it
 1646         -- produces no new notification.
 1647         accum <- readIORef (ssAccumulated ss)
 1648         let mInput = extractToolInput val
 1649             line = formatToolLine mempty toolName (fromMaybe "" mInput)
 1650         modifyIORef' (ssSplices ss) (<> [(T.length accum, line)])
 1651         -- Gated on the SAME bucket as the frame path. The first tool call of a
 1652         -- turn is the bubble's create ('hasBubble' is false, so this branch's
 1653         -- edit creates it via 'updateBubble'); a create does not consume the
 1654         -- bucket, so the first real edit after it lands immediately.
 1655         hadBubble <- isJust <$> readIORef (ssMsgId ss)
 1656         open <- editGateOpen cfg chan ss
 1657         when open $ do
 1658           edited <- updateBubble cfg chan ss (spliceContent ss (Just ""))
 1659           -- A CREATE is not an edit: it does not consume the bucket, so the
 1660           -- next event can edit immediately and the progress becomes visible
 1661           -- straight away.
 1662           when (edited && hadBubble) (markBubbleEdited cfg ss)
 1663     PsSeparateMessages -> do
 1664       mMsgId <- readIORef (ssMsgId ss)
 1665       accum <- readIORef (ssAccumulated ss)
 1666       -- For non-streaming channels, finalize when there's accumulated text
 1667       -- even without a streaming bubble (mMsgId is Nothing). For streaming
 1668       -- channels, only finalize when there's an in-progress bubble.
 1669       when (not (T.null accum) &&
 1670             (not (ccSupportsStreaming chan) || isJust mMsgId)) $
 1671         finalizeBubble cfg chan ss accum False
 1672       case extractToolName val of
 1673         Nothing -> pure ()
 1674         Just toolName -> do
 1675           let mInput = extractToolInput val
 1676               line = formatToolLine mempty toolName (fromMaybe "" mInput)
 1677           ccSend chan line

 1679 -- | Handle an @ask@ event: render the question on the platform. When the
 1680 -- ask has options and the channel supports inline keyboards, sends the
 1681 -- question + an inline keyboard (one button per option). Otherwise falls
 1682 -- back to the numbered-list text rendering. Registers the pending ask so
 1683 -- the callback handler can resolve a button tap to the option label.
 1684 -- The ask is registered even when it offers no options (an open-ended
 1685 -- 'ASK_HUMAN', or the confirmation gate's bare prompt), so it can still be
 1686 -- resolved and dequeued instead of leaking into the pending map.
 1687 handleAsk
 1688   :: ChatChannel c
 1689   => ChatChannelConfig -> c -> ConversationKey -> PendingAsks -> SessionId -> Value
 1690   -> IO ()
 1691 handleAsk _cfg chan _key pendingAsks sid val = do
 1692   let question = extractAskQuestion val
 1693       askId = extractAskId val
 1694       opts = extractAskOptions val
 1695   -- One renderer for both shapes, so an open-ended ask carries a
 1696   -- "Reply with your answer." hint instead of arriving as a bare question
 1697   -- the human cannot tell is waiting on them. A channel with inline
 1698   -- keyboards renders the buttons instead; its body carries the same
 1699   -- question + option descriptions.
 1700   if null opts
 1701     then ccSend chan (formatAsk question [])
 1702     else do
 1703       let prefix = T.take 8 askId
 1704       mMsgId <- ccSendWithOptions chan question opts prefix
 1705       case mMsgId of
 1706         Just _  -> pure ()

 1709   -- the option label. An ask without an id can never be resolved — it is

 1712     seqNo <- atomically $ do
 1713       n <- readTVar (paNext pendingAsks)
 1714       writeTVar (paNext pendingAsks) (n + 1)
 1715       pure n
 1716     let entry = PendingAsk { paAskId = askId, paOptions = opts, paWatched = False, paSeq = seqNo }
 1717     atomically (modifyTVar' (paStore pendingAsks) (Map.insertWith (++) sid [entry]))

 1758 -- | Handle /new: mint a fresh session and make it the conversation's
 1759 -- focused session.
 1760 --
 1761 -- Exactly ONE mint happens per @/new@:
 1762 --
 1763 --   * the conversation already has a session ⇒ @POST \/api\/sessions\/:id\/new@
 1764 --     ('cccRebindNew'), which mints AND rebinds the tab in one call;
 1765 --   * no prior session ⇒ @POST \/api\/sessions\/new@ ('cccNewSession').
 1766 --
 1767 -- Minting on both paths would create two sessions and reproduce the exact
 1768 -- orphan-session bug this work exists to fix.
 1769 --
 1770 -- The arguments are the CLI's (@-p\/--provider@, @-m\/--model@,
 1771 -- @-r\/--repo@, @-b\/--branch@), parsed by the shared 'parseNewArgs' so chat
 1772 -- and CLI accept the same flags. @-b@ only qualifies @-r@: without a repo
 1773 -- there is nothing to check out, so no setup is dispatched. Message order:
 1774 -- rebind warning, then repo failure, then the confirmation.
 1775 handleNewSession
 1776   :: ChatChannel c
 1777   => ChatChannelConfig -> c -> SessionMap -> TVar (Map ConversationKey WsConn)
 1778   -> PendingAsks -> BgSessions -> TVar (Map ConversationKey (SessionId, [TabJson]))
 1779   -> WatchState -> ThinkingTabs
 1780   -> ConversationKey -> Text
 1781   -> IO ()
 1782 handleNewSession cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key args = do
 1783   let apiBase = gcApiBase (cccGateway cfg)
 1784       mgr = cccHttpManager cfg
 1785       na = parseNewArgs args
 1786   mPrior <- sessionLookup sessions key
 1787   eOutcome <- case mPrior of
 1788     Just oldSid -> do
 1789       -- Absent provider/model inherit the old session's (the gateway applies
 1790       -- that); the channel is this channel's label, so the new session stays
 1791       -- attributable to the medium it was created from.
 1792       eRes <- cccRebindNew cfg mgr apiBase oldSid
 1793         (rebindBody (naProvider na) (naModel na) (Just (ccLabel chan)))
 1794       pure $ fmap (\res -> NewOutcome
 1795         { noSid         = nsrSessionId res
 1796         , noRebindError = nsrRebindError res
 1797         , noOldSid      = Just (sessionIdText oldSid)
 1798         , noProvider    = Just (nsrProvider res)
 1799         , noModel       = Just (nsrModel res)
 1800         }) eRes
 1801     Nothing -> do
 1802       eSid <- cccNewSession cfg mgr apiBase (mintBody (naProvider na) (naModel na) (Just (ccLabel chan)))
 1803       pure $ fmap (\sidText -> NewOutcome
 1804         { noSid         = sidText
 1805         , noRebindError = Nothing
 1806         , noOldSid      = Nothing
 1807         , noProvider    = Nothing
 1808         , noModel       = Nothing
 1809         }) eSid
 1810   case eOutcome of
 1811     Left e -> ccSend chan ("new session failed: " <> e)
 1812     Right out -> case mkSessionId (noSid out) of
 1813       Left _ -> ccSend chan "new session failed: invalid session id"
 1814       Right sid -> do
 1815         sessionInsert sessions key sid
 1816         -- Re-target the conversation's WS connection on BOTH paths: without
 1817         -- this the new session's replies and asks are filtered out (WU-5).
 1818         ensureWsConn cfg chan wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key sid
 1819         -- A rebind failure is a warning, not a failure: the session exists.
 1820         for_ (noRebindError out) $ \e ->
 1821           ccSend chan ("warning: /new tab rebind failed: " <> e)
 1822         mRepoErr <- setupRepoFor cfg mgr apiBase sid (naRepo na) (naBranch na)
 1823         for_ mRepoErr $ \e -> ccSend chan ("repo setup failed: " <> e)
 1824         ccSend chan (newConfirmation out)

 1932 -- | Handle a callback query (button tap) from the platform. The callback_data
 1933 -- is @"<8hex>:<index>"@. Resolves the 8-hex prefix to a pending ask, resolves
 1934 -- the index to the option label, answers via the HTTP API, acknowledges the
 1935 -- callback (dismiss the spinner), removes the keyboard, and sends a
 1936 -- @✓ <label>@ confirmation. If no pending ask matches, the callback is
 1937 -- silently dropped (stale button tap).
 1938 handleCallback
 1939   :: ChatChannel c
 1940   => ChatChannelConfig -> c -> SessionMap -> PendingAsks -> BgSessions
 1941   -> ConversationKey -> MessageSource -> Text
 1942   -> IO ()
 1943 handleCallback cfg chan sessions pendingAsks bgSessions key _src cbData = do
 1944   case parseCallbackData cbData of
 1945     Nothing -> pure ()  -- not a valid callback_data format
 1946     Just (prefix, idx) -> do
 1947       -- Only asks owned by this conversation are answerable: its focused
 1948       -- session, or one of its own background sessions. A tap in another
 1949       -- conversation can never resolve them.
 1950       sids <- answerableSids sessions bgSessions key
 1951       asksMap <- readTVarIO (paStore pendingAsks)
 1952       let matches =
 1953             [ (sid, ask)
 1954             | sid <- sids
 1955             , Just asks <- [Map.lookup sid asksMap]
 1956             , ask <- asks
 1957             , T.isPrefixOf prefix (paAskId ask)
 1958             ]
 1959       case matches of
 1960         [] -> pure ()
 1961         ((sid, ask) : _) ->
 1962           case atIndex (paOptions ask) idx of
 1963             Nothing -> pure ()
 1964             Just opt -> answerAsk cfg chan pendingAsks sid (paAskId ask) (qoLabel opt)

 2023 -- | Answer a pending ask belonging to one of this conversation's background
 2024 -- sessions with an inbound plain-text message. Returns 'True' when the
 2025 -- message was consumed as an answer (so it must NOT be routed to the
 2026 -- focused session).
 2027 --
 2028 -- Background sessions only, deliberately. The conversation's OWN (focused)
 2029 -- session is not consulted here: a plain message for it is routed to the
 2030 -- gateway, whose @POST \/api\/sessions\/:id\/send@ offers the text to that
 2031 -- session's pending ask before starting a turn. That keeps the answer path
 2032 -- working when the loop never saw the @ask@ event (a reconnect or a channel
 2033 -- restart) and guarantees the reply never queues behind the session lock the
 2034 -- blocked turn holds.
 2035 --
 2036 -- A numeric body (1-based) selects the corresponding option label; any other
 2037 -- body is delivered verbatim -- which is what the confirmation gate needs
 2038 -- (@y@/@yes@ approve once, anything else rejects). The oldest pending ask
 2039 -- wins: bg sessions are scanned in creation order, and each session's list
 2040 -- is newest-first ('handleAsk' prepends), so the oldest ask for a session is
 2041 -- its last element.
 2042 answerPendingBgAsk
 2043   :: ChatChannel c
 2044   => ChatChannelConfig -> c -> PendingAsks -> BgSessions -> ConversationKey -> Text
 2045   -> IO Bool
 2046 answerPendingBgAsk cfg chan pendingAsks bgSessions key body = do
 2047   bg <- bgSessionsFor bgSessions key
 2048   asksMap <- readTVarIO (paStore pendingAsks)
 2049   let bgIds = map bgSid bg
 2050       -- A question raised on a non-focused tab never reaches 'handleAsk' (the
 2051       -- broker focus-filters asks), so the watch path registers it with
 2052       -- 'paWatched' set. Those are answerable here too — otherwise the
 2053       -- notification could be read but not replied to.
 2054       watched =
 2055         [ (sess, ask)
 2056         | (sess, asks) <- Map.toList asksMap
 2057         , sess `notElem` bgIds
 2058         , Just ask <- [earliestRegistered (filter paWatched asks)]
 2059         ]
 2060   -- Per session the earliest-registered ask is the list tail ('handleAsk'
 2061   -- prepends); across sessions 'paSeq' orders them, which session order
 2062   -- cannot.
 2063   let candidates =
 2064         [ (bgSid sess, ask)
 2065         | sess <- bg
 2066         , Just asks <- [Map.lookup (bgSid sess) asksMap]
 2067         , Just ask <- [earliestRegistered asks]
 2068         ] <> watched
 2069   case oldestAcross candidates of
 2070     Nothing -> pure False
 2071     Just (sid, ask) -> do
 2072       answerAsk cfg chan pendingAsks sid (paAskId ask) (resolveAnswer (paOptions ask) body)
 2073       pure True

 2110 -- | Handle a @lists@ WS event: compare the new tab list with the last known
 2111 -- one for this conversation. If the focused session's tab was removed (closed),
 2112 -- send the "tab closed" notification and clear the tracking state so the next
 2113 -- message creates a fresh tab. Also seeds the 'ThinkingTabs' set from the
 2114 -- snapshot's @thinkingSessionIds@ so tabs already thinking before the WS
 2115 -- connection was established still trigger a finished notification.
 2116 handleLists
 2117   :: ChatChannel c
 2118   => ChatChannelConfig -> c -> ConversationKey
 2119   -> TVar (Map ConversationKey (SessionId, [TabJson]))
 2120   -> WatchState -> ThinkingTabs -> Value -> IO ()
 2121 handleLists cfg chan key tabTracker watchState thinkingTabs val = do
 2122   -- Seed the thinking-tabs set from the snapshot's thinkingSessionIds.
 2123   -- This is the core fix for the "missed thinking event" problem: when a
 2124   -- WS connection is established (or a lists snapshot is broadcast on tab
 2125   -- changes), sessions that were already thinking before the connection
 2126   -- existed are included in the snapshot. Without this seeding, the
 2127   -- ThinkingTabs set starts empty and the thinking→idle transition is
 2128   -- never detected for those sessions (the idle event arrives but
 2129   -- wasThinking is false → no notification).
 2130   seedThinkingTabsFromLists watchState thinkingTabs key val
 2131   -- Fetch the current tabs from the gateway to get an accurate list.
 2132   let apiBase = gcApiBase (cccGateway cfg)
 2133       mgr = cccHttpManager cfg
 2134   eTabs <- httpGetTabs mgr apiBase
 2135   case eTabs of
 2136     Left _ -> pure ()
 2137     Right newTabs -> do
 2138       tracker <- readTVarIO tabTracker
 2139       case Map.lookup key tracker of
 2140         Nothing -> pure ()  -- no tracked session for this conversation
 2141         Just (focusedSid, _oldTabs) -> do
 2142           -- Check if the focused session's tab was removed.
 2143           let newSessionIds = mapMaybe tjSessionId newTabs
 2144           if focusedSid `elem` mapMaybe mkSessionId' newSessionIds
 2145             then do
 2146               -- Tab still exists; update the tracked tab list.
 2147               atomically (modifyTVar' tabTracker (Map.insert key (focusedSid, newTabs)))
 2148             else do
 2149               -- The focused session's tab was closed. Send the notification.
 2150               ccSend chan ("tab closed (session " <> sessionIdText focusedSid
 2151                         <> "); a new tab will be created on your next message")
 2152               -- Clear the tracking state so the next message creates a fresh tab.
 2153               atomically (modifyTVar' tabTracker (Map.delete key))
 2154   where
 2155     mkSessionId' t = case mkSessionId t of Right s -> Just s; Left _ -> Nothing

   99 -- | Per-conversation watch-all-tabs toggle state. When 'True' for a
  100 -- conversation, the channel receives a notification every time any
  101 -- non-focused tab finishes thinking (harness-status transitions from
  102 -- @thinking@ to @idle@). Thread-safe via 'TVar'.
  103 data WatchState = WatchState
  104   { wsVar  :: TVar (Map ConversationKey Bool)
  105   , wsSave :: Maybe (IO ())
  106   -- ^ The save action (snapshots the TVar and writes to disk). Called
  107   -- inside the 'wsLock' critical section so the snapshot is consistent
  108   -- with the mutation.
  109   , wsLock :: MVar ()
  110   -- ^ Serializes the mutation+persist sequence so a stale snapshot from
  111   -- one thread cannot overwrite a newer on-disk write from another.
  112   -- Without this lock, concurrent toggles from Signal + Telegram (which
  113   -- share one 'WatchState') can lose state after a restart.
  114   }
  115 
  116 -- | Create a new empty watch-state map (watch mode off for all
  117 -- conversations).
  118 newWatchState :: IO WatchState
  119 newWatchState = do
  120   tv <- newTVarIO Map.empty
  121   lock <- newMVar ()
  122   pure (WatchState tv Nothing lock)
  123 
  124 -- | Create a persisting watch-state store. The supplied save function is
  125 -- called after every mutation (snapshotting the full current map, so the
  126 -- last writer wins with a consistent view). Mirrors
  127 -- 'Seal.Channels.Cursor.newPersistingCursorStore'.
  128 newPersistingWatchState :: (Map ConversationKey Bool -> IO ()) -> IO WatchState
  129 newPersistingWatchState saveFn = do
  130   tv <- newTVarIO Map.empty
  131   lock <- newMVar ()
  132   let ws = WatchState tv (Just (saveAction ws)) lock
  133       saveAction s = saveFn =<< snapshotWatch s
  134   pure ws
  135 
  136 -- | Replace the store's map in one STM transaction. Used at boot to seed
  137 -- the in-memory store from the persisted @watch_state.json@. Does NOT
  138 -- persist (the caller is loading FROM disk).
  139 seedWatchState :: WatchState -> Map ConversationKey Bool -> IO ()
  140 seedWatchState ws m = atomically (writeTVar (wsVar ws) m)
  141 
  142 -- | Snapshot the current map. Used by the save action.
  143 snapshotWatch :: WatchState -> IO (Map ConversationKey Bool)
  144 snapshotWatch ws = readTVarIO (wsVar ws)
  145 
  146 -- | Look up whether watch mode is enabled for a conversation. 'False'
  147 -- when the conversation has no entry (the default).
  148 lookupWatch :: WatchState -> ConversationKey -> IO Bool
  149 lookupWatch ws key = fromMaybe False . Map.lookup key <$> readTVarIO (wsVar ws)
  150 
  151 -- | Toggle watch mode for a conversation. Returns the new state.
  152 toggleWatch :: WatchState -> ConversationKey -> IO Bool
  153 toggleWatch ws key = do
  154   -- Hold the lock through mutation+persist so no concurrent thread can
  155   -- mutate the TVar between our snapshot and our save (which would let a
  156   -- stale snapshot overwrite a newer on-disk write).
  157   withMVar (wsLock ws) $ \_ -> do
  158     newVal <- atomically $ do
  159       m <- readTVar (wsVar ws)
  160       let v = not (fromMaybe False (Map.lookup key m))
  161       writeTVar (wsVar ws) (Map.insert key v m)
  162       pure v
  163     persistWatch ws
  164     pure newVal
  165 
  166 -- | Run the persist action (if any) after a successful mutation. A save
  167 -- failure is logged to stderr and swallowed — the in-memory store stays
  168 -- authoritative within the session; the next successful mutation will
  169 -- retry the save (writing the full current map, so a missed save
  170 -- self-heals). Mirrors 'Seal.Channels.Cursor.persistCursor'.
  171 persistWatch :: WatchState -> IO ()
  172 persistWatch ws =
  173   case wsSave ws of
  174     Nothing  -> pure ()
  175     Just act -> act `catch` \e ->
  176       dbg ("[watch] watch_state.json save failed: " <> T.pack (show (e :: SomeException)))

  211 -- | Run the generic chat-channel loop. Blocks until the channel's
  212 -- 'ccReceive' returns EOF. Each conversation gets its own WS connection
  213 -- for streaming.
  214 runChatChannel :: ChatChannel c => ChatChannelConfig -> c -> WatchState -> IO ()
  215 runChatChannel cfg chan watchState = do
  216   sessions <- newSessionMap
  217   -- Map of conversation keys to their WS client + streaming state.
  218   wsConns <- newTVarIO Map.empty :: IO (TVar (Map ConversationKey (WsClient, StreamingState)))
  219   pendingAsks <- newTVarIO Map.empty :: IO PendingAsks
  220   tabTracker <- newTVarIO Map.empty :: IO (TVar (Map ConversationKey (SessionId, [TabJson])))
  221   thinkingTabs <- newThinkingTabs
  222   loop sessions wsConns pendingAsks tabTracker watchState thinkingTabs
  223   where
  224     loop sessions wsConns pendingAsks tabTracker ws tt = do
  225       mMsg <- ccReceive chan
  226       case mMsg of
  227         Nothing -> pure ()  -- EOF
  228         Just (InboundMessage src body mCbData) -> do
  229           dbg ("received: " <> body)
  230           let key = convKeyFromSource src
  231           case mCbData of
  232             Just cbData -> do
  233               -- Callback query (button tap): resolve to a pending ask,
  234               -- answer via HTTP, acknowledge, and remove keyboard.
  235               handleCallback cfg chan sessions pendingAsks key src cbData
  236             Nothing ->
  237               -- Regular text message: route normally.
  238               handleInbound cfg chan sessions wsConns pendingAsks tabTracker ws tt key body
  239           loop sessions wsConns pendingAsks tabTracker ws tt

  241 -- | Handle one inbound message: resolve the session, route, and dispatch.
  242 handleInbound
  243   :: ChatChannel c
  244   => ChatChannelConfig -> c -> SessionMap
  245   -> TVar (Map ConversationKey (WsClient, StreamingState))
  246   -> PendingAsks
  247   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  248   -> WatchState
  249   -> ThinkingTabs
  250   -> ConversationKey -> Text
  251   -> IO ()
  252 handleInbound cfg chan sessions wsConns pendingAsks tabTracker watchState thinkingTabs key body = do
  253   let apiBase = gcApiBase (cccGateway cfg)
  254       mgr = cccHttpManager cfg
  255   -- Intercept /watch (and /watch on|off) before normal routing — it
  256   -- toggles per-conversation loop state, not a gateway command.
  257   if isWatchCommand body
  258     then do
  259       dbg ("[watch] /watch command received, key=" <> ckConv key)
  260       handleWatchToggle chan watchState key body
  261       -- When watch mode is turned ON, ensure a WS connection exists so
  262       -- the channel receives BeActivity events for all tabs. Without
  263       -- this, /watch as the first message would enable watch mode but
  264       -- never receive any events (no WS connection = no event source).
  265       watchOn <- lookupWatch watchState key
  266       dbg ("[watch] watchOn=" <> (if watchOn then "true" else "false") <> ", ensuring WS conn")
  267       when watchOn $
  268         void (resolveSession cfg chan sessions wsConns pendingAsks tabTracker watchState thinkingTabs key)
  269     else case parseTabFocus body of
  270     Just idx -> do
  271       handleFocus cfg chan sessions wsConns pendingAsks tabTracker watchState thinkingTabs key idx
  272     Nothing -> case route body of
  273       Right (ChatFocus idx) ->
  274         handleFocus cfg chan sessions wsConns pendingAsks tabTracker watchState thinkingTabs key idx
  275       Right (ChatInject idx payload) -> do
  276         -- Send the payload to the target tab's session WITHOUT changing
  277         -- the focused tab (no FocusOp, no "focused tab N" confirmation).
  278         handleInject cfg chan idx payload
  279       Right ChatCurrentTab -> do
  280         -- Send the current tab info via HTTP (the gateway routes /tab).
  281         mSid <- sessionLookup sessions key
  282         case mSid of
  283           Just sid -> do
  284             eResult <- httpSend mgr apiBase sid "/tab"
  285             case eResult of
  286               Right sr -> ccSend chan (srResponse sr)
  287               Left e -> ccSend chan ("error: " <> e)
  288           Nothing -> ccSend chan "no current tab"
  289       Right (ChatNewSession args) -> do
  290         -- Create a new session via HTTP, update the session map.
  291         handleNewSession cfg chan sessions wsConns pendingAsks tabTracker watchState thinkingTabs key args
  292       Right (ChatSlash _cmd) ->
  293         sendSlash cfg chan sessions wsConns pendingAsks tabTracker watchState thinkingTabs key body
  294       Right (ChatTabCommand _) ->
  295         -- Tab commands go through the HTTP API (the gateway routes them).
  296         sendSlash cfg chan sessions wsConns pendingAsks tabTracker watchState thinkingTabs key body
  297       Right (ChatPlain text) ->
  298         sendPlain cfg chan sessions wsConns pendingAsks tabTracker watchState thinkingTabs key text
  299       Left _ -> ccSend chan "error: invalid command"

  301 -- | Handle a focus command: resolve the tab index to a session id via
  302 -- @GET /api/tabs@, send a @FocusOp@ over WS, send "focused tab N" to the
  303 -- platform, and optionally send the last assistant reply.
  304 handleFocus
  305   :: ChatChannel c
  306   => ChatChannelConfig -> c -> SessionMap
  307   -> TVar (Map ConversationKey (WsClient, StreamingState))
  308   -> PendingAsks
  309   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  310   -> WatchState
  311   -> ThinkingTabs
  312   -> ConversationKey -> TabIndex
  313   -> IO ()
  314 handleFocus cfg chan sessions wsConns pendingAsks tabTracker watchState thinkingTabs key idx = do
  315   let apiBase = gcApiBase (cccGateway cfg)
  316       mgr = cccHttpManager cfg
  317   eTabs <- httpGetTabs mgr apiBase
  318   case eTabs of
  319     Left e -> ccSend chan ("focus failed: " <> e)
  320     Right tabs ->
  321       case [ t | t <- tabs, tjIndex t == tabIndexToInt idx ] of
  322         [] -> ccSend chan "focus failed: tab index out of range"
  323         (tab : _) ->
  324           case tjSessionId tab of
  325             Nothing -> ccSend chan "focus failed: tab has no session"
  326             Just sidText -> case mkSessionId sidText of
  327               Left _ -> ccSend chan "focus failed: invalid session id"
  328               Right sid -> do
  329                 -- Update the session map so subsequent messages route here.
  330                 sessionInsert sessions key sid
  331                 -- Track the focused session + current tab list for tab-close detection.
  332                 atomically (modifyTVar' tabTracker (Map.insert key (sid, tabs)))
  333                 -- Ensure a WS connection exists for this conversation.
  334                 ensureWsConn cfg chan wsConns pendingAsks tabTracker watchState thinkingTabs key sid
  335                 -- Send "focused tab N" confirmation.
  336                 ccSend chan ("focused tab " <> T.singleton (tabIndexToChar idx))
  337                 -- Fetch and send the last assistant reply for context.
  338                 sendLastReply cfg chan sid
  339 
  340 -- | Handle an inject command (@\/N payload@): resolve the tab index to a
  341 -- session id via @GET /api/tabs@, then send the payload to that session
  342 -- via @POST /api/sessions/:id/send@. Unlike 'handleFocus', this does NOT
  343 -- change the focused tab, send a FocusOp over WS, update the session map,
  344 -- send a "focused tab N" confirmation, or send the last assistant reply.
  345 -- The payload is delivered to the target session; any response will be
  346 -- visible when the user later focuses on that tab.
  347 handleInject
  348   :: ChatChannel c
  349   => ChatChannelConfig -> c -> TabIndex -> Text
  350   -> IO ()
  351 handleInject cfg chan idx payload = do
  352   let apiBase = gcApiBase (cccGateway cfg)
  353       mgr = cccHttpManager cfg
  354   eTabs <- httpGetTabs mgr apiBase
  355   case eTabs of
  356     Left e -> ccSend chan ("inject failed: " <> e)
  357     Right tabs ->
  358       case [ t | t <- tabs, tjIndex t == tabIndexToInt idx ] of
  359         [] -> ccSend chan "inject failed: tab index out of range"
  360         (tab : _) ->
  361           case tjSessionId tab of
  362             Nothing -> ccSend chan "inject failed: tab has no session"
  363             Just sidText -> case mkSessionId sidText of
  364               Left _ -> ccSend chan "inject failed: invalid session id"
  365               Right sid -> do
  366                 -- Send the payload to the target session via HTTP.
  367                 -- Do NOT update the session map, send a FocusOp, or
  368                 -- send a focus confirmation — the current focused tab
  369                 -- is unchanged.
  370                 eResult <- httpSend mgr apiBase sid payload
  371                 case eResult of
  372                   Right sr | srKind sr == "error" -> ccSend chan (fromMaybe "error" (srError sr))
  373                   Right sr | not (T.null (srResponse sr)) -> ccSend chan (srResponse sr)
  374                   Right _ -> pure ()  -- assistant response comes via WS
  375                   Left e -> ccSend chan ("error: " <> e)

  377 -- | Ensure a WS connection exists for the conversation. If one exists,
  378 -- send a FocusOp to change focus. If not, start a new WS connection with
  379 -- the streaming event handler and send a FocusOp.
  380 ensureWsConn
  381   :: ChatChannel c
  382   => ChatChannelConfig -> c
  383   -> TVar (Map ConversationKey (WsClient, StreamingState))
  384   -> PendingAsks
  385   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  386   -> WatchState
  387   -> ThinkingTabs
  388   -> ConversationKey -> SessionId
  389   -> IO ()
  390 ensureWsConn cfg chan wsConns pendingAsks tabTracker watchState thinkingTabs key sid = do
  391   let gwCfg = cccGateway cfg
  392   conns <- readTVarIO wsConns
  393   dbg ("[watch] ensureWsConn key=" <> ckConv key <> " hasConn=" <> (case Map.lookup key conns of Just _ -> "true"; Nothing -> "false"))
  394   case Map.lookup key conns of
  395     Just (ws, _) -> wcFocus ws sid  -- already connected; just change focus
  396     Nothing -> do
  397       -- Start a new WS connection with the streaming event handler.
  398       let callback = handleServerEvent cfg chan key wsConns pendingAsks tabTracker watchState thinkingTabs sid
  399       eWs <- startWsClient (gcHost gwCfg) (gcWsPort gwCfg) callback
  400       case eWs of
  401         Left e -> dbg ("[watch] WS connect FAILED: " <> e)
  402         Right ws -> do
  403           ss <- newStreamingState
  404           atomically (modifyTVar' wsConns (Map.insert key (ws, ss)))
  405           wcFocus ws sid

  407 -- | The WS event handler: processes streaming events for one conversation.
  408 -- Creates/edits/finalizes platform messages from entry-update/entry/activity
  409 -- events.
  410 handleServerEvent
  411   :: ChatChannel c
  412   => ChatChannelConfig -> c -> ConversationKey
  413   -> TVar (Map ConversationKey (WsClient, StreamingState))
  414   -> PendingAsks
  415   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  416   -> WatchState
  417   -> ThinkingTabs
  418   -> SessionId -> ServerEvent -> IO ()
  419 handleServerEvent cfg chan key wsConns pendingAsks tabTracker watchState thinkingTabs focusedSid ev =
  420   case ev of
  421     SeEntryUpdate sid val
  422       | sid == focusedSid -> handleEntryUpdate cfg chan key wsConns val
  423     SeEntry sid val
  424       | sid == focusedSid -> handleEntry cfg chan key wsConns val
  425     SeActivity sid val
  426       | sid == focusedSid -> do
  427           dbg ("[watch] SeActivity focused sid=" <> sessionIdText sid <> " kind=" <> extractActivityKind val <> " status=" <> extractActivityStatus val)
  428           handleActivity cfg chan key wsConns val
  429       | otherwise         -> do
  430           dbg ("[watch] SeActivity non-focused sid=" <> sessionIdText sid <> " focusedSid=" <> sessionIdText focusedSid <> " kind=" <> extractActivityKind val <> " status=" <> extractActivityStatus val)
  431           handleWatchActivity cfg chan key watchState thinkingTabs focusedSid sid val
  432     -- Tool-call events are BeActivity with kind="tool-call", broadcast
  433     -- by the server's aeOnToolCall hook. They arrive as SeActivity.
  434     -- handleActivity dispatches on kind internally.
  435     SeAsk sid val
  436       | sid == focusedSid -> handleAsk cfg chan key pendingAsks sid val
  437     SeLists val -> handleLists cfg chan key tabTracker watchState thinkingTabs val
  438     _ -> pure ()  -- ignore events for other sessions or irrelevant types

  440 -- | Check whether the inbound body is a @/watch@ command (bare @/watch@,
  441 -- or @/watch on@ / @/watch off@ / @/watch status@ / @/watch -h@). Pure.
  442 isWatchCommand :: Text -> Bool
  443 isWatchCommand body =
  444   case T.words (T.toLower (T.strip body)) of
  445     ["/watch"]           -> True
  446     ["/watch", "on"]     -> True
  447     ["/watch", "off"]    -> True
  448     ["/watch", "status"] -> True
  449     ["/watch", "-h"]     -> True
  450     _                    -> False

  452 -- | Handle the @/watch@ slash command: toggle (or set) watch-all-tabs
  453 -- mode for the conversation and send a confirmation to the platform.
  454 -- @/watch@ toggles; @/watch on@ and @/watch off@ set explicitly;
  455 -- @/watch status@ prints the current state; @/watch -h@ prints help.
  456 handleWatchToggle
  457   :: ChatChannel c => c -> WatchState -> ConversationKey -> Text -> IO ()
  458 handleWatchToggle chan watchState key body =
  459   case T.words (T.toLower body) of
  460     ["/watch", "on"]  -> setWatch True
  461     ["/watch", "off"] -> setWatch False
  462     ["/watch", "status"] -> do
  463       cur <- lookupWatch watchState key
  464       ccSend chan (watchStatusMsg cur)
  465     ["/watch", "-h"] -> ccSend chan watchHelp
  466     _ -> do  -- bare /watch — toggle
  467       cur <- lookupWatch watchState key
  468       dbg ("[watch] toggle: currently " <> (if cur then "on" else "off"))
  469       newVal <- toggleWatch watchState key
  470       ccSend chan (watchConfirm newVal)
  471   where
  472     setWatch v = do
  473       -- Hold the lock through mutation+persist (same reason as
  474       -- 'toggleWatch').
  475       withMVar (wsLock watchState) $ \_ -> do
  476         atomically (modifyTVar' (wsVar watchState) (Map.insert key v))
  477         persistWatch watchState
  478       ccSend chan (watchConfirm v)

  511 -- | Handle a @harness-status@ activity event for a non-focused session.
  512 -- When watch mode is enabled for the conversation:
  513 --
  514 -- * @thinking@ — record the session as thinking (so we can detect the
  515 --   transition to @idle@).
  516 -- * @idle@ — if the session was previously thinking, remove it from the
  517 --   thinking set and send a \"tab finished thinking\" notification with
  518 --   the last assistant reply.
  519 --
  520 -- When watch mode is disabled, this is a no-op.
  521 handleWatchActivity
  522   :: ChatChannel c
  523   => ChatChannelConfig -> c -> ConversationKey
  524   -> WatchState -> ThinkingTabs
  525   -> SessionId -> SessionId -> Value -> IO ()
  526 handleWatchActivity cfg chan key watchState thinkingTabs _focusedSid sid val = do
  527   watchOn <- lookupWatch watchState key
  528   dbg ("[watch] handleWatchActivity sid=" <> sessionIdText sid <> " watchOn=" <> (if watchOn then "true" else "false"))
  529   when watchOn $ do
  530     let kind = extractActivityKind val
  531     dbg ("[watch] kind=" <> kind <> " status=" <> extractActivityStatus val)
  532     case kind of
  533       "harness-status" -> do
  534         let status = extractActivityStatus val
  535         case status of
  536           "thinking" -> do
  537             dbg ("[watch] adding to thinking set: sid=" <> sessionIdText sid)
  538             atomically (modifyTVar' thinkingTabs (Map.insertWith Set.union key (Set.singleton sid)))
  539           "idle" -> do
  540             wasThinking <- atomically $ do
  541               m <- readTVar thinkingTabs
  542               case Map.lookup key m of
  543                 Just sids | sid `Set.member` sids -> do
  544                   let sids' = Set.delete sid sids
  545                   writeTVar thinkingTabs (if Set.null sids' then Map.delete key m else Map.insert key sids' m)
  546                   pure True
  547                 _ -> pure False
  548             dbg ("[watch] idle received, wasThinking=" <> (if wasThinking then "true" else "false") <> " sid=" <> sessionIdText sid)
  549             when wasThinking $
  550               sendWatchNotification cfg chan sid
  551           _ -> pure ()
  552       _ -> pure ()

  554 -- | Send a \"tab finished thinking\" notification to the channel. Fetches
  555 -- the tab list (to resolve the tab label) and the transcript (to get the
  556 -- last assistant reply). If either fetch fails, the notification is still
  557 -- sent with whatever information is available — the notification must not
  558 -- be silently dropped just because a gateway call failed.
  559 sendWatchNotification
  560   :: ChatChannel c
  561   => ChatChannelConfig -> c -> SessionId -> IO ()
  562 sendWatchNotification cfg chan sid = do
  563   let apiBase = gcApiBase (cccGateway cfg)
  564       mgr = cccHttpManager cfg
  565       sidText = sessionIdText sid
  566   dbg ("[watch] sendWatchNotification sid=" <> sidText)
  567   -- Resolve the tab index (for the "Tab N" prefix) from GET /api/tabs,
  568   -- and the session display title (the same label the web frontend's
  569   -- sidebar shows) from GET /api/sessions. The title cascade mirrors
  570   -- the frontend's sessionDisplayTitle: description, autoSummary,
  571   -- firstMessageSnippet, agent, short id.
  572   tabIdx <- do
  573     eTabs <- httpGetTabs mgr apiBase
  574     case eTabs of
  575       Right tabs ->
  576         case [ t | t <- tabs, tjSessionId t == Just sidText ] of
  577           (t : _) -> pure (case mkTabIndex (tjIndex t) of
  578             Right idx -> T.singleton (tabIndexToChar idx)
  579             Left _    -> T.pack (show (tjIndex t)))
  580           []      -> pure "?"
  581       Left _ -> pure "?"
  582   sessionTitle <- do
  583     eSessions <- httpGetSessions mgr apiBase
  584     case eSessions of
  585       Right sessions ->
  586         case [ s | s <- sessions, sessionJsonId s == Just sidText ] of
  587           (s : _) -> pure (sessionDisplayTitle s)
  588           []      -> pure sidText
  589       Left _ -> pure sidText
  590   mReply <- do
  591     eEntries <- httpGetTranscript mgr apiBase sid
  592     case eEntries of
  593       Right entries -> pure (lastAssistantText entries)
  594       Left _        -> pure Nothing
  595   dbg ("[watch] notification tabIdx=" <> tabIdx <> " sessionTitle=" <> sessionTitle <> " hasReply=" <> (case mReply of Just _ -> "true"; Nothing -> "false"))
  596   let header = "\x1F4D4 Tab " <> tabIdx <> " (" <> sessionTitle <> ") finished thinking"
  597   case mReply of
  598     Just reply | not (T.null reply) -> ccSend chan (header <> ":\n" <> reply)
  599     _ -> ccSend chan header

  601 -- | Extract the @id@ field from a session info JSON object (from
  602 -- @GET /api/sessions@). Returns 'Nothing' when the field is missing or
  603 -- not a string. Pure.
  604 sessionJsonId :: Value -> Maybe Text
  605 sessionJsonId (A.Object o) = asText =<< KeyMap.lookup (Key.fromText "id") o
  606 sessionJsonId _ = Nothing

  608 -- | Derive the display title from a session info JSON object, mirroring
  609 -- the web frontend's @sessionDisplayTitle@ cascade:
  610 -- @description -> autoSummary -> firstMessageSnippet -> agent -> short id@.
  611 -- Pure.
  612 sessionDisplayTitle :: Value -> Text
  613 sessionDisplayTitle val =
  614   case val of
  615     A.Object o ->
  616       fromMaybe (shortId o) (firstNonEmpty
  617         [ asText =<< KeyMap.lookup (Key.fromText "description") o
  618         , asText =<< KeyMap.lookup (Key.fromText "auto_summary") o
  619         , asText =<< KeyMap.lookup (Key.fromText "autoSummary") o
  620         , asText =<< KeyMap.lookup (Key.fromText "first_message_snippet") o
  621         , asText =<< KeyMap.lookup (Key.fromText "firstMessageSnippet") o
  622         , asText =<< KeyMap.lookup (Key.fromText "agent") o
  623         ])
  624     _ -> "?"
  625   where
  626     firstNonEmpty = foldr (\m acc -> case m of
  627       Just t | not (T.null t) -> Just t
  628       _                       -> acc) Nothing
  629     shortId o =
  630       maybe "?" (T.take 12)
  631         (asText =<< KeyMap.lookup (Key.fromText "id") o)

  634 -- | Handle an @entry-update@ event: create or edit the streaming bubble.
  635 handleEntryUpdate
  636   :: ChatChannel c
  637   => ChatChannelConfig -> c -> ConversationKey
  638   -> TVar (Map ConversationKey (WsClient, StreamingState))
  639   -> Value -> IO ()
  640 handleEntryUpdate cfg chan key wsConns val = do
  641   conns <- readTVarIO wsConns
  642   case Map.lookup key conns of
  643     Nothing -> pure ()  -- no streaming state; ignore
  644     Just (_, ss) -> do
  645       let streamCfg = cccStreamCfg cfg
  646           text = extractEntryText val
  647       if T.null text
  648         then pure ()
  649         else do
  650           -- Post-finalize guard (issue #198 follow-up): a late
  651           -- entry-update can arrive AFTER the turn's recorded entry
  652           -- (the server's streaming path and the post-turn broadcast
  653           -- are unsynchronized). Editing now would overwrite the
  654           -- finalized text and re-add the cursor. Ignore updates
  655           -- entirely while the turn is finalized.
  656           finalized <- readIORef (ssFinalized ss)
  657           unless finalized $ do
  658             writeIORef (ssAccumulated ss) text
  659             -- For channels that don't support streaming (Signal), just
  660             -- accumulate the text without sending intermediate edits.
  661             -- The final text is sent as a single message when the
  662             -- recorded entry or idle status arrives (finalizeBubble).
  663             when (ccSupportsStreaming chan) $ do
  664               now <- getCurrentTime
  665               mLastEdit <- readIORef (ssLastEdit ss)
  666               mMsgId <- readIORef (ssMsgId ss)
  667               -- Gate on NEW text since the last edit, not the total length
  668               -- (issue #198, part 3). The accumulator only grows within a
  669               -- response, so a total-length threshold degenerates to an edit
  670               -- on EVERY frame once the text passes spcBufferThreshold —
  671               -- one outbound platform edit per arriving WS frame (the
  672               -- 'lots of small updates' flood). Measuring the delta since
  673               -- the last edit keeps the cadence the config intends: an edit
  674               -- per interval, or per 80 NEW codepoints, whichever first.
  675               lastLen <- readIORef (ssLastLen ss)
  676               when (shouldEdit streamCfg now mLastEdit (T.length text - lastLen)) $ do
  677                 let content = addCursor streamCfg text
  678                 case mMsgId of
  679                   Nothing -> do
  680                     mId <- ccSendWithId chan content
  681                     case mId of
  682                       Just id' -> do
  683                         writeIORef (ssMsgId ss) (Just id')
  684                         writeIORef (ssLastEdit ss) (Just now)
  685                         writeIORef (ssLastLen ss) (T.length text)
  686                       Nothing -> pure ()
  687                   Just id' -> do
  688                     ok <- ccEditMessage chan id' content
  689                     when ok $ do
  690                       writeIORef (ssLastEdit ss) (Just now)
  691                       writeIORef (ssLastLen ss) (T.length text)
  692 
  693 -- | Finalize the current streaming bubble for one conversation: edit it
  694 -- without the cursor (or, if the edit fails, send the full text as a new
  695 -- message), then reset the bubble state so the next entry-update starts
  696 -- a fresh bubble. Does nothing when no bubble exists or nothing was
  697 -- streamed. When @markFinal@ is 'True' (the turn's recorded response
  698 -- entry — the definitive delivery), sets 'ssFinalized' so late
  699 -- entry-updates are ignored until the next turn starts.
  700 finalizeBubble
  701   :: ChatChannel c
  702   => ChatChannelConfig -> c -> StreamingState -> Text -> Bool -> IO ()
  703 finalizeBubble cfg chan ss text markFinal = do
  704   let streamCfg = cccStreamCfg cfg
  705       finalText = stripCursor streamCfg text
  706   mMsgId <- readIORef (ssMsgId ss)
  707   delivered <- case mMsgId of
  708     Nothing -> isJust <$> ccSendWithId chan finalText
  709     Just id' -> do
  710       ok <- ccEditMessage chan id' finalText
  711       if ok
  712         then pure True
  713         else isJust <$> ccSendWithId chan finalText
  714   when (markFinal && delivered) $ writeIORef (ssFinalized ss) True
  715   resetStreamingState ss

  717 -- | Handle an @entry@ event (complete transcript entry): finalize the
  718 -- streaming bubble with the definitive text (mark the turn final).
  719 handleEntry
  720   :: ChatChannel c
  721   => ChatChannelConfig -> c -> ConversationKey
  722   -> TVar (Map ConversationKey (WsClient, StreamingState))
  723   -> Value -> IO ()
  724 handleEntry cfg chan key wsConns val = do
  725   -- Only process response entries (skip request entries which would
  726   -- incorrectly finalize the streaming bubble with the user's text).
  727   let direction = extractDirection val
  728   if direction /= "response"
  729     then pure ()
  730     else do
  731       conns <- readTVarIO wsConns
  732       case Map.lookup key conns of
  733         Nothing -> pure ()
  734         Just (_, ss) -> do
  735           let text = extractEntryText val
  736           if T.null text
  737             then pure ()
  738             else do
  739               -- The recorded entry is the definitive delivery: mark the
  740               -- turn finalized so late entry-updates are ignored.
  741               -- For non-streaming channels, the recorded entry's text
  742               -- is the authoritative full response — send it even if
  743               -- we already sent the accumulated text via idle (the
  744               -- recorded entry is the canonical source). Skip if we
  745               -- already delivered via finalizeBubble (mMsgId was set
  746               -- and finalized).
  747               alreadyDelivered <- readIORef (ssFinalized ss)
  748               if alreadyDelivered
  749                 then pure ()
  750                 else finalizeBubble cfg chan ss text True

  752 -- | Handle an @activity@ event: if harness-status is idle, finalize any
  753 -- in-progress streaming bubble.
  754 handleActivity
  755   :: ChatChannel c
  756   => ChatChannelConfig -> c -> ConversationKey
  757   -> TVar (Map ConversationKey (WsClient, StreamingState))
  758   -> Value -> IO ()
  759 handleActivity cfg chan key wsConns val = do
  760   let kind = extractActivityKind val
  761   case kind of
  762     "harness-status" -> handleHarnessStatus cfg chan key wsConns val
  763     "tool-call" -> handleToolCallActivity cfg chan key wsConns val
  764     _ -> pure ()

  766 -- | Handle a @harness-status@ activity: on @thinking@ (turn start),
  767 -- clear the finalized flag so the new turn streams. On @idle@ (turn
  768 -- end), finalize any in-progress streaming bubble with the accumulated
  769 -- text (a safety net — the recorded entry is the normal finalize path)
  770 -- and reset all streaming state.
  771 handleHarnessStatus
  772   :: ChatChannel c
  773   => ChatChannelConfig -> c -> ConversationKey
  774   -> TVar (Map ConversationKey (WsClient, StreamingState))
  775   -> Value -> IO ()
  776 handleHarnessStatus cfg chan key wsConns val = do
  777     let status = extractActivityStatus val
  778     case status of
  779       "thinking" -> do
  780         conns <- readTVarIO wsConns
  781         for_ (Map.lookup key conns) $ \(_, ss) ->
  782           writeIORef (ssFinalized ss) False
  783       "idle" -> do
  784         conns <- readTVarIO wsConns
  785         case Map.lookup key conns of
  786           Nothing -> pure ()
  787           Just (_, ss) -> do
  788             mMsgId <- readIORef (ssMsgId ss)
  789             accum <- readIORef (ssAccumulated ss)
  790             -- For non-streaming channels (Signal), the accumulated text
  791             -- is the full response. Send it as a new message (mMsgId is
  792             -- Nothing because no streaming bubble was created). For
  793             -- streaming channels, only finalize if there's an active
  794             -- streaming bubble (mMsgId is Just).
  795             -- Don't mark as finalized — the recorded entry is the
  796             -- canonical finalize path. This is just a safety net.
  797             when (not (T.null accum) &&
  798                   (not (ccSupportsStreaming chan) || isJust mMsgId)) $
  799               finalizeBubble cfg chan ss accum False
  800             resetStreamingState ss
  801             writeIORef (ssFinalized ss) True
  802       _ -> pure ()

  804 -- | Handle a @tool-call@ activity: finalize the current text bubble
  805 -- (segment break — the pre-tool text stays its own message, and the
  806 -- next entry-update starts a NEW bubble below the tool line), then send
  807 -- the tool-progress line as its own platform message.
  808 handleToolCallActivity
  809   :: ChatChannel c
  810   => ChatChannelConfig -> c -> ConversationKey
  811   -> TVar (Map ConversationKey (WsClient, StreamingState))
  812   -> Value -> IO ()
  813 handleToolCallActivity cfg chan key wsConns val = do
  814   -- Segment break (issue #198 follow-up, symptom 1): the pre-tool
  815   -- streamed text and the post-tool streamed text must be SEPARATE
  816   -- platform messages, with the tool line between them. Without this,
  817   -- the next entry-update edits the pre-tool bubble (ssMsgId is still
  818   -- set) and appends the post-tool text onto the pre-tool text.
  819   conns <- readTVarIO wsConns
  820   for_ (Map.lookup key conns) $ \(_, ss) -> do
  821     mMsgId <- readIORef (ssMsgId ss)
  822     accum <- readIORef (ssAccumulated ss)
  823     -- For non-streaming channels, finalize when there's accumulated text
  824     -- even without a streaming bubble (mMsgId is Nothing). For streaming
  825     -- channels, only finalize when there's an in-progress bubble.
  826     when (not (T.null accum) &&
  827           (not (ccSupportsStreaming chan) || isJust mMsgId)) $
  828       finalizeBubble cfg chan ss accum False
  829   case extractToolName val of
  830     Nothing -> pure ()
  831     Just toolName -> do
  832       let mInput = extractToolInput val
  833           line = formatToolLine mempty toolName (fromMaybe "" mInput)
  834       ccSend chan line

  912 -- | Resolve the conversation's session. If the conversation has no session
  913 -- yet, create one via the HTTP API.
  914 resolveSession
  915   :: ChatChannel c => ChatChannelConfig -> c -> SessionMap -> TVar (Map ConversationKey (WsClient, StreamingState))
  916   -> PendingAsks -> TVar (Map ConversationKey (SessionId, [TabJson]))
  917   -> WatchState -> ThinkingTabs
  918   -> ConversationKey
  919   -> IO SessionId
  920 resolveSession cfg chan sessions wsConns pendingAsks tabTracker watchState thinkingTabs key = do
  921   mSid <- sessionLookup sessions key
  922   case mSid of
  923     Just sid -> pure sid
  924     Nothing -> do
  925       let apiBase = gcApiBase (cccGateway cfg)
  926           mgr = cccHttpManager cfg

  928       case eSid of
  929         Right sidText -> case mkSessionId sidText of
  930           Right sid -> do
  931             sessionInsert sessions key sid
  932             ensureWsConn cfg chan wsConns pendingAsks tabTracker watchState thinkingTabs key sid

  934           Left _ -> fallbackSid

  951 -- | Handle a callback query (button tap) from the platform. The callback_data
  952 -- is @"<8hex>:<index>"@. Resolves the 8-hex prefix to a pending ask, resolves
  953 -- the index to the option label, answers via the HTTP API, acknowledges the
  954 -- callback (dismiss the spinner), removes the keyboard, and sends a
  955 -- @✓ <label>@ confirmation. If no pending ask matches, the callback is
  956 -- silently dropped (stale button tap).
  957 handleCallback
  958   :: ChatChannel c
  959   => ChatChannelConfig -> c -> SessionMap -> PendingAsks
  960   -> ConversationKey -> MessageSource -> Text
  961   -> IO ()
  962 handleCallback cfg chan sessions pendingAsks key _src cbData = do
  963   case parseCallbackData cbData of
  964     Nothing -> pure ()  -- not a valid callback_data format
  965     Just (prefix, idx) -> do
  966       mSid <- sessionLookup sessions key
  967       case mSid of
  968         Nothing -> pure ()
  969         Just sid -> do
  970           asksMap <- readTVarIO pendingAsks
  971           case Map.lookup sid asksMap of
  972             Nothing -> pure ()
  973             Just asks ->
  974               case [ a | a <- asks, T.isPrefixOf prefix (paAskId a) ] of
  975                 (ask : _) ->
  976                   case atIndex (paOptions ask) idx of
  977                     Nothing -> pure ()
  978                     Just opt -> do
  979                       let apiBase = gcApiBase (cccGateway cfg)
  980                           mgr = cccHttpManager cfg
  981                       -- Answer the question via the HTTP API.
  982                       _ <- httpAnswerQuestion mgr apiBase sid (paAskId ask) (qoLabel opt) "once"
  983                       -- Acknowledge the callback (dismiss spinner).
  984                       -- We don't have the callback_query_id here (it's in
  985                       -- the InboundMessage but not passed through). The
  986                       -- ccAnswerCallback method needs it. For now, skip
  987                       -- the callback ack — the button still works, the
  988                       -- spinner just persists a bit longer.
  989                       -- Send a confirmation to the chat.
  990                       ccSend chan ("✓ " <> qoLabel opt)
  991                       let remaining = filter (not . T.isPrefixOf prefix . paAskId) asks
  992                       atomically (modifyTVar' pendingAsks (Map.insert sid remaining))
  993                 [] -> pure ()

  995 -- | Handle a @lists@ WS event: compare the new tab list with the last known
  996 -- one for this conversation. If the focused session's tab was removed (closed),
  997 -- send the "tab closed" notification and clear the tracking state so the next
  998 -- message creates a fresh tab. Also seeds the 'ThinkingTabs' set from the
  999 -- snapshot's @thinkingSessionIds@ so tabs already thinking before the WS
 1000 -- connection was established still trigger a finished notification.
 1001 handleLists
 1002   :: ChatChannel c
 1003   => ChatChannelConfig -> c -> ConversationKey
 1004   -> TVar (Map ConversationKey (SessionId, [TabJson]))
 1005   -> WatchState -> ThinkingTabs -> Value -> IO ()
 1006 handleLists cfg chan key tabTracker watchState thinkingTabs val = do
 1007   -- Seed the thinking-tabs set from the snapshot's thinkingSessionIds.
 1008   -- This is the core fix for the "missed thinking event" problem: when a
 1009   -- WS connection is established (or a lists snapshot is broadcast on tab
 1010   -- changes), sessions that were already thinking before the connection
 1011   -- existed are included in the snapshot. Without this seeding, the
 1012   -- ThinkingTabs set starts empty and the thinking→idle transition is
 1013   -- never detected for those sessions (the idle event arrives but
 1014   -- wasThinking is false → no notification).
 1015   seedThinkingTabsFromLists watchState thinkingTabs key val
 1016   -- Fetch the current tabs from the gateway to get an accurate list.
 1017   let apiBase = gcApiBase (cccGateway cfg)
 1018       mgr = cccHttpManager cfg
 1019   eTabs <- httpGetTabs mgr apiBase
 1020   case eTabs of
 1021     Left _ -> pure ()
 1022     Right newTabs -> do
 1023       tracker <- readTVarIO tabTracker
 1024       case Map.lookup key tracker of
 1025         Nothing -> pure ()  -- no tracked session for this conversation
 1026         Just (focusedSid, _oldTabs) -> do
 1027           -- Check if the focused session's tab was removed.
 1028           let newSessionIds = mapMaybe tjSessionId newTabs
 1029           if focusedSid `elem` mapMaybe mkSessionId' newSessionIds
 1030             then do
 1031               -- Tab still exists; update the tracked tab list.
 1032               atomically (modifyTVar' tabTracker (Map.insert key (focusedSid, newTabs)))
 1033             else do
 1034               -- The focused session's tab was closed. Send the notification.
 1035               ccSend chan ("tab closed (session " <> sessionIdText focusedSid
 1036                         <> "); a new tab will be created on your next message")
 1037               -- Clear the tracking state so the next message creates a fresh tab.
 1038               atomically (modifyTVar' tabTracker (Map.delete key))
 1039   where
 1040     mkSessionId' t = case mkSessionId t of Right s -> Just s; Left _ -> Nothing
 1041 
 1042 -- | Seed the 'ThinkingTabs' set for a conversation from the
 1043 -- @thinkingSessionIds@ field in a @lists@ WS event payload. This ensures
 1044 -- sessions that were already thinking before the WS connection was
 1045 -- established are tracked, so their eventual @idle@ transition triggers a
 1046 -- watch notification. Replaces the per-conversation thinking set (rather
 1047 -- than unioning) so sessions that finished between snapshots are not kept
 1048 -- stale — the next @thinking@ activity event will re-add them if needed.
 1049 -- No-op when watch mode is off (the thinking set is only consulted by
 1050 -- 'handleWatchActivity', which itself checks watch mode, but seeding
 1051 -- when off would waste memory for no benefit).
 1052 seedThinkingTabsFromLists :: WatchState -> ThinkingTabs -> ConversationKey -> Value -> IO ()
 1053 seedThinkingTabsFromLists watchState thinkingTabs key val = do
 1054   watchOn <- lookupWatch watchState key
 1055   when watchOn $ do
 1056     let sids = extractThinkingSessionIds val
 1057     unless (null sids) $ do
 1058       dbg ("[watch] seeding thinking tabs from lists snapshot: key=" <> ckConv key
 1059         <> " count=" <> T.pack (show (length sids)))
 1060       atomically (modifyTVar' thinkingTabs (Map.insert key (Set.fromList sids)))
 1061 
 1062 -- | Extract the @thinkingSessionIds@ array from a @lists@ WS event
 1063 -- payload. Returns the list of valid 'SessionId's. Pure.
 1064 extractThinkingSessionIds :: Value -> [SessionId]
 1065 extractThinkingSessionIds val =
 1066   case val of
 1067     A.Object o -> case KeyMap.lookup (Key.fromText "thinkingSessionIds") o of

 1074 -- | Parse callback_data of the form @"<8hex>:<index>"@. Returns 'Nothing'
 1075 -- for malformed data. Pure.
 1076 parseCallbackData :: Text -> Maybe (Text, Int)
 1077 parseCallbackData cbData =
 1078   case T.splitOn ":" cbData of
 1079     [prefix, idxTxt]
 1080       | T.length prefix == 8 && T.all isHexChar prefix
 1081         -> case T.unpack idxTxt of
 1082              [] -> Nothing
 1083              s  -> case reads s of
 1084                      [(n, "")] | n >= 0 -> Just (prefix, n)
 1085                      _ -> Nothing
 1086     _ -> Nothing
 1087   where
 1088     isHexChar c = isDigit c || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F')

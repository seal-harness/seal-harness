  104 -- | Run the generic chat-channel loop. Blocks until the channel's
  105 -- 'ccReceive' returns EOF. Each conversation gets its own WS connection
  106 -- for streaming.
  107 runChatChannel :: ChatChannel c => ChatChannelConfig -> c -> IO ()
  108 runChatChannel cfg chan = do
  109   sessions <- newSessionMap
  110   -- Map of conversation keys to their WS client + streaming state.
  111   wsConns <- newTVarIO Map.empty :: IO (TVar (Map ConversationKey (WsClient, StreamingState)))
  112   pendingAsks <- newTVarIO Map.empty :: IO PendingAsks
  113   tabTracker <- newTVarIO Map.empty :: IO (TVar (Map ConversationKey (SessionId, [TabJson])))
  114   loop sessions wsConns pendingAsks tabTracker
  115   where
  116     loop sessions wsConns pendingAsks tabTracker = do
  117       mMsg <- ccReceive chan
  118       case mMsg of
  119         Nothing -> pure ()  -- EOF
  120         Just (InboundMessage src body mCbData) -> do
  121           dbg ("received: " <> body)
  122           let key = convKeyFromSource src
  123           case mCbData of
  124             Just cbData -> do
  125               -- Callback query (button tap): resolve to a pending ask,
  126               -- answer via HTTP, acknowledge, and remove keyboard.
  127               handleCallback cfg chan sessions pendingAsks key src cbData
  128             Nothing ->
  129               -- Regular text message: route normally.
  130               handleInbound cfg chan sessions wsConns pendingAsks tabTracker key body
  131           loop sessions wsConns pendingAsks tabTracker

  133 -- | Handle one inbound message: resolve the session, route, and dispatch.
  134 handleInbound
  135   :: ChatChannel c
  136   => ChatChannelConfig -> c -> SessionMap
  137   -> TVar (Map ConversationKey (WsClient, StreamingState))
  138   -> PendingAsks
  139   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  140   -> ConversationKey -> Text
  141   -> IO ()
  142 handleInbound cfg chan sessions wsConns pendingAsks tabTracker key body = do
  143   let apiBase = gcApiBase (cccGateway cfg)
  144       mgr = cccHttpManager cfg
  145   -- First, check if this is a /tab focus N (intercepted locally).
  146   case parseTabFocus body of
  147     Just idx -> do
  148       handleFocus cfg chan sessions wsConns pendingAsks tabTracker key idx
  149     Nothing -> case route body of
  150       Right (ChatFocus idx) ->
  151         handleFocus cfg chan sessions wsConns pendingAsks tabTracker key idx
  152       Right (ChatInject idx payload) -> do
  153         -- Focus the tab, then send the payload as a plain message.
  154         handleFocus cfg chan sessions wsConns pendingAsks tabTracker key idx
  155         sendPlain cfg chan sessions wsConns pendingAsks tabTracker key payload
  156       Right ChatCurrentTab -> do
  157         -- Send the current tab info via HTTP (the gateway routes /tab).
  158         mSid <- sessionLookup sessions key
  159         case mSid of
  160           Just sid -> do
  161             eResult <- httpSend mgr apiBase sid "/tab"
  162             case eResult of
  163               Right sr -> ccSend chan (srResponse sr)
  164               Left e -> ccSend chan ("error: " <> e)
  165           Nothing -> ccSend chan "no current tab"
  166       Right (ChatNewSession args) -> do
  167         -- Create a new session via HTTP, update the session map.
  168         handleNewSession cfg chan sessions wsConns pendingAsks tabTracker key args
  169       Right (ChatSlash _cmd) ->
  170         sendSlash cfg chan sessions wsConns pendingAsks tabTracker key body
  171       Right (ChatTabCommand _) ->
  172         -- Tab commands go through the HTTP API (the gateway routes them).
  173         sendSlash cfg chan sessions wsConns pendingAsks tabTracker key body
  174       Right (ChatPlain text) ->
  175         sendPlain cfg chan sessions wsConns pendingAsks tabTracker key text
  176       Left _ -> ccSend chan "error: invalid command"

  178 -- | Handle a focus command: resolve the tab index to a session id via
  179 -- @GET /api/tabs@, send a @FocusOp@ over WS, send "focused tab N" to the
  180 -- platform, and optionally send the last assistant reply.
  181 handleFocus
  182   :: ChatChannel c
  183   => ChatChannelConfig -> c -> SessionMap
  184   -> TVar (Map ConversationKey (WsClient, StreamingState))
  185   -> PendingAsks
  186   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  187   -> ConversationKey -> TabIndex
  188   -> IO ()
  189 handleFocus cfg chan sessions wsConns pendingAsks tabTracker key idx = do
  190   let apiBase = gcApiBase (cccGateway cfg)
  191       mgr = cccHttpManager cfg
  192   eTabs <- httpGetTabs mgr apiBase
  193   case eTabs of
  194     Left e -> ccSend chan ("focus failed: " <> e)
  195     Right tabs ->
  196       case [ t | t <- tabs, tjIndex t == tabIndexToInt idx ] of
  197         [] -> ccSend chan "focus failed: tab index out of range"
  198         (tab : _) ->
  199           case tjSessionId tab of
  200             Nothing -> ccSend chan "focus failed: tab has no session"
  201             Just sidText -> case mkSessionId sidText of
  202               Left _ -> ccSend chan "focus failed: invalid session id"
  203               Right sid -> do
  204                 -- Update the session map so subsequent messages route here.
  205                 sessionInsert sessions key sid
  206                 -- Track the focused session + current tab list for tab-close detection.
  207                 atomically (modifyTVar' tabTracker (Map.insert key (sid, tabs)))
  208                 -- Ensure a WS connection exists for this conversation.
  209                 ensureWsConn cfg chan wsConns pendingAsks tabTracker key sid
  210                 -- Send "focused tab N" confirmation.
  211                 ccSend chan ("focused tab " <> T.singleton (tabIndexToChar idx))
  212                 -- Fetch and send the last assistant reply for context.
  213                 sendLastReply cfg chan sid

  215 -- | Ensure a WS connection exists for the conversation. If one exists,
  216 -- send a FocusOp to change focus. If not, start a new WS connection with
  217 -- the streaming event handler and send a FocusOp.
  218 ensureWsConn
  219   :: ChatChannel c
  220   => ChatChannelConfig -> c
  221   -> TVar (Map ConversationKey (WsClient, StreamingState))
  222   -> PendingAsks
  223   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  224   -> ConversationKey -> SessionId
  225   -> IO ()
  226 ensureWsConn cfg chan wsConns pendingAsks tabTracker key sid = do
  227   let gwCfg = cccGateway cfg
  228   conns <- readTVarIO wsConns
  229   case Map.lookup key conns of
  230     Just (ws, _) -> wcFocus ws sid  -- already connected; just change focus
  231     Nothing -> do
  232       -- Start a new WS connection with the streaming event handler.
  233       let callback = handleServerEvent cfg chan key wsConns pendingAsks tabTracker sid
  234       eWs <- startWsClient (gcHost gwCfg) (gcWsPort gwCfg) callback
  235       case eWs of
  236         Left _ -> pure ()  -- WS failed; the channel still works via HTTP
  237         Right ws -> do
  238           ss <- newStreamingState
  239           atomically (modifyTVar' wsConns (Map.insert key (ws, ss)))
  240           wcFocus ws sid

  242 -- | The WS event handler: processes streaming events for one conversation.
  243 -- Creates/edits/finalizes platform messages from entry-update/entry/activity
  244 -- events.
  245 handleServerEvent
  246   :: ChatChannel c
  247   => ChatChannelConfig -> c -> ConversationKey
  248   -> TVar (Map ConversationKey (WsClient, StreamingState))
  249   -> PendingAsks
  250   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  251   -> SessionId -> ServerEvent -> IO ()
  252 handleServerEvent cfg chan key wsConns pendingAsks tabTracker focusedSid ev =
  253   case ev of
  254     SeEntryUpdate sid val
  255       | sid == focusedSid -> handleEntryUpdate cfg chan key wsConns val
  256     SeEntry sid val
  257       | sid == focusedSid -> handleEntry cfg chan key wsConns val
  258     SeActivity sid val
  259       | sid == focusedSid -> handleActivity cfg chan key wsConns val
  260     -- Tool-call events are BeActivity with kind="tool-call", broadcast
  261     -- by the server's aeOnToolCall hook. They arrive as SeActivity.
  262     -- handleActivity dispatches on kind internally.
  263     SeAsk sid val
  264       | sid == focusedSid -> handleAsk cfg chan key pendingAsks sid val
  265     SeLists val -> handleLists cfg chan key tabTracker val
  266     _ -> pure ()  -- ignore events for other sessions or irrelevant types

  268 -- | Handle an @entry-update@ event: create or edit the streaming bubble.
  269 handleEntryUpdate
  270   :: ChatChannel c
  271   => ChatChannelConfig -> c -> ConversationKey
  272   -> TVar (Map ConversationKey (WsClient, StreamingState))
  273   -> Value -> IO ()
  274 handleEntryUpdate cfg chan key wsConns val = do
  275   conns <- readTVarIO wsConns
  276   case Map.lookup key conns of
  277     Nothing -> pure ()  -- no streaming state; ignore
  278     Just (_, ss) -> do
  279       let streamCfg = cccStreamCfg cfg
  280           text = extractEntryText val
  281       if T.null text
  282         then pure ()
  283         else do
  284           -- Post-finalize guard (issue #198 follow-up): a late
  285           -- entry-update can arrive AFTER the turn's recorded entry
  286           -- (the server's streaming path and the post-turn broadcast
  287           -- are unsynchronized). Editing now would overwrite the
  288           -- finalized text and re-add the cursor. Ignore updates
  289           -- entirely while the turn is finalized.
  290           finalized <- readIORef (ssFinalized ss)
  291           unless finalized $ do
  292             writeIORef (ssAccumulated ss) text
  293             -- For channels that don't support streaming (Signal), just
  294             -- accumulate the text without sending intermediate edits.
  295             -- The final text is sent as a single message when the
  296             -- recorded entry or idle status arrives (finalizeBubble).
  297             when (ccSupportsStreaming chan) $ do
  298               now <- getCurrentTime
  299               mLastEdit <- readIORef (ssLastEdit ss)
  300               mMsgId <- readIORef (ssMsgId ss)
  301               -- Gate on NEW text since the last edit, not the total length
  302               -- (issue #198, part 3). The accumulator only grows within a
  303               -- response, so a total-length threshold degenerates to an edit
  304               -- on EVERY frame once the text passes spcBufferThreshold —
  305               -- one outbound platform edit per arriving WS frame (the
  306               -- 'lots of small updates' flood). Measuring the delta since
  307               -- the last edit keeps the cadence the config intends: an edit
  308               -- per interval, or per 80 NEW codepoints, whichever first.
  309               lastLen <- readIORef (ssLastLen ss)
  310               when (shouldEdit streamCfg now mLastEdit (T.length text - lastLen)) $ do
  311                 let content = addCursor streamCfg text
  312                 case mMsgId of
  313                   Nothing -> do
  314                     mId <- ccSendWithId chan content
  315                     case mId of
  316                       Just id' -> do
  317                         writeIORef (ssMsgId ss) (Just id')
  318                         writeIORef (ssLastEdit ss) (Just now)
  319                         writeIORef (ssLastLen ss) (T.length text)
  320                       Nothing -> pure ()
  321                   Just id' -> do
  322                     ok <- ccEditMessage chan id' content
  323                     when ok $ do
  324                       writeIORef (ssLastEdit ss) (Just now)
  325                       writeIORef (ssLastLen ss) (T.length text)

  327 -- | Finalize the current streaming bubble for one conversation: edit it
  328 -- without the cursor (or, if the edit fails, send the full text as a new
  329 -- message), then reset the bubble state so the next entry-update starts
  330 -- a fresh bubble. Does nothing when no bubble exists or nothing was
  331 -- streamed. When @markFinal@ is 'True' (the turn's recorded response
  332 -- entry — the definitive delivery), sets 'ssFinalized' so late
  333 -- entry-updates are ignored until the next turn starts.
  334 finalizeBubble
  335   :: ChatChannel c
  336   => ChatChannelConfig -> c -> StreamingState -> Text -> Bool -> IO ()
  337 finalizeBubble cfg chan ss text markFinal = do
  338   let streamCfg = cccStreamCfg cfg
  339       finalText = stripCursor streamCfg text
  340   mMsgId <- readIORef (ssMsgId ss)
  341   delivered <- case mMsgId of
  342     Nothing -> isJust <$> ccSendWithId chan finalText
  343     Just id' -> do
  344       ok <- ccEditMessage chan id' finalText
  345       if ok
  346         then pure True
  347         else isJust <$> ccSendWithId chan finalText
  348   when (markFinal && delivered) $ writeIORef (ssFinalized ss) True
  349   resetStreamingState ss

  351 -- | Handle an @entry@ event (complete transcript entry): finalize the
  352 -- streaming bubble with the definitive text (mark the turn final).
  353 handleEntry
  354   :: ChatChannel c
  355   => ChatChannelConfig -> c -> ConversationKey
  356   -> TVar (Map ConversationKey (WsClient, StreamingState))
  357   -> Value -> IO ()
  358 handleEntry cfg chan key wsConns val = do
  359   -- Only process response entries (skip request entries which would
  360   -- incorrectly finalize the streaming bubble with the user's text).
  361   let direction = extractDirection val
  362   if direction /= "response"
  363     then pure ()
  364     else do
  365       conns <- readTVarIO wsConns
  366       case Map.lookup key conns of
  367         Nothing -> pure ()
  368         Just (_, ss) -> do
  369           let text = extractEntryText val
  370           if T.null text
  371             then pure ()
  372             else do
  373               -- The recorded entry is the definitive delivery: mark the
  374               -- turn finalized so late entry-updates are ignored.
  375               -- For non-streaming channels, the recorded entry's text
  376               -- is the authoritative full response — send it even if
  377               -- we already sent the accumulated text via idle (the
  378               -- recorded entry is the canonical source). Skip if we
  379               -- already delivered via finalizeBubble (mMsgId was set
  380               -- and finalized).
  381               alreadyDelivered <- readIORef (ssFinalized ss)
  382               if alreadyDelivered
  383                 then pure ()
  384                 else finalizeBubble cfg chan ss text True

  386 -- | Handle an @activity@ event: if harness-status is idle, finalize any
  387 -- in-progress streaming bubble.
  388 handleActivity
  389   :: ChatChannel c
  390   => ChatChannelConfig -> c -> ConversationKey

  393 handleActivity cfg chan key wsConns val = do
  394   let kind = extractActivityKind val

  396     "harness-status" -> handleHarnessStatus cfg chan key wsConns val
  397     "tool-call" -> handleToolCallActivity cfg chan key wsConns val

  400 -- | Handle a @harness-status@ activity: on @thinking@ (turn start),
  401 -- clear the finalized flag so the new turn streams. On @idle@ (turn
  402 -- end), finalize any in-progress streaming bubble with the accumulated
  403 -- text (a safety net — the recorded entry is the normal finalize path)
  404 -- and reset all streaming state.
  405 handleHarnessStatus
  406   :: ChatChannel c
  407   => ChatChannelConfig -> c -> ConversationKey
  408   -> TVar (Map ConversationKey (WsClient, StreamingState))
  409   -> Value -> IO ()
  410 handleHarnessStatus cfg chan key wsConns val = do
  411     let status = extractActivityStatus val
  412     case status of
  413       "thinking" -> do
  414         conns <- readTVarIO wsConns
  415         for_ (Map.lookup key conns) $ \(_, ss) ->
  416           writeIORef (ssFinalized ss) False
  417       "idle" -> do
  418         conns <- readTVarIO wsConns
  419         case Map.lookup key conns of
  420           Nothing -> pure ()
  421           Just (_, ss) -> do
  422             mMsgId <- readIORef (ssMsgId ss)
  423             accum <- readIORef (ssAccumulated ss)
  424             -- For non-streaming channels (Signal), the accumulated text
  425             -- is the full response. Send it as a new message (mMsgId is
  426             -- Nothing because no streaming bubble was created). For
  427             -- streaming channels, only finalize if there's an active
  428             -- streaming bubble (mMsgId is Just).
  429             -- Don't mark as finalized — the recorded entry is the
  430             -- canonical finalize path. This is just a safety net.
  431             when (not (T.null accum) &&
  432                   (not (ccSupportsStreaming chan) || isJust mMsgId)) $
  433               finalizeBubble cfg chan ss accum False
  434             resetStreamingState ss
  435             writeIORef (ssFinalized ss) True
  436       _ -> pure ()

  438 -- | Handle a @tool-call@ activity: finalize the current text bubble
  439 -- (segment break — the pre-tool text stays its own message, and the
  440 -- next entry-update starts a NEW bubble below the tool line), then send
  441 -- the tool-progress line as its own platform message.
  442 handleToolCallActivity
  443   :: ChatChannel c
  444   => ChatChannelConfig -> c -> ConversationKey
  445   -> TVar (Map ConversationKey (WsClient, StreamingState))
  446   -> Value -> IO ()
  447 handleToolCallActivity cfg chan key wsConns val = do
  448   -- Segment break (issue #198 follow-up, symptom 1): the pre-tool
  449   -- streamed text and the post-tool streamed text must be SEPARATE
  450   -- platform messages, with the tool line between them. Without this,
  451   -- the next entry-update edits the pre-tool bubble (ssMsgId is still
  452   -- set) and appends the post-tool text onto the pre-tool text.
  453   conns <- readTVarIO wsConns
  454   for_ (Map.lookup key conns) $ \(_, ss) -> do
  455     mMsgId <- readIORef (ssMsgId ss)
  456     accum <- readIORef (ssAccumulated ss)
  457     -- For non-streaming channels, finalize when there's accumulated text
  458     -- even without a streaming bubble (mMsgId is Nothing). For streaming
  459     -- channels, only finalize when there's an in-progress bubble.
  460     when (not (T.null accum) &&
  461           (not (ccSupportsStreaming chan) || isJust mMsgId)) $
  462       finalizeBubble cfg chan ss accum False
  463   case extractToolName val of
  464     Nothing -> pure ()
  465     Just toolName -> do
  466       let mInput = extractToolInput val
  467           line = formatToolLine mempty toolName (fromMaybe "" mInput)
  468       ccSend chan line

  470 -- | Handle an @ask@ event: render the question on the platform. When the
  471 -- ask has options and the channel supports inline keyboards, sends the
  472 -- question + an inline keyboard (one button per option). Otherwise falls
  473 -- back to the numbered-list text rendering. Registers the pending ask so
  474 -- the callback handler can resolve a button tap to the option label.
  475 handleAsk
  476   :: ChatChannel c
  477   => ChatChannelConfig -> c -> ConversationKey -> PendingAsks -> SessionId -> Value
  478   -> IO ()
  479 handleAsk _cfg chan _key pendingAsks sid val = do
  480   let question = extractAskQuestion val
  481       askId = extractAskId val
  482       opts = extractAskOptions val
  483   if null opts
  484     then ccSend chan question
  485     else do
  486       let prefix = T.take 8 askId
  487       mMsgId <- ccSendWithOptions chan question opts prefix
  488       case mMsgId of
  489         Just _  -> pure ()
  490         Nothing -> ccSend chan (formatQuestionWithOptions question opts)
  491       let entry = PendingAsk { paAskId = askId, paOptions = opts }
  492       atomically (modifyTVar' pendingAsks (Map.insertWith (++) sid [entry]))

  494 -- | Send a plain text message via the HTTP API. Resolves the conversation's
  495 -- session (creating one if it doesn't exist yet).
  496 sendPlain
  497   :: ChatChannel c
  498   => ChatChannelConfig -> c -> SessionMap -> TVar (Map ConversationKey (WsClient, StreamingState))
  499   -> PendingAsks -> TVar (Map ConversationKey (SessionId, [TabJson]))
  500   -> ConversationKey -> Text
  501   -> IO ()
  502 sendPlain cfg chan sessions wsConns pendingAsks tabTracker key text = do
  503   sid <- resolveSession cfg chan sessions wsConns pendingAsks tabTracker key
  504   let apiBase = gcApiBase (cccGateway cfg)
  505       mgr = cccHttpManager cfg
  506   eResult <- httpSend mgr apiBase sid text
  507   case eResult of
  508     Right sr | srKind sr == "error" -> ccSend chan (fromMaybe "error" (srError sr))
  509     Right sr | not (T.null (srResponse sr)) -> ccSend chan (srResponse sr)
  510     Right _ -> pure ()  -- assistant response comes via WS
  511     Left e -> ccSend chan ("error: " <> e)

  522 -- | Handle /new: create a new session via HTTP, update the session map.
  523 handleNewSession
  524   :: ChatChannel c
  525   => ChatChannelConfig -> c -> SessionMap -> TVar (Map ConversationKey (WsClient, StreamingState))
  526   -> PendingAsks -> TVar (Map ConversationKey (SessionId, [TabJson]))
  527   -> ConversationKey -> Text
  528   -> IO ()
  529 handleNewSession cfg chan sessions wsConns pendingAsks tabTracker key _args = do
  530   let apiBase = gcApiBase (cccGateway cfg)
  531       mgr = cccHttpManager cfg
  532   eSid <- httpNewSession mgr apiBase (A.object [])
  533   case eSid of
  534     Left e -> ccSend chan ("new session failed: " <> e)
  535     Right sidText -> case mkSessionId sidText of
  536       Left _ -> ccSend chan "new session failed: invalid session id"
  537       Right sid -> do
  538         sessionInsert sessions key sid
  539         -- Ensure a WS connection exists for this conversation.
  540         ensureWsConn cfg chan wsConns pendingAsks tabTracker key sid
  541         ccSend chan ("new session " <> sessionIdText sid)

  543 -- | Resolve the conversation's session. If the conversation has no session
  544 -- yet, create one via the HTTP API.
  545 resolveSession
  546   :: ChatChannel c => ChatChannelConfig -> c -> SessionMap -> TVar (Map ConversationKey (WsClient, StreamingState))
  547   -> PendingAsks -> TVar (Map ConversationKey (SessionId, [TabJson]))
  548   -> ConversationKey
  549   -> IO SessionId
  550 resolveSession cfg chan sessions wsConns pendingAsks tabTracker key = do
  551   mSid <- sessionLookup sessions key
  552   case mSid of
  553     Just sid -> pure sid
  554     Nothing -> do
  555       let apiBase = gcApiBase (cccGateway cfg)
  556           mgr = cccHttpManager cfg
  557       eSid <- httpNewSession mgr apiBase (A.object [])
  558       case eSid of
  559         Right sidText -> case mkSessionId sidText of
  560           Right sid -> do
  561             sessionInsert sessions key sid
  562             ensureWsConn cfg chan wsConns pendingAsks tabTracker key sid
  563             pure sid
  564           Left _ -> fallbackSid
  565         Left _ -> fallbackSid
  566   where
  567     fallbackSid = pure (case mkSessionId "default" of Right s -> s; Left _ -> error "unreachable")

  569 -- | Fetch and send the last assistant reply from the session's transcript.
  570 sendLastReply
  571   :: ChatChannel c
  572   => ChatChannelConfig -> c -> SessionId -> IO ()
  573 sendLastReply cfg chan sid = do
  574   let apiBase = gcApiBase (cccGateway cfg)
  575       mgr = cccHttpManager cfg
  576   eEntries <- httpGetTranscript mgr apiBase sid
  577   case eEntries of
  578     Left _ -> pure ()
  579     Right entries -> for_ (lastAssistantText entries) (ccSend chan)

  581 -- | Handle a callback query (button tap) from the platform. The callback_data
  582 -- is @"<8hex>:<index>"@. Resolves the 8-hex prefix to a pending ask, resolves
  583 -- the index to the option label, answers via the HTTP API, acknowledges the
  584 -- callback (dismiss the spinner), removes the keyboard, and sends a
  585 -- @✓ <label>@ confirmation. If no pending ask matches, the callback is
  586 -- silently dropped (stale button tap).
  587 handleCallback
  588   :: ChatChannel c
  589   => ChatChannelConfig -> c -> SessionMap -> PendingAsks
  590   -> ConversationKey -> MessageSource -> Text
  591   -> IO ()
  592 handleCallback cfg chan sessions pendingAsks key _src cbData = do
  593   case parseCallbackData cbData of
  594     Nothing -> pure ()  -- not a valid callback_data format
  595     Just (prefix, idx) -> do
  596       mSid <- sessionLookup sessions key
  597       case mSid of
  598         Nothing -> pure ()
  599         Just sid -> do
  600           asksMap <- readTVarIO pendingAsks
  601           case Map.lookup sid asksMap of
  602             Nothing -> pure ()
  603             Just asks ->
  604               case [ a | a <- asks, T.isPrefixOf prefix (paAskId a) ] of
  605                 (ask : _) ->
  606                   case atIndex (paOptions ask) idx of
  607                     Nothing -> pure ()
  608                     Just opt -> do
  609                       let apiBase = gcApiBase (cccGateway cfg)
  610                           mgr = cccHttpManager cfg
  611                       -- Answer the question via the HTTP API.
  612                       _ <- httpAnswerQuestion mgr apiBase sid (paAskId ask) (qoLabel opt) "once"
  613                       -- Acknowledge the callback (dismiss spinner).
  614                       -- We don't have the callback_query_id here (it's in
  615                       -- the InboundMessage but not passed through). The
  616                       -- ccAnswerCallback method needs it. For now, skip
  617                       -- the callback ack — the button still works, the
  618                       -- spinner just persists a bit longer.
  619                       -- Send a confirmation to the chat.
  620                       ccSend chan ("✓ " <> qoLabel opt)
  621                       let remaining = filter (not . T.isPrefixOf prefix . paAskId) asks
  622                       atomically (modifyTVar' pendingAsks (Map.insert sid remaining))
  623                 [] -> pure ()
  624 
  625 -- | Handle a @lists@ WS event: compare the new tab list with the last known
  626 -- one for this conversation. If the focused session's tab was removed (closed),
  627 -- send the "tab closed" notification and clear the tracking state so the next
  628 -- message creates a fresh tab.
  629 handleLists
  630   :: ChatChannel c
  631   => ChatChannelConfig -> c -> ConversationKey
  632   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  633   -> Value -> IO ()
  634 handleLists cfg chan key tabTracker _val = do
  635   -- Fetch the current tabs from the gateway to get an accurate list.
  636   let apiBase = gcApiBase (cccGateway cfg)
  637       mgr = cccHttpManager cfg
  638   eTabs <- httpGetTabs mgr apiBase
  639   case eTabs of
  640     Left _ -> pure ()
  641     Right newTabs -> do
  642       tracker <- readTVarIO tabTracker
  643       case Map.lookup key tracker of
  644         Nothing -> pure ()  -- no tracked session for this conversation
  645         Just (focusedSid, _oldTabs) -> do
  646           -- Check if the focused session's tab was removed.
  647           let newSessionIds = mapMaybe tjSessionId newTabs
  648           if focusedSid `elem` mapMaybe mkSessionId' newSessionIds
  649             then do
  650               -- Tab still exists; update the tracked tab list.
  651               atomically (modifyTVar' tabTracker (Map.insert key (focusedSid, newTabs)))
  652             else do
  653               -- The focused session's tab was closed. Send the notification.
  654               ccSend chan ("tab closed (session " <> sessionIdText focusedSid
  655                         <> "); a new tab will be created on your next message")
  656               -- Clear the tracking state so the next message creates a fresh tab.
  657               atomically (modifyTVar' tabTracker (Map.delete key))
  658   where
  659     mkSessionId' t = case mkSessionId t of Right s -> Just s; Left _ -> Nothing

  661 -- | Parse callback_data of the form @"<8hex>:<index>"@. Returns 'Nothing'
  662 -- for malformed data. Pure.
  663 parseCallbackData :: Text -> Maybe (Text, Int)
  664 parseCallbackData cbData =
  665   case T.splitOn ":" cbData of
  666     [prefix, idxTxt]
  667       | T.length prefix == 8 && T.all isHexChar prefix
  668         -> case T.unpack idxTxt of
  669              [] -> Nothing
  670              s  -> case reads s of
  671                      [(n, "")] | n >= 0 -> Just (prefix, n)
  672                      _ -> Nothing
  673     _ -> Nothing
  674   where
  675     isHexChar c = isDigit c || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F')
  676 
  677 -- | Safe indexing: return the element at position @n@ or 'Nothing'.
  678 atIndex :: [a] -> Int -> Maybe a
  679 atIndex [] _ = Nothing
  680 atIndex (x:_) 0 = Just x
  681 atIndex (_:xs) n = atIndex xs (n - 1)

  690 -- | Extract the @options@ array from an @ask@ event payload. Each option
  691 -- is a JSON object with @label@ and @description@ fields.
  692 extractAskOptions :: Value -> [QuestionOption]
  693 extractAskOptions val =
  694   case val of
  695     A.Object o -> case KeyMap.lookup (Key.fromText "options") o of
  696       Just (A.Array arr) -> mapMaybe parseOption (foldr (:) [] arr)
  697       _ -> []
  698     _ -> []
  699   where
  700     parseOption (A.Object ob) =
  701       QuestionOption
  702         <$> (asText =<< KeyMap.lookup (Key.fromText "label") ob)
  703         <*> pure (fromMaybe "" (asText =<< KeyMap.lookup (Key.fromText "description") ob))
  704     parseOption _ = Nothing

  706 -- | Format a question + its options as a numbered list for chat channels
  707 -- that don't support inline keyboards (Signal, CLI). Mirrors
  708 -- 'Seal.Handles.AskReply.formatQuestionWithOptions'. Pure.
  709 formatQuestionWithOptions :: Text -> [QuestionOption] -> Text
  710 formatQuestionWithOptions question [] = question
  711 formatQuestionWithOptions question opts =
  712   question
  713   <> "\n\n"
  714   <> T.intercalate "\n" (zipWith formatLine [1 :: Int ..] opts)
  715   <> "\n\nReply with a number or type your own answer."
  716   where
  717     formatLine n (QuestionOption lbl desc)
  718       | T.null desc = T.pack (show n <> ") ") <> lbl
  719       | otherwise   = T.pack (show n <> ") ") <> lbl <> " — " <> desc

  725 -- | Extract the accumulated text from an @entry-update@ or @entry@ event's
  726 -- @entry@ payload. The payload is a JSON object with a @payload@ field
  727 -- containing the message content.
  728 extractEntryText :: Value -> Text
  729 extractEntryText val =
  730   case val of
  731     A.Object o -> case KeyMap.lookup (Key.fromText "payload") o of
  732       Just (A.Object p) -> case KeyMap.lookup (Key.fromText "content") p of
  733         Just (A.Array arr) ->
  734           -- content is an array of {text: "...", type: "text"} blocks
  735           T.concat (mapMaybe extractTextBlock (toList arr))
  736         Just (A.String t) -> t
  737         _ -> ""
  738       Just (A.String t) -> t
  739       _ -> ""
  740     _ -> ""
  741   where
  742     extractTextBlock (A.Object b) = case KeyMap.lookup (Key.fromText "text") b of
  743       Just (A.String t) -> Just t
  744       _ -> Nothing
  745     extractTextBlock _ = Nothing
  746     toList = foldr (:) []

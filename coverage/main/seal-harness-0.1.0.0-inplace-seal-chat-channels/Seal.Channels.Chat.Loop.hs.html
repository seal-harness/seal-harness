  135 -- | Per-conversation watch-all-tabs toggle state. When 'True' for a
  136 -- conversation, the channel receives a notification every time any
  137 -- non-focused tab finishes thinking (harness-status transitions from
  138 -- @thinking@ to @idle@). Thread-safe via 'TVar'.
  139 data WatchState = WatchState
  140   { wsVar  :: TVar (Map ConversationKey Bool)
  141   , wsSave :: Maybe (IO ())
  142   -- ^ The save action (snapshots the TVar and writes to disk). Called
  143   -- inside the 'wsLock' critical section so the snapshot is consistent
  144   -- with the mutation.
  145   , wsLock :: MVar ()
  146   -- ^ Serializes the mutation+persist sequence so a stale snapshot from
  147   -- one thread cannot overwrite a newer on-disk write from another.
  148   -- Without this lock, concurrent toggles from Signal + Telegram (which
  149   -- share one 'WatchState') can lose state after a restart.
  150   }
  151 
  152 -- | Create a new empty watch-state map (watch mode off for all
  153 -- conversations).
  154 newWatchState :: IO WatchState
  155 newWatchState = do
  156   tv <- newTVarIO Map.empty
  157   lock <- newMVar ()
  158   pure (WatchState tv Nothing lock)
  159 
  160 -- | Create a persisting watch-state store. The supplied save function is
  161 -- called after every mutation (snapshotting the full current map, so the
  162 -- last writer wins with a consistent view). Mirrors
  163 -- 'Seal.Channels.Cursor.newPersistingCursorStore'.
  164 newPersistingWatchState :: (Map ConversationKey Bool -> IO ()) -> IO WatchState
  165 newPersistingWatchState saveFn = do
  166   tv <- newTVarIO Map.empty
  167   lock <- newMVar ()
  168   let ws = WatchState tv (Just (saveAction ws)) lock
  169       saveAction s = saveFn =<< snapshotWatch s
  170   pure ws
  171 
  172 -- | Replace the store's map in one STM transaction. Used at boot to seed
  173 -- the in-memory store from the persisted @watch_state.json@. Does NOT
  174 -- persist (the caller is loading FROM disk).
  175 seedWatchState :: WatchState -> Map ConversationKey Bool -> IO ()
  176 seedWatchState ws m = atomically (writeTVar (wsVar ws) m)
  177 
  178 -- | Snapshot the current map. Used by the save action.
  179 snapshotWatch :: WatchState -> IO (Map ConversationKey Bool)
  180 snapshotWatch ws = readTVarIO (wsVar ws)

  182 -- | Look up whether watch mode is enabled for a conversation. 'False'
  183 -- when the conversation has no entry (the default).
  184 lookupWatch :: WatchState -> ConversationKey -> IO Bool
  185 lookupWatch ws key = fromMaybe False . Map.lookup key <$> readTVarIO (wsVar ws)
  186 
  187 -- | Toggle watch mode for a conversation. Returns the new state.
  188 toggleWatch :: WatchState -> ConversationKey -> IO Bool
  189 toggleWatch ws key = do
  190   -- Hold the lock through mutation+persist so no concurrent thread can
  191   -- mutate the TVar between our snapshot and our save (which would let a
  192   -- stale snapshot overwrite a newer on-disk write).
  193   withMVar (wsLock ws) $ \_ -> do
  194     newVal <- atomically $ do
  195       m <- readTVar (wsVar ws)
  196       let v = not (fromMaybe False (Map.lookup key m))
  197       writeTVar (wsVar ws) (Map.insert key v m)
  198       pure v
  199     persistWatch ws
  200     pure newVal
  201 
  202 -- | Run the persist action (if any) after a successful mutation. A save
  203 -- failure is logged to stderr and swallowed — the in-memory store stays
  204 -- authoritative within the session; the next successful mutation will
  205 -- retry the save (writing the full current map, so a missed save
  206 -- self-heals). Mirrors 'Seal.Channels.Cursor.persistCursor'.
  207 persistWatch :: WatchState -> IO ()
  208 persistWatch ws =
  209   case wsSave ws of
  210     Nothing  -> pure ()
  211     Just act -> act `catch` \e ->
  212       dbg ("[watch] watch_state.json save failed: " <> T.pack (show (e :: SomeException)))

  227 -- A bg session is a freshly minted, headless session (no tab) plus the

  238 -- | The per-conversation background-session store. Each conversation's
  239 -- list is in creation order (oldest first) so the per-conversation cap can
  240 -- evict the oldest session when a new one is started. Thread-safe via
  241 -- 'TVar'.
  242 --
  243 -- The store also remembers the session ids this conversation has owned for
  244 -- a bounded window after they are retired ('bsKnown'). That memory is what
  245 -- keeps a bg turn out of the watch-all-tabs path: 'BeActivity' is broadcast
  246 -- to every subscriber, so the conversation's own connection receives
  247 -- @harness-status idle@ / @tool-call@ for the bg session too -- possibly
  248 -- after the bg connection's reader has already retired it. Without the
  249 -- memory those late events would be classified as "not a bg session" and
  250 -- produce a bogus "Tab ?" watch notification plus a duplicate reply.
  251 data BgSessions = BgSessions
  252   { bsLive  :: TVar (Map ConversationKey [BgSession])
  253   , bsKnown :: TVar (Map ConversationKey [SessionId])
  254   }
  255 
  256 -- | How many retired bg session ids a conversation remembers for the watch
  257 -- exclusion. Bounded so a long-lived conversation cannot grow without
  258 -- limit; the window only has to outlive one turn's activity broadcasts.
  259 maxKnownBgSids :: Int
  260 maxKnownBgSids = 32
  261 
  262 -- | Create a new empty background-session store.
  263 newBgSessions :: IO BgSessions
  264 newBgSessions = BgSessions <$> newTVarIO Map.empty <*> newTVarIO Map.empty

  266 -- | The conversation's live background sessions, oldest first.
  267 bgSessionsFor :: BgSessions -> ConversationKey -> IO [BgSession]
  268 bgSessionsFor bgs key =
  269   fromMaybe [] . Map.lookup key <$> readTVarIO (bsLive bgs)

  271 -- | Does @sid@ belong to one of this conversation's background sessions
  272 -- (live, or retired within the remembered window)? Used to keep bg turns
  273 -- out of the watch-all-tabs path.
  274 bgOwnsSession :: BgSessions -> ConversationKey -> SessionId -> IO Bool
  275 bgOwnsSession bgs key sid =
  276   elem sid . fromMaybe [] . Map.lookup key <$> readTVarIO (bsKnown bgs)

  278 -- | Register a background session at the end of its conversation's list
  279 -- (creation order) and remember its id for the watch exclusion.
  280 bgInsert :: BgSessions -> ConversationKey -> BgSession -> IO ()
  281 bgInsert bgs key sess = do
  282   atomically (modifyTVar' (bsLive bgs)
  283     (Map.insertWith (flip (++)) key [sess]))
  284   atomically (modifyTVar' (bsKnown bgs)
  285     (Map.insertWith (\new old -> take maxKnownBgSids (new ++ filter (`notElem` new) old))
  286        key [bgSid sess]))
  287 
  288 -- | Untrack a background session. Returns the session when it was tracked,
  289 -- 'Nothing' when it was already gone (so retirements are idempotent).
  290 -- Retiring does NOT forget the session id (see 'bsKnown'): late activity
  291 -- events for it still must stay out of the watch path.
  292 bgRemove :: BgSessions -> ConversationKey -> SessionId -> IO (Maybe BgSession)
  293 bgRemove bgs key sid = atomically $ do
  294   let tv = bsLive bgs
  295   m <- readTVar tv
  296   let sessions = fromMaybe [] (Map.lookup key m)
  297       (dropped, kept) = partition ((== sid) . bgSid) sessions
  298       m' = case kept of
  299              [] -> Map.delete key m
  300              _  -> Map.insert key kept m
  301   writeTVar tv m'
  302   pure (case dropped of
  303           (s : _) -> Just s
  304           []      -> Nothing)

  306 -- | Drop every pending ask registered for a session. A retired bg session
  307 -- can no longer be answered, so its asks must not accumulate (or be
  308 -- resolvable by a later button tap).
  309 dropPendingAsks :: PendingAsks -> SessionId -> IO ()
  310 dropPendingAsks pendingAsks sid =
  311   atomically (modifyTVar' (paStore pendingAsks) (Map.delete sid))

  383 -- | Run the generic chat-channel loop. Blocks until the channel's
  384 -- 'ccReceive' returns EOF. Each conversation gets its own WS connection
  385 -- for streaming.
  386 runChatChannel :: ChatChannel c => ChatChannelConfig -> c -> WatchState -> IO ()
  387 runChatChannel cfg chan watchState = do
  388   sessions <- newSessionMap
  389   -- Map of conversation keys to their WS client + streaming state.
  390   wsConns <- newTVarIO Map.empty :: IO (TVar (Map ConversationKey WsConn))
  391   pendingAsks <- newPendingAsks
  392   tabTracker <- newTVarIO Map.empty :: IO (TVar (Map ConversationKey (SessionId, [TabJson])))
  393   thinkingTabs <- newThinkingTabs
  394   bgSessions <- newBgSessions
  395   loop sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs
  396   where
  397     loop sessions wsConns pendingAsks bgSessions tabTracker ws tt = do
  398       mMsg <- ccReceive chan
  399       case mMsg of
  400         Nothing -> pure ()  -- EOF
  401         Just (InboundMessage src body mCbData) -> do
  402           dbg ("received: " <> body)
  403           let key = convKeyFromSource src
  404           case mCbData of
  405             Just cbData -> do
  406               -- Callback query (button tap): resolve to a pending ask,
  407               -- answer via HTTP, acknowledge, and remove keyboard.
  408               handleCallback cfg chan sessions pendingAsks bgSessions key src cbData
  409             Nothing ->
  410               -- Regular text message: route normally.
  411               handleInbound cfg chan sessions wsConns pendingAsks bgSessions tabTracker ws tt key body
  412           loop sessions wsConns pendingAsks bgSessions tabTracker ws tt

  414 -- | Handle one inbound message: resolve the session, route, and dispatch.
  415 handleInbound
  416   :: ChatChannel c
  417   => ChatChannelConfig -> c -> SessionMap
  418   -> TVar (Map ConversationKey WsConn)
  419   -> PendingAsks
  420   -> BgSessions
  421   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  422   -> WatchState
  423   -> ThinkingTabs
  424   -> ConversationKey -> Text
  425   -> IO ()
  426 handleInbound cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key body = do
  427   let apiBase = gcApiBase (cccGateway cfg)
  428       mgr = cccHttpManager cfg
  429   -- Intercept /watch (and /watch on|off) before normal routing — it
  430   -- toggles per-conversation loop state, not a gateway command.
  431   if isWatchCommand body
  432     then do
  433       dbg ("[watch] /watch command received, key=" <> ckConv key)
  434       handleWatchToggle chan watchState key body
  435       -- When watch mode is turned ON, ensure a WS connection exists so
  436       -- the channel receives BeActivity events for all tabs. Without
  437       -- this, /watch as the first message would enable watch mode but
  438       -- never receive any events (no WS connection = no event source).
  439       watchOn <- lookupWatch watchState key
  440       dbg ("[watch] watchOn=" <> (if watchOn then "true" else "false") <> ", ensuring WS conn")
  441       when watchOn $
  442         void (resolveSession cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key)
  443     else case parseTabFocus body of
  444     Just idx -> do
  445       handleFocus cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key idx
  446     Nothing -> case route body of
  447       Right (ChatFocus idx) ->
  448         handleFocus cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key idx
  449       Right (ChatInject idx payload) -> do
  450         -- Send the payload to the target tab's session WITHOUT changing
  451         -- the focused tab (no FocusOp, no "focused tab N" confirmation).
  452         handleInject cfg chan idx payload
  453       Right ChatCurrentTab -> do
  454         -- Send the current tab info via HTTP (the gateway routes /tab).
  455         mSid <- sessionLookup sessions key
  456         case mSid of
  457           Just sid -> do
  458             eResult <- cccSendMessage cfg mgr apiBase sid "/tab"
  459             case eResult of
  460               Right sr -> ccSend chan (srResponse sr)
  461               Left e -> ccSend chan ("error: " <> e)
  462           Nothing -> ccSend chan "no current tab"
  463       Right (ChatNewSession args) -> do
  464         -- Create a new session via HTTP, update the session map.
  465         handleNewSession cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key args
  466       Right (ChatBackground prompt) -> do
  467         -- Run the prompt in a freshly minted headless session. The
  468         -- conversation's focused session and tab list are untouched.
  469         handleBackground cfg chan pendingAsks bgSessions key prompt
  470       Right (ChatSlash _cmd) ->
  471         sendSlash cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key body
  472       Right (ChatTabCommand _) ->
  473         -- Tab commands go through the HTTP API (the gateway routes them).
  474         sendSlash cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key body
  475       Right ChatTabList -> handleTabList cfg chan
  476       Right (ChatPlain text) -> do
  477         -- A plain message answers a pending ask from one of this
  478         -- conversation's background sessions, if there is one. Checked
  479         -- here (not before routing) so a /command is never consumed as an
  480         -- answer.
  481         consumed <- answerPendingBgAsk cfg chan pendingAsks bgSessions key text
  482         unless consumed $
  483           sendPlain cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key text
  484       Left _ -> ccSend chan "error: invalid command"

  486 -- | Handle a focus command: resolve the tab index to a session id via
  487 -- @GET /api/tabs@, send a @FocusOp@ over WS, send "focused tab N" to the
  488 -- platform, and optionally send the last assistant reply.
  489 handleFocus
  490   :: ChatChannel c
  491   => ChatChannelConfig -> c -> SessionMap
  492   -> TVar (Map ConversationKey WsConn)
  493   -> PendingAsks
  494   -> BgSessions
  495   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  496   -> WatchState
  497   -> ThinkingTabs
  498   -> ConversationKey -> TabIndex
  499   -> IO ()
  500 handleFocus cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key idx = do
  501   let apiBase = gcApiBase (cccGateway cfg)
  502       mgr = cccHttpManager cfg
  503   eTabs <- httpGetTabs mgr apiBase
  504   case eTabs of
  505     Left e -> ccSend chan ("focus failed: " <> e)
  506     Right tabs ->
  507       case [ t | t <- tabs, tjIndex t == tabIndexToInt idx ] of
  508         [] -> ccSend chan "focus failed: tab index out of range"
  509         (tab : _) ->
  510           case tjSessionId tab of
  511             Nothing -> ccSend chan "focus failed: tab has no session"
  512             Just sidText -> case mkSessionId sidText of
  513               Left _ -> ccSend chan "focus failed: invalid session id"
  514               Right sid -> do
  515                 -- Update the session map so subsequent messages route here.
  516                 sessionInsert sessions key sid
  517                 -- Track the focused session + current tab list for tab-close detection.
  518                 atomically (modifyTVar' tabTracker (Map.insert key (sid, tabs)))
  519                 -- Ensure a WS connection exists for this conversation.
  520                 ensureWsConn cfg chan wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key sid
  521                 -- Send "focused tab N" confirmation.
  522                 ccSend chan ("focused tab " <> T.singleton (tabIndexToChar idx))
  523                 -- Fetch and send the last assistant reply for context.
  524                 sendLastReply cfg chan sid

  526 -- | Handle an inject command (@\/N payload@): resolve the tab index to a
  527 -- session id via @GET /api/tabs@, then send the payload to that session
  528 -- via @POST /api/sessions/:id/send@. Unlike 'handleFocus', this does NOT
  529 -- change the focused tab, send a FocusOp over WS, update the session map,
  530 -- send a "focused tab N" confirmation, or send the last assistant reply.
  531 -- The payload is delivered to the target session; any response will be
  532 -- visible when the user later focuses on that tab.
  533 handleInject
  534   :: ChatChannel c
  535   => ChatChannelConfig -> c -> TabIndex -> Text
  536   -> IO ()

  538   let apiBase = gcApiBase (cccGateway cfg)
  539       mgr = cccHttpManager cfg
  540   eTabs <- httpGetTabs mgr apiBase
  541   case eTabs of
  542     Left e -> ccSend chan ("inject failed: " <> e)
  543     Right tabs ->
  544       case [ t | t <- tabs, tjIndex t == tabIndexToInt idx ] of
  545         [] -> ccSend chan "inject failed: tab index out of range"
  546         (tab : _) ->
  547           case tjSessionId tab of
  548             Nothing -> ccSend chan "inject failed: tab has no session"
  549             Just sidText -> case mkSessionId sidText of
  550               Left _ -> ccSend chan "inject failed: invalid session id"
  551               Right sid -> do
  552                 -- Send the payload to the target session via HTTP.
  553                 -- Do NOT update the session map, send a FocusOp, or
  554                 -- send a focus confirmation — the current focused tab
  555                 -- is unchanged.
  556                 eResult <- cccSendMessage cfg mgr apiBase sid payload
  557                 case eResult of
  558                   Right sr | srKind sr == "error" -> ccSend chan (fromMaybe "error" (srError sr))
  559                   Right sr | not (T.null (srResponse sr)) -> ccSend chan (srResponse sr)
  560                   Right _ -> pure ()  -- assistant response comes via WS
  561                   Left e -> ccSend chan ("error: " <> e)

  598 -- | Handle @/bg <prompt>@: mint a headless session (no tab), open a
  599 -- dedicated WS connection focused on it, register it, and run the prompt on
  600 -- it. The reply is posted back to this conversation by
  601 -- 'handleBgServerEvent'; the conversation's focused session, session map,
  602 -- and tab tracking are never touched.
  603 --
  604 -- A blank prompt answers 'bgUsage' and creates nothing; @-h@ / @--help@
  605 -- answers 'bgHelp' and creates nothing. Every failure (mint, connect, send)
  606 -- closes whatever it opened, untracks the session, drops its pending asks,
  607 -- and reports on the channel.
  608 handleBackground
  609   :: ChatChannel c
  610   => ChatChannelConfig -> c -> PendingAsks -> BgSessions
  611   -> ConversationKey -> Text
  612   -> IO ()
  613 handleBackground cfg chan pendingAsks bgSessions key prompt
  614   | T.null (T.strip prompt) = ccSend chan bgUsage
  615   | isBgHelpArg prompt      = ccSend chan bgHelp
  616   | otherwise               = do
  617       let apiBase = gcApiBase (cccGateway cfg)
  618           mgr = cccHttpManager cfg
  619           gwCfg = cccGateway cfg
  620       -- (a) mint a headless session, (b) validate its id.
  621       eMinted <- cccNewSession cfg mgr apiBase (A.object ["headless" .= A.Bool True])
  622       case eMinted >>= mkSessionId of
  623         Left e -> ccSend chan ("bg session failed: " <> e)
  624         Right bgSid -> do
  625           -- (c) per-session streaming state + delivery flag.
  626           ss <- newStreamingState
  627           delivered <- newIORef False
  628           let callback =
  629                 handleBgServerEvent cfg chan key pendingAsks bgSessions bgSid ss delivered
  630           -- (d) the dedicated connection for the bg session.
  631           eWs <- cccStartWs cfg (gcHost gwCfg) (gcWsPort gwCfg) callback
  632           case eWs of
  633             Left e -> ccSend chan ("bg session failed: " <> e)
  634             Right ws -> do
  635               -- (e) focus before registering or sending, so no turn event
  636               -- can arrive for an untracked session (which would leak the
  637               -- connection and skip the watch exclusion).
  638               wcFocus ws bgSid
  639               evictOldestBgSession pendingAsks bgSessions key
  640               bgInsert bgSessions key BgSession { bgSid = bgSid, bgWs = ws }
  641               -- (f) run the prompt on the bg session (never on the
  642               -- conversation's session) and confirm.
  643               eSent <- cccSendMessage cfg mgr apiBase bgSid prompt
  644               case eSent of
  645                 Right sr | srKind sr /= "error" ->
  646                   ccSend chan ("bg session " <> shortSessionId bgSid <> " started")
  647                 failure -> do
  648                   _ <- bgRemove bgSessions key bgSid
  649                   wcClose ws
  650                   dropPendingAsks pendingAsks bgSid
  651                   ccSend chan ("bg session failed: " <> bgSendFailure failure)
  652   where
  653     bgSendFailure (Left e)   = e
  654     bgSendFailure (Right sr) = fromMaybe "send failed" (srError sr)

  656 -- | Enforce the per-conversation cap: when the conversation already holds
  657 -- 'maxBgSessions' live bg sessions, retire the oldest one so a further
  658 -- session can start.
  659 evictOldestBgSession :: PendingAsks -> BgSessions -> ConversationKey -> IO ()
  660 evictOldestBgSession pendingAsks bgSessions key = do
  661   existing <- bgSessionsFor bgSessions key
  662   when (length existing >= maxBgSessions) $
  663     case existing of
  664       (oldest : _) -> void (retireBgSession pendingAsks bgSessions key (bgSid oldest))

  667 -- | Retire one background session: close its WS connection, untrack it, and
  668 -- drop its pending asks. Returns 'True' when it was still tracked, so a
  669 -- repeated @idle@ is a no-op.
  670 retireBgSession
  671   :: PendingAsks -> BgSessions -> ConversationKey -> SessionId -> IO Bool
  672 retireBgSession pendingAsks bgSessions key sid = do
  673   mSess <- bgRemove bgSessions key sid
  674   for_ mSess (wcClose . bgWs)
  675   dropPendingAsks pendingAsks sid
  676   pure (isJust mSess)

  678 -- | The WS event handler for a background session's dedicated connection.
  679 -- Acts only on events for 'bgSid':
  680 --
  681 -- * a @response@ 'SeEntry' — post the recorded reply to the conversation and
  682 --   mark it delivered. Token deltas are deliberately not streamed for bg
  683 --   turns (pre-regression parity: the reply arrives on the recorded entry).
  684 -- * 'SeAsk' — render the question on the conversation (and register it as
  685 --   pending, so a button tap can answer it).
  686 -- * @tool-call@ activity — post the tool-progress line.
  687 -- * @harness-status@ @idle@ — retire the session (close, untrack, drop its
  688 --   asks) and report a turn that produced no reply.
  689 -- * everything else (entry-updates, list snapshots, replay markers, other
  690 --   sessions) — ignored.
  691 handleBgServerEvent
  692   :: ChatChannel c
  693   => ChatChannelConfig -> c -> ConversationKey -> PendingAsks -> BgSessions
  694   -> SessionId -> StreamingState -> IORef Bool -> ServerEvent -> IO ()
  695 handleBgServerEvent cfg chan key pendingAsks bgSessions bgSid ss delivered ev =
  696   case ev of
  697     SeEntry sid val
  698       | sid == bgSid
  699       , extractDirection val == "response"
  700       , not (T.null (extractEntryText val)) -> do
  701           handleEntry cfg chan ss val
  702           writeIORef delivered True
  703     SeAsk sid val
  704       | sid == bgSid -> handleAsk cfg chan key pendingAsks sid val
  705     SeAskResolved _ val -> forgetAskByPrefix pendingAsks (extractAskId val)
  706     SeActivity sid val
  707       | sid == bgSid -> case extractActivityKind val of
  708           "tool-call" -> handleToolCallActivity cfg chan ss val
  709           "harness-status"
  710             | extractActivityStatus val == "idle" -> do
  711                 -- The close trigger is harness-status idle only (not
  712                 -- reply-delivered, which also fires at every turn end when
  713                 -- sub-agent children keep the session thinking).
  714                 wasTracked <- retireBgSession pendingAsks bgSessions key bgSid
  715                 when wasTracked $ do
  716                   wasDelivered <- readIORef delivered
  717                   unless wasDelivered $
  718                     ccSend chan ("bg session " <> shortSessionId bgSid <> " produced no reply")
  719           _ -> pure ()
  720     _ -> pure ()

  722 -- | Ensure a WS connection exists for the conversation. If one exists,
  723 -- send a FocusOp to change focus. If not, start a new WS connection with
  724 -- the streaming event handler and send a FocusOp.
  725 --
  726 -- A connection is only reusable for the session it was built for.
  727 -- 'handleServerEvent' closes over the focused session **by value**, so a
  728 -- FocusOp cannot re-target the filter: an existing connection focused on
  729 -- session A keeps dropping session B's entries and asks. Re-targeting
  730 -- therefore closes the old connection and builds a new one whose callback
  731 -- is bound to the new session.
  732 ensureWsConn
  733   :: ChatChannel c
  734   => ChatChannelConfig -> c
  735   -> TVar (Map ConversationKey WsConn)
  736   -> PendingAsks
  737   -> BgSessions
  738   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  739   -> WatchState
  740   -> ThinkingTabs
  741   -> ConversationKey -> SessionId
  742   -> IO ()
  743 ensureWsConn cfg chan wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key sid = do
  744   conns <- readTVarIO wsConns
  745   dbg ("[watch] ensureWsConn key=" <> ckConv key <> " hasConn=" <> (case Map.lookup key conns of Just _ -> "true"; Nothing -> "false"))
  746   case Map.lookup key conns of
  747     Just (boundSid, ws, _)
  748       | boundSid == sid -> wcFocus ws sid  -- already connected; just change focus
  749       -- Re-target: the callback closed over 'boundSid' by value, so a
  750       -- FocusOp alone would leave this conversation deaf to 'sid'.
  751       | otherwise -> do
  752           dbg ("[watch] ensureWsConn re-target " <> sessionIdText boundSid
  753                <> " -> " <> sessionIdText sid)
  754           wcClose ws
  755           atomically (modifyTVar' wsConns (Map.delete key))
  756           connectWs
  757     Nothing -> do
  758       connectWs
  759   where
  760     -- Start a new WS connection whose event handler is bound to 'sid'.
  761     connectWs = do
  762       let gwCfg = cccGateway cfg
  763           callback = handleServerEvent cfg chan key wsConns pendingAsks
  764             bgSessions tabTracker watchState thinkingTabs sid
  765       eWs <- cccStartWs cfg (gcHost gwCfg) (gcWsPort gwCfg) callback
  766       case eWs of
  767         Left e -> dbg ("[watch] WS connect FAILED: " <> e)
  768         Right ws -> do
  769           ss <- newStreamingState
  770           atomically (modifyTVar' wsConns (Map.insert key (sid, ws, ss)))
  771           wcFocus ws sid

  773 -- | The WS event handler: processes streaming events for one conversation.
  774 -- Creates/edits/finalizes platform messages from entry-update/entry/activity
  775 -- events.
  776 handleServerEvent
  777   :: ChatChannel c
  778   => ChatChannelConfig -> c -> ConversationKey
  779   -> TVar (Map ConversationKey WsConn)
  780   -> PendingAsks
  781   -> BgSessions
  782   -> TVar (Map ConversationKey (SessionId, [TabJson]))
  783   -> WatchState
  784   -> ThinkingTabs
  785   -> SessionId -> ServerEvent -> IO ()
  786 handleServerEvent cfg chan key wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs focusedSid ev =
  787   case ev of
  788     SeEntryUpdate sid val
  789       | sid == focusedSid ->
  790           withStreamingState wsConns key (\ss -> handleEntryUpdate cfg chan ss val)
  791     SeEntry sid val
  792       | sid == focusedSid ->
  793           withStreamingState wsConns key (\ss -> handleEntry cfg chan ss val)
  794     SeActivity sid val
  795       | sid == focusedSid -> do
  796           dbg ("[watch] SeActivity focused sid=" <> sessionIdText sid <> " kind=" <> extractActivityKind val <> " status=" <> extractActivityStatus val)
  797           withStreamingState wsConns key (\ss -> handleActivity cfg chan ss val)
  798       | otherwise         -> do
  799           -- A background session's activity arrives on that session's own
  800           -- WS connection, but it can also be broadcast to this one. It
  801           -- must never reach the watch-all-tabs path: a bg turn is tracked
  802           -- by 'handleBgServerEvent', and a watch notification would emit
  803           -- a bogus "Tab ?" line plus a duplicate reply for every bg turn.
  804           isBg <- bgOwnsSession bgSessions key sid
  805           dbg ("[watch] SeActivity non-focused sid=" <> sessionIdText sid <> " focusedSid=" <> sessionIdText focusedSid <> " kind=" <> extractActivityKind val <> " status=" <> extractActivityStatus val <> " bg=" <> (if isBg then "true" else "false"))
  806           unless isBg $
  807             handleWatchActivity cfg chan key watchState thinkingTabs focusedSid sid val
  808     -- Tool-call events are BeActivity with kind="tool-call", broadcast
  809     -- by the server's aeOnToolCall hook. They arrive as SeActivity.
  810     -- handleActivity dispatches on kind internally.
  811     SeAsk sid val
  812       | sid == focusedSid -> handleAsk cfg chan key pendingAsks sid val
  813     -- A question answered/cancelled/timed out elsewhere (the web UI) is no
  814     -- longer answerable from this channel.
  815     SeAskResolved _ val -> forgetAskByPrefix pendingAsks (extractAskId val)
  816     SeLists val -> handleLists cfg chan key tabTracker watchState thinkingTabs val
  817     _ -> pure ()  -- ignore events for other sessions or irrelevant types

  907 --   thinking set and send a \"tab finished thinking\" notification with
  908 --   the last assistant reply.

  920     let kind = extractActivityKind val
  921     dbg ("[watch] kind=" <> kind <> " status=" <> extractActivityStatus val)

  923       "harness-status" -> do

  926           "thinking" -> do

  944 -- | Send a \"tab finished thinking\" notification to the channel. Fetches
  945 -- the tab list (to resolve the tab label) and the transcript (to get the
  946 -- last assistant reply). If either fetch fails, the notification is still
  947 -- sent with whatever information is available — the notification must not
  948 -- be silently dropped just because a gateway call failed.
  949 sendWatchNotification
  950   :: ChatChannel c
  951   => ChatChannelConfig -> c -> SessionId -> IO ()
  952 sendWatchNotification cfg chan sid = do
  953   let apiBase = gcApiBase (cccGateway cfg)
  954       mgr = cccHttpManager cfg
  955       sidText = sessionIdText sid
  956   dbg ("[watch] sendWatchNotification sid=" <> sidText)
  957   -- Resolve the tab index (for the "Tab N" prefix) from GET /api/tabs,
  958   -- and the session display title (the same label the web frontend's
  959   -- sidebar shows) from GET /api/sessions. The title cascade mirrors
  960   -- the frontend's sessionDisplayTitle: description, autoSummary,
  961   -- firstMessageSnippet, agent, short id.
  962   tabIdx <- do
  963     eTabs <- httpGetTabs mgr apiBase
  964     case eTabs of
  965       Right tabs ->
  966         case [ t | t <- tabs, tjSessionId t == Just sidText ] of
  967           (t : _) -> pure (case mkTabIndex (tjIndex t) of
  968             Right idx -> T.singleton (tabIndexToChar idx)
  969             Left _    -> T.pack (show (tjIndex t)))
  970           []      -> pure "?"
  971       Left _ -> pure "?"
  972   sessionTitle <- do
  973     eSessions <- httpGetSessions mgr apiBase
  974     case eSessions of
  975       Right sessions ->
  976         case [ s | s <- sessions, sessionJsonId s == Just sidText ] of
  977           (s : _) -> pure (sessionDisplayTitle s)
  978           []      -> pure sidText
  979       Left _ -> pure sidText
  980   mReply <- do
  981     eEntries <- httpGetTranscript mgr apiBase sid
  982     case eEntries of
  983       Right entries -> pure (lastAssistantText entries)
  984       Left _        -> pure Nothing
  985   dbg ("[watch] notification tabIdx=" <> tabIdx <> " sessionTitle=" <> sessionTitle <> " hasReply=" <> (case mReply of Just _ -> "true"; Nothing -> "false"))
  986   let header = "\x1F4D4 Tab " <> tabIdx <> " (" <> sessionTitle <> ") finished thinking"
  987   case mReply of
  988     Just reply | not (T.null reply) -> ccSend chan (header <> ":\n" <> reply)
  989     _ -> ccSend chan header

  999 -- the web frontend's @sessionDisplayTitle@ cascade:
 1000 -- @description -> autoSummary -> firstMessageSnippet -> agent -> short id@.
 1001 -- Pure.

 1005     A.Object o ->

 1008         , asText =<< KeyMap.lookup (Key.fromText "auto_summary") o
 1009         , asText =<< KeyMap.lookup (Key.fromText "autoSummary") o

 1011         , asText =<< KeyMap.lookup (Key.fromText "firstMessageSnippet") o

 1024 -- | Handle an @entry-update@ event: create or edit the streaming bubble.
 1025 handleEntryUpdate
 1026   :: ChatChannel c
 1027   => ChatChannelConfig -> c -> StreamingState -> Value -> IO ()
 1028 handleEntryUpdate cfg chan ss val = do
 1029   let streamCfg = cccStreamCfg cfg
 1030       text = extractEntryText val
 1031   unless (T.null text) $ do
 1032     -- Post-finalize guard (issue #198 follow-up): a late
 1033     -- entry-update can arrive AFTER the turn's recorded entry
 1034     -- (the server's streaming path and the post-turn broadcast
 1035     -- are unsynchronized). Editing now would overwrite the
 1036     -- finalized text and re-add the cursor. Ignore updates
 1037     -- entirely while the turn is finalized.
 1038     finalized <- readIORef (ssFinalized ss)
 1039     unless finalized $ do
 1040       writeIORef (ssAccumulated ss) text
 1041       -- For channels that don't support streaming (Signal), just
 1042       -- accumulate the text without sending intermediate edits.
 1043       -- The final text is sent as a single message when the
 1044       -- recorded entry or idle status arrives (finalizeBubble).
 1045       when (ccSupportsStreaming chan) $ do
 1046         now <- getCurrentTime

 1068                 Nothing -> pure ()

 1070               ok <- ccEditMessage chan id' content

 1072                 writeIORef (ssLastEdit ss) (Just now)

 1075 -- | Finalize the current streaming bubble for one conversation: edit it
 1076 -- without the cursor (or, if the edit fails, send the full text as a new
 1077 -- message), then reset the bubble state so the next entry-update starts
 1078 -- a fresh bubble. Does nothing when no bubble exists or nothing was
 1079 -- streamed. When @markFinal@ is 'True' (the turn's recorded response
 1080 -- entry — the definitive delivery), sets 'ssFinalized' so late
 1081 -- entry-updates are ignored until the next turn starts.

 1083   :: ChatChannel c
 1084   => ChatChannelConfig -> c -> StreamingState -> Text -> Bool -> IO ()

 1086   let streamCfg = cccStreamCfg cfg
 1087       finalText = stripCursor streamCfg text
 1088   mMsgId <- readIORef (ssMsgId ss)
 1089   delivered <- case mMsgId of
 1090     Nothing -> isJust <$> ccSendWithId chan finalText
 1091     Just id' -> do
 1092       ok <- ccEditMessage chan id' finalText
 1093       if ok
 1094         then pure True
 1095         else isJust <$> ccSendWithId chan finalText
 1096   when (markFinal && delivered) $ writeIORef (ssFinalized ss) True
 1097   resetStreamingState ss

 1264 -- | Handle /new: mint a fresh session and make it the conversation's
 1265 -- focused session.
 1266 --
 1267 -- Exactly ONE mint happens per @/new@:
 1268 --
 1269 --   * the conversation already has a session ⇒ @POST \/api\/sessions\/:id\/new@
 1270 --     ('cccRebindNew'), which mints AND rebinds the tab in one call;
 1271 --   * no prior session ⇒ @POST \/api\/sessions\/new@ ('cccNewSession').
 1272 --
 1273 -- Minting on both paths would create two sessions and reproduce the exact
 1274 -- orphan-session bug this work exists to fix.
 1275 --
 1276 -- The arguments are the CLI's (@-p\/--provider@, @-m\/--model@,
 1277 -- @-r\/--repo@), parsed by the shared 'parseNewArgs' so chat and CLI accept
 1278 -- the same flags. Message order: rebind warning, then repo failure, then the
 1279 -- confirmation.
 1280 handleNewSession
 1281   :: ChatChannel c
 1282   => ChatChannelConfig -> c -> SessionMap -> TVar (Map ConversationKey WsConn)
 1283   -> PendingAsks -> BgSessions -> TVar (Map ConversationKey (SessionId, [TabJson]))
 1284   -> WatchState -> ThinkingTabs
 1285   -> ConversationKey -> Text
 1286   -> IO ()
 1287 handleNewSession cfg chan sessions wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key args = do
 1288   let apiBase = gcApiBase (cccGateway cfg)
 1289       mgr = cccHttpManager cfg
 1290       na = parseNewArgs args
 1291   mPrior <- sessionLookup sessions key
 1292   eOutcome <- case mPrior of
 1293     Just oldSid -> do
 1294       -- Absent provider/model inherit the old session's (the gateway applies
 1295       -- that); the channel is this channel's label, so the new session stays
 1296       -- attributable to the medium it was created from.
 1297       eRes <- cccRebindNew cfg mgr apiBase oldSid
 1298         (rebindBody (naProvider na) (naModel na) (Just (ccLabel chan)))
 1299       pure $ fmap (\res -> NewOutcome
 1300         { noSid         = nsrSessionId res
 1301         , noRebindError = nsrRebindError res
 1302         , noOldSid      = Just (sessionIdText oldSid)
 1303         , noProvider    = Just (nsrProvider res)
 1304         , noModel       = Just (nsrModel res)
 1305         }) eRes
 1306     Nothing -> do
 1307       eSid <- cccNewSession cfg mgr apiBase (mintBody (naProvider na) (naModel na))
 1308       pure $ fmap (\sidText -> NewOutcome
 1309         { noSid         = sidText
 1310         , noRebindError = Nothing
 1311         , noOldSid      = Nothing
 1312         , noProvider    = Nothing
 1313         , noModel       = Nothing
 1314         }) eSid
 1315   case eOutcome of
 1316     Left e -> ccSend chan ("new session failed: " <> e)
 1317     Right out -> case mkSessionId (noSid out) of
 1318       Left _ -> ccSend chan "new session failed: invalid session id"
 1319       Right sid -> do
 1320         sessionInsert sessions key sid
 1321         -- Re-target the conversation's WS connection on BOTH paths: without
 1322         -- this the new session's replies and asks are filtered out (WU-5).
 1323         ensureWsConn cfg chan wsConns pendingAsks bgSessions tabTracker watchState thinkingTabs key sid
 1324         -- A rebind failure is a warning, not a failure: the session exists.
 1325         for_ (noRebindError out) $ \e ->
 1326           ccSend chan ("warning: /new tab rebind failed: " <> e)
 1327         mRepoErr <- setupRepoFor cfg mgr apiBase sid (naRepo na)
 1328         for_ mRepoErr $ \e -> ccSend chan ("repo setup failed: " <> e)
 1329         ccSend chan (newConfirmation out)

 1512 -- its last element.

 1573 -- message creates a fresh tab. Also seeds the 'ThinkingTabs' set from the
 1574 -- snapshot's @thinkingSessionIds@ so tabs already thinking before the WS
 1575 -- connection was established still trigger a finished notification.
 1576 handleLists
 1577   :: ChatChannel c
 1578   => ChatChannelConfig -> c -> ConversationKey
 1579   -> TVar (Map ConversationKey (SessionId, [TabJson]))
 1580   -> WatchState -> ThinkingTabs -> Value -> IO ()
 1581 handleLists cfg chan key tabTracker watchState thinkingTabs val = do
 1582   -- Seed the thinking-tabs set from the snapshot's thinkingSessionIds.
 1583   -- This is the core fix for the "missed thinking event" problem: when a
 1584   -- WS connection is established (or a lists snapshot is broadcast on tab
 1585   -- changes), sessions that were already thinking before the connection
 1586   -- existed are included in the snapshot. Without this seeding, the
 1587   -- ThinkingTabs set starts empty and the thinking→idle transition is
 1588   -- never detected for those sessions (the idle event arrives but
 1589   -- wasThinking is false → no notification).
 1590   seedThinkingTabsFromLists watchState thinkingTabs key val
 1591   -- Fetch the current tabs from the gateway to get an accurate list.
 1592   let apiBase = gcApiBase (cccGateway cfg)
 1593       mgr = cccHttpManager cfg
 1594   eTabs <- httpGetTabs mgr apiBase
 1595   case eTabs of
 1596     Left _ -> pure ()
 1597     Right newTabs -> do
 1598       tracker <- readTVarIO tabTracker
 1599       case Map.lookup key tracker of
 1600         Nothing -> pure ()  -- no tracked session for this conversation
 1601         Just (focusedSid, _oldTabs) -> do
 1602           -- Check if the focused session's tab was removed.
 1603           let newSessionIds = mapMaybe tjSessionId newTabs
 1604           if focusedSid `elem` mapMaybe mkSessionId' newSessionIds
 1605             then do
 1606               -- Tab still exists; update the tracked tab list.
 1607               atomically (modifyTVar' tabTracker (Map.insert key (focusedSid, newTabs)))
 1608             else do
 1609               -- The focused session's tab was closed. Send the notification.
 1610               ccSend chan ("tab closed (session " <> sessionIdText focusedSid
 1611                         <> "); a new tab will be created on your next message")
 1612               -- Clear the tracking state so the next message creates a fresh tab.
 1613               atomically (modifyTVar' tabTracker (Map.delete key))
 1614   where
 1615     mkSessionId' t = case mkSessionId t of Right s -> Just s; Left _ -> Nothing

 1617 -- | Seed the 'ThinkingTabs' set for a conversation from the
 1618 -- @thinkingSessionIds@ field in a @lists@ WS event payload. This ensures
 1619 -- sessions that were already thinking before the WS connection was
 1620 -- established are tracked, so their eventual @idle@ transition triggers a
 1621 -- watch notification. Replaces the per-conversation thinking set (rather
 1622 -- than unioning) so sessions that finished between snapshots are not kept
 1623 -- stale — the next @thinking@ activity event will re-add them if needed.
 1624 -- No-op when watch mode is off (the thinking set is only consulted by
 1625 -- 'handleWatchActivity', which itself checks watch mode, but seeding
 1626 -- when off would waste memory for no benefit).
 1627 seedThinkingTabsFromLists :: WatchState -> ThinkingTabs -> ConversationKey -> Value -> IO ()
 1628 seedThinkingTabsFromLists watchState thinkingTabs key val = do
 1629   watchOn <- lookupWatch watchState key
 1630   when watchOn $ do
 1631     let sids = extractThinkingSessionIds val
 1632     unless (null sids) $ do
 1633       dbg ("[watch] seeding thinking tabs from lists snapshot: key=" <> ckConv key
 1634         <> " count=" <> T.pack (show (length sids)))
 1635       atomically (modifyTVar' thinkingTabs (Map.insert key (Set.fromList sids)))

 1649 -- | Parse callback_data of the form @"<8hex>:<index>"@. Returns 'Nothing'
 1650 -- for malformed data. Pure.
 1651 parseCallbackData :: Text -> Maybe (Text, Int)
 1652 parseCallbackData cbData =
 1653   case T.splitOn ":" cbData of
 1654     [prefix, idxTxt]
 1655       | T.length prefix == 8 && T.all isHexChar prefix
 1656         -> case T.unpack idxTxt of
 1657              [] -> Nothing
 1658              s  -> case reads s of
 1659                      [(n, "")] | n >= 0 -> Just (prefix, n)
 1660                      _ -> Nothing
 1661     _ -> Nothing
 1662   where
 1663     isHexChar c = isDigit c || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F')

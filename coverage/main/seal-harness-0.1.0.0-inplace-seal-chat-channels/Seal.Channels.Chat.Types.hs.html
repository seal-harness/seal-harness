   44 -- | One inbound message from the platform (Signal, Telegram).
   45 data InboundMessage = InboundMessage
   46   { imSource :: MessageSource
   47   , imBody   :: Text
   48   , imCallbackData :: Maybe Text
   49     -- ^ @Just data@ when this message is a @callback_query@ (button tap);
   50     -- @Nothing@ for a regular text message. The loop uses this to route
   51     -- callback taps to the ASK_HUMAN answer-delivery path.
   52   } deriving stock (Eq, Show)
   53 
   54 -- | A raw received message from the transport: the conversation id, the
   55 -- sender's user id (if available), and the message body. The transport
   56 -- derives these from authenticated transport metadata (e.g. Telegram's
   57 -- @chat.id@ + @from.id@, Signal's peer phone number + UUID). The body is
   58 -- the message text. Pure data; the adapter's reader loop constructs a
   59 -- 'MessageSource' from these fields.
   60 data ReceivedMessage = ReceivedMessage
   61   { rmConversationId :: ConversationId
   62   , rmSender         :: Maybe Text
   63   , rmReplyTo        :: Text
   64     -- ^ The platform-specific reply target (Telegram chat id, Signal
   65     -- phone number). Used by the adapter to address sends/edits.
   66   , rmBody           :: Text
   67   , rmCallbackData   :: Maybe Text
   68     -- ^ @Just data@ when the message is a @callback_query@ (button tap);
   69     -- @Nothing@ for a regular text message. The data is the
   70     -- @callback_data@ string the keyboard was built with.
   71   , rmCallbackId     :: Maybe Text
   72     -- ^ @Just id@ for callback_query (for answerCallbackQuery); @Nothing@
   73     -- for regular messages.
   74   , rmCallbackMessageId :: Maybe Text
   75     -- ^ @Just msgId@ for callback_query (the message holding the
   76     -- keyboard, so the adapter can remove it after a tap); @Nothing@
   77     -- for regular messages.
   78   } deriving stock (Eq, Show)

   80 -- | A platform message identifier (opaque: a Telegram @message_id@ or a
   81 -- Signal timestamp). Used for editing streaming bubbles.
   82 newtype ChatMessageId = ChatMessageId Text
   83   deriving stock (Eq, Show)

   85 -- | The per-conversation routing key: (channel-kind-text, conversation-id-text).
   86 -- Derived from the 'MessageSource' (both fields are server-derived, never
   87 -- user-supplied, so a sender cannot forge a key).
   88 data ConversationKey = ConversationKey
   89   { ckChannel :: Text
   90   , ckConv    :: Text
   91   } deriving stock (Eq, Ord, Show)
   92 
   93 -- | Derive the conversation key from a 'MessageSource'.
   94 convKeyFromSource :: MessageSource -> ConversationKey
   95 convKeyFromSource ms = ConversationKey
   96   { ckChannel = channelKindToText (msChannelKind ms)
   97   , ckConv = conversationIdText (msConversationId ms)
   98   }

  100 -- | A client-side session map: conversation key → session id. Replaces the
  101 -- server-side 'CursorStore'. Thread-safe via 'TVar'.
  102 newtype SessionMap = SessionMap (TVar (Map ConversationKey SessionId))
  103 
  104 -- | Create a new empty session map.
  105 newSessionMap :: IO SessionMap
  106 newSessionMap = SessionMap <$> newTVarIO Map.empty
  107 
  108 -- | Look up the session id for a conversation key.
  109 sessionLookup :: SessionMap -> ConversationKey -> IO (Maybe SessionId)
  110 sessionLookup (SessionMap tv) key =
  111   Map.lookup key <$> readTVarIO tv
  112 
  113 -- | Insert (or replace) a conversation's session id.
  114 sessionInsert :: SessionMap -> ConversationKey -> SessionId -> IO ()
  115 sessionInsert (SessionMap tv) key sid =
  116   atomically (modifyTVar' tv (Map.insert key sid))

  137 -- | Mutable streaming state for one conversation's streaming bubble. Lives
  138 -- in 'IORef's so the WS event handler can call create/edit without passing
  139 -- state around.
  140 data StreamingState = StreamingState
  141   { ssMsgId       :: IORef (Maybe ChatMessageId)
  142   , ssAccumulated :: IORef Text
  143   , ssLastEdit    :: IORef (Maybe UTCTime)
  144   , ssLastLen     :: IORef Int
  145     -- ^ Codepoints in the accumulated text at the last platform edit.
  146     -- The edit gate measures NEW text since the last edit, not the total.
  147   , ssFinalized   :: IORef Bool
  148     -- ^ 'True' once the turn's final entry has been delivered (the
  149     -- bubble was finalized). Late entry-updates — which can arrive after
  150     -- the recorded entry, because the server's streaming path and the
  151     -- post-turn broadcast are unsynchronized — are ignored while set.
  152     -- Cleared when a new turn starts (harness-status: thinking).
  153   }
  154 
  155 -- | Create fresh streaming state for one conversation.
  156 newStreamingState :: IO StreamingState
  157 newStreamingState = do
  158   msgId <- newIORef Nothing
  159   accum <- newIORef ""
  160   lastEdit <- newIORef Nothing
  161   lastLen <- newIORef 0
  162   finalized <- newIORef False
  163   pure StreamingState
  164     { ssMsgId = msgId
  165     , ssAccumulated = accum
  166     , ssLastEdit = lastEdit
  167     , ssLastLen = lastLen
  168     , ssFinalized = finalized
  169     }
  170 
  171 -- | Reset the streaming state for a new segment/turn: the bubble id is
  172 -- cleared (the next entry-update creates a new bubble), the accumulated
  173 -- text is dropped, and the rate-limit clock is reset. Does NOT clear
  174 -- 'ssFinalized' — the finalize paths set it /after/ resetting, so a
  175 -- blanket clear here would race. Callers that start a new turn (the
  176 -- @thinking@ harness-status) clear it explicitly.
  177 resetStreamingState :: StreamingState -> IO ()
  178 resetStreamingState ss = do
  179   writeIORef (ssMsgId ss) Nothing
  180   writeIORef (ssAccumulated ss) ""
  181   writeIORef (ssLastEdit ss) Nothing
  182   writeIORef (ssLastLen ss) 0

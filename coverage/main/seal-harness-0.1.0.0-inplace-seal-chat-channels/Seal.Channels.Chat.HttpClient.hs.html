   40 -- | The result of a @POST /api/sessions/:id/send@ call.
   41 data SendResult = SendResult
   42   { srKind      :: Text       -- ^ @"assistant"@, @"slash"@, or @"error"@
   43   , srResponse  :: Text       -- ^ the response text (slash output or empty)
   44   , srSessionId :: Maybe Text -- ^ new session id (from @\/new@)
   45   , srError     :: Maybe Text -- ^ error message (when kind is @"error"@)
   46   } deriving stock (Eq, Show)
   47 
   48 -- | Parse the JSON response from @POST /api/sessions/:id/send@.
   49 parseSendResult :: BL.ByteString -> Either Text SendResult
   50 parseSendResult body =
   51   case A.decode body :: Maybe Value of
   52     Nothing -> Left "failed to parse send response JSON"
   53     Just (A.Object o) ->
   54       Right SendResult
   55         { srKind = fromMaybe "" (asText =<< KeyMap.lookup (Key.fromText "kind") o)
   56         , srResponse = fromMaybe "" (asText =<< KeyMap.lookup (Key.fromText "response") o)
   57         , srSessionId = asText =<< KeyMap.lookup (Key.fromText "session_id") o
   58         , srError = asText =<< KeyMap.lookup (Key.fromText "error") o
   59         }
   60     Just _ -> Left "send response is not a JSON object"

   62 -- | A tab from @GET /api/tabs@.
   63 data TabJson = TabJson
   64   { tjIndex     :: Int          -- ^ positional index (0-based)
   65   , tjSessionId :: Maybe Text   -- ^ session id (Nothing for harness tabs)
   66   , tjKind      :: Text         -- ^ tab kind wire string
   67   , tjLabel     :: Maybe Text   -- ^ optional user-set label
   68   } deriving stock (Eq, Show)

   70 -- | @POST /api/sessions/:id/send@ — send a message to a session.
   71 httpSend :: Manager -> Text -> SessionId -> Text -> IO (Either Text SendResult)
   72 httpSend mgr apiBase sid message = do
   73   let url = T.unpack apiBase <> "/sessions/" <> T.unpack (sessionIdText sid) <> "/send"
   74   doRequest mgr url methodPost (Just (A.object ["message" .= message])) >>= \case
   75     Left e -> pure (Left e)
   76     Right body -> pure (parseSendResult body)
   77 
   78 -- | @GET /api/tabs@ — list all tabs.
   79 httpGetTabs :: Manager -> Text -> IO (Either Text [TabJson])
   80 httpGetTabs mgr apiBase = do
   81   let url = T.unpack apiBase <> "/tabs"
   82   doRequest mgr url methodGet Nothing >>= \case
   83     Left e -> pure (Left e)
   84     Right body -> pure (parseTabsBody body)
   85 
   86 -- | @POST /api/sessions/new@ — create a new session.
   87 httpNewSession :: Manager -> Text -> Value -> IO (Either Text Text)
   88 httpNewSession mgr apiBase body = do
   89   let url = T.unpack apiBase <> "/sessions/new"
   90   doRequest mgr url methodPost (Just body) >>= \case
   91     Left e -> pure (Left e)
   92     Right respBody -> pure (extractSessionId respBody)
   93 
   94 -- | @GET /api/sessions/:id/transcript@ — get the transcript as a JSON array.
   95 httpGetTranscript :: Manager -> Text -> SessionId -> IO (Either Text [Value])
   96 httpGetTranscript mgr apiBase sid = do
   97   let url = T.unpack apiBase <> "/sessions/" <> T.unpack (sessionIdText sid) <> "/transcript"
   98   doRequest mgr url methodGet Nothing >>= \case
   99     Left e -> pure (Left e)
  100     Right body ->
  101       case A.decode body :: Maybe [Value] of
  102         Just vs -> pure (Right vs)
  103         Nothing -> pure (Left "failed to parse transcript JSON array")
  104 
  105 -- | @POST /api/sessions/:id/stop@ — abort the session's in-flight turn.
  106 httpStopSession :: Manager -> Text -> SessionId -> IO (Either Text ())
  107 httpStopSession mgr apiBase sid = do
  108   let url = T.unpack apiBase <> "/sessions/" <> T.unpack (sessionIdText sid) <> "/stop"
  109   doRequest mgr url methodPost Nothing >>= \case
  110     Left e -> pure (Left e)
  111     Right _ -> pure (Right ())

  113 -- | @POST /api/sessions/:id/questions/:qid/answer@ — answer a pending
  114 -- ASK_HUMAN question.
  115 httpAnswerQuestion :: Manager -> Text -> SessionId -> Text -> Text -> Text -> IO (Either Text ())
  116 httpAnswerQuestion mgr apiBase sid qid answer scope = do
  117   let url = T.unpack apiBase <> "/sessions/" <> T.unpack (sessionIdText sid)
  118             <> "/questions/" <> T.unpack qid <> "/answer"
  119   doRequest mgr url methodPost (Just (A.object ["answer" .= answer, "scope" .= scope])) >>= \case
  120     Left e -> pure (Left e)
  121     Right _ -> pure (Right ())

  137 -- | Parse a JSON body into a 'Value'. Returns 'Left' on parse failure.
  138 parseJsonBody :: BL.ByteString -> Either Text Value
  139 parseJsonBody body =
  140   case A.decode body :: Maybe Value of
  141     Just v  -> Right v
  142     Nothing -> Left "failed to parse JSON"
  143 
  144 -- | Perform an HTTP request with a method, optional JSON body, and return
  145 -- the response body on success (2xx). Returns 'Left' on network or HTTP
  146 -- errors.
  147 doRequest :: Manager -> String -> Method -> Maybe Value -> IO (Either Text BL.ByteString)
  148 doRequest mgr url m mBody = do
  149   eReq <- try (parseRequest url) :: IO (Either HttpException Request)
  150   case eReq of
  151     Left _ -> pure (Left ("invalid URL: " <> T.pack url))
  152     Right req0 -> do
  153       let req1 = req0 { method = m }
  154           req2 = case mBody of
  155             Just bodyVal ->
  156               req1 { requestBody = RequestBodyLBS (A.encode bodyVal)
  157                    , requestHeaders = [("content-type", "application/json")]
  158                    }
  159             Nothing -> req1
  160       eResp <- try (httpLbs req2 mgr) :: IO (Either HttpException (Response BL.ByteString))
  161       case eResp of
  162         Left _ -> pure (Left "HTTP request failed")
  163         Right resp ->
  164           let code = statusCode (responseStatus resp)
  165           in if code >= 200 && code <= 299
  166                then pure (Right (responseBody resp))
  167                else pure (Left ("HTTP " <> T.pack (show code)))
  168 
  169 -- | Extract the @id@ field from a session info JSON object.
  170 extractSessionId :: BL.ByteString -> Either Text Text
  171 extractSessionId body =
  172   case A.decode body :: Maybe Value of
  173     Just (A.Object o) ->
  174       case asText =<< KeyMap.lookup (Key.fromText "session_id") o of
  175         Just sid -> Right sid
  176         Nothing  -> Left "session response missing 'session_id' field"
  177     Just _ -> Left "session response is not a JSON object"
  178     Nothing -> Left "failed to parse session response JSON"
  179 
  180 -- | Parse the @GET /api/tabs@ response body.
  181 parseTabsBody :: BL.ByteString -> Either Text [TabJson]
  182 parseTabsBody body =
  183   case A.decode body :: Maybe [Value] of
  184     Nothing -> Left "failed to parse tabs JSON array"
  185     Just vs -> traverse parseTabJson vs
  186 
  187 -- | Parse one tab JSON object.
  188 parseTabJson :: Value -> Either Text TabJson
  189 parseTabJson (A.Object o) =
  190   Right TabJson
  191     { tjIndex = case KeyMap.lookup (Key.fromText "index") o of
  192         Just (A.Number n) -> round n
  193         _ -> 0
  194     , tjSessionId = asText =<< KeyMap.lookup (Key.fromText "session_id") o
  195     , tjKind = fromMaybe "" (asText =<< KeyMap.lookup (Key.fromText "kind") o)
  196     , tjLabel = asText =<< KeyMap.lookup (Key.fromText "label") o
  197     }
  198 parseTabJson _ = Left "tab is not a JSON object"
  199 
  200 -- | Extract text from a JSON string value.
  201 asText :: Value -> Maybe Text
  202 asText (A.String t) = Just t
  203 asText _ = Nothing

   57 -- | The result of a @POST /api/sessions/:id/send@ call.
   58 data SendResult = SendResult
   59   { srKind      :: Text       -- ^ @"assistant"@, @"slash"@, or @"error"@
   60   , srResponse  :: Text       -- ^ the response text (slash output or empty)
   61   , srSessionId :: Maybe Text -- ^ new session id (from @\/new@)
   62   , srError     :: Maybe Text -- ^ error message (when kind is @"error"@)
   63   } deriving stock (Eq, Show)

   65 -- | Parse the JSON response from @POST /api/sessions/:id/send@.
   66 parseSendResult :: BL.ByteString -> Either Text SendResult
   67 parseSendResult body =
   68   case A.decode body :: Maybe Value of
   69     Nothing -> Left "failed to parse send response JSON"
   70     Just (A.Object o) ->
   71       Right SendResult
   72         { srKind = fromMaybe "" (asText =<< KeyMap.lookup (Key.fromText "kind") o)
   73         , srResponse = fromMaybe "" (asText =<< KeyMap.lookup (Key.fromText "response") o)
   74         , srSessionId = asText =<< KeyMap.lookup (Key.fromText "session_id") o
   75         , srError = asText =<< KeyMap.lookup (Key.fromText "error") o
   76         }
   77     Just _ -> Left "send response is not a JSON object"

   97 -- | @POST /api/sessions/:id/send@ — send a message to a session.
   98 httpSend :: Manager -> Text -> SessionId -> Text -> IO (Either Text SendResult)
   99 httpSend mgr apiBase sid message = do
  100   let url = T.unpack apiBase <> "/sessions/" <> T.unpack (sessionIdText sid) <> "/send"
  101   doRequest mgr url methodPost (Just (A.object ["message" .= message])) >>= \case
  102     Left e -> pure (Left e)
  103     Right body -> pure (parseSendResult body)

  137 -- | One entry of the gateway's slash-command menu: the canonical command name
  138 -- (no leading slash, aliases are never listed) and its one-line synopsis.

  144 -- | @GET /api/commands@ — the slash-command menu the chat channels render as
  145 -- @\/help@ (and register as the platform's command menu). Malformed entries
  146 -- are skipped rather than failing the whole list.
  147 httpGetCommands :: Manager -> Text -> IO (Either Text [CommandJson])
  148 httpGetCommands mgr apiBase = do
  149   let url = T.unpack apiBase <> "/commands"
  150   doRequest mgr url methodGet Nothing >>= \case
  151     Left e -> pure (Left e)
  152     Right body -> case A.decode body :: Maybe [Value] of
  153       Just vs -> pure (Right (mapMaybe parseCommandJson vs))
  154       Nothing -> pure (Left "failed to parse commands JSON array")

  156 -- | Parse one @\/api\/commands@ entry.
  157 parseCommandJson :: Value -> Maybe CommandJson
  158 parseCommandJson (A.Object o) = do
  159   n <- asText =<< KeyMap.lookup (Key.fromText "name") o
  160   let syn = fromMaybe "" (asText =<< KeyMap.lookup (Key.fromText "synopsis") o)
  161   pure (CommandJson n syn)
  162 parseCommandJson _ = Nothing

  196 -- | The request body for @POST \/api\/sessions\/:id\/new@. Absent overrides
  197 -- are omitted entirely — never sent as JSON @null@.
  198 rebindBody :: Maybe Text -> Maybe Text -> Maybe Text -> Value
  199 rebindBody mProvider mModel mChannel = A.object (catMaybes
  200   [ ("provider" .=) <$> mProvider
  201   , ("model" .=) <$> mModel
  202   , ("channel" .=) <$> mChannel

  240 -- | Parse the @POST \/api\/sessions\/:id\/new@ response body.
  241 parseNewSessionResult :: BL.ByteString -> Either Text NewSessionResult
  242 parseNewSessionResult body =
  243   case A.decode body :: Maybe Value of
  244     Just (A.Object o) ->
  245       case asText =<< KeyMap.lookup (Key.fromText "session_id") o of
  246         Nothing -> Left "new-session response missing 'session_id' field"
  247         Just sid
  248           | T.null sid -> Left "new-session response has an empty 'session_id'"
  249           | otherwise  -> Right NewSessionResult
  250               { nsrSessionId   = sid
  251               , nsrTabIndex    = asInt =<< KeyMap.lookup (Key.fromText "tab_index") o
  252               , nsrRebound     = fromMaybe False
  253                                    (asBool =<< KeyMap.lookup (Key.fromText "rebound") o)
  254               , nsrRebindError = asText =<< KeyMap.lookup (Key.fromText "rebind_error") o
  255               , nsrProvider    = fromMaybe ""
  256                                    (asText =<< KeyMap.lookup (Key.fromText "provider") o)
  257               , nsrModel       = fromMaybe ""
  258                                    (asText =<< KeyMap.lookup (Key.fromText "model") o)
  259               }
  260     Just _  -> Left "new-session response is not a JSON object"
  261     Nothing -> Left "failed to parse new-session response JSON"
  262   where
  263     asInt (A.Number n) = Just (round n)
  264     asInt _            = Nothing
  265     asBool (A.Bool b)  = Just b
  266     asBool _           = Nothing

  287 -- | @POST /api/sessions/:id/questions/:qid/answer@ — answer a pending
  288 -- ASK_HUMAN question (or the confirmation gate) with free text.
  289 --
  290 -- The body carries an @answer@ and nothing else: the server's parser rejects
  291 -- a body that carries both @answer@ and @scope@ with 400 ("ambiguous: send
  292 -- either {scope} or {answer}, not both", 'parseAnswerBody'). The
  293 -- confirmation gate parses the text itself — @y@/@yes@ approve once,
  294 -- anything else rejects — so one shape serves both ask kinds.
  295 --
  296 -- The endpoint answers @200 {ok:true, accepted:<bool>}@; @accepted:false@
  297 -- means the question was already answered, cancelled, or unknown, which is
  298 -- a resolution but not an accepted answer.
  299 httpAnswerQuestion :: Manager -> Text -> SessionId -> Text -> Text -> IO (Either Text Bool)
  300 httpAnswerQuestion mgr apiBase sid qid answer = do
  301   let url = T.unpack apiBase <> "/sessions/" <> T.unpack (sessionIdText sid)
  302             <> "/questions/" <> T.unpack qid <> "/answer"
  303   doRequest mgr url methodPost (Just (answerBodyJson answer)) >>= \case
  304     Left e -> pure (Left e)
  305     Right body -> pure (Right (acceptedFrom body))
  306 
  307 -- | Read the @accepted@ flag from an answer response. Absent/malformed
  308 -- (e.g. a server that predates the flag) counts as accepted.
  309 acceptedFrom :: BL.ByteString -> Bool
  310 acceptedFrom body =
  311   case A.decode body :: Maybe Value of
  312     Just (A.Object o) -> case KeyMap.lookup (Key.fromText "accepted") o of
  313       Just (A.Bool b) -> b
  314       _               -> True
  315     _ -> True

  343 -- | Perform an HTTP request with a method, optional JSON body, and return
  344 -- the response body on success (2xx). Returns 'Left' on network or HTTP
  345 -- errors.
  346 doRequest :: Manager -> String -> Method -> Maybe Value -> IO (Either Text BL.ByteString)
  347 doRequest mgr url m mBody = do
  348   eReq <- try (parseRequest url) :: IO (Either HttpException Request)
  349   case eReq of
  350     Left _ -> pure (Left ("invalid URL: " <> T.pack url))
  351     Right req0 -> do
  352       let req1 = req0 { method = m }
  353           req2 = case mBody of
  354             Just bodyVal ->
  355               req1 { requestBody = RequestBodyLBS (A.encode bodyVal)
  356                    , requestHeaders = [("content-type", "application/json")]
  357                    }
  358             Nothing -> req1
  359       eResp <- try (httpLbs req2 mgr) :: IO (Either HttpException (Response BL.ByteString))
  360       case eResp of
  361         Left _ -> pure (Left "HTTP request failed")
  362         Right resp ->
  363           let code = statusCode (responseStatus resp)
  364           in if code >= 200 && code <= 299
  365                then pure (Right (responseBody resp))
  366                else pure (Left (parseErrorBody code (responseBody resp)))
  367 
  368 -- | The text of a failed HTTP response. Prefers the gateway's
  369 -- @{\"error\": \"…\"}@ shape ('errJson' in 'Seal.Gateway.API') so callers can
  370 -- surface the real reason (e.g. @clone failed: …@); falls back to
  371 -- @HTTP \<code\>@ for a non-JSON or shapeless body (a proxy, a plain-text
  372 -- 5xx).
  373 parseErrorBody :: Int -> BL.ByteString -> Text
  374 parseErrorBody code body =
  375   case A.decode body :: Maybe Value of
  376     Just (A.Object o) -> case asText =<< KeyMap.lookup (Key.fromText "error") o of
  377       Just msg | not (T.null msg) -> msg
  378       _                           -> fallback
  379     _ -> fallback
  380   where
  381     fallback = "HTTP " <> T.pack (show code)
  382 
  383 -- | Extract the @id@ field from a session info JSON object.
  384 extractSessionId :: BL.ByteString -> Either Text Text
  385 extractSessionId body =
  386   case A.decode body :: Maybe Value of
  387     Just (A.Object o) ->
  388       case asText =<< KeyMap.lookup (Key.fromText "session_id") o of
  389         Just sid -> Right sid
  390         Nothing  -> Left "session response missing 'session_id' field"
  391     Just _ -> Left "session response is not a JSON object"
  392     Nothing -> Left "failed to parse session response JSON"

  394 -- | Parse the @GET /api/lists@ response body.
  395 parseListsBody :: BL.ByteString -> Either Text ListsJson
  396 parseListsBody body =
  397   case A.decode body :: Maybe Value of
  398     Just (A.Object o) -> do
  399       tabs <- case KeyMap.lookup (Key.fromText "tabs") o of
  400         Just (A.Array arr) -> traverse parseTabJson (foldr (:) [] arr)
  401         _                  -> Left "lists response missing 'tabs' array"
  402       let arrOf k = case KeyMap.lookup (Key.fromText k) o of
  403             Just (A.Array arr) -> foldr (:) [] arr
  404             _                  -> []
  405       pure ListsJson
  406         { ljTabs               = tabs
  407         , ljTabSessions        = arrOf "tabSessions"
  408         , ljThinkingSessionIds = mapMaybe asText (arrOf "thinkingSessionIds")
  409         }
  410     _ -> Left "failed to parse lists response JSON object"
  411 
  412 -- | Parse the @GET /api/tabs@ response body.
  413 parseTabsBody :: BL.ByteString -> Either Text [TabJson]
  414 parseTabsBody body =
  415   case A.decode body :: Maybe [Value] of
  416     Nothing -> Left "failed to parse tabs JSON array"
  417     Just vs -> traverse parseTabJson vs

  419 -- | Parse one tab JSON object.
  420 parseTabJson :: Value -> Either Text TabJson
  421 parseTabJson (A.Object o) =
  422   Right TabJson
  423     { tjIndex = case KeyMap.lookup (Key.fromText "index") o of
  424         Just (A.Number n) -> round n
  425         _ -> 0
  426     , tjSessionId = asText =<< KeyMap.lookup (Key.fromText "session_id") o
  427     , tjKind = fromMaybe "" (asText =<< KeyMap.lookup (Key.fromText "kind") o)
  428     , tjLabel = asText =<< KeyMap.lookup (Key.fromText "label") o
  429     , tjStatus = asText =<< KeyMap.lookup (Key.fromText "status") o
  430     }
  431 parseTabJson _ = Left "tab is not a JSON object"

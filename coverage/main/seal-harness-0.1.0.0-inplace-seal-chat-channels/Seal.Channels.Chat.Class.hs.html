    1 -- | The 'ChatChannel' type class — the seam a chat channel implements to
    2 -- be wired into the generic loop ('Seal.Channels.Chat.Loop'). Each method
    3 -- is an 'IO' action so the type is uniform between real and mock variants.
    4 --
    5 -- The class is minimal: receive from the platform, send to the platform,
    6 -- send with a returned message id (for editing), edit a message, and a
    7 -- label. The generic loop handles all routing, session tracking, WS
    8 -- streaming, and ASK_HUMAN through the gateway API.
    9 --
   10 -- Optional methods ('ccSendWithOptions', 'ccAnswerCallback',
   11 -- 'ccEditReplyMarkup', 'ccLastChatId') have default no-op / 'Nothing'
   12 -- implementations so channels that don't support inline keyboards (Signal,
   13 -- mock) fall back to the text-based question rendering.
   14 module Seal.Channels.Chat.Class
   15   ( ChatChannel (..)
   16   , QuestionOption (..)
   17   ) where
   18 
   19 import Data.Text (Text)
   20 
   21 import Seal.Channels.Chat.Types (InboundMessage, ChatMessageId)
   22 
   23 -- | One discrete choice offered alongside an ASK_HUMAN question. Mirrors
   24 -- 'Seal.Handles.AskReply.QuestionOption' but lives in the chat-channel
   25 -- package (no dependency on server internals). The label is the value
   26 -- returned when the human picks this choice; the description is a one-line
   27 -- explanation shown alongside the button.
   28 data QuestionOption = QuestionOption
   29   { qoLabel       :: !Text
   30   , qoDescription :: !Text
   31   } deriving stock (Eq, Show)
   32 
   33 -- | The seam a chat channel implements. Each method is an 'IO' action.
   34 class ChatChannel c where
   35   -- | Receive one inbound message from the platform. Returns 'Nothing' on
   36   -- EOF (the platform connection closed + inbox drained).
   37   ccReceive :: c -> IO (Maybe InboundMessage)
   38 
   39   -- | Send a message to the platform (plain text, no returned id).
   40   ccSend :: c -> Text -> IO ()
   41 
   42   -- | Send a message and return its platform id (for later editing via
   43   -- 'ccEditMessage'). 'Nothing' if the send failed or the platform doesn't
   44   -- support message ids.
   45   ccSendWithId :: c -> Text -> IO (Maybe ChatMessageId)
   46 
   47   -- | Edit a previously sent message: message id, new content. Returns
   48   -- 'True' on success, 'False' on failure (or unsupported).
   49   ccEditMessage :: c -> ChatMessageId -> Text -> IO Bool
   50 
   51   -- | The channel's label (e.g. @"signal"@, @"telegram"@).
   52   ccLabel :: c -> Text
   53 
   54   -- | Whether the channel supports progressive message editing (streaming
   55   -- text bubbles). When 'False', the loop sends the final text as a single
   56   -- message after the turn completes instead of creating/editing a
   57   -- streaming bubble per delta. Default: 'True' (most channels support
   58   -- editing). Signal sets this to 'False' to match the old behavior.
   59   ccSupportsStreaming :: c -> Bool
   60   ccSupportsStreaming _ = True
   61 
   62   -- | Send a question with multiple-choice options as an inline keyboard
   63   -- (one button per option). The @askIdPrefix@ is the 8-hex prefix of the
   64   -- ask id — the channel embeds it in the callback_data
   65   -- (@\"<prefix>:<index>\"@) so the loop can tie a button tap back to the
   66   -- specific pending question. The body carries the full question + option
   67   -- descriptions; the buttons carry just the short labels. Returns the
   68   -- platform message id of the sent message (for later keyboard removal
   69   -- via 'ccEditReplyMarkup'). 'Nothing' if the channel doesn't support
   70   -- inline keyboards — the loop falls back to 'ccSend' with a numbered
   71   -- list. Default: 'Nothing' (unsupported).
   72   ccSendWithOptions
   73     :: c -> Text -> [QuestionOption] -> Text -> IO (Maybe ChatMessageId)
   74   ccSendWithOptions _ _ _ _ = pure Nothing
   75 
   76   -- | Acknowledge a callback query (dismiss the button's loading spinner).
   77   -- Best-effort: never throws. Default: no-op (unsupported).
   78   ccAnswerCallback :: c -> Text -> IO ()
   79   ccAnswerCallback _ _ = pure ()
   80 
   81   -- | Remove the inline keyboard from a previously sent message (disable
   82   -- buttons after a tap so they can't be re-clicked). Best-effort: never
   83   -- throws. Default: no-op (unsupported).
   84   ccEditReplyMarkup :: c -> ChatMessageId -> IO ()
   85   ccEditReplyMarkup _ _ = pure ()
   86 
   87   -- | The last chat id (the platform-specific reply target). 'Nothing' if
   88   -- no message has been received yet. Used by the loop to determine whether
   89   -- inline keyboard sends are possible. Default: 'Nothing'.
   90   ccLastChatId :: c -> IO (Maybe Text)
   91   ccLastChatId _ = pure Nothing

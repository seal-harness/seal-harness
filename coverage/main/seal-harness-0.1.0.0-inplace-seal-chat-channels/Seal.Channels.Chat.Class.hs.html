    1 -- | The 'ChatChannel' type class — the seam a chat channel implements to
    2 -- be wired into the generic loop ('Seal.Channels.Chat.Loop'). Each method
    3 -- is an 'IO' action so the type is uniform between real and mock variants.
    4 --
    5 -- The class is minimal: receive from the platform, send to the platform,
    6 -- send with a returned message id (for editing), edit a message, and a
    7 -- label. The generic loop handles all routing, session tracking, WS
    8 -- streaming, and ASK_HUMAN through the gateway API.
    9 --
   10 -- Optional methods ('ccSendWithOptions', 'ccAnswerCallback',
   11 -- 'ccEditReplyMarkup', 'ccLastChatId') have default no-op / 'Nothing'
   12 -- implementations so channels that don't support inline keyboards (Signal,
   13 -- mock) fall back to the text-based question rendering.
   14 module Seal.Channels.Chat.Class
   15   ( ChatChannel (..)
   16   , ChunkUnit (..)
   17   , QuestionOption (..)
   18   , ProgressStyle (..)
   19   ) where
   20 
   21 import Data.Text (Text)
   22 
   23 import Seal.Channels.Chat.Types (InboundMessage, ChatMessageId)
   24 
   25 -- | How the loop renders a turn's progress on the platform.
   26 --
   27 -- 'PsSeparateMessages' (the default) is the long-standing behaviour: each
   28 -- text segment and each tool call becomes its own message. 'PsSingleBubble'
   29 -- creates one message per turn and edits it in place, so a turn produces a
   30 -- single notification instead of one per tool call.
   31 --
   32 -- Deliberately opt-in per channel: it depends on the platform's edit support
   33 -- being reliable in practice. Signal and Telegram both opt in; every other
   34 -- channel keeps the default.
   35 data ProgressStyle
   36   = PsSeparateMessages
   37   | PsSingleBubble
   38   deriving stock (Eq, Show)
   39 
   40 -- | The unit a platform expresses its maximum message length in.
   41 --
   42 -- This is not cosmetic. Signal's limit is a byte budget (it inlines the first
   43 -- ~2048 /utf-8 bytes/ and moves the rest to an attachment), while Telegram's
   44 -- 4096 is measured in /utf-16 code units/. Counting code points — 'T.length' —
   45 -- is correct for neither: an emoji is 4 bytes and 2 UTF-16 units but a single
   46 -- code point. Chunking a multi-byte message by code-point count silently
   47 -- overruns the platform's real limit and the tail gets truncated or demoted to
   48 -- an attachment.
   49 --
   50 -- Lives here rather than in "Seal.Channels.Chat.Bubble" so the 'ChatChannel'
   51 -- class can name it without an import cycle ('Bubble' already imports this
   52 -- module).
   53 data ChunkUnit
   54   = ChunkUtf8Bytes
   55   | ChunkUtf16Units
   56   deriving stock (Eq, Show)
   57 
   58 -- | One discrete choice offered alongside an ASK_HUMAN question. Mirrors
   59 -- 'Seal.Handles.AskReply.QuestionOption' but lives in the chat-channel
   60 -- package (no dependency on server internals). The label is the value
   61 -- returned when the human picks this choice; the description is a one-line
   62 -- explanation shown alongside the button.
   63 data QuestionOption = QuestionOption
   64   { qoLabel       :: !Text
   65   , qoDescription :: !Text
   66   } deriving stock (Eq, Show)

   68 -- | The seam a chat channel implements. Each method is an 'IO' action.
   69 class ChatChannel c where
   70   -- | Receive one inbound message from the platform. Returns 'Nothing' on
   71   -- EOF (the platform connection closed + inbox drained).
   72   ccReceive :: c -> IO (Maybe InboundMessage)
   73 
   74   -- | Send a message to the platform (plain text, no returned id).
   75   ccSend :: c -> Text -> IO ()
   76 
   77   -- | Send a message and return its platform id (for later editing via
   78   -- 'ccEditMessage'). 'Nothing' if the send failed or the platform doesn't
   79   -- support message ids.
   80   ccSendWithId :: c -> Text -> IO (Maybe ChatMessageId)
   81 
   82   -- | Edit a previously sent message: message id, new content. Returns
   83   -- 'True' on success, 'False' on failure (or unsupported).
   84   ccEditMessage :: c -> ChatMessageId -> Text -> IO Bool
   85 
   86   -- | Send a message that must NOT raise a platform notification, returning
   87   -- its id so it can be edited later. Used for a turn's progress bubble: the
   88   -- progress should be visible without pinging the human.
   89   --
   90   -- 'Nothing' means the platform cannot send silently, or the send failed; the
   91   -- loop then falls back to 'ccSendWithId' (which notifies) so progress is
   92   -- never lost. Default: 'ccSendWithId' — i.e. a channel that cannot send
   93   -- silently keeps today's behaviour exactly.
   94   --
   95   -- Deliberately NOT used for @ASK_HUMAN@ (or any message that must reach the
   96   -- human): those always use the notifying path.
   97   ccSendWithIdSilently :: c -> Text -> IO (Maybe ChatMessageId)
   98   ccSendWithIdSilently c t = ccSendWithId c t
   99 
  100   -- | The channel's label (e.g. @"signal"@, @"telegram"@).
  101   ccLabel :: c -> Text
  102 
  103   -- | Whether the channel supports progressive message editing (streaming
  104   -- text bubbles). Only consulted by the @PsSeparateMessages@ style - a
  105   -- @PsSingleBubble@ channel always edits, so this is inert for Signal and
  106   -- Telegram. Default: 'True' (most channels support editing).
  107   ccSupportsStreaming :: c -> Bool
  108   ccSupportsStreaming _ = True
  109 
  110   -- | How to render a turn's progress. Default 'PsSeparateMessages' — the
  111   -- long-standing one-message-per-segment/tool behaviour — so a channel is
  112   -- unaffected unless it explicitly opts in.
  113   ccProgressStyle :: c -> ProgressStyle
  114   ccProgressStyle _ = PsSeparateMessages
  115 
  116   -- | The platform's maximum message length in codepoints, when it has one.
  117   -- Used by 'PsSingleBubble' to stop live-editing a bubble that has outgrown
  118   -- a single message and fall back to chunked sends. 'Nothing' (the default)
  119   -- means no known limit. The loop is polymorphic in the channel, so the
  120   -- limit has to be reachable through the class rather than from a
  121   -- channel-specific record field.
  122   ccChunkLimit :: c -> Maybe Int
  123   ccChunkLimit _ = Nothing
  124 
  125   -- | The unit 'ccChunkLimit' is expressed in. Only consulted by the chunked
  126   -- send path ('Seal.Channels.Chat.Bubble.chunkMessage').
  127   --
  128   -- Defaults to 'ChunkUtf8Bytes', the stricter of the two: a byte budget
  129   -- never overruns a platform that actually counts UTF-16 units only for
  130   -- characters outside the BMP, whereas counting code points overruns both.
  131   -- A channel whose documented limit is in code units should say so, because
  132   -- a byte budget then splits earlier than it needs to.
  133   ccChunkUnit :: c -> ChunkUnit
  134   ccChunkUnit _ = ChunkUtf8Bytes
  135 
  136   -- | Minimum milliseconds between live edits of a turn's single progress
  137   -- bubble. Both edit sources (streaming frames and tool calls) share one
  138   -- bucket, so a turn cannot exceed one edit per interval.
  139   --
  140   -- Defaults to 'spcEditIntervalMs' of the streaming config (1500 ms), i.e.
  141   -- the same cadence the separate-messages style already uses - so this is
  142   -- not a second, drift-prone constant. It exists because that style's
  143   -- 'shouldEdit' is an @||@: its 80-codepoint branch forces an edit on every
  144   -- frame once the accumulator passes the threshold, which is unbounded. The
  145   -- single-bubble path needs an interval-only gate.
  146   --
  147   -- Only consulted by 'PsSingleBubble'. The @idle@ closing edit is never
  148   -- gated, so a turn always ends on its final content.
  149   ccEditIntervalMs :: c -> Int
  150   ccEditIntervalMs _ = 1500
  151 
  152   -- | Send a question with multiple-choice options as an inline keyboard
  153   -- (one button per option). The @askIdPrefix@ is the 8-hex prefix of the
  154   -- ask id — the channel embeds it in the callback_data
  155   -- (@\"<prefix>:<index>\"@) so the loop can tie a button tap back to the
  156   -- specific pending question. The body carries the full question + option
  157   -- descriptions; the buttons carry just the short labels. Returns the
  158   -- platform message id of the sent message (for later keyboard removal
  159   -- via 'ccEditReplyMarkup'). 'Nothing' if the channel doesn't support
  160   -- inline keyboards — the loop falls back to 'ccSend' with a numbered
  161   -- list. Default: 'Nothing' (unsupported).
  162   ccSendWithOptions
  163     :: c -> Text -> [QuestionOption] -> Text -> IO (Maybe ChatMessageId)
  164   ccSendWithOptions _ _ _ _ = pure Nothing
  165 
  166   -- | Acknowledge a callback query (dismiss the button's loading spinner).
  167   -- Best-effort: never throws. Default: no-op (unsupported).
  168   ccAnswerCallback :: c -> Text -> IO ()
  169   ccAnswerCallback _ _ = pure ()
  170 
  171   -- | Remove the inline keyboard from a previously sent message (disable
  172   -- buttons after a tap so they can't be re-clicked). Best-effort: never
  173   -- throws. Default: no-op (unsupported).
  174   ccEditReplyMarkup :: c -> ChatMessageId -> IO ()
  175   ccEditReplyMarkup _ _ = pure ()
  176 
  177   -- | The last chat id (the platform-specific reply target). 'Nothing' if
  178   -- no message has been received yet. Used by the loop to determine whether
  179   -- inline keyboard sends are possible. Default: 'Nothing'.
  180   ccLastChatId :: c -> IO (Maybe Text)
  181   ccLastChatId _ = pure Nothing

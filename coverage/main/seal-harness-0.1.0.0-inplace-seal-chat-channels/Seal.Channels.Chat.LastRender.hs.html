   64 -- | Everything @\/last@ can tell the user went wrong.
   65 --
   66 -- A bespoke ADT rather than @Either Text@ because 'renderLastError' matches on
   67 -- the constructor to build a distinct string for each, and the tests assert
   68 -- every constructor.
   69 data LastError
   70   = LENoSuchTab Text      -- ^ the token was not a tab index
   71   | LEBadCount Text       -- ^ the second token was not a number
   72   | LENoSession Char      -- ^ the tab is a harness tab (no session)
   73   | LENoCurrentTab        -- ^ no focused tab to default to
   74   | LETabs                -- ^ @GET \/api\/lists@ failed
   75   | LETranscript Char     -- ^ the transcript fetch failed
   76   | LEQuestions Char      -- ^ the pending-questions fetch failed
   77   deriving stock (Eq, Show)

   79 -- | Parse the raw argument text carried by
   80 -- 'Seal.Channels.Chat.Route.ChatLast'.
   81 --
   82 -- The first token is a tab iff it is exactly one character AND
   83 -- 'tabIndexFromChar' accepts it. That is stricter than 'parseTabFocus' (which
   84 -- takes the first character of an arbitrary token): @\/last bb@ naming a bad
   85 -- token back is better than silently meaning tab @b@.
   86 parseLastArgs :: Text -> Either LastError LastArgs
   87 parseLastArgs raw =
   88   case T.words (T.strip raw) of
   89     [] -> Right (LastArgs Nothing 1)
   90     [tabTok] -> LastArgs <$> (Just <$> tabOf tabTok) <*> pure 1
   91     (tabTok : countTok : _) ->
   92       LastArgs <$> (Just <$> tabOf tabTok) <*> countOf countTok
   93   where
   94     tabOf tok = case T.uncons tok of
   95       Just (c, rest)
   96         | T.null rest -> either (const (Left (LENoSuchTab tok))) Right (tabIndexFromChar c)
   97       _ -> Left (LENoSuchTab tok)
   98 
   99     countOf tok
  100       | T.null t || not (T.all isDigit t) = Left (LEBadCount tok)
  101       | otherwise = Right (clamp (T.foldl' step 0 t))
  102       where
  103         t = T.strip tok
  104         step acc c = acc * 10 + (fromEnum c - fromEnum '0')

  106 -- | Clamp a requested count into the supported @1..5@ range.
  107 clamp :: Int -> Int
  108 clamp = max 1 . min 5

  139 -- | Render an error for display on the channel.
  140 --
  141 -- The four strings the requirements specify are used verbatim; errors the
  142 -- requirements do not cover follow the same @last failed:@ shape as the
  143 -- existing @focus failed:@ / @inject failed:@ / @tab list failed:@ family.
  144 renderLastError :: LastError -> Text
  145 renderLastError err = case err of
  146   LENoSuchTab t   -> "last failed: no such tab " <> singleLine t <> " \8212 /tab list"
  147   LEBadCount t    -> "last failed: \"" <> singleLine t
  148                      <> "\" isn't a number \8212 use /last <tab> 1-5"
  149   LENoSession c   -> "last failed: tab " <> T.singleton c
  150                      <> " has no session (harness tab)"
  151   LENoCurrentTab  -> "last failed: no current tab"
  152   LETabs          -> "last failed: couldn't read the tab list \8212 try again in a moment"
  153   LETranscript c  -> "last failed: couldn't read tab " <> T.singleton c
  154                      <> "'s transcript \8212 try again in a moment"
  155   LEQuestions c   -> "last failed: couldn't read tab " <> T.singleton c
  156                      <> "'s question \8212 try again in a moment"

  158 -- | Flatten and bound text that came from the user, before it is echoed back.
  159 --
  160 -- Without this an inbound message of arbitrary length could be echoed into the
  161 -- chat, and its newlines would let a user spoof a harness-authored line (the
  162 -- same argument that motivates block-quoting a question). 16 characters is
  163 -- enough to show which token was rejected.
  164 singleLine :: Text -> Text
  165 singleLine t =
  166   let flattened = T.unwords (T.words (T.map newlineToSpace t))
  167   in if T.length flattened > 16
  168        then T.take 16 flattened <> "\8230"
  169        else flattened
  170   where
  171     newlineToSpace c = if c == '\n' || c == '\r' || c == '\t' then ' ' else c

  177 -- | A tool call as the renderer sees it: an id (for the pass/fail join) and a
  178 -- shape-gated name. The newtype keeps a raw model-authored string from
  179 -- reaching the renderer by accident.
  180 data ToolName = ToolName
  181   { tnId   :: Text          -- ^ @tool_use.id@, the join key (never rendered)
  182   , tnName :: Maybe Text    -- ^ 'safeOpLabel' result; 'Nothing' ⇒ render as unknown
  183   } deriving stock (Eq)

  185 -- | What one transcript entry contributes. Deliberately has **no** field able
  186 -- to hold a tool input, a tool-result body, the system prompt, or a thinking
  187 -- block: the "never tool inputs" guarantee is structural, not testimonial.
  188 data EntryFacts = EntryFacts
  189   { lefText    :: Text                 -- ^ response text blocks, joined "\n\n"
  190   , lefTools   :: [ToolName]           -- ^ tool_use names, no inputs
  191   , lefResults :: [(Text, Bool)]       -- ^ (tool_use_id, is_error) from requests
  192   , lefLegacy  :: Bool                 -- ^ payload has neither content nor messages
  193   } deriving stock (Eq)
  194 
  195 -- | Shape-gate a model-authored tool name.
  196 --
  197 -- @tool_use.name@ arrives straight from the provider response with no
  198 -- validation, so rendering it unbounded would let a model put arbitrary
  199 -- content (JSON, paths, URLs, keys, newlines) into a trusted-looking tool
  200 -- line. Every real opcode name matches @[A-Z][A-Z0-9_]{0,31}@; anything else
  201 -- is rejected and rendered as @(unknown opcode)@.
  202 --
  203 -- This is a shape gate, not an allowlist: a model can still emit a real
  204 -- opcode's name. It stops content injection, not lying.
  205 safeOpLabel :: Text -> Maybe Text
  206 safeOpLabel t
  207   | T.null t = Nothing
  208   | T.length t > 32 = Nothing
  209   | otherwise =
  210       let (h, rest) = T.splitAt 1 t
  211       in if T.all isUpperish h && T.all isNameChar rest then Just t else Nothing
  212   where
  213     isUpperish c = c >= 'A' && c <= 'Z'
  214     isNameChar c = isUpperish c || (c >= '0' && c <= '9') || c == '_'
  215 
  216 -- | Extract the facts one frontend-shaped transcript entry carries.
  217 --
  218 -- Rules, all allowlist-based:
  219 --
  220 -- * only @direction == "response"@ contributes text, and only from a block
  221 --   whose @type@ is @\"text\"@ carrying a string @text@ — a @\"thinking\"@
  222 --   block is dropped, and every other type is dropped with no fallback;
  223 -- * only @direction == "request"@ contributes the pass/fail join, from
  224 --   @type == \"tool_result\"@ blocks keyed by @tool_use_id@.
  225 --
  226 -- Tool inputs are never read into 'EntryFacts'.
  227 extractEntry :: Value -> EntryFacts
  228 extractEntry val =
  229   case val of
  230     Object o ->
  231       let dir = strField "direction" o
  232           payload = fromMaybe Null (KeyMap.lookup (Key.fromText "payload") o)
  233           legacy = isLegacyPayload val
  234       in case dir of
  235            Just "response" -> EntryFacts
  236              { lefText = textOf payload
  237              , lefTools = toolsOf payload
  238              , lefResults = []
  239              , lefLegacy = legacy
  240              }
  241            Just "request" -> EntryFacts "" [] (resultsOf payload) legacy
  242            _ -> EntryFacts "" [] [] legacy
  243     _ -> EntryFacts "" [] [] False
  244 
  245 -- | Does this entry use the legacy payload shape (no @content@, no
  246 -- @messages@)? Those predate the current encoding; the renderer reports that
  247 -- honestly rather than claiming the session is empty.
  248 isLegacyPayload :: Value -> Bool
  249 isLegacyPayload val = case val of
  250   Object o -> case KeyMap.lookup (Key.fromText "payload") o of
  251     Just (Object p) ->
  252       not (KeyMap.member (Key.fromText "content") p)
  253         && not (KeyMap.member (Key.fromText "messages") p)
  254     _ -> False
  255   _ -> False
  256 
  257 -- | The response text: @text@ blocks joined with a blank line. Deliberately
  258 -- not @extractEntryText@'s 'T.concat', which runs paragraphs together.
  259 textOf :: Value -> Text
  260 textOf payload = case blockArray payload of
  261   Just blocks -> T.intercalate "\n\n"
  262     [ t | b <- blocks, blockType b == Just "text"
  263         , Just t <- [strField "text" =<< asObject b] ]
  264   Nothing -> ""
  265 
  266 -- | The @tool_use@ calls in a response, shape-gated.
  267 toolsOf :: Value -> [ToolName]
  268 toolsOf payload = case blockArray payload of
  269   Just blocks ->
  270     [ ToolName (fromMaybe "" tid) (safeOpLabel raw)
  271     | b <- blocks
  272     , blockType b == Just "tool_use"
  273     , Object bo <- [b]
  274     , let raw = fromMaybe "" (strField "name" bo)
  275     , let tid = strField "id" bo
  276     ]
  277   Nothing -> []
  278 
  279 -- | The @(tool_use_id, is_error)@ pairs a request entry carries. This is the
  280 -- only thing read from tool-result blocks — never their content.
  281 resultsOf :: Value -> [(Text, Bool)]
  282 resultsOf payload =
  283   case KeyMap.lookup (Key.fromText "messages") =<< asObject payload of
  284     Just (Array msgs) -> concatMap fromMessage (V.toList msgs)
  285     _ -> []
  286   where
  287     fromMessage (Object m) = case KeyMap.lookup (Key.fromText "content") m of
  288       Just (Array blocks) -> mapMaybe fromBlock (V.toList blocks)
  289       _ -> []
  290     fromMessage _ = []
  291     fromBlock (Object b)
  292       | blockType (Object b) == Just "tool_result"
  293       , Just tid <- strField "tool_use_id" b =
  294           Just (tid, fromMaybe False (boolField "is_error" b))
  295       | otherwise = Nothing
  296     fromBlock _ = Nothing
  297 
  298 blockArray :: Value -> Maybe [Value]
  299 blockArray payload = case KeyMap.lookup (Key.fromText "content") =<< asObject payload of
  300   Just (Array arr) -> Just (V.toList arr)
  301   _ -> Nothing
  302 
  303 blockType :: Value -> Maybe Text
  304 blockType b = strField "type" =<< asObject b
  305 
  306 asObject :: Value -> Maybe (KeyMap.KeyMap Value)
  307 asObject (Object o) = Just o
  308 asObject _ = Nothing
  309 
  310 strField :: Text -> KeyMap.KeyMap Value -> Maybe Text
  311 strField k o = case KeyMap.lookup (Key.fromText k) o of
  312   Just (String t) -> Just t
  313   _ -> Nothing
  314 
  315 boolField :: Text -> KeyMap.KeyMap Value -> Maybe Bool
  316 boolField k o = case KeyMap.lookup (Key.fromText k) o of
  317   Just (Bool b) -> Just b
  318   _ -> Nothing

  324 -- | Whether a tool call succeeded. Tri-state because a 'Bool' cannot express
  325 -- "no result in the window": a tool-only response must still name what ran
  326 -- (DoD 2 wants the opcode), and rendering an unfinished call bare would
  327 -- silently assert success.
  328 data ToolState = LtsOk | LtsFailed | LtsUnfinished
  329   deriving stock (Eq, Show)
  330 
  331 data ToolOutcome = ToolOutcome
  332   { lsaName  :: Maybe Text   -- ^ 'safeOpLabel' result; 'Nothing' ⇒ unknown opcode
  333   , lsaState :: !ToolState
  334   } deriving stock (Eq)      -- no Show, as a rule for content-bearing types

  336 data AssistantItem
  337   = LriText Text
  338   | LriToolOnly ToolOutcome
  339   deriving stock (Eq)
  340 
  341 -- | Pair responses with their text or their tool calls, chronologically,
  342 -- resolving each tool's pass/fail by joining on @tool_use_id@.
  343 --
  344 -- The join needs both halves: @tool_use@ ids live in responses, their results
  345 -- in requests. A @tool_use@ with no matching result is 'LtsUnfinished'.
  346 assistantItems :: [Value] -> [AssistantItem]
  347 assistantItems entries = mapMaybe toItem facts
  348   where
  349     facts = map extractEntry entries
  350     results = concatMap lefResults facts
  351 
  352     toItem f
  353       | not (T.null (lefText f)) = Just (LriText (lefText f))
  354       | (t : _) <- lefTools f = Just (LriToolOnly (outcome t))
  355       | otherwise = Nothing
  356 
  357     outcome t = ToolOutcome (tnName t) (stateOf (tnId t))
  358     stateOf tid = case lookup tid results of
  359       Just True  -> LtsFailed
  360       Just False -> LtsOk
  361       Nothing    -> LtsUnfinished

  377 data LastView
  378   = LastMessages TabHeading [AssistantItem]
  379   | LastQuestion TabHeading PendingQuestion
  380   | LastWorking  TabHeading (Maybe ToolOutcome)
  381   | LastEmpty    TabHeading
  382   deriving stock (Eq)        -- no Show, as a rule for content-bearing types

  384 data ResolveInput = ResolveInput
  385   { lriHeading   :: TabHeading
  386   , lriItems     :: [AssistantItem]
  387   , lriThinking  :: Bool
  388   , lriQuestion  :: Maybe PendingQuestion
  389   , lriCount     :: Int        -- ^ requested n, already clamped 1..5
  390   , lriLegacy    :: Bool
  391   } deriving stock (Eq)        -- no Show: holds AssistantItem/PendingQuestion
  392 
  393 -- | Pick the one view that describes this tab right now.
  394 --
  395 -- Precedence, and why: a pending question is the most actionable state and a
  396 -- blocked turn is *also* thinking, so it must win or it is unreachable;
  397 -- thinking beats messages so a busy tab never implies it finished; anything
  398 -- else with items is 'LastMessages'; otherwise 'LastEmpty'. Pure.
  399 resolveLast :: ResolveInput -> LastView
  400 resolveLast ri
  401   | Just q <- lriQuestion ri = LastQuestion (lriHeading ri) q
  402   | lriLegacy ri, null (lriItems ri) = LastEmpty (lriHeading ri)
  403   | lriThinking ri =
  404       LastWorking (lriHeading ri) (lastCompleted (lriItems ri))
  405   | not (null (lriItems ri)) =
  406       LastMessages (lriHeading ri) (takeLast (lriCount ri) (lriItems ri))
  407   | otherwise = LastEmpty (lriHeading ri)
  408   where
  409     -- The newest *completed* tool, so "last completed" is never a lie.
  410     lastCompleted items =
  411       case [ o | LriToolOnly o <- items, lsaState o /= LtsUnfinished ] of
  412         [] -> Nothing
  413         os -> Just (last os)
  414 
  415 -- | The newest @n@ items, in chronological order.
  416 takeLast :: Int -> [AssistantItem] -> [AssistantItem]
  417 takeLast n xs = drop (max 0 (length xs - n)) xs
  418 
  419 -- | A stable tag for a view, for tests and callers that branch without
  420 -- rendering. The view types carry no 'Show', so this is how structure is
  421 -- asserted.
  422 viewTag :: LastView -> Text
  423 viewTag v = case v of
  424   LastMessages {} -> "messages"
  425   LastQuestion {} -> "question"
  426   LastWorking {}  -> "working"
  427   LastEmpty {}    -> "empty"

  429 -- | The tool lines a view contains, in order.
  430 renderedTools :: LastView -> [Text]
  431 renderedTools v = case v of
  432   LastMessages _ items -> [ toolLine o | LriToolOnly o <- items ]
  433   LastWorking _ (Just o) -> [ toolLine o ]
  434   LastWorking _ Nothing -> []
  435   LastQuestion {} -> []
  436   LastEmpty {} -> []

  438 -- | The tool line for the newest tool in a turn, or 'Nothing' when the turn
  439 -- ran no tool. Used by the watch notification, so a bodyless notification
  440 -- names what ran instead of being empty.
  441 --
  442 -- Deliberately the same 3-state rendering as 'renderedTools', so a tool reads
  443 -- identically wherever it appears.
  444 mostRecentToolLine :: [AssistantItem] -> Maybe Text
  445 mostRecentToolLine items = case [ o | LriToolOnly o <- items ] of
  446   [] -> Nothing
  447   os -> Just (toolLine (last os))

  449 -- | One tool line. 'LtsUnfinished' is rendered distinctly from 'LtsOk',
  450 -- because it means "no result in the window" — i.e. unknown — and a bare name
  451 -- would assert success.
  452 toolLine :: ToolOutcome -> Text
  453 toolLine o =
  454   opEmoji nameText <> " " <> nameText <> suffix
  455   where
  456     nameText = fromMaybe "(unknown opcode)" (lsaName o)
  457     suffix = case lsaState o of
  458       LtsOk         -> ""
  459       LtsFailed     -> " (failed)"
  460       LtsUnfinished -> " (no result)"

  466 -- | Render a view to the exact bytes the channel sends.
  467 --
  468 -- One contract: a heading ends with @:@ iff a body follows, and the body
  469 -- begins after exactly one blank line. A heading with no body ends with @.@.
  470 renderLast :: LastView -> Text
  471 renderLast view = case view of
  472   LastEmpty h -> heading h <> " \8212 no messages yet."
  473   LastMessages h [] -> renderLast (LastEmpty h)
  474   LastMessages h items ->
  475     heading h <> " \8212 " <> countLabel (length items) <> ":\n\n"
  476       <> renderItems (lthIndex h) items <> hintFor (lthIndex h) (length items) items
  477   LastWorking h mt ->
  478     heading h <> " \8212 still working:\n\n"
  479       <> maybe "no tools have completed yet" (\o -> "last completed: " <> toolLine o) mt
  480   LastQuestion h q ->
  481     heading h <> " \8212 waiting on your answer:\n\n"
  482       <> quoteBody (formatQ q) <> "\n\n" <> answerFooter (lthIndex h) q
  483   where
  484     formatQ q = case lqOptions q of
  485       [] -> lqText q
  486       os -> lqText q <> "\n\n"
  487               <> T.intercalate "\n" (zipWith optLine [1 :: Int ..] os)
  488     optLine n (QuestionOption lbl desc)
  489       | T.null desc = T.pack (show n) <> ") " <> lbl
  490       | otherwise   = T.pack (show n) <> ") " <> lbl <> " \8212 " <> desc
  491 
  492 -- | @📔 Tab b (title)@ — the title is omitted when empty.
  493 heading :: TabHeading -> Text
  494 heading h =
  495   "\128212 Tab " <> T.singleton (tabIndexToChar (lthIndex h)) <> titlePart
  496   where
  497     titlePart = case lthTitle h of
  498       Just t | not (T.null (T.strip t)) -> " (" <> t <> ")"
  499       _ -> ""
  500 
  501 countLabel :: Int -> Text
  502 countLabel 1 = "last message"
  503 countLabel n = "last " <> T.pack (show n) <> " messages"
  504 
  505 -- | Items, each attributed with the tab char when there is more than one
  506 -- (a chunk boundary then lands inside or between attributed items).
  507 renderItems :: TabIndex -> [AssistantItem] -> Text
  508 renderItems idx items = T.intercalate "\n\n" (map renderItem items)
  509   where
  510     -- More than one item ⇒ each is attributed, so a chunk boundary that falls
  511     -- between items still tells the reader which tab the text came from.
  512     prefix = if length items > 1 then T.singleton (tabIndexToChar idx) <> ": " else ""
  513     renderItem item = prefix <> case item of
  514       LriText t -> t
  515       LriToolOnly o -> toolLine o <> " \8212 no reply text."
  516 
  517 -- | The "for more" hint, for a text-free block whose window was saturated (so
  518 -- more may exist). Without the saturation gate the hint repeats forever, which
  519 -- is the dead end it exists to avoid. It names the actual tab char, and never
  520 -- re-offers the same count.
  521 hintFor :: TabIndex -> Int -> [AssistantItem] -> Text
  522 hintFor idx n items
  523   | any isText items = ""
  524   | n >= 5 = " /b to switch to that tab."
  525   | otherwise = " /last " <> T.singleton (tabIndexToChar idx) <> " 5 for more."
  526   where
  527     isText (LriText _) = True
  528     isText _ = False

  530 -- | Quote model-authored text so it cannot be mistaken for harness output.
  531 --
  532 -- Every non-blank line gets @"> "@; a blank line becomes a bare @">"@ (no
  533 -- trailing space). Line separators other than @\n@ are normalised first,
  534 -- because @\r@, U+0085, U+2028 and U+2029 render as line breaks but would
  535 -- otherwise slip past 'T.lines' and escape the quote.
  536 quoteBody :: Text -> Text
  537 quoteBody t =
  538   T.intercalate "\n"
  539     [ if T.null l then ">" else "> " <> l | l <- T.lines (normalizeNewlines t) ]
  540 
  541 -- | Fold the line separators that clients render as breaks into @\n@.
  542 --
  543 -- @\r\n@ collapses to a single newline first, so a CRLF does not become two
  544 -- blank lines. The remaining separators are handled per character.
  545 normalizeNewlines :: Text -> Text
  546 normalizeNewlines = T.map other . T.replace "\r\n" "\n"
  547   where
  548     other '\r' = '\n'
  549     other '\x85' = '\n'
  550     other '\x2028' = '\n'
  551     other '\x2029' = '\n'
  552     other c = c
  553 
  554 -- | The answer footer. Both ask kinds get the same one, because a gate ask and
  555 -- an open-ended ask are indistinguishable on the wire (both carry no options).
  556 answerFooter :: TabIndex -> PendingQuestion -> Text
  557 answerFooter idx q = case lqOptions q of

  559   _  -> "Answer with /" <> c <> " 1 or /" <> c <> " <your answer>."

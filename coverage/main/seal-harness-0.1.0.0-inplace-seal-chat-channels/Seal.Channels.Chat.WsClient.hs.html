    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | WebSocket client for the gateway's stream server. Connects, sends
    3 -- @FocusOp@s to change the focused session, and spawns a background thread
    4 -- that decodes incoming @ServerEvent@s and calls a callback for each.
    5 --
    6 -- One WS connection per conversation (per the design's open question #4).
    7 -- The connection's focused session changes on @/N@ focus.
    8 module Seal.Channels.Chat.WsClient
    9   ( WsClient (..)
   10   , startWsClient
   11   , WsEventCallback
   12   ) where
   13 
   14 import Control.Concurrent (forkIO)
   15 import Control.Monad (void)
   16 import Control.Concurrent.MVar (newEmptyMVar, putMVar, takeMVar)
   17 import Control.Exception (SomeException, try)
   18 import Data.Aeson qualified as A
   19 import Data.ByteString.Lazy qualified as BL
   20 import Data.IORef (IORef, newIORef, readIORef, writeIORef)
   21 import Data.Text (Text)
   22 import Data.Text qualified as T

   24   (ClientApp, Connection, receiveData, runClient, sendTextData, sendClose)
   25 import System.Timeout (timeout)
   26

   28 import Seal.Gateway.Types.Stream

   31 -- | The callback the background reader calls for each decoded 'ServerEvent'.
   32 type WsEventCallback = ServerEvent -> IO ()
   33 
   34 -- | A handle to a running WS client connection.
   35 data WsClient = WsClient
   36   { wcFocus      :: SessionId -> IO ()
   37     -- ^ Send a 'FocusOp' for the given session (change focus).
   38   , wcFocusSince :: SessionId -> Maybe Text -> IO ()
   39     -- ^ Send a 'FocusOp' with an optional @since@ entry id (for replay).
   40   , wcClose      :: IO ()
   41     -- ^ Close the WS connection and stop the background reader.
   42   }
   43 
   44 -- | Connect to the gateway's WS server. Forks the connection in a
   45 -- background thread and returns a 'WsClient' handle immediately. Returns
   46 -- 'Left' if the connection fails within 5 seconds.
   47 startWsClient
   48   :: Text    -- ^ Host (e.g. @"127.0.0.1"@)
   49   -> Int     -- ^ Port (e.g. 8081)
   50   -> WsEventCallback
   51   -> IO (Either Text WsClient)
   52 startWsClient host port callback = do
   53   connRef <- newIORef Nothing
   54   readyVar <- newEmptyMVar
   55   let app :: ClientApp ()
   56       app conn = do
   57         writeIORef connRef (Just conn)
   58         putMVar readyVar ()
   59         readerLoop conn callback
   60   -- Fork the client connection so we can return immediately.
   61   _ <- forkIO $ void (try (runClient (T.unpack host) port "/" app) :: IO (Either SomeException ()))
   62   -- Wait up to 5s for the connection to be established.
   63   mReady <- timeout 5000000 (takeMVar readyVar)
   64   case mReady of
   65     Nothing -> pure (Left ("WS connect timeout: " <> host <> ":" <> T.pack (show port)))
   66     Just () -> do
   67       let client = WsClient
   68             { wcFocus = \sid -> sendFocusOpRef connRef sid Nothing
   69             , wcFocusSince = sendFocusOpRef connRef
   70             , wcClose = closeConnRef connRef
   71             }
   72       pure (Right client)
   73 
   74 -- | The read loop: decode incoming frames, call the callback for each.
   75 -- Exits when the connection closes or an error occurs.
   76 readerLoop :: Connection -> WsEventCallback -> IO ()
   77 readerLoop conn callback = go
   78   where
   79     go = do
   80       eData <- try (receiveData conn) :: IO (Either SomeException BL.ByteString)
   81       case eData of
   82         Left _ -> pure ()  -- connection closed
   83         Right bs ->
   84           case decodeServerEvent bs of
   85             Just ev -> callback ev >> go
   86             Nothing -> go  -- unparseable frame; skip
   87 
   88 -- | Send a 'FocusOp' via the connection stored in the 'IORef'.
   89 sendFocusOpRef :: IORef (Maybe Connection) -> SessionId -> Maybe Text -> IO ()
   90 sendFocusOpRef ref sid mSince = do
   91   mConn <- readIORef ref
   92   case mConn of
   93     Nothing -> pure ()  -- connection not established or closed
   94     Just conn -> sendFocusOp conn sid mSince
   95 
   96 -- | Close the connection stored in the 'IORef'.
   97 closeConnRef :: IORef (Maybe Connection) -> IO ()
   98 closeConnRef ref = do
   99   mConn <- readIORef ref
  100   case mConn of
  101     Nothing -> pure ()
  102     Just conn -> do
  103       _ <- try (sendClose conn ("client closed" :: Text)) :: IO (Either SomeException ())
  104       writeIORef ref Nothing
  105 
  106 -- | Send a 'FocusOp' over the WS connection.
  107 sendFocusOp :: Connection -> SessionId -> Maybe Text -> IO ()
  108 sendFocusOp conn sid mSince =
  109   sendTextData conn (A.encode (FocusOp (sessionIdText sid) mSince))

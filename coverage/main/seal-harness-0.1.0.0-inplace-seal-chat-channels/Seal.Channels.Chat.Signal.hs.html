   49 -- | The testability seam over signal-cli. Mirrors the existing
   50 -- 'Seal.Channels.Signal.Transport.SignalTransport' but in the
   51 -- chat-channels package namespace.
   52 data SignalChatTransport = SignalChatTransport
   53   { sctReceive :: IO (Either Text ReceivedMessage)
   54     -- ^ Next inbound message.
   55   , sctSend    :: Text -> Text -> IO ()
   56     -- ^ Send a message: recipient, body.
   57   , sctSendWithId :: Text -> Text -> IO (Maybe Text)
   58     -- ^ Send a message and return the timestamp (for editing).
   59   , sctEditMessage :: Text -> Text -> Text -> IO Bool
   60     -- ^ Edit a previously sent message: recipient, timestamp, new content.
   61   , sctClose   :: IO ()
   62   }
   63 
   64 -- | Chunk a message to the given character limit. Splits on word
   65 -- boundaries when possible.
   66 chunkMessage :: Int -> Text -> [Text]
   67 chunkMessage limit msg
   68   | T.length msg <= limit = [msg]
   69   | otherwise = go msg
   70   where
   71     go t
   72       | T.null t = []
   73       | T.length t <= limit = [t]
   74       | otherwise = T.take limit t : go (T.drop limit t)

   76 -- | A mock transport backed by a 'TQueue' of inbound messages and an
   77 -- 'IORef's of captured sends and captured edits. For testing. The third
   78 -- return value exposes the captured @(timestamp, content)@ edit pairs so
   79 -- streaming-edit cadence is observable in tests.
   80 mkMockSignalChatTransport :: [ReceivedMessage] -> IO (SignalChatTransport, IO [Text], IO [(Text, Text)])
   81 mkMockSignalChatTransport scripted = do
   82   q <- newTQueueIO
   83   mapM_ (atomically . writeTQueue q) scripted
   84   capRef <- newIORef []
   85   editRef <- newIORef ([] :: [(Text, Text)])
   86   tsRef <- newIORef (1000 :: Int)
   87   let transport = SignalChatTransport
   88         { sctReceive = do
   89             m <- atomically (tryReadTQueue q)
   90             case m of
   91               Just msg -> pure (Right msg)
   92               Nothing -> pure (Left "inbox empty")
   93         , sctSend = \_r b -> modifyIORef' capRef (b :)
   94         , sctSendWithId = \_r b -> do
   95             n <- readIORef tsRef
   96             writeIORef tsRef (n + 1)
   97             modifyIORef' capRef (b :)  -- sendWithId is also a visible send
   98             pure (Just (T.pack (show (n + 1))))
   99         , sctEditMessage = \_r ts content -> do
  100             modifyIORef' editRef ((ts, content) :)
  101             pure True
  102         , sctClose = pure ()
  103         }
  104       getCaptured = reverse <$> readIORef capRef
  105       getEdits = reverse <$> readIORef editRef
  106   pure (transport, getCaptured, getEdits)

  112 -- | Spawn @signal-cli --output=json --trust-new-identities=always
  113 -- -u <account> jsonRpc@ as a child process, line-buffered JSON-RPC over
  114 -- stdin/stdout. Returns 'Left' if signal-cli is not installed or fails to
  115 -- start.
  116 mkRealSignalChatTransport :: Text -> IO (Either Text SignalChatTransport)
  117 mkRealSignalChatTransport account = do
  118   versionOk <- probeSignalCli
  119   if not versionOk
  120     then pure (Left "signal-cli not installed or on PATH")
  121     else do
  122       eStarted <- try @IOException $
  123         createProcess
  124           ( (proc "signal-cli"
  125               [ "--output=json"
  126               , "--trust-new-identities=always"
  127               , "-u", T.unpack account
  128               , "jsonRpc"
  129               ])
  130               { std_in = CreatePipe, std_out = CreatePipe, std_err = Inherit }
  131           )
  132       case eStarted of
  133         Left e -> pure (Left ("signal-cli launch failed: " <> T.pack (show e)))
  134         Right (mIn, mOut, _err, ph) -> do
  135           (hIn, hOut) <- case (mIn, mOut) of
  136             (Just a, Just b) -> pure (a, b)
  137             _ -> error "mkRealSignalChatTransport: pipe creation failed (unreachable)"
  138           hSetBuffering hIn (BlockBuffering Nothing)
  139           hSetBuffering hOut LineBuffering
  140           inbox <- newTQueueIO
  141           readerDead <- newTVarIO False
  142           idRef <- newIORef (0 :: Int)
  143           respMap <- newIORef (Map.empty :: Map.Map Int (MVar (Maybe Value)))
  144           _ <- forkIO (demuxReader hOut inbox respMap readerDead)
  145           sendLock <- newMVar ()
  146           let writeFrame :: Value -> IO ()
  147               writeFrame frame = withMVar sendLock $ \_ -> do
  148                 BL.hPutStr hIn (A.encode frame)
  149                 TIO.hPutStrLn hIn ""
  150                 hFlush hIn
  151           let sendRequest :: Value -> IO (Maybe Value)
  152               sendRequest frame = do
  153                 rid <- atomicModifyIORef' idRef (\n -> (n + 1, n))
  154                 mv <- newEmptyMVar
  155                 _ <- atomicModifyIORef' respMap (\m -> (Map.insert rid mv m, m))
  156                 let ridVal = A.Number (fromIntegral rid)
  157                     framed = case frame of
  158                       A.Object o -> A.Object (KeyMap.insert (Key.fromString "id") ridVal o)
  159                       _ -> frame
  160                 writeFrame framed
  161                 mResult <- timeout 10000000 (takeMVar mv)
  162                 _ <- atomicModifyIORef' respMap (\m -> (Map.delete rid m, m))
  163                 pure (case mResult of
  164                   Just (Just v) -> Just v
  165                   _ -> Nothing)
  166           pure (Right SignalChatTransport
  167             { sctReceive = receiveBlocking inbox readerDead
  168             , sctSend = \recipient body -> do
  169                 let frame = A.object
  170                       [ "jsonrpc" A..= ("2.0" :: Text)
  171                       , "method"  A..= ("send" :: Text)
  172                       , "params"  A..= A.object
  173                           [ "recipient" A..= [recipient]
  174                           , "message"   A..= body
  175                           ]
  176                       ]
  177                 writeFrame frame
  178             , sctSendWithId = \recipient body -> do
  179                 let frame = A.object
  180                       [ "jsonrpc" A..= ("2.0" :: Text)
  181                       , "method"  A..= ("send" :: Text)
  182                       , "params"  A..= A.object
  183                           [ "recipient" A..= [recipient]
  184                           , "message"   A..= body
  185                           ]
  186                       ]
  187                 mResp <- sendRequest frame
  188                 pure (extractTimestamp =<< mResp)
  189             , sctEditMessage = \recipient ts content -> do
  190                 let frame = A.object
  191                       [ "jsonrpc" A..= ("2.0" :: Text)
  192                       , "method"  A..= ("send" :: Text)
  193                       , "params"  A..= A.object
  194                           [ "recipient" A..= [recipient]
  195                           , "message"   A..= content
  196                           , "editTimestamp" A..= (read (T.unpack ts) :: Int)
  197                           ]
  198                       ]
  199                 mResp <- sendRequest frame
  200                 pure (case mResp of
  201                   Just _  -> True
  202                   Nothing -> False)
  203             , sctClose = do
  204                 _ <- try @IOException (hClose hIn)
  205                 terminateProcess ph
  206                 _ <- timeout 5000000 (waitForProcess ph)
  207                 pure ()
  208             })

  210 -- | The background demux reader: reads all lines from signal-cli's
  211 -- stdout and classifies each:
  212 -- * JSON-RPC response (has @result@ or @error@ + @id@) → route to the
  213 --   waiting caller via the MVar keyed by @id@ in 'respMap'.
  214 -- * JSON-RPC notification (has @method@ + @params@, no @id@) → parse
  215 --   into a 'ReceivedMessage' and push to the inbox 'TQueue'.
  216 demuxReader
  217   :: Handle -> TQueue ReceivedMessage
  218   -> IORef (Map.Map Int (MVar (Maybe Value))) -> TVar Bool
  219   -> IO ()
  220 demuxReader hOut inbox respMap readerDead = go
  221   where
  222     go = do
  223       eLine <- try @IOException (hGetLine hOut)
  224       case eLine of
  225         Left _ -> atomically (writeTVar readerDead True)
  226         Right line -> do
  227           case A.decode (BL.fromStrict (TE.encodeUtf8 (T.pack line))) of
  228             Nothing -> go
  229             Just v -> do
  230               case extractResponseId v of
  231                 Just rid -> do
  232                   m <- readIORef respMap
  233                   case Map.lookup rid m of
  234                     Just mv -> putMVar mv (Just v)
  235                     Nothing -> pure ()
  236                 Nothing -> do
  237                   case parseSignalEnvelope v of
  238                     Right msg -> atomically (writeTQueue inbox msg)
  239                     Left _    -> pure ()
  240           go
  241 
  242 -- | Block until the demux reader's inbox yields a value; return 'Left'
  243 -- only when the reader is dead AND the inbox is drained (EOF).
  244 receiveBlocking :: TQueue ReceivedMessage -> TVar Bool -> IO (Either Text ReceivedMessage)
  245 receiveBlocking inbox readerDead =
  246   atomically $
  247     (Right <$> readTQueue inbox)
  248       `orElse` do
  249         dead <- readTVar readerDead
  250         check dead
  251         isEmpty <- isEmptyTQueue inbox
  252         check isEmpty
  253         pure (Left "signal inbox empty")
  254 
  255 -- | Extract the @id@ field from a JSON-RPC response.
  256 extractResponseId :: Value -> Maybe Int
  257 extractResponseId v =
  258   case v of
  259     A.Object o -> case KeyMap.lookup (Key.fromString "id") o of
  260       Just (A.Number n) -> Just (round n)
  261       _ -> Nothing
  262     _ -> Nothing
  263 
  264 -- | Extract the @result.timestamp@ from a JSON-RPC @send@ response.
  265 extractTimestamp :: Value -> Maybe Text
  266 extractTimestamp v =
  267   case v of
  268     A.Object o -> case KeyMap.lookup (Key.fromString "result") o of
  269       Just (A.Object ro) -> case KeyMap.lookup (Key.fromString "timestamp") ro of
  270         Just (A.Number n) -> Just (T.pack (show (round n :: Int)))
  271         _ -> Nothing
  272       _ -> Nothing
  273     _ -> Nothing

  275 -- | Preflight @signal-cli --version@.
  276 probeSignalCli :: IO Bool
  277 probeSignalCli = do
  278   r <- try @IOException $
  279         withCreateProcess
  280           ( (proc "signal-cli" ["--version"])
  281               { std_out = CreatePipe, std_err = CreatePipe }
  282           ) $ \_ _ _ ph -> waitForProcess ph
  283   pure $ case r of
  284     Right ExitSuccess -> True
  285     _                 -> False

  287 -- | Parse a raw signal-cli JSON value into a 'ReceivedMessage'. Handles
  288 -- both raw envelopes (@{"envelope": {...}}@) and JSON-RPC-wrapped messages
  289 -- (@{"jsonrpc":"2.0","method":"receive","params":{"envelope":{...}}}@).
  290 parseSignalEnvelope :: Value -> Either Text ReceivedMessage
  291 parseSignalEnvelope v = do
  292   env <- unwrapEnvelope v
  293   source <- fieldText "source" env
  294   let mUuid = fieldTextMaybe "sourceUuid" env
  295   cid <- conversationIdForSignal (Just source) mUuid
  296   let body = extractBody env
  297       replyTo = source
  298   Right ReceivedMessage
  299     { rmConversationId = cid
  300     , rmSender = Just source
  301     , rmReplyTo = replyTo
  302     , rmBody = body
  303     , rmCallbackData = Nothing
  304     , rmCallbackId = Nothing
  305     , rmCallbackMessageId = Nothing
  306     }
  307 
  308 -- | Derive the 'ConversationId' from the peer's authenticated transport
  309 -- metadata.
  310 conversationIdForSignal :: Maybe Text -> Maybe Text -> Either Text ConversationId
  311 conversationIdForSignal mSource mUuid =
  312   case mSource of
  313     Nothing        -> Left "signal envelope missing source (peer phone number)"
  314     Just src
  315       | T.null src -> Left "signal envelope source is empty"
  316       | otherwise  -> mkConversationId full
  317       where
  318         full = case mUuid of
  319           Nothing   -> "sig:" <> src
  320           Just uuid -> "sig:" <> src <> ":" <> uuid
  321 
  322 -- | Unwrap a raw envelope or a JSON-RPC-wrapped message.
  323 unwrapEnvelope :: Value -> Either Text Value
  324 unwrapEnvelope v =
  325   case v of
  326     A.Object o ->
  327       case KeyMap.lookup (Key.fromString "envelope") o of
  328         Just env -> Right env
  329         Nothing -> case KeyMap.lookup (Key.fromString "params") o of
  330           Just (A.Object p) -> case KeyMap.lookup (Key.fromString "envelope") p of
  331             Just env -> Right env
  332             Nothing  -> Left "signal envelope: no envelope in params"
  333           _ -> Left "signal envelope: no envelope and no params"
  334     _ -> Left "signal envelope: not an object"
  335 
  336 -- | Extract a required text field from an object.
  337 fieldText :: Text -> Value -> Either Text Text
  338 fieldText key v =
  339   case v of
  340     A.Object o -> case KeyMap.lookup (Key.fromString (T.unpack key)) o of
  341       Just (A.String t) -> Right t
  342       _ -> Left ("signal envelope missing or non-text field: " <> key)
  343     _ -> Left ("signal envelope field " <> key <> ": not an object")
  344 
  345 -- | Extract an optional text field from an object.
  346 fieldTextMaybe :: Text -> Value -> Maybe Text
  347 fieldTextMaybe key v = do
  348   inner <- fieldValueMaybe key v
  349   case inner of
  350     A.String t -> Just t
  351     _ -> Nothing
  352 
  353 -- | Extract an optional sub-value from an object.
  354 fieldValueMaybe :: Text -> Value -> Maybe Value
  355 fieldValueMaybe key v =
  356   case v of
  357     A.Object o -> KeyMap.lookup (Key.fromString (T.unpack key)) o
  358     _ -> Nothing

  360 -- | Extract the message body from @dataMessage.message@.
  361 extractBody :: Value -> Text
  362 extractBody v = fromMaybe "" $ do
  363   dm <- fieldValueMaybe "dataMessage" v
  364   fieldTextMaybe "message" dm

  370 -- | The live Signal chat channel state.
  371 data SignalChatChannel = SignalChatChannel
  372   { sccAllowList   :: AllowList UserId
  373   , sccChunkLimit  :: Int
  374   , sccInbox       :: TQueue InboundMessage
  375   , sccTransport   :: SignalChatTransport
  376   , sccLastSender  :: IORef (Maybe Text)
  377   , sccReaderAlive :: IORef Bool
  378   }
  379 
  380 instance ChatChannel SignalChatChannel where
  381   ccReceive ch = do
  382     mMsg <- atomically (tryReadTQueue (sccInbox ch))
  383     case mMsg of
  384       Just msg -> pure (Just msg)
  385       Nothing -> do
  386         alive <- readIORef (sccReaderAlive ch)
  387         if alive
  388           then threadDelay 1000 >> ccReceive ch
  389           else pure Nothing
  390 
  391   ccSend ch t =
  392     mapM_ (sendRaw ch) (chunkMessage (sccChunkLimit ch) t)
  393 
  394   ccSendWithId ch t = do
  395     mSender <- readIORef (sccLastSender ch)
  396     case mSender of
  397       Nothing -> pure Nothing
  398       Just sender -> do
  399         mId <- sctSendWithId (sccTransport ch) sender t
  400         pure (ChatMessageId <$> mId)
  401 
  402   ccEditMessage ch (ChatMessageId msgId) content = do
  403     mSender <- readIORef (sccLastSender ch)
  404     case mSender of
  405       Nothing -> pure False
  406       Just sender -> sctEditMessage (sccTransport ch) sender msgId content
  407 
  408   ccLabel _ = "signal"
  409 
  410   -- Signal doesn't reliably support message editing (signal-cli's
  411   -- editTimestamp is flaky in practice). Disable streaming: send the
  412   -- final text as a single message after the turn completes.
  413   ccSupportsStreaming _ = False
  414 
  415 -- | Send one chunk verbatim to the last sender.
  416 sendRaw :: SignalChatChannel -> Text -> IO ()
  417 sendRaw ch t = do
  418   mSender <- readIORef (sccLastSender ch)
  419   case mSender of
  420     Nothing -> pure ()
  421     Just sender -> sctSend (sccTransport ch) sender t

  423 -- | Run the reader thread with cleanup. Spawns a background thread that
  424 -- loops 'sctReceive', creates a 'MessageSource', and pushes
  425 -- 'InboundMessage's to 'sccInbox'. On transport close or exception, the
  426 -- thread exits.
  427 withSignalChatChannel
  428   :: AllowList UserId
  429   -> Int
  430   -> SignalChatTransport
  431   -> (SignalChatChannel -> IO a)
  432   -> IO a
  433 withSignalChatChannel allow chunkLimit transport action =
  434   bracket before after (action . snd)
  435   where
  436     before = do
  437       inbox <- newTQueueIO
  438       lastSender <- newIORef (Nothing :: Maybe Text)
  439       alive <- newIORef True
  440       let ch = SignalChatChannel
  441             { sccAllowList = allow
  442             , sccChunkLimit = chunkLimit
  443             , sccInbox = inbox
  444             , sccTransport = transport
  445             , sccLastSender = lastSender
  446             , sccReaderAlive = alive
  447             }
  448       tid <- forkIO (readerLoop ch)
  449       pure (tid, ch)
  450     after (tid, _) = do
  451       killThread tid
  452       sctClose transport
  453 
  454 -- | The background reader: loop 'sctReceive', construct a 'MessageSource'
  455 -- from the received conversation id + sender, update the last sender, and
  456 -- push to the inbox. Exits when 'sctReceive' returns 'Left'.
  457 readerLoop :: SignalChatChannel -> IO ()
  458 readerLoop ch = go
  459   where
  460     go = do
  461       eVal <- try (sctReceive (sccTransport ch)) :: IO (Either SomeException (Either Text ReceivedMessage))
  462       case eVal of
  463         Left _ -> writeIORef (sccReaderAlive ch) False
  464         Right (Left _) -> writeIORef (sccReaderAlive ch) False
  465         Right (Right (ReceivedMessage cid mSender replyTo body _ _ _))
  466           | T.null body -> go
  467           | otherwise -> do
  468               case mkMessageSource cid Signal (mkSender <$> mSender) mempty of
  469                 Right ms -> do
  470                   writeIORef (sccLastSender ch) (Just replyTo)
  471                   atomically (writeTQueue (sccInbox ch) (InboundMessage ms body Nothing))
  472                 Left _ -> pure ()
  473               go

  475 -- | Construct a 'UserId' from a sender text, best-effort.
  476 mkSender :: Text -> UserId
  477 mkSender t = case mkUserId t of
  478   Right u -> u
  479   Left _ -> case mkUserId "unknown" of Right u -> u; Left _ -> error "unreachable"

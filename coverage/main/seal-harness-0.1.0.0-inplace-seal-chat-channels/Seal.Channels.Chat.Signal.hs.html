   51 -- | The testability seam over signal-cli. Mirrors the existing
   52 -- 'Seal.Channels.Signal.Transport.SignalTransport' but in the

   54 data SignalChatTransport = SignalChatTransport
   55   { sctReceive :: IO (Either Text ReceivedMessage)

   60     -- ^ Send a message and return the timestamp (for editing).

   62     -- ^ Edit a previously sent message: recipient, timestamp, new content.

   66 -- | Split a reply to Signal's byte budget, breaking at the most natural
   67 -- boundary that fits (paragraph, line, sentence, clause, word).
   68 --
   69 -- Delegates to 'Seal.Channels.Chat.Bubble.chunkMessage' and measures in UTF-8
   70 -- BYTES: Signal inlines only the first ~2048 bytes of a message and moves the
   71 -- rest to an attachment, so a code-point budget would overrun it by up to 4x
   72 -- for emoji or CJK text and silently demote the tail to a file.
   73 chunkMessage :: Int -> Text -> [Text]
   74 chunkMessage = Bubble.chunkMessage ChunkUtf8Bytes

   76 -- | A mock transport backed by a 'TQueue' of inbound messages and an
   77 -- 'IORef's of captured sends and captured edits. For testing. The third
   78 -- return value exposes the captured @(timestamp, content)@ edit pairs so
   79 -- streaming-edit cadence is observable in tests.
   80 mkMockSignalChatTransport :: [ReceivedMessage] -> IO (SignalChatTransport, IO [Text], IO [(Text, Text)])
   81 mkMockSignalChatTransport scripted = do
   82   q <- newTQueueIO
   83   mapM_ (atomically . writeTQueue q) scripted
   84   capRef <- newIORef []
   85   editRef <- newIORef ([] :: [(Text, Text)])
   86   tsRef <- newIORef (1000 :: Int)
   87   let transport = SignalChatTransport
   88         { sctReceive = do
   89             m <- atomically (tryReadTQueue q)
   90             case m of
   91               Just msg -> pure (Right msg)
   92               Nothing -> pure (Left "inbox empty")
   93         , sctSend = \_r b -> modifyIORef' capRef (b :)
   94         , sctSendWithId = \_r b -> do
   95             n <- readIORef tsRef
   96             writeIORef tsRef (n + 1)
   97             modifyIORef' capRef (b :)  -- sendWithId is also a visible send
   98             pure (Just (T.pack (show (n + 1))))
   99         , sctEditMessage = \_r ts content -> do
  100             modifyIORef' editRef ((ts, content) :)
  101             pure True
  102         , sctClose = pure ()
  103         }
  104       getCaptured = reverse <$> readIORef capRef
  105       getEdits = reverse <$> readIORef editRef
  106   pure (transport, getCaptured, getEdits)

  112 -- | Spawn @signal-cli --output=json --trust-new-identities=always
  113 -- -u <account> jsonRpc@ as a child process, line-buffered JSON-RPC over
  114 -- stdin/stdout. Returns 'Left' if signal-cli is not installed or fails to
  115 -- start.
  116 mkRealSignalChatTransport :: Text -> IO (Either Text SignalChatTransport)
  117 mkRealSignalChatTransport account = do
  118   versionOk <- probeSignalCli
  119   if not versionOk
  120     then pure (Left "signal-cli not installed or on PATH")
  121     else do
  122       eStarted <- try @IOException $
  123         createProcess
  124           ( (proc "signal-cli"
  125               [ "--output=json"
  126               , "--trust-new-identities=always"
  127               , "-u", T.unpack account
  128               , "jsonRpc"
  129               ])
  130               { std_in = CreatePipe, std_out = CreatePipe, std_err = Inherit }
  131           )
  132       case eStarted of
  133         Left e -> pure (Left ("signal-cli launch failed: " <> T.pack (show e)))
  134         Right (mIn, mOut, _err, ph) -> do
  135           (hIn, hOut) <- case (mIn, mOut) of
  136             (Just a, Just b) -> pure (a, b)
  137             _ -> error "mkRealSignalChatTransport: pipe creation failed (unreachable)"
  138           hSetBuffering hIn (BlockBuffering Nothing)
  139           hSetBuffering hOut LineBuffering
  140           inbox <- newTQueueIO
  141           readerDead <- newTVarIO False
  142           idRef <- newIORef (0 :: Int)
  143           respMap <- newIORef (Map.empty :: Map.Map Int (MVar (Maybe Value)))
  144           _ <- forkIO (demuxReader hOut inbox respMap readerDead)
  145           sendLock <- newMVar ()
  146           let writeFrame :: Value -> IO ()
  147               writeFrame frame = withMVar sendLock $ \_ -> do
  148                 BL.hPutStr hIn (A.encode frame)
  149                 TIO.hPutStrLn hIn ""
  150                 hFlush hIn
  151           let sendRequest :: Value -> IO (Maybe Value)
  152               sendRequest frame = do
  153                 rid <- atomicModifyIORef' idRef (\n -> (n + 1, n))
  154                 mv <- newEmptyMVar
  155                 _ <- atomicModifyIORef' respMap (\m -> (Map.insert rid mv m, m))
  156                 let ridVal = A.Number (fromIntegral rid)
  157                     framed = case frame of
  158                       A.Object o -> A.Object (KeyMap.insert (Key.fromString "id") ridVal o)
  159                       _ -> frame
  160                 writeFrame framed
  161                 mResult <- timeout 10000000 (takeMVar mv)
  162                 _ <- atomicModifyIORef' respMap (\m -> (Map.delete rid m, m))
  163                 pure (case mResult of
  164                   Just (Just v) -> Just v
  165                   _ -> Nothing)
  166           pure (Right SignalChatTransport
  167             { sctReceive = receiveBlocking inbox readerDead
  168             , sctSend = \recipient body -> do
  169                 -- MUST await the response before returning, exactly like
  170                 -- 'sctSendWithId'. signal-cli >= 0.12.3 executes JSON-RPC
  171                 -- requests in PARALLEL, and the timestamp it assigns is the
  172                 -- message id: two sends in flight at once can be committed out
  173                 -- of order, and two that land on the same millisecond cause
  174                 -- the Signal apps to discard all but one (signal-cli #1783,
  175                 -- fixed upstream in 0.13.16 by "ensure every sent message gets
  176                 -- a unique timestamp"). 'ccSend' chunks a long answer and
  177                 -- 'mapM_'s over the chunks, so a fire-and-forget 'sctSend'
  178                 -- wrote every chunk frame back-to-back with nothing awaited
  179                 -- between them - which split the answer across messages and
  180                 -- delivered them in the wrong order. Awaiting here makes that
  181                 -- 'mapM_' a strict sequence.
  182                 let frame = signalSendFrame recipient body Nothing
  183                 mResp <- sendRequest frame
  184                 case mResp of
  185                   Just _  -> pure ()
  186                   Nothing -> logSendFailure "no-response"
  187             , sctSendWithId = \recipient body -> do
  188                 let frame = signalSendFrame recipient body Nothing
  189                 mResp <- sendRequest frame
  190                 pure (extractTimestamp =<< mResp)
  191             , sctEditMessage = \recipient ts content ->
  192                 -- The timestamp is an opaque identifier we produced ourselves,
  193                 -- but it still reaches a subprocess argument, so parse it
  194                 -- without a partial function: an unparseable value fails the
  195                 -- edit (the loop then falls back to a fresh send).
  196                 case TR.decimal ts of
  197                   Right (n, rest) | T.null rest, n >= 0 -> do
  198                     let frame = signalSendFrame recipient content (Just n)
  199                     mResp <- sendRequest frame
  200                     pure (case mResp of
  201                       Just _  -> True
  202                       Nothing -> False)
  203                   _ -> pure False
  204             , sctClose = do
  205                 _ <- try @IOException (hClose hIn)
  206                 terminateProcess ph
  207                 _ <- timeout 5000000 (waitForProcess ph)
  208                 pure ()
  209             })

  211 -- | Report a failed @send@ so a dropped message is diagnosable instead of
  212 -- silent. Carries ONLY a fixed classifier - never the message body, the
  213 -- recipient (a phone number or group id), or a raw exception, since this
  214 -- value reaches stderr and stderr reaches the operator's logs.
  215 logSendFailure :: Text -> IO ()
  216 logSendFailure reason =
  217   hPutStrLn stderr ("[signal] send failed: " <> T.unpack reason)

  219 -- | The background demux reader: reads all lines from signal-cli's
  220 -- stdout and classifies each:
  221 -- * JSON-RPC response (has @result@ or @error@ + @id@) → route to the
  222 --   waiting caller via the MVar keyed by @id@ in 'respMap'.
  223 -- * JSON-RPC notification (has @method@ + @params@, no @id@) → parse
  224 --   into a 'ReceivedMessage' and push to the inbox 'TQueue'.
  225 demuxReader
  226   :: Handle -> TQueue ReceivedMessage
  227   -> IORef (Map.Map Int (MVar (Maybe Value))) -> TVar Bool
  228   -> IO ()
  229 demuxReader hOut inbox respMap readerDead = go
  230   where
  231     go = do
  232       eLine <- try @IOException (hGetLine hOut)
  233       case eLine of
  234         Left _ -> atomically (writeTVar readerDead True)
  235         Right line -> do
  236           case A.decode (BL.fromStrict (TE.encodeUtf8 (T.pack line))) of
  237             Nothing -> go
  238             Just v -> do
  239               case extractResponseId v of
  240                 Just rid -> do
  241                   m <- readIORef respMap
  242                   case Map.lookup rid m of
  243                     Just mv -> putMVar mv (Just v)
  244                     Nothing -> pure ()
  245                 Nothing -> do
  246                   case parseSignalEnvelope v of
  247                     Right msg -> atomically (writeTQueue inbox msg)
  248                     Left _    -> pure ()
  249           go

  251 -- | Block until the demux reader's inbox yields a value; return 'Left'
  252 -- only when the reader is dead AND the inbox is drained (EOF).
  253 receiveBlocking :: TQueue ReceivedMessage -> TVar Bool -> IO (Either Text ReceivedMessage)
  254 receiveBlocking inbox readerDead =
  255   atomically $
  256     (Right <$> readTQueue inbox)
  257       `orElse` do
  258         dead <- readTVar readerDead
  259         check dead
  260         isEmpty <- isEmptyTQueue inbox
  261         check isEmpty
  262         pure (Left "signal inbox empty")

  276   case v of
  277     A.Object o -> case KeyMap.lookup (Key.fromString "result") o of
  278       Just (A.Object ro) -> case KeyMap.lookup (Key.fromString "timestamp") ro of

  296 -- | Parse a raw signal-cli JSON value into a 'ReceivedMessage'. Handles
  297 -- both raw envelopes (@{"envelope": {...}}@) and JSON-RPC-wrapped messages
  298 -- (@{"jsonrpc":"2.0","method":"receive","params":{"envelope":{...}}}@).
  299 parseSignalEnvelope :: Value -> Either Text ReceivedMessage
  300 parseSignalEnvelope v = do
  301   env <- unwrapEnvelope v
  302   source <- fieldText "source" env
  303   let mUuid = fieldTextMaybe "sourceUuid" env
  304       -- Check for a group ID in the dataMessage. V1 groups carry
  305       -- @groupInfo.groupId@; V2 groups carry @groupV2.id@. Both are
  306       -- base64-encoded. When present, the conversation is the group
  307       -- (not the individual sender), so the conversation ID and reply
  308       -- target must reference the group, not the sender's phone.
  309       mGroupId = extractGroupId env
  310   cid <- conversationIdForSignal (Just source) mUuid mGroupId
  311   let body = extractBody env
  312       replyTo = case mGroupId of
  313         Just gid -> "group:" <> gid
  314         Nothing  -> source
  315   Right ReceivedMessage
  316     { rmConversationId = cid
  317     , rmSender = Just source
  318     , rmReplyTo = replyTo
  319     , rmBody = body
  320     , rmCallbackData = Nothing
  321     , rmCallbackId = Nothing
  322     , rmCallbackMessageId = Nothing
  323     }

  325 -- | Derive the 'ConversationId' from the peer's authenticated transport
  326 -- metadata, or from the group ID when the message is from a group chat.
  327 -- For group messages, the conversation ID is @sig:group:<sanitized-id>@
  328 -- so watch state and session routing are per-group-chat, not per-sender.
  329 -- For DMs, the conversation ID is @sig:<phone>:<uuid>@ as before.
  330 conversationIdForSignal :: Maybe Text -> Maybe Text -> Maybe Text -> Either Text ConversationId
  331 conversationIdForSignal mSource mUuid mGroupId =
  332   case mGroupId of
  333     Just gid -> mkConversationId ("sig:group:" <> sanitizeGroupId gid)
  334     Nothing  -> dmConversationId
  335   where
  336     dmConversationId =
  337       case mSource of
  338         Nothing        -> Left "signal envelope missing source (peer phone number)"
  339         Just src
  340           | T.null src -> Left "signal envelope source is empty"
  341           | otherwise  -> mkConversationId full
  342           where
  343             full = case mUuid of
  344               Nothing   -> "sig:" <> src
  345               Just uuid -> "sig:" <> src <> ":" <> uuid
  346 
  347 -- | Extract a group ID from a signal-cli envelope. Checks V1 groups
  348 -- (@dataMessage.groupInfo.groupId@) and V2 groups
  349 -- (@dataMessage.groupV2.id@). Returns 'Nothing' for DMs.
  350 extractGroupId :: Value -> Maybe Text
  351 extractGroupId env = do
  352   dm <- fieldValueMaybe "dataMessage" env
  353   -- V2 groups take precedence (modern Signal uses V2).
  354   case fieldTextMaybe "id" =<< fieldValueMaybe "groupV2" dm of
  355     Just gid -> Just gid
  356     Nothing  -> fieldTextMaybe "groupId" =<< fieldValueMaybe "groupInfo" dm
  357 
  358 -- | Sanitize a base64 group ID for use in a 'ConversationId'. The valid
  359 -- charset is @A-Za-z0-9_-:+@; base64 uses @A-Za-z0-9+/=@. Replace @/@
  360 -- with @_@ and strip @=@ padding. The result is unique per group and
  361 -- round-trippable (base64url without padding).
  362 sanitizeGroupId :: Text -> Text
  363 sanitizeGroupId = T.filter (/= '=') . T.map (\c -> if c == '/' then '_' else c)
  364 
  365 -- | Build a signal-cli JSON-RPC send frame. If the recipient starts with
  366 -- @group:@, the remainder is the group ID and the frame uses
  367 -- @groupId@ instead of @recipient@ (signal-cli routes group messages
  368 -- via @--group-id@, not @--recipient@). The optional edit timestamp is
  369 -- included for message edits.
  370 signalSendFrame :: Text -> Text -> Maybe Int -> Value
  371 signalSendFrame recipient body mEditTs =
  372   A.object
  373     [ "jsonrpc" A..= ("2.0" :: Text)
  374     , "method"  A..= ("send" :: Text)
  375     , "params"  A..= A.object (target <> [ "message" A..= body ] <> editField)
  376     ]
  377   where
  378     (target, editField) = case T.stripPrefix "group:" recipient of
  379       Just gid -> (["groupId" A..= gid], maybe [] (\ts -> ["editTimestamp" A..= ts]) mEditTs)
  380       Nothing  -> (["recipient" A..= [recipient]], maybe [] (\ts -> ["editTimestamp" A..= ts]) mEditTs)
  381 
  382 -- | Unwrap a raw envelope or a JSON-RPC-wrapped message.
  383 unwrapEnvelope :: Value -> Either Text Value
  384 unwrapEnvelope v =
  385   case v of
  386     A.Object o ->
  387       case KeyMap.lookup (Key.fromString "envelope") o of
  388         Just env -> Right env
  389         Nothing -> case KeyMap.lookup (Key.fromString "params") o of
  390           Just (A.Object p) -> case KeyMap.lookup (Key.fromString "envelope") p of
  391             Just env -> Right env
  392             Nothing  -> Left "signal envelope: no envelope in params"
  393           _ -> Left "signal envelope: no envelope and no params"
  394     _ -> Left "signal envelope: not an object"
  395 
  396 -- | Extract a required text field from an object.
  397 fieldText :: Text -> Value -> Either Text Text
  398 fieldText key v =
  399   case v of
  400     A.Object o -> case KeyMap.lookup (Key.fromString (T.unpack key)) o of
  401       Just (A.String t) -> Right t
  402       _ -> Left ("signal envelope missing or non-text field: " <> key)
  403     _ -> Left ("signal envelope field " <> key <> ": not an object")

  431 data SignalChatChannel = SignalChatChannel
  432   { sccAllowList   :: AllowList UserId
  433   , sccChunkLimit  :: Int
  434   , sccInbox       :: TQueue InboundMessage

  436   , sccLastSender  :: IORef (Maybe Text)

  440 instance ChatChannel SignalChatChannel where
  441   ccReceive ch = do
  442     mMsg <- atomically (tryReadTQueue (sccInbox ch))
  443     case mMsg of
  444       Just msg -> pure (Just msg)
  445       Nothing -> do
  446         alive <- readIORef (sccReaderAlive ch)
  447         if alive
  448           then threadDelay 1000 >> ccReceive ch
  449           else pure Nothing
  450 
  451   ccSend ch t =
  452     mapM_ (sendRaw ch) (chunkMessage (sccChunkLimit ch) t)
  453 
  454   ccSendWithId ch t = do
  455     mSender <- readIORef (sccLastSender ch)
  456     case mSender of
  457       Nothing -> pure Nothing
  458       Just sender -> do
  459         mId <- sctSendWithId (sccTransport ch) sender t
  460         pure (ChatMessageId <$> mId)
  461 
  462   ccEditMessage ch (ChatMessageId msgId) content = do
  463     mSender <- readIORef (sccLastSender ch)
  464     case mSender of
  465       Nothing -> pure False
  466       Just sender -> sctEditMessage (sccTransport ch) sender msgId content
  467 
  468   ccLabel _ = "signal"
  469 
  470   -- Signal supports editing a sent message: signal-cli >= 0.14 exposes
  471   -- editTimestamp on the JSON-RPC `send` method, and `sctSendWithId` /
  472   -- `sctEditMessage` above are built on it. Editing is what lets a turn
  473   -- produce ONE notification instead of one per text segment and tool call,
  474   -- so opts into the single-edited-bubble style. A failed edit falls back to
  475   -- a fresh send, so unreliable edit support degrades to the old behaviour
  476   -- rather than losing content.
  477   ccSupportsStreaming _ = True
  478 
  479   ccProgressStyle _ = PsSingleBubble
  480 
  481   ccChunkLimit ch = Just (sccChunkLimit ch)
  482 
  483   -- Signal's budget is UTF-8 BYTES: it inlines the first ~2048 bytes and
  484   -- moves the remainder to an attachment, so a code-point count undercounts
  485   -- multi-byte text by up to 4x.
  486   ccChunkUnit _ = ChunkUtf8Bytes

  488 -- | Send one chunk verbatim to the last sender.
  489 sendRaw :: SignalChatChannel -> Text -> IO ()
  490 sendRaw ch t = do
  491   mSender <- readIORef (sccLastSender ch)
  492   case mSender of
  493     Nothing -> pure ()
  494     Just sender -> sctSend (sccTransport ch) sender t
  495 -- | Run the reader thread with cleanup. Spawns a background thread that
  496 -- loops 'sctReceive', creates a 'MessageSource', and pushes
  497 -- 'InboundMessage's to 'sccInbox'. On transport close or exception, the
  498 -- thread exits. The initial backoff (microseconds) is used for retry
  499 -- delays on transient errors; it doubles on each consecutive error
  500 -- (capped at 30s) and resets on a successful receive.
  501 withSignalChatChannel
  502   :: AllowList UserId
  503   -> Int
  504   -> Int
  505   -- ^ Initial backoff in microseconds (for retry on transient errors)
  506   -> SignalChatTransport
  507   -> (SignalChatChannel -> IO a)
  508   -> IO a
  509 withSignalChatChannel allow chunkLimit initialBackoff transport action =
  510   bracket before after (action . snd)
  511   where
  512     before = do
  513       inbox <- newTQueueIO
  514       lastSender <- newIORef (Nothing :: Maybe Text)
  515       alive <- newIORef True
  516       let ch = SignalChatChannel
  517             { sccAllowList = allow
  518             , sccChunkLimit = chunkLimit
  519             , sccInbox = inbox
  520             , sccTransport = transport
  521             , sccLastSender = lastSender
  522             , sccReaderAlive = alive
  523             }
  524       tid <- forkIO (readerLoop ch initialBackoff)
  525       pure (tid, ch)
  526     after (tid, _) = do
  527       killThread tid
  528       sctClose transport

  530 -- | The background reader: loop 'sctReceive', construct a 'MessageSource'
  531 -- from the received conversation id + sender, update the last sender, and
  532 -- push to the inbox. Exits when 'sctReceive' returns 'Left'.
  533 -- Retries with exponential backoff on transient errors instead of dying.
  534 readerLoop :: SignalChatChannel -> Int -> IO ()
  535 readerLoop ch initialBackoff = go initialBackoff
  536   where
  537     maxBackoff = 30_000_000  -- 30 seconds
  538     go backoff = do
  539       eVal <- try (sctReceive (sccTransport ch)) :: IO (Either SomeException (Either Text ReceivedMessage))
  540       case eVal of
  541         Left ex -> do
  542           logErr ("reader exception: " <> T.pack (show ex))
  543           threadDelay backoff
  544           go (min (backoff * 2) maxBackoff)
  545         Right (Left err) -> do
  546           logErr ("receive error: " <> err)
  547           threadDelay backoff
  548           go (min (backoff * 2) maxBackoff)
  549         Right (Right (ReceivedMessage cid mSender replyTo body _ _ _))
  550           | T.null body -> go initialBackoff
  551           | otherwise -> do
  552               case mkMessageSource cid Signal (mkSender <$> mSender) mempty of
  553                 Right ms -> do
  554                   writeIORef (sccLastSender ch) (Just replyTo)
  555                   atomically (writeTQueue (sccInbox ch) (InboundMessage ms body Nothing))
  556                 Left _ -> pure ()
  557               go initialBackoff
  558     logErr msg = hPutStrLn stderr ("[signal] " <> T.unpack msg)
  559 
  560 -- | Construct a 'UserId' from a sender text, best-effort.
  561 mkSender :: Text -> UserId
  562 mkSender t = case mkUserId t of
  563   Right u -> u
  564   Left _ -> case mkUserId "unknown" of Right u -> u; Left _ -> error "unreachable"

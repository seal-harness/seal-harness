    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Pure rate-limiting functions for streaming text edits. Copied from
    3 -- 'Seal.Channels.StreamProgress' (the server-side implementation) so the
    4 -- chat-channel package can rate-limit WS @entry-update@ edits without
    5 -- depending on server internals.
    6 --
    7 -- These are the same pure functions: 'shouldEdit' decides when enough time
    8 -- or text has accumulated, 'addCursor' appends the cursor character for
    9 -- intermediate edits, 'stripCursor' removes it for the final edit.
   10 module Seal.Channels.Chat.RateLimit
   11   ( StreamProgressConfig (..)
   12   , defaultStreamProgressConfig
   13   , resolveStreamProgressConfig
   14   , shouldEdit
   15   , addCursor
   16   , stripCursor
   17   ) where
   18 
   19 import Data.Maybe (fromMaybe)
   20 import Data.Text (Text)
   21 import Data.Text qualified as T
   22 import Data.Time (UTCTime, diffUTCTime)
   23 
   24 -- | Configuration for streaming text edits. Matches the server-side
   25 -- 'Seal.Channels.StreamProgress.StreamProgressConfig' field-for-field.
   26 data StreamProgressConfig = StreamProgressConfig
   27   { spcEnabled         :: Bool
   28   , spcEditIntervalMs  :: Int    -- ^ minimum ms between edits
   29   , spcBufferThreshold :: Int    -- ^ codepoint count that forces an edit
   30   , spcCursor          :: Text   -- ^ the cursor character for intermediate edits
   31   } deriving stock (Eq, Show)
   32 
   33 -- | The default config: enabled, with sensible defaults matching the
   34 -- server-side config.
   35 defaultStreamProgressConfig :: StreamProgressConfig
   36 defaultStreamProgressConfig = StreamProgressConfig
   37   { spcEnabled = True
   38   , spcEditIntervalMs = 1500
   39   , spcBufferThreshold = 80
   40   , spcCursor = "\x2589"  -- █ left-seven-eighths block
   41   }
   42 
   43 -- | Resolve an optional config into a fully-populated one. 'Nothing' returns
   44 -- the default. A present-but-partial config fills missing fields from
   45 -- 'defaultStreamProgressConfig'. Pure.
   46 resolveStreamProgressConfig :: Maybe StreamProgressConfig -> StreamProgressConfig
   47 resolveStreamProgressConfig = fromMaybe defaultStreamProgressConfig
   48 
   49 -- | Should the manager send an edit now? Returns 'True' when enough time has
   50 -- passed since the last edit, or the buffer threshold is exceeded. Pure
   51 -- (takes the current time as an argument).
   52 shouldEdit :: StreamProgressConfig -> UTCTime -> Maybe UTCTime -> Int -> Bool
   53 shouldEdit cfg now mLastEdit accumLen =
   54   accumLen >= spcBufferThreshold cfg || timeElapsed
   55   where
   56     timeElapsed = case mLastEdit of
   57       Nothing -> True
   58       Just lastEdit ->
   59         diffUTCTime now lastEdit * 1000
   60           >= fromIntegral (spcEditIntervalMs cfg)
   61 
   62 -- | Append the cursor to the text for intermediate edits. Pure.
   63 addCursor :: StreamProgressConfig -> Text -> Text
   64 addCursor cfg text =
   65   if T.null (spcCursor cfg) then text else text <> spcCursor cfg
   66 
   67 -- | Strip the cursor from the text for the final edit. Pure.
   68 stripCursor :: StreamProgressConfig -> Text -> Text
   69 stripCursor cfg text =
   70   if T.null (spcCursor cfg)
   71     then text
   72     else fromMaybe text (T.stripSuffix (spcCursor cfg) text)

    2 -- | The WebSocket stream protocol types — the wire shapes for the WS event
    3 -- union and the client-to-server focus op. These are the types both the
    4 -- server (WS server in 'Seal.Gateway.Stream') and channel clients (WS
    5 -- client in 'Seal.Channels.WsClient') import.
    6 --
    7 -- 'BrokerEvent' is the in-process event the broker fans out. The WS server
    8 -- encodes it to wire JSON; the WS client decodes from wire JSON. Only the
    9 -- 'BrokerEvent' data type and 'FocusOp' live here — the 'Subscriber' and
   10 -- 'StreamBroker' runtime types stay in the server (they carry STM state).
   11 --
   12 -- Canonical home; 'Seal.Gateway.Stream' and 'Seal.Gateway.StreamBroker' in
   13 -- the server re-export 'FocusOp' and 'BrokerEvent' from here.

   16   , FocusOp (..)
   17   , RequestEntriesOp (..)
   18   , ClientMessage (..)
   19   , ServerEvent (..)
   20   , decodeServerEvent
   21   , StreamErrorCode (..)
   22   ) where

   35 -- | One event the broker fans out to subscribers. The 'Value' payloads are
   36 -- the pre-encoded JSON the WS peer receives — the server's 'sendEvent'
   37 -- wraps them in the wire envelope.
   38 data BrokerEvent
   39   = BeEntryRecorded SessionId Value   -- ^ a transcript entry (the JSON the WS peer receives)
   40   | BeEntryUpdate SessionId Value      -- ^ a streaming entry whose text is still growing (the WS peer renders as an @entry-update@ event, replacing the in-place entry by id)
   41   | BeHarnessStatus Value             -- ^ a harness liveness change
   42   | BeListsSnapshot Value             -- ^ a refreshed tab/session snapshot
   43   | BeAsk SessionId Value             -- ^ a pending human-question from ASK_HUMAN (the JSON the WS peer renders)
   44   | BeAskResolved SessionId Value      -- ^ a pending question was answered/cancelled (the JSON carries the ask id)
   45   | BeActivity SessionId Value          -- ^ a per-session activity signal (harness-status / reply-delivered) the WS peer renders as an @activity@ envelope
   46   | BeAgentDefsChanged                 -- ^ agent defs were created/updated/deleted; clients should re-fetch
   47   | BeSkillsChanged                    -- ^ skills were created/updated/deleted; clients should re-fetch
   48   | BeReposChanged                     -- ^ the source-control repo registry was mutated; clients should re-fetch /api/repos
   49   deriving stock (Eq, Show)

   51 -- | The focus op the client sends to change its focused session. Accepts
   52 -- both the frontend's shape (@{"op":"focus","sessionId":"..."}@) and the
   53 -- legacy shape (@{"session":"..."}@) for robustness. The optional @since@
   54 -- field requests replay of entries after the given entry id.
   55 data FocusOp = FocusOp
   56   { foSession :: Text
   57   , foSince   :: Maybe Text
   58   }
   59   deriving stock (Eq, Show)
   60 
   61 instance A.FromJSON FocusOp where
   62   parseJSON = A.withObject "focus" $ \o ->
   63     FocusOp
   64       <$> (o .: "sessionId" <|> o .: "session")
   65       <*> (o .:? "since")
   66 
   67 instance A.ToJSON FocusOp where
   68   toJSON (FocusOp sess mSince) = A.object
   69     ( [ "op" A..= ("focus" :: Text)
   70       , "sessionId" A..= sess
   71       ]
   72       <> maybe [] (\s -> ["since" A..= s]) mSince
   73     )

   75 -- | The @request-entries@ op the client sends to load a chunk of
   76 -- transcript entries (for chunked / infinite-scroll-upward loading).
   77 -- @before@ is the entry-id cursor (exclusive); when 'Nothing' the server
   78 -- returns the latest N entries. @limit@ is advisory — the server clamps
   79 -- it to [1, 200] with a default of 50.
   80 data RequestEntriesOp = RequestEntriesOp
   81   { reoSessionId :: Text
   82   , reoBefore    :: Maybe Text
   83   , reoLimit     :: Maybe Int
   84   } deriving stock (Eq, Show)
   85 
   86 instance A.FromJSON RequestEntriesOp where
   87   parseJSON = A.withObject "request-entries" $ \o ->
   88     RequestEntriesOp
   89       <$> o .: "sessionId"
   90       <*> o .:? "before"
   91       <*> o .:? "limit"
   92 
   93 -- | The discriminated union of client-to-server messages. Dispatches on
   94 -- the @op@ field: @"focus"@ → 'CmFocus', @"request-entries"@ →
   95 -- 'CmRequestEntries'. When @op@ is absent, falls back to 'CmFocus'
   96 -- (preserving the existing tolerance for the legacy @{"session":"..."}@
   97 -- shape).
   98 data ClientMessage = CmFocus FocusOp | CmRequestEntries RequestEntriesOp
   99   deriving stock (Eq, Show)
  100 
  101 instance A.FromJSON ClientMessage where
  102   parseJSON = A.withObject "ClientMessage" $ \o -> do
  103     mOp <- o .:? "op"
  104     case (mOp :: Maybe Text) of
  105       Just "focus"           -> CmFocus <$> A.parseJSON (A.Object o)
  106       Just "request-entries" -> CmRequestEntries <$> A.parseJSON (A.Object o)
  107       Nothing                -> CmFocus <$> A.parseJSON (A.Object o)
  108       _                      -> fail "unknown op"

  110 -- | Decode a WS wire frame (raw JSON bytes) into a 'ServerEvent'. Returns
  111 -- 'Nothing' for unparseable frames. Used by the chat-channel WS client's
  112 -- background reader. Pure.
  113 decodeServerEvent :: BL.ByteString -> Maybe ServerEvent
  114 decodeServerEvent bs = A.decode bs >>= parseServerEvent

  116 -- | The WS wire event union — the discriminated union the WS peer
  117 -- receives. Tagged by the @type@ field. This is the Haskell mirror of the
  118 -- frontend's @ServerEvent@ TypeScript type
  119 -- (@frontend/src/types/stream.ts@). The server encodes 'BrokerEvent's into
  120 -- these wire shapes; the channel WS client decodes them.
  121 --
  122 -- Currently only the 'FocusOp' and 'BrokerEvent' types are wired. The full
  123 -- 'ServerEvent' union will be fleshed out as the chat-channel WS client
  124 -- is implemented (Step 2). For now, this is the type declaration that
  125 -- establishes the contract.
  126 data ServerEvent
  127   = SeHello Text Text          -- ^ @hello@: protocolVersion + serverStartedAt
  128   | SeEntry SessionId Value    -- ^ @entry@: a complete transcript entry
  129   | SeEntryUpdate SessionId Value  -- ^ @entry-update@: a streaming entry update
  130   | SeActivity SessionId Value     -- ^ @activity@: a per-session activity signal
  131   | SeReplayEnd SessionId (Maybe Text)  -- ^ @replay-end@: replay finished, last replayed entry id
  132   | SeLists Value              -- ^ @lists@: a tab/session snapshot
  133   | SeAsk SessionId Value      -- ^ @ask@: a pending human question
  134   | SeAskResolved SessionId Value  -- ^ @ask_resolved@: a question was answered/cancelled
  135   | SeAgentDefsChanged         -- ^ @agent-defs-changed@
  136   | SeSkillsChanged            -- ^ @skills-changed@
  137   | SeReposChanged             -- ^ @repos-changed@
  138   | SeEntriesChunk SessionId Value  -- ^ @entries-chunk@: a chunk of transcript entries (whole object, matching 'SeLists')
  139   | SeError StreamErrorCode Text  -- ^ @error@: an error code + message
  140   deriving stock (Eq, Show)
  141 
  142 -- | The error codes the WS server can send in an @error@ event. Mirrors
  143 -- the frontend's @StreamErrorCode@ type.
  144 data StreamErrorCode
  145   = SecInvalidOp
  146   | SecInvalidFrame
  147   | SecSessionNotFound
  148   | SecFrameTooLarge
  149   | SecReplayFailed
  150   | SecReplayAborted
  151   | SecInternal
  152   deriving stock (Eq, Show)
  153 
  154 instance A.ToJSON StreamErrorCode where
  155   toJSON = \case
  156     SecInvalidOp      -> A.String "invalid-op"
  157     SecInvalidFrame   -> A.String "invalid-frame"
  158     SecSessionNotFound -> A.String "session-not-found"
  159     SecFrameTooLarge  -> A.String "frame-too-large"
  160     SecReplayFailed   -> A.String "replay-failed"
  161     SecReplayAborted  -> A.String "replay-aborted"
  162     SecInternal       -> A.String "internal"
  163 
  164 instance A.FromJSON StreamErrorCode where
  165   parseJSON = A.withText "StreamErrorCode" $ \case
  166     "invalid-op"        -> pure SecInvalidOp
  167     "invalid-frame"     -> pure SecInvalidFrame
  168     "session-not-found" -> pure SecSessionNotFound
  169     "frame-too-large"   -> pure SecFrameTooLarge
  170     "replay-failed"     -> pure SecReplayFailed
  171     "replay-aborted"    -> pure SecReplayAborted
  172     "internal"          -> pure SecInternal
  173     other               -> fail ("unknown stream error code: " <> show other)

  175 -- | Parse a raw JSON 'Value' into a 'ServerEvent' by dispatching on the
  176 -- @type@ field. Returns 'Nothing' for unknown types or missing fields.
  177 -- Pure. Used by 'decodeServerEvent'.
  178 parseServerEvent :: Value -> Maybe ServerEvent
  179 parseServerEvent (A.Object o) =
  180   case KeyMap.lookup (Key.fromText "type") o of
  181     Just (A.String t) -> case t of
  182       "hello" -> SeHello
  183         <$> (asText =<< KeyMap.lookup (Key.fromText "protocolVersion") o)
  184         <*> (asText =<< KeyMap.lookup (Key.fromText "serverStartedAt") o)
  185       "entry" -> SeEntry
  186         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  187         <*> pure (fromMaybe A.Null (KeyMap.lookup (Key.fromText "entry") o))
  188       "entry-update" -> SeEntryUpdate
  189         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  190         <*> pure (fromMaybe A.Null (KeyMap.lookup (Key.fromText "entry") o))
  191       "activity" -> SeActivity
  192         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  193         <*> pure (fromMaybe A.Null (KeyMap.lookup (Key.fromText "activity") o))
  194       "replay-end" -> SeReplayEnd
  195         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  196         <*> pure (asText =<< KeyMap.lookup (Key.fromText "lastEntryId") o)
  197       "lists" -> Just (SeLists (A.Object o))
  198       "ask" -> SeAsk
  199         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  200         <*> pure (fromMaybe A.Null (KeyMap.lookup (Key.fromText "ask") o))
  201       "ask_resolved" -> SeAskResolved
  202         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  203         <*> pure (fromMaybe A.Null (KeyMap.lookup (Key.fromText "ask") o))
  204       "agent-defs-changed" -> Just SeAgentDefsChanged
  205       "skills-changed" -> Just SeSkillsChanged
  206       "repos-changed" -> Just SeReposChanged
  207       "entries-chunk" -> SeEntriesChunk
  208         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  209         <*> pure (A.Object o)  -- whole object, matching SeLists pattern
  210       "error" -> SeError
  211         <$> (asErrCode =<< KeyMap.lookup (Key.fromText "code") o)
  212         <*> pure (fromMaybe "" (asText =<< KeyMap.lookup (Key.fromText "message") o))
  213       _ -> Nothing
  214     _ -> Nothing
  215 parseServerEvent _ = Nothing
  216 
  217 asText :: Value -> Maybe Text
  218 asText (A.String t) = Just t
  219 asText _ = Nothing
  220 
  221 asSessionId :: Value -> Maybe SessionId
  222 asSessionId (A.String t) =
  223   case mkSessionId t of Right s -> Just s; Left _ -> Nothing
  224 asSessionId _ = Nothing
  225 
  226 asErrCode :: Value -> Maybe StreamErrorCode
  227 asErrCode v = case A.fromJSON v of
  228   A.Success c -> Just c
  229   A.Error _   -> Nothing

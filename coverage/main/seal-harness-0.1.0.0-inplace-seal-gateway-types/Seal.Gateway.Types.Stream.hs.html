    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The WebSocket stream protocol types — the wire shapes for the WS event
    3 -- union and the client-to-server focus op. These are the types both the
    4 -- server (WS server in 'Seal.Gateway.Stream') and channel clients (WS
    5 -- client in 'Seal.Channels.WsClient') import.
    6 --
    7 -- 'BrokerEvent' is the in-process event the broker fans out. The WS server
    8 -- encodes it to wire JSON; the WS client decodes from wire JSON. Only the
    9 -- 'BrokerEvent' data type and 'FocusOp' live here — the 'Subscriber' and
   10 -- 'StreamBroker' runtime types stay in the server (they carry STM state).
   11 --
   12 -- Canonical home; 'Seal.Gateway.Stream' and 'Seal.Gateway.StreamBroker' in
   13 -- the server re-export 'FocusOp' and 'BrokerEvent' from here.
   14 module Seal.Gateway.Types.Stream
   15   ( BrokerEvent (..)
   16   , FocusOp (..)
   17   , ServerEvent (..)
   18   , decodeServerEvent
   19   , StreamErrorCode (..)
   20   ) where

   33 -- | One event the broker fans out to subscribers. The 'Value' payloads are
   34 -- the pre-encoded JSON the WS peer receives — the server's 'sendEvent'
   35 -- wraps them in the wire envelope.
   36 data BrokerEvent
   37   = BeEntryRecorded SessionId Value   -- ^ a transcript entry (the JSON the WS peer receives)
   38   | BeEntryUpdate SessionId Value      -- ^ a streaming entry whose text is still growing (the WS peer renders as an @entry-update@ event, replacing the in-place entry by id)
   39   | BeHarnessStatus Value             -- ^ a harness liveness change
   40   | BeListsSnapshot Value             -- ^ a refreshed tab/session snapshot
   41   | BeAsk SessionId Value             -- ^ a pending human-question from ASK_HUMAN (the JSON the WS peer renders)
   42   | BeAskResolved SessionId Value      -- ^ a pending question was answered/cancelled (the JSON carries the ask id)
   43   | BeActivity SessionId Value          -- ^ a per-session activity signal (harness-status / reply-delivered) the WS peer renders as an @activity@ envelope
   44   | BeAgentDefsChanged                 -- ^ agent defs were created/updated/deleted; clients should re-fetch
   45   | BeSkillsChanged                    -- ^ skills were created/updated/deleted; clients should re-fetch
   46   | BeReposChanged                     -- ^ the source-control repo registry was mutated; clients should re-fetch /api/repos
   47   deriving stock (Eq, Show)

   49 -- | The focus op the client sends to change its focused session. Accepts
   50 -- both the frontend's shape (@{"op":"focus","sessionId":"..."}@) and the
   51 -- legacy shape (@{"session":"..."}@) for robustness. The optional @since@
   52 -- field requests replay of entries after the given entry id.
   53 data FocusOp = FocusOp
   54   { foSession :: Text
   55   , foSince   :: Maybe Text
   56   }
   57   deriving stock (Eq, Show)
   58 
   59 instance A.FromJSON FocusOp where
   60   parseJSON = A.withObject "focus" $ \o ->
   61     FocusOp
   62       <$> (o .: "sessionId" <|> o .: "session")
   63       <*> (o .:? "since")
   64 
   65 instance A.ToJSON FocusOp where
   66   toJSON (FocusOp sess mSince) = A.object
   67     ( [ "op" A..= ("focus" :: Text)
   68       , "sessionId" A..= sess
   69       ]
   70       <> maybe [] (\s -> ["since" A..= s]) mSince
   71     )

   73 -- | Decode a WS wire frame (raw JSON bytes) into a 'ServerEvent'. Returns
   74 -- 'Nothing' for unparseable frames. Used by the chat-channel WS client's
   75 -- background reader. Pure.
   76 decodeServerEvent :: BL.ByteString -> Maybe ServerEvent
   77 decodeServerEvent bs = A.decode bs >>= parseServerEvent

   79 -- | The WS wire event union — the discriminated union the WS peer
   80 -- receives. Tagged by the @type@ field. This is the Haskell mirror of the
   81 -- frontend's @ServerEvent@ TypeScript type
   82 -- (@frontend/src/types/stream.ts@). The server encodes 'BrokerEvent's into
   83 -- these wire shapes; the channel WS client decodes them.
   84 --
   85 -- Currently only the 'FocusOp' and 'BrokerEvent' types are wired. The full
   86 -- 'ServerEvent' union will be fleshed out as the chat-channel WS client
   87 -- is implemented (Step 2). For now, this is the type declaration that
   88 -- establishes the contract.
   89 data ServerEvent
   90   = SeHello Text Text          -- ^ @hello@: protocolVersion + serverStartedAt
   91   | SeEntry SessionId Value    -- ^ @entry@: a complete transcript entry
   92   | SeEntryUpdate SessionId Value  -- ^ @entry-update@: a streaming entry update
   93   | SeActivity SessionId Value     -- ^ @activity@: a per-session activity signal
   94   | SeReplayEnd SessionId (Maybe Text)  -- ^ @replay-end@: replay finished, last replayed entry id
   95   | SeLists Value              -- ^ @lists@: a tab/session snapshot
   96   | SeAsk SessionId Value      -- ^ @ask@: a pending human question
   97   | SeAskResolved SessionId Value  -- ^ @ask_resolved@: a question was answered/cancelled
   98   | SeAgentDefsChanged         -- ^ @agent-defs-changed@
   99   | SeSkillsChanged            -- ^ @skills-changed@
  100   | SeReposChanged             -- ^ @repos-changed@
  101   | SeError StreamErrorCode Text  -- ^ @error@: an error code + message
  102   deriving stock (Eq, Show)
  103 
  104 -- | The error codes the WS server can send in an @error@ event. Mirrors
  105 -- the frontend's @StreamErrorCode@ type.
  106 data StreamErrorCode
  107   = SecInvalidOp
  108   | SecInvalidFrame
  109   | SecSessionNotFound
  110   | SecFrameTooLarge
  111   | SecReplayFailed
  112   | SecReplayAborted
  113   | SecInternal
  114   deriving stock (Eq, Show)
  115 
  116 instance A.ToJSON StreamErrorCode where
  117   toJSON = \case
  118     SecInvalidOp      -> A.String "invalid-op"
  119     SecInvalidFrame   -> A.String "invalid-frame"
  120     SecSessionNotFound -> A.String "session-not-found"
  121     SecFrameTooLarge  -> A.String "frame-too-large"
  122     SecReplayFailed   -> A.String "replay-failed"
  123     SecReplayAborted  -> A.String "replay-aborted"
  124     SecInternal       -> A.String "internal"
  125 
  126 instance A.FromJSON StreamErrorCode where
  127   parseJSON = A.withText "StreamErrorCode" $ \case
  128     "invalid-op"        -> pure SecInvalidOp
  129     "invalid-frame"     -> pure SecInvalidFrame
  130     "session-not-found" -> pure SecSessionNotFound
  131     "frame-too-large"   -> pure SecFrameTooLarge
  132     "replay-failed"     -> pure SecReplayFailed
  133     "replay-aborted"    -> pure SecReplayAborted
  134     "internal"          -> pure SecInternal
  135     other               -> fail ("unknown stream error code: " <> show other)
  136 
  137 -- | Parse a raw JSON 'Value' into a 'ServerEvent' by dispatching on the
  138 -- @type@ field. Returns 'Nothing' for unknown types or missing fields.
  139 -- Pure. Used by 'decodeServerEvent'.
  140 parseServerEvent :: Value -> Maybe ServerEvent
  141 parseServerEvent (A.Object o) =
  142   case KeyMap.lookup (Key.fromText "type") o of
  143     Just (A.String t) -> case t of
  144       "hello" -> SeHello
  145         <$> (asText =<< KeyMap.lookup (Key.fromText "protocolVersion") o)
  146         <*> (asText =<< KeyMap.lookup (Key.fromText "serverStartedAt") o)
  147       "entry" -> SeEntry
  148         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  149         <*> pure (fromMaybe A.Null (KeyMap.lookup (Key.fromText "entry") o))
  150       "entry-update" -> SeEntryUpdate
  151         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  152         <*> pure (fromMaybe A.Null (KeyMap.lookup (Key.fromText "entry") o))
  153       "activity" -> SeActivity
  154         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  155         <*> pure (fromMaybe A.Null (KeyMap.lookup (Key.fromText "activity") o))
  156       "replay-end" -> SeReplayEnd
  157         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  158         <*> pure (asText =<< KeyMap.lookup (Key.fromText "lastEntryId") o)
  159       "lists" -> Just (SeLists (A.Object o))
  160       "ask" -> SeAsk
  161         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  162         <*> pure (fromMaybe A.Null (KeyMap.lookup (Key.fromText "ask") o))
  163       "ask_resolved" -> SeAskResolved
  164         <$> (asSessionId =<< KeyMap.lookup (Key.fromText "sessionId") o)
  165         <*> pure (fromMaybe A.Null (KeyMap.lookup (Key.fromText "ask") o))
  166       "agent-defs-changed" -> Just SeAgentDefsChanged
  167       "skills-changed" -> Just SeSkillsChanged
  168       "repos-changed" -> Just SeReposChanged
  169       "error" -> SeError
  170         <$> (asErrCode =<< KeyMap.lookup (Key.fromText "code") o)
  171         <*> pure (fromMaybe "" (asText =<< KeyMap.lookup (Key.fromText "message") o))
  172       _ -> Nothing
  173     _ -> Nothing
  174 parseServerEvent _ = Nothing
  175 
  176 asText :: Value -> Maybe Text
  177 asText (A.String t) = Just t
  178 asText _ = Nothing
  179 
  180 asSessionId :: Value -> Maybe SessionId
  181 asSessionId (A.String t) =
  182   case mkSessionId t of Right s -> Just s; Left _ -> Nothing
  183 asSessionId _ = Nothing
  184 
  185 asErrCode :: Value -> Maybe StreamErrorCode
  186 asErrCode v = case A.fromJSON v of
  187   A.Success c -> Just c
  188   A.Error _   -> Nothing

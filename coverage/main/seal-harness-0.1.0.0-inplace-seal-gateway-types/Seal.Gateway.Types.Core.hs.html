   51 -- | Opaque session label. No parse invariant on construction history, but a
   52 -- single strict predicate guards every path-join / network boundary.
   53 newtype SessionId = SessionId Text
   54   deriving stock (Eq, Ord, Show)
   55   deriving newtype (ToJSON, FromJSON)
   56 
   57 isValidSessionId :: Text -> Bool
   58 isValidSessionId t =
   59   not (T.null t)
   60     && T.head t /= '.'
   61     && T.all (`elem` chars) t
   62   where
   63     chars = ['A' .. 'Z'] <> ['a' .. 'z'] <> ['0' .. '9'] <> "_-"
   64 
   65 mkSessionId :: Text -> Either Text SessionId
   66 mkSessionId t
   67   | isValidSessionId t = Right (SessionId t)
   68   | otherwise          = Left ("invalid session id: " <> T.pack (show t))
   69 
   70 -- | Total constructor for known-safe system strings (e.g. @"web"@,
   71 -- @"manual"@). Calls 'isValidSessionId' internally and errors on
   72 -- failure — it does NOT bypass validation. Use only for compile-time-
   73 -- known literals; runtime-derived strings should use 'mkSessionId' (the
   74 -- 'Either'-returning variant).
   75 mkSystemSessionId :: Text -> SessionId
   76 mkSystemSessionId t
   77   | isValidSessionId t = SessionId t
   78   | otherwise          = error ("mkSystemSessionId: invalid session id: " <> T.unpack t)

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Pure, channel-agnostic pieces of the @\/new@ command, shared between the
    3 -- main library (@Seal.Command.New@) and the gateway-API chat channels
    4 -- (@seal-chat-channels@), which cannot import the main library.
    5 --
    6 -- Two things live here:
    7 --
    8 -- * 'NewArgs' and its parser — the @-p@\/@-m@\/@-r@ flags of @\/new@. The
    9 --   chat-channel loop parses the same flags as the CLI/web registry, so the
   10 --   parser must be shared rather than re-implemented (handoff §4 item 9:
   11 --   "reuse it, do not re-parse by hand").
   12 -- * 'renderNewSessionConfirmation' — the one-line confirmation naming the new
   13 --   session, the old session, and the resume hint. Extracted so the chat loop
   14 --   and 'Seal.Command.New.renderNewConfirmation' emit byte-identical text.
   15 module Seal.Gateway.Types.NewArgs
   16   ( NewArgs (..)
   17   , parseNewArgs
   18   , emptyNewArgs
   19   , renderNewSessionConfirmation
   20   ) where
   21 
   22 import Data.Text (Text)
   23 import Data.Text qualified as T
   24 
   25 -- | Optional arguments for @/new@. All fields are 'Maybe' — 'Nothing' means
   26 -- "use the config default" (for provider/model) or "no repo" (for repo).
   27 data NewArgs = NewArgs
   28   { naProvider :: Maybe Text
   29   , naModel    :: Maybe Text
   30   , naRepo     :: Maybe Text
   31   } deriving stock (Eq, Show)
   32 
   33 -- | The empty 'NewArgs' — no overrides, use all config defaults.
   34 emptyNewArgs :: NewArgs
   35 emptyNewArgs = NewArgs { naProvider = Nothing, naModel = Nothing, naRepo = Nothing }
   36 
   37 -- | Parse the raw argument text from the routing layer into 'NewArgs'.
   38 -- This is a simple manual parse used by the inbox-channel loop path
   39 -- ('Seal.Channels.Chat.Loop'), which doesn't go through
   40 -- optparse-applicative. Supports @-p@\/@--provider@, @-m@\/@--model@, and
   41 -- @-r@\/@--repo@ with a single value each. Unknown flags are ignored.
   42 parseNewArgs :: Text -> NewArgs
   43 parseNewArgs raw =
   44   go (splitArgs raw) emptyNewArgs
   45   where
   46     go [] acc = acc
   47     go (flag : val : rest) acc
   48       | flag `elem` ["-p", "--provider"] = go rest acc { naProvider = Just val }
   49       | flag `elem` ["-m", "--model"]    = go rest acc { naModel    = Just val }
   50       | flag `elem` ["-r", "--repo"]     = go rest acc { naRepo     = Just val }
   51       | otherwise                        = go (val : rest) acc
   52     go (_flag : rest) acc = go rest acc
   53 
   54     -- Split on whitespace, preserving non-empty tokens.
   55     splitArgs t = case T.words t of
   56       [] -> []
   57       ws -> ws
   58 
   59 -- | Render the one-line @/new@ confirmation. Names the old session + resume
   60 -- hint so the safety net (old session kept on disk) is user-visible.
   61 --
   62 -- @new session \<sid\> (\<provider\>/\<model\>) — prior session \<oldSid\> kept in
   63 -- /session list@
   64 renderNewSessionConfirmation :: Text -> Text -> Text -> Text -> Text
   65 renderNewSessionConfirmation newSid provider model oldSid =
   66   "new session " <> newSid
   67     <> " (" <> provider <> "/" <> model <> ")"
   68     <> " — prior session " <> oldSid
   69     <> " kept in /session list"

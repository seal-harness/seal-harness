   49 -- ---------------------------------------------------------------------------
   50 -- Minimal inline of the bits of @Text.Casing@ the @lensy*Snake@/@lens*Kebab@
   51 -- helpers need. Vendored here so this module stays dependency-free (the
   52 -- @text-casing@ Hackage package isn't in the pinned haskell.nix index).
   53 -- @fromHumps@ splits @FooBarBaz@ / @fooBarBaz@ into @[Foo, Bar, Baz]@ /
   54 -- @[foo, Bar, Baz]@; @toQuietSnake@ joins with @_@ and lowercases;
   55 -- @toKebab@ joins with @-@ and lowercases.
   56 -- ---------------------------------------------------------------------------
   57 
   58 fromHumps :: String -> [String]
   59 fromHumps = foldr go []
   60   where
   61     go c []          = [[c]]
   62     go c (w:ws)
   63       | isUpper c    = [c] : w : ws
   64       | otherwise    = (c:w) : ws
   65 
   66 toQuietSnake :: [String] -> String
   67 toQuietSnake = concatWith '_' . map (map toLower)
   68   where concatWith _ []       = ""
   69         concatWith sep (x:xs) = x <> foldr (\a acc -> sep : a <> acc) "" xs
   70 
   71 toKebab :: [String] -> String
   72 toKebab = concatWith '-' . map (map toLower)
   73   where concatWith _ []       = ""
   74         concatWith sep (x:xs) = x <> foldr (\a acc -> sep : a <> acc) "" xs

   76 lensyLenToJSON
   77   :: (Generic a, GToJSON Zero (Rep a)) => Int -> a -> Value
   78 lensyLenToJSON n = genericToJSON (lensyLenOptions n)
   79 
   80 lensyLenParseJSON
   81   :: (Generic a, GFromJSON Zero (Rep a)) => Int -> Value -> Parser a
   82 lensyLenParseJSON n = genericParseJSON (lensyLenOptions n)

   84 lensyLenOptions :: Int -> Options
   85 lensyLenOptions n = defaultOptions { fieldLabelModifier = lensyLenConstructorToNiceJson n }
   86 
   87 lensyLenConstructorToNiceJson :: Int -> String -> String
   88 lensyLenConstructorToNiceJson n fieldName = firstToLower $ drop n fieldName
   89   where
   90     firstToLower (c:cs) = toLower c : cs
   91     firstToLower _ = error $ "lensyLenConstructorToNiceJson: bad arguments: " <> show (n,fieldName)

  118 lensyFieldNameToNiceJson :: String -> String
  119 lensyFieldNameToNiceJson fieldName = dropWhile (=='_') $ dropWhile (/='_') $ dropWhile (=='_') fieldName

  121 lensyFieldNameToSnakeJson :: String -> String
  122 lensyFieldNameToSnakeJson = toQuietSnake . fromHumps . dropWhile (=='_') . dropWhile (/='_') . dropWhile (=='_')

  146 -- | Generic 'ToJSON' using 'stripPrefixOptions' — strips the leading run
  147 -- of lowercase letters (the lens prefix) from each record field and
  148 -- lowercases the first remaining char, keeping camelCase for the rest.
  149 --   @teId → id@, @crMaxTokens → maxTokens@, @cbInputSchema → inputSchema@,
  150 --   @msgRole → role@, @uInput → input@.
  151 stripPrefixToJSON
  152   :: (Generic a, GToJSON Zero (Rep a)) => a -> Value
  153 stripPrefixToJSON = genericToJSON stripPrefixOptions
  154 
  155 -- | Generic 'FromJSON' using 'stripPrefixOptions'. See 'stripPrefixToJSON'.
  156 stripPrefixParseJSON
  157   :: (Generic a, GFromJSON Zero (Rep a)) => Value -> Parser a
  158 stripPrefixParseJSON = genericParseJSON stripPrefixOptions
  159 
  160 -- | 'Options' that strip the leading lowercase lens prefix from record
  161 -- field names (camelCase preserved). Constructor tags are left as-is
  162 -- (Seal's enums encode as bare tags like @"User"@ / @"Request"@).
  163 stripPrefixOptions :: Options
  164 stripPrefixOptions = defaultOptions { fieldLabelModifier = stripLensPrefixCamel }
  165 
  166 -- | Drop the leading run of lowercase letters, then lowercase the first
  167 -- remaining character. Examples:
  168 --
  169 --   * @teId       → id@
  170 --   * @teTimestamp → timestamp@
  171 --   * @crMaxTokens → maxTokens@
  172 --   * @cbInputSchema → inputSchema@
  173 --   * @cbForId    → forId@
  174 --   * @msgRole    → role@
  175 --   * @uInput     → input@
  176 --   * @tdName     → name@
  177 --   * @rsContent  → content@
  178 --
  179 -- Field names with no leading lowercase run (e.g. already-@lowerCamel@
  180 -- keys produced by hand-written instances) are returned unchanged except
  181 -- for the (no-op) first-char lowercasing.
  182 stripLensPrefixCamel :: String -> String
  183 stripLensPrefixCamel fieldName =
  184   case dropWhile isLower fieldName of
  185     ""           -> fieldName
  186     firstUpper:rest -> toLower firstUpper : rest

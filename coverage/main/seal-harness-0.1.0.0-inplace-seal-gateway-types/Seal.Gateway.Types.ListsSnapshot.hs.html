    1 {-# LANGUAGE DeriveGeneric #-}
    2 {-# LANGUAGE OverloadedStrings #-}
    3 -- | The wire snapshot type for the WS @lists@ frame and the REST @GET
    4 -- /api/lists@ endpoint. Carries the partitioned session lists plus the set
    5 -- of sessions currently in a @thinking@ turn.
    6 --
    7 -- Only the data type + 'ToJSON' instance live here. The builder
    8 -- ('buildListsSnapshot') stays in the server ('Seal.Gateway.ListsSnapshot')
    9 -- because it depends on 'TabsHandle', 'SealPaths', and 'SessionStore' —
   10 -- server-internal types.
   11 --
   12 -- Canonical home; 'Seal.Gateway.ListsSnapshot' in the server re-exports the
   13 -- type from here.
   14 module Seal.Gateway.Types.ListsSnapshot
   15   ( ListsSnapshotWire (..)
   16   ) where
   17 
   18 import Data.Aeson (ToJSON (..), object, (.=))
   19 import Data.Aeson qualified as A
   20 import Data.Text (Text)
   21 import GHC.Generics (Generic)
   22 
   23 -- | The partitioned snapshot. Haskell record fields use the @lsw@ prefix;
   24 -- the 'ToJSON' instance drops it (wire keys: @tabs@, @recentSessions@,
   25 -- @archivedSessions@, @tabSessions@, @thinkingSessionIds@).
   26 data ListsSnapshotWire = ListsSnapshotWire
   27   { lswTabs             :: [A.Value]
   28   , lswRecentSessions   :: [A.Value]
   29   , lswArchivedSessions :: [A.Value]
   30   , lswTabSessions      :: [A.Value]
   31   , lswThinkingSessionIds :: [Text]
   32   } deriving stock (Eq, Show, Generic)
   33 
   34 instance ToJSON ListsSnapshotWire where
   35   toJSON s = object
   36     [ "tabs"             .= lswTabs s
   37     , "recentSessions"   .= lswRecentSessions s
   38     , "archivedSessions" .= lswArchivedSessions s
   39     , "tabSessions"      .= lswTabSessions s
   40     , "thinkingSessionIds" .= lswThinkingSessionIds s
   41     ]

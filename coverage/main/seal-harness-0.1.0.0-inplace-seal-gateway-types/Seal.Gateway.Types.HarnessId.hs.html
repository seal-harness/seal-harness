    1 {-# LANGUAGE DeriveGeneric #-}
    2 {-# LANGUAGE OverloadedStrings #-}
    3 -- | A UUID-backed durable harness identity (the registry key). Minted at
    4 -- spawn time; stamped as a tmux @seal_id marker on the harness window so a
    5 -- window can be re-identified after a rename or reconnect. The UUID is
    6 -- generated in-repo from 'System.Random' (the repo has @uuid-types@ but
    7 -- not the full @uuid@ package, so v4 generation is hand-rolled from two
    8 -- random 'Word64's with the v4 version/variant bits set).
    9 --
   10 -- Canonical home for the gateway API contract; 'Seal.Harness.Id' in the
   11 -- server re-exports from here. The 'newHarnessId' IO action stays in the
   12 -- server (it requires 'System.Random'); only the type + pure functions
   13 -- live here.
   14 module Seal.Gateway.Types.HarnessId
   15   ( HarnessId (..)
   16   , parseHarnessId
   17   , harnessIdToText
   18   , isValidHarnessIdText
   19   ) where
   20 
   21 import Data.Aeson (FromJSON, ToJSON)
   22 import Data.Text (Text)
   23 import Data.Text qualified as T
   24 import Data.UUID.Types qualified as U
   25 import GHC.Generics (Generic)

   27 -- | A UUID-backed durable harness identity.
   28 newtype HarnessId = HarnessId Text
   29   deriving stock (Eq, Ord, Show, Generic)
   30   deriving newtype (ToJSON, FromJSON)
   31 
   32 -- | The text form (a UUID string) — the value stamped as the @seal_id marker.
   33 harnessIdToText :: HarnessId -> Text
   34 harnessIdToText (HarnessId t) = t
   35 
   36 -- | Parse a HarnessId from its text form. 'Left' on malformed input.
   37 parseHarnessId :: Text -> Either Text HarnessId
   38 parseHarnessId t
   39   | isValidHarnessIdText t = Right (HarnessId t)
   40   | otherwise              = Left ("invalid HarnessId: " <> t)
   41 
   42 -- | True if the text is a valid UUID (8-4-4-4-12 hex, case-insensitive).
   43 -- Used to defend the @seal_id marker stamp against a malformed id.
   44 isValidHarnessIdText :: Text -> Bool
   45 isValidHarnessIdText t =
   46   case U.fromString (T.unpack t) of
   47     Just _  -> True
   48     Nothing -> False

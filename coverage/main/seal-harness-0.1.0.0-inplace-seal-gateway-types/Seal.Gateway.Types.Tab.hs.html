    4 -- | The validated 'TabIndex' (0..35) — the single index type reused
    5 -- everywhere (TabList slots, /N routing, /tab close <N>, /tab focus <N>).
    6 -- Smart-constructed so an out-of-range index fails to compile into any path.
    7 -- Plus 'TabKind', the closed enumeration of tab kinds.
    8 --
    9 -- Canonical home for the gateway API contract; 'Seal.Handles.Tab' in the
   10 -- server re-exports from here.
   11 module Seal.Gateway.Types.Tab
   12   ( TabIndex (..)
   13   , mkTabIndex
   14   , tabIndexToInt
   15   , tabIndexToChar
   16   , tabIndexFromChar
   17   , maxTabIndex
   18   , TabKind (..)
   19   ) where
   20 
   21 import Data.Aeson (FromJSON, ToJSON)
   22 import Data.Char (isAsciiLower, isDigit, toLower)
   23 import Data.Text (Text)
   24 import Data.Text qualified as T
   25 import GHC.Generics (Generic)
   26 
   27 -- | The maximum tab index (0-based): 35 = 'z'. 36 slots total.
   28 maxTabIndex :: Int
   29 maxTabIndex = 35
   30 
   31 -- | A validated tab index: 0..35 (the terse grammar maps 0-9a-z to 0..35).
   32 -- Smart-constructed; the predicate rejects <0 and >35.
   33 newtype TabIndex = TabIndex Int
   34   deriving stock (Eq, Ord, Show, Generic)
   35   deriving anyclass (ToJSON, FromJSON)
   36 
   37 mkTabIndex :: Int -> Either Text TabIndex
   38 mkTabIndex n
   39   | n < 0           = Left ("tab index out of range: " <> T.pack (show n))
   40   | n > maxTabIndex = Left ("tab index out of range: " <> T.pack (show n))
   41   | otherwise       = Right (TabIndex n)
   42 
   43 tabIndexToInt :: TabIndex -> Int
   44 tabIndexToInt (TabIndex n) = n
   45 
   46 -- | 0->'0', 9->'9', 10->'a', 35->'z'.
   47 tabIndexToChar :: TabIndex -> Char
   48 tabIndexToChar (TabIndex n)
   49   | n < 10   = toEnum (fromEnum '0' + n)
   50   | otherwise = toEnum (fromEnum 'a' + (n - 10))
   51 
   52 -- | Inverse of 'tabIndexToChar', case-insensitive. 'Left' for non-[0-9a-z].
   53 tabIndexFromChar :: Char -> Either Text TabIndex
   54 tabIndexFromChar c =
   55   let lo = toLower c
   56   in case lo of
   57        _ | isDigit lo      -> mkTabIndex (fromEnum lo - fromEnum '0')
   58          | isAsciiLower lo -> mkTabIndex (fromEnum lo - fromEnum 'a' + 10)
   59          | otherwise       -> Left ("not a tab index char: " <> T.pack [c])
   60 
   61 -- | The closed enumeration of tab kinds.
   62 data TabKind
   63   = KindAi | KindProvider | KindHarness | KindShell | KindSsh | KindTmux
   64   deriving stock (Eq, Show, Enum, Bounded, Generic)
   65   deriving anyclass (ToJSON, FromJSON)

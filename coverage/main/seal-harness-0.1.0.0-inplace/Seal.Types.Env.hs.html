   12 -- | The runtime environment, built from the resolved 'Config'. Holds resolved
   13 -- config values; a mutable resource (counter 'IORef', HTTP 'Manager', DB pool,
   14 -- etc.) would live here as well.
   15 data Env = Env
   16   { envLogLevel :: !Text
   17   , envServerHost :: !Text
   18   , envServerPort :: !Int
   19   , envLogger :: !SealLogger
   20   -- , envHttpManager :: !Manager
   21   }
   22 
   23 -- | Build the runtime 'Env' from the resolved configuration and the shared
   24 -- 'SealLogger' (built once at startup via 'withSealLogger').
   25 mkEnv :: SealLogger -> Config -> IO Env
   26 mkEnv logger cfg = pure Env
   27   { envLogLevel = view config_logLevel cfg
   28   , envServerHost = view (config_server . serverConfig_host) cfg
   29   , envServerPort = view (config_server . serverConfig_port) cfg
   30   , envLogger = logger
   31   }

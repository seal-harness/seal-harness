  138 -- | Full @seal serve@ startup wiring. Mirrors 'Seal.Tui.runTui': paths →
  139 -- config → vault → session → backends → tabsH → broker → gateway + WS
  140 -- server.
  141 --
  142 -- Ctrl-C is handled here: SIGINT reaches the RTS as a 'UserInterrupt' on the
  143 -- main thread, which 'handleServeInterrupt' turns into a shutdown log line
  144 -- naming the live sub-agents (whose forked threads die with this process)
  145 -- followed by exit 130.
  146 runServeMain :: AutonomyLevel -> Maybe FilePath -> Bool -> SealLogger -> IO ()
  147 runServeMain autonomy mCliStaticDir noEnrollToken logger = do
  148   -- Filled in once the backends exist; 'Nothing' means an interrupt landed
  149   -- during startup, before there was a runtime to report on.
  150   runtimeRef <- newIORef Nothing
  151   runServeMainInner autonomy mCliStaticDir noEnrollToken logger runtimeRef
  152     `catch` handleServeInterrupt logger runtimeRef
  153 
  154 runServeMainInner
  155   :: AutonomyLevel -> Maybe FilePath -> Bool -> SealLogger -> IORef (Maybe AgentRuntime) -> IO ()
  156 runServeMainInner autonomy mCliStaticDir noEnrollToken logger runtimeRef = do
  157   paths <- getSealPaths
  158   ensureSealDirs paths
  159   -- Claim this SEAL_HOME for the life of the process before anything else
  160   -- starts listening. The lock is advisory and kernel-held, so a previous
  161   -- daemon that crashed does not block this one (design §12).
  162   _serveLock <- claimSealHome paths
  163   -- Probe + sweep the persistent ssh-agent registry: GC dead agents from
  164   -- a crashed/killed seal process so their stale entries don't accumulate
  165   -- (#88). Live agents are reused (the key is still loaded).
  166   -- This handle is THE shared handle for the entire process — threaded
  167   -- through ChannelDeps, SendDeps, ApiDeps, and mkRepoTestSeam so all
  168   -- git-op call sites share the same arhLive/arhLock/arhCache.
  169   agentRegH <- mkAgentRegistryHandle (sshAgentsDir paths)
  170   arProbeAndSweep agentRegH
  171   -- GC the content-addressed agent-metadata snapshot cache: keep the
  172   -- newest N entries, drop the rest. Best-effort — failures are ignored.
  173   gcAgentMetaCache (agentMetaCacheDir paths) agentMetaCacheKeepN
  174   migrateSecurityConfig paths
  175   -- First-run scaffolding, after the migration so the security template cannot
  176   -- pre-empt it (see 'Seal.Config.Templates.seedSecurity'). Never overwrites and
  177   -- never fatal: a missing template must not keep the harness from booting.
  178   (createdTemplates, templateWarnings) <- ensureConfigTemplates paths
  179   forM_ templateWarnings (logIO logger WarningS . ls)
  180   forM_ createdTemplates $ \fp ->
  181     logIO logger InfoS ("wrote a documented config template at " <> ls (T.pack fp))
  182   let cfgPath = configFilePath paths
  183   cfg <- loadRuntimeConfig cfgPath >>= \case
  184     Left err -> do
  185       logIO logger WarningS ("could not load config: " <> ls err)
  186       pure defaultRuntimeConfig
  187     Right c  -> pure c
  188   secCfg <- loadSecurityConfig (securityFilePath paths) >>= \case
  189     Left err -> do
  190       -- Fail closed. Degrading an unparseable security.toml to the defaults
  191       -- would silently reopen the access-control enrollment window on every
  192       -- boot, so the gateway refuses to start and prints the diagnostics.
  193       logIO logger ErrorS ("could not load security config: " <> ls err)
  194       ioError (userError ("security.toml could not be parsed; refusing to start. "
  195                           <> T.unpack err))
  196     Right c  -> pure c
  197   -- Access control is resolved ONCE here, before any channel is forked or any
  198   -- port is bound, so the refuse-to-start matrix (design §10) runs first.
  199   let gwCfgForAccess = gatewayConfigFromSecurity secCfg
  200   let gwCfgAccess = case mCliStaticDir of
  201         Just dir -> gwCfgForAccess { gcStaticDir = Just (T.pack dir) }
  202         Nothing  -> gwCfgForAccess
  203   -- TLS is validated before anything binds. A half-configured certificate/key
  204   -- pair is a boot refusal rather than a handshake failure on the first
  205   -- connection, where the message would be about a cipher, not a typo.
  206   case tlsConfigError gwCfgAccess of
  207     Just err -> do
  208       logIO logger ErrorS ("gateway: " <> ls err)
  209       ioError (userError (T.unpack err))
  210     Nothing -> pure ()
  211   -- `host` is handed to Warp, so an unusable value is a boot refusal rather
  212   -- than a string in a log line: a malformed or unresolvable host used to be
  213   -- harmless (it only fed the printed URL) and now aborts inside Warp, after
  214   -- the banner. Both halves run here, beside the TLS check, before anything
  215   -- binds.
  216   case gatewayHostError (gcHost gwCfgAccess) of
  217     Just err -> do
  218       logIO logger ErrorS ("gateway: " <> ls err)
  219       ioError (userError (T.unpack err))
  220     Nothing -> pure ()
  221   gatewayHostResolutionError hostResolves (gcHost gwCfgAccess) >>= \case
  222     Just err -> do
  223       logIO logger ErrorS ("gateway: " <> ls err)
  224       ioError (userError (T.unpack err))
  225     Nothing -> pure ()
  226   -- TLS paths default under keys/tls, so `tls = true` is the whole
  227   -- configuration. A missing pair is generated with openssl rather than left
  228   -- to the operator; see 'Seal.Security.TlsCert' for why that is not a
  229   -- substitute for a trusted certificate.
  230   --
  231   -- A pair Seal generated at the default paths is Seal's to maintain: it is
  232   -- replaced when it cannot serve a browser a legal certificate. A pair the
  233   -- operator pointed at with tls_cert/tls_key is never validated or replaced.
  234   gwCfgTls <- if tlsEnabled gwCfgAccess
  235     then do
  236       let (certPath, keyPath) =
  237             effectiveTlsPaths (tlsCertPath paths) (tlsKeyPath paths) gwCfgAccess
  238           managed = isNothing (gcTlsCert gwCfgAccess) && isNothing (gcTlsKey gwCfgAccess)
  239           warnSelfSigned = logIO logger WarningS (ls (generatedCertificateTrustHint certPath))
  240       systemId <- getSystemID
  241       addresses <- localAddressNames
  242       let hosts = sanHostsFor
  243                     (sanInputsFor gwCfgAccess
  244                        (T.pack (nodeName systemId))
  245                        (scAccessControl secCfg >>= acfcOrigin >>= originHostOf)
  246                        addresses)
  247       ensured <- (if managed then ensureGeneratedCertificate else ensureCertificateAt)
  248                    certPath keyPath hosts
  249       case ensured of
  250         Left err -> do
  251           logIO logger ErrorS ("gateway: " <> ls err)
  252           ioError (userError (T.unpack err))
  253         Right TlsCertUnchanged ->
  254           logIO logger DebugS ("TLS certificate unchanged at " <> ls (T.pack certPath))
  255         Right TlsCertCreated -> do
  256           logIO logger InfoS ("generated a self-signed TLS certificate for "
  257             <> ls (T.intercalate ", " hosts) <> " at " <> ls (T.pack certPath)
  258             <> " (private key " <> ls (T.pack keyPath) <> ", mode 0600)")
  259           warnSelfSigned
  260         Right (TlsCertReplaced why) -> do
  261           logIO logger WarningS
  262             ("replaced the generated TLS certificate at " <> ls (T.pack certPath)
  263               <> ": " <> ls why <> ".")
  264           logIO logger WarningS
  265             ("the previous pair is kept at " <> ls (T.pack (certPath <> ".bak"))
  266               <> " and " <> ls (T.pack (keyPath <> ".bak"))
  267               <> "; trust the new certificate on every device that uses this gateway.")
  268           warnSelfSigned
  269       expiry <- certificateExpiryWarning certPath
  270       forM_ expiry (logIO logger WarningS . ls)
  271       pure gwCfgAccess { gcTlsCert = Just (T.pack certPath)
  272                         , gcTlsKey  = Just (T.pack keyPath) }
  273     else pure gwCfgAccess
  274   -- The §10 matrix runs here, before any channel is forked or port bound.
  275   bootInputs <- preflight paths
  276   let planResult = bootInputs >>= bootPlan (BindCoordinates (gcHost gwCfgAccess)
  277                                                              (gcPort gwCfgAccess))
  278   access <- case planResult of
  279     Left refusal -> do
  280       let reason = bootRefusalMessage refusal
  281       logIO logger ErrorS ("access control: " <> ls reason)
  282       ioError (userError (T.unpack reason))
  283     -- Serving HTTPS makes the https origin authoritative; a verified origin
  284     -- that still says http:// would fail every ceremony's origin check.
  285     Right plan -> case tlsOriginError (tlsEnabled gwCfgAccess) (bpOrigin plan) of
  286       Just err -> do
  287         logIO logger ErrorS ("gateway: " <> ls err)
  288         ioError (userError (T.unpack err))
  289       Nothing -> do
  290         -- @--no-enrollment-token@: the operator asserts the network is
  291         -- secured (firewall/VPN), so a 'CliWindow' grant (no secret) is
  292         -- minted instead of a 'CliToken' grant. The browser can enroll a
  293         -- passkey without pasting a token. Passkey auth stays enabled —
  294         -- this is not @mode = "none"@.
  295         let adjustedPlan = if noEnrollToken
  296               then plan { bpMintGrant = CliWindow <$ bpMintGrant plan }
  297               else plan
  298         newAccessControlHandle paths adjustedPlan
  299   printAccessControlBanner paths access
  300   mHandle <- tryOpenVault paths secCfg logger
  301   ref     <- newIORef mHandle
  302   let rt = VaultRuntime
  303             { vrPaths      = paths
  304             , vrConfigPath = cfgPath
  305             , vrHandleRef  = ref
  306             }
  307   mgr <- newTlsManager
  308   callCounter <- newIORef 0
  309   let pr = ProviderRuntime
  310             { prConfigPath  = cfgPath
  311             , prVault       = rt
  312             , prManager     = mgr
  313             , prCallCounter = callCounter
  314             }
  315       cfgRoot = spConfig paths
  316   ensureConfigRepo cfgRoot
  317   let repo = openConfigRepo cfgRoot
  318   embedding <- resolveEmbeddingBackend (rcEmbedding cfg) (spHome paths)
  319   backends <- newBackends paths repo embedding
  320   -- Publish the sub-agent runtime so an interrupt during the gateway run can
  321   -- log exactly which children are being taken down with the process
  322   -- ('Seal.Command.Shutdown').
  323   writeIORef runtimeRef (Just (bRuntime backends))
  324   wakeMutex <- newSessionWakeMutex
  325   -- W4: the source-control repo registry handle (closes over
  326   -- repos.toml). Built once at startup and threaded into ApiDeps for
  327   -- /api/repos CRUD. The handle's rrhList/rrhMutate re-read the file on
  328   -- each call so mutations from other processes are reflected.
  329   repoRegH <- mkRepoRegistryHandle (reposFilePath paths)
  330   -- W5: the /repo slash command's test seam. The ls-remote arm runs real
  331   -- git against repoCloneStateDir; the vault-list arm backs the /repo info
  332   -- non-blocking vault-key advisory. The vault handle (mHandle) may be
  333   -- 'Nothing' if the vault is not configured — in that case /repo test
  334   -- surfaces 'vault locked' (fail-closed) and /repo info skips the advisory.
  335   repoSeam <- mkRepoTestSeam rt repoRegH agentRegH paths
  336   -- W5: persisting tab handle. Load the persisted tab list, drop tabs whose
  337   -- session.json is missing on disk (stale), and seed the TVar. Harness tabs
  338   -- (BoundHarness) are kept as-is; the periodic reconcile sweep (run later)
  339   -- re-resolves their liveness and marks missing windows 'orphaned'. No
  340   -- broadcast at boot (no WS client connected yet); the first subscriber's
  341   -- broadcast trigger refreshes the sidebar.
  342   let tabsPath = tabListPath paths
  343   tabsH <- newPersistingTabsHandle tabsPath
  344   mPersisted <- loadTabList tabsPath
  345   case mPersisted of
  346     Nothing -> pure ()
  347     Just tl -> do
  348       kept <- filterM (sessionTabExists paths) (tlTabs tl)
  349       seedTabsHandle tabsH (renumberTabs kept)
  350   -- Persisting cursor store: load + seed so existing Telegram/Signal
  351   -- conversations re-resolve to their prior sessions (carrying the user's
  352   -- /model use choice) after a seal serve restart. A cursor pointing at a
  353   -- session whose session.json is missing degrades gracefully —
  354   -- resolveTabSession returns Nothing and the loop mints a fresh session
  355   -- on the next message (exactly as if the cursor were absent). No
  356   -- boot-time stale sweep is needed.
  357   cursorsH <- newPersistingCursorStore (cursorMapPath paths)
  358   mCursors <- loadCursorMap (cursorMapPath paths)
  359   for_ mCursors (seedCursorStore cursorsH)
  360   -- Persisting watch-state store: load + seed so a conversation's
  361   -- watch-all-tabs mode is restored after a @seal serve@ restart.
  362   -- Mirrors the cursor-store boot path above.
  363   watchState <- newPersistingWatchState (saveWatchMap (watchMapPath paths))
  364   mWatch <- loadWatchMap (watchMapPath paths)
  365   for_ mWatch (seedWatchState watchState)
  366   reg     <- newHarnessRegistry
  367   tmuxR   <- mkRealTmuxRunner
  368   uiState <- newUiStateHandle paths
  369   askReply <- newAskReplyStore 0  -- 0 = block indefinitely (no timeout); a
  370                                  -- future phase may surface a configurable
  371                                  -- per-session timeout.
  372   approvals <- newApprovalCache
  373   -- Build an in-memory active session (NOT persisted to disk) so the
  374   -- active-session ref has valid provider/model fallbacks. The session
  375   -- only lands on disk when the user sends the first message (the web send
  376   -- handler writes the transcript to the session dir). This avoids
  377   -- polluting the sessions list with an empty session on every `seal serve`.
  378   sessionMeta <- initSessionMeta paths cfg (bAgentDefs backends)
  379   activeRef   <- newIORef sessionMeta
  380   broker <- newStreamBroker 1024
  381   -- Build the shared ChannelDeps early so the reply registry + write locks
  382   -- can be shared with the web send handler (SendDeps). The cursor store,
  383   -- reply registry, and write locks are created inside newChannelDeps.
  384   let loadCfg = fromRight defaultRuntimeConfig <$> loadRuntimeConfig cfgPath
  385       isRemoteExec = case untrustedExecConfigFromSecurity secCfg of
  386         Just uec -> uecMode uec == UemRemote
  387         Nothing  -> False
  388   chanDeps <- newChannelDeps
  389         paths rt repoRegH agentRegH pr backends autonomy (Just broker)
  390         reg tmuxR (Just mgr) approvals loadCfg isRemoteExec tabsH logger cursorsH
  391   let sr = SessionRuntime
  392              { srPaths      = paths
  393              , srConfigPath = cfgPath
  394              , srActive     = activeRef
  395              }
  396       -- The /new command for the web: mints a fresh session, swaps srActive,
  397       -- inserts a new tab into the TabsHandle, swaps srActive to the new
  398       -- session, and returns the old sid. When -r/--repo is given, the
  399       -- repo is cloned via SETUP_REPO (same as the web "Set up repo").
  400       newDeps = NewDeps
  401         { ndPaths = paths
  402         , ndCfg = loadCfg
  403         , ndAgentDefs = backends
  404         , ndChannelLabel = "web"
  405         , ndOldMeta = readIORef activeRef
  406         , ndInsertTab = \_caps newMeta -> do
  407             oldMeta <- readIORef activeRef
  408             let oldSid = smId oldMeta
  409             _ <- insertTabH tabsH (BoundSession (smId newMeta)) KindProvider Nothing
  410             writeIORef activeRef newMeta
  411             pure oldSid
  412         , ndSetupRepo = Just (handleSetupRepo sendDeps)
  413         , ndRepoReg = Just repoRegH
  414         }
  415       -- The slash-command registry is built from the shared
  416       -- 'coreCommandSpecs' (the channel-agnostic core: vault, provider,
  417       -- session, model, agent, tab, stop, terseGrammar, new, repo) plus
  418       -- the web's call/skill placeholders. 'runSlash' rebuilds call/skill
  419       -- AND stop per-request via 'replaceCallSkillSpecs' so those commands
  420       -- target the request's explicit 'SessionId' (the web is
  421       -- multi-session; srActive is not authoritative). The placeholders
  422       -- here never run.
  423       placeholderDispatcher _ _ = pure (Left (OpNotFound (OpName "placeholder")))
  424       coreDeps = CoreCommandDeps
  425         { ccdVault       = rt
  426         , ccdProvider    = pr
  427         , ccdSession     = sr
  428         , ccdAgentDefs   = bAgentDefs backends
  429         , ccdCfgPath     = cfgPath
  430         , ccdPaths       = paths
  431         , ccdTabs        = tabsH
  432         , ccdTabCloseNotifier = mkTabCloseNotifier (cdCursors chanDeps) (cdReplies chanDeps)
  433         , ccdAbortReg    = cdAbortReg chanDeps
  434         , ccdRunRecords  = bRunRecords backends
  435         , ccdStopWriter  = mkStopTranscriptWriter paths (Just broker)
  436         , ccdModelWriter = mkModelTranscriptWriter paths (Just broker)
  437         , ccdRepoReg     = repoRegH
  438         , ccdRepoSeam    = Just repoSeam
  439         }
  440       registry = mkRegistry
  441         ( coreCommandSpecs coreDeps
  442           <> [ skillCommandSpec (bSkills backends) placeholderDispatcher
  443              , callCommandSpec placeholderDispatcher
  444              , newCommandSpec newDeps
  445              ]
  446         )
  447       sendDeps = SendDeps
  448         { sdPaths      = paths
  449         , sdVault      = rt
  450         , sdRepoReg    = repoRegH
  451         , sdAgentReg   = agentRegH
  452         , sdProvider   = pr
  453         , sdSession    = sr
  454         , sdBackends   = backends
  455         , sdConfigRepo = repo
  456         , sdPreprocess = emptyChain
  457         , sdRegistry   = registry
  458         , sdResolve    = resolveSessionProvider pr
  459         , sdAutonomy   = autonomy
  460         , sdBroker     = Just broker
  461         , sdHarnessRegistry = reg
  462         , sdTmuxRunner  = tmuxR
  463         , sdHttpManager = Just mgr
  464         , sdAskReply    = askReply
  465         , sdApprovals   = approvals
  466         , sdReplies     = cdReplies chanDeps
  467         , sdLocks       = cdLocks chanDeps

  568   -- so there is no second listener and no second port to collide with.
  569   --
  570   -- The Origin allowlist is the configured list PLUS origins derived from
  571   -- the HTTP server's host + port. A wildcard host (0.0.0.0) means "bind all
  572   -- interfaces" — the browser may reach the server via any of them, so we
  573   -- can't enumerate the allowed origins ahead of time. In that case, pass an
  574   -- empty allowlist to the WS guard, which triggers wildcard mode (accept
  575   -- any Origin) — overriding even the default loopback origin. For a specific
  576   -- host, derive the origin + prepend it to the configured list.

  579       -- send an https Origin, and an http-only allowlist would reject the
  580       -- WebSocket handshake.
  581       scheme = if tlsEnabled gwCfg then "https" else "http"
  582       httpOrigins = [ scheme <> "://" <> gcHost gwCfg <> ":" <> T.pack (show (gcPort gwCfg))
  583                     | not isWildcard ]
  584       origins = if isWildcard
  585                   then []  -- wildcard host → empty allowlist → Stream.hs accepts any
  586                   else httpOrigins <> gcAllowedOrigins gwCfg

  588         { sgAllowedOrigins = origins
  589         , sgGlobalCap = 1024
  590         , sgTabsHandle = tabsH
  591         , sgPaths = paths
  592         , sgHeartbeatSeconds = 30

  602   -- that died without its bracket cleanup broadcasting idle). Runs every 5
  603   -- minutes and clears only sessions that have been SILENT for 30 minutes
  604   -- — every entry/tool-call broadcast refreshes the session's stamp — and
  605   -- are not busy (no turn in flight, no running background children). A
  606   -- long turn that is still producing output is never cleared: reporting
  607   -- it idle mid-turn mis-sorts the sidebar, drops the chat channels'
  608   -- /watch state, and closes a /bg connection.
  609   _ <- forkIO $ forever $ do
  610     threadDelay (5 * 60 * 1_000_000)
  611     void (reconcileStaleThinking broker
  612             (\sid -> (||) <$> sessionTurnInFlight (sdLocks sendDeps) sid
  613                           <*> hasRunningChildren (bRuntime backends) sid)
  614             (30 * 60))

  658 -- | Take the per-@SEAL_HOME@ daemon lock for the life of this process.
  659 --
  660 -- Refuses rather than queueing: two daemons on one @SEAL_HOME@ would fight for
  661 -- the same port and the same transcript directories, and the second one is
  662 -- always a mistake. The returned lock is never released explicitly — process
  663 -- exit is the release.
  664 claimSealHome :: SealPaths -> IO ServeLock
  665 claimSealHome paths = do
  666   locked <- acquireServeLock (serveLockPath paths)
  667   case locked of
  668     Left err -> ioError (userError (T.unpack err))
  669     Right lock -> pure lock

  702 -- | Renumber a tab list so slots stay contiguous 0..n-1 after dropping stale
  703 -- tabs (I1: the tab list invariant). The TVar's smart constructors would
  704 -- do this via 'removeTab', but here we renumber in bulk after a boot-time
  705 -- filter, so the indices are recomputed from position.

  707 renumberTabs ts = TabList (zipWith renumber [0..] ts)

  717 -- | The address a co-located chat channel uses to reach the gateway. A
  718 -- wildcard bind is not dialable, so it maps to loopback.
  719 clientHostFor :: Seal.Gateway.Config.GatewayConfig -> T.Text
  720 clientHostFor gwCfg
  721   | Seal.Gateway.Config.gcHost gwCfg == "0.0.0.0" = "127.0.0.1"
  722   | otherwise = Seal.Gateway.Config.gcHost gwCfg

  724 -- | The manager every chat channel uses to call the gateway API.
  725 --
  726 -- It carries the process-local service token, so the channel passes the
  727 -- access-control gate without a browser passkey session, and under TLS it
  728 -- trusts the gateway's own PEM certificate: the generated pair is
  729 -- self-signed, so the system store cannot verify it. The token is injected by
  730 -- the manager rather than threaded through every 'HttpClient' call.
  731 gatewayManager
  732   :: SealLogger
  733   -> Seal.Gateway.Config.GatewayConfig
  734   -> T.Text   -- ^ Client host
  735   -> T.Text   -- ^ Gateway service token
  736   -> IO Manager
  737 gatewayManager logger gwCfg clientHost serviceToken = do
  738   base <- if tlsEnabled gwCfg then tlsSettings else pure defaultManagerSettings
  739   newManager (base { managerModifyRequest = pure . authorize })
  740   where
  741     authorize req = req
  742       { requestHeaders =
  743           ("Authorization", "Bearer " <> TE.encodeUtf8 serviceToken)
  744             : filter ((/= "Authorization") . fst) (requestHeaders req)
  745       }
  746     tlsSettings = case Seal.Gateway.Config.gcTlsCert gwCfg of
  747       Nothing -> pure tlsManagerSettings
  748       Just certPath -> do
  749         mPinned <- readCertificateStore (T.unpack certPath)
  750         case mPinned of
  751           Nothing -> do
  752             logIO logger WarningS
  753               ("gateway: cannot read TLS certificate at " <> ls certPath
  754                  <> "; chat channels may fail to verify the TLS handshake")
  755             pure tlsManagerSettings
  756           Just pinned -> do
  757             eSystem <- try getSystemCertificateStore
  758               :: IO (Either SomeException CertificateStore)
  759             let systemCerts = either (const []) listCertificates eSystem
  760                 store = makeCertificateStore (systemCerts <> listCertificates pinned)
  761                 baseParams = defaultParamsClient (T.unpack clientHost) BS.empty
  762                 params = baseParams
  763                   { clientShared = (clientShared baseParams) { sharedCAStore = store } }
  764             pure (mkManagerSettings (TLSSettings params) Nothing)

  766 -- | Fork a single resolved chat channel. This is the post-resolution body
  767 -- extracted from the former monolithic @forkNewSignalChatChannel@ /
  768 -- @forkNewTelegramChatChannel@ (design §3.5): resolution has already
  769 -- happened ('resolveChannelConfig' in the caller); this function builds the
  770 -- gateway-client transport and forks 'withSignalChatChannel' or
  771 -- 'withTelegramChatChannel'. The existing library functions are reused
  772 -- unchanged. The 'ChannelId' is used only for diagnostic logging.
  773 forkResolvedChannel
  774   :: SealLogger
  775   -> Seal.Gateway.Config.GatewayConfig
  776   -> Manager
  777     -- ^ Gateway API manager (pinned to the gateway certificate under TLS).
  778   -> Manager
  779     -- ^ Shared manager for Telegram's own API.
  780   -> T.Text
  781     -- ^ Gateway service token (sent on the WS handshake).
  782   -> ChannelId
  783   -> ResolvedChannel
  784   -> WatchState
  785   -> IO ()
  786 forkResolvedChannel logger gwCfg gwMgr mgr serviceToken chanId resolved watchState = do
  787   let clientHost = clientHostFor gwCfg
  788       -- The WS stream is served on the SAME port as the HTTP API. When the
  789       -- gateway serves TLS, the channel client must speak https/wss too and
  790       -- trust the gateway's own certificate: plain http against the TLS
  791       -- listener fails with 'InsecureConnectionDenied', and the generated
  792       -- certificate is self-signed, so the system store cannot verify it.
  793       chatGwCfg = ChatTypes.mkGatewayConfig clientHost
  794         (Seal.Gateway.Config.gcPort gwCfg)
  795         (tlsEnabled gwCfg)
  796         (Seal.Gateway.Config.gcTlsCert gwCfg)
  797       -- The WS handshake also passes the access-control gate, so the
  798       -- channel's connection carries the same service token as its HTTP
  799       -- calls (which the manager injects).
  800       chanCfg = (defaultChatChannelConfig gwMgr chatGwCfg)
  801         { cccStartWs = ChatWs.startWsClient
  802             (ChatTypes.gcWsSecure chatGwCfg)
  803             (ChatTypes.gcTlsCert chatGwCfg)
  804             (Just serviceToken) }
  805       cid = channelIdText chanId
  806   case resolved of
  807     ResolvedSignal account chunkLimit allow -> do
  808       let accountText = Seal.Signal.Config.signalAccountText account
  809       eTransport <- ChatSignal.mkRealSignalChatTransport accountText
  810       case eTransport of
  811         Left err -> logIO logger WarningS
  812                       ("channel " <> ls cid <> ": signal chat channel: " <> ls err)
  813         Right transport ->
  814           void (forkIO (withSignalChatChannel
  815                           allow
  816                           chunkLimit
  817                          1000000
  818                           transport
  819                           (\chan -> runChatChannel chanCfg chan watchState)))
  820     ResolvedTelegram token chunkLimit allow -> do
  821       let tokenText = Seal.Telegram.Config.telegramTokenText token
  822       transport <- ChatTelegram.mkRealTelegramChatTransport tokenText mgr
  823       -- Register the bot's command menu (/help on Telegram shows it) from
  824       -- the gateway's slash-command registry. Forked: the chat listeners are
  825       -- started before 'runGateway' binds the HTTP port, so an immediate
  826       -- fetch always races the bind — 'registerWithRetry' retries for ~10s.
  827       -- Failure is non-fatal (the bot still works; the menu is stale).
  828       void (forkIO (do
  829         eRes <- registerWithRetry 10 1000000
  830                   (ChatHttp.httpGetCommands gwMgr (ChatTypes.gcApiBase chatGwCfg))
  831                   (tgSetCommandsViaApi mgr tokenText)
  832         case eRes of
  833           Right () -> pure ()
  834           Left e   -> logIO logger WarningS
  835                         ("channel " <> ls cid
  836                            <> ": telegram chat channel: command menu not registered: " <> ls e)))
  837       void (forkIO (withTelegramChatChannel
  838                       allow
  839                       chunkLimit
  840                      1000000
  841                       transport
  842                       (\chan -> runChatChannel chanCfg chan watchState)))

  856 -- | Fork chat-channel listeners for every entry in 'rcChannels' (design
  857 -- §3.5). Each 'ChannelConfig' is resolved via the pure
  858 -- 'resolveChannelConfig'; the caller pre-fetches the vault token for
  859 -- Telegram channels via 'vhGet' + 'telegramVaultKeyFor'. A channel that
  860 -- fails resolution logs a warning rather than crashing serve startup.
  861 -- Replaces the former single-fork @forkNewSignalChatChannel@ /
  862 -- @forkNewTelegramChatChannel@.
  863 forkChatChannels
  864   :: SealLogger
  865   -> Seal.Gateway.Config.GatewayConfig
  866   -> Manager
  867   -> T.Text
  868     -- ^ Gateway service token.
  869   -> RuntimeConfig
  870   -> Maybe VaultHandle
  871   -> WatchState
  872   -> IO ()
  873 forkChatChannels logger gwCfg mgr serviceToken cfg mHandle watchState = do
  874   let clientHost = clientHostFor gwCfg
  875   gwMgr <- gatewayManager logger gwCfg clientHost serviceToken
  876   void (Map.traverseWithKey (forkOne gwMgr) (rcChannels cfg))
  877   where
  878     forkOne gwMgr chanId chanCfg = do
  879       -- Pre-fetch the vault token for Telegram channels (IO); Signal
  880       -- channels pass Nothing (the account lives in config, not the vault).
  881       mVaultToken <- case (ccKind chanCfg, mHandle) of
  882         (Telegram, Just vh) -> do
  883           r <- vhGet vh (telegramVaultKeyFor chanId)
  884           pure (case r of
  885             Right bs -> Just (TE.decodeUtf8 bs)
  886             Left _   -> Nothing)
  887         _ -> pure Nothing
  888       case resolveChannelConfig chanId chanCfg mVaultToken of
  889         Left err -> logIO logger WarningS
  890                       ("channel " <> ls (channelIdText chanId) <> ": " <> ls err)
  891         Right resolved ->
  892           void (forkIO (forkResolvedChannel logger gwCfg gwMgr mgr serviceToken
  893                           chanId resolved watchState))

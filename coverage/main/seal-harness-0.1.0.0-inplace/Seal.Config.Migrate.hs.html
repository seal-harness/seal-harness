    2 -- | One-time, idempotent migration of security-critical fields (vault
    3 -- settings, @[untrusted_execution]@ and @[gateway]@) from
    4 -- @config\/config.toml@ to @~\/.seal\/security.toml@ (design §4 Approach B).
    5 --
    6 -- The migration runs at boot, before 'Seal.Config.Security.loadSecurityConfig' /
    7 -- 'Seal.Config.File.loadRuntimeConfig'. It is idempotent and fail-open (a write
    8 -- failure logs an error and boot proceeds — the legacy values stay where they
    9 -- are, and the next boot retries).
   10 --
   11 -- Two rules keep it from losing data:
   12 --
   13 -- * __Strip only after a successful write.__ Re-encoding @config.toml@ (which
   14 --   'cleanConfigToml' does) drops any section the runtime codec does not know,
   15 --   so it only happens once the new copy exists in @security.toml@.
   16 -- * __@security.toml@ wins, except where it has no answer.__ A stale
   17 --   @[untrusted_execution]@ / vault field in @config.toml@ is discarded, as
   18 --   before. The @[gateway]@ table is different: on an upgraded install
   19 --   @security.toml@ exists (seeded on first run) but has never carried a
   20 --   @[gateway]@ table, so "security wins" would reset every operator's bind
   21 --   address to loopback. When security has no table and config has one, the
   22 --   table is moved across first.
   23 --
   24 -- A third rule covers a subtler case: a live @[gateway]@ header whose values do
   25 -- not survive decoding (every key is 'Toml.dioptional', which demotes a type
   26 -- mismatch to "absent"). Moving that would silently drop a value, and stripping
   27 -- it would destroy the table, so the migration does neither and says so.
   28 module Seal.Config.Migrate
   29   ( migrateSecurityConfig
   30   , configHasLegacySecurityFields
   31   ) where

   54 -- | One-time, idempotent migration. Examines both @config.toml@ and
   55 -- @security.toml@ and moves any legacy security fields from @config.toml@ to
   56 -- @security.toml@. Cases (design §4 B):
   57 --
   58 --   1. @security.toml@ absent, @config.toml@ has legacy fields → migrate.
   59 --   2. @security.toml@ exists, @config.toml@ has stale legacy fields →
   60 --      @security.toml@ wins (precedence); clean stale fields from @config.toml@.
   61 --      The @[gateway]@ table is merged across instead when @security.toml@ has
   62 --      none — see 'reconcileWithExisting'.
   63 --   3. @security.toml@ exists, @config.toml@ clean → no-op (idempotent).
   64 --   4. @config.toml@ absent → no-op (defaults apply; @security.toml@ may exist).
   65 --   5. Write fails (permissions) → log a loud error and proceed (fail-open).
   66 --   6. @config.toml@ carries a field but cannot be read, or carries a
   67 --      @[gateway]@ header whose values do not decode → log a loud error and
   68 --      change nothing (never strip what was not moved).
   69 migrateSecurityConfig :: SealPaths -> IO ()
   70 migrateSecurityConfig paths = do
   71   let cfgPath = configFilePath paths
   72       secPath = securityFilePath paths
   73   secExists <- doesFileExist secPath
   74   cfgExists <- doesFileExist cfgPath
   75   case (secExists, cfgExists) of
   76     -- Case 4 (and case 3 with no config.toml): nothing in config.toml to move.
   77     (_, False)    -> pure ()
   78     (False, True) -> migrateIntoNewFile cfgPath secPath
   79     (True, True)  -> reconcileWithExisting cfgPath secPath
   80 
   81 -- | Do these @config.toml@ bytes still carry the pre-split security fields?
   82 --
   83 -- The same substring scan 'migrateSecurityConfig' uses, exported because the
   84 -- first-run scaffolder ('Seal.Config.Templates.ensureConfigTemplates') has to
   85 -- know: creating @security.toml@ while these fields are still in @config.toml@
   86 -- would move the install onto the "security.toml wins" branch below.
   87 --
   88 -- The scan is deliberately crude: it also trips on a field name mentioned in a
   89 -- comment, which at worst defers the template and lets the migration run. The
   90 -- reverse mistake — missing a live field — is the one that loses data.
   91 configHasLegacySecurityFields :: Text -> Bool
   92 configHasLegacySecurityFields contents = any (`T.isInfixOf` contents) legacyMarkers
   93 
   94 -- | Lines/tokens that indicate legacy security fields are present in config.toml.
   95 legacyMarkers :: [Text]
   96 legacyMarkers =
   97   [ "vault_path"
   98   , "vault_recipient"
   99   , "vault_identity"
  100   , "vault_unlock"
  101   , "vault_key_type"
  102   , "[untrusted_execution]"
  103   , "[gateway]"
  104   ]

  106 -- | The keys the @[gateway]@ codec actually reads. A key outside this set was
  107 -- never gateway configuration — TOML puts every bare key after a table header
  108 -- inside that table, so a stray @default_provider@ under @[gateway]@ is ignored
  109 -- by the loader and loses no setting when it is dropped.
  110 gatewayKeys :: [Text]
  111 gatewayKeys =
  112   [ "port", "host", "static_dir", "allowed_origins"
  113   , "tls_cert", "tls_key", "tls", "tls_hosts"
  114   ]
  115 
  116 -- | Leaf paths of the keys the gateway codec reads, as @gateway.port@.
  117 -- Empty for a commented-out block (it is not a table node at all).
  118 gatewayLeafPaths :: Toml.TOML -> [Text]
  119 gatewayLeafPaths toml =
  120   [ T.intercalate "." pieces
  121   | path <- tomlKeyPaths toml
  122   , let pieces = map Toml.unPiece path
  123   , startsWithGateway pieces
  124   , readsKey pieces
  125   ]
  126   where
  127     startsWithGateway ("gateway" : _) = True
  128     startsWithGateway _               = False
  129     readsKey pieces = case reverse pieces of
  130       (key : _) -> key `elem` gatewayKeys
  131       []        -> False
  132 
  133 -- | Can every key of @config.toml@'s @[gateway]@ table be written back out?
  134 --
  135 -- Every gateway key is optional at the codec layer, and @Toml.dioptional@
  136 -- demotes a value it cannot read to "absent": @port = "8443"@ reads as no port
  137 -- at all. Moving such a table would silently drop the key, and stripping
  138 -- @config.toml@ would destroy it, so the migration does neither and says why.
  139 -- A commented-out block has no leaf paths, so it never blocks the migration.
  140 gatewayTableSurvives :: Text -> SecurityConfig -> Bool
  141 gatewayTableSurvives contents secCfg = case Toml.parse contents of
  142   Left _     -> True     -- a parse failure is reported by its own branch
  143   Right raw  -> all (`elem` reEncoded) (gatewayLeafPaths raw)
  144   where
  145     reEncoded = case scGateway secCfg of
  146       Nothing -> []
  147       Just gw -> case Toml.parse (Toml.encode gatewayTableCodec (Just gw)) of
  148         Left _     -> []
  149         Right toml -> gatewayLeafPaths toml
  150 
  151 -- | Case 1: no @security.toml@. Everything the security codec understands is
  152 -- written to a new one, then stripped from @config.toml@.
  153 migrateIntoNewFile :: FilePath -> FilePath -> IO ()
  154 migrateIntoNewFile cfgPath secPath = do
  155   contents <- TIO.readFile cfgPath
  156   if not (configHasLegacySecurityFields contents)
  157     then pure ()
  158     else case decodeSecurityFields contents of
  159       Left err -> warnUndecodable cfgPath err
  160       Right secCfg
  161         | not (gatewayTableSurvives contents secCfg) ->
  162             warnUndecodableGateway cfgPath
  163         | otherwise -> do
  164             written <- tryWriteSecurityConfig secPath secCfg
  165             case written of
  166               Left err -> warnWriteFailed secPath err
  167               Right () -> do
  168                 cleanConfigToml cfgPath
  169                 say ("[seal] migrated security settings from " <> T.pack cfgPath
  170                       <> " to " <> T.pack secPath
  171                       <> " (one-time). Edit " <> T.pack secPath <> " for future changes.")
  172 
  173 -- | Case 2: @security.toml@ exists. Strip the stale fields from @config.toml@,
  174 -- moving the @[gateway]@ table across first when security has none of its own.
  175 reconcileWithExisting :: FilePath -> FilePath -> IO ()
  176 reconcileWithExisting cfgPath secPath = do
  177   contents <- TIO.readFile cfgPath
  178   if not (configHasLegacySecurityFields contents)
  179     then pure ()
  180     else case decodeSecurityFields contents of
  181       Left err -> warnUndecodable cfgPath err
  182       Right fromCfg
  183         | not (gatewayTableSurvives contents fromCfg) ->
  184             warnUndecodableGateway cfgPath
  185         | otherwise -> do
  186             loaded <- loadSecurityConfig secPath
  187             case loaded of
  188               Left err ->
  189                 -- Never rewrite a file we could not read: the merge would start
  190                 -- from defaults and drop whatever else the operator has there.
  191                 say ("[seal] WARNING: " <> T.pack secPath <> " could not be read ("
  192                       <> err <> "). Leaving " <> T.pack cfgPath <> " untouched; fix "
  193                       <> T.pack secPath <> " and restart.")
  194               Right sec
  195                 | isJust (scGateway fromCfg) && isNothing (scGateway sec) -> do
  196                     written <- saveSecurityConfig secPath
  197                                  (sec { scGateway = scGateway fromCfg })
  198                                  PreserveCredentials
  199                     case written of
  200                       Left err -> warnWriteFailed secPath err
  201                       Right () -> do
  202                         cleanConfigToml cfgPath
  203                         say ("[seal] moved [gateway] from " <> T.pack cfgPath <> " to "
  204                               <> T.pack secPath <> " (one-time). Edit " <> T.pack secPath
  205                               <> " for future changes.")
  206                 | otherwise -> do
  207                     cleanConfigToml cfgPath
  208                     say ("[seal] removed stale [untrusted_execution]/vault_*/[gateway] from "
  209                           <> T.pack cfgPath <> " (security.toml takes precedence)")
  210 
  211 -- | Decode the security-shaped subset of a @config.toml@ document.
  212 decodeSecurityFields :: Text -> Either Text SecurityConfig
  213 decodeSecurityFields contents = case Toml.parse contents of
  214   Left err     -> Left (Toml.unTomlParseError err)
  215   Right toml   -> case Toml.runTomlCodec securityConfigCodec toml of
  216     Success cfg  -> Right cfg
  217     Failure errs -> Left (Toml.prettyTomlDecodeErrors errs)
  218 
  219 -- | A field is present but does not decode. Changing nothing is the only safe
  220 -- outcome: stripping it would destroy the value, and the migration is a move,
  221 -- not a validation.
  222 warnUndecodable :: FilePath -> Text -> IO ()
  223 warnUndecodable cfgPath err =
  224   say ("[seal] WARNING: " <> T.pack cfgPath <> " carries security settings that could "
  225         <> "not be read, so nothing was moved or removed:\n  " <> err
  226         <> "\n  Fix the value, or move the table to ~/.seal/security.toml by hand. "
  227         <> "The gateway uses its defaults until then.")
  228 
  229 -- | A live @[gateway]@ header whose values decode to "absent".
  230 --
  231 -- Every gateway key is optional at the codec layer, and @Toml.dioptional@
  232 -- demotes a type mismatch to "absent": @port = "8443"@ (quoted) reads as no
  233 -- port at all. Moving that would drop the value silently; stripping it would
  234 -- destroy the table. Do neither and say why.
  235 warnUndecodableGateway :: FilePath -> IO ()
  236 warnUndecodableGateway cfgPath =
  237   say ("[seal] WARNING: " <> T.pack cfgPath <> " has a [gateway] table whose values could "
  238         <> "not all be read (every gateway key is optional, so a quoted number or an "
  239         <> "unexpected spelling reads as absent). Nothing was moved or removed.\n"
  240         <> "  Fix the value, or move the table to ~/.seal/security.toml by hand. "
  241         <> "The gateway uses its defaults until then.")
  242 
  243 -- | Clean config.toml: re-save the loaded RuntimeConfig, which drops the
  244 -- legacy fields (the RuntimeConfig codec no longer parses them).
  245 --
  246 -- Only ever called after the corresponding write to @security.toml@ succeeded.
  247 cleanConfigToml :: FilePath -> IO ()
  248 cleanConfigToml cfgPath = do
  249   eCfg <- loadRuntimeConfig cfgPath
  250   case eCfg of
  251     Right cfg -> saveRuntimeConfig cfgPath cfg
  252     Left _    -> pure ()
  253 
  254 -- | Try to write the security config (fail-open on error).
  255 tryWriteSecurityConfig :: FilePath -> SecurityConfig -> IO (Either Text ())
  256 tryWriteSecurityConfig path cfg = do
  257   -- Two layers now: the write can throw, and it can refuse. Both are failures
  258   -- the caller reports the same way, so they are flattened here rather than
  259   -- letting a nested Either escape.
  260   result <- try @SomeException (saveSecurityConfig path cfg PreserveCredentials)
  261   pure $ case result of
  262     Left err         -> Left (T.pack (show err))
  263     Right (Left err) -> Left err
  264     Right (Right ()) -> Right ()
  265 
  266 -- | A failed write leaves @config.toml@ alone, so the settings survive to the
  267 -- next boot. Say that, because "your settings were not moved" is very different
  268 -- from "your settings are gone".
  269 warnWriteFailed :: FilePath -> Text -> IO ()
  270 warnWriteFailed secPath err =
  271   say ("[seal] WARNING: could not write " <> T.pack secPath <> " — " <> err
  272         <> "\n  Your settings are still in config.toml and nothing was removed. "
  273         <> "Boot continues on defaults; the next boot retries. "
  274         <> "Fix permissions at ~/.seal/ and restart.")
  275 
  276 say :: Text -> IO ()
  277 say = hPutStrLn stderr . T.unpack

    2 -- | Seeding a branched session: copy the first @N@ conversation lines of a
    3 -- parent session into a new child session directory.
    4 --
    5 -- The transcript is the only thing a branch inherits. The workdir is
    6 -- deliberately /not/ copied: the transcript records tool inputs and
    7 -- outputs, never filesystem snapshots, so a historical workdir cannot be
    8 -- reconstructed — and copying the /current/ one would be misleading the
    9 -- moment the branch point is earlier than the present. A new session id
   10 -- already gets a fresh empty workdir via 'ensureSessionWorkdir', so the
   11 -- correct behaviour is the default.
   12 module Seal.Session.Branch
   13   ( branchSession
   14   , clampBranchLen
   15   , foldEntryPrefix
   16   ) where

   40 -- | Clamp a client-supplied branch length into @[0, parentLen]@.
   41 --
   42 -- The value reaches us from the browser, so it is untrusted. Clamping
   43 -- (rather than rejecting) means a stale frontend whose transcript view is
   44 -- ahead of the parent's file is harmless: the branch simply copies what
   45 -- exists.
   46 clampBranchLen :: Int -> Int -> Int
   47 clampBranchLen parentLen n = max 0 (min parentLen n)
   48 
   49 -- | Select the prefix of @entries.jsonl@ that belongs to the first @n@
   50 -- conversation lines.
   51 --
   52 -- This replicates the /writer's/ @erConvLen@ convention, which is not
   53 -- uniform per kind ('Seal.Handles.Transcript.writeOne'):
   54 --
   55 --   * 'EKRequest', 'EKResponse', 'EKCompaction' — @erConvLen@ is the
   56 --     conversation length after the entry (absolute).
   57 --   * 'EKHarness' — @erConvLen@ is a delta: how many lines that entry
   58 --     contributed (0 for a plain opcode invocation, 1 for a skill load or
   59 --     setup-repo result).
   60 --
   61 -- An entry is included iff its cursor-after is @<= n@. Entries are
   62 -- append-ordered, so the first entry that overflows ends the prefix.
   63 foldEntryPrefix :: Int -> [EntryRecord] -> [EntryRecord]
   64 foldEntryPrefix n = go 0
   65   where
   66     go _ [] = []
   67     go cursor (e : es)
   68       | after > n = []
   69       | otherwise = e : go after es
   70       where
   71         after = case erKind e of
   72           EKHarness -> cursor + erConvLen e
   73           _         -> erConvLen e

   75 -- | Copy the parent's transcript prefix into a new child session directory.
   76 --
   77 -- Copies @conversation.jsonl@ byte-exactly (so the inherited lines are
   78 -- identical to the parent's) and the matching @entries.jsonl@ prefix, then
   79 -- builds @conversation.idx@. Writes no @session.json@ — the caller does
   80 -- that last, so a failed seed leaves nothing that 'listSessions' would
   81 -- surface.
   82 --
   83 -- The anchor count is derived from the /newline-terminated/ lines actually
   84 -- present rather than from the index, which also makes a torn trailing
   85 -- line (an in-flight turn mid-write) harmless: a partial final line has no
   86 -- newline, so it never counts toward @n@ and is never copied.
   87 branchSession
   88   :: SealPaths
   89   -> SessionId   -- ^ parent
   90   -> SessionId   -- ^ child
   91   -> Int         -- ^ requested conversation length
   92   -> IO (Either Text ())
   93 branchSession paths parentSid childSid requestedLen = do
   94   mParentMeta <- readSessionMeta paths parentSid
   95   -- Refuse if the child id already names a real session. Session ids are
   96   -- timestamp-derived to the millisecond, so a same-millisecond mint could
   97   -- collide; clobbering an existing session's transcript would be
   98   -- unacceptable, so this is a hard guard rather than an overwrite.
   99   childExists <- doesFileExist (sessionMetaPath paths childSid)
  100   case (mParentMeta, childExists) of
  101     (Nothing, _) -> pure (Left ("parent session not found: " <> sessionIdText parentSid))
  102     (Just _, True) ->
  103       pure (Left ("refusing to branch into an existing session: " <> sessionIdText childSid))
  104     (Just _, False) -> do
  105       let parentConv  = sessionConversationPath paths parentSid
  106           parentEntry = sessionEntriesPath paths parentSid
  107           childDir    = sessionDir paths childSid
  108           childConv   = sessionConversationPath paths childSid
  109           childEntry  = sessionEntriesPath paths childSid
  110           childIdx    = sessionConversationIndexPath paths childSid
  111 
  112       eConv <- readConversationByLines parentConv
  113       case eConv of
  114         Left err -> pure (Left err)
  115         Right convBytes -> do
  116           let n = clampBranchLen (length convBytes) requestedLen
  117               copiedBytes = take n convBytes
  118           eResult <- try @IOException $ do
  119             createDirectoryIfMissing True childDir
  120             when (n > 0) $ do
  121               BS.writeFile childConv (BS.concat copiedBytes)
  122               eEntries <- readEntryPrefix parentEntry n
  123               case eEntries of
  124                 Right er -> BS.writeFile childEntry er
  125                 Left err -> ioError (userError (T.unpack err))
  126               -- Build-if-missing, matching 'withIndexedTranscript'. Only
  127               -- when the conversation file exists: a zero-length branch
  128               -- must look exactly like a fresh session (no conversation
  129               -- file, no index), because a lone [0] offset is rejected by
  130               -- 'validateIndex' as corrupt.
  131               _ <- ensureIndex childConv childIdx
  132               pure ()
  133           case eResult :: Either IOException () of
  134             Right () -> pure (Right ())
  135             Left ioe -> do
  136               -- Leave no partial child directory behind.
  137               exists <- doesDirectoryExist childDir
  138               when exists (ignoreIO (removeDirectoryRecursive childDir))
  139               pure (Left ("could not seed branch: " <> T.pack (show ioe)))

  141 -- | Read a conversation file as (per-line bytes, per-line bytes). The
  142 -- line list is used for counting; the byte list is what we copy verbatim.
  143 --
  144 -- An absent file is @Right ([], [])@ — a brand-new parent has nothing to
  145 -- inherit, which is not an error.
  146 readConversationByLines :: FilePath -> IO (Either Text [BS.ByteString])
  147 readConversationByLines path = do
  148   exists <- doesFileExist path
  149   if not exists
  150     then pure (Right [])
  151     else do
  152       eBs <- try @IOException (BS.readFile path)
  153       pure $ case eBs of
  154         Left ioe -> Left ("could not read parent conversation: " <> T.pack (show ioe))
  155         Right bs ->
  156           -- Split on newlines (0x0a). A trailing partial line (no final
  157           -- newline) is DROPPED: it belongs to an in-flight write.
  158           let complete = dropTrailingPartial (BS.split 0x0a bs)
  159           in Right [ p <> "\n" | p <- complete ]
  160 
  161 -- | 'BS.split' on a file ending with a newline yields a final empty
  162 -- element. That empty element is the artifact of the trailing newline, not
  163 -- a line; drop it. Any /other/ final element is a genuinely partial line
  164 -- and is dropped too.
  165 dropTrailingPartial :: [BS.ByteString] -> [BS.ByteString]
  166 dropTrailingPartial pieces = case reverse pieces of
  167   ("" : rest) -> reverse rest   -- file ended with '\n': no partial line
  168   (_  : rest) -> reverse rest   -- partial line mid-write: drop it
  169   []          -> []

  171 -- | Read the parent's @entries.jsonl@ and encode the prefix for the child.
  172 readEntryPrefix :: FilePath -> Int -> IO (Either Text BS.ByteString)
  173 readEntryPrefix path n = do
  174   exists <- doesFileExist path
  175   if not exists
  176     then pure (Right BS.empty)
  177     else do
  178       eRaw <- try @IOException (readFileTextStrict path)
  179       pure $ case eRaw of
  180         Left ioe -> Left ("could not read parent entries: " <> T.pack (show ioe))
  181         Right raw ->
  182           let decoded = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  183                                 (filter (not . T.null) (T.lines raw))
  184           in Right (BS.concat (map (\e -> encodeEntryRecordRaw e <> "\n")
  185                                    (foldEntryPrefix n decoded)))

   21 -- Raw input
   22 -- ---------------------------------------------------------------------------
   23 
   24 newtype RawInbound = RawInbound Text
   25   deriving stock (Eq, Show)

   27 -- ---------------------------------------------------------------------------
   28 -- Preprocess chain
   29 -- ---------------------------------------------------------------------------
   30 
   31 -- | A single preprocess stage. 'Left' aborts the chain with a rejection message.
   32 type PreprocessStage = RawInbound -> IO (Either Text RawInbound)
   33 
   34 -- | An ordered sequence of 'PreprocessStage's run before classification.
   35 newtype PreprocessChain = PreprocessChain [PreprocessStage]
   36 
   37 -- | The empty chain: all input passes through unchanged.
   38 emptyChain :: PreprocessChain
   39 emptyChain = PreprocessChain []
   40 
   41 -- | Run every stage in order, short-circuiting on the first 'Left'.
   42 runChain :: PreprocessChain -> RawInbound -> IO (Either Text RawInbound)
   43 runChain (PreprocessChain stages) = go stages
   44   where
   45     go []       r = pure (Right r)
   46     go (s : ss) r = s r >>= \case
   47       Left err -> pure (Left err)
   48       Right r' -> go ss r'
   49 
   50 -- ---------------------------------------------------------------------------
   51 -- Disposition
   52 -- ---------------------------------------------------------------------------
   53 
   54 data Disposition
   55   = DispatchAction CommandAction  -- ^ a parsed command to run
   56   | ShowText Text                 -- ^ help text or parse error to echo
   57   | PlainMessage Text             -- ^ non-slash input (MVP stub)
   58   | Rejected Text                 -- ^ preprocess chain rejected the input
   59 
   60 -- | Classify one inbound line. The chain runs FIRST; if it rejects the
   61 -- input the result is 'Rejected' regardless of content.
   62 --
   63 -- Classification (after chain passes):
   64 --
   65 -- * Leading @\/@  → 'parseSlash' → 'DispatchAction' | 'ShowText'
   66 -- * Otherwise     → 'PlainMessage'
   67 ingest :: Registry -> PreprocessChain -> RawInbound -> IO Disposition
   68 ingest registry chain raw = do
   69   chainResult <- runChain chain raw
   70   case chainResult of
   71     Left msg             -> pure (Rejected msg)
   72     Right (RawInbound t) ->
   73       if T.isPrefixOf "/" t
   74         then pure $ case parseSlash registry t of
   75           ParsedAction a     -> DispatchAction a
   76           ParseHelp Nothing  -> ShowText (renderHelpIndex registry)
   77           ParseHelp (Just n) -> ShowText (renderHelpFor registry n)
   78           ParseFailure txt   -> ShowText txt
   79         else pure (PlainMessage t)

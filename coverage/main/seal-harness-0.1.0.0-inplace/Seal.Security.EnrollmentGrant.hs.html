   83 -- | How a grant was minted — and, for an enrolled credential, how it was added
   84 -- (@added_via@ plus @legacy@, which no grant can produce).
   85 --
   86 -- The distinction is load-bearing, not descriptive: a 'CliWindow' grant is
   87 -- authority with no secret at all, so it exists only in the bootstrap state
   88 -- (nothing enrolled, loopback bind, loopback origin). Any other combination is
   89 -- a refusal, which is why the type flows into both the boot plan and the
   90 -- resolver rather than being flattened to a boolean.
   91 data GrantSource
   92   = CliToken
   93     -- ^ Minted by @seal access-control enroll@ or by boot; presented as a
   94     -- request header.
   95   | CliWindow
   96     -- ^ @enroll --no-token@: loopback-confined, bootstrap-only, no secret
   97     -- presented.
   98   | Assertion
   99     -- ^ Minted by a successful assertion from an already-enrolled credential;
  100     -- the authority for adding a second passkey.
  101   | UiToken
  102     -- ^ Minted by an authenticated browser session that has just completed a
  103     -- step-up assertion (@POST /api/auth/enroll/verify@ with
  104     -- @grant: portable@), so the operator can open the link on a *different*
  105     -- device. Token-presented, file-backed, exactly like 'CliToken' — it
  106     -- differs only in provenance, which the audit log and @minted_via@ report.
  107   deriving stock (Eq, Show)
  108 
  109 -- | The wire/TOML spelling (@minted_via@, @added_via@).
  110 grantSourceText :: GrantSource -> Text
  111 grantSourceText source = case source of
  112   CliToken  -> "cli_token"
  113   CliWindow -> "cli_window"
  114   Assertion -> "assertion"
  115   UiToken   -> "ui_token"

  117 -- | Parse the stored spelling. 'Nothing' for anything unrecognised — callers
  118 -- decide whether that is a refusal (grants) or a fallback to @legacy@
  119 -- (credentials, where claiming nothing about the enrolling authority is safe).
  120 parseGrantSource :: Text -> Maybe GrantSource
  121 parseGrantSource raw = case raw of
  122   "cli_token"  -> Just CliToken
  123   "cli_window" -> Just CliWindow
  124   "assertion"  -> Just Assertion
  125   "ui_token"   -> Just UiToken
  126   _            -> Nothing

  128 -- | Where a grant is in its lifecycle (design §7.3).
  129 --
  130 -- @Minted@ may be bound to exactly one challenge; @GrantBound@ records which,
  131 -- so a grant bound to challenge A cannot redeem B.
  132 data GrantPhase
  133   = GrantMinted
  134   | GrantBound !ChallengeId
  135   deriving stock (Eq, Show)

  145 -- | An outstanding enrollment grant.
  146 --
  147 -- @grTokenSha@ is the SHA-256 of the token the client will present — never the
  148 -- token itself, so a copy of the state directory does not hand over enrollment
  149 -- authority. It is 'Nothing' only for 'CliWindow', which presents no secret at
  150 -- all and is confined to a loopback-bound, credential-free bootstrap.
  151 data Grant = Grant
  152   { grId        :: !GrantId
  153   , grExpiresAt :: !UTCTime
  154   , grSource    :: !GrantSource
  155   , grPhase     :: !GrantPhase
  156   , grTokenSha  :: !(Maybe ByteString)
  157   } deriving stock (Eq, Show)

  159 -- | One ceremony's exclusive hold on a grant (design §7.3).
  160 --
  161 -- Created @O_CREAT|O_EXCL@ when @options@ binds, so the /create/ result decides
  162 -- which of two racing verifies wins — a prior read cannot, because two readers
  163 -- can both see "no claim" before either writes.
  164 --
  165 -- @clDeadline@ is 'effectiveDeadline': the claim may never outlive the grant it
  166 -- belongs to.
  167 data GrantClaim = GrantClaim
  168   { clGrantId     :: !GrantId
  169   , clChallengeId :: !ChallengeId
  170   , clDeadline    :: !UTCTime
  171   } deriving stock (Eq, Show)

  176 type EnrollmentAuthority = GrantSource

  178 -- | Why enrollment authority was refused. The constructors select the HTTP
  179 -- status, the wire code and the UI branch, which is why this is an ADT and not
  180 -- @Either Text@.
  181 data GrantDenied
  182   = GrantRequired
  183     -- ^ No authority was presented and none may be inherited.
  184   | StepUpRequired
  185     -- ^ A session exists but presented no token: prove the device, then retry.
  186   | GrantInvalid
  187     -- ^ A token was presented and does not match a live grant.
  188   deriving stock (Eq, Show)
  189 
  190 -- | Resolve the authority a request holds (design §7.2).
  191 --
  192 -- Order: a presented token is matched against the file grant's digest, then the
  193 -- session grant's digest — the latter only if the requesting session is the one
  194 -- the grant was minted for.
  195 --
  196 -- __The load-bearing rule__ (gate round 4, B-1): /authority is presented, not
  197 -- inherited/. An absent token is acceptable only for 'CliWindow', which by
  198 -- construction carries no secret. Rev 4 let 'Nothing' fall through to a
  199 -- session-held grant, so a stolen cookie could redeem the victim's live
  200 -- assertion grant merely by omitting the header. There is now no path from a
  201 -- cookie alone to enrollment authority.
  202 --
  203 -- This is a pure read: a mismatch never rebinds, consumes or extends anything,
  204 -- so a stale token cannot shadow a live grant and cannot be used to probe the
  205 -- grant's state.
  206 findAuthority
  207   :: Maybe Text        -- ^ presented token (@X-Seal-Enroll-Token@)
  208   -> Maybe Grant       -- ^ the file-backed grant
  209   -> Maybe Grant       -- ^ the grant carried by the requesting session
  210   -> Bool              -- ^ does the requesting session match that session grant?
  211   -> Either GrantDenied Grant
  212 findAuthority presented fileGrant sessionGrant sessionMatches =
  213   case presented of
  214     -- A token was presented, so it must resolve. Nothing is inherited.
  215     Just raw ->
  216       case matchesToken raw fileGrant of
  217         Just grant -> Right grant
  218         -- The same token is tried against the session grant, and only when the
  219         -- requesting session is the one it was minted for. A token that matches
  220         -- neither is GrantInvalid: it must not inherit the session grant, which
  221         -- would make a wrong token as good as an absent one.
  222         Nothing -> case matchesToken raw (if sessionMatches then sessionGrant else Nothing) of
  223           Just grant -> Right grant
  224           Nothing    -> Left GrantInvalid
  225     Nothing -> case fileGrant of
  226       Just grant@Grant { grSource = CliWindow } -> Right grant
  227       _ -> Left (if sessionMatches && isJust sessionGrant
  228                    then StepUpRequired
  229                    else GrantRequired)
  230   where
  231     matchesToken raw mGrant = mGrant >>= \grant ->
  232       case grTokenSha grant of
  233         Just digest | not (BS.null digest)
  234                     , constantTimeEq digest (tokenDigest raw) -> Just grant
  235         _ -> Nothing
  236 
  237 -- | SHA-256 of a token, as stored. The token itself is never persisted.
  238 tokenDigest :: Text -> ByteString
  239 tokenDigest = sha256Hash . TE.encodeUtf8

  245 grantToJson :: Grant -> Value
  246 grantToJson grant = object
  247   [ "grant_id" .= grantIdText (grId grant)
  248   , "minted_via" .= grantSourceText (grSource grant)
  249   , "expires_at" .= isoTime (grExpiresAt grant)
  250   , "bound_challenge" .= boundChallenge
  251   , "token_sha256" .= fmap base64Text (grTokenSha grant)
  252   ]
  253   where
  254     boundChallenge = case grPhase grant of
  255       GrantBound cid -> Just (challengeIdText cid)
  256       GrantMinted    -> Nothing

  258 -- | Decode a stored grant.
  259 --
  260 -- Deliberately total in the permissive direction: an unreadable field is
  261 -- treated as /absent/, never as a boot refusal. Absent is never more
  262 -- permissive than present (design §7.3) — the caller deletes a grant it cannot
  263 -- read, and a missing grant is just "no authority".
  264 grantFromJson :: Value -> Maybe Grant
  265 grantFromJson = parseMaybe parseGrant
  266 
  267 parseGrant :: Value -> Parser Grant
  268 parseGrant = withObject "Grant" $ \o -> do
  269   ident@(GrantId rawId) <- GrantId <$> o .: "grant_id"
  270   -- The id names the claim file, so a separator in it would move the file.
  271   -- Grant ids are minted as hex, so this only rejects a hand-edited or
  272   -- corrupted document.
  273   when (any (`elem` ("/\\" :: String)) (T.unpack rawId)
  274           || ".." `T.isInfixOf` rawId)
  275     (fail "grant_id must not contain a path separator")
  276   source <- o .: "minted_via" >>= requireSource
  277   expires <- o .: "expires_at" >>= requireTime
  278   phase <- maybe GrantMinted (GrantBound . ChallengeId) <$> o .:? "bound_challenge"
  279   digest <- (o .:? "token_sha256") >>= traverse requireDigest
  280   -- CliWindow carries no secret; every other source must.
  281   case (source, digest) of
  282     (CliWindow, _)     -> pure (Grant ident expires source phase Nothing)
  283     (_, Just d)        -> pure (Grant ident expires source phase (Just d))
  284     (_, Nothing)       -> fail "a token-backed grant must carry token_sha256"
  285   where
  286     requireSource raw = case parseGrantSource raw of
  287       Just source -> pure source
  288       Nothing     -> fail ("unknown minted_via: " <> T.unpack raw)
  289     requireTime raw = case parseIsoTimeMaybe raw of
  290       Just t  -> pure t
  291       Nothing -> fail ("unparseable expires_at: " <> T.unpack raw)
  292     -- The digest is stored base64url, because JSON has no byte string.
  293     requireDigest raw = case B64Url.decodeUnpadded (TE.encodeUtf8 raw) of
  294       Left e   -> fail ("token_sha256 is not base64url: " <> e)
  295       Right bs -> pure bs
  296 
  297 -- | The grant as one line of JSON, for the state file.
  298 grantBody :: Grant -> LBS.ByteString
  299 grantBody = encode . grantToJson

  301 grantBodyText :: Grant -> Text
  302 grantBodyText = TE.decodeUtf8 . LBS.toStrict . grantBody

  328 -- | Everything the enrollment decision depends on.
  329 --
  330 -- There is no clock here on purpose. A grant's expiry is decided when it is
  331 -- read ('findAuthority' returns a grant that is already known live, and the
  332 -- claim protocol re-checks the deadline), so this function cannot decide it and
  333 -- a @gcNow@ field would suggest otherwise. Design §7.2 lists one; it is
  334 -- deliberately omitted rather than carried unused.
  335 data GrantContext = GrantContext
  336   { gcMode         :: !AccessMode
  337   , gcAvailability :: !CredentialAvailability
  338   , gcAuthorized   :: !(Maybe EnrollmentAuthority)
  339     -- ^ What 'findAuthority' resolved. 'Nothing' means no authority was
  340     -- presented, which is a state this function must still answer for: the
  341     -- status endpoint asks it without a request.
  342   , gcHasSession   :: !Bool
  343   } deriving stock (Eq, Show)
  344 
  345 -- | Which authorities are acceptable in this boot state.
  346 --
  347 -- Total over 3 modes x 3 availabilities x 5 authorities. Two rules do most of
  348 -- the work:
  349 --
  350 -- * __The loopback window exists only with nothing enrolled.__ It is authority
  351 --   that presents no secret, so allowing it once a credential exists — or in
  352 --   @UnusableOnly@, which is the state an attacker who can change
  353 --   @origin@\\/@rp_id@ aims to create — would hand enrollment to anyone who
  354 --   can reach the port. That is gate round 4's B-2, and the reason
  355 --   @cli_window@ outside @NoCredentials@ is @GrantRequired@ rather than
  356 --   @GrantInvalid@: the token is well-formed, the state no longer permits it.
  357 -- * __Enrolling always needs authority, even in @open@ mode.__ A public gateway
  358 --   that let a network client enroll would be flipped from open to locked with
  359 --   a key the operator does not hold.
  360 enrollmentAuthority :: GrantContext -> Either GrantDenied EnrollmentAuthority
  361 enrollmentAuthority ctx
  362   -- Bootstrap. Nothing is enrolled, so a token (or the confined window) is the
  363   -- only thing that can authorize; an assertion cannot exist yet.
  364   | gcAvailability ctx == NoCredentials = case gcAuthorized ctx of
  365       Just CliToken  -> Right CliToken
  366       Just CliWindow -> Right CliWindow
  367       _              -> Left GrantRequired
  368   | otherwise = case (gcMode ctx, gcAuthorized ctx) of
  369       -- Open mode is fully open to *use*, not to enroll.
  370       (AccessOpen, Just CliToken) -> Right CliToken
  371       (AccessOpen, Just UiToken)  -> Right UiToken
  372       (AccessOpen, _)             -> Left GrantRequired
  373       -- passkey + UnusableOnly is rp_mismatch: token only, never the window.
  374       (AccessPasskey, Just CliToken) -> Right CliToken
  375       -- A browser-minted token is a peer of a terminal-minted one everywhere a
  376       -- token is accepted, *except* bootstrap (@NoCredentials@, above) and
  377       -- @unconfigured@ (below): those slots are reserved for authority minted
  378       -- with local access, because they are what stops a reachable network
  379       -- client enrolling a key the operator does not hold. Placed after the
  380       -- CliToken clause so @UnusableOnly@ accepts it too, matching CliToken.
  381       (AccessPasskey, Just UiToken) -> Right UiToken
  382       (AccessPasskey, _) | gcAvailability ctx == UnusableOnly -> Left GrantRequired
  383       -- passkey + Usable: an existing credential's assertion can add another,
  384       -- and a token still works.
  385       (AccessPasskey, Just Assertion) -> Right Assertion
  386       -- Only "no authority presented at all" can ask for a step-up. A window
  387       -- that arrives in the wrong state is well-formed but not permitted, so it
  388       -- is GrantRequired: telling that client to prove its device would be a
  389       -- dead end.
  390       (AccessPasskey, Nothing)
  391         | gcHasSession ctx -> Left StepUpRequired
  392         | otherwise        -> Left GrantRequired
  393       (AccessPasskey, Just _) -> Left GrantRequired
  394       -- Unconfigured with credentials present is not a state boot produces
  395       -- (it resolves to passkey or open); answering it keeps the function total
  396       -- without granting anything.
  397       (AccessUnconfigured, Just CliToken) -> Right CliToken
  398       (AccessUnconfigured, _)             -> Left GrantRequired

  404 -- | Is the grant still within its lifetime?
  405 --
  406 -- Expiry is an upper bound, never a quota: a grant that has expired is
  407 -- indistinguishable from no grant at all, which is why every reader treats it
  408 -- that way instead of refusing.
  409 grantLiveAt :: UTCTime -> Grant -> Bool
  410 grantLiveAt now grant = now < grExpiresAt grant
  411 
  412 -- | Read the outstanding grant.
  413 --
  414 -- Three states collapse to @Nothing@, deliberately:
  415 --
  416 -- * absent — the normal state;
  417 -- * unparseable — treated as absent /and deleted/, because a file the gateway
  418 --   cannot read must never be more permissive than no file (design §7.3);
  419 -- * expired — treated as absent, but left on disk for the reaper. Reaping on a
  420 --   polled read would turn @GET \/status@ into a writer.
  421 --
  422 -- Never cached: the CLI mints grants while the daemon runs, so a cached copy
  423 -- would be invisible until restart.
  424 loadGrantFile :: FilePath -> IO (Maybe Grant)
  425 loadGrantFile path = do
  426   exists <- doesFileExist path
  427   if not exists
  428     then pure Nothing
  429     else do
  430       raw <- LBS.readFile path
  431       case eitherDecode raw of
  432         Right value -> case grantFromJson value of
  433           Just grant -> pure (Just grant)
  434           Nothing    -> reap path
  435         Left _ -> reap path
  436   where
  437     reap p = removeFile p >> pure Nothing
  438 
  439 -- | Write the grant atomically, mode @0600@.
  440 --
  441 -- A unique temp name and an atomic rename are the only cross-process
  442 -- guarantees available here: the write lock is process-local, and the CLI writes
  443 -- this file while the daemon reads it.
  444 saveGrantFile :: FilePath -> Grant -> IO (Either Text ())
  445 saveGrantFile path grant = do
  446   let dir = takeDirectory path
  447   (tmp, h) <- openBinaryTempFile dir (takeBaseName path <> ".tmp")
  448   LBS.hPut h (grantBody grant)
  449   hClose h
  450   setFileMode tmp 0o600
  451   renameFile tmp path
  452   pure (Right ())
  453 
  454 deleteGrantFile :: FilePath -> IO ()
  455 deleteGrantFile path = do
  456   exists <- doesFileExist path
  457   when exists (removeFile path)

  463 -- | The one deadline rule: a claim expires at whichever comes first, the
  464 -- grant's own expiry or the challenge's TTL from now.
  465 --
  466 -- Both bounds are needed. Only the challenge TTL would let a claim outlive a
  467 -- grant with a shorter TTL; only the grant expiry would let a ceremony keep a
  468 -- bound claim alive for the grant's whole lifetime after its prompt was
  469 -- abandoned.
  470 effectiveDeadline :: Grant -> UTCTime -> NominalDiffTime -> UTCTime
  471 effectiveDeadline grant now challengeTtl =
  472   min (grExpiresAt grant) (addUTCTime challengeTtl now)
  473 
  474 -- | Is the claim still within its deadline?
  475 --
  476 -- @now == deadline@ is expired: the deadline is the first instant at which the
  477 -- claim may no longer be used.
  478 claimLiveAt :: UTCTime -> GrantClaim -> Bool
  479 claimLiveAt now claim = now < clDeadline claim
  480 
  481 -- | May this grant redeem @challenge@?
  482 --
  483 -- Only a grant that @options@ bound to this exact challenge can. A grant still
  484 -- in 'GrantMinted' means the client skipped @options@, and a grant bound
  485 -- elsewhere is a race or a forgery — both refuse, and neither silently rebinds.
  486 grantRedeems :: ChallengeId -> Grant -> Bool
  487 grantRedeems challenge grant = grPhase grant == GrantBound challenge
  488 
  489 -- | Take the exclusive claim for a grant, or lose the race.
  490 --
  491 -- @True@ means this caller holds it. The file is created @O_CREAT|O_EXCL@ and
  492 -- written directly: there is no temp-and-rename here, because the atomicity
  493 -- this needs is over /existence/, not over content. A crash mid-write leaves a
  494 -- claim the reader cannot parse, which is treated as no claim — the next
  495 -- @options@ rebinds.
  496 claimExclusively :: FilePath -> GrantClaim -> IO Bool
  497 claimExclusively path claim = do
  498   opened <- try (openFd path WriteOnly defaultFileFlags
  499                    { creat = Just 0o600, exclusive = True })
  500   case opened of
  501     Left (_ :: IOException) -> pure False
  502     Right fd -> do
  503       h <- fdToHandle fd
  504       LBS.hPut h (claimBody claim)
  505       hClose h
  506       setFileMode path 0o600
  507       pure True
  508 
  509 -- | Give the claim back, but only if it is still /this/ ceremony's claim.
  510 --
  511 -- Compare-and-delete: a failed verify must not delete a claim that a newer
  512 -- @options@ has already replaced, or the retry would find the grant claimed by
  513 -- nobody and the loser of that race would win the next one.
  514 releaseClaim :: FilePath -> ChallengeId -> IO ()
  515 releaseClaim path challenge = do
  516   held <- loadClaimFile path
  517   case held of
  518     Just claim | clChallengeId claim == challenge -> deleteGrantFile path
  519     _ -> pure ()
  520 
  521 -- | Bind a claim, superseding one left behind by an abandoned ceremony.
  522 --
  523 -- The ordinary case is 'claimExclusively'. The exception is a claim nobody
  524 -- finished: the operator dismissed the OS passkey prompt, or the tab died, so
  525 -- no @verify@ ever arrives and the marker sits there for the whole challenge
  526 -- window. Refusing every retry in that window makes the first accidental
  527 -- dismissal burn the link — and since the SPA treats @grant_invalid@ as "this
  528 -- token is spent", it also throws the token away.
  529 --
  530 -- A fresh bind therefore wins: last bind wins. Exactly-once does not depend on
  531 -- this function. It is carried one level up by the single-use challenge
  532 -- (consumed before verification) and by 'grantRedeems', which refuses a grant
  533 -- bound to a different challenge — so of two ceremonies only the one the grant
  534 -- is currently bound to can complete. A third party that also holds the link
  535 -- can make the operator's ceremony lose, which is a denial, not an escalation:
  536 -- it already has the link.
  537 bindClaimSuperseding :: FilePath -> GrantClaim -> IO Bool
  538 bindClaimSuperseding path claim = do
  539   won <- claimExclusively path claim
  540   if won
  541     then pure True
  542     else do
  543       deleteGrantFile path
  544       claimExclusively path claim
  545 
  546 -- | Read a claim. Unparseable or absent both mean "nobody holds it".
  547 loadClaimFile :: FilePath -> IO (Maybe GrantClaim)
  548 loadClaimFile path = do
  549   exists <- doesFileExist path
  550   if not exists
  551     then pure Nothing
  552     else do
  553       raw <- LBS.readFile path
  554       pure (either (const Nothing) claimFromJson (eitherDecode raw))

  564 claimFromJson = parseMaybe $ withObject "GrantClaim" $ \o -> do

  566   challenge <- ChallengeId <$> o .: "challenge_id"
  567   deadline  <- o .: "deadline" >>= requireTime
  568   pure (GrantClaim ident challenge deadline)
  569   where
  570     requireTime raw = case parseIsoTimeMaybe raw of
  571       Just t  -> pure t
  572       Nothing -> fail ("unparseable deadline: " <> T.unpack raw)

  576 --------------------------------------------------------------------------------

  578 -- | A grant that was just written to @state/@, plus the plaintext token the
  579 -- client must present.
  580 --
  581 -- The token is the only copy: the file stores its digest. Losing it means
  582 -- @--force@ (or @--cancel@), which is why the command prints it to stdout.
  583 -- 'Nothing' for a 'CliWindow' grant, which presents no secret at all.
  584 data MintedGrant = MintedGrant
  585   { mgGrant :: !Grant
  586   , mgToken :: !(Maybe Text)
  587   } deriving stock (Eq, Show)
  588 
  589 -- | The default lifetime of a grant minted from the terminal (design §7.1).
  590 defaultGrantTtl :: GrantSource -> NominalDiffTime
  591 defaultGrantTtl CliWindow = 5 * 60
  592 -- Five minutes, not ten: a link minted in a browser is exposed in the DOM, in
  593 -- clipboard managers and cloud clipboard sync, and on a screen share, which a
  594 -- token printed to a terminal is not. Nothing in the flow needs longer.
  595 defaultGrantTtl UiToken   = 5 * 60
  596 defaultGrantTtl _         = 10 * 60
  597 
  598 -- | Mint a grant, unless one is already outstanding.
  599 --
  600 -- Refusing rather than overwriting is what makes a lost token recoverable: the
  601 -- refusal names the outstanding grant's source and remaining lifetime, and
  602 -- @--force@ is the deliberate way to replace it. Minting also reaps every stale
  603 -- claim marker and the previous token file, because a new grant invalidates all
  604 -- of them.
  605 enrollGrant
  606   :: SealPaths
  607   -> GrantSource
  608   -> NominalDiffTime
  609   -> Bool              -- ^ force: replace an outstanding grant
  610   -> IO (Either Text MintedGrant)
  611 enrollGrant paths source ttl force = do
  612   now <- getCurrentTime
  613   -- The grant lives in state/, which boot creates; a CLI invocation may be the
  614   -- first thing to touch it (a fresh SEAL_HOME, or state/ removed by hand).
  615   createDirectoryIfMissing True (spState paths)
  616   existing <- loadGrantFile (enrollmentGrantPath paths)
  617   case existing of
  618     Just grant | grantLiveAt now grant, not force ->
  619       pure (Left (outstandingRefusal now grant))
  620     _ -> do
  621       token <- case source of
  622         CliWindow -> pure Nothing
  623         _         -> Just <$> generateToken 32
  624       ident <- GrantId <$> generateToken 16
  625       let grant = Grant
  626             { grId        = ident
  627             , grExpiresAt = addUTCTime ttl now
  628             , grSource    = source
  629             , grPhase     = GrantMinted
  630             , grTokenSha  = tokenDigest <$> token
  631             }
  632       reapClaims paths
  633       reapTokenFile paths
  634       saved <- saveGrantFile (enrollmentGrantPath paths) grant
  635       pure $ case saved of
  636         Left err  -> Left err
  637         Right ()  -> Right (MintedGrant grant token)
  638 
  639 -- | Drop every trace of the outstanding grant: the grant file, every claim
  640 -- marker and the plaintext token.
  641 --
  642 -- Idempotent and silent when nothing is outstanding — @--cancel@ with nothing
  643 -- to cancel is a success, not an error.
  644 cancelGrant :: SealPaths -> IO ()
  645 cancelGrant paths = do
  646   deleteGrantFile (enrollmentGrantPath paths)
  647   reapClaims paths
  648   reapTokenFile paths
  649 
  650 -- | Delete every claim marker. A new grant (or a cancel) invalidates all of
  651 -- them: they name grants that no longer exist.
  652 reapClaims :: SealPaths -> IO ()
  653 reapClaims paths = do
  654   -- A fresh SEAL_HOME has no state/ yet; nothing to reap is not an error.
  655   present <- doesDirectoryExist (spState paths)
  656   when present $ do
  657     entries <- listDirectory (spState paths)
  658     forM_ entries $ \entry -> when (claimPrefix `T.isPrefixOf` T.pack entry) $
  659       removeFile (spState paths </> entry)
  660   where
  661     claimPrefix = "enrollment-grant.claimed-"
  662 
  663 reapTokenFile :: SealPaths -> IO ()
  664 reapTokenFile paths = do
  665   let path = enrollmentTokenPath paths
  666   exists <- doesFileExist path
  667   when exists (removeFile path)
  668 
  669 -- | How long the grant has left. Negative once expired.
  670 grantRemaining :: UTCTime -> Grant -> NominalDiffTime
  671 grantRemaining now grant = diffUTCTime (grExpiresAt grant) now
  672 
  673 outstandingRefusal :: UTCTime -> Grant -> Text
  674 outstandingRefusal now grant =
  675   "an enrollment grant is already outstanding (minted via "
  676     <> grantSourceText (grSource grant) <> ", "
  677     <> tshowSeconds (grantRemaining now grant) <> " left).\n"
  678     <> "  Use the token you were given, or re-run with --force to replace it,\n"
  679     <> "  or run `seal access-control enroll --cancel` to drop it."
  680 
  681 tshowSeconds :: NominalDiffTime -> Text
  682 tshowSeconds delta =
  683   T.pack (show (max 0 (ceiling (realToFrac delta :: Double) :: Int))) <> "s"
  684 
  685 -- | Parse a @--ttl@ value: @\<N\>\<s|m|h\>@.
  686 --
  687 -- A bare number is a usage error, not a default unit: "10" meaning ten minutes
  688 -- and "10" meaning ten seconds are both plausible, so guessing is how an
  689 -- operator ends up with a window that closed before they used it. The result is
  690 -- clamped to 30 s – 24 h, the range the design allows.
  691 parseTtl :: Text -> Either Text NominalDiffTime
  692 parseTtl raw =
  693   case T.unsnoc (T.strip raw) of
  694     Just (digits, unit) | not (T.null digits), T.all isAsciiDigit digits ->
  695       let multiplier = case unit of
  696             's' -> 1
  697             'm' -> 60
  698             'h' -> 3600
  699             _   -> 0
  700       in if multiplier == 0
  701            then Left (badTtl raw)
  702            else Right (fromIntegral (max 30 (min 86400 (digitsToSeconds digits * multiplier))))
  703     _ -> Left (badTtl raw)
  704   where
  705     badTtl value =
  706       "--ttl \"" <> value <> "\" is not a duration.\n"
  707         <> "  Use a number followed by s, m or h — for example 90s, 10m or 1h."
  708 
  709 isAsciiDigit :: Char -> Bool
  710 isAsciiDigit = isDigit
  711 
  712 digitsToSeconds :: Text -> Integer
  713 digitsToSeconds = T.foldl' (\acc c -> acc * 10 + toInteger (fromEnum c - fromEnum '0')) 0
  714 
  715 -- | The lifetime the command should use when @--ttl@ is absent.
  716 defaultTtlFor :: GrantSource -> NominalDiffTime
  717 defaultTtlFor = defaultGrantTtl

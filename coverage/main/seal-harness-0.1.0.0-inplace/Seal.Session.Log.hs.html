    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Best-effort per-session diagnostic log (@\<sessionDir\>\/seal.log@).
    3 --
    4 -- Records turn lifecycle events (start/end/duration) and failures (exceptions,
    5 -- provider errors) that are NOT already captured in @entries.jsonl@ or
    6 -- @conversation.jsonl@. The transcript files record what the model + opcodes
    7 -- did; this log records what the /harness/ did — where it started, where it
    8 -- stopped, and why it failed when it failed.
    9 --
   10 -- Design constraints:
   11 --
   12 -- 1. /Best-effort, never throws./ A write error (full disk, permissions,
   13 --    missing parent dir) is swallowed — the log must never break the agent
   14 --    loop. The worst case is a missing log line, not a crashed session.
   15 -- 2. /Append-only, one line per event./ Each line carries an ISO-8601
   16 --    timestamp prefix + a level tag + the message. Human-readable,
   17 --    grep-friendly.
   18 -- 3. /No duplication of transcript data./ Message content, tool-call inputs,
   19 --    and opcode results are already in @conversation.jsonl@ / @entries.jsonl@.
   20 --    This log records only lifecycle boundaries + failure diagnostics.
   21 module Seal.Session.Log
   22   ( appendSessionLog
   23   , logTurnStart
   24   , logTurnEnd
   25   , logTurnError
   26   , logProviderError
   27   , logMaxTurns
   28   , logTruncation
   29   , logTruncationGiveUp
   30   , logModelResponse
   31   , logStreamResult
   32   , logToolDispatch
   33   , logDegenerateResponse
   34   , logRetryAttempt
   35   , logRetryExhausted
   36   , maxLengthContinueRetries
   37   ) where
   38 
   39 import Control.Exception (catch, SomeException)
   40 import Data.Text (Text)
   41 import Data.Text qualified as T
   42 import Data.Time (UTCTime, getCurrentTime, formatTime, defaultTimeLocale)
   43 import System.IO (withFile, IOMode (AppendMode), hPutStr)
   44 import Seal.Providers.Class (StopReason (..), ContentBlock (..))
   45 import Seal.Core.Types (OpName (..))

   47 -- | Maximum number of auto-continuation retries when the provider returns
   48 -- 'StopMaxTokens' with no tool calls (text was truncated mid-generation).
   49 -- After this many retries the loop surfaces a user-visible truncation
   50 -- notice and stops. Mirrors Hermes' @length_continue_retries < 3@ cap.
   51 maxLengthContinueRetries :: Int
   52 maxLengthContinueRetries = 3
   53 
   54 -- | Append a single line to the session log. Best-effort: any IO error is
   55 -- swallowed. The line is prefixed with an ISO-8601 timestamp and the given
   56 -- level tag. A 'Nothing' path is a no-op (logging disabled).
   57 appendSessionLog :: Maybe FilePath -> Text -> Text -> IO ()
   58 appendSessionLog Nothing       _     _ = pure ()
   59 appendSessionLog (Just path) level msg = do
   60   now <- getCurrentTime
   61   let line = T.unpack (formatLogLine now level msg)
   62   withFile path AppendMode (`hPutStr` (line <> "\n"))
   63     `catch` \(_ :: SomeException) -> pure ()
   64 
   65 -- | Format a log line: @\<ISO-8601 ts\> [LEVEL] message@
   66 formatLogLine :: UTCTime -> Text -> Text -> Text
   67 formatLogLine ts level msg =
   68   T.pack (formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%S%3QZ" ts)
   69   <> " [" <> level <> "] " <> msg
   70 
   71 -- | Log a turn start. @turnN@ is the 0-based remaining-turns counter from
   72 -- 'runTurn' (the first turn is @aeMaxTurns@).
   73 logTurnStart :: Maybe FilePath -> Int -> IO ()
   74 logTurnStart mPath turnN =
   75   appendSessionLog mPath "TURN" ("start (turns remaining: " <> T.pack (show turnN) <> ")")
   76 
   77 -- | Log a turn end with a duration in milliseconds.
   78 logTurnEnd :: Maybe FilePath -> Int -> Integer -> IO ()
   79 logTurnEnd mPath turnN ms =
   80   appendSessionLog mPath "TURN" ("end (turns remaining: " <> T.pack (show turnN)
   81     <> ", " <> T.pack (show ms) <> "ms)")
   82 
   83 -- | Log a turn failure (caught exception). The message is the exception text.
   84 logTurnError :: Maybe FilePath -> Text -> IO ()
   85 logTurnError mPath err =
   86   appendSessionLog mPath "ERROR" ("turn failed: " <> err)
   87 
   88 -- | Log a provider error (the 'Left' branch of 'providerComplete').
   89 logProviderError :: Maybe FilePath -> Text -> IO ()
   90 logProviderError mPath err =
   91   appendSessionLog mPath "ERROR" ("provider error: " <> err)
   92 
   93 -- | Log the max-turns stop (the loop hit @aeMaxTurns@ without a final text
   94 -- response).
   95 logMaxTurns :: Maybe FilePath -> IO ()
   96 logMaxTurns mPath =
   97   appendSessionLog mPath "WARN" "stopped: too many tool turns (hit aeMaxTurns)"
   98 
   99 -- | Log a @StopMaxTokens@ truncation that triggered an auto-continuation
  100 -- retry. @attempt@ is 1-based (1 = first retry requested). Mirrors Hermes'
  101 -- @↻ Requesting continuation (N/3)@ log line.
  102 logTruncation :: Maybe FilePath -> Int -> IO ()
  103 logTruncation mPath attempt =
  104   appendSessionLog mPath "WARN"
  105     ("response truncated (StopMaxTokens) - requesting continuation ("
  106       <> T.pack (show attempt) <> "/" <> T.pack (show maxLengthContinueRetries) <> ")")
  107 
  108 -- | Log a @StopMaxTokens@ truncation after all continuation retries are
  109 -- exhausted. The loop surfaces a user-visible truncation notice and stops.
  110 logTruncationGiveUp :: Maybe FilePath -> IO ()
  111 logTruncationGiveUp mPath =
  112   appendSessionLog mPath "WARN"
  113     ("stopped: response remained truncated after "
  114       <> T.pack (show maxLengthContinueRetries)
  115       <> " continuation attempts")

  119 -- | Log a summary of the model response after aggregation. Records metadata
  120 -- (stop reason, content block count, tool call count, text length) — NOT the
  121 -- actual text content (which is in conversation.jsonl).
  122 logModelResponse :: Maybe FilePath -> StopReason -> [ContentBlock] -> IO ()
  123 logModelResponse mPath stopReason blocks =
  124   appendSessionLog mPath "DEBUG"
  125     ("model response: stop=" <> stopReasonText stopReason
  126       <> " blocks=" <> T.pack (show (length blocks))
  127       <> " toolCalls=" <> T.pack (show (length [ () | CbToolUse{} <- blocks ]))
  128       <> " textLen=" <> T.pack (show (sum [ T.length t | CbText t <- blocks ]))
  129       <> " thinkingLen=" <> T.pack (show (sum [ T.length t | CbThinking t <- blocks ])))
  130 
  131 -- | Log the result of the stream race (success, timeout, abort, or error).
  132 logStreamResult :: Maybe FilePath -> Either Text a -> IO ()
  133 logStreamResult mPath result =
  134   appendSessionLog mPath "DEBUG" $
  135     case result of
  136       Right _ -> "stream result: success"
  137       Left err
  138         | err == "(stopped)" -> "stream result: aborted (user stop)"
  139         | "stream timed out" `T.isInfixOf` err -> "stream result: timeout"
  140         | otherwise -> "stream result: error (" <> err <> ")"
  141 
  142 -- | Log a tool dispatch outcome. Records tool name, success/error, and
  143 -- result size — NOT the actual result content.
  144 logToolDispatch :: Maybe FilePath -> OpName -> Bool -> Int -> IO ()
  145 logToolDispatch mPath (OpName name) isError resultSize =
  146   appendSessionLog mPath "DEBUG"
  147     ("tool dispatch: " <> name
  148       <> " result=" <> (if isError then "error" else "ok")
  149       <> " parts=" <> T.pack (show resultSize))
  150 
  151 -- | Log a degenerate response warning (no tool calls, no text, stop ≠ MaxTokens).
  152 logDegenerateResponse :: Maybe FilePath -> StopReason -> [ContentBlock] -> IO ()
  153 logDegenerateResponse mPath stopReason blocks =
  154   appendSessionLog mPath "WARN"
  155     ("degenerate response: stop=" <> stopReasonText stopReason
  156       <> " blocks=" <> T.pack (show (length blocks))
  157       <> " — model returned no usable content (no tool calls, no text)")
  158 
  159 -- | Log a retry attempt before it fires. Records attempt number, max,
  160 -- reason, and backoff delay.
  161 logRetryAttempt :: Maybe FilePath -> Int -> Int -> Text -> Int -> IO ()
  162 logRetryAttempt mPath attempt maxRetries reason delayMs =
  163   appendSessionLog mPath "INFO"
  164     ("retry attempt " <> T.pack (show attempt) <> "/" <> T.pack (show maxRetries)
  165       <> ": reason=" <> reason
  166       <> " backoff=" <> T.pack (show delayMs) <> "ms")
  167 
  168 -- | Log retry exhaustion after all attempts fail.
  169 logRetryExhausted :: Maybe FilePath -> Int -> Text -> IO ()
  170 logRetryExhausted mPath maxRetries reason =
  171   appendSessionLog mPath "ERROR"
  172     ("retry exhausted after " <> T.pack (show maxRetries)
  173       <> " attempts: " <> reason
  174       <> " — see seal.log for full diagnostic details")
  175 
  176 -- | Render a 'StopReason' as short text for log lines.
  177 stopReasonText :: StopReason -> Text
  178 stopReasonText StopEnd        = "StopEnd"
  179 stopReasonText StopToolUse    = "StopToolUse"
  180 stopReasonText StopMaxTokens  = "StopMaxTokens"
  181 stopReasonText (StopOther t)  = "StopOther:" <> t

   27 -- | The known harness flavours + a smart-constructed 'HCustom' that rejects
   28 -- path separators (a custom-tool launch command must not smuggle a path).
   29 data HarnessFlavour
   30   = HfClaudeCode
   31   | HfCodex
   32   | HfGeneric
   33   | HCustom Text
   34   deriving stock (Eq, Show, Generic)
   35 
   36 -- | Smart-construct an 'HCustom' flavour: non-empty, no path separators
   37 -- (@/@ @\\@ — argv injection defense for a custom-tool launch command),
   38 -- no control chars, no leading dash.
   39 mkHCustom :: Text -> Either Text HarnessFlavour
   40 mkHCustom t
   41   | T.null t              = Left "custom harness flavour is empty"
   42   | T.head t == '-'       = Left "custom harness flavour must not start with '-'"
   43   | T.any isPathSep t     = Left "custom harness flavour must not contain path separators"
   44   | not (T.all validChar t) = Left "custom harness flavour has invalid characters"
   45   | otherwise             = Right (HCustom t)
   46   where
   47     isPathSep c = c == '/' || c == '\\'
   48     validChar c = c `Set.member` flavourChars
   49     flavourChars = Set.fromList
   50       $ ['A'..'Z'] <> ['a'..'z'] <> ['0'..'9'] <> "_.-+:@"
   51 
   52 -- | A harness spec: the flavour + the tmux coordinates + the launch args.
   53 data HarnessSpec = HarnessSpec
   54   { hsFlavour     :: HarnessFlavour
   55   , hsTmuxSession :: TmuxIdent          -- ^ the tmux session the harness lives in
   56   , hsCwd         :: Maybe FilePath
   57   , hsArgs        :: [Text]
   58   , hsDurableId   :: Maybe HarnessId    -- ^ adopt an existing id (re-attach)
   59   } deriving stock (Eq, Show)
   60 
   61 -- | A provider-backed chat session spec (provider label + model).
   62 data ProviderSpec = ProviderSpec
   63   { psProvider :: Text
   64   , psModel    :: Text
   65   } deriving stock (Eq, Show)
   66 
   67 -- | The session-kind enumeration.
   68 data SessionKind
   69   = SkProvider ProviderSpec
   70   | SkHarness HarnessSpec
   71   deriving stock (Eq, Show)

   73 -- | Infer a 'ProviderId' from a model-prefix heuristic (e.g.
   74 -- @claude-…@ -> @anthropic@). A minimal table; the real resolution lives
   75 -- in 'Seal.Providers.Registry'.
   76 inferProviderId :: Text -> Maybe ProviderId
   77 inferProviderId model
   78   | "claude"  `T.isPrefixOf` model = Just (ProviderId "anthropic")
   79   | "gpt"     `T.isPrefixOf` model = Just (ProviderId "openai")
   80   | "llama"   `T.isPrefixOf` model = Just (ProviderId "ollama")
   81   | "gemini"  `T.isPrefixOf` model = Just (ProviderId "google")
   82   | otherwise                      = Nothing

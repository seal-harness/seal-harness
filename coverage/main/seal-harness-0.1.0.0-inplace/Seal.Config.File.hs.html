   78 -- | The agent/operator-tunable runtime configuration persisted in
   79 -- @config\/config.toml@. Every field is optional; a missing key decodes as
   80 -- 'Nothing'. Security-critical boot-only fields (vault, untrusted_execution)
   81 -- live in 'Seal.Config.Security.SecurityConfig' (loaded from
   82 -- @security.toml@) and are NOT present here.
   83 data RuntimeConfig = RuntimeConfig
   84   { rcDefaultProvider :: Maybe Text
   85     -- ^ Provider id used for new sessions (e.g. @\"anthropic\"@).
   86   , rcDefaultModel :: Maybe Text
   87     -- ^ Model id used for new sessions (e.g. @\"claude-opus-4-8\"@).
   88   , rcDefaultAgent :: Maybe Text
   89     -- ^ Agent def id used by default for agent-driven flows (set via
   90     -- @\/agent default@). 'Nothing' means no default agent is selected.
   91   , rcProviders :: Map Text ProviderConfig
   92     -- ^ Per-provider config sections (@[providers.<label>]@).
   93   , rcRetrieval :: Maybe RetrievalConfig
   94     -- ^ Optional @[retrieval]@ section (Dynamic Retrieval tuning). Absent
   95     -- means 'defaultRetrievalConfig' applies at resolution time.
   96   , rcSignal :: Maybe SignalConfig
   97     -- ^ Optional @[signal]@ section (Signal channel config). Absent means
   98     -- the Signal channel is not configured.
   99   , rcTelegram :: Maybe TelegramConfig
  100     -- ^ Optional @[telegram]@ section (Telegram channel config). Absent
  101     -- means the Telegram channel is not configured.
  102   , rcGateway :: Maybe PartialGatewayConfig
  103     -- ^ Optional @[gateway]@ section (web gateway config). Absent means the
  104     -- gateway is not configured. Each field inside is optional too; the
  105     -- call site merges with 'Seal.Gateway.Config.withGatewayDefaults'.
  106   , rcDebugSessionTranscript :: Maybe Bool
  107     -- ^ Optional @debug_session_transcript@ flag. When @true@, every
  108     -- 'CompletionRequest' sent to the LLM is also appended (redundantly,
  109     -- in full) to a @requests.jsonl@ file alongside the session's
  110     -- @conversation.jsonl@ / @entries.jsonl@. The contract: each line is
  111     -- the complete JSON-encoded 'CompletionRequest' exactly as passed to
  112     -- the provider, including the full @crMessages@ history. Used to
  113     -- debug whether the two-file storage format is correctly feeding the
  114     -- session history to the LLM. Absent (the default) means the file is
  115     -- not written.
  116   , rcOnDemandSchemas :: Maybe Bool
  117     -- ^ Optional @on_demand_schemas@ flag. When @true@, the full
  118     -- @input_schema@ JSON for every opcode is replaced with a minimal
  119     -- stub (@{\"type\":\"object\"}@) in the @tools@ field of each
  120     -- 'CompletionRequest' (saving tokens on every turn), and a
  121     -- read-only @OPCODE_DESCRIBE@ / @OPCODE_LIST@ pair is registered so
  122     -- the model can retrieve an opcode's full input/output schema
  123     -- on-demand before calling it. Absent (the default) preserves the
  124     -- existing behavior: full schemas are sent inline and the describe
  125     -- opcodes are not registered.
  126   , rcDelegation :: Maybe DelegationFileConfig
  127     -- ^ Optional @[delegation]@ section (subagent spawning tunables).
  128     -- Absent means 'defaultDelegationConfig' applies at resolution time.
  129   , rcWeb :: Maybe WebConfig
  130     -- ^ Optional @[web]@ section (web tool config: search endpoint +
  131     -- fetch/search allow-lists + max fetch bytes). Absent means the web
  132     -- tools are fail-closed (no endpoint configured).
  133   , rcWorkdir :: Maybe WorkdirConfig
  134     -- ^ Optional @[workdir]@ section (per-session workdir lifecycle).
  135     -- Absent means defaults apply (persist, no chroot — chroot is
  136     -- deferred). Per-session workdirs are always on regardless of this
  137     -- section; this config only controls cleanup-on-exit.
  138   , rcSkills :: Maybe SkillsConfig
  139     -- ^ Optional @[skills]@ section (skill auto-injection at session
  140     -- start). Absent means the built-in default applies: the
  141     -- @seal-usage@ skill is auto-injected into the system prompt. Set
  142     -- @[skills] autoload = ""@ to disable auto-injection explicitly.
  143   , rcAgent :: Maybe AgentConfig
  144     -- ^ Optional @[agent]@ section (static behavioral guidance blocks
  145     -- injected into the system prompt). Absent means all three blocks
  146     -- are injected (the defaults); individual flags disable each block.
  147   , rcChatStreaming :: Maybe ChatStreamingFileConfig
  148     -- ^ Optional @[chat_streaming]@ section (chat channel streaming +
  149     -- tool-call progress). Absent means the feature is disabled.
  150   , rcMaxTurns :: Maybe Int
  151     -- ^ Optional top-level @max_turns@ key: the maximum number of
  152     -- tool-use iterations per turn before the loop stops. Absent →
  153     -- 'defaultMaxTurns' (90). Values < 1 are clamped to 1 at resolution
  154     -- time. Mirrors Hermes' default.
  155   , rcToolTimeout :: Maybe ToolTimeoutFileConfig
  156     -- ^ Optional @[tool_timeout]@ section (per-call timeout/abort/retry
  157     -- behavior for opcode dispatch). Absent means
  158     -- 'Seal.Tools.Timeout.defaultToolTimeoutConfig' applies at resolution
  159     -- time. Each field inside is optional too; the resolver fills defaults.
  160   } deriving stock (Eq, Show)

  205 -- | The @[tool_timeout]@ section: per-call timeout/abort/retry behavior for

  222     -- ^ SIGTERM→SIGKILL grace period in microseconds (default 5_000_000).

  225   , ttfcAbortPollMicros :: Maybe Int

  384 -- | Resolve the per-turn max-iterations bound from the loaded config. Absent
  385 -- or non-positive values fall back to 'defaultMaxTurns'; values below 1 are
  386 -- clamped to 1 (one tool call before the stop).
  387 maxTurnsConfig :: RuntimeConfig -> Int
  388 maxTurnsConfig cfg = case rcMaxTurns cfg of
  389   Nothing      -> defaultMaxTurns
  390   Just n | n < 1 -> 1
  391   Just n       -> n

  393 -- | Resolve the effective 'ToolTimeoutConfig': the config-file section if
  394 -- present (with defaults filling absent fields), else
  395 -- 'defaultToolTimeoutConfig'.
  396 toolTimeoutConfig :: RuntimeConfig -> ToolTimeoutConfig
  397 toolTimeoutConfig cfg = case rcToolTimeout cfg of
  398   Nothing    -> defaultToolTimeoutConfig
  399   Just ttfc  -> ToolTimeoutConfig
  400     { ttcDefaultSeconds  = fromMaybe (ttcDefaultSeconds  defaultToolTimeoutConfig) (ttfcDefaultSeconds  ttfc)
  401     , ttcMaxSeconds      = fromMaybe (ttcMaxSeconds      defaultToolTimeoutConfig) (ttfcMaxSeconds      ttfc)
  402     , ttcRetryMax        = fromMaybe (ttcRetryMax        defaultToolTimeoutConfig) (ttfcRetryMax        ttfc)
  403     , ttcRetryBaseMicros = fromMaybe (ttcRetryBaseMicros defaultToolTimeoutConfig) (ttfcRetryBaseMicros ttfc)
  404     , ttcRetryFactor     = fromMaybe (ttcRetryFactor     defaultToolTimeoutConfig) (ttfcRetryFactor     ttfc)
  405     , ttcKillGraceMicros = fromMaybe (ttcKillGraceMicros defaultToolTimeoutConfig) (ttfcKillGraceMicros ttfc)
  406     , ttcMaxOutputBytes  = fromMaybe (ttcMaxOutputBytes  defaultToolTimeoutConfig) (ttfcMaxOutputBytes  ttfc)
  407     , ttcAbortPollMicros = fromMaybe (ttcAbortPollMicros defaultToolTimeoutConfig) (ttfcAbortPollMicros ttfc)
  408     }

  559 -- ---------------------------------------------------------------------------
  560 -- @providers@ table normalization
  561 -- ---------------------------------------------------------------------------
  562 
  563 -- | tomland's 'Toml.tableMap' \/ 'Toml.table' combinators look up the
  564 -- @providers@ node by requiring it to carry an explicit value in the parsed
  565 -- AST. A TOML file that declares only @[providers.\<label\>]@ sub-tables
  566 -- (the idiomatic style, and the one every hand-written config uses) never
  567 -- writes a bare @[providers]@ header, so that node is /implicit/ in
  568 -- tomland's prefix tree and the lookup silently returns an empty map
  569 -- instead of the sub-tables' contents.
  570 --
  571 -- This walks the parsed AST once before decoding and makes that node
  572 -- explicit — without disturbing anything else — so both styles decode
  573 -- correctly. Round-tripped files (written by 'saveRuntimeConfig') already have
  574 -- an explicit node and pass through unchanged.
  575 normalizeProvidersTable :: Toml.TOML -> Toml.TOML
  576 normalizeProvidersTable t =
  577   t { Toml.tomlTables = HashMap.adjust explicitNode "providers" (Toml.tomlTables t) }
  578   where
  579     -- 'Toml.tableMap' only ever looks at the 'Just' value of a matching
  580     -- node, never its 'Toml.Branch' children — so an implicit @providers@
  581     -- 'Toml.Branch' must be collapsed into a single 'Toml.Leaf' whose value
  582     -- embeds those children as its own 'Toml.tomlTables', mirroring exactly
  583     -- what a bare @[providers]@ header would have produced.
  584     explicitNode :: Toml.PrefixTree Toml.TOML -> Toml.PrefixTree Toml.TOML
  585     explicitNode tree = case tree of
  586       Toml.Branch pref Nothing children ->
  587         Toml.Leaf pref (mempty { Toml.tomlTables = children })
  588       Toml.Branch _ (Just _) _ -> tree
  589       Toml.Leaf (_ :|| [])  _  -> tree
  590       Toml.Leaf (_ :|| (p : ps)) v ->
  591         Toml.Leaf ("providers" :|| []) (mempty { Toml.tomlTables = Toml.single (p :|| ps) v })

  597 -- | Load the config file at @path@.
  598 --
  599 -- * File absent  → @Right 'defaultRuntimeConfig'@
  600 -- * Parse error  → @Left@ with the rendered tomland diagnostics
  601 loadRuntimeConfig :: FilePath -> IO (Either Text RuntimeConfig)
  602 loadRuntimeConfig path = do
  603   exists <- doesFileExist path
  604   if not exists
  605     then pure (Right defaultRuntimeConfig)
  606     else do
  607       contents <- TIO.readFile path
  608       pure $ case Toml.parse contents of
  609         Left err   -> Left (Toml.unTomlParseError err)
  610         Right toml -> case Toml.runTomlCodec runtimeConfigCodec (normalizeProvidersTable toml) of
  611           Success cfg  -> Right cfg
  612           Failure errs -> Left (Toml.prettyTomlDecodeErrors errs)
  613 
  614 -- | Save @cfg@ to @path@ atomically: write @path.tmp@, rename over @path@.
  615 -- The file is not chmod-restricted (config.toml is not secret material;
  616 -- unlike vault.age / security.toml which are handled with 0600).
  617 saveRuntimeConfig :: FilePath -> RuntimeConfig -> IO ()
  618 saveRuntimeConfig path cfg = do
  619   let encoded = Toml.encode runtimeConfigCodec cfg
  620       tmp     = path <> ".tmp"
  621   TIO.writeFile tmp encoded
  622   renameFile tmp path
  623 
  624 -- | Process-wide lock serializing config writes to prevent lost-update
  625 -- races (design V7). Multiple concurrent 'updateRuntimeConfig' callers
  626 -- (e.g. Gateway PUT + a future CONFIG_UPDATE, or two sessions) can silently
  627 -- clobber each other without this lock. The MVar is initialized once via
  628 -- 'unsafePerformIO' — idiomatic for a process-wide lock (mirrors
  629 -- 'Seal.Security.Vault.stWriteLock').
  630 {-# NOINLINE configWriteLock #-}
  631 configWriteLock :: MVar ()
  632 configWriteLock = unsafePerformIO (newMVar ())
  633 
  634 -- | Load the config at @path@, apply @f@, save. Propagates any load
  635 -- error as @Left Text@ without writing. The update function operates on
  636 -- 'RuntimeConfig' only — it physically cannot touch security-critical
  637 -- fields (vault, untrusted_execution) because they live in
  638 -- 'Seal.Config.Security.SecurityConfig', a different type (design §4 E).
  639 -- The load-modify-save is serialized behind 'configWriteLock' to prevent
  640 -- lost-update races (design V7).
  641 updateRuntimeConfig :: FilePath -> (RuntimeConfig -> RuntimeConfig) -> IO (Either Text ())
  642 updateRuntimeConfig path f = withMVar configWriteLock $ \_ -> do
  643   result <- loadRuntimeConfig path
  644   case result of
  645     Left err  -> pure (Left err)
  646     Right cfg -> saveRuntimeConfig path (f cfg) >> pure (Right ())

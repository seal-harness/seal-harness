   85 -- | The agent/operator-tunable runtime configuration persisted in
   86 -- @config\/config.toml@. Every field is optional; a missing key decodes as
   87 -- 'Nothing'. Security-critical boot-only fields (vault, untrusted_execution)
   88 -- live in 'Seal.Config.Security.SecurityConfig' (loaded from
   89 -- @security.toml@) and are NOT present here.
   90 data RuntimeConfig = RuntimeConfig
   91   { rcDefaultProvider :: Maybe Text
   92     -- ^ Provider id used for new sessions (e.g. @\"anthropic\"@).
   93   , rcDefaultModel :: Maybe Text
   94     -- ^ Model id used for new sessions (e.g. @\"claude-opus-4-8\"@).
   95   , rcDefaultAgent :: Maybe Text
   96     -- ^ Agent def id used by default for agent-driven flows (set via
   97     -- @\/agent default@). 'Nothing' means no default agent is selected.
   98   , rcProviders :: Map Text ProviderConfig
   99     -- ^ Per-provider config sections (@[providers.<label>]@).
  100   , rcRetrieval :: Maybe RetrievalConfig
  101     -- ^ Optional @[retrieval]@ section (Dynamic Retrieval tuning). Absent
  102     -- means 'defaultRetrievalConfig' applies at resolution time.
  103   , rcSignal :: Maybe SignalConfig
  104     -- ^ Optional @[signal]@ section (Signal channel config). Absent means
  105     -- the Signal channel is not configured.
  106   , rcTelegram :: Maybe TelegramConfig
  107     -- ^ Optional @[telegram]@ section (Telegram channel config). Absent
  108     -- means the Telegram channel is not configured.
  109   , rcDebugSessionTranscript :: Maybe Bool
  110     -- ^ Optional @debug_session_transcript@ flag. When @true@, every
  111     -- 'CompletionRequest' sent to the LLM is also appended (redundantly,
  112     -- in full) to a @requests.jsonl@ file alongside the session's
  113     -- @conversation.jsonl@ / @entries.jsonl@. The contract: each line is
  114     -- the complete JSON-encoded 'CompletionRequest' exactly as passed to
  115     -- the provider, including the full @crMessages@ history. Used to
  116     -- debug whether the indexed transcript storage format is correctly feeding the
  117     -- session history to the LLM. Absent (the default) means the file is
  118     -- not written.
  119   , rcDebugModelResponses :: Maybe Bool
  120   , rcMaxModelRetries :: Maybe Int
  121   , rcRetryBaseDelayMs :: Maybe Int
  122     -- ^ Optional @debug_model_responses@ flag. When @true@, the exact
  123     -- raw 'StreamEvent' values received from the model at every turn
  124     -- are written to a @responses.jsonl@ file alongside the session's
  125     -- other debug files. Independent from @debug_session_transcript@
  126     -- (which controls @requests.jsonl@). Absent (the default) means
  127     -- the file is not written.
  128   , rcOnDemandSchemas :: Maybe Bool
  129     -- ^ Optional @on_demand_schemas@ flag. When @true@, the full
  130     -- @input_schema@ JSON for every opcode is replaced with a minimal
  131     -- stub (@{\"type\":\"object\"}@) in the @tools@ field of each
  132     -- 'CompletionRequest' (saving tokens on every turn), and a
  133     -- read-only @OPCODE_DESCRIBE@ / @OPCODE_LIST@ pair is registered so
  134     -- the model can retrieve an opcode's full input/output schema
  135     -- on-demand before calling it. Absent (the default) preserves the
  136     -- existing behavior: full schemas are sent inline and the describe
  137     -- opcodes are not registered.
  138   , rcDelegation :: Maybe DelegationFileConfig
  139     -- ^ Optional @[delegation]@ section (subagent spawning tunables).
  140     -- Absent means 'defaultDelegationConfig' applies at resolution time.
  141   , rcWeb :: Maybe WebConfig
  142     -- ^ Optional @[web]@ section (web tool config: search endpoint +
  143     -- fetch/search allow-lists + max fetch bytes). Absent means the web
  144     -- tools are fail-closed (no endpoint configured).
  145   , rcWorkdir :: Maybe WorkdirConfig
  146     -- ^ Optional @[workdir]@ section (per-session workdir lifecycle).
  147     -- Absent means defaults apply (persist, no chroot — chroot is
  148     -- deferred). Per-session workdirs are always on regardless of this
  149     -- section; this config only controls cleanup-on-exit.
  150   , rcSkills :: Maybe SkillsConfig
  151     -- ^ Optional @[skills]@ section (skill auto-injection at session
  152     -- start). Absent means the built-in default applies: the
  153     -- @seal-usage@ skill is auto-injected into the system prompt. Set
  154     -- @[skills] autoload = ""@ to disable auto-injection explicitly.
  155   , rcAgent :: Maybe AgentConfig
  156     -- ^ Optional @[agent]@ section (static behavioral guidance blocks
  157     -- injected into the system prompt). Absent means all three blocks
  158     -- are injected (the defaults); individual flags disable each block.
  159   , rcMaxTurns :: Maybe Int
  160     -- ^ Optional top-level @max_turns@ key: the maximum number of
  161     -- tool-use iterations per turn before the loop stops. Absent →
  162     -- 'defaultMaxTurns' (200). Values < 1 are clamped to 1 at resolution
  163     -- time. Mirrors Hermes' default.
  164   , rcToolTimeout :: Maybe ToolTimeoutFileConfig
  165     -- ^ Optional @[tool_timeout]@ section (per-call timeout/abort/retry
  166     -- behavior for opcode dispatch). Absent means
  167     -- 'Seal.Tools.Timeout.defaultToolTimeoutConfig' applies at resolution
  168     -- time. Each field inside is optional too; the resolver fills defaults.
  169   , rcEmbedding :: Maybe EmbeddingConfig
  170     -- ^ Optional @[embedding]@ section (semantic search backend for
  171     -- memory). Absent means the null backend is used (substring search
  172     -- only). Set @backend = "engram"@ to enable engram subprocess search.
  173   , rcChannels :: Map ChannelId ChannelConfig
  174     -- ^ Per-channel config sections (@[channels.\<id\>]@). Multi-instance
  175     -- chat channel management (design §3.1). Legacy @[signal]@\/@[telegram]@
  176     -- sections are migrated into this map at load time ('migrateLegacyChannels').
  177   } deriving stock (Eq, Show)

  435 -- | Resolve the per-turn max-iterations bound from the loaded config. Absent
  436 -- or non-positive values fall back to 'defaultMaxTurns'; values below 1 are
  437 -- clamped to 1 (one tool call before the stop).
  438 maxTurnsConfig :: RuntimeConfig -> Int
  439 maxTurnsConfig cfg = case rcMaxTurns cfg of
  440   Nothing      -> defaultMaxTurns
  441   Just n | n < 1 -> 1
  442   Just n       -> n

  444 -- | Resolve the effective 'ToolTimeoutConfig': the config-file section if
  445 -- present (with defaults filling absent fields), else
  446 -- 'defaultToolTimeoutConfig'.
  447 toolTimeoutConfig :: RuntimeConfig -> ToolTimeoutConfig
  448 toolTimeoutConfig cfg = case rcToolTimeout cfg of
  449   Nothing    -> defaultToolTimeoutConfig
  450   Just ttfc  -> ToolTimeoutConfig
  451     { ttcDefaultSeconds  = fromMaybe (ttcDefaultSeconds  defaultToolTimeoutConfig) (ttfcDefaultSeconds  ttfc)
  452     , ttcMaxSeconds      = fromMaybe (ttcMaxSeconds      defaultToolTimeoutConfig) (ttfcMaxSeconds      ttfc)
  453     , ttcRetryMax        = fromMaybe (ttcRetryMax        defaultToolTimeoutConfig) (ttfcRetryMax        ttfc)
  454     , ttcRetryBaseMicros = fromMaybe (ttcRetryBaseMicros defaultToolTimeoutConfig) (ttfcRetryBaseMicros ttfc)
  455     , ttcRetryFactor     = fromMaybe (ttcRetryFactor     defaultToolTimeoutConfig) (ttfcRetryFactor     ttfc)
  456     , ttcKillGraceMicros = fromMaybe (ttcKillGraceMicros defaultToolTimeoutConfig) (ttfcKillGraceMicros ttfc)
  457     , ttcMaxOutputBytes  = fromMaybe (ttcMaxOutputBytes  defaultToolTimeoutConfig) (ttfcMaxOutputBytes  ttfc)
  458     , ttcAbortPollMicros = fromMaybe (ttcAbortPollMicros defaultToolTimeoutConfig) (ttfcAbortPollMicros ttfc)
  459     }

  464 -- | Bidirectional tomland codec for 'RuntimeConfig'.
  465 -- 'Toml.dioptional' wraps each key: absent → 'Nothing' on decode,
  466 -- 'Nothing' → key omitted on encode.
  467 runtimeConfigCodec :: Toml.TomlCodec RuntimeConfig
  468 runtimeConfigCodec = RuntimeConfig
  469   <$> Toml.dioptional (Toml.text "default_provider") .= rcDefaultProvider
  470   <*> Toml.dioptional (Toml.text "default_model")    .= rcDefaultModel
  471   <*> Toml.dioptional (Toml.text "default_agent")    .= rcDefaultAgent
  472   <*> Toml.tableMap Toml._KeyText (Toml.table providerConfigCodec) "providers" .= rcProviders
  473   <*> Toml.dioptional (Toml.table retrievalConfigCodec "retrieval") .= rcRetrieval
  474   <*> Toml.dioptional (Toml.table signalConfigCodec "signal")       .= rcSignal
  475   <*> Toml.dioptional (Toml.table telegramConfigCodec "telegram")   .= rcTelegram
  476   <*> Toml.dioptional (Toml.bool "debug_session_transcript") .= rcDebugSessionTranscript
  477   <*> Toml.dioptional (Toml.bool "debug_model_responses") .= rcDebugModelResponses
  478   <*> Toml.dioptional (Toml.int "max_model_retries") .= rcMaxModelRetries
  479   <*> Toml.dioptional (Toml.int "model_retry_base_delay_ms") .= rcRetryBaseDelayMs
  480   <*> Toml.dioptional (Toml.bool "on_demand_schemas") .= rcOnDemandSchemas
  481   <*> Toml.dioptional (Toml.table delegationConfigCodec "delegation") .= rcDelegation
  482   <*> Toml.dioptional (Toml.table webConfigCodec "web") .= rcWeb

  580 -- ---------------------------------------------------------------------------
  581 -- @providers@ table normalization
  582 -- ---------------------------------------------------------------------------
  583 
  584 -- | tomland's 'Toml.tableMap' \/ 'Toml.table' combinators look up the
  585 -- @providers@ node by requiring it to carry an explicit value in the parsed
  586 -- AST. A TOML file that declares only @[providers.\<label\>]@ sub-tables
  587 -- (the idiomatic style, and the one every hand-written config uses) never
  588 -- writes a bare @[providers]@ header, so that node is /implicit/ in
  589 -- tomland's prefix tree and the lookup silently returns an empty map
  590 -- instead of the sub-tables' contents.
  591 --
  592 -- This walks the parsed AST once before decoding and makes that node
  593 -- explicit — without disturbing anything else — so both styles decode
  594 -- correctly. Round-tripped files (written by 'saveRuntimeConfig') already have
  595 -- an explicit node and pass through unchanged.
  596 normalizeProvidersTable :: Toml.TOML -> Toml.TOML
  597 normalizeProvidersTable t =
  598   t { Toml.tomlTables = HashMap.adjust explicitNode "providers" (Toml.tomlTables t) }
  599   where
  600     -- 'Toml.tableMap' only ever looks at the 'Just' value of a matching
  601     -- node, never its 'Toml.Branch' children — so an implicit @providers@
  602     -- 'Toml.Branch' must be collapsed into a single 'Toml.Leaf' whose value
  603     -- embeds those children as its own 'Toml.tomlTables', mirroring exactly
  604     -- what a bare @[providers]@ header would have produced.
  605     explicitNode :: Toml.PrefixTree Toml.TOML -> Toml.PrefixTree Toml.TOML
  606     explicitNode tree = case tree of
  607       Toml.Branch pref Nothing children ->
  608         Toml.Leaf pref (mempty { Toml.tomlTables = children })
  609       Toml.Branch _ (Just _) _ -> tree
  610       Toml.Leaf (_ :|| [])  _  -> tree
  611       Toml.Leaf (_ :|| (p : ps)) v ->
  612         Toml.Leaf ("providers" :|| []) (mempty { Toml.tomlTables = Toml.single (p :|| ps) v })
  613

  616 -- ---------------------------------------------------------------------------

  618 -- | Parallel to 'normalizeProvidersTable' for the @[channels.\<id\>]@
  619 -- table. tomland's 'Toml.tableMap' silently returns an empty map when a
  620 -- config declares only @[channels.\<id\>]@ sub-tables and no bare
  621 -- @[channels]@ header (the idiomatic hand-written style). This walks the
  622 -- parsed AST once before decoding and makes that node explicit so both
  623 -- styles decode correctly. Round-tripped files (written by
  624 -- 'saveRuntimeConfig') already have an explicit node and pass through
  625 -- unchanged.
  626 normalizeChannelsTable :: Toml.TOML -> Toml.TOML

  628   t { Toml.tomlTables = HashMap.adjust explicitNode "channels" (Toml.tomlTables t) }
  629   where
  630     explicitNode :: Toml.PrefixTree Toml.TOML -> Toml.PrefixTree Toml.TOML
  631     explicitNode tree = case tree of
  632       Toml.Branch pref Nothing children ->
  633         Toml.Leaf pref (mempty { Toml.tomlTables = children })
  634       Toml.Branch _ (Just _) _ -> tree
  635       Toml.Leaf (_ :|| [])  _  -> tree
  636       Toml.Leaf (_ :|| (p : ps)) v ->
  637         Toml.Leaf ("channels" :|| []) (mempty { Toml.tomlTables = Toml.single (p :|| ps) v })

  639 -- ---------------------------------------------------------------------------
  640 -- Legacy channel migration (design §3.3)
  641 -- ---------------------------------------------------------------------------
  642 
  643 -- | Migrate legacy @[signal]@\/@[telegram]@ sections into 'rcChannels' on
  644 -- load. Per-id synthesis (not whole-map-empty): a legacy section is
  645 -- synthesized only when its default id (@\"signal\"@\/@\"telegram\"@) is
  646 -- /not/ already present in 'rcChannels', so an explicit
  647 -- @[channels.signal]@ wins over a legacy @[signal]@. A legacy section is
  648 -- migrated even when 'rcChannels' is non-empty (coexistence during the
  649 -- upgrade transition). 'fromTelegramConfig' preserves 'tcToken' in
  650 -- 'ccToken' so token-in-config setups survive migration. This is read-only
  651 -- migration — the legacy sections are NOT removed from the file.
  652 migrateLegacyChannels :: RuntimeConfig -> RuntimeConfig
  653 migrateLegacyChannels cfg =
  654   let chans0 = rcChannels cfg
  655       chans1 = case (Map.notMember (ChannelId "signal") chans0, rcSignal cfg) of
  656         (True, Just sig) -> Map.insert (ChannelId "signal") (fromSignalConfig sig) chans0
  657         _                -> chans0
  658       chans2 = case (Map.notMember (ChannelId "telegram") chans1, rcTelegram cfg) of
  659         (True, Just tg) -> Map.insert (ChannelId "telegram") (fromTelegramConfig tg) chans1
  660         _               -> chans1
  661   in cfg { rcChannels = chans2 }

  667 -- | Load the config file at @path@.
  668 --
  669 -- * File absent  → @Right 'defaultRuntimeConfig'@
  670 -- * Parse error  → @Left@ with the rendered tomland diagnostics
  671 loadRuntimeConfig :: FilePath -> IO (Either Text RuntimeConfig)
  672 loadRuntimeConfig path = do
  673   exists <- doesFileExist path
  674   if not exists
  675     then pure (Right defaultRuntimeConfig)
  676     else do
  677       contents <- TIO.readFile path
  678       pure $ case Toml.parse contents of
  679         Left err   -> Left (Toml.unTomlParseError err)
  680         Right toml -> case Toml.runTomlCodec runtimeConfigCodec (normalizeChannelsTable (normalizeProvidersTable toml)) of
  681           Success cfg  -> Right (migrateLegacyChannels cfg)
  682           Failure errs -> Left (Toml.prettyTomlDecodeErrors errs)
  683 
  684 -- | Save @cfg@ to @path@ atomically: write @path.tmp@, rename over @path@.
  685 -- The file is not chmod-restricted (config.toml is not secret material;
  686 -- unlike vault.age / security.toml which are handled with 0600).
  687 saveRuntimeConfig :: FilePath -> RuntimeConfig -> IO ()
  688 saveRuntimeConfig path cfg = do
  689   let encoded = Toml.encode runtimeConfigCodec cfg
  690       tmp     = path <> ".tmp"
  691   TIO.writeFile tmp encoded
  692   renameFile tmp path

  694 -- | Process-wide lock serializing config writes to prevent lost-update
  695 -- races (design V7). Multiple concurrent 'updateRuntimeConfig' callers
  696 -- (e.g. Gateway PUT + a future CONFIG_UPDATE, or two sessions) can silently
  697 -- clobber each other without this lock. The MVar is initialized once via
  698 -- 'unsafePerformIO' — idiomatic for a process-wide lock (mirrors
  699 -- 'Seal.Security.Vault.stWriteLock').
  700 {-# NOINLINE configWriteLock #-}
  701 configWriteLock :: MVar ()
  702 configWriteLock = unsafePerformIO (newMVar ())
  703 
  704 -- | Load the config at @path@, apply @f@, save. Propagates any load
  705 -- error as @Left Text@ without writing. The update function operates on
  706 -- 'RuntimeConfig' only — it physically cannot touch security-critical
  707 -- fields (vault, untrusted_execution) because they live in
  708 -- 'Seal.Config.Security.SecurityConfig', a different type (design §4 E).
  709 -- The load-modify-save is serialized behind 'configWriteLock' to prevent
  710 -- lost-update races (design V7).
  711 updateRuntimeConfig :: FilePath -> (RuntimeConfig -> RuntimeConfig) -> IO (Either Text ())
  712 updateRuntimeConfig path f = withMVar configWriteLock $ \_ -> do
  713   stillLegacy <- loadLegacyGatewayTable path
  714   case stillLegacy of
  715     Just _    -> pure (Left gatewayTableRefusal)
  716     Nothing   -> do
  717       result <- loadRuntimeConfig path
  718       case result of
  719         Left err  -> pure (Left err)
  720         Right cfg -> saveRuntimeConfig path (f cfg) >> pure (Right ())
  721 
  722 -- | Why a @config.toml@ write stops while the file still carries @[gateway]@.
  723 --
  724 -- The gateway table lives in @security.toml@. A re-encode through
  725 -- 'runtimeConfigCodec' cannot carry a section the codec no longer knows, so
  726 -- writing here would delete the operator's bind address, and the migration
  727 -- could never recover it. Refusing is the only outcome that cannot lose the
  728 -- setting: the next @seal serve@ or @seal tui@ boot moves the table into
  729 -- @security.toml@ and the refusal clears.
  730 gatewayTableRefusal :: Text
  731 gatewayTableRefusal =
  732   "config.toml still carries a [gateway] table, which this build cannot write "
  733     <> "back. Start `seal serve` or `seal tui` once to move it into "
  734     <> "~/.seal/security.toml, then retry."

   74 -- | The agent/operator-tunable runtime configuration persisted in
   75 -- @config\/config.toml@. Every field is optional; a missing key decodes as
   76 -- 'Nothing'. Security-critical boot-only fields (vault, untrusted_execution)
   77 -- live in 'Seal.Config.Security.SecurityConfig' (loaded from
   78 -- @security.toml@) and are NOT present here.
   79 data RuntimeConfig = RuntimeConfig
   80   { rcDefaultProvider :: Maybe Text
   81     -- ^ Provider id used for new sessions (e.g. @\"anthropic\"@).
   82   , rcDefaultModel :: Maybe Text
   83     -- ^ Model id used for new sessions (e.g. @\"claude-opus-4-8\"@).
   84   , rcDefaultAgent :: Maybe Text
   85     -- ^ Agent def id used by default for agent-driven flows (set via
   86     -- @\/agent default@). 'Nothing' means no default agent is selected.
   87   , rcProviders :: Map Text ProviderConfig
   88     -- ^ Per-provider config sections (@[providers.<label>]@).
   89   , rcRetrieval :: Maybe RetrievalConfig
   90     -- ^ Optional @[retrieval]@ section (Dynamic Retrieval tuning). Absent
   91     -- means 'defaultRetrievalConfig' applies at resolution time.
   92   , rcSignal :: Maybe SignalConfig
   93     -- ^ Optional @[signal]@ section (Signal channel config). Absent means
   94     -- the Signal channel is not configured.
   95   , rcTelegram :: Maybe TelegramConfig
   96     -- ^ Optional @[telegram]@ section (Telegram channel config). Absent
   97     -- means the Telegram channel is not configured.
   98   , rcGateway :: Maybe PartialGatewayConfig
   99     -- ^ Optional @[gateway]@ section (web gateway config). Absent means the
  100     -- gateway is not configured. Each field inside is optional too; the
  101     -- call site merges with 'Seal.Gateway.Config.withGatewayDefaults'.
  102   , rcDebugSessionTranscript :: Maybe Bool
  103     -- ^ Optional @debug_session_transcript@ flag. When @true@, every
  104     -- 'CompletionRequest' sent to the LLM is also appended (redundantly,
  105     -- in full) to a @requests.jsonl@ file alongside the session's
  106     -- @conversation.jsonl@ / @entries.jsonl@. The contract: each line is
  107     -- the complete JSON-encoded 'CompletionRequest' exactly as passed to
  108     -- the provider, including the full @crMessages@ history. Used to
  109     -- debug whether the two-file storage format is correctly feeding the
  110     -- session history to the LLM. Absent (the default) means the file is
  111     -- not written.
  112   , rcOnDemandSchemas :: Maybe Bool
  113     -- ^ Optional @on_demand_schemas@ flag. When @true@, the full
  114     -- @input_schema@ JSON for every opcode is replaced with a minimal
  115     -- stub (@{\"type\":\"object\"}@) in the @tools@ field of each
  116     -- 'CompletionRequest' (saving tokens on every turn), and a
  117     -- read-only @OPCODE_DESCRIBE@ / @OPCODE_LIST@ pair is registered so
  118     -- the model can retrieve an opcode's full input/output schema
  119     -- on-demand before calling it. Absent (the default) preserves the
  120     -- existing behavior: full schemas are sent inline and the describe
  121     -- opcodes are not registered.
  122   , rcDelegation :: Maybe DelegationFileConfig
  123     -- ^ Optional @[delegation]@ section (subagent spawning tunables).
  124     -- Absent means 'defaultDelegationConfig' applies at resolution time.
  125   , rcWeb :: Maybe WebConfig
  126     -- ^ Optional @[web]@ section (web tool config: search endpoint +
  127     -- fetch/search allow-lists + max fetch bytes). Absent means the web
  128     -- tools are fail-closed (no endpoint configured).
  129   , rcWorkdir :: Maybe WorkdirConfig
  130     -- ^ Optional @[workdir]@ section (per-session workdir lifecycle).
  131     -- Absent means defaults apply (persist, no chroot — chroot is
  132     -- deferred). Per-session workdirs are always on regardless of this
  133     -- section; this config only controls cleanup-on-exit.
  134   , rcSkills :: Maybe SkillsConfig
  135     -- ^ Optional @[skills]@ section (skill auto-injection at session
  136     -- start). Absent means the built-in default applies: the
  137     -- @seal-usage@ skill is auto-injected into the system prompt. Set
  138     -- @[skills] autoload = ""@ to disable auto-injection explicitly.
  139   , rcAgent :: Maybe AgentConfig
  140     -- ^ Optional @[agent]@ section (static behavioral guidance blocks
  141     -- injected into the system prompt). Absent means all three blocks
  142     -- are injected (the defaults); individual flags disable each block.
  143   , rcMaxTurns :: Maybe Int
  144     -- ^ Optional top-level @max_turns@ key: the maximum number of
  145     -- tool-use iterations per turn before the loop stops. Absent →
  146     -- 'defaultMaxTurns' (90). Values < 1 are clamped to 1 at resolution
  147     -- time. Mirrors Hermes' default.
  148   , rcToolTimeout :: Maybe ToolTimeoutFileConfig
  149     -- ^ Optional @[tool_timeout]@ section (per-call timeout/abort/retry
  150     -- behavior for opcode dispatch). Absent means
  151     -- 'Seal.Tools.Timeout.defaultToolTimeoutConfig' applies at resolution
  152     -- time. Each field inside is optional too; the resolver fills defaults.
  153   , rcEmbedding :: Maybe EmbeddingConfig
  154     -- ^ Optional @[embedding]@ section (semantic search backend for
  155     -- memory). Absent means the null backend is used (substring search
  156     -- only). Set @backend = "engram"@ to enable engram subprocess search.
  157   } deriving stock (Eq, Show)

  396 -- | Resolve the per-turn max-iterations bound from the loaded config. Absent
  397 -- or non-positive values fall back to 'defaultMaxTurns'; values below 1 are
  398 -- clamped to 1 (one tool call before the stop).
  399 maxTurnsConfig :: RuntimeConfig -> Int
  400 maxTurnsConfig cfg = case rcMaxTurns cfg of
  401   Nothing      -> defaultMaxTurns
  402   Just n | n < 1 -> 1
  403   Just n       -> n

  405 -- | Resolve the effective 'ToolTimeoutConfig': the config-file section if
  406 -- present (with defaults filling absent fields), else
  407 -- 'defaultToolTimeoutConfig'.
  408 toolTimeoutConfig :: RuntimeConfig -> ToolTimeoutConfig
  409 toolTimeoutConfig cfg = case rcToolTimeout cfg of
  410   Nothing    -> defaultToolTimeoutConfig
  411   Just ttfc  -> ToolTimeoutConfig
  412     { ttcDefaultSeconds  = fromMaybe (ttcDefaultSeconds  defaultToolTimeoutConfig) (ttfcDefaultSeconds  ttfc)
  413     , ttcMaxSeconds      = fromMaybe (ttcMaxSeconds      defaultToolTimeoutConfig) (ttfcMaxSeconds      ttfc)
  414     , ttcRetryMax        = fromMaybe (ttcRetryMax        defaultToolTimeoutConfig) (ttfcRetryMax        ttfc)
  415     , ttcRetryBaseMicros = fromMaybe (ttcRetryBaseMicros defaultToolTimeoutConfig) (ttfcRetryBaseMicros ttfc)
  416     , ttcRetryFactor     = fromMaybe (ttcRetryFactor     defaultToolTimeoutConfig) (ttfcRetryFactor     ttfc)
  417     , ttcKillGraceMicros = fromMaybe (ttcKillGraceMicros defaultToolTimeoutConfig) (ttfcKillGraceMicros ttfc)
  418     , ttcMaxOutputBytes  = fromMaybe (ttcMaxOutputBytes  defaultToolTimeoutConfig) (ttfcMaxOutputBytes  ttfc)
  419     , ttcAbortPollMicros = fromMaybe (ttcAbortPollMicros defaultToolTimeoutConfig) (ttfcAbortPollMicros ttfc)
  420     }

  425 -- | Bidirectional tomland codec for 'RuntimeConfig'.

  430   <$> Toml.dioptional (Toml.text "default_provider") .= rcDefaultProvider

  434   <*> Toml.dioptional (Toml.table retrievalConfigCodec "retrieval") .= rcRetrieval
  435   <*> Toml.dioptional (Toml.table signalConfigCodec "signal")       .= rcSignal
  436   <*> Toml.dioptional (Toml.table telegramConfigCodec "telegram")   .= rcTelegram
  437   <*> Toml.dioptional (Toml.table gatewayConfigCodec "gateway")    .= rcGateway
  438   <*> Toml.dioptional (Toml.bool "debug_session_transcript") .= rcDebugSessionTranscript
  439   <*> Toml.dioptional (Toml.bool "on_demand_schemas") .= rcOnDemandSchemas
  440   <*> Toml.dioptional (Toml.table delegationConfigCodec "delegation") .= rcDelegation
  441   <*> Toml.dioptional (Toml.table webConfigCodec "web") .= rcWeb
  442   <*> Toml.dioptional (Toml.table workdirConfigCodec "workdir") .= rcWorkdir
  443   <*> Toml.dioptional (Toml.table skillsConfigCodec "skills")   .= rcSkills
  444   <*> Toml.dioptional (Toml.table agentConfigCodec "agent")     .= rcAgent

  537 -- ---------------------------------------------------------------------------
  538 -- @providers@ table normalization
  539 -- ---------------------------------------------------------------------------
  540 
  541 -- | tomland's 'Toml.tableMap' \/ 'Toml.table' combinators look up the
  542 -- @providers@ node by requiring it to carry an explicit value in the parsed
  543 -- AST. A TOML file that declares only @[providers.\<label\>]@ sub-tables
  544 -- (the idiomatic style, and the one every hand-written config uses) never
  545 -- writes a bare @[providers]@ header, so that node is /implicit/ in
  546 -- tomland's prefix tree and the lookup silently returns an empty map
  547 -- instead of the sub-tables' contents.
  548 --
  549 -- This walks the parsed AST once before decoding and makes that node
  550 -- explicit — without disturbing anything else — so both styles decode
  551 -- correctly. Round-tripped files (written by 'saveRuntimeConfig') already have
  552 -- an explicit node and pass through unchanged.
  553 normalizeProvidersTable :: Toml.TOML -> Toml.TOML
  554 normalizeProvidersTable t =
  555   t { Toml.tomlTables = HashMap.adjust explicitNode "providers" (Toml.tomlTables t) }
  556   where
  557     -- 'Toml.tableMap' only ever looks at the 'Just' value of a matching
  558     -- node, never its 'Toml.Branch' children — so an implicit @providers@
  559     -- 'Toml.Branch' must be collapsed into a single 'Toml.Leaf' whose value
  560     -- embeds those children as its own 'Toml.tomlTables', mirroring exactly
  561     -- what a bare @[providers]@ header would have produced.
  562     explicitNode :: Toml.PrefixTree Toml.TOML -> Toml.PrefixTree Toml.TOML
  563     explicitNode tree = case tree of
  564       Toml.Branch pref Nothing children ->
  565         Toml.Leaf pref (mempty { Toml.tomlTables = children })
  566       Toml.Branch _ (Just _) _ -> tree
  567       Toml.Leaf (_ :|| [])  _  -> tree
  568       Toml.Leaf (_ :|| (p : ps)) v ->
  569         Toml.Leaf ("providers" :|| []) (mempty { Toml.tomlTables = Toml.single (p :|| ps) v })

  575 -- | Load the config file at @path@.
  576 --
  577 -- * File absent  → @Right 'defaultRuntimeConfig'@
  578 -- * Parse error  → @Left@ with the rendered tomland diagnostics
  579 loadRuntimeConfig :: FilePath -> IO (Either Text RuntimeConfig)
  580 loadRuntimeConfig path = do
  581   exists <- doesFileExist path
  582   if not exists
  583     then pure (Right defaultRuntimeConfig)
  584     else do
  585       contents <- TIO.readFile path
  586       pure $ case Toml.parse contents of
  587         Left err   -> Left (Toml.unTomlParseError err)
  588         Right toml -> case Toml.runTomlCodec runtimeConfigCodec (normalizeProvidersTable toml) of
  589           Success cfg  -> Right cfg
  590           Failure errs -> Left (Toml.prettyTomlDecodeErrors errs)
  591 
  592 -- | Save @cfg@ to @path@ atomically: write @path.tmp@, rename over @path@.
  593 -- The file is not chmod-restricted (config.toml is not secret material;
  594 -- unlike vault.age / security.toml which are handled with 0600).
  595 saveRuntimeConfig :: FilePath -> RuntimeConfig -> IO ()
  596 saveRuntimeConfig path cfg = do
  597   let encoded = Toml.encode runtimeConfigCodec cfg
  598       tmp     = path <> ".tmp"
  599   TIO.writeFile tmp encoded
  600   renameFile tmp path
  601 
  602 -- | Process-wide lock serializing config writes to prevent lost-update
  603 -- races (design V7). Multiple concurrent 'updateRuntimeConfig' callers
  604 -- (e.g. Gateway PUT + a future CONFIG_UPDATE, or two sessions) can silently
  605 -- clobber each other without this lock. The MVar is initialized once via
  606 -- 'unsafePerformIO' — idiomatic for a process-wide lock (mirrors
  607 -- 'Seal.Security.Vault.stWriteLock').
  608 {-# NOINLINE configWriteLock #-}
  609 configWriteLock :: MVar ()
  610 configWriteLock = unsafePerformIO (newMVar ())
  611 
  612 -- | Load the config at @path@, apply @f@, save. Propagates any load
  613 -- error as @Left Text@ without writing. The update function operates on
  614 -- 'RuntimeConfig' only — it physically cannot touch security-critical
  615 -- fields (vault, untrusted_execution) because they live in
  616 -- 'Seal.Config.Security.SecurityConfig', a different type (design §4 E).
  617 -- The load-modify-save is serialized behind 'configWriteLock' to prevent
  618 -- lost-update races (design V7).
  619 updateRuntimeConfig :: FilePath -> (RuntimeConfig -> RuntimeConfig) -> IO (Either Text ())
  620 updateRuntimeConfig path f = withMVar configWriteLock $ \_ -> do
  621   result <- loadRuntimeConfig path
  622   case result of
  623     Left err  -> pure (Left err)
  624     Right cfg -> saveRuntimeConfig path (f cfg) >> pure (Right ())

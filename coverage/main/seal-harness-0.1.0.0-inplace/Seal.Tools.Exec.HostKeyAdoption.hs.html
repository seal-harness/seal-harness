    2 -- | TOFU (Trust On First Use) host-key adoption with human confirmation.
    3 --
    4 -- When the harness connects to a remote execution machine via SSH and the
    5 -- host key is not in the pinned @known_hosts@ file, SSH fails with
    6 -- @StrictHostKeyChecking=yes@ + @BatchMode=yes@. This module provides the
    7 -- capability to adopt the key: probe the host via @ssh-keyscan@, display the
    8 -- fingerprint, ask the human for confirmation, and append the key to the
    9 -- @known_hosts@ file on approval.
   10 --
   11 -- Security properties:
   12 --
   13 --   * Only @ExecHostKeyUnknown@ (no key known) triggers adoption — a key
   14 --     mismatch (@ExecHostKeyMismatch@, the key CHANGED) is a hard failure
   15 --     that NEVER offers adoption (possible MITM).
   16 --   * The human sees the host, the key type, and the fingerprint before
   17 --     approving.
   18 --   * The key is appended to the pinned @known_hosts@ file (the same file
   19 --     @ssh -o UserKnownHostsFile=\<pinned\>@ uses), not the default
   20 --     @~/.ssh/known_hosts@.
   21 --   * On rejection or failure, returns @Left@ with a descriptive message;
   22 --     the caller fail-closes.
   23 --
   24 --   * This module writes to @~/.seal/exec-known-hosts@ — a security-sensitive
   25 --     file under the @~/.seal/@ tree (see CONTRIBUTING.md §"Security-first").
   26 --     This is **not** an opcode; it is a harness-level capability invoked at
   27 --     session-exec build time (before any agent turn), gated by a
   28 --     human-confirmation callback. No agent-driven opcode can reach this
   29 --     path — the @HostKeyAdoption@ handle is threaded through @TurnDeps@,
   30 --     not through the @UIO@/@UntrustedIO@ capability surface.

   68 -- | The capability handle for host-key adoption. The single method probes
   69 -- the remote host, presents the key to the human, and on approval appends
   70 -- it to the pinned @known_hosts@ file. The constructor is NOT exported;
   71 -- use 'mkHostKeyAdoption' (production) or 'mkHostKeyAdoptionStub' (tests).
   72 newtype HostKeyAdoption = HostKeyAdoption
   73   { hkaAdopt :: SshConfig -> IO (Either Text ())
   74     -- ^ Probe the host, ask the human, append the key on approval.
   75     -- Returns @Right ()@ on success (key adopted), @Left msg@ on
   76     -- rejection or failure.
   77   }
   78 
   79 -- | Information about a host key discovered by @ssh-keyscan@.
   80 data HostKeyInfo = HostKeyInfo
   81   { hkiHost    :: Text
   82     -- ^ The host (user@host form for display)
   83   , hkiKeyType :: Text
   84     -- ^ e.g. @ssh-ed25519@
   85   , hkiKeyLine :: Text
   86     -- ^ The full @ssh-keyscan@ output line (host keytype keydata)
   87   , hkiFingerprint :: Text
   88     -- ^ The SHA256 fingerprint (computed via @ssh-keygen -lf@)
   89   }
   90 
   91 -- | Production host-key adoption. Probes via @ssh-keyscan@, shows the
   92 -- fingerprint, calls the @confirm@ callback, and appends the key to the
   93 -- pinned @known_hosts@ file on approval. The @confirm@ callback is the
   94 -- human-interaction seam (e.g. @ccPrompt@ from 'ChannelCaps').
   95 mkHostKeyAdoption
   96   :: (Text -> IO Bool)
   97   -- ^ Confirm callback: given the prompt text, returns @True@ to approve,
   98   -- @False@ to reject.
   99   -> HostKeyAdoption
  100 mkHostKeyAdoption confirm =
  101   HostKeyAdoption
  102     { hkaAdopt = adoptHostKeyIO confirm
  103     }
  104 
  105 -- | A stub that always returns @Left "host-key adoption not available"@.
  106 -- Used in tests or when no human-interaction surface is wired.
  107 mkHostKeyAdoptionStub :: HostKeyAdoption
  108 mkHostKeyAdoptionStub =
  109   HostKeyAdoption
  110     { hkaAdopt = \_ -> pure (Left "host-key adoption not available")
  111     }

  113 -- | Like 'mkHostKeyAdoption' but with an injected probe function (tests).
  114 -- The probe returns the 'HostKeyInfo' (or an error message). The real
  115 -- 'probeHostKey' runs @ssh-keyscan@ + @ssh-keygen -lf@.
  116 mkHostKeyAdoptionWithProbe
  117   :: (SshConfig -> IO (Either Text HostKeyInfo))
  118   -- ^ Injected probe (replaces 'probeHostKey')
  119   -> (Text -> IO Bool)
  120   -- ^ Confirm callback
  121   -> HostKeyAdoption
  122 mkHostKeyAdoptionWithProbe probe confirm =
  123   HostKeyAdoption
  124     { hkaAdopt = \cfg -> do
  125         eInfo <- probe cfg
  126         case eInfo of
  127           Left err -> pure (Left err)
  128           Right info -> do
  129             let prompt = buildAdoptionPrompt info
  130             approved <- confirm prompt
  131             if approved
  132               then appendKeyToKnownHosts (scKnownHosts cfg) (hkiKeyLine info)
  133               else pure (Left ("host-key adoption rejected for " <> hkiHost info))
  134     }
  135 
  136 -- | The full adoption flow: probe → display → confirm → append.
  137 -- Used by 'mkHostKeyAdoption'; exported for direct testing.
  138 adoptHostKeyIO :: (Text -> IO Bool) -> SshConfig -> IO (Either Text ())
  139 adoptHostKeyIO confirm cfg = do
  140   eInfo <- probeHostKey cfg
  141   case eInfo of
  142     Left err -> pure (Left err)
  143     Right info -> do
  144       let prompt = buildAdoptionPrompt info
  145       globalLogIO InfoS (K.ls ("Host-key adoption prompt: " <> prompt))
  146       approved <- confirm prompt
  147       if approved
  148         then do
  149           r <- appendKeyToKnownHosts (scKnownHosts cfg) (hkiKeyLine info)
  150           case r of
  151             Left err -> pure (Left err)
  152             Right _  -> pure (Right ())
  153         else pure (Left ("host-key adoption rejected for " <> hkiHost info))
  154 
  155 -- | Probe a remote host via @ssh-keyscan -t ed25519,rsa,ecdsa <host>@
  156 -- and compute the fingerprint via @ssh-keygen -lf -@. Returns the
  157 -- 'HostKeyInfo' or an error message.
  158 probeHostKey :: SshConfig -> IO (Either Text HostKeyInfo)
  159 probeHostKey cfg = do
  160   let host = T.unpack (getSshHost (scHost cfg))
  161       port = scPort cfg
  162       keyTypes = "ed25519,rsa,ecdsa"
  163       portArg = if port == 22 then [] else ["-p", show port]
  164       scanArgv = ["ssh-keyscan", "-T", "10"] <> portArg
  165                  <> ["-t", keyTypes, host]
  166   eScanOutput <- try (readProcessWithStderr scanArgv)
  167                     :: IO (Either IOException (ExitCode, Text, Text))
  168   case eScanOutput of
  169     Left ioErr -> pure (Left ("ssh-keyscan failed: " <> T.pack (show ioErr)))
  170     Right (ExitSuccess, stdout, _stderr) ->
  171       case extractFirstKeyLine stdout of
  172         Nothing -> pure (Left ("ssh-keyscan returned no keys for " <> T.pack host))
  173         Just (keyType, keyLine) -> do
  174           eFp <- computeFingerprint keyLine
  175           case eFp of
  176             Left err -> pure (Left err)
  177             Right fp -> pure (Right HostKeyInfo
  178               { hkiHost = getSshUser (scUser cfg) <> "@" <> T.pack host
  179               , hkiKeyType = keyType
  180               , hkiKeyLine = keyLine
  181               , hkiFingerprint = fp
  182               })
  183     Right (ExitFailure n, _stdout, stderr) ->
  184       pure (Left ("ssh-keyscan exited " <> T.pack (show n) <> ": " <> stderr))
  185 
  186 -- | Extract the first key line from @ssh-keyscan@ output. Returns the key
  187 -- type and the full line. @ssh-keyscan@ output format:
  188 -- @<host> <keytype> <keydata>@
  189 extractFirstKeyLine :: Text -> Maybe (Text, Text)
  190 extractFirstKeyLine output =
  191   case [ l | l <- T.lines output, not (T.null (T.strip l)), not ("#" `T.isPrefixOf` T.strip l) ] of
  192     (line : _) ->
  193       let parts = T.words line
  194       in case parts of
  195            (_host : keyType : _) -> Just (keyType, line)
  196            _ -> Nothing
  197     [] -> Nothing
  198 
  199 -- | Compute the SHA256 fingerprint of a host key line via
  200 -- @ssh-keygen -lf -@ (the key line on stdin).
  201 computeFingerprint :: Text -> IO (Either Text Text)
  202 computeFingerprint keyLine = do
  203   let argv = ["ssh-keygen", "-lf", "-"]
  204   eRes <- try (readProcessWithStdin argv (TE.encodeUtf8 keyLine))
  205             :: IO (Either IOException (ExitCode, Text, Text))
  206   case eRes of
  207     Left ioErr -> pure (Left ("ssh-keygen failed: " <> T.pack (show ioErr)))
  208     Right (ExitSuccess, stdout, _stderr) ->
  209       case T.lines (T.strip stdout) of
  210         (fpLine : _) -> pure (Right (T.strip fpLine))
  211         [] -> pure (Left "ssh-keygen returned no fingerprint")
  212     Right (ExitFailure n, _stdout, stderr) ->
  213       pure (Left ("ssh-keygen exited " <> T.pack (show n) <> ": " <> stderr))

  215 -- | Build the human-facing adoption prompt.
  216 buildAdoptionPrompt :: HostKeyInfo -> Text
  217 buildAdoptionPrompt info =
  218   "The remote execution host " <> hkiHost info
  219   <> " is not in the known_hosts file.\n"
  220   <> "Key type: " <> hkiKeyType info <> "\n"
  221   <> "Fingerprint: " <> hkiFingerprint info <> "\n"
  222   <> "Adopt this host key? [y/N]"
  223 
  224 -- | Append a key line to the @known_hosts@ file. Creates the file if it
  225 -- does not exist (the parent directory must already exist). Avoids
  226 -- duplicate entries: skips if the key line is already present.
  227 appendKeyToKnownHosts :: FilePath -> Text -> IO (Either Text ())
  228 appendKeyToKnownHosts knownHostsPath keyLine = do
  229   exists <- doesFileExist knownHostsPath
  230   if not exists
  231     then do
  232       let dir = takeDirectory knownHostsPath
  233       dirExists <- doesDirectoryExist dir
  234       if not dirExists
  235         then pure (Left ("known_hosts directory does not exist: " <> T.pack dir))
  236         else do
  237           TIO.writeFile knownHostsPath (keyLine <> "\n")
  238           pure (Right ())
  239     else do
  240       existing <- TIO.readFile knownHostsPath
  241       if keyLine `T.isInfixOf` existing
  242         then pure (Right ())
  243         else do
  244           TIO.appendFile knownHostsPath (keyLine <> "\n")
  245           pure (Right ())

  251 -- | Run a process with stdout+stderr captured, no stdin.
  252 readProcessWithStderr :: [String] -> IO (ExitCode, Text, Text)
  253 readProcessWithStderr argv = do
  254   let (program, args) = case argv of
  255         (p : as) -> (p, as)
  256         []       -> error "readProcessWithStderr: empty argv (unreachable)"
  257       cp = (proc program args)
  258               { std_in = NoStream, std_out = CreatePipe, std_err = CreatePipe
  259               }
  260   withCreateProcess cp $ \_ mOut mErr ph -> do
  261     outTxt <- maybe (pure "") TIO.hGetContents mOut
  262     errTxt <- maybe (pure "") TIO.hGetContents mErr
  263     ec <- waitForProcess ph
  264     pure (ec, outTxt, errTxt)
  265 
  266 -- | Run a process with a stdin payload, stdout+stderr captured.
  267 readProcessWithStdin :: [String] -> ByteString -> IO (ExitCode, Text, Text)
  268 readProcessWithStdin argv stdinBytes = do
  269   let (program, args) = case argv of
  270         (p : as) -> (p, as)
  271         []       -> error "readProcessWithStdin: empty argv (unreachable)"
  272       cp = (proc program args)
  273               { std_in = CreatePipe, std_out = CreatePipe, std_err = CreatePipe
  274               }
  275   withCreateProcess cp $ \mIn mOut mErr ph -> do
  276     case mIn of
  277       Just hIn -> do
  278         BS.hPut hIn stdinBytes
  279         hClose hIn
  280       Nothing -> pure ()
  281     outTxt <- maybe (pure "") TIO.hGetContents mOut
  282     errTxt <- maybe (pure "") TIO.hGetContents mErr
  283     ec <- waitForProcess ph
  284     pure (ec, outTxt, errTxt)

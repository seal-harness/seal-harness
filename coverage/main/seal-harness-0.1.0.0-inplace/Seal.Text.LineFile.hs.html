    2 -- | The reusable line-oriented text-file abstraction.
    3 --
    4 -- A 'LineWindow' is built on 'Seal.Core.Paging'. The pure core ('windowLines')
    5 -- windows a list of lines already split; the bounded IO function
    6 -- 'readLineWindow' reads an opaque 'SafePath' and produces a window WITHOUT
    7 -- materializing the whole file (peak memory is @O(window) + O(1)@). Lines are

    9 -- semantics table in the design spec.

   31 -- | A windowed slice of a line-oriented text file. The @lw*@ counts are
   32 -- 0-based line indices.
   33 data LineWindow = LineWindow
   34   { lwLines     :: ![Text]  -- ^ the windowed lines, in file order
   35   , lwStart     :: !Int     -- ^ 0-based index of first returned line (== 'pgOffset')
   36   , lwEnd       :: !Int     -- ^ 0-based index just past the last returned line
   37                             --   (@lwStart + length lwLines@)
   38   , lwTotal     :: !Int     -- ^ total line count (lower bound if 'lwTruncated')
   39   , lwHasMore   :: !Bool    -- ^ @lwEnd < lwTotal || lwTruncated@
   40   , lwTruncated :: !Bool    -- ^ @True@ if the file exceeded the scan byte ceiling
   41                             --   (then 'lwTotal' is a lower bound)
   42   } deriving stock (Eq, Show)
   43 
   44 -- | Pure: window over a list of lines already split. Built directly on
   45 -- 'Seal.Core.Paging'. @lwTruncated = False@.
   46 windowLines :: PageParams -> Int -> Maybe Int -> [Text] -> LineWindow
   47 windowLines params offset mLimit ls =
   48   let Page items off total hasMore = paginate params offset mLimit ls
   49   in LineWindow
   50        { lwLines     = items
   51        , lwStart     = off
   52        , lwEnd       = off + length items
   53        , lwTotal     = total
   54        , lwHasMore   = hasMore
   55        , lwTruncated = False
   56        }

   58 -- | The compiled-in default scan byte ceiling (>= 65536, the prior FILE_READ
   59 -- | The hard character ceiling for a single 'FILE_READ' response. If the
   60 -- file content exceeds this many characters AND the caller did not supply
   61 -- an explicit @offset@\/@limit@ (i.e. is not paginating), the read is
   62 -- rejected outright — the model must use @offset@\/@limit@ to chunk the
   63 -- file. This mirrors Hermes' @read_file@ behavior: reads exceeding
   64 -- ~100K characters are rejected. When the caller IS paginating (supplied
   65 -- @offset@ or @limit@), no rejection — the window is returned normally.
   66 -- The byte-level @maxScanBytes@ ceiling remains the memory safety bound;
   67 -- this is a content-level guard that improves the model's pagination UX.
   68 maxReadChars :: Int
   69 maxReadChars = 100000   -- 100K chars
   70 
   71 -- bound). Used by callers that do not have an operator-configured ceiling;
   72 -- 'FILE_READ' resolves the ceiling from the @[retrieval]@ config section (via
   73 -- 'Seal.Config.File.retrievalMaxScanBytes') and passes it in, so this default
   74 -- only applies to bare reuse (e.g. tests, future CRUD) that wants the same
   75 -- out-of-the-box bound.
   76 maxScanBytes :: Int
   77 maxScanBytes = 131072   -- 128 KiB

   79 -- | Bounded IO: read an opaque, already-confined 'SafePath' and window it
   80 -- WITHOUT materializing the whole file. The @maxScanBytes@ argument is the
   81 -- byte ceiling for both streaming passes; the caller ('FILE_READ') resolves
   82 -- it from the config-layer operator bound and the call-layer model request
   83 -- (clamped down, never above the operator bound). Two streaming passes over
   84 -- the file 'Handle', each reading at most @maxScanBytes@ bytes:
   85 --
   86 --   1. __Count pass__: stream lines, counting @lwTotal@, holding O(1) memory.
   87 --      The byte ceiling is enforced at the chunk level while accumulating a
   88 --      line (never read a whole line and then check), so a single newline-free
   89 --      file cannot blow the bound before the check runs. If the ceiling is hit
   90 --      before EOF, stop; set @lwTruncated = True@ and let @lwTotal@ be the
   91 --      count so far (a lower bound; may be @0@ for a newline-free file).
   92 --   2. __Window pass__: stream again, @drop lwStart@, @take size@ lines into the
   93 --      window (O(window) memory).
   94 readLineWindow :: PageParams -> Int -> Maybe Int -> Int -> SafePath -> IO LineWindow
   95 readLineWindow params offset mLimit scanBytes safe =
   96   withFile (getSafePath safe) ReadMode $ \h -> do
   97     (total, truncated) <- countLines h scanBytes
   98     hSeek h AbsoluteSeek 0
   99     let size  = windowSize params total mLimit
  100         start = clamp 0 total offset
  101     win <- windowPass h start size scanBytes truncated
  102     pure LineWindow
  103       { lwLines     = win
  104       , lwStart     = start
  105       , lwEnd       = start + length win
  106       , lwTotal     = total
  107       , lwHasMore   = start + length win < total || truncated
  108       , lwTruncated = truncated
  109       }

  111 -- | Count complete lines in the file, streaming, up to @maxScanBytes@.
  112 -- Returns @(count, truncated)@. At EOF (not truncated) a non-empty trailing
  113 -- partial line is counted (matching @Data.Text.lines@). If the byte ceiling is
  114 -- hit before EOF, the in-progress partial line is NOT counted (so @lwTotal@ is
  115 -- a lower bound; a newline-free file yields @0@).
  116 countLines :: Handle -> Int -> IO (Int, Bool)
  117 countLines h scanBytes = do
  118   bytesRef <- newIORef (0 :: Int)
  119   countRef <- newIORef (0 :: Int)
  120   resRef   <- newIORef BS.empty
  121   let go = do
  122         bytes <- readIORef bytesRef
  123         if bytes >= scanBytes
  124           then pure True
  125           else do
  126             chunk <- BS.hGet h 4096
  127             if BS.null chunk
  128               then do
  129                 res <- readIORef resRef
  130                 if BS.null res
  131                   then pure False
  132                   else do modifyIORef' countRef (+ 1); pure False
  133               else do
  134                 -- Enforce the ceiling at the byte/chunk level: truncate this
  135                 -- chunk to the remaining budget so a single read never blows
  136                 -- the bound. If the chunk is clipped, the next loop iteration
  137                 -- sees bytes == scanBytes and stops (truncated).
  138                 let remaining = scanBytes - bytes
  139                     chunk'    = if BS.length chunk <= remaining
  140                                   then chunk
  141                                   else BS.take remaining chunk
  142                     clipped   = BS.length chunk' < BS.length chunk
  143                 modifyIORef' bytesRef (+ BS.length chunk')
  144                 res0 <- readIORef resRef
  145                 let (ls, newRes) = splitLines (res0 <> chunk')
  146                 writeIORef resRef newRes
  147                 modifyIORef' countRef (+ length ls)
  148                 if clipped
  149                   then pure True   -- hit the ceiling mid-chunk
  150                   else go
  151   truncated <- go
  152   count <- readIORef countRef
  153   pure (count, truncated)
  154 
  155 -- | Re-stream the file, drop @start@ complete lines, take up to @size@ into the
  156 -- window. Stops at @maxScanBytes@ or EOF. When @not truncated@, a non-empty
  157 -- trailing partial line at EOF is emitted as a final line (matching
  158 -- @Data.Text.lines@); when @truncated@ the trailing partial is discarded (it
  159 -- was not counted in the count pass).
  160 windowPass :: Handle -> Int -> Int -> Int -> Bool -> IO [Text]
  161 windowPass h start size scanBytes truncated
  162   | size <= 0 = pure []
  163   | otherwise = do
  164       bytesRef    <- newIORef (0 :: Int)
  165       curRef      <- newIORef BS.empty     -- bytes of the in-progress line
  166       skippedRef  <- newIORef (0 :: Int)   -- complete lines passed over
  167       takenRef    <- newIORef (0 :: Int)   -- lines added to the window
  168       winRef      <- newIORef ([] :: [Text])  -- collected in reverse
  169       let stop = do taken <- readIORef takenRef; pure (taken >= size)
  170           processChunk chunk = do
  171             cur0 <- readIORef curRef
  172             let (completeLines, residual) = splitLines (cur0 <> chunk)
  173             writeIORef curRef residual
  174             mapM_ emitLine completeLines
  175           emitLine lbs = do
  176             skipped <- readIORef skippedRef
  177             taken   <- readIORef takenRef
  178             if skipped < start
  179               then modifyIORef' skippedRef (+ 1)
  180               else if taken < size
  181                      then do modifyIORef' winRef (TE.decodeUtf8Lenient lbs :)
  182                              modifyIORef' takenRef (+ 1)
  183                      else modifyIORef' skippedRef (+ 1)
  184           go = do
  185             done <- stop
  186             if done
  187               then pure ()
  188               else do
  189                 bytes <- readIORef bytesRef
  190                 if bytes >= scanBytes
  191                   then pure ()
  192                   else do
  193                     chunk <- BS.hGet h 4096
  194                     if BS.null chunk
  195                       then pure ()
  196                       else do
  197                         -- Truncate this chunk to the remaining byte budget so
  198                         -- a single read never blows the bound; matches the
  199                         -- count pass.
  200                         let remaining = scanBytes - bytes
  201                             chunk'    = if BS.length chunk <= remaining
  202                                           then chunk
  203                                           else BS.take remaining chunk
  204                         modifyIORef' bytesRef (+ BS.length chunk')
  205                         processChunk chunk'
  206                         go
  207       go
  208       -- Flush a trailing partial line at EOF (only when not truncated; if we
  209       -- stopped at the byte ceiling, the partial is not a counted line).
  210       cur     <- readIORef curRef
  211       taken   <- readIORef takenRef
  212       skipped <- readIORef skippedRef
  213       when (not truncated && taken < size && not (BS.null cur) && skipped >= start)
  214         (modifyIORef' winRef (TE.decodeUtf8Lenient cur :))
  215       reverse <$> readIORef winRef

  217 -- | Split a ByteString into complete lines (without their trailing newline)
  218 -- plus the residual partial line. A line is complete when it ends with a
  219 -- newline (@0x0A@); the newline is consumed. Matches @Data.Text.lines@: a
  220 -- trailing newline does NOT produce an empty final line, and a final partial
  221 -- line (no trailing newline) is returned as the residual.
  222 splitLines :: BS.ByteString -> ([BS.ByteString], BS.ByteString)
  223 splitLines s
  224   | BS.null s = ([], BS.empty)
  225   | otherwise =
  226       case BS.elemIndex 0x0A s of
  227         Nothing -> ([], s)
  228         Just i  ->
  229           let line = BS.take i s
  230               rest = BS.drop (i + 1) s
  231               (ls, residual) = splitLines rest
  232           in (line : ls, residual)

  234 -- | Render the window: content + a machine-actionable footer telling the model
  235 -- how to page forward. Lines are joined with @"\n"@ (@T.intercalate@,
  236 -- preserving line content), then a blank line, then exactly one footer line.
  237 -- Footers use ASCII hyphen-minus and 1-based inclusive display line numbers
  238 -- (@lwStart+1 .. lwEnd@); the @offset=@ value in the footer is the **0-based**
  239 -- @lwEnd@ (copy-paste ready).
  240 --
  241 -- The footer states overlap (an empty window at end-of-file satisfies both
  242 -- "offset past end" and "final"; a newline-free file over the cap satisfies
  243 -- both "empty" and "truncated"), so the footer is chosen by a **total, ordered**
  244 -- guard — first match wins. Implemented in exactly this order.
  245 renderWindow :: LineWindow -> Text
  246 renderWindow lw =
  247   let body   = T.intercalate "\n" (lwLines lw)
  248       footer = renderFooter lw
  249   in if T.null body then footer else body <> "\n\n" <> footer
  250 
  251 renderFooter :: LineWindow -> Text
  252 renderFooter lw =
  253   let n     = lwTotal lw
  254       s     = lwStart lw
  255       e     = lwEnd lw
  256       trunc = lwTruncated lw
  257   in
  258     -- Guard 1: empty file but truncated (newline-free file over the cap).
  259     if n == 0 && trunc
  260       then "[no line break within the scan limit; file may be a single long line or non-line-oriented]"
  261     -- Guard 2: genuinely empty file (0 lines, not truncated).
  262     else if n == 0
  263       then "[empty file (0 lines)]"
  264     -- Guard 3: empty window on a non-empty file (offset at/past the counted end).
  265     else if s == e
  266       then if trunc
  267              then "[offset " <> tshow s <> " reached the scan limit (" <> tshow n <> "+ lines counted so far); read with offset=0 to restart]"
  268              else "[offset " <> tshow s <> " is past end of file (" <> tshow n <> " lines); read with offset=0 to start over]"
  269     -- Guards 4-6 fire only when s < e, so no inverted range is possible.
  270     -- Guard 4: truncated (file exceeds scan limit; more may exist).
  271     else if trunc
  272       then "[lines " <> tshow (s + 1) <> "-" <> tshow e <> " of >=" <> tshow n <> " (file exceeds scan limit; more may exist) - read with offset=" <> tshow e <> " for the next window]"
  273     -- Guard 5: more lines remain.
  274     else if e < n
  275       then "[lines " <> tshow (s + 1) <> "-" <> tshow e <> " of " <> tshow n <> "; " <> tshow (n - e) <> " more - read with offset=" <> tshow e <> " for the next window]"
  276     -- Guard 6: final window (e == n, s < e).
  277     else
  278       "[lines " <> tshow (s + 1) <> "-" <> tshow e <> " of " <> tshow n <> " (end of file)]"

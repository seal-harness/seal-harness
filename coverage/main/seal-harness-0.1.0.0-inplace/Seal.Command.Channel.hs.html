   74 -- | The runtime the @\/channel@ command closes over: where to find the
   75 -- config file (so the wizard can read-modify-write @config.toml@), a
   76 -- 'SignalCli' seam for shelling out to @signal-cli@, a 'TelegramBotApi'
   77 -- seam for the Telegram Bot API, and a 'VaultStore' seam for storing
   78 -- channel secrets in the vault. Tests inject mock seams so @cabal test@
   79 -- never needs the binary, network, or vault.
   80 data ChannelRuntime = ChannelRuntime
   81   { crConfigPath    :: FilePath
   82   , crSignalCli     :: SignalCli
   83   , crTelegramBotApi :: TelegramBotApi
   84   , crVaultStore    :: VaultStore
   85   }
   86 
   87 -- | A seam over the vault for storing channel secrets (like the Telegram
   88 -- bot token). The wizard writes the token to the vault (not @config.toml@)
   89 -- so it is never stored in cleartext. 'vsStore' stores a key+value;
   90 -- 'vsAvailable' checks whether the vault is open and ready.
   91 data VaultStore = VaultStore
   92   { vsStore     :: Text -> ByteString -> IO (Either Text ())
   93     -- ^ Store a secret under a key. 'Right ()' on success; 'Left' diagnostic.
   94   , vsAvailable :: IO Bool
   95     -- ^ Is the vault open and ready to accept writes?
   96   }
   97 
   98 -- | A small seam over the @signal-cli@ binary. Each field is one subprocess
   99 -- operation the wizard needs. 'mkRealSignalCli' shells out; tests supply an
  100 -- in-memory 'SignalCli'. The seam keeps the wizard logic pure with respect
  101 -- to the binary and makes the interactive flow unit-testable.
  102 data SignalCli = SignalCli
  103   { scCheckInstalled :: IO (Either Text Text)
  104     -- ^ @signal-cli --version@. Right version-string on success; Left
  105     -- diagnostic if absent or failing.
  106   , scLink           :: (Text -> IO ()) -> IO LinkOutcome
  107     -- ^ @signal-cli link -n Seal@. The callback is invoked with the
  108     -- @sgnl:\/\/@ URI the instant it appears (before the scan completes),
  109     -- so the wizard can show it immediately. 'scLink' then blocks until
  110     -- the user scans it (or signal-cli times out). See 'LinkOutcome'.
  111   , scRegister       :: Text -> Maybe Text -> IO RegisterOutcome
  112     -- ^ @signal-cli -u <phone> register [--captcha <token>]@.
  113   , scVerify         :: Text -> Text -> IO VerifyOutcome
  114     -- ^ @signal-cli -u <phone> verify <code>@.
  115   , scListAccounts   :: IO AccountsOutcome
  116     -- ^ @signal-cli listAccounts@. Used to detect the linked account.
  117   , scReceiveSender  :: Text -> Int -> IO ReceiveOutcome
  118     -- ^ @signal-cli -u <account> receive --max-messages 1 -t <timeout>@.
  119     -- Blocks until one message arrives or the timeout elapses. Used by the
  120     -- wizard to capture the sender's user id and lock down @allow_from@.
  121   }
  122 
  123 -- | Outcome of @signal-cli link@: the @sgnl:\/\/@ URI to show the user (or a
  124 -- failure), then the exit status after the user scans. The URI is emitted
  125 -- before blocking, so the wizard can show it immediately.
  126 data LinkOutcome
  127   = LinkFailed Text
  128   | LinkTimedOut
  129     -- ^ signal-cli's provisioning WebSocket closed before the scan completed
  130     -- (60s idle timeout). The wizard may retry to give the user a fresh window.
  131   | LinkSucceeded Text
  132     -- ^ The @sgnl:\/\/@ URI; the scan completed with @ExitSuccess@.
  133 
  134 -- | Outcome of @signal-cli -u <phone> register@.
  135 data RegisterOutcome
  136   = RegisterOk
  137   | RegisterCaptchaRequired
  138   | RegisterFailed Text
  139 
  140 -- | Outcome of @signal-cli -u <phone> verify <code>@.
  141 data VerifyOutcome
  142   = VerifyOk
  143   | VerifyFailed Text
  144 
  145 -- | Outcome of @signal-cli listAccounts@.
  146 data AccountsOutcome
  147   = AccountsFailed Text
  148   | AccountsFound [Text]
  149     -- ^ The detected @+…@ phone numbers (E.164).
  150 
  151 -- | Outcome of @signal-cli receive@ (one message, with a timeout). The
  152 -- sender's user id is the phone number signal-cli reports in the envelope's
  153 -- @source@ field (used to populate @allow_from@).
  154 data ReceiveOutcome
  155   = ReceiveFailed Text
  156   | ReceiveTimedOut
  157   | ReceiveSender Text

  163 -- | A small seam over the Telegram Bot API. The wizard needs only one

  176   = GetMeOk Text
  177     -- ^ The bot's username (without the leading @).
  178   | GetMeFailed Text

  218 -- | Run the Signal setup wizard. Checks @signal-cli@, offers link or
  219 -- register, walks the chosen flow, writes config. All interaction goes
  220 -- through 'ChannelCaps' (sends + prompts) so the wizard works on the CLI
  221 -- TUI and any interactive channel.
  222 signalSetupCmd :: ChannelRuntime -> CommandAction
  223 signalSetupCmd rt = commandAction $ \caps -> do
  224   let cli = crSignalCli rt
  225   -- Step 1: Check signal-cli is installed.
  226   eVer <- scCheckInstalled cli
  227   case eVer of
  228     Left _ -> ccSend caps $ T.intercalate "\n"
  229       [ "signal-cli is not installed."
  230       , ""
  231       , "Install it first:"
  232       , "  macOS:  brew install signal-cli"
  233       , "  Nix:    nix-env -i signal-cli"
  234       , "  Other:  https://github.com/AsamK/signal-cli"
  235       , ""
  236       , "Then run /channel signal again."
  237       ]
  238     Right version -> do
  239       ccSend caps ("Found signal-cli " <> version)
  240       -- Step 2: Offer link or register.
  241       ccSend caps $ T.intercalate "\n"
  242         [ ""
  243         , "How would you like to connect?"
  244         , "  [1] Link to an existing Signal account (adds Seal as secondary device)"
  245         , "  [2] Register with a phone number (becomes primary device for that number)"
  246         , ""
  247         , "Note: Option 2 will take over the number from any existing Signal registration."
  248         ]
  249       choice <- T.strip <$> ccPrompt caps (AskPrompt "Choice [1]: " [])
  250       let effectiveChoice = if T.null choice then "1" else choice
  251       case effectiveChoice of
  252         "1" -> signalLinkFlow rt caps
  253         "2" -> signalRegisterFlow rt caps
  254         _   -> ccSend caps "Invalid choice. Setup cancelled."

  256 -- | Link to an existing Signal account by scanning a @sgnl:\/\/@ URI.
  257 -- Retries up to 3 times on timeout (signal-cli's provisioning WebSocket has a
  258 -- fixed 60s idle timeout); each retry mints a fresh URI + QR so the user
  259 -- gets a new window. A real failure exits immediately.
  260 signalLinkFlow :: ChannelRuntime -> ChannelCaps -> IO ()
  261 signalLinkFlow rt caps = linkAttempt (3 :: Int)
  262   where
  263     linkAttempt 0 = ccSend caps "Link timed out. Run /channel signal again to retry."
  264     linkAttempt triesLeft = do
  265       ccSend caps "Generating link... (this may take a moment)"
  266       let onUri uri = do
  267             mQr <- renderQR uri
  268             let qrBlock = case mQr of
  269                   Just qr -> [ "Scan this QR code with Signal on your phone:"
  270                              , ""
  271                              , qr
  272                              , ""
  273                              , "Or paste this link into a QR code generator:"
  274                              ]
  275                   Nothing -> [ "Scan this link (or paste into a QR code generator):" ]
  276             ccSend caps $ T.intercalate "\n"
  277               ( [ "Open Signal on your phone:"
  278                 , "  Settings \x2192 Linked Devices \x2192 Link New Device"
  279                 , ""
  280                 ] <> qrBlock <> [ ""
  281                 , "  " <> uri
  282                 , ""
  283                 , "Waiting for you to scan... (this will complete automatically)"
  284                 ] )
  285       outcome <- scLink (crSignalCli rt) onUri
  286       case outcome of
  287         LinkFailed err -> ccSend caps ("Link failed or was cancelled.\n" <> err)
  288         LinkTimedOut -> do
  289           ccSend caps "Link timed out after 60s. Generating a fresh link..."
  290           linkAttempt (triesLeft - 1)
  291         LinkSucceeded _uri -> do
  292           ccSend caps "Linked successfully!"
  293           detectAndWriteSignalConfig rt caps

  295 -- | Register a new phone number (primary device). Prompts for the E.164
  296 -- number, calls @signal-cli register@ (handling a captcha challenge if
  297 -- required), then @verify@.
  298 signalRegisterFlow :: ChannelRuntime -> ChannelCaps -> IO ()
  299 signalRegisterFlow rt caps = do
  300   phoneNumber <- T.strip <$> ccPrompt caps (AskPrompt "Phone number (E.164 format, e.g. +15555550123): " [])
  301   if T.null phoneNumber || not ("+" `T.isPrefixOf` phoneNumber)
  302     then ccSend caps "Invalid phone number. Must start with + (E.164 format)."
  303     else signalRegister rt caps phoneNumber Nothing
  304 
  305 -- | Attempt @signal-cli -u <phone> register@, handling the captcha loop.
  306 signalRegister :: ChannelRuntime -> ChannelCaps -> Text -> Maybe Text -> IO ()
  307 signalRegister rt caps phoneNumber mCaptcha = do
  308   let cli = crSignalCli rt
  309   ccSend caps ("Sending verification SMS to " <> phoneNumber <> "...")
  310   result <- scRegister cli phoneNumber mCaptcha
  311   case result of
  312     RegisterOk -> signalVerify rt caps phoneNumber
  313     RegisterCaptchaRequired -> do
  314       ccSend caps $ T.intercalate "\n"
  315         [ "Signal requires a captcha before sending the SMS."
  316         , ""
  317         , "1. Open this URL in a browser:"
  318         , "   https://signalcaptchas.org/registration/generate.html"
  319         , "2. Solve the captcha"
  320         , "3. Open DevTools (F12), go to Network tab"
  321         , "4. Click \"Open Signal\" \x2014 find the signalcaptcha:// URL in the Network tab"
  322         , "5. Copy and paste the full URL here (starts with signalcaptcha://)"
  323         ]
  324       captchaInput <- T.strip <$> ccPrompt caps (AskPrompt "Captcha token: " [])
  325       let token = T.strip (T.replace "signalcaptcha://" "" captchaInput)
  326       if T.null token
  327         then ccSend caps "No captcha provided. Setup cancelled."
  328         else signalRegister rt caps phoneNumber (Just token)
  329     RegisterFailed err -> ccSend caps ("Registration failed: " <> err)
  330 
  331 -- | Verify a phone number after the registration SMS was sent.
  332 signalVerify :: ChannelRuntime -> ChannelCaps -> Text -> IO ()
  333 signalVerify rt caps phoneNumber = do
  334   ccSend caps "Verification code sent! Check your SMS."
  335   code <- T.strip <$> ccPrompt caps (AskPrompt "Verification code: " [])
  336   result <- scVerify (crSignalCli rt) phoneNumber code
  337   case result of
  338     VerifyOk -> do
  339       ccSend caps "Phone number verified!"
  340       writeSignalConfig rt caps phoneNumber
  341     VerifyFailed err -> ccSend caps ("Verification failed: " <> err)

  343 -- | Detect the linked account number via @signal-cli listAccounts@, then
  344 -- write config. Falls back to prompting if detection fails.
  345 detectAndWriteSignalConfig :: ChannelRuntime -> ChannelCaps -> IO ()
  346 detectAndWriteSignalConfig rt caps = do
  347   result <- scListAccounts (crSignalCli rt)
  348   case result of
  349     AccountsFailed _ -> do
  350       phoneNumber <- T.strip <$> ccPrompt caps (AskPrompt "What phone number was linked? (E.164 format): " [])
  351       writeSignalConfig rt caps phoneNumber
  352     AccountsFound (phone : _) -> do
  353       ccSend caps ("Detected account: " <> phone)
  354       writeSignalConfig rt caps phone
  355     AccountsFound [] -> do
  356       phoneNumber <- T.strip <$> ccPrompt caps (AskPrompt "Could not detect account. Phone number (E.164 format): " [])
  357       writeSignalConfig rt caps phoneNumber

  359 -- | Write the @[signal]@ section into @config.toml@ and confirm. Preserves
  360 -- all other config; sets the account and a permissive initial DM policy
  361 -- (@AllowAll@), then immediately prompts the user to send a message and
  362 -- locks down @allow_from@ to the captured sender (see 'lockDownAllowFrom').
  363 writeSignalConfig :: ChannelRuntime -> ChannelCaps -> Text -> IO ()
  364 writeSignalConfig rt caps phoneNumber = do
  365   let cfgPath = crConfigPath rt
  366   createDirectoryIfMissing True (takeDirectory cfgPath)
  367   existing <- loadRuntimeConfig cfgPath
  368   let baseCfg = fromRight defaultRuntimeConfig existing
  369       updated = baseCfg
  370         { rcSignal = Just SignalConfig
  371             { scAccount        = Just phoneNumber
  372             , scTextChunkLimit = Just defaultSignalChunkLimit
  373             , scAllowFrom      = AllowAll
  374             }
  375         }
  376   saveRuntimeConfig cfgPath updated
  377   ccSend caps $ T.intercalate "\n"
  378     [ ""
  379     , "Signal configured!"
  380     , "  Account: " <> phoneNumber
  381     , "  Default channel: signal"
  382     , ""
  383     , "One more step to secure your setup:"
  384     , "  Send any message from Signal to " <> phoneNumber
  385     , "  and I'll lock down access to just that sender."
  386     ]
  387   lockDownAllowFrom rt caps phoneNumber
  388 
  389 -- | Prompt the user to send a Signal message, capture the sender's user id
  390 -- via @signal-cli receive@, and rewrite the @[signal].allow_from@ to
  391 -- 'AllowOnly' that sender. Retries up to 3 times on timeout (120s each). On
  392 -- failure, leaves the config at 'AllowAll' with a warning to lock down
  393 -- manually. Idempotent: only writes when a sender is captured.
  394 lockDownAllowFrom :: ChannelRuntime -> ChannelCaps -> Text -> IO ()
  395 lockDownAllowFrom rt caps account = lockAttempt (3 :: Int)
  396   where
  397     lockAttempt 0 = ccSend caps $ T.intercalate "\n"
  398       [ ""
  399       , "No message received. Your config currently accepts messages from"
  400       , "anyone. To lock it down later, edit the [signal] section of your"
  401       , "config and set allow_from = [\"<your-phone-or-uuid>\"]."
  402       ]
  403     lockAttempt triesLeft = do
  404       ccSend caps "Waiting for your message... (this may take a moment)"
  405       result <- scReceiveSender (crSignalCli rt) account 120
  406       case result of
  407         ReceiveFailed err -> ccSend caps ("Receiving failed: " <> err)
  408         ReceiveTimedOut -> do
  409           ccSend caps "No message received in time. Let's try again."
  410           lockAttempt (triesLeft - 1)
  411         ReceiveSender sender -> do
  412           let cfgPath = crConfigPath rt
  413           existing <- loadRuntimeConfig cfgPath
  414           let baseCfg = fromRight defaultRuntimeConfig existing
  415               mSig = rcSignal baseCfg
  416               sigCfg = fromMaybe SignalConfig
  417                          { scAccount = Just account
  418                          , scTextChunkLimit = Just defaultSignalChunkLimit
  419                          , scAllowFrom = AllowAll
  420                          } mSig
  421               updated = baseCfg
  422                 { rcSignal = Just sigCfg { scAllowFrom = AllowOnly (Set.singleton sender) }
  423                 }
  424           saveRuntimeConfig cfgPath updated
  425           ccSend caps $ T.intercalate "\n"
  426             [ ""
  427             , "Locked down! Access is now restricted to: " <> sender
  428             , ""
  429             , "To start chatting:"
  430             , "  1. Restart Seal (or run: seal signal)"
  431             , "  2. Open Signal on your phone"
  432             , "  3. Send a message to " <> account
  433             , ""
  434             , "To allow more senders later, edit " <> T.pack cfgPath <> ":"
  435             , "  [signal]"
  436             , "  allow_from = [\"" <> sender <> "\", \"<other-sender>\"]"
  437             ]
  438

  443 -- | Run the Telegram setup wizard. Prompts for a BotFather token, validates
  444 -- it via the Bot API @getMe@ (so a typo is caught here, not at runtime),
  445 -- stores the token in the vault (NOT in @config.toml@), and writes the
  446 -- @[telegram]@ config section with the non-secret fields. All interaction
  447 -- goes through 'ChannelCaps' (sends + prompts) so the wizard works on the
  448 -- CLI TUI and any interactive channel.
  449 telegramSetupCmd :: ChannelRuntime -> CommandAction
  450 telegramSetupCmd rt = commandAction $ \caps -> do
  451   ccSend caps $ T.intercalate "\n"
  452     [ "Setting up the Telegram channel."
  453     , ""
  454     , "You need a bot token from BotFather (@BotFather on Telegram):"
  455     , "  1. Open a chat with @BotFather"
  456     , "  2. Send /newbot and follow the prompts"
  457     , "  3. Copy the token it gives you (looks like 123456:ABC-DEF...)"
  458     ]
  459   tokenInput <- T.strip <$> ccPrompt caps (AskPrompt "Bot token: " [])
  460   case mkTelegramToken tokenInput of
  461     Left err -> ccSend caps ("Invalid token: " <> err)
  462     Right token -> do
  463       ccSend caps "Validating token with Telegram..."
  464       result <- tbaGetMe (crTelegramBotApi rt) (telegramTokenText token)
  465       case result of
  466         GetMeFailed err -> ccSend caps ("Token validation failed: " <> err)
  467         GetMeOk username -> do
  468           ccSend caps ("Connected! Bot username: @" <> username)
  469           storeTelegramToken rt caps (telegramTokenText token) username
  470 
  471 -- | Store the token in the vault and write the @[telegram]@ config section
  472 -- (without the token — only non-secret fields). If the vault is not
  473 -- available, warns the user to set up the vault first.
  474 storeTelegramToken :: ChannelRuntime -> ChannelCaps -> Text -> Text -> IO ()
  475 storeTelegramToken rt caps token username = do
  476   let store = crVaultStore rt
  477   available <- vsAvailable store
  478   if not available
  479     then ccSend caps $ T.intercalate "\n"
  480       [ ""
  481       , "Vault is not set up or not unlocked."
  482       , "The Telegram bot token is a secret and must be stored in the vault."
  483       , ""
  484       , "Run /vault setup first, then /channel telegram again."
  485       ]
  486     else do
  487       result <- vsStore store telegramVaultKey (TE.encodeUtf8 token)
  488       case result of
  489         Left err -> ccSend caps ("Failed to store token in vault: " <> err)
  490         Right () -> writeTelegramConfig rt caps username
  491 
  492 -- | Write the @[telegram]@ section into @config.toml@ (without the token)
  493 -- and confirm. The token is stored in the vault, not the config file.
  494 -- Preserves all other config; sets a permissive default DM policy
  495 -- (@AllowAll@), which the user can tighten later by editing @allow_from@.
  496 writeTelegramConfig :: ChannelRuntime -> ChannelCaps -> Text -> IO ()
  497 writeTelegramConfig rt caps username = do
  498   let cfgPath = crConfigPath rt
  499   createDirectoryIfMissing True (takeDirectory cfgPath)
  500   existing <- loadRuntimeConfig cfgPath
  501   let baseCfg = fromRight defaultRuntimeConfig existing
  502       updated = baseCfg
  503         { rcTelegram = Just TelegramConfig
  504             { tcToken         = Nothing  -- token lives in the vault, not config
  505             , tcTextChunkLimit = Just defaultTelegramChunkLimit
  506             , tcAllowFrom      = AllowAll
  507             }
  508         }
  509   saveRuntimeConfig cfgPath updated
  510   ccSend caps $ T.intercalate "\n"
  511     [ ""
  512     , "Telegram configured!"
  513     , "  Bot: @" <> username
  514     , "  Token: stored in vault (key: " <> telegramVaultKey <> ")"
  515     , "  DM policy: open (accepts messages from anyone)"
  516     , ""
  517     , "To start chatting:"
  518     , "  1. Restart Seal (or run: seal telegram)"
  519     , "  2. Open Telegram and DM @" <> username
  520     , ""
  521     , "To restrict access later, edit " <> T.pack cfgPath <> ":"
  522     , "  [telegram]"
  523     , "  allow_from = [\"<your-telegram-user-id>\"]"
  524     , ""
  525     , "Your Telegram user id will appear in the logs on first message."
  526     ]

  531 
  532 -- | The real 'SignalCli': every field shells out to the @signal-cli@ binary
  533 -- via "System.Process". Returned by 'mkRealSignalCli'. If @signal-cli@ is
  534 -- absent the fields surface a clear 'Left' (the wizard shows an install
  535 -- hint and bails).
  536 mkRealSignalCli :: IO SignalCli
  537 mkRealSignalCli = pure SignalCli
  538   { scCheckInstalled = do
  539       (ec, out, _err) <- readProcessNoInput "signal-cli" ["--version"]
  540       pure $ case ec of
  541         ExitSuccess ->
  542           Right (T.strip (decodeUtf8 out))

  549         (ec, _out, err) <- readProcessNoInput "signal-cli" args

  551         pure $ case ec of
  552           ExitSuccess -> RegisterOk
  553           _ | "captcha" `T.isInfixOf` T.toLower errText -> RegisterCaptchaRequired
  554             | otherwise -> RegisterFailed errText

  559       pure $ case ec of
  560         ExitSuccess -> VerifyOk
  561         _ -> VerifyFailed errText

  563       (ec, out, _err) <- readProcessNoInput "signal-cli" ["listAccounts"]

  565         ExitSuccess ->

  567                          (map T.strip (T.lines (decodeUtf8 out)))
  568           in AccountsFound phones

  591 -- | The real 'TelegramBotApi': validates a bot token via a GET request to
  592 -- @https:\/\/api.telegram.org\/bot<token>\/getMe@. Uses a shared TLS manager.
  593 -- On a non-200 response or a network error the wizard surfaces a clear
  594 -- 'GetMeFailed'. The bot's username is extracted from the JSON response body
  595 -- (the @\"result\": {\"username\": \"...\"}@ field). We avoid pulling in a

  597 -- text scan — the Bot API response shape for getMe is stable.
  598 mkRealTelegramBotApi :: IO TelegramBotApi

  600   mgr <- newTlsManager
  601   pure TelegramBotApi
  602     { tbaGetMe = validateToken mgr

  609 -- | Build a 'VaultStore' from a 'VaultHandle'. 'vsStore' encodes the text
  610 -- to a ByteString and calls 'vhPut'; 'vsAvailable' checks that the vault
  611 -- is open (not locked) by calling 'vhStatus'.
  612 mkRealVaultStore :: Maybe Vault.VaultHandle -> IO VaultStore
  613 mkRealVaultStore mVh = pure $ case mVh of
  614   Nothing -> noVaultStore
  615   Just vh -> VaultStore
  616     { vsStore = \key val -> do
  617         r <- Vault.vhPut vh key val
  618         pure $ case r of
  619           Right () -> Right ()
  620           Left err -> Left (T.pack (show err))
  621     , vsAvailable = do
  622         st <- Vault.vhStatus vh
  623         pure (not (Vault.vsLocked st))
  624     }
  625 
  626 -- | A 'VaultStore' that is never available (no vault configured). The
  627 -- wizard shows a "set up the vault first" message when this is used.
  628 noVaultStore :: VaultStore
  629 noVaultStore = VaultStore
  630   { vsStore = \_ _ -> pure (Left "vault not configured")
  631   , vsAvailable = pure False
  632   }

  634 -- | Perform the @getMe@ request and extract the bot username. Returns
  635 -- 'GetMeOk' with the username (without the leading @) on success,
  636 -- 'GetMeFailed' with a diagnostic otherwise.
  637 validateToken :: Manager -> Text -> IO GetMeOutcome
  638 validateToken mgr token = do
  639   let url = "https://api.telegram.org/bot" <> T.unpack token <> "/getMe"
  640   eReq <- try @SomeException (parseRequest url)
  641   case eReq of
  642     Left ex -> pure (GetMeFailed ("request error: " <> T.pack (show ex)))
  643     Right req -> do
  644       eResp <- try @SomeException (httpLbs req mgr)
  645       case eResp of
  646         Left ex -> pure (GetMeFailed ("network error: " <> T.pack (show ex)))
  647         Right resp ->
  648           let code = statusCode (responseStatus resp)
  649               body = decodeUtf8 (BL.toStrict (responseBody resp))
  650           in if code == 200
  651                then case extractUsername body of
  652                  Just u  -> pure (GetMeOk u)
  653                  Nothing -> pure (GetMeFailed "could not parse username from getMe response")
  654                else pure (GetMeFailed ("getMe returned HTTP " <> T.pack (show code) <> ": " <> body))
  655 
  656 -- | Extract the @\"username\"@ field value from a getMe JSON response body
  657 -- via a lightweight text scan: find @\"username\"@ then the next quoted
  658 -- string. The Bot API response shape is stable: @{"ok":true,"result":{"id":...,"is_bot":true,"first_name":"...","username":"..."}}@.
  659 -- Returns the username without the leading @.
  660 extractUsername :: Text -> Maybe Text
  661 extractUsername body = do
  662   let key = "\"username\""
  663       afterKey = T.dropWhile (`notElem` ("\"" :: String))
  664                    (T.drop (T.length key)
  665                            (snd (T.breakOn key body)))
  666   -- afterKey starts with "username-value"..., drop the leading quote
  667   case T.uncons (T.drop 1 afterKey) of
  668     Just (_, rest) ->
  669       let val = T.takeWhile (/= '"') rest
  670       in if T.null val then Nothing else Just val
  671     Nothing -> Nothing

  673 -- ---------------------------------------------------------------------------
  674 -- runLink — signal-cli link subprocess
  675 -- ---------------------------------------------------------------------------
  676 runLink :: (Text -> IO ()) -> IO LinkOutcome
  677 runLink onUri =
  678   withCreateProcess
  679     ( (proc "signal-cli" ["link", "-n", "Seal"])
  680         { std_out = CreatePipe, std_err = CreatePipe }
  681     ) $ \_ mOut mErr ph -> do
  682       let stderrH = fromMaybe (error "runLink: no stderr handle") mErr
  683           stdoutH = fromMaybe (error "runLink: no stdout handle") mOut
  684       hSetBuffering stderrH LineBuffering
  685       hSetBuffering stdoutH LineBuffering
  686       mUri <- readUntilLink stdoutH stderrH
  687       case mUri of
  688         Nothing -> do
  689           _ <- waitForProcess ph
  690           pure (LinkFailed "no sgnl:// URI found in signal-cli link output")
  691         Just uri -> do
  692           onUri uri
  693           ec <- waitForProcess ph
  694           case ec of
  695             ExitSuccess -> pure (LinkSucceeded uri)
  696             _ -> do
  697               -- Read all remaining stderr (signal-cli emits many debug log
  698               -- lines before the "Connection closed!" error; a single
  699               -- hGetLine would miss it). Process has exited so the handle
  700               -- is at EOF — hGetContents returns immediately.
  701               errRest <- BS.hGetContents stderrH
  702               let errText = decodeUtf8 errRest
  703               if "Connection closed" `T.isInfixOf` errText
  704                 then pure LinkTimedOut
  705                 else pure (LinkFailed (T.strip errText))
  706 
  707 -- | Read lines from stdout then stderr looking for a @sgnl:\/\/@ URI. Max
  708 -- 50 lines to prevent an infinite loop on a misbehaving signal-cli.
  709 readUntilLink :: Handle -> Handle -> IO (Maybe Text)
  710 readUntilLink stdoutH stderrH = go (50 :: Int)
  711   where
  712     go 0 = pure Nothing
  713     go n = do
  714       lineResult <- try @IOException (hGetLine stdoutH)
  715       case lineResult of
  716         Right line ->
  717           let t = T.pack line
  718           in if "sgnl://" `T.isInfixOf` t
  719              then pure (Just (T.strip t))
  720              else go (n - 1)
  721         Left _ -> do
  722           errResult <- try @IOException (hGetLine stderrH)
  723           case errResult of
  724             Right line ->
  725               let t = T.pack line
  726               in if "sgnl://" `T.isInfixOf` t
  727                  then pure (Just (T.strip t))
  728                  else go (n - 1)
  729             Left _ -> pure Nothing

  731 -- | Read a process's stdout/stderr as 'ByteString' and wait for it. Mirrors
  732 -- 'Seal.Vault.Backend.readProcessNoInput' (kept local to avoid a cross-
  733 -- module dependency for one helper).
  734 readProcessNoInput :: FilePath -> [String] -> IO (ExitCode, ByteString, ByteString)
  735 readProcessNoInput cmdPath args =
  736   withCreateProcess
  737     ( (proc cmdPath args)
  738         { std_in = NoStream, std_out = CreatePipe, std_err = CreatePipe }
  739     ) $ \_ mOut mErr ph -> do
  740       (hOut, hErr) <- case (mOut, mErr) of
  741         (Just a, Just b) -> pure (a, b)
  742         _ -> error "readProcessNoInput: pipe creation failed (unreachable)"
  743       out <- BS.hGetContents hOut
  744       err <- BS.hGetContents hErr
  745       ec  <- waitForProcess ph
  746       let !_ = BS.length out
  747           !_ = BS.length err
  748       pure (ec, out, err)

  750 decodeUtf8 :: ByteString -> Text
  751 decodeUtf8 = decodeUtf8With lenientDecode

  753 -- | Render a @sgnl:\/\/@ URI (or any text) as an ANSI QR code by shelling out
  754 -- to @qrencode -t ANSIUTF8@. Returns 'Nothing' if @qrencode@ is not installed
  755 -- or exits non-zero, so the wizard can fall back to showing the raw URI. The
  756 -- QR is emitted immediately (before the 60s scan window opens) so the user
  757 -- has the full window to scan it.
  758 renderQR :: Text -> IO (Maybe Text)
  759 renderQR uri = do
  760   eResult <- try @IOException $
  761     readProcessNoInput "qrencode" ["-t", "ANSIUTF8", "-o", "-", T.unpack uri]
  762   pure $ case eResult of
  763     Left _ -> Nothing
  764     Right (ExitSuccess, out, _)
  765       | not (BS.null out) -> Just (T.strip (decodeUtf8 out))
  766     Right _ -> Nothing

  768 -- | Extract the @source@ (sender phone number) from one
  769 -- @signal-cli receive --output=json@ envelope. The output is a JSON array of
  770 -- envelope objects (or a single object); each has @envelope.source@. Returns
  771 -- the first sender found, or 'Nothing' if the shape doesn't match.
  772 parseReceiveSender :: Text -> Maybe Text
  773 parseReceiveSender body =
  774   case A.decode (BL.fromStrict (TE.encodeUtf8 body)) of
  775     Just (vals :: [A.Value]) -> firstSource vals
  776     Nothing                  ->
  777       case A.decode (BL.fromStrict (TE.encodeUtf8 body)) of
  778         Just (val :: A.Value) -> sourceOf val
  779         Nothing              -> Nothing
  780   where
  781     sourceOf v = do
  782       o <- asObject v
  783       env <- KeyMap.lookup (Key.fromString "envelope") o
  784       envO <- asObject env
  785       A.String src <- KeyMap.lookup (Key.fromString "source") envO
  786       Just src
  787     firstSource = foldr (\v acc -> sourceOf v <|> acc) Nothing
  788     asObject (A.Object o) = Just o
  789     asObject _            = Nothing

   81 -- | Entry point for @seal access-control@.
   82 runAccessControl :: AccessControlAction -> IO ExitCode
   83 runAccessControl action = do
   84   paths <- getSealPaths
   85   -- Scaffold the two config files on first use, exactly as `serve` and `tui` do.
   86   -- This is not decorative here: `disable`, `reset` and `migrate-credentials`
   87   -- write security.toml, and 'Seal.Config.Security.saveSecurityConfig' creates it
   88   -- when absent — so without this a fresh install that runs one of those first
   89   -- would keep a comment-free file and never see the template.
   90   (created, warnings) <- ensureConfigTemplates paths
   91   forM_ created $ \fp ->
   92     hPutStrLn stderr ("[seal] wrote a documented config template at " <> fp)
   93   forM_ warnings $ \w -> hPutStrLn stderr ("[seal] " <> T.unpack w)
   94   runAccessControlWith paths action

   96 -- | 'runAccessControl' against an explicit 'SealPaths'.
   97 --
   98 -- Split out so tests can point the command at a temporary home without
   99 -- mutating @SEAL_HOME@ for the whole process (which every other test in the
  100 -- suite shares).
  101 runAccessControlWith :: SealPaths -> AccessControlAction -> IO ExitCode
  102 runAccessControlWith paths action = case action of
  103   AcStatus                   -> status paths
  104   AcEnroll opts              -> enroll paths opts
  105   AcConfigure opts           -> configureWith defaultOutputs hostResolves paths opts
  106   AcDisable yes              -> disable paths yes
  107   AcReset                    -> reset paths
  108   AcMigrateCredentials force -> migrate paths force

  110 -- | Say why, on stderr, and report a refusal.
  111 --
  112 -- stdout carries data (the status report, the enrollment token); stderr carries
  113 -- reasons. Keeping them apart is what lets an operator do
  114 -- @seal access-control enroll >token.txt@ and still see what happened.
  115 refuse :: T.Text -> IO ExitCode
  116 refuse message = hPutStrLn stderr (T.unpack message) >> pure (ExitFailure 1)

  118 -- | A usage error: the flags do not make sense together, or a value is
  119 -- malformed. Exit code 2, distinguishing "you typed it wrong" from "the state
  120 -- refused you" (design §12).
  121 usage :: T.Text -> IO ExitCode
  122 usage message = hPutStrLn stderr (T.unpack message) >> pure (ExitFailure 2)

  124 -- | Refuse to touch the store while a daemon owns this @SEAL_HOME@.
  125 --
  126 -- The probe is the kernel lock, not a pid file: a daemon that crashed leaves no
  127 -- false refusal behind (design §12).
  128 requireDaemonStopped :: SealPaths -> T.Text -> IO (Either T.Text ())
  129 requireDaemonStopped paths action = do
  130   running <- daemonHoldsLock (serveLockPath paths)
  131   pure $ if running
  132     then Left ("refusing to " <> action <> ": `seal serve` is running for this SEAL_HOME.\n"
  133                 <> "  Stop it first — it has the credential store and state files open.")
  134     else Right ()

  136 -- | Print the current mode, the resolved origin/rpId and the enrolled
  137 -- credentials (labels and counters — never key material).
  138 status :: SealPaths -> IO ExitCode
  139 status paths = do
  140   gwCfg <- gatewayConfig paths
  141   inputs <- preflight paths
  142   let coords = BindCoordinates (gcHost gwCfg) (gcPort gwCfg)
  143       decision = inputs >>= bootPlan coords
  144   case decision of
  145     Left refusal ->
  146       refuse ("Access control: INVALID — " <> bootRefusalMessage refusal)
  147     Right plan -> do
  148       TIO.putStrLn ("Access control: " <> describe (bpMode plan))
  149       case (bpOrigin plan, bpRpId plan) of
  150         (Just origin, Just rpId) -> do
  151           TIO.putStrLn ("  origin: " <> origin)
  152           TIO.putStrLn ("  rp id:  " <> rpId)
  153         _ -> TIO.putStrLn "  (no origin resolved — set [access_control] origin for a wildcard bind)"
  154       let creds = Map.toList (bpCreds plan)
  155       if null creds
  156         then TIO.putStrLn "  credentials: none enrolled"
  157         else do
  158           TIO.putStrLn ("  credentials: " <> T.pack (show (length creds)))
  159           mapM_ (printCred (bpRpId plan)) creds
  160           when (any (\(_, cred) -> Just (credRpId cred) /= bpRpId plan) creds) $
  161             TIO.putStrLn
  162               "  hint: a credential is bound to another rp_id. Recover with\n\
  163               \  `seal access-control reset --yes` (deletes all passkeys), then\n\
  164               \  `seal access-control configure ...`, serve, and enroll again."
  165       TIO.putStrLn (case bpMintGrant plan of
  166         Just _  -> "  enrollment: boot mints a token (run `seal access-control enroll`)"
  167         Nothing -> "  enrollment: a usable passkey authorizes a new one")
  168       pure ExitSuccess
  169   where
  170     describe mode = case mode of
  171       AccessPasskey      -> "passkey"
  172       AccessOpen         -> "none (explicitly waived — anyone who can reach the gateway has full control)"
  173       AccessUnconfigured -> "not configured (the gateway serves only the setup screen)"
  174 
  175     printCred rpId (credIdText, cred) = TIO.putStrLn
  176       ("    - " <> credIdText <> "  " <> credLabel cred
  177         <> "  rp_id=" <> credRpId cred
  178         <> "  " <> (if Just (credRpId cred) == rpId then "usable" else "not usable here")
  179         <> "  sign_count=" <> T.pack (show (credSignCount cred)))
  180 
  181 -- | @enroll@: mint the one-time authority that adds a credential.
  182 --
  183 -- stdout carries the token (and nothing else), stderr carries the hints, so
  184 -- @seal access-control enroll > token.txt@ works and still tells the operator
  185 -- what to do with it.
  186 enroll :: SealPaths -> EnrollOptions -> IO ExitCode
  187 enroll paths opts
  188   | eoCancel opts, eoNoToken opts || eoForce opts || isJust (eoTtl opts) =
  189       usage "--cancel cannot be combined with --no-token, --ttl or --force."
  190   | otherwise = case traverse (parseTtl . T.pack) (eoTtl opts) of
  191       Left err -> usage err
  192       Right mTtl
  193         | eoCancel opts -> do
  194             cancelGrant paths
  195             TIO.putStrLn "Cancelled any outstanding enrollment grant."
  196             pure ExitSuccess
  197         | otherwise -> do
  198             let source = if eoNoToken opts then CliWindow else CliToken
  199                 ttl = fromMaybe (defaultTtlFor source) mTtl
  200             allowed <- windowAllowed paths source
  201             case allowed of
  202               Left err -> refuse err
  203               Right () -> do
  204                 minted <- enrollGrant paths source ttl (eoForce opts)
  205                 case minted of
  206                   Left err     -> refuse err
  207                   Right result -> announce paths result
  208 
  209 -- | @--no-token@ is authority with no secret, so it is confined to the
  210 -- bootstrap state (design §7.1, §12): a loopback bind, a loopback resolved
  211 -- origin, and nothing enrolled. There is no override — a window on a reachable
  212 -- gateway is anyone's enrollment.
  213 windowAllowed :: SealPaths -> GrantSource -> IO (Either T.Text ())
  214 windowAllowed _ CliToken = pure (Right ())
  215 windowAllowed _ Assertion = pure (Right ())
  216 -- A browser-minted grant is created by the gateway, never by this command, so
  217 -- reaching here would be a bug. Refuse rather than wave it through: the
  218 -- window preflight is what keeps authority-without-a-secret off a reachable port.
  219 windowAllowed _ UiToken = pure (Left "ui_token grants are minted by the gateway, not by `seal access-control enroll`.")
  220 windowAllowed paths CliWindow = do
  221   gwCfg <- gatewayConfig paths
  222   inputs <- preflight paths
  223   let coords = BindCoordinates (gcHost gwCfg) (gcPort gwCfg)
  224   pure $ case inputs >>= bootPlan coords of
  225     Left refusal -> Left (windowRefusal
  226       ("this SEAL_HOME cannot serve as configured: " <> bootRefusalMessage refusal))
  227     Right plan
  228       | not (isLoopbackHost (bcHost coords)) ->
  229           Left (windowRefusal "the gateway binds a non-loopback address")
  230       | not (maybe False isLoopbackHost (bpRpId plan)) ->
  231           Left (windowRefusal "the resolved origin is not loopback")
  232       | credentialAvailability (bpRpId plan) (bpCreds plan) /= NoCredentials ->
  233           Left (windowRefusal "credentials are already enrolled; use a token or an assertion")
  234       | otherwise -> Right ()
  235   where
  236     windowRefusal reason =
  237       "--no-token is not available here: " <> reason <> ".\n"
  238         <> "  An enrollment window presents no secret, so it exists only while\n"
  239         <> "  nothing is enrolled and the gateway is reachable from this machine\n"
  240         <> "  alone. Mint a token instead: `seal access-control enroll`."
  241 
  242 -- | Print the minted grant: token on stdout, everything else on stderr.
  243 announce :: SealPaths -> MintedGrant -> IO ExitCode
  244 announce paths minted = do
  245   now <- getCurrentTime
  246   _ <- appendAudit paths now (emptyAuditFields GrantMinted)
  247          { afVia = Just (grSource (mgGrant minted))
  248          , afRemainingSeconds = Just (secondsLeft now (mgGrant minted))
  249          }
  250   interactive <- hIsTerminalDevice stderr
  251   case mgToken minted of
  252     Just token -> do
  253       TIO.putStrLn token
  254       unless interactive $ do
  255         writeSecretFile (enrollmentTokenPath paths) token
  256         hPutStrLn stderr ("The token is also stored at "
  257                             <> enrollmentTokenPath paths <> " (mode 0600), because "
  258                             <> "stderr is not a terminal.")
  259     Nothing ->
  260       TIO.putStrLn "No token: this is a loopback enrollment window (5 minutes)."
  261   hPutStrLn stderr (T.unpack (summary now (mgGrant minted)))
  262   pure ExitSuccess
  263   where
  264     secondsLeft now grant =
  265       max 0 (ceiling (realToFrac (diffUTCTime (grExpiresAt grant) now) :: Double) :: Int)
  266 
  267     summary now grant =
  268       "Grant minted via " <> grantSourceText (grSource grant) <> ", expiring in "
  269         <> T.pack (show (max 0 (ceiling
  270              (realToFrac (diffUTCTime (grExpiresAt grant) now) :: Double) :: Int)))
  271         <> "s. Open the web UI and enroll a passkey; the grant is consumed on success."

  273 -- | @disable@: the only way to turn access control off.
  274 --
  275 -- No HTTP request can do this (design §5 invariant 7), which is what removes
  276 -- the "one token enrolls and disables" blast radius. Credentials are kept: the
  277 -- operator who wants them gone runs @reset --yes@.
  278 disable :: SealPaths -> Bool -> IO ExitCode
  279 disable _ False =
  280   usage "disable requires --yes: it turns access control off, and anyone who can reach the gateway gets full control."
  281 disable paths True = do
  282   stopped <- requireDaemonStopped paths "disable access control"
  283   case stopped of
  284     Left err -> refuse err
  285     Right () -> do
  286       written <- updateAccessControlSection (securityFilePath paths) $ \cfg ->
  287         cfg { acfcMode = Just "none" }
  288       case written of
  289         Left err -> refuse ("could not disable access control: " <> err)
  290         Right _ -> do
  291           now <- getCurrentTime
  292           _ <- appendAudit paths now (emptyAuditFields ModeDisabled)
  293           TIO.putStrLn "Access control is off ([access_control] mode = \"none\")."
  294           TIO.putStrLn "  Credentials are kept. `seal access-control enroll` turns it back on."
  295           pure ExitSuccess

  297 -- | Move credentials out of @security.toml@ and into @credentials.toml@.
  298 migrate :: SealPaths -> Bool -> IO ExitCode

  300   stopped <- requireDaemonStopped paths "migrate credentials"
  301   case stopped of
  302     Left err -> refuse err

  305 migrateWith :: SealPaths -> Bool -> IO ExitCode
  306 migrateWith paths force = do
  307   result <- migrateLegacyCredentials (securityFilePath paths)
  308                                      (credentialsFilePath paths) force
  309   case result of
  310     Left err -> refuse ("Could not migrate credentials: " <> err)
  311     Right NothingToMigrate -> do
  312       TIO.putStrLn "Nothing to migrate: security.toml holds no credential tables."
  313       pure ExitSuccess
  314     Right (Migrated count) -> do
  315       TIO.putStrLn ("Moved " <> T.pack (show count) <> " credential(s) to "
  316                      <> T.pack (credentialsFilePath paths) <> ".")
  317       TIO.putStrLn "  Removed the [access_control.credentials] tables from security.toml."
  318       TIO.putStrLn "  Restart `seal serve` to pick up the new store."
  319       pure ExitSuccess

  321 -- | Remove the @[access_control]@ table, returning the gateway to the setup
  322 -- screen.
  323 --
  324 -- 'DropCredentials' is correct here and nowhere else on this path: @reset@ is
  325 -- CLI-only and @--yes@ is mandatory, so a human decided to destroy the
  326 -- credential tables rather than be blocked by them. Every other writer keeps
  327 -- 'PreserveCredentials' and is refused instead.
  328 reset :: SealPaths -> IO ExitCode
  329 reset paths = do
  330   stopped <- requireDaemonStopped paths "reset access control"
  331   case stopped of
  332     Left err -> refuse err
  333     Right () -> resetWith paths
  334 
  335 resetWith :: SealPaths -> IO ExitCode
  336 resetWith paths = do
  337   result <- clearAccessControlSection (securityFilePath paths) DropCredentials
  338   case result of
  339     Left err -> refuse ("Could not reset access control: " <> err)
  340     Right () -> do
  341       TIO.putStrLn "Removed [access_control] from ~/.seal/security.toml."
  342       TIO.putStrLn "  This also deleted every enrolled passkey. Enrolling again is required."
  343       TIO.putStrLn "  A running `seal serve` keeps its current mode until you restart it."
  344       TIO.putStrLn "  To target a hostname other than localhost, run"
  345       TIO.putStrLn "  `seal access-control configure --origin https://HOST --rp-id HOST`"
  346       TIO.putStrLn "  before restarting, then open the web UI to enroll a new passkey."
  347       -- `reset` is the lost-every-device escape hatch, so it also destroys what
  348       -- would otherwise be left half-configured: the store, the outstanding
  349       -- grant and its token. The [access_control] section is already gone, so a
  350       -- crash from here leaves the loud `unconfigured` state, not a locked-out
  351       -- one.
  352       cancelGrant paths
  353       removeStoreFile paths
  354       pure ExitSuccess

  356 -- | Delete @credentials.toml@ if it exists. Only @reset --yes@ may do this.

  377 -- | A configure failure. The constructor decides the exit code: malformed
  378 -- flags are usage (2), a state/environment refusal is 1.
  379 data ConfigureError
  380   = ConfigureUsage !Text
  381   | ConfigureRefusal !Text
  382   deriving stock (Eq, Show)
  383 
  384 configureExit :: ConfigureError -> ExitCode
  385 configureExit = \case
  386   ConfigureUsage _   -> ExitFailure 2
  387   ConfigureRefusal _ -> ExitFailure 1
  388 
  389 configureMessage :: ConfigureError -> Text
  390 configureMessage = \case
  391   ConfigureUsage m   -> m
  392   ConfigureRefusal m -> m

  402 -- | Persist the gateway coordinates. Acquires the serve lock so it cannot race
  403 -- a running daemon; a concurrent @tui@/@vault setup@ writer can still race the
  404 -- later write (accepted residual risk, see the plan).
  405 configureWith
  406   :: Outputs
  407   -> (Text -> IO Bool)
  408   -> SealPaths
  409   -> ConfigureOptions
  410   -> IO ExitCode
  411 configureWith out resolves paths opts = do
  412   ensureSealDirs paths
  413   acquired <- acquireServeLock (serveLockPath paths)
  414   case acquired of
  415     Left _ -> report out (ConfigureRefusal
  416       "`seal serve` is running for this SEAL_HOME; stop it before configuring coordinates.")
  417     Right lock -> configureLocked out resolves paths opts
  418                      `finally` releaseServeLock lock
  419 
  420 configureLocked
  421   :: Outputs -> (Text -> IO Bool) -> SealPaths -> ConfigureOptions -> IO ExitCode
  422 configureLocked out resolves paths opts = do
  423   legacy <- legacyConfigProblem paths
  424   case legacy of
  425     Just problem -> report out problem
  426     Nothing -> do
  427       loaded <- loadSecurityConfig (securityFilePath paths)
  428       case loaded of
  429         Left err -> report out
  430           (ConfigureRefusal ("security.toml could not be read: " <> err))
  431         Right cfg -> do
  432           inputs <- preflight paths
  433           case inputs of
  434             Left refusal -> report out
  435               (ConfigureRefusal (bootRefusalMessage refusal))
  436             Right bi -> case planConfigure cfg (biCredentials bi) opts of
  437               Left problem -> report out problem
  438               Right plan -> applyPlan out resolves paths opts cfg plan
  439 
  440 applyPlan
  441   :: Outputs -> (Text -> IO Bool) -> SealPaths -> ConfigureOptions
  442   -> SecurityConfig -> PlannedConfigure -> IO ExitCode
  443 applyPlan out resolves paths opts oldCfg plan = do
  444   let eff = gatewayConfigFromSecurity (plConfig plan)
  445   resolution <- gatewayHostResolutionError resolves (gcHost eff)
  446   case resolution of
  447     Just err -> report out (ConfigureRefusal err)
  448     Nothing -> do
  449       keys <- validateKeyFiles opts
  450       case keys of
  451         Left problem -> report out problem
  452         Right () -> do
  453           written <- updateSecurityConfig (securityFilePath paths)
  454                        (const (plConfig plan))
  455           case written of
  456             Left err -> report out
  457               (ConfigureRefusal ("could not write security.toml: " <> err))
  458             Right () -> do
  459               printConfigure out oldCfg plan
  460               printHints out paths plan
  461               pure ExitSuccess
  462 
  463 report :: Outputs -> ConfigureError -> IO ExitCode
  464 report out problem = errLine out (configureMessage problem)
  465                        >> pure (configureExit problem)

  467 -- | Refuse when @config.toml@ still carries pre-split security fields. Running
  468 -- @configure@ then would create a @security.toml@ that makes the next boot's
  469 -- migration take the "security.toml wins" branch and drop the un-migrated
  470 -- vault/execution settings, so we send the operator through one migration boot
  471 -- first instead.
  472 legacyConfigProblem :: SealPaths -> IO (Maybe ConfigureError)
  473 legacyConfigProblem paths = do
  474   let path = configFilePath paths
  475   exists <- doesFileExist path
  476   if not exists then pure Nothing else do
  477     read' <- try (TIO.readFile path) :: IO (Either IOException Text)
  478     pure $ case read' of
  479       Left err -> Just (ConfigureRefusal
  480         ("config.toml could not be read: " <> T.pack (show err)))
  481       Right raw
  482         | configHasLegacySecurityFields raw -> Just (ConfigureRefusal
  483             ("config.toml still carries pre-split security fields.\n"
  484               <> "  Run `seal serve` once to migrate them into security.toml, "
  485               <> "then retry configure."))
  486         | otherwise -> Nothing
  487 
  488 validateKeyFiles :: ConfigureOptions -> IO (Either ConfigureError ())
  489 validateKeyFiles opts = do
  490   let named = catMaybes [coTlsCert opts, coTlsKey opts]
  491   bad <- filterM (fmap not . readable) named
  492   pure $ case bad of
  493     []      -> Right ()
  494     (p : _) -> Left (ConfigureRefusal
  495       ("TLS key file \"" <> T.pack p <> "\" is missing or unreadable."))
  496   where
  497     readable p = do
  498       exists <- doesFileExist p
  499       if not exists then pure False else do
  500         content <- try (BS.readFile p) :: IO (Either IOException BS.ByteString)
  501         pure (either (const False) (const True) content)

  503 printConfigure :: Outputs -> SecurityConfig -> PlannedConfigure -> IO ()
  504 printConfigure out oldCfg plan = do
  505   let oldGw = gatewayConfigFromSecurity oldCfg
  506       newGw = gatewayConfigFromSecurity (plConfig plan)
  507       oldAc = scAccessControl oldCfg
  508       newAc = scAccessControl (plConfig plan)
  509       fmtList xs = if null xs then "(none)" else T.intercalate "," xs
  510       fields =
  511         [ ("bind_host", gcHost oldGw, gcHost newGw)
  512         , ("bind_port", tshow (gcPort oldGw), tshow (gcPort newGw))
  513         , ("tls", onOff (tlsEnabled oldGw), onOff (tlsEnabled newGw))
  514         , ("origin", fromMaybe "(none)" (oldAc >>= acfcOrigin)
  515                    , fromMaybe "(none)" (newAc >>= acfcOrigin))
  516         , ("rp_id", fromMaybe "(none)" (oldAc >>= acfcRpId)
  517                   , fromMaybe "(none)" (newAc >>= acfcRpId))
  518         , ("allowed_origins", fmtList (gcAllowedOrigins oldGw)
  519                             , fmtList (gcAllowedOrigins newGw))
  520         , ("tls_hosts", fmtList (gcTlsHosts oldGw), fmtList (gcTlsHosts newGw))
  521         ]
  522   forM_ [ (k, o, n) | (k, o, n) <- fields, o /= n ] $ \(k, o, n) ->
  523     outLine out (k <> ": " <> o <> " -> " <> n)
  524   outLine out ("bind_host: " <> gcHost newGw)
  525   outLine out ("bind_port: " <> tshow (gcPort newGw))
  526   outLine out ("tls: " <> onOff (tlsEnabled newGw))
  527   outLine out ("origin: " <> fromMaybe "(none)" (plOrigin plan))
  528   outLine out ("rp_id: " <> fromMaybe "(none)" (plRpId plan))
  529 
  530 printHints :: Outputs -> SealPaths -> PlannedConfigure -> IO ()
  531 printHints out paths plan = do
  532   let gw = gatewayConfigFromSecurity (plConfig plan)
  533       mode = scAccessControl (plConfig plan) >>= acfcMode
  534       (certPath, _) =
  535         effectiveTlsPaths (tlsCertPath paths) (tlsKeyPath paths) gw
  536   when (tlsEnabled gw && isNothing (gcTlsCert gw)) $
  537     errLine out (generatedCertificateTrustHint certPath)
  538   when (mode == Just "none"
  539          && not (isLoopbackHost (gcHost gw))
  540          && maybe False (not . isHttps) (plOrigin plan)) $
  541     errLine out
  542       "warning: mode \"none\" on a non-loopback plain-http bind exposes the \
  543       \gateway without authentication."
  544   errLine out "next: run `seal serve`, then `seal access-control status`."
  545   errLine out "      Enroll through the origin clients use; changing it later orphans passkeys."
  546 
  547 -- | Pure planning and validation. No I/O, no filesystem; the handler supplies
  548 -- credentials and runs the resolver/certificate checks.
  549 planConfigure
  550   :: SecurityConfig
  551   -> CredentialsPresence
  552   -> ConfigureOptions
  553   -> Either ConfigureError PlannedConfigure
  554 planConfigure sec creds opts = do
  555   whenE (not (anyOption opts)) (usageE
  556     "configure needs at least one of --host, --port, --tls, --no-tls, \
  557     \--tls-cert, --tls-key, --tls-host, --origin or --rp-id.")
  558   mPort <- traverse parsePort (coPort opts)
  559   whenE (coTls opts && coNoTls opts) (usageE
  560     "--tls and --no-tls cannot be combined.")
  561   whenE (coNoTls opts && any isJust [coTlsCert opts, coTlsKey opts]
  562           || coNoTls opts && not (null (coTlsHosts opts))) (usageE
  563     "--no-tls cannot be combined with --tls-cert, --tls-key or --tls-host.")
  564   case (coTlsCert opts, coTlsKey opts) of
  565     (Just _, Nothing) -> usageE "--tls-cert requires --tls-key."
  566     (Nothing, Just _) -> usageE "--tls-key requires --tls-cert."
  567     _                 -> Right ()
  568   mapM_ checkCertPath (catMaybes [coTlsCert opts, coTlsKey opts])
  569   mapM_ checkTlsHost (coTlsHosts opts)
  570   requestedOrigin <- traverse parseOriginStrict (coOrigin opts)
  571   requestedRpId <- traverse parseRpIdText (coRpId opts)
  572   let oldGw = fromMaybe emptyGateway (scGateway sec)
  573       oldAc = fromMaybe emptyAccess (scAccessControl sec)
  574       oldEff = gatewayConfigFromSecurity sec
  575       newGw0 = applyGatewayDeltas oldGw mPort opts
  576       newGw1 = if coTls opts || isJust (coTlsCert opts) || isJust (coTlsKey opts)
  577                  then newGw0 { pgcTls = Just True }
  578                  else newGw0
  579       newGw2 = if null (coTlsHosts opts)
  580                  then newGw1
  581                  else newGw1 { pgcTlsHosts = coTlsHosts opts }
  582       newGw3 = if coNoTls opts
  583                  then newGw1 { pgcTls = Just False, pgcTlsCert = Nothing
  584                              , pgcTlsKey = Nothing, pgcTlsHosts = [] }
  585                  else newGw2
  586       effGw = withGatewayDefaults newGw3
  587       effTls = tlsEnabled effGw
  588   whenE (not (null (coTlsHosts opts)) && not effTls) (usageE
  589     "--tls-host requires TLS: pass --tls or --tls-cert/--tls-key.")
  590   chosenOrigin <- chooseOrigin opts oldAc effGw effTls requestedOrigin
  591   let derivedRpId = chosenOrigin >>= originHost
  592       effectiveRpId = requestedRpId <|> acfcRpId oldAc <|> derivedRpId
  593   validateRpId chosenOrigin effectiveRpId
  594   let oldDerived = derivedOriginForm (gcHost oldEff) (gcPort oldEff)
  595       newDerived = if effTls
  596                      then httpsOriginFor (gcHost effGw) (gcPort effGw)
  597                      else derivedOriginForm (gcHost effGw) (gcPort effGw)
  598       managedNew = chosenOrigin <|> newDerived
  599       removed = catMaybes [acfcOrigin oldAc, oldDerived]
  600       kept = filter (`notElem` removed)
  601                (map normaliseEntry (pgcAllowedOrigins oldGw))
  602       merged = nub (kept ++ maybeToList managedNew)
  603       merged' = if null merged then maybeToList managedNew else merged
  604       newGw = newGw3 { pgcAllowedOrigins = merged' }
  605       newAc = oldAc { acfcOrigin = chosenOrigin
  606                     , acfcRpId = requestedRpId <|> acfcRpId oldAc }
  607       newCfg = sec { scGateway = Just newGw, scAccessControl = Just newAc }
  608       raw = Toml.encode securityConfigCodec newCfg
  609       coords = BindCoordinates (gcHost effGw) (gcPort effGw)
  610       inputs = BootInputs raw (scAccessControl newCfg) creds
  611   plan <- case bootPlan coords inputs of
  612     Left refusal -> refusalE (bootRefusalMessage refusal)
  613     Right p      -> Right p
  614   maybe (Right ()) usageE (gatewayHostError (gcHost effGw))
  615   maybe (Right ()) refusalE (tlsConfigError effGw)
  616   maybe (Right ()) refusalE
  617     (tlsOriginError (tlsEnabled effGw) (bpOrigin plan))
  618   whenE (not (isLoopbackHost (gcHost effGw))
  619           && not (maybe False isHttps (bpOrigin plan))
  620           && bpMode plan /= AccessOpen) (refusalE
  621     "refusing to write a non-loopback plain-http gateway. Bind loopback (and \
  622     \use a TLS reverse proxy), enable TLS, or use mode \"none\" explicitly.")
  623   let enrolled = bpCreds plan
  624       rp = bpRpId plan
  625       orphans = maybe True (\r -> any ((/= r) . credRpId) (Map.elems enrolled)) rp
  626   whenE (not (Map.null enrolled) && orphans) (refusalE
  627     ("this change would orphan every enrolled passkey. Recover with:\n"
  628       <> "  seal access-control reset --yes   (deletes all enrolled passkeys)\n"
  629       <> "  seal access-control configure ... (the coordinates you want)\n"
  630       <> "  seal serve, then enroll a new passkey."))
  631   case decodeSecurityConfig raw of
  632     Right decoded | decoded == newCfg -> Right ()
  633     _ -> refusalE
  634       "refusing to write security.toml: the encoded document does not round-trip."
  635   pure PlannedConfigure
  636     { plConfig = newCfg
  637     , plOrigin = bpOrigin plan
  638     , plRpId = bpRpId plan
  639     }

  641 anyOption :: ConfigureOptions -> Bool
  642 anyOption opts =
  643   any isJust [coHost opts, coPort opts, coOrigin opts, coRpId opts]
  644     || any isJust [coTlsCert opts, coTlsKey opts]
  645     || coTls opts || coNoTls opts || not (null (coTlsHosts opts))
  646 
  647 parsePort :: Text -> Either ConfigureError Int
  648 parsePort raw = case readMaybe (T.unpack raw) of
  649   Just n | n >= 1 && n <= 65535 -> Right n
  650   _ -> usageE ("--port must be an integer between 1 and 65535, got \"" <> raw <> "\".")
  651 
  652 checkCertPath :: FilePath -> Either ConfigureError ()
  653 checkCertPath p
  654   | isAbsolute p = Right ()
  655   | otherwise = usageE ("--tls-cert/--tls-key must be absolute paths, got \""
  656                           <> T.pack p <> "\".")
  657 
  658 checkTlsHost :: Text -> Either ConfigureError ()
  659 checkTlsHost h
  660   | T.null (T.strip h) = usageE "--tls-host must not be empty."
  661   | T.any (\c -> isSpace c || isControl c) h = usageE
  662       "--tls-host must not contain whitespace or control characters."
  663   | otherwise = Right ()
  664 
  665 parseOriginStrict :: Text -> Either ConfigureError Text
  666 parseOriginStrict raw
  667   | T.any isSpace raw = usageE
  668       "an origin must not contain whitespace."
  669   | T.any (`elem` ("@?#\\" :: String)) raw = usageE
  670       "an origin must be scheme://host[:port] with no userinfo, query, fragment or path."
  671   | otherwise = case normalizeOrigin raw of
  672       Just o | isHttpOrigin o -> Right o
  673       _ -> usageE ("\"" <> raw <> "\" is not an absolute http:// or https:// origin.")
  674 
  675 parseRpIdText :: Text -> Either ConfigureError Text
  676 parseRpIdText raw
  677   | T.null rp = usageE "--rp-id must not be empty."
  678   | isIpLiteral rp = usageE
  679       ("--rp-id \"" <> rp <> "\" is an IP address; a WebAuthn rp_id must be a domain name.")
  680   | otherwise = Right rp
  681   where
  682     rp = T.toLower (T.strip raw)
  683 
  684 validateRpId :: Maybe Text -> Maybe Text -> Either ConfigureError ()
  685 validateRpId Nothing (Just _) = refusalE
  686   "an rp_id needs an origin; pass --origin as well."
  687 validateRpId _ Nothing = Right ()
  688 validateRpId (Just origin) (Just rp) =
  689   let host = fromMaybe "" (originHost origin)
  690   in if rp == host
  691        || (("." <> rp) `T.isSuffixOf` host && T.count "." rp >= 1)
  692        then Right ()
  693        else usageE ("--rp-id \"" <> rp <> "\" is not a parent domain of the origin host \""
  694                       <> host <> "\".")
  695 
  696 chooseOrigin
  697   :: ConfigureOptions
  698   -> AccessControlFileConfig
  699   -> GatewayConfig
  700   -> Bool
  701   -> Maybe Text
  702   -> Either ConfigureError (Maybe Text)
  703 chooseOrigin opts oldAc effGw effTls requested = case requested of
  704   Just o -> Right (Just o)
  705   Nothing
  706     | coNoTls opts -> Right (acfcOrigin oldAc)
  707     | not effTls   -> Right (acfcOrigin oldAc)
  708     | otherwise -> case acfcOrigin oldAc of
  709         Just o | isHttps o -> Right (Just o)
  710                | otherwise -> refusalE
  711                    ("TLS is enabled but [access_control] origin is \"" <> o
  712                      <> "\"; pass --origin https://<the host clients use>.")
  713         Nothing -> case httpsOriginFor (gcHost effGw) (gcPort effGw) of
  714           Just o  -> Right (Just o)
  715           Nothing -> refusalE
  716             ("cannot derive an https origin for bind host \"" <> gcHost effGw
  717               <> "\"; pass --origin.")
  718 
  719 applyGatewayDeltas
  720   :: PartialGatewayConfig -> Maybe Int -> ConfigureOptions -> PartialGatewayConfig
  721 applyGatewayDeltas oldGw mPort opts = oldGw
  722   { pgcHost = coHost opts <|> pgcHost oldGw
  723   , pgcPort = mPort <|> pgcPort oldGw
  724   , pgcTlsCert = fmap T.pack (coTlsCert opts) <|> pgcTlsCert oldGw
  725   , pgcTlsKey = fmap T.pack (coTlsKey opts) <|> pgcTlsKey oldGw
  726   }
  727 
  728 derivedOriginForm :: Text -> Int -> Maybe Text
  729 derivedOriginForm host port
  730   | isWildcardHost host = Nothing
  731   | otherwise = Just (canonicalOriginFor host port)
  732 
  733 httpsOriginFor :: Text -> Int -> Maybe Text
  734 httpsOriginFor host port
  735   | isWildcardHost host = Nothing
  736   | isLoopbackHost host = Just (httpsUrl "localhost" port)
  737   | isIpLiteral host || T.isInfixOf ":" host = Nothing
  738   | otherwise = Just (httpsUrl (T.toLower host) port)
  739   where
  740     httpsUrl h p
  741       | p == 443 = "https://" <> h
  742       | otherwise = "https://" <> h <> ":" <> tshow p
  743 
  744 originHost :: Text -> Maybe Text
  745 originHost origin =
  746   let (_, rest) = T.breakOn "://" origin
  747   in if T.null rest
  748        then Nothing
  749        else let authority = T.drop 3 rest
  750             in Just (T.toLower (T.takeWhile (/= ':') authority))
  751 
  752 normaliseEntry :: Text -> Text
  753 normaliseEntry raw = fromMaybe raw (normalizeOrigin raw)
  754 
  755 isHttpOrigin :: Text -> Bool
  756 isHttpOrigin o = "http://" `T.isPrefixOf` o || "https://" `T.isPrefixOf` o
  757 
  758 isHttps :: Text -> Bool
  759 isHttps o = "https://" `T.isPrefixOf` o
  760 
  761 onOff :: Bool -> Text
  762 onOff True  = "on"
  763 onOff False = "off"
  764 
  765 tshow :: Show a => a -> Text
  766 tshow = T.pack . show
  767 
  768 usageE :: Text -> Either ConfigureError a
  769 usageE = Left . ConfigureUsage
  770 
  771 refusalE :: Text -> Either ConfigureError a
  772 refusalE = Left . ConfigureRefusal
  773 
  774 whenE :: Bool -> Either ConfigureError () -> Either ConfigureError ()
  775 whenE True  action = action
  776 whenE False _      = Right ()

  800 gatewayConfig :: SealPaths -> IO GatewayConfig
  801 gatewayConfig paths = do
  802   sec <- loadSecurityConfig (securityFilePath paths)
  803   case sec of
  804     Right cfg | isJust (scGateway cfg) -> pure (gatewayConfigFromSecurity cfg)
  805     -- Upgrade window: this SEAL_HOME has not booted serve/tui since the
  806     -- [gateway] section moved, so the table is still in config.toml. Read it
  807     -- from there — read-only. `status` and `enroll` must not rewrite
  808     -- security.toml: a pre-upgrade daemon may be running, and the write lock
  809     -- is process-local. The next serve/tui boot moves the table and this path
  810     -- goes quiet.
  811     _ -> do
  812       legacy <- loadLegacyGatewayTable (configFilePath paths)
  813       pure (maybe defaultGatewayConfig withGatewayDefaults legacy)

   74 -- | Spawn the real signal-cli transport, resolve the account + chunk limit
   75 -- + allow-list, and run the agent loop against the Signal channel. Fails
   76 -- fast with a stderr diagnostic if signal-cli is absent or the account is
   77 -- unresolved.
   78 runSignal
   79   :: ChannelDeps -> Registry -> PreprocessChain -> TabsHandle
   80   -> (SignalAccount, Int, AllowList UserId)
   81   -> AskReplyStore
   82   -> IO ()
   83 runSignal deps registry chain tabsH (account, chunkLimit, allow) askReply = do
   84   let accountLabel = signalAccountText account
   85   eTransport <- mkRealSignalTransport accountLabel
   86   case eTransport of
   87     Left err -> logIO (cdLogger deps) ErrorS ("seal signal: " <> ls err)
   88     Right transport -> do
   89       let withCh = withSignalChannel (allow, chunkLimit) account transport (cdLogger deps)
   90           plainHandler h = plainTurn deps h askReply
   91       runChannelLoop deps withCh plainHandler registry chain askReply tabsH Nothing Nothing

   93 -- | The inbox-driven loop. Spawns the Signal channel via 'withSignalChannel',
   94 -- pulls @(MessageSource, body)@ from 'chReceive', classifies via
   95 -- 'Seal.Routing.Route' (Layer-1 terse grammar + /tab commands BEFORE the
   96 -- /-command registry), dispatches slash commands via a 'ChannelCaps' adapter
   97 -- over the 'ChannelHandle', and routes plain messages to the supplied
   98 -- 'plainHandler' (which runs 'runTurn' with 'aeMessageSource' = @Just ms@,
   99 -- using the supplied 'ChannelHandle' for any sends).
  100 -- Terminates when 'chReceive' returns EOF (@(Nothing, "")@ with the reader
  101 -- exited). The 'withSignalChannel' bracket owns cleanup.
  102 runSignalLoop
  103   :: Registry
  104   -> PreprocessChain
  105   -> (AllowList UserId, Int)
  106   -> SignalAccount
  107   -> SignalTransport
  108   -> TabsHandle
  109   -> AskReplyStore
  110   -> SessionRuntime
  111   -> (ChannelHandle -> Maybe MessageSource -> Text -> IO ())
  112   -> SealLogger
  113   -> IO ()
  114 runSignalLoop registry chain (allow, chunkLimit) account transport tabsH askReply sr plainHandler logger =
  115   withSignalChannel (allow, chunkLimit) account transport logger $ \ch -> do
  116     let h = toHandle ch
  117         handleCaps = Data.Default.def
  118           { ccSend         = chSend h
  119           , ccShowHuman    = chSend h
  120           , ccPrompt       = \(AskPrompt q opts) -> do
  121               -- Bind the pending question to the active session so the
  122               -- next inbound message from the peer (delivered via
  123               -- 'deliverNextAnswer' in the loop below) unblocks this thread.
  124               meta <- readIORef (srActive sr)
  125               let sid = smId meta
  126               outcome <- askHumanWithOptions askReply sid q opts
  127                            (\_qid -> chSend h (formatQuestionWithOptions q opts))
  128               pure (fromRight "" outcome)
  129           , ccPromptSecret = fmap (fromRight "") . chPromptSecret h
  130           , ccStreaming    = False  -- Signal: send accumulated text once, not per-delta
  131           }
  132     loop h handleCaps
  133   where
  134     loop h handleCaps = do
  135       (mSrc, body) <- chReceive h
  136       case mSrc of
  137         Nothing -> pure ()  -- EOF: reader exited + inbox drained
  138         Just _ms -> do
  139           -- First, try to deliver the inbound message as the answer to the
  140           -- oldest pending ASK_HUMAN question for the active session. If it
  141           -- matches, the waiting agent-loop thread unblocks and the message
  142           -- is consumed as the answer (NOT routed as a plain turn). If no
  143           -- question is pending, the message is a normal inbound turn.
  144           meta <- readIORef (srActive sr)
  145           let sid = smId meta
  146           (_resolved, delivered) <- deliverNextAnswerResolved askReply sid body
  147           if delivered
  148             then loop h handleCaps
  149             else do
  150               -- Layer-1 routing: check the terse /N grammar + /tab commands
  151               -- BEFORE the /-command registry. Plain turns are forked so the
  152               -- main loop keeps receiving + delivering answers while the
  153               -- agent loop blocks on ASK_HUMAN.
  154               case Seal.Routing.Route.route body of
  155                 Right (Seal.Routing.Route.Focus idx) -> do
  156                   _ <- focusTabH tabsH idx
  157                   chSend h ("focused tab " <> T.singleton (tabIndexToChar idx))
  158                   loop h handleCaps
  159                 Right (Seal.Routing.Route.Inject idx payload) -> do
  160                   _ <- focusTabH tabsH idx
  161                   void (forkIO (plainHandler h mSrc payload))
  162                   loop h handleCaps
  163                 Right (Seal.Routing.Route.TabCommand tsc) -> do
  164                   _ <- handleTabCommand' h tabsH tsc
  165                   loop h handleCaps
  166                 Right Seal.Routing.Route.CurrentTab -> do
  167                   tl <- snapshotTabs tabsH
  168                   case lookupByRef tl (BoundSession (smId meta)) of
  169                     Just t  -> chSend h (renderTab t)
  170                     Nothing -> chSend h "no current tab"
  171                   loop h handleCaps
  172                 Right (Seal.Routing.Route.NewSession _args) -> do
  173                   -- /new is registered as a CommandSpec in the registry
  174                   -- (this standalone loop tracks "current" via srActive, not
  175                   -- a cursor). Fall through to the registry path.
  176                   d <- ingest registry chain (RawInbound body)
  177                   case d of
  178                     DispatchAction a -> do
  179                       mFollowUp <- runCommandAction a handleCaps
  180                       case mFollowUp of
  181                         Just t  -> void (forkIO (plainHandler h mSrc t))
  182                         Nothing -> pure ()
  183                       loop h handleCaps
  184                     ShowText t       -> chSend h t >> loop h handleCaps
  185                     PlainMessage t   -> void (forkIO (plainHandler h mSrc t)) >> loop h handleCaps
  186                     Rejected msg     -> chSend h msg >> loop h handleCaps
  187                 Right (Seal.Routing.Route.SlashCommand _) -> do
  188                   d <- ingest registry chain (RawInbound body)
  189                   case d of
  190                     DispatchAction a -> do
  191                       mFollowUp <- runCommandAction a handleCaps
  192                       case mFollowUp of
  193                         Just t  -> void (forkIO (plainHandler h mSrc t))
  194                         Nothing -> pure ()
  195                       loop h handleCaps
  196                     ShowText t       -> chSend h t >> loop h handleCaps
  197                     PlainMessage t   -> void (forkIO (plainHandler h mSrc t)) >> loop h handleCaps
  198                     Rejected msg     -> chSend h msg >> loop h handleCaps
  199                 Right (Seal.Routing.Route.Plain t) -> do
  200                   void (forkIO (plainHandler h mSrc t))
  201                   loop h handleCaps
  202                 Left (Seal.Routing.Route.ParseError e) -> do
  203                   chSend h e
  204                   loop h handleCaps

  206 -- | Handle a parsed 'TabSlashCommand' over the Signal channel (mutates the
  207 -- TabsHandle, replies via chSend). Mirrors Seal.Channel.Cli.handleTabCommand.
  208 handleTabCommand' :: ChannelHandle -> TabsHandle -> TabSlashCommand -> IO ()
  209 handleTabCommand' h tabsH = \case
  210   TabListCmd -> do
  211     tl <- snapshotTabs tabsH
  212     if tabCount tl == 0
  213       then chSend h "no tabs"
  214       else mapM_ (chSend h . renderTab) (tlTabs tl)
  215   TabNewCmd _mKind -> do
  216     r <- insertTabH tabsH (BoundSession placeholderSid) KindAi Nothing
  217     case r of
  218       Left e  -> chSend h ("tab new failed: " <> e)
  219       Right i -> chSend h ("tab " <> T.singleton (tabIndexToChar i) <> " created")
  220   TabCloseCmd idx force -> do
  221     r <- removeTabH tabsH idx
  222     case r of
  223       Left e  -> chSend h (if force == Force then "force close: " <> e else "close failed: " <> e)
  224       Right _ -> chSend h ("tab " <> T.singleton (tabIndexToChar idx) <> " closed")
  225   TabFocusCmd idx -> do
  226     r <- focusTabH tabsH idx
  227     case r of
  228       Left e  -> chSend h ("focus failed: " <> e)
  229       Right _ -> chSend h ("focused tab " <> T.singleton (tabIndexToChar idx))
  230   TabResumeCmd sid -> do
  231     r <- insertTabH tabsH (BoundSession sid) KindAi Nothing
  232     case r of
  233       Left e  -> chSend h ("resume failed: " <> e)
  234       Right i -> chSend h ("tab " <> T.singleton (tabIndexToChar i) <> " resumed")
  235   TabRenameCmd idx name -> do
  236     r <- renameTabH tabsH idx name
  237     case r of
  238       Left e  -> chSend h ("rename failed: " <> e)
  239       Right _ -> chSend h ("tab " <> T.singleton (tabIndexToChar idx) <> " renamed to " <> name)
  240   where
  241     placeholderSid = case mkSessionId "tab-session" of
  242       Right s -> s
  243       Left _  -> error "placeholder session id"

  245 -- | Render one tab as a single line: @<index>  <kind>  [label]@.
  246 renderTab :: Tab -> Text
  247 renderTab t =
  248   T.singleton (tabIndexToChar (tIndex t)) <> "  " <> T.pack (show (tKind t))
  249     <> maybe "" ("  " <>) (tLabel t)

  251 -- | Full @seal signal@ startup wiring: paths -> config -> vault -> session
  252 -- -> backends -> registry -> spawn the Signal channel -> run the loop.
  253 -- Mirrors 'Seal.Tui.runTui' but drives the Signal channel instead of the
  254 -- Haskeline TUI. Resolves the @[signal]@ config section + an optional
  255 -- vault-supplied account label; fails fast with a stderr diagnostic if
  256 -- the account is unresolved or signal-cli is absent. The autonomy level
  257 -- threads through to 'mkSessionAgentEnv' so 'Supervised' (the default)
  258 -- prompts before running Untrusted opcodes (the next inbound message from
  259 -- the peer delivers the answer via the ask/reply store).
  260 runSignalMain :: Seal.Security.Policy.AutonomyLevel -> SealLogger -> IO ()
  261 runSignalMain autonomy logger = do
  262   paths <- getSealPaths
  263   ensureSealDirs paths
  264   migrateSecurityConfig paths
  265   let cfgPath = configFilePath paths
  266   cfg <- loadRuntimeConfig cfgPath >>= \case
  267     Left err -> do
  268       logIO logger WarningS ("could not load config: " <> ls err)
  269       pure defaultRuntimeConfig
  270     Right c  -> pure c
  271   secCfg <- loadSecurityConfig (securityFilePath paths) >>= \case
  272     Left err -> do
  273       logIO logger WarningS ("could not load security config: " <> ls err)
  274       pure defaultSecurityConfig
  275     Right c  -> pure c
  276   -- Vault (mirrors Tui.tryOpenVault but inlined to keep this module standalone)
  277   mHandle <- tryOpenVault paths secCfg logger
  278   ref     <- newIORef mHandle
  279   let rt = VaultRuntime
  280             { vrPaths      = paths
  281             , vrConfigPath = cfgPath
  282             , vrHandleRef  = ref
  283             }
  284   mgr <- newTlsManager
  285   callCounter <- newIORef 0
  286   let pr = ProviderRuntime
  287             { prConfigPath  = cfgPath
  288             , prVault       = rt
  289             , prManager     = mgr
  290             , prCallCounter = callCounter
  291             }
  292   -- The config directory is a git repo (versioning for the evolutionary stores)
  293   let cfgRoot = spConfig paths
  294   ensureConfigRepo cfgRoot
  295   let repo = openConfigRepo cfgRoot
  296   backends <- newBackends cfgRoot repo
  297   sessionMeta <- initSession paths cfg (bAgentDefs backends)
  298   activeRef   <- newIORef sessionMeta
  299   let sr = SessionRuntime
  300              { srPaths      = paths
  301              , srConfigPath = cfgPath
  302              , srActive     = activeRef
  303              }
  304   tabsH <- newTabsHandle
  305   cli <- mkRealSignalCli
  306   tgApi <- mkRealTelegramBotApi
  307   vaultStore <- mkRealVaultStore mHandle
  308   let channelRt = ChannelRuntime { crConfigPath = cfgPath, crSignalCli = cli
  309                                  , crTelegramBotApi = tgApi
  310                                  , crVaultStore = vaultStore }
  311   -- Resolve the [signal] section + an optional vault-supplied account.
  312   -- For now the vault-supplied account is Nothing (the account comes from
  313   -- config); a future phase may pull it from the vault via CPS.
  314   askReply <- newAskReplyStore 0  -- 0 = block indefinitely (no timeout)
  315   approvals <- newApprovalCache
  316   harnessReg <- Seal.Harness.Registry.newHarnessRegistry
  317   tmuxR <- Seal.Harness.Tmux.mkRealTmuxRunner
  318   let loadCfg = do
  319         lc <- loadRuntimeConfig cfgPath
  320         pure (fromRight defaultRuntimeConfig lc)
  321   repoRegH <- mkRepoRegistryHandle (reposFilePath paths)
  322   cursorsH <- newPersistingCursorStore (cursorMapPath paths)
  323   mCursors <- loadCursorMap (cursorMapPath paths)
  324   for_ mCursors (seedCursorStore cursorsH)
  325   chanDeps <- newChannelDeps
  326         paths rt repoRegH pr backends autonomy Nothing
  327         harnessReg tmuxR (Just mgr) approvals loadCfg (isJust (untrustedExecConfigFromSecurity secCfg)) tabsH logger cursorsH
  328   let coreDeps = CoreCommandDeps
  329         { ccdVault       = rt
  330         , ccdProvider    = pr
  331         , ccdSession     = sr
  332         , ccdAgentDefs   = bAgentDefs backends
  333         , ccdCfgPath     = cfgPath
  334         , ccdPaths       = paths
  335         , ccdTabs        = tabsH
  336         , ccdTabCloseNotifier = mkTabCloseNotifier (cdCursors chanDeps) (cdReplies chanDeps)
  337         , ccdAbortReg    = cdAbortReg chanDeps
  338         , ccdStopWriter  = mkStopTranscriptWriter paths (cdBroker chanDeps)
  339         , ccdModelWriter = mkModelTranscriptWriter paths (cdBroker chanDeps)
  340         , ccdRepoReg     = repoRegH
  341         , ccdRepoSeam    = Nothing
  342         }
  343       registry = mkRegistry
  344         ( coreCommandSpecs coreDeps
  345           <> [ channelCommandSpec channelRt
  346              ]
  347         )
  348   case resolveSignalConfig (rcSignal cfg) Nothing of
  349     Left err -> logIO logger ErrorS ("seal signal: " <> ls err)
  350     Right resolved -> runSignal chanDeps registry emptyChain tabsH resolved askReply

  352 -- | Open the vault if both recipient and identity are configured. Mirrors
  353 -- 'Seal.Tui.tryOpenVault'; duplicated here to keep this module standalone
  354 -- (a later refactor can extract the shared startup).
  355 tryOpenVault :: SealPaths -> SecurityConfig -> SealLogger -> IO (Maybe Vault.VaultHandle)
  356 tryOpenVault paths cfg logger =
  357   case (scVaultRecipient cfg, scVaultIdentity cfg) of
  358     (Just _, Just _) ->
  359       resolveEncryptor cfg >>= \case
  360         Left err -> do
  361           logIO logger WarningS ("vault not available: " <> ls (T.pack (show err)))
  362           pure Nothing
  363         Right enc -> do
  364           let vcfg = Vault.VaultConfig
  365                 { Vault.vcPath    = maybe (vaultFilePath paths) T.unpack (scVaultPath cfg)
  366                 , Vault.vcKeyType = fromMaybe "x25519" (scVaultKeyType cfg)
  367                 , Vault.vcUnlock  = parseUnlockMode (scVaultUnlock cfg)
  368                 }
  369           Just <$> Vault.openVault vcfg enc
  370     _ -> pure Nothing

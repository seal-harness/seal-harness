   57 -- | Opcodes that a child agent must NEVER have access to. Mirrors Hermes'
   58 -- @DELEGATE_BLOCKED_TOOLS@. Stripping these means a child cannot:
   59 --
   60 --   * recursively spawn its own subagents (@AGENT_START@) — that would
   61 --     bypass the depth cap;
   62 --   * mutate agent definitions (@AGENT_DEF_WRITE@ / @AGENT_DEF_DELETE@) —
   63 --     only the parent should evolve the def store;
   64 --   * introspect or control live instances (@AGENT_INSTANCES@ /
   65 --     @AGENT_STATUS@ / @AGENT_STOP@ / @AGENT_INTERRUPT@) — those are
   66 --     parent/operator controls, not child tools.
   67 delegationBlocklist :: Set.Set OpName
   68 delegationBlocklist = Set.fromList
   69   [ OpName "AGENT_START"
   70   , OpName "AGENT_DEF_WRITE"
   71   , OpName "AGENT_DEF_DELETE"
   72   , OpName "AGENT_INSTANCES"
   73   , OpName "AGENT_STATUS"
   74   , OpName "AGENT_STOP"
   75   , OpName "AGENT_INTERRUPT"
   76   ]

   78 -- | The role-aware child blocklist (issue #154 §3.2): the delegation
   79 -- blocklist a CHILD's registry must apply. AGENT_START is ALWAYS
   80 -- present-but-rejecting in a child registry — the gate (its authorize)
   81 -- is the enforcement, returning the dedicated leaf/kill-switch messages
   82 -- (§3.2 item 6) instead of unknown-tool. So the blocklist drops only
   83 -- AGENT_START; every other entry (def mutation, lifecycle control)
   84 -- always applies — those stay parent/operator-only.
   85 childBlocklist :: Maybe Text -> Bool -> Set.Set OpName
   86 childBlocklist _ _ = Set.delete (OpName "AGENT_START") delegationBlocklist
   87 
   88 -- | The effective role for a spawned child (issue #154 §3.1): the def's
   89 -- role is AUTHORITATIVE; the per-task @role@ hint may only NARROW (an
   90 -- orchestrator def downgraded to leaf). Any other task hint is ignored —
   91 -- a leaf def can never be widened by task input. Returns @Nothing@ for
   92 -- an unrole'd def (implicit leaf).
   93 effectiveRole :: Maybe Text -> Maybe Text -> Maybe Text
   94 effectiveRole defRole (Just "leaf") = case defRole of
   95   Just "orchestrator" -> Just "leaf"
   96   other               -> other
   97 effectiveRole defRole _ = defRole

   99 -- | Intersect a def's @tools@ allow-list with the set of opcodes the
  100 -- harness's base child registry actually exposes ('AllowOnly' ⇒ keep only
  101 -- members; 'AllowAll' passes through unchanged — the full base registry).
  102 -- Unknown names silently drop (intersection semantics, not grants).
  103 intersectAllowList :: AllowList OpName -> (OpName -> Bool) -> AllowList OpName
  104 intersectAllowList AllowAll _         = AllowAll
  105 intersectAllowList (AllowOnly xs) inBase = AllowOnly (Set.filter inBase xs)
  106 
  107 -- | Apply a (role-aware) blocklist to a child's tool allow-list. Only
  108 -- narrows 'AllowOnly' (set-difference with the blocklist); 'AllowAll' is
  109 -- returned unchanged because the blocklist is enforced at registry-build
  110 -- time by omitting blocklisted opcodes from the ops list (we can't enumerate
  111 -- the universe of opcode names to form a complement here).
  112 narrowAllowList :: AllowList OpName -> AllowList OpName
  113 narrowAllowList = narrowAllowListWith delegationBlocklist
  114 
  115 -- | 'narrowAllowList' over an explicit (role-aware) blocklist — W2's
  116 -- second chokepoint: a def that explicitly lists @AGENT_START@ keeps it
  117 -- only when the computed blocklist also omits it.
  118 narrowAllowListWith :: Set.Set OpName -> AllowList OpName -> AllowList OpName
  119 narrowAllowListWith _ AllowAll       = AllowAll
  120 narrowAllowListWith bl (AllowOnly xs) = AllowOnly (Set.difference xs bl)
  121 
  122 -- | Filter a list of opcodes to remove any whose name is in the
  123 -- 'delegationBlocklist'. The wiring layer calls this on its base ops list
  124 -- before passing to 'Seal.ISA.Registry.mkRegistry' to build the child's
  125 -- narrowed registry. This is the primary blocklist enforcement — it works
  126 -- regardless of whether the def's @adTools@ is 'AllowAll' or 'AllowOnly'.
  127 filterBlocklisted :: [opcode] -> (opcode -> OpName) -> [opcode]
  128 filterBlocklisted ops getName =
  129   [ o | o <- ops, not (getName o `Set.member` delegationBlocklist) ]
  130 
  131 -- | Filter a list of opcodes to remove any whose name is in an explicit
  132 -- (role-aware) blocklist. W2's generalized form — 'filterBlocklisted' is
  133 -- the static-blocklist special case.
  134 filterBlocklistedWith :: [opcode] -> Set.Set OpName -> (opcode -> OpName) -> [opcode]
  135 filterBlocklistedWith ops bl getName =
  136   [ o | o <- ops, not (getName o `Set.member` bl) ]

  138 -- | The per-channel deps the worker-builder closes over. The wiring layer
  139 -- (Cli.hs, Channels.Loop.hs, Gateway.Send.hs) builds this from its own
  140 -- per-turn closure and passes it to 'mkDelegateWorker'.
  141 data DelegationWorkerDeps = DelegationWorkerDeps
  142   { dwdPaths        :: SealPaths
  143   , dwdParentSid    :: SessionId
  144     -- ^ The parent's session id — the child's transcript nests under it.
  145   , dwdAppEnv       :: Env
  146     -- ^ The top-level app env (katip logging, config) — re-used for the
  147     -- child's 'runApp'.
  148   , dwdMkUIOEnv :: Maybe FilePath -> SessionId -> IO UIOEnv
  149     -- ^ Construct the child's 'UIOEnv' (carrying the 'UntrustedIO'
  150     -- capability handle + Git 'CloneDeps') from an optional anchor
  151     -- workdir + the child's session id. When the anchor is 'Just path'
  152     -- (WU-4: the parent's workdir, @ctIsolateWorkdir = False@), the
  153     -- child reuses that workdir so repo clones and plan files are
  154     -- visible without re-cloning. When 'Nothing' (WU-4:
  155     -- @ctIsolateWorkdir = True@), the child gets a fresh empty workdir
  156     -- at @cache/workdirs/<child-session>@. The wiring layer resolves
  157     -- the security config into the env. Called at child-start time
  158     -- (after the child's sid is minted).
  159   , dwdParentWorkdir :: Maybe FilePath
  160     -- ^ The parent's workdir (WU-4). 'Just' the parent's
  161     -- @sessionWorkdir paths parentSid@ in production wiring; 'Nothing'
  162     -- in tests that don't exercise anchoring. Used by
  163     -- 'mkDelegateWorker' to compute the anchor: when the task's
  164     -- 'ctIsolateWorkdir' is 'False', the child inherits this workdir;
  165     -- when 'True', the child is isolated (anchor = 'Nothing').
  166   , dwdAutonomy     :: AutonomyLevel
  167   , dwdApprovals    :: ApprovalCache
  168   , dwdOnDemand     :: Bool
  169   , dwdParentDepth  :: Int
  170     -- ^ The parent's delegation depth; the child's depth is this + 1.
  171   , dwdResolveProvider :: AgentDef -> IO (Either Text (SomeProvider, ModelId))
  172     -- ^ Resolve the child's provider+model from the def, applying any
  173     -- delegation.provider/model/base_url override (the wiring layer reads
  174     -- the override from the RuntimeConfig and threads it here).
  175   , dwdResolveProviderOverride
  176       :: Maybe (AgentDef -> IO (Either Text (SomeProvider, ModelId)))
  177     -- ^ Test seam (issue #154): when 'Just', REPLACES 'dwdResolveProvider'
  178     -- for every child spawn. 'Nothing' in production. Gateway API
  179     -- integration tests inject a resolver returning the harness's
  180     -- 'ScriptProvider' so child turns pop the same scripted queue as the
  181     -- parent turn (no real provider call). Mirrors the 'tdMkWorker' seam
  182     -- pattern ('Nothing' = production behavior).
  183   , dwdUnionDefBackend :: AgentDefBackend
  184     -- ^ The per-turn workdir ⊕ user union agent-def backend (the SAME
  185     -- one the parent session's turn assembled). The nested wiring (§3.2)
  186     -- resolves grandchildren against it so repo-shipped orchestrators can
  187     -- spawn repo-shipped specialists. In mode=remote this reuses the
  188     -- parent's cachedWorkdirScan result — no new control-plane FS reads.
  189   , dwdChildRegistry
  190       :: AgentDef -> Int -> Maybe Text -> SessionId -> ChannelCaps -> IO Registry
  191     -- ^ Build the child's narrowed ISA registry. W2 signature (was
  192     -- @AgentDef -> SessionId -> ChannelCaps@): gains the CHILD's own
  193     -- delegation depth ('dwdParentDepth' + 1) and the child's effective
  194     -- role, both computed by 'mkDelegateWorker'. The wiring layer is
  195     -- responsible for applying the role-aware blocklist
  196     -- ('childBlocklist') to the ops list AND the def's @adTools@
  197     -- allow-list ('narrowAllowListWith' / 'intersectAllowList') and
  198     -- constructing the registry — including, for orchestrator children,
  199     -- the role-conditioned nested @AGENT_START@. The caps + sid are
  200     -- passed in so the registry can close over them (ASK_HUMAN etc.).
  201   , dwdChildSystemPrompt :: AgentDef -> ChildTask -> IO (Maybe Text)
  202     -- ^ Build the child's system prompt from the def's @adSystem@ + the
  203     -- task's @ctContext@. Runs in 'IO' so the wiring layer can load the
  204     -- auto-injected skill (default @seal-usage@) from the skill backend and
  205     -- append it. 'Nothing' means no system prompt.
  206   , dwdOnEntry :: IO ()
  207     -- ^ The on-entry hook for the child's transcript (live broadcast).
  208     -- 'pure ()' for the CLI; 'broadcastNewEntries' for web/channels.
  209   , dwdChannel :: Text
  210     -- ^ The parent session's channel label (e.g. @\"telegram\"@, @\"web\"@,
  211     -- @\"cli\"@), inherited by the child so the child's request entries are
  212     -- attributed to the same channel the parent turn ran on. Stamped into
  213     -- the child's 'aeChannel' so 'runTurn' attributes the child's user
  214     -- messages (the task goals) to the originating channel.
  215   , dwdAbortFlag :: SessionId -> IO AbortFlag
  216     -- ^ Construct the child's abort flag from the child's session id
  217     -- (looked up from the 'Seal.Tools.Exec.Abort.SessionAbortRegistry').
  218     -- The child's abort is independent of the parent's in v1 (a parent
  219     -- abort doesn't auto-abort the child, and vice versa).
  220   , dwdToolTimeout :: ToolTimeoutConfig
  221     -- ^ The per-call timeout/retry config, inherited from the parent
  222     -- (loaded once from @config.toml@ @[tool_timeout]@ at startup).
  223   }

  225 -- | Build the 'AgentWorkerBuilder' the AGENT_START opcode closes over. This
  226 -- is the shared delegation worker: open the child transcript, build the
  227 -- child env, run 'runTurn' with the goal as the first user message, capture
  228 -- the final text response as the summary, and report the outcome.
  229 --
  230 -- The summary is captured via a 'ChannelCaps' whose 'ccSend' writes to an
  231 -- IORef; 'runTurn' calls 'ccSend' with the final text response, so we read
  232 -- the IORef after the run. The child's @ccPrompt@ is a no-op (children don't
  233 -- prompt the human — that would deadlock the parent).
  234 mkDelegateWorker :: DelegationWorkerDeps -> AgentWorkerBuilder
  235 mkDelegateWorker deps agentDef childSid task _hooks = do
  236   let childDir = agentSessionDir (dwdPaths deps) (dwdParentSid deps) childSid
  237       childDepth = dwdParentDepth deps + 1
  238   createDirectoryIfMissing True childDir
  239   let resolve = case dwdResolveProviderOverride deps of
  240         Just testResolve -> testResolve agentDef
  241         Nothing          -> dwdResolveProvider deps agentDef
  242   eProv <- resolve
  243   case eProv of
  244     Left err -> pure (ChildWorkerOutcome
  245                        (Just ("agent start failed: " <> err))
  246                        CerError 0 0 (Just childSid))
  247     Right (prov, model) ->
  248       withTwoFileTranscript childDir $ \childTHandle -> do
  249         summaryRef <- newIORef (Nothing :: Maybe Text)
  250         stopFanoutDoneRef <- newIORef False
  251         let capturingCaps = def
  252               { ccSend = \t -> atomicModifyIORef' summaryRef (const (Just t, ()))
  253               , ccStreaming    = False  -- children: capture final summary, no per-delta sends
  254               }
  255         childReg <- dwdChildRegistry deps agentDef childDepth
  256                                      (effectiveRole (adRole agentDef) (ctRole task))
  257                                      childSid capturingCaps
  258         let anchor = if ctIsolateWorkdir task then Nothing else dwdParentWorkdir deps
  259         childUioEnv <- dwdMkUIOEnv deps anchor childSid
  260         childSystem <- dwdChildSystemPrompt deps agentDef task
  261         childAbortFlag <- dwdAbortFlag deps childSid
  262         -- Capture the final answer via 'aeOnStop': the loop's
  263         -- final-answer path calls 'notifyStop' (the replyFanout hook)
  264         -- rather than 'ccSend' — the double-delivery fix. The
  265         -- summaryRef-capture hooks BOTH, taking the last write.
  266         let childEnvOnStop = Just (\t -> atomicModifyIORef' summaryRef (const (Just t, ())))
  267         let env = AgentEnv
  268               { aeProvider   = prov
  269               , aeProviderLabel = providerLabel agentDef
  270               , aeModel      = model
  271               , aeSystem     = childSystem
  272               , aeRegistry   = childReg
  273               , aeTranscript = childTHandle
  274               , aeBackend    = localBackend
  275               , aeUIOEnv     = childUioEnv
  276               , aeCaps       = capturingCaps
  277               , aeSession    = childSid
  278               , aeMaxTurns   = 90
  279               , aeChannel    = dwdChannel deps
  280               , aeMessageSource = Nothing
  281               , aeAutonomy   = dwdAutonomy deps
  282               , aeApprovals  = dwdApprovals deps
  283               , aeDebugRequestsPath = Nothing
  284               , aeOnEntry    = dwdOnEntry deps
  285               , aeOnUserMessage = Nothing
  286               , aeOnStop     = childEnvOnStop
  287               , aeStopFanoutDone = stopFanoutDoneRef
  288               , aeOnToolCall = \_ _ -> pure ()
  289               , aeOnTextDelta = Nothing
  290               , aeOnDemandSchemas = dwdOnDemand deps
  291               , aeLogPath    = Nothing
  292               , aeAbortFlag  = childAbortFlag
  293               , aeToolTimeout = dwdToolTimeout deps
  294               }
  295         runApp (dwdAppEnv deps) (runTurn env (ctGoal task))
  296           `catch` \e -> writeIORef summaryRef
  297                        (Just ("child runTurn raised: " <> T.pack (show (e :: SomeException))))
  298         summary <- readIORef summaryRef
  299         let exitReason = case summary of
  300               Just _  -> CerCompleted
  301               Nothing -> CerMaxIterations
  302         -- Token counts: the child's per-turn usage is recorded in the
  303         -- transcript entries; we don't aggregate them here (would require
  304         -- reading entries.jsonl). Report 0 for now — a follow-up can sum
  305         -- the child's EKResponse entries.
  306         pure (ChildWorkerOutcome summary exitReason 0 0 (Just childSid))
  307   where
  308     providerLabel d = if T.null (adProvider d) then "ollama" else adProvider d

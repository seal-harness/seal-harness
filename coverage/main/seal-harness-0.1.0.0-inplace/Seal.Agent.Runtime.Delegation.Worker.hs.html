   74 -- | Opcodes that a child agent must NEVER have access to. Mirrors Hermes'
   75 -- @DELEGATE_BLOCKED_TOOLS@. Stripping these means a child cannot:
   76 --
   77 --   * recursively spawn its own subagents (@AGENT_START@) — that would
   78 --     bypass the depth cap;
   79 --   * mutate agent definitions (@AGENT_DEF_WRITE@ / @AGENT_DEF_DELETE@) —
   80 --     only the parent should evolve the def store;
   81 --   * introspect or control live instances (@AGENT_INSTANCES@ /
   82 --     @AGENT_STATUS@ / @AGENT_STOP@ / @AGENT_INTERRUPT@) — those are
   83 --     parent/operator controls, not child tools.
   84 delegationBlocklist :: Set.Set OpName
   85 delegationBlocklist = Set.fromList
   86   [ OpName "AGENT_START"
   87   , OpName "AGENT_DEF_WRITE"
   88   , OpName "AGENT_DEF_DELETE"
   89   , OpName "AGENT_INSTANCES"
   90   , OpName "AGENT_STATUS"
   91   , OpName "AGENT_STOP"
   92   , OpName "AGENT_INTERRUPT"
   93   ]

   95 -- | The role-aware child blocklist (issue #154 §3.2): the delegation
   96 -- blocklist a CHILD's registry must apply. AGENT_START is ALWAYS
   97 -- present-but-rejecting in a child registry — the gate (its authorize)
   98 -- is the enforcement, returning the dedicated leaf/kill-switch messages
   99 -- (§3.2 item 6) instead of unknown-tool. So the blocklist drops only
  100 -- AGENT_START; every other entry (def mutation, lifecycle control)
  101 -- always applies — those stay parent/operator-only.
  102 childBlocklist :: Maybe Text -> Bool -> Set.Set OpName
  103 childBlocklist _ _ = Set.delete (OpName "AGENT_START") delegationBlocklist
  104 
  105 -- | The effective role for a spawned child (issue #154 §3.1): the def's
  106 -- role is AUTHORITATIVE; the per-task @role@ hint may only NARROW (an
  107 -- orchestrator def downgraded to leaf). Any other task hint is ignored —
  108 -- a leaf def can never be widened by task input. Returns @Nothing@ for
  109 -- an unrole'd def (implicit leaf).
  110 effectiveRole :: Maybe Text -> Maybe Text -> Maybe Text
  111 effectiveRole defRole (Just "leaf") = case defRole of
  112   Just "orchestrator" -> Just "leaf"
  113   other               -> other
  114 effectiveRole defRole _ = defRole

  116 -- | Intersect a def's @tools@ allow-list with the set of opcodes the
  117 -- harness's base child registry actually exposes ('AllowOnly' ⇒ keep only
  118 -- members; 'AllowAll' passes through unchanged — the full base registry).
  119 -- Unknown names silently drop (intersection semantics, not grants).
  120 intersectAllowList :: AllowList OpName -> (OpName -> Bool) -> AllowList OpName
  121 intersectAllowList AllowAll _         = AllowAll
  122 intersectAllowList (AllowOnly xs) inBase = AllowOnly (Set.filter inBase xs)
  123 
  124 -- | Apply a (role-aware) blocklist to a child's tool allow-list. Only
  125 -- narrows 'AllowOnly' (set-difference with the blocklist); 'AllowAll' is
  126 -- returned unchanged because the blocklist is enforced at registry-build
  127 -- time by omitting blocklisted opcodes from the ops list (we can't enumerate
  128 -- the universe of opcode names to form a complement here).
  129 narrowAllowList :: AllowList OpName -> AllowList OpName
  130 narrowAllowList = narrowAllowListWith delegationBlocklist
  131 
  132 -- | 'narrowAllowList' over an explicit (role-aware) blocklist — W2's
  133 -- second chokepoint: a def that explicitly lists @AGENT_START@ keeps it
  134 -- only when the computed blocklist also omits it.
  135 narrowAllowListWith :: Set.Set OpName -> AllowList OpName -> AllowList OpName
  136 narrowAllowListWith _ AllowAll       = AllowAll
  137 narrowAllowListWith bl (AllowOnly xs) = AllowOnly (Set.difference xs bl)

  139 -- | Filter a list of opcodes to remove any whose name is in the
  140 -- 'delegationBlocklist'. The wiring layer calls this on its base ops list
  141 -- before passing to 'Seal.ISA.Registry.mkRegistry' to build the child's
  142 -- narrowed registry. This is the primary blocklist enforcement — it works
  143 -- regardless of whether the def's @adTools@ is 'AllowAll' or 'AllowOnly'.
  144 filterBlocklisted :: [opcode] -> (opcode -> OpName) -> [opcode]
  145 filterBlocklisted ops getName =
  146   [ o | o <- ops, not (getName o `Set.member` delegationBlocklist) ]
  147 
  148 -- | Filter a list of opcodes to remove any whose name is in an explicit
  149 -- (role-aware) blocklist. W2's generalized form — 'filterBlocklisted' is
  150 -- the static-blocklist special case.
  151 filterBlocklistedWith :: [opcode] -> Set.Set OpName -> (opcode -> OpName) -> [opcode]
  152 filterBlocklistedWith ops bl getName =
  153   [ o | o <- ops, not (getName o `Set.member` bl) ]

  270 -- | Build the 'AgentWorkerBuilder' the AGENT_START opcode closes over. This
  271 -- is the shared delegation worker: open the child transcript, build the
  272 -- child env, run 'runTurn' with the goal as the first user message, capture
  273 -- the final text response as the summary, and report the outcome.
  274 --
  275 -- The summary is captured via a 'ChannelCaps' whose 'ccSend' writes to an
  276 -- IORef; 'runTurn' calls 'ccSend' with the final text response, so we read
  277 -- the IORef after the run. The child's @ccPrompt@ is a no-op (children don't
  278 -- prompt the human — that would deadlock the parent).
  279 mkDelegateWorker :: DelegationWorkerDeps -> AgentWorkerBuilder
  280 mkDelegateWorker deps agentDef childSid task _hooks = do
  281   let topLevel = dwdTopLevel deps
  282       childDir = if topLevel
  283                    then sessionDir (dwdPaths deps) childSid
  284                    else agentSessionDir (dwdPaths deps) (dwdParentSid deps) childSid
  285       childDepth = if topLevel then 0 else dwdParentDepth deps + 1
  286   createDirectoryIfMissing True childDir
  287   let resolve = case dwdResolveProviderOverride deps of
  288         Just testResolve -> testResolve agentDef
  289         Nothing          -> dwdResolveProvider deps agentDef
  290   eProv <- resolve
  291   case eProv of
  292     Left err -> pure (ChildWorkerOutcome
  293                        (Just ("agent start failed: " <> err))
  294                        CerError 0 0 (Just childSid))
  295     Right (prov, model) ->
  296       withIndexedTranscript childDir $ \childTHandle -> do
  297         -- Top-level (handoff): write the session's own meta and resolve the
  298         -- repo it carries (explicit @ctRepoUrl@ else the caller's).
  299         mRepo <- if topLevel
  300                    then writeHandoffMeta deps childSid agentDef (ctRepo task)
  301                    else writeChildMeta deps childSid >> pure Nothing
  302         summaryRef <- newIORef (Nothing :: Maybe Text)
  303         stopFanoutDoneRef <- newIORef False
  304         let capturingCaps = def
  305               { ccSend = \t -> atomicModifyIORef' summaryRef (const (Just t, ()))
  306               , ccStreaming    = False  -- children: capture final summary, no per-delta sends
  307               }
  308         -- The child's full lineage, self first: the child's own id
  309         -- prepended to the parent's baked chain. Passed into the
  310         -- registry seam so the child's session ops get the whole chain.
  311         -- For top-level (handoff): lineage is just [childSid] — no parent chain.
  312         let childLineage = if topLevel
  313                              then [childSid]
  314                              else childSid : dwdParentLineage deps
  315         childReg <- dwdChildRegistry deps childLineage agentDef childDepth
  316                                       (effectiveRole (adRole agentDef) (ctRole task))
  317                                       childSid capturingCaps
  318         let anchor = if ctIsolateWorkdir task then Nothing else dwdParentWorkdir deps
  319         childUioEnv <- dwdMkUIOEnv deps anchor childSid
  320         childSystem <- dwdChildSystemPrompt deps agentDef task
  321         childAbortFlag <- dwdAbortFlag deps childSid
  322         childRepeatTrack <- dwdRepeatTrack deps childSid
  323         -- Handoff: materialize the session's repo in its OWN workdir (the
  324         -- task set @ctIsolateWorkdir@, so @dwdMkUIOEnv@ above built a fresh
  325         -- per-session workdir) through the audited SETUP_REPO opcode — the
  326         -- same op the gateway's setup-repo endpoint and @/new -r@ dispatch.
  327         -- It gets the same arguments the creating session used (url +
  328         -- branch). Runs BEFORE the first turn so the goal sees a checkout.
  329         case mRepo of
  330           Nothing -> pure ()
  331           Just repo -> do
  332             cloneDeps <- runUIOWithEnv childUioEnv uioCdCloneDeps
  333             runApp (dwdAppEnv deps) $
  334               dispatchHandoffRepo childTHandle
  335                 (handoffRepoRegistry cloneDeps (dwdAutonomy deps))
  336                 childUioEnv (dwdToolTimeout deps) childAbortFlag
  337                 repo (Just (dwdChannel deps))
  338         -- Capture the final answer via 'aeOnStop': the loop's
  339         -- final-answer path calls 'notifyStop' (the replyFanout hook)
  340         -- rather than 'ccSend' — the double-delivery fix. The
  341         -- summaryRef-capture hooks BOTH, taking the last write.
  342         let childEnvOnStop = Just (\t -> atomicModifyIORef' summaryRef (const (Just t, ())))
  343         let env = AgentEnv
  344               { aeProvider   = prov
  345               , aeProviderLabel = providerLabel agentDef
  346               , aeModel      = model
  347               , aeSystem     = childSystem
  348               , aeRegistry   = childReg
  349               , aeTranscript = childTHandle
  350               , aeBackend    = localBackend
  351               , aeUIOEnv     = childUioEnv
  352               , aeCaps       = capturingCaps
  353               , aeSession    = childSid
  354               , aeMaxTurns   = 90
  355               , aeChannel    = dwdChannel deps
  356               , aeMessageSource = Nothing
  357               , aeAutonomy   = dwdAutonomy deps
  358               , aeApprovals  = dwdApprovals deps
  359               , aeDebugRequestsPath = Nothing
  360                   , aeDebugResponsePath = Nothing
  361                   , aeMaxModelRetries = 3
  362                   , aeRetryBaseDelayMs = 1000
  363               , aeOnEntry    = dwdOnEntry deps
  364               , aeOnUserMessage = Nothing
  365               , aeOnStop     = childEnvOnStop
  366               , aeStopFanoutDone = stopFanoutDoneRef
  367               , aeOnToolCall = \_ _ -> pure ()
  368               , aeOnTextDelta = Nothing
  369               , aeOnDemandSchemas = dwdOnDemand deps
  370               , aeLogPath    = Nothing
  371                , aeAbortFlag  = childAbortFlag
  372                , aeRepeatTrack = childRepeatTrack
  373                , aeToolTimeout = dwdToolTimeout deps
  374               }
  375         runApp (dwdAppEnv deps) (runTurn env (ctGoal task))
  376           `catch` \e -> writeIORef summaryRef
  377                        (Just ("child runTurn raised: " <> T.pack (show (e :: SomeException))))
  378         summary <- readIORef summaryRef
  379         let exitReason = case summary of
  380               Just _  -> CerCompleted
  381               Nothing -> CerMaxIterations
  382         -- Token counts: the child's per-turn usage is recorded in the
  383         -- transcript entries; we don't aggregate them here (would require
  384         -- reading entries.jsonl). Report 0 for now — a follow-up can sum
  385         -- the child's EKResponse entries.
  386         pure (ChildWorkerOutcome summary exitReason 0 0 (Just childSid))
  387   where
  388     providerLabel d = if T.null (adProvider d) then "ollama" else adProvider d

  409     , smParent = Just (dwdParentSid deps)
  410     , smAgent = Nothing
  411     , smSystemOverride = Nothing
  412     , smAgentName = Nothing
  413     , smDescription = Nothing

  426 -- | The repo checkout a handoff session carries, from the caller's optional
  427 -- overrides and the calling session's own record. Pure, and used by BOTH the
  428 -- opcode (to name it in the result text) and the worker (to write the meta
  429 -- and dispatch the clone) -- one rule, applied twice, so the two never
  430 -- disagree.
  431 --
  432 -- * With no explicit repo, the caller's 'smRepoUrl' *and* 'smRepoBranch' are
  433 --   inherited: the handoff session gets the same repo and the same branch as
  434 --   the session that handed off.
  435 -- * An explicit branch overrides the inherited one (same repo, different
  436 --   branch).
  437 -- * An explicit repo stands alone: its branch is whatever the caller passed
  438 --   (or the remote default). A branch belonging to a different repo is never
  439 --   carried across.
  440 -- * A blank value on either side counts as absent; an inherited branch that
  441 --   no longer validates (a hand-edited @session.json@) is dropped rather
  442 --   than failing the handoff.
  443 handoffRepo :: Maybe Text -> Maybe Text -> Maybe SessionMeta -> Maybe RepoSpec
  444 handoffRepo explicitUrl explicitBranch mParent = case nonBlank explicitUrl of
  445   Just url -> Just (RepoSpec url (validBranch (nonBlank explicitBranch)))
  446   Nothing  -> do
  447     url <- mParent >>= nonBlank . smRepoUrl
  448     let inherited = mParent >>= validBranch . smRepoBranch
  449         branch = validBranch (nonBlank explicitBranch)
  450     pure (RepoSpec url (pick branch inherited))
  451   where
  452     nonBlank :: Maybe Text -> Maybe Text
  453     nonBlank Nothing  = Nothing
  454     nonBlank (Just t)
  455       | T.null (T.strip t) = Nothing
  456       | otherwise          = Just (T.strip t)
  457     -- A branch is only carried when it still parses as one.
  458     validBranch :: Maybe Text -> Maybe Text
  459     validBranch Nothing  = Nothing
  460     validBranch (Just b) = case mkBranchName b of
  461       Right bn -> Just (branchNameText bn)
  462       Left _   -> Nothing
  463     pick (Just x) _ = Just x
  464     pick Nothing y  = y

  466 -- | The registry the handoff session's pre-turn repo checkout dispatches
  467 -- through: exactly one opcode, the real 'SETUP_REPO', with the child's own
  468 -- 'CloneDeps' and the caller's autonomy level.
  469 --
  470 -- Deliberately NOT the session's own (def-narrowed) registry: a definition's
  471 -- @tools@ allow-list is intersected into that one, so a def that does not
  472 -- list 'SETUP_REPO' would silently skip the checkout. Only that allow-list
  473 -- is bypassed — no blocklist control is skipped ('SETUP_REPO' is not in
  474 -- 'delegationBlocklist').
  475 handoffRepoRegistry :: CloneDeps -> AutonomyLevel -> Registry
  476 handoffRepoRegistry cloneDeps autonomy = mkRegistry [setupRepoOp cloneDeps autonomy]
  477 
  478 -- | Dispatch 'SETUP_REPO' into the handoff session's transcript and record
  479 -- its result for the user (the clone / no-op / conflict / failure text),
  480 -- exactly as the gateway's @POST /api/sessions/:id/setup-repo@ endpoint and
  481 -- @/new -r@ do. ACK-before-execute: 'dispatch' durably records the
  482 -- invocation before the clone runs. A dispatch failure is logged; a clone
  483 -- failure is recorded in the transcript. Neither aborts the session — the
  484 -- goal may not need the files.
  485 dispatchHandoffRepo
  486   :: IndexedTranscriptHandle -> Registry -> UIOEnv -> ToolTimeoutConfig
  487   -> AbortFlag -> RepoSpec -> Maybe Text -> App ()
  488 dispatchHandoffRepo h reg uioEnv toolTimeout abortFlag repo mChannel = do
  489   -- Same argument shape the creating session used: @{url, branch?}@. The
  490   -- branch key is omitted (never null) when there is none, so the opcode
  491   -- asks for the remote's default branch.
  492   let input = object
  493         ([ "url" .= rsUrl repo ]
  494          <> maybe [] (\b -> ["branch" .= b]) (rsBranch repo))
  495   res <- dispatch reg h localBackend uioEnv toolTimeout abortFlag
  496                     (OpName "SETUP_REPO") input
  497   case res of
  498     Left dErr ->
  499       liftIO (globalLogIO WarningS
  500                (ls ("[handoff] SETUP_REPO dispatch failed for " <> rsUrl repo
  501                     <> ": " <> T.pack (show dErr))))
  502     Right opRes ->
  503       liftIO (recordSetupRepoResult h (OpName "SETUP_REPO") input opRes mChannel)

  505 -- | Write a 'SessionMeta' for a top-level handoff session into its
  506 -- top-level dir (@sessions\/\<sid\>\/session.json@): the session's own id,
  507 -- the originating session's provider/model (for audit trail), channel
  508 -- @"agent"@, 'smParent' = 'Nothing' (top-level — no parent), the def the
  509 -- session runs (so a resume keeps its agent), and the repo it carries
  510 -- (the task's @ctRepo@ when the opcode resolved one — the caller's explicit
  511 -- values, else the calling session's own repo *and* branch; see
  512 -- 'handoffRepo', which is idempotent and also covers a worker invoked with
  513 -- no task repo at all). The handoff session IS visible to
  514 -- 'Seal.Session.Store.listSessions' (it's a top-level dir, not nested under
  515 -- @agents\/@). Returns the effective checkout, which the caller dispatches.
  516 writeHandoffMeta
  517   :: DelegationWorkerDeps -> SessionId -> AgentDef -> Maybe RepoSpec
  518   -> IO (Maybe RepoSpec)
  519 writeHandoffMeta deps childSid agentDef taskRepo = do
  520   mParent <- loadParentMeta deps
  521   now <- getCurrentTime
  522   let (provider, model) = case mParent of
  523         Just pm -> (smProvider pm, smModel pm)
  524         Nothing -> ("unknown", "unknown")
  525       repo = handoffRepo (rsUrl <$> taskRepo) (taskRepo >>= rsBranch) mParent
  526       agentName = let n = T.strip (adName agentDef)
  527                   in if T.null n then Nothing else Just n
  528   saveSessionMeta (dwdPaths deps) SessionMeta
  529     { smId = childSid
  530     , smProvider = provider
  531     , smModel = model
  532     , smChannel = "agent"
  533     , smParent = Nothing
  534     , smAgent = Just (adId agentDef)
  535     , smSystemOverride = Nothing
  536     , smAgentName = agentName
  537     , smDescription = Nothing
  538     , smSealFrontMatter = True
  539     , smRepoUrl = rsUrl <$> repo
  540     , smRepoBranch = repo >>= rsBranch
  541     , smCreatedAt = now
  542     , smLastActive = now
  543     }
  544   pure repo

   68 vaultCommandSpec :: VaultRuntime -> CommandSpec
   69 vaultCommandSpec rt = CommandSpec
   70   { csName         = CommandName "vault"
   71   , csAliases      = []
   72   , csGroup        = GroupVault
   73   , csSynopsis     = "Manage the encrypted secret vault"
   74   , csParserInfo   = vaultParserInfo rt
   75   , csAvailability = InteractiveOnly
   76   }
   77 
   78 vaultParserInfo :: VaultRuntime -> ParserInfo CommandAction
   79 vaultParserInfo rt =
   80   info (vaultParser rt <**> helper)
   81     (  progDesc "Encrypted secret vault operations"
   82     <> header   "vault — manage secrets in the on-disk encrypted vault"
   83     )
   84 
   85 vaultParser :: VaultRuntime -> Parser CommandAction
   86 vaultParser rt = hsubparser
   87   (  command "setup"
   88        (info (pure (setupCmd rt))
   89              (progDesc "Set up the vault backend and create the vault"))
   90   <> command "add"
   91        (info (addCmd rt <$> nameArg)
   92              (progDesc "Add or update a secret (value entered via hidden prompt)"))
   93   <> command "get"
   94        (info (getCmd rt <$> nameArg)
   95              (progDesc "Retrieve and reveal a secret"))
   96   <> command "list"
   97        (info (pure (listCmd rt))
   98              (progDesc "List all secret names (values are never shown)"))
   99   <> command "delete"
  100        (info (deleteCmd rt <$> nameArg)
  101              (progDesc "Delete a secret"))
  102   <> command "lock"
  103        (info (pure (lockCmd rt))
  104              (progDesc "Lock the vault (clear the decrypted cache)"))
  105   <> command "unlock"
  106        (info (pure (unlockCmd rt))
  107              (progDesc "Unlock the vault (decrypt into memory cache)"))
  108   <> command "status"
  109        (info (pure (statusCmd rt))
  110              (progDesc "Show vault status: locked, secret count, key type"))
  111   <> metavar "COMMAND"
  112   )

  114 nameArg :: Parser Text
  115 nameArg = T.pack <$> strArgument (metavar "NAME" <> help "Secret name")

  121 -- | Run k with the vault handle, or send "not configured" and return.
  122 withHandle :: VaultRuntime -> ChannelCaps -> (VaultHandle -> IO ()) -> IO ()
  123 withHandle rt caps k = do
  124   mh <- readIORef (vrHandleRef rt)
  125   case mh of
  126     Nothing -> ccSend caps "vault not configured — run /vault setup"
  127     Just h  -> k h
  128 
  129 vaultErrMsg :: VaultError -> Text
  130 vaultErrMsg VaultLocked           = "vault is locked — run /vault unlock"
  131 vaultErrMsg VaultNotFound         = "vault not found — run /vault setup"
  132 vaultErrMsg VaultAlreadyExists    = "vault already exists"
  133 vaultErrMsg (VaultKeyNotFound k)  = "no such secret: " <> k
  134 vaultErrMsg (VaultBackendError t) = "backend error: " <> t
  135 
  136 handleResult :: ChannelCaps -> Either VaultError a -> (a -> IO ()) -> IO ()
  137 handleResult caps (Left e)  _ = ccSend caps (vaultErrMsg e)
  138 handleResult _    (Right a) k = k a

  144 setupCmd :: VaultRuntime -> CommandAction
  145 setupCmd rt = commandAction $ \caps -> do
  146   -- Snapshot old security config BEFORE any modifications so rekeyExisting can use it.
  147   let secPath = securityFilePath (vrPaths rt)
  148   oldCfg <- loadSecurityConfig secPath
  149   let vaultPath = case oldCfg of
  150         Left _    -> vaultFilePath (vrPaths rt)
  151         Right cfg -> maybe (vaultFilePath (vrPaths rt)) T.unpack (scVaultPath cfg)
  152       hasKey = case oldCfg of
  153         Right cfg -> isJust (scVaultRecipient cfg) && isJust (scVaultIdentity cfg)
  154         Left _    -> False
  155   vaultExists <- doesFileExist vaultPath
  156   -- An existing vault WITH a recorded key means re-running setup rotates that
  157   -- key; confirm first and reassure that secrets are preserved (the old key is
  158   -- kept and nothing is decrypted-then-lost). A vault file with no key recorded
  159   -- in config is an orphan from an interrupted setup — fall through so the
  160   -- orphan branch in rekeyExisting can tell the user to delete it.
  161   proceed <-
  162     if vaultExists && hasKey
  163       then do
  164         ccSend caps
  165           (  "A vault already exists. Continuing will ROTATE its key: every "
  166           <> "secret is re-encrypted to a new key. Your existing secrets are "
  167           <> "preserved, and the current key file is kept (never overwritten).")
  168         ans <- ccPrompt caps (AskPrompt "Rotate the vault key? [y/N]: " [])
  169         pure (T.toLower (T.strip ans) `elem` ["y", "yes"])
  170       else pure True
  171   if not proceed
  172     then ccSend caps "Setup cancelled; the vault is unchanged."
  173     else runSetup rt caps oldCfg
  174 
  175 -- | Backend selection + key generation + vault init/rekey. Split out of
  176 -- 'setupCmd' so the existing-vault rotation guard stays readable. New keys are
  177 -- always written to a fresh identity file (see 'freshKeyName' in
  178 -- "Seal.Vault.Backend"), so a rotation never overwrites the key the current
  179 -- vault still depends on.
  180 runSetup :: VaultRuntime -> ChannelCaps -> Either Text SecurityConfig -> IO ()
  181 runSetup rt caps oldCfg = do
  182   plugins <- detectAgePlugins
  183   let hasYubi = "yubikey" `elem` plugins
  184   ccSend caps "Available vault backends:"
  185   ccSend caps "  1. Local age key (age-keygen) — key stored on disk"
  186   when hasYubi $
  187     ccSend caps "  2. YubiKey (age-plugin-yubikey) — key stays on token [recommended]"
  188   let userNum = if hasYubi then "3" else "2"
  189   ccSend caps ("  " <> userNum <> ". User-supplied (bring your own key)")
  190   choice <- T.strip <$> ccPrompt caps (AskPrompt "Choose backend [1]: " [])
  191   let effective = if T.null choice then "1" else choice
  192   rkResult <- case (effective, hasYubi) of
  193     ("1", _)     -> setupLocalAgeKey (vrPaths rt) "default"
  194     ("2", True)  -> do
  195       tp <- ccPrompt caps (AskPrompt "Require touch? [y/N]: " [])
  196       let touch = T.toLower (T.strip tp) `elem` ["y", "yes"]
  197       pp <- ccPrompt caps
  198         (AskPrompt "Require PIN on each decrypt session? [Y/n] (choose n for no PIN): " [])
  199       let pin = T.toLower (T.strip pp) `notElem` ["n", "no"]
  200       setupYubiKey (vrPaths rt) "default" touch pin caps
  201     ("2", False) -> setupUserSupplied caps
  202     ("3", True)  -> setupUserSupplied caps
  203     (other, _)   -> pure (Left ("Invalid choice: " <> other))
  204   case rkResult of
  205     Left err -> ccSend caps ("Setup failed: " <> err)
  206     Right rk -> do
  207       encResult <- mkAgeEncryptor
  208         (AgeRecipient (rkRecipient rk))
  209         (AgeIdentity  (rkIdentity  rk))
  210       case encResult of
  211         Left e -> ccSend caps (vaultErrMsg e)
  212         Right enc -> do
  213           -- Honor vault_path from config if set; fall back to the default path.
  214           -- tryOpenVault in Seal.Tui uses the same expression to stay in sync.
  215           let vaultPath = case oldCfg of
  216                 Left _    -> vaultFilePath (vrPaths rt)
  217                 Right cfg -> maybe (vaultFilePath (vrPaths rt)) T.unpack (scVaultPath cfg)
  218               vaultCfg = VaultConfig
  219                 { vcPath   = vaultPath
  220                 , vcKeyType = rkKeyType rk
  221                 , vcUnlock  = UnlockOnDemand
  222                 }
  223           h <- openVault vaultCfg enc
  224           initResult <- vhInit h
  225           case initResult of
  226             Right () -> do
  227               ur <- updateSecurityConfig (securityFilePath (vrPaths rt)) $ \sc -> sc
  228                 { scVaultRecipient = Just (rkRecipient rk)
  229                 , scVaultIdentity  = Just (rkIdentity  rk)
  230                 , scVaultKeyType   = Just (rkKeyType   rk)
  231                 }
  232               case ur of
  233                 Left err -> ccSend caps ("Config write failed: " <> err)
  234                 Right () -> do
  235                   writeIORef (vrHandleRef rt) (Just h)
  236                   ccSend caps "Vault created successfully."
  237             Left VaultAlreadyExists ->
  238               rekeyExisting rt caps enc rk oldCfg
  239             Left e ->
  240               ccSend caps (vaultErrMsg e)
  241 
  242 -- | Rekey an existing vault with a new encryptor.
  243 -- Uses the OLD config snapshot (captured before any modifications) to build
  244 -- the old encryptor; updates the config after a successful rekey.
  245 rekeyExisting
  246   :: VaultRuntime
  247   -> ChannelCaps
  248   -> VaultEncryptor
  249   -> ResolvedKey
  250   -> Either Text SecurityConfig
  251   -> IO ()
  252 rekeyExisting rt caps newEnc rk eCfg = do
  253   case eCfg of
  254     Left err ->
  255       ccSend caps ("Cannot read existing config for rekey: " <> err)
  256     Right oldCfg ->
  257       -- Honor vault_path from config if set; fall back to the default path.
  258       -- tryOpenVault in Seal.Tui uses the same expression to stay in sync.
  259       let vaultPath = maybe (vaultFilePath (vrPaths rt)) T.unpack (scVaultPath oldCfg)
  260       in case (scVaultRecipient oldCfg, scVaultIdentity oldCfg) of
  261         (Nothing, Nothing) ->
  262           -- Vault file exists but config has no key: a previous setup was
  263           -- interrupted before the config was written.  Tell the user to
  264           -- remove the orphaned file rather than trying to re-encrypt it.
  265           ccSend caps
  266             (  "vault file exists at "
  267             <> T.pack vaultPath
  268             <> " but the config has no key recorded — it was likely left from"
  269             <> " an interrupted setup. Delete that file and re-run '/vault setup'.")
  270         _ -> do
  271           oldEncResult <- resolveEncryptor oldCfg
  272           case oldEncResult of
  273             Left e ->
  274               ccSend caps ("Cannot load existing key for rekey: " <> vaultErrMsg e)
  275             Right oldEnc -> do
  276               let oldVaultCfg = VaultConfig
  277                     { vcPath    = vaultPath
  278                     , vcKeyType = fromMaybe "unknown" (scVaultKeyType oldCfg)
  279                     , vcUnlock  = UnlockOnDemand
  280                     }
  281               oldH <- openVault oldVaultCfg oldEnc
  282               -- vhRekey reads from disk directly; no explicit unlock needed.
  283               -- Rotation was already confirmed up front in setupCmd, so show the
  284               -- old/new key summary and proceed without a second prompt.
  285               let confirmRekey msg = ccSend caps msg >> pure True
  286               rekeyResult <- vhRekey oldH newEnc (rkKeyType rk) confirmRekey
  287               case rekeyResult of
  288                 Right () -> do
  289                   ur <- updateSecurityConfig (securityFilePath (vrPaths rt)) $ \sc -> sc
  290                     { scVaultRecipient = Just (rkRecipient rk)
  291                     , scVaultIdentity  = Just (rkIdentity  rk)
  292                     , scVaultKeyType   = Just (rkKeyType   rk)
  293                     }
  294                   case ur of
  295                     Left err -> ccSend caps
  296                       (  "ERROR: vault was re-encrypted with the new key, but saving the"
  297                       <> " config failed (" <> err <> "). The vault on disk now uses the"
  298                       <> " NEW key, but security.toml still names the OLD key. Do NOT re-run"
  299                       <> " '/vault setup' — that would attempt to decrypt with the old key"
  300                       <> " and fail. Instead, manually edit config.toml to set"
  301                       <> " vault_recipient, vault_identity, and vault_key_type to the"
  302                       <> " values just configured, or restore config.toml from a backup.")
  303                     Right () -> do
  304                       writeIORef (vrHandleRef rt) (Just oldH)
  305                       ccSend caps "Vault rekeyed successfully."
  306                 Left e ->
  307                   ccSend caps (vaultErrMsg e)

  309 addCmd :: VaultRuntime -> Text -> CommandAction
  310 addCmd rt name = commandAction $ \caps ->
  311   withHandle rt caps $ \h -> do
  312     val <- ccPromptSecret caps ("Value for " <> name <> ": ")
  313     result <- vhPut h name (TE.encodeUtf8 val)
  314     handleResult caps result $ \() ->
  315       ccSend caps ("Secret '" <> name <> "' stored.")
  316 
  317 getCmd :: VaultRuntime -> Text -> CommandAction
  318 getCmd rt name = commandAction $ \caps ->
  319   withHandle rt caps $ \h -> do
  320     result <- vhGet h name
  321     handleResult caps result (ccSend caps . TE.decodeUtf8Lenient)
  322 
  323 listCmd :: VaultRuntime -> CommandAction
  324 listCmd rt = commandAction $ \caps ->
  325   withHandle rt caps $ \h -> do
  326     result <- vhList h
  327     handleResult caps result (mapM_ (ccSend caps))
  328 
  329 deleteCmd :: VaultRuntime -> Text -> CommandAction
  330 deleteCmd rt name = commandAction $ \caps ->
  331   withHandle rt caps $ \h -> do
  332     result <- vhDelete h name
  333     handleResult caps result $ \() ->
  334       ccSend caps ("Secret '" <> name <> "' deleted.")
  335 
  336 lockCmd :: VaultRuntime -> CommandAction
  337 lockCmd rt = commandAction $ \caps ->
  338   withHandle rt caps $ \h -> do
  339     vhLock h
  340     ccSend caps "Vault locked."
  341 
  342 unlockCmd :: VaultRuntime -> CommandAction
  343 unlockCmd rt = commandAction $ \caps ->
  344   withHandle rt caps $ \h -> do
  345     result <- vhUnlock h
  346     handleResult caps result $ \() ->
  347       ccSend caps "Vault unlocked."
  348 
  349 statusCmd :: VaultRuntime -> CommandAction
  350 statusCmd rt = commandAction $ \caps ->
  351   withHandle rt caps $ \h -> do
  352     st <- vhStatus h
  353     ccSend caps $ T.unlines
  354       [ "locked:  " <> (if vsLocked st then "yes" else "no")
  355       , "secrets: " <> T.pack (show (vsSecretCount st))
  356       , "key:     " <> vsKeyType st

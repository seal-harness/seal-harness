    2 -- | The gateway-owned credential store: @~\/.seal\/credentials.toml@.
    3 --
    4 -- Credentials live here rather than in @security.toml@ because the two have
    5 -- opposite write patterns. @security.toml@ is the boot-critical document —
    6 -- unparseable means the gateway does not start — and it is rewritten only by
    7 -- human-driven admin commands. Credentials change on a successful login (the
    8 -- signature counter) and on every enrollment or deletion, so keeping them in
    9 -- the boot-critical file means a login rewrites the document the next boot
   10 -- depends on.
   11 --
   12 -- Format (design §6.2):
   13 --
   14 -- > version = 1
   15 -- >
   16 -- > [credentials.<credential-id-base64url>]
   17 -- > public_key = "<base64url, 64-byte x||y>"
   18 -- > sign_count = 0
   19 -- > label      = "passkey"
   20 -- > created_at = "2026-10-30T14:00:00Z"
   21 -- > rp_id      = "localhost"
   22 -- > added_via  = "cli_token"
   23 --
   24 -- The file is a sibling of @security.toml@ (not under @config\/@), written
   25 -- atomically with mode @0600@. The temp name is unique per call so two
   26 -- processes cannot collide on it before the rename.

   70 -- | The decoded @credentials.toml@: a format version plus the credential map
   71 -- keyed by base64url credential id.
   72 --
   73 -- @version@ is explicit so a future format change is a boot refusal
   74 -- ('BrCredentialsVersion') rather than a silent misread.
   75 data CredentialsFile = CredentialsFile
   76   { crdVersion     :: !Int
   77   , crdCredentials :: !(Map Text CredentialFileConfig)
   78   } deriving stock (Eq, Show)

   80 -- | Bidirectional codec for the whole file.
   81 --
   82 -- 'Seal.Config.Security.credentialCodec' is reused rather than duplicated: the
   83 -- same @[..credentials.\<id\>]@ table shape is still decoded out of
   84 -- @security.toml@ by the legacy-layout check until the migration lands.
   85 credentialsCodec :: Toml.TomlCodec CredentialsFile
   86 credentialsCodec = CredentialsFile
   87   <$> Toml.int "version" .= crdVersion
   88   <*> Toml.tableMap Toml._KeyText (Toml.table credentialCodec) "credentials"
   89         .= crdCredentials
   90 
   91 -- | Make the implicit @credentials@ node explicit before decoding.
   92 --
   93 -- tomland's 'Toml.tableMap' looks the table up under its @[credentials]@
   94 -- header, so an operator-written file that carries only @[credentials.\<id\>]@
   95 -- sub-tables leaves that node implicit in the prefix tree and the lookup
   96 -- yields an EMPTY map (learnings 2026-10-30 §4). An empty store is not a parse
   97 -- error, so without this the operator's passkeys disappear silently — the
   98 -- worst failure mode available here, because the file still "loads".
   99 --
  100 -- A file written by 'saveCredentials' already has the explicit header and
  101 -- passes through unchanged.
  102 normalizeCredentialsTable :: Toml.TOML -> Toml.TOML
  103 normalizeCredentialsTable toml =
  104   toml { Toml.tomlTables =
  105            HashMap.adjust (fixNode "credentials") "credentials" (Toml.tomlTables toml) }
  106 
  107 -- | Render a 'CredentialsFile' as TOML.
  108 encodeCredentials :: CredentialsFile -> Text
  109 encodeCredentials = Toml.encode credentialsCodec
  110 
  111 -- | Parse and decode a 'CredentialsFile' from raw TOML text.
  112 decodeCredentials :: Text -> Either Text CredentialsFile
  113 decodeCredentials raw = case Toml.parse raw of
  114   Left err -> Left (Toml.unTomlParseError err)
  115   Right toml -> case Toml.runTomlCodec credentialsCodec
  116                        (normalizeCredentialsTable toml) of
  117     Success file -> Right file
  118     Failure errs -> Left (Toml.prettyTomlDecodeErrors errs)

  120 -- | Load @credentials.toml@.
  121 --
  122 -- Three outcomes, distinguished because boot treats them differently
  123 -- (design §10): @Right Nothing@ — the file is absent (a normal state: the
  124 -- operator has not enrolled anything yet); @Right (Just file)@ — decoded;
  125 -- @Left@ — present but unparseable or semantically invalid.
  126 loadCredentials :: FilePath -> IO (Either Text (Maybe CredentialsFile))
  127 loadCredentials path = do
  128   exists <- doesFileExist path
  129   if not exists
  130     then pure (Right Nothing)
  131     else do
  132       raw <- TIO.readFile path
  133       pure (Just <$> decodeCredentials raw)
  134 
  135 -- | Write @file@ to @path@ atomically, mode @0600@.
  136 --
  137 -- The temp file gets a unique name from 'openBinaryTempFile' (not a fixed
  138 -- @.tmp@ suffix) so two processes writing concurrently cannot collide on the
  139 -- temp path and produce a truncated document; the rename is atomic, and
  140 -- @0600@ is applied at creation and again explicitly to defeat a permissive
  141 -- umask. The security write lock is process-local, so the unique temp name
  142 -- plus the atomic rename are the only cross-process guarantees.
  143 saveCredentials :: FilePath -> CredentialsFile -> IO ()
  144 saveCredentials path file = do
  145   let dir = takeDirectory path
  146   (tmp, h) <- openBinaryTempFile dir (takeBaseName path <> ".tmp")
  147   BS.hPut h (TE.encodeUtf8 (encodeCredentials file))
  148   hClose h
  149   setFileMode tmp 0o600
  150   renameFile tmp path
  151 
  152 -- | The format version this build writes.
  153 credentialsVersion :: Int
  154 credentialsVersion = 1
  155 
  156 -- | Create a store that does not exist yet.
  157 --
  158 -- Refuses when the file is already there, and refuses an empty credential map.
  159 -- The two refusals are the store's integrity rules (design §9.2): the store is
  160 -- never empty, and an existing store is never silently replaced — re-creating
  161 -- it would discard credentials the operator cannot get back.
  162 --
  163 -- The destination is opened with @O_CREAT|O_EXCL@ rather than written through a
  164 -- temp file: @rename@ would happily replace an existing store, whereas
  165 -- @O_EXCL@ is the only way to make "does not already exist" atomic across
  166 -- processes. @0600@ is applied at creation, so credential public keys are
  167 -- never world-readable.
  168 createCredentialsFile :: FilePath -> Map Text CredentialFileConfig -> IO (Either Text ())
  169 createCredentialsFile path creds =
  170   withMVar securityWriteLock $ \_ -> createCredentialsFileUnlocked path creds
  171 
  172 -- | 'createCredentialsFile' without the lock, for a caller that already holds it.
  173 createCredentialsFileUnlocked :: FilePath -> Map Text CredentialFileConfig -> IO (Either Text ())
  174 createCredentialsFileUnlocked path creds
  175   | Map.null creds = pure (Left emptyStoreRefusal)
  176   | otherwise = do
  177       exists <- doesFileExist path
  178       if exists
  179         then pure (Left (existsRefusal path))
  180         else do
  181           opened <- try (openFd path WriteOnly defaultFileFlags
  182                            { creat = Just 0o600, exclusive = True })
  183           case opened of
  184             Left (err :: IOException) -> pure (Left (T.pack path
  185               <> ": could not create the credential store: "
  186               <> T.pack (displayException err)))
  187             Right fd -> do
  188               h <- fdToHandle fd
  189               BS.hPut h (TE.encodeUtf8
  190                            (encodeCredentials (CredentialsFile credentialsVersion creds)))
  191               hClose h
  192               setFileMode path 0o600
  193               pure (Right ())
  194 
  195 -- | Replace the credential map of an existing store.
  196 --
  197 -- The callback returns 'Either' so a rule that depends on the /current/
  198 -- contents — the last-usable-credential guard — is evaluated inside the write,
  199 -- not by a read-then-write in the caller that another request can invalidate in
  200 -- between. A @Left@ from the callback writes nothing.
  201 --
  202 -- Refuses a missing file: enrollment creates the store, every later mutation
  203 -- requires it. Without that rule, a store deleted underneath a running gateway
  204 -- would read as empty and the next write would report success.
  205 updateCredentialsFile
  206   :: FilePath
  207   -> (Map Text CredentialFileConfig -> Either Text (Map Text CredentialFileConfig))
  208   -> IO (Either Text ())
  209 updateCredentialsFile path f =
  210   withMVar securityWriteLock $ \_ -> updateCredentialsFileUnlocked path f
  211 
  212 -- | 'updateCredentialsFile' without the lock, for a caller that already holds it.
  213 updateCredentialsFileUnlocked
  214   :: FilePath
  215   -> (Map Text CredentialFileConfig -> Either Text (Map Text CredentialFileConfig))
  216   -> IO (Either Text ())
  217 updateCredentialsFileUnlocked path f = do
  218   loaded <- loadCredentials path
  219   case loaded of
  220     Left err          -> pure (Left err)
  221     Right Nothing     -> pure (Left (missingRefusal path))
  222     Right (Just file) -> case f (crdCredentials file) of
  223       Left err -> pure (Left err)
  224       Right creds
  225         | Map.null creds -> pure (Left emptyStoreRefusal)
  226         | otherwise -> do
  227             let updated = file { crdCredentials = creds }
  228             -- Verify the encoded document decodes back to what we intend
  229             -- before it replaces the store, so a codec change that drops a
  230             -- field is a refusal rather than silent data loss.
  231             case decodeCredentials (encodeCredentials updated) of
  232               Right back | back == updated ->
  233                 saveCredentials path updated >> pure (Right ())
  234               _ -> pure (Left roundTripRefusal)
  235 
  236 emptyStoreRefusal :: Text
  237 emptyStoreRefusal =
  238   "refusing to write an empty credential store: removing the last credential \
  239   \would lock the gateway out with no way back in"
  240 
  241 existsRefusal :: FilePath -> Text
  242 existsRefusal path =
  243   "refusing to create the credential store: " <> T.pack path <> " already exists"
  244 
  245 missingRefusal :: FilePath -> Text
  246 missingRefusal path =
  247   "refusing to update the credential store: " <> T.pack path <> " does not exist"
  248 
  249 roundTripRefusal :: Text
  250 roundTripRefusal =
  251   "refusing to write credentials.toml: the encoded document does not round-trip \
  252   \the new credential map"

  255 -- Migration from the pre-split layout
  256 --------------------------------------------------------------------------------
  257 
  258 -- | What a migration did, so the caller can print something true. A no-op is a
  259 -- success, not an error: re-running the command is how you finish an
  260 -- interrupted run.
  261 data MigrationOutcome
  262   = NothingToMigrate
  263   | Migrated !Int
  264     -- ^ How many credentials moved.
  265   deriving stock (Eq, Show)
  266 
  267 -- | Move the credential tables out of @security.toml@.
  268 --
  269 -- Order is deliberate: the store is written first, the tables are removed
  270 -- second. A crash between the two leaves both copies, which boot refuses with
  271 -- @BrLegacyLayout@ — loud, and re-running finishes the job. Removing first
  272 -- would leave nothing anywhere.
  273 --
  274 -- @force@ re-runs only the /table removal/. It never overwrites an existing
  275 -- @credentials.toml@: that file is the only copy of credentials the operator
  276 -- may still be able to sign in with, and a migration is not allowed to be the
  277 -- thing that destroys it. If the store already exists, the credentials in
  278 -- @security.toml@ are stale leftovers and only the tables are dropped.
  279 migrateLegacyCredentials
  280   :: FilePath      -- ^ security.toml
  281   -> FilePath      -- ^ credentials.toml
  282   -> Bool          -- ^ force: re-run the table removal
  283   -> IO (Either Text MigrationOutcome)
  284 migrateLegacyCredentials securityPath credentialsPath force = do
  285   securityExists <- doesFileExist securityPath
  286   if not securityExists
  287     then pure (Right NothingToMigrate)
  288     else do
  289       raw <- TIO.readFile securityPath
  290       case readLegacyCredentials raw of
  291         Left err -> pure (Left ("could not read the pre-split credential tables: " <> err))
  292         Right legacy
  293           | Map.null legacy -> pure (Right NothingToMigrate)
  294           | otherwise -> do
  295               storeExists <- doesFileExist credentialsPath
  296               if storeExists && not force
  297                 then pure (Left (storeExistsRefusal credentialsPath))
  298                 else do
  299                   written <-
  300                     if storeExists
  301                       then pure (Right ())
  302                       else createCredentialsFile credentialsPath (Map.map markLegacy legacy)
  303                   case written of
  304                     Left err -> pure (Left err)
  305                     Right () -> dropTables raw >> pure (Right (Migrated (Map.size legacy)))
  306   where
  307     -- Every credential the old layout produced claims only what is true of it:
  308     -- that it predates the grant vocabulary.
  309     markLegacy cfg = cfg { cfcAddedVia = Just "legacy" }
  310 
  311     -- Re-encode security.toml from the decoded config. The tables go away
  312     -- because the codec does not know them; 'DropCredentials' is the exemption
  313     -- that lets this write through the credential-preserving guard, and this is
  314     -- one of the two @--yes@-gated callers allowed to use it.
  315     dropTables _raw = do
  316       loaded <- loadSecurityConfig securityPath
  317       case loaded of
  318         Left err -> pure (Left err)
  319         Right cfg -> do
  320           saveSecurityConfig securityPath cfg DropCredentials
  321 
  322 storeExistsRefusal :: FilePath -> Text
  323 storeExistsRefusal path =
  324   "refusing to migrate: " <> T.pack path <> " already exists.\n"
  325     <> "  It is the only copy of credentials the gateway can sign in with, so it is\n"
  326     <> "  never overwritten. Re-run with --force to drop the stale tables in\n"
  327     <> "  security.toml and keep the existing store."

  329 --------------------------------------------------------------------------------
  330 -- Enrollment: raise the mode and write the credential as one operation
  331 --------------------------------------------------------------------------------
  332 
  333 -- | Turn access control on and add a credential, atomically (design §6.1).
  334 --
  335 -- The two writes go to two files, so neither is atomic on its own; what has to
  336 -- be atomic is the /pair/. Two failures are possible and both matter:
  337 --
  338 -- * a crash between the two — mode-first ordering leaves @passkey@ with no
  339 --   credentials, which resolves to the loud, grant-gated bootstrap. The reverse
  340 --   ordering would leave a credential on a still-open gateway: silent
  341 --   non-enforcement;
  342 -- * two ceremonies interleaving — one raises, the other raises and writes, the
  343 --   first then writes and fails. Without one lock around both, a store can end
  344 --   up holding a credential while the mode says @none@.
  345 --
  346 -- So the lock is taken exactly once, here, and the unlocked primitives are used
  347 -- inside it. 'securityWriteLock' is non-reentrant: calling the public writers
  348 -- from within would self-deadlock.
  349 --
  350 -- A failed credential write restores the previous mode via 'restoreMode', which
  351 -- only restores if the on-disk mode is still the @passkey@ this raise wrote.
  352 enrollCredential
  353   :: FilePath                        -- ^ security.toml
  354   -> FilePath                        -- ^ credentials.toml
  355   -> Map Text CredentialFileConfig   -- ^ entries to add
  356   -> IO (Either Text ())
  357 enrollCredential securityPath credentialsPath creds =
  358   withMVar securityWriteLock $ \_ -> do
  359     raised <- raiseMode securityPath
  360     case raised of
  361       Left err -> pure (Left err)
  362       Right witness -> do
  363         written <- writeStoreUnlocked credentialsPath creds
  364         case written of
  365           Right () -> pure (Right ())
  366           Left err -> do
  367             -- Put the mode back. Best effort by design: if this also fails, the
  368             -- gateway is left closed (passkey, no credential), which is the safe
  369             -- direction and is loud on the next boot.
  370             _ <- restoreMode securityPath witness
  371             pure (Left err)
  372 
  373 -- | Create the store, or merge into it, without taking the lock.
  374 writeStoreUnlocked :: FilePath -> Map Text CredentialFileConfig -> IO (Either Text ())
  375 writeStoreUnlocked path creds = do
  376   exists <- doesFileExist path
  377   if exists
  378     then updateCredentialsFileUnlocked path (Right . Map.union creds)
  379     else createCredentialsFileUnlocked path creds

    2 -- | The engram embedding backend — a subprocess-based 'EmbeddingBackend'
    3 -- that shells out to the @engram@ CLI for semantic search. engram uses
    4 -- SQLite-vec with ollama/nomic-embed-text embeddings under the hood.
    5 --
    6 -- The memory system uses a separate engram index (e.g.
    7 -- @~\/.seal\/memory\/engram.db@) from the KB index (@~\/.engram\/index.db@)
    8 -- via engram's @--index@ flag.
    9 --
   10 -- The engram CLI commands used:
   11 --
   12 --   * @engram add <file> --index <path> --no-progress@ — index a file
   13 --   * @engram remove <file> --index <path>@ — remove a file from the index
   14 --   * @engram search "<query>" --limit N --index <path> --json@ — search (JSON output)
   15 --
   16 -- JSON search output format:
   17 --
   18 -- @
   19 -- [
   20 --   { "path": "\/path\/to\/file.md", "snippet": "...", "distance": 0.68 }
   21 -- ]
   22 -- @
   23 --
   24 -- The @distance@ is converted to a relevance score via @1 - distance@.
   25 module Seal.Memory.EngramBackend
   26   ( engramEmbeddingBackend
   27   , resolveEmbeddingBackend
   28   ) where

   49 -- | Construct an 'EmbeddingBackend' backed by the engram CLI.
   50 --
   51 -- @binaryPath@ is the absolute path to the engram binary (e.g.
   52 -- @~\/.local\/bin\/engram@). @indexPath@ is the path to the SQLite index
   53 -- file (e.g. @~\/.seal\/memory\/engram.db@). The index is created on first
   54 -- use by engram itself.
   55 engramEmbeddingBackend :: Text -> FilePath -> EmbeddingBackend
   56 engramEmbeddingBackend binaryPath indexPath = EmbeddingBackend
   57   { ebIndex = \filePath content -> do
   58       -- Write content to the file so engram can index it by path.
   59       BS.writeFile filePath (TE.encodeUtf8 content)
   60       _ <- runEngram binaryPath indexPath ["add", filePath, "--no-progress"]
   61       pure ()
   62   , ebUnindex = \filePath -> do
   63       _ <- runEngram binaryPath indexPath ["remove", filePath]
   64       pure ()
   65   , ebSearch = \query limit -> do
   66       let limitStr = show limit
   67           queryStr = T.unpack query
   68       result <- runEngram binaryPath indexPath
   69         ["search", queryStr, "--limit", limitStr, "--json"]
   70       case result of
   71         Left _ -> pure []
   72         Right output -> pure (parseEngramResults output)
   73   , ebBackendName = "engram"
   74   }
   75 
   76 -- | Run an engram command with fixed argv (no shell). Returns the stdout
   77 -- on success, or an error message on failure. Best-effort — a failed
   78 -- engram command returns 'Left' and the caller (the opcode) falls back
   79 -- to substring search.
   80 runEngram :: Text -> FilePath -> [String] -> IO (Either Text BS.ByteString)
   81 runEngram binaryPath indexPath args = do
   82   let binStr = T.unpack binaryPath
   83       fullArgs = args <> ["--index", indexPath]
   84       cp = (proc binStr fullArgs)
   85         { std_in = CreatePipe, std_out = CreatePipe, std_err = CreatePipe }
   86   withCreateProcess cp $ \mIn mOut mErr ph -> do
   87     for_ mIn hClose
   88     out <- maybe (pure BS.empty) BS.hGetContents mOut
   89     _err <- maybe (pure BS.empty) BS.hGetContents mErr
   90     ec <- waitForProcess ph
   91     let !_ = BS.length out
   92     case ec of
   93       ExitSuccess -> pure (Right out)
   94       ExitFailure _ -> pure (Left ("engram command failed: " <> T.pack (unwords (binStr : fullArgs))))

   96 -- | The JSON shape of one engram search result (from @engram search --json@).
   97 data EngramResult = EngramResult
   98   { erPath     :: Text
   99   , erSnippet  :: Text
  100   , erDistance :: Double
  101   }
  102 
  103 instance FromJSON EngramResult where
  104   parseJSON = withObject "EngramResult" $ \o ->
  105     EngramResult
  106       <$> o .: "path"
  107       <*> o .: "snippet"
  108       <*> o .: "distance"
  109 
  110 -- | Parse engram's @--json@ search output into a list of 'SearchResult'.
  111 -- The @distance@ is converted to a relevance score via @1 - distance@
  112 -- (engram reports distance; we invert so higher score = more relevant).
  113 -- Malformed entries are skipped (defensive parsing).
  114 parseEngramResults :: BS.ByteString -> [SearchResult]
  115 parseEngramResults output =
  116   case decode (BL.fromStrict output) :: Maybe [EngramResult] of
  117     Just results -> mapMaybe toSearchResult results
  118     Nothing      -> []
  119   where
  120     toSearchResult er =
  121       Just SearchResult
  122         { srPath = erPath er
  123         , srContent = erSnippet er
  124         , srScore = 1.0 - erDistance er
  125         }

  127 -- | Resolve an 'EmbeddingBackend' from the optional config and seal paths.
  128 -- If the config is absent or @backend@ is @"null"@ (or absent), returns
  129 -- 'nullEmbeddingBackend'. If @backend@ is @"engram"@, constructs an
  130 -- 'engramEmbeddingBackend' with the configured (or default) binary path
  131 -- and index path.
  132 resolveEmbeddingBackend
  133   :: Maybe EmbeddingConfig
  134   -> FilePath
  135   -> IO EmbeddingBackend
  136 resolveEmbeddingBackend mCfg sealHome =
  137   case mCfg of
  138     Nothing -> pure nullEmbeddingBackend
  139     Just cfg -> case ecBackend cfg of
  140       Just b | b == "engram" -> do
  141         let binPath = maybe "~/.local/bin/engram" T.unpack (ecBinaryPath cfg)
  142             indexPath = maybe (sealHome </> "memory" </> "engram.db")
  143                              T.unpack (ecIndexPath cfg)
  144         exists <- doesFileExist binPath
  145         if exists
  146           then pure (engramEmbeddingBackend (T.pack binPath) indexPath)
  147           else pure nullEmbeddingBackend
  148       _ -> pure nullEmbeddingBackend

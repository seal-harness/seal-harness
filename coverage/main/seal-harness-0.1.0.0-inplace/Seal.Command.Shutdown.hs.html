    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Shutdown logging for @seal serve@.
    3 --
    4 -- Ctrl-C is the normal way to stop the server, and it is also the moment when
    5 -- every in-process sub-agent dies: children run as forked threads in this
    6 -- process ('Seal.Agent.Runtime.Registry'), so terminating the server
    7 -- terminates them. Nothing marks that in the log today, which makes
    8 -- "my sub-agents died and never came back" hard to correlate with "I stopped
    9 -- the server". This module closes that gap:
   10 --
   11 -- * On 'UserInterrupt' (what the GHC RTS raises on the main thread for SIGINT)
   12 --   it logs a shutdown marker plus an inventory of the live sub-agents — id,
   13 --   parent session, child session, depth, status — and then exits with
   14 --   'shutdownExitCode' (130, the conventional @128 + SIGINT@).
   15 -- * Every other async exception is re-raised untouched: this handler must
   16 --   never swallow 'ThreadKilled' or a timeout.
   17 --
   18 -- No POSIX signal handler is installed. The RTS already converts SIGINT into
   19 -- 'UserInterrupt' and restores the default SIGINT disposition first, so a
   20 -- second Ctrl-C still hard-kills a wedged process. Installing our own handler
   21 -- would take that escape hatch away.
   22 module Seal.Command.Shutdown
   23   ( handleServeInterrupt
   24   , isLiveSubagent
   25   , renderLiveSubagents
   26   , shutdownExitCode
   27   ) where
   28 
   29 import Control.Exception (AsyncException (..), throwIO)
   30 import Data.IORef (IORef, readIORef)
   31 import Data.Text (Text)

   33 import System.Exit (ExitCode (..), exitWith)
   34 
   35 import Katip (Severity (..), ls)

   43 -- | What a SIGINT-killed server exits with: 128 + 2 (SIGINT). Scripts and
   44 -- @make serve@ then see an interrupted server rather than a clean exit 0.
   45 shutdownExitCode :: ExitCode
   46 shutdownExitCode = ExitFailure 130
   47 
   48 -- | Handle an async exception raised on the @seal serve@ main thread.
   49 --
   50 -- 'UserInterrupt' (Ctrl-C) logs the shutdown marker + the live sub-agent
   51 -- inventory and exits with 'shutdownExitCode'. The inventory comes from the
   52 -- @IORef@ the serve wiring fills in once 'Seal.Core.Backends.newBackends' has
   53 -- run; 'Nothing' means the interrupt landed during startup, before there was
   54 -- a runtime to report on.
   55 --
   56 -- Any other async exception is re-raised as-is.
   57 handleServeInterrupt :: SealLogger -> IORef (Maybe AgentRuntime) -> AsyncException -> IO ()
   58 handleServeInterrupt logger runtimeRef = \case
   59   UserInterrupt -> do
   60     logShutdown logger runtimeRef
   61     exitWith shutdownExitCode
   62   other -> throwIO other
   63 
   64 -- | Emit the shutdown marker followed by one line per live sub-agent.
   65 logShutdown :: SealLogger -> IORef (Maybe AgentRuntime) -> IO ()
   66 logShutdown logger runtimeRef = do
   67   mRuntime <- readIORef runtimeRef
   68   case mRuntime of
   69     Nothing ->
   70       logIO logger InfoS
   71         "SIGINT received (Ctrl-C); shutting down during startup\
   72         \ (the sub-agent runtime was not created yet)"
   73     Just runtime -> do
   74       instances <- listAgents runtime
   75       let live = filter isLiveSubagent instances
   76       logIO logger InfoS (ls (headline (length live)))
   77       mapM_ (logIO logger InfoS . ls) (renderLiveSubagents live)
   78   where
   79     -- Plain ASCII on purpose: 'Seal.Logging.Logger.logIO' renders messages
   80     -- through 'show', so a non-ASCII glyph shows up escaped (@\\8212@ for an
   81     -- em dash) in seal.log.
   82     headline :: Int -> Text
   83     headline 0 = "SIGINT received (Ctrl-C); shutting down: no live sub-agents"
   84     headline 1 =
   85       "SIGINT received (Ctrl-C); shutting down: 1 live sub-agent\
   86       \ will be killed with this process"
   87     headline n =
   88       "SIGINT received (Ctrl-C); shutting down: " <> T.pack (show n)
   89         <> " live sub-agents will be killed with this process"
   90 
   91 -- | A sub-agent whose worker thread is still alive in this process, and so
   92 -- dies with it. 'Starting' and 'Running' are the statuses that carry a live
   93 -- forked worker; the rest ('Stopped', 'Crashed', 'Interrupted') linger in the
   94 -- registry as history.
   95 isLiveSubagent :: AgentInstance -> Bool
   96 isLiveSubagent inst = case aiStatus inst of
   97   Starting -> True
   98   Running  -> True
   99   _        -> False
  100 
  101 -- | Render one line per live sub-agent: id, parent session, child session,
  102 -- depth, status. Non-live entries are dropped. The id already embeds the
  103 -- agent def id (see 'Seal.Agent.Runtime.Delegation.mkSubagentId'), so the def
  104 -- is not repeated here.
  105 renderLiveSubagents :: [AgentInstance] -> [Text]
  106 renderLiveSubagents = map render . filter isLiveSubagent
  107   where
  108     render inst =
  109       "  sub-agent " <> subagentIdText (aiSubagentId inst)
  110         <> " parent=" <> sessionIdText (aiParentSession inst)
  111         <> " child=" <> sessionIdText (aiSession inst)
  112         <> " depth=" <> T.pack (show (aiDepth inst))
  113         <> " status=" <> statusText (aiStatus inst)
  114     statusText = \case
  115       Starting    -> "starting"
  116       Running     -> "running"
  117       Idle        -> "idle"
  118       Stopped     -> "stopped"
  119       Interrupted -> "interrupted"
  120       Crashed err -> "crashed: " <> err

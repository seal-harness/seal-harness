    2 -- | Static behavioral guidance blocks injected into the system prompt.
    3 -- Each block is a few sentences of operational guidance that improves
    4 -- model behavior across providers without being model-specific. The
    5 -- blocks are config-gated (see 'Seal.Config.File': @[agent]@
    6 -- @parallel_tool_guidance@, @tool_use_enforcement@,
    7 -- @task_completion_guidance@); each defaults to injected (true).
    8 --
    9 -- The blocks are appended /before/ the dynamic @\<available_skills\>@
   10 -- catalog so the stable text precedes the volatile catalog (cache-
   11 -- friendly ordering: stable identity prefix → static guidance →
   12 -- autoload body → available-skills catalog).

   32 -- | Parallel tool-call guidance. Tells the model to batch independent
   33 -- tool calls into one turn rather than running them sequentially across
   34 -- turns. A real latency/token win with no risk.
   35 parallelToolGuidance :: Text
   36 parallelToolGuidance =
   37   "## Parallel tool calls\n\n\
   38   \When several tool calls are independent (no output of one is needed to \
   39   \call another), batch them into a single turn. Do not call them \
   40   \sequentially across turns when you could have called them together."
   41 
   42 -- | Tool-use enforcement. Tells the model to actually call tools instead
   43 -- of describing actions. Improves reliability on weaker models that tend
   44 -- to narrate instead of act.
   45 toolUseEnforcement :: Text
   46 toolUseEnforcement =
   47   "## Tool use\n\n\
   48   \When a task requires a tool, actually call the tool. Do not describe \
   49   \or narrate the action you would take (\"I would run X\") and then stop \
   50   \— call it. Do not answer from memory when a current tool call could \
   51   \get the real state."
   52 
   53 -- | Task-completion / anti-fabrication guidance. Tells the model not to
   54 -- stop after a stub and not to fabricate output when a path is blocked.
   55 taskCompletionGuidance :: Text
   56 taskCompletionGuidance =
   57   "## Task completion\n\n\
   58   \Do not stop after producing a stub, TODO, or skeleton and report the \
   59   \task done. Complete the implementation through to working, tested \
   60   \code. If a path is genuinely blocked (missing input, failed tool, \
   61   \permission denied), say so explicitly and ask for what you need — do \
   62   \not fabricate output to fill the gap."
   63 
   64 -- | Credential-bearing-tool guidance. Tells the model to use @BIN_EXEC@
   65 -- (not @SHELL_EXEC@) for @git@ and @gh@, because Seal Harness injects
   66 -- credentials from the vault only through the @BIN_EXEC@ path.
   67 credentialToolGuidance :: Text
   68 credentialToolGuidance =
   69   "## Credential-bearing tools\n\n\
   70  \Use `BIN_EXEC` (not `SHELL_EXEC`) for `git` and `gh` — Seal Harness \
   71  \injects credentials from the vault only through `BIN_EXEC`. Running \
   72  \`gh` via `SHELL_EXEC` gets no credential injection and will fail with \
   73  \an opaque authentication error on push, PR creation, or any operation \
   74  \that requires auth."
   75 
   76 -- | Render the enabled guidance blocks as a single section, joined with
   77 -- blank lines. Returns the empty 'Text' when no block is enabled (so the
   78 -- caller can skip emitting anything). Each enabled block is a
   79 -- @## header@ + body, so the whole section reads as a sequence of
   80 -- short guidance notes.
   81 staticGuidanceBlock :: Bool -> Bool -> Bool -> Bool -> Text
   82 staticGuidanceBlock parallel toolUse taskCompletion credentialTool =
   83   let parts = [ parallelToolGuidance    | parallel ]
   84            <> [ toolUseEnforcement      | toolUse ]
   85            <> [ taskCompletionGuidance  | taskCompletion ]
   86            <> [ credentialToolGuidance  | credentialTool ]
   87   in if null parts then "" else T.intercalate "\n\n" parts
   88 
   89 -- | Append the enabled static guidance blocks to the resolved system
   90 -- prompt. The blocks go /before/ the auto-loaded skill body and the
   91 -- available-skills catalog (the caller runs this first, then
   92 -- 'Seal.Skills.Autoload.injectAutoloadSkill', then
   93 -- 'Seal.Skills.Prompt.injectAvailableSkills'). Returns the prompt
   94 -- unchanged when no block is enabled.
   95 injectStaticGuidance
   96   :: Bool    -- ^ parallel tool-call guidance
   97   -> Bool    -- ^ tool-use enforcement
   98   -> Bool    -- ^ task-completion guidance
   99   -> Bool    -- ^ credential-bearing-tool guidance
  100   -> Maybe Text
  101   -> Maybe Text
  102 injectStaticGuidance parallel toolUse taskCompletion credentialTool mPrompt =
  103   let block = staticGuidanceBlock parallel toolUse taskCompletion credentialTool
  104   in if T.null block
  105        then mPrompt
  106        else Just (case mPrompt of
  107                     Nothing  -> block
  108                     Just base -> base <> "\n\n" <> block)

  113 -- | The per-def catalog budget (chars). Matches the skills catalog
  114 -- (4096) — with repo-prefixed ids and 256-char descriptions, ~19
  115 -- metaswarm agents fit comfortably; truncation is observable via the
  116 -- marker, and AGENT_DEF_LIST always stays complete.
  117 availableAgentsBudget :: Int
  118 availableAgentsBudget = 4096
  119 
  120 -- | The one-line note injected into a LEAF child's prompt instead of the
  121 -- catalog (the child's registry has no spawn capability, so advertising
  122 -- agents it cannot delegate to would waste turns).
  123 leafAgentNote :: Text
  124 leafAgentNote = "You are a leaf agent; delegation is not available."
  125 
  126 -- | The untruncated @\<available_agents\>@ catalog block: one bullet per
  127 -- def (@- \<full-id\> [\<role\>]: \<description|name-fallback\>@), grouped
  128 -- by 'adGroup' with @## \<group\>@ headers (ungrouped defs fall under
  129 -- @## Agents@), ending with the AGENT_START nudge line. The block is
  130 -- truncated to 'availableAgentsBudget' with the elided-count marker.
  131 -- The empty list renders @\"\"@ (the caller skips injection — no empty
  132 -- tags are ever emitted). Bullets always use the FULL merged-backend id
  133 -- (workdir-prefixed where applicable) — that is what AGENT_START accepts.
  134 availableAgentsBlock :: [AgentDef] -> Text
  135 availableAgentsBlock [] = ""
  136 availableAgentsBlock defs = truncateBlock budget (fullBlock defs)
  137   where
  138     budget = availableAgentsBudget
  139     fullBlock ds =
  140       "<available_agents>\n"
  141       <> T.intercalate "\n\n" (map renderGroup grouped)
  142       <> "\n\nDelegate with AGENT_START using an id before relying on an agent."
  143       <> "\n</available_agents>"
  144       where
  145         sorted = sortOn (\d -> (groupKey d, agentDefIdText (adId d))) ds
  146         grouped = groupBy (\a b -> groupKey a == groupKey b) sorted
  147         groupKey d = fromMaybe "" (adGroup d)
  148 
  149 -- | Render one group's section: a header line (the group name, or
  150 -- \"Agents\" for the ungrouped section) followed by the @- id [role]: text@
  151 -- bullets.
  152 renderGroup :: [AgentDef] -> Text
  153 renderGroup [] = ""
  154 renderGroup group@(d0:_) =
  155   header <> "\n" <> T.intercalate "\n" (map bullet group)
  156   where
  157     header = case adGroup d0 of
  158       Just g  -> "## " <> g
  159       Nothing -> "## Agents"
  160     bullet d = "- " <> agentDefIdText (adId d) <> roleSuffix (adRole d)
  161                <> ": " <> primaryText d
  162     primaryText d = case adDescription d of
  163       Just desc | not (T.null (T.strip desc)) -> desc
  164       _         -> adName d
  165     roleSuffix (Just r) = " [" <> r <> "]"
  166     roleSuffix Nothing  = ""
  167 
  168 -- | Truncate the block to the budget, appending the elided count when
  169 -- truncation occurs (character boundary — 'Text' is Unicode-correct).
  170 truncateBlock :: Int -> Text -> Text
  171 truncateBlock budget block
  172   | T.length block <= budget = block
  173   | otherwise =
  174       T.take budget block
  175         <> "\n[...catalog truncated at "
  176         <> T.pack (show budget)
  177         <> " chars; "
  178         <> T.pack (show (T.length block - budget))
  179         <> " more chars elided...]"
  180 
  181 -- | Append the @\<available_agents\>@ catalog to the resolved system
  182 -- prompt. Pure over the def list (the caller passes 'adbList' from the
  183 -- per-turn union backend). Returns the prompt unchanged (no catalog)
  184 -- when the list is empty, so no empty tags are ever emitted. The catalog
  185 -- is appended AFTER everything else (skills catalog last per the
  186 -- cache-friendly ordering; agents follow).
  187 injectAvailableAgents :: [AgentDef] -> Maybe Text -> Maybe Text
  188 injectAvailableAgents defs mPrompt =
  189   let block = availableAgentsBlock defs
  190   in if T.null block
  191        then mPrompt
  192        else pure (case mPrompt of
  193                     Nothing  -> block
  194                     Just base -> base <> "\n\n" <> block)

   29 -- | The assembled WAI application: the REST API routes, then static file
   30 -- serving with SPA fallback (if the static dir is configured). Each
   31 -- request is wrapped with 'requestTimingMiddleware' so the method, path,
   32 -- status, and duration are logged via Katip for observability.
   33 gatewayApp :: SealLogger -> ApiDeps -> Maybe FilePath -> Application
   34 gatewayApp logger deps mStaticDir =
   35   requestTimingMiddleware logger $ \req respond ->
   36     case pathInfo req of
   37       ("api" : _) -> apiApp deps req respond
   38       _ -> case mStaticDir of
   39         Nothing -> respond (responseLBS status404 [("Content-Type", "application/json")] "{\"error\":\"not found\"}")
   40         Just staticDir -> serveStatic staticDir req respond
   41 
   42 -- | WAI middleware that logs each request's HTTP method, path, response
   43 -- status, and total duration (in milliseconds) via the Katip 'SealLogger'.
   44 -- Captures the start time before the inner app runs, and logs the elapsed
   45 -- time in the @respond@ callback (after the inner app has produced its
   46 -- response but before it is sent to the client). The log line format is:
   47 --
   48 -- @HTTP GET /api/lists -> 200 (15ms)@
   49 --
   50 -- This covers ALL endpoints (API + static) because it wraps the entire
   51 -- 'gatewayApp', giving observability into every HTTP request.
   52 requestTimingMiddleware :: SealLogger -> Application -> Application
   53 requestTimingMiddleware logger = requestTimingMiddlewareWith (logIO logger InfoS . ls)
   54 
   55 -- | Like 'requestTimingMiddleware' but with an injectable logger (for
   56 -- tests). The logger receives a single 'String' per request.
   57 requestTimingMiddlewareWith :: (String -> IO ()) -> Application -> Application
   58 requestTimingMiddlewareWith logger innerApp req respond = do
   59   startTime <- getCurrentTime
   60   innerApp req $ \resp -> do
   61     endTime <- getCurrentTime
   62     let durationMs = floor (diffUTCTime endTime startTime * 1000) :: Int
   63         method = BC.unpack (requestMethod req)
   64         path = "/" <> T.unpack (T.intercalate "/" (pathInfo req))
   65         status = HT.statusCode (responseStatus resp)
   66     logger ("HTTP " <> method <> " " <> path <> " -> " <> show status <> " (" <> show durationMs <> "ms)")
   67     respond resp
   68 
   69 -- | Serve a static file with SPA fallback (serve @index.html@ if the path
   70 -- doesn't match a file).
   71 serveStatic :: FilePath -> Application
   72 serveStatic staticDir req respond = do
   73   let pathParts = map T.unpack (pathInfo req)
   74       filePath = staticDir </> case pathParts of
   75         [] -> "index.html"
   76         _ -> case xs of
   77           [p] -> p
   78           _ -> T.unpack (T.intercalate "/" (map T.pack xs))
   79       xs = pathParts
   80   exists <- doesFileExist filePath
   81   if exists
   82     then respond (responseFile status200 [("Content-Type", contentTypeFor (takeExtension filePath))] filePath Nothing)
   83     else respond (responseFile status200 [("Content-Type", "text/html")] (staticDir </> "index.html") Nothing)
   84 
   85 -- | The content-type for a file extension.
   86 contentTypeFor :: String -> BC.ByteString
   87 contentTypeFor ext = case ext of
   88   ".html" -> "text/html"
   89   ".js"   -> "application/javascript"
   90   ".css"  -> "text/css"
   91   ".json" -> "application/json"
   92   ".svg"  -> "image/svg+xml"
   93   ".png"  -> "image/png"
   94   ".ico"  -> "image/x-icon"
   95   _       -> "application/octet-stream"

   97 -- | Run the gateway. Logs a startup banner with the bind address via the
   98 -- Katip 'SealLogger'. When the host is non-loopback:
   99 --   - If @failClosedOnNonLoopback@ is True (mode=remote), REFUSES to start
  100 --     (design V6: a non-loopback gateway is a config-tamper surface — the
  101 --     unauthenticated @updateRuntimeConfig@ caller is reachable by anything
  102 --     that can reach the address).
  103 --   - Otherwise, warns (the full slash-command surface is reachable).
  104 runGateway :: SealLogger -> GatewayConfig -> Bool -> ApiDeps -> IO ()
  105 runGateway logger cfg failClosedOnNonLoopback deps = do
  106   let host = gcHost cfg
  107       port = gcPort cfg
  108   logIO logger InfoS "Seal Harness gateway"
  109   logIO logger InfoS ("  URL:      http://" <> ls host <> ":" <> ls (show port))
  110   if host /= "127.0.0.1" && host /= "::1" && host /= "localhost"
  111     then
  112       if failClosedOnNonLoopback
  113         then do
  114           logIO logger ErrorS ("refusing to start gateway on non-loopback address " <> ls host
  115                                <> " in remote-only mode — bind 127.0.0.1 or disable remote-only")
  116           pure ()  -- do NOT call run; exit without serving
  117         else do
  118           logIO logger WarningS ("binding to " <> ls host
  119                                  <> " — the full slash-command surface is reachable by anything that can reach this address")
  120           let mStaticDir = fmap T.unpack (gcStaticDir cfg)
  121               app = gatewayApp logger deps mStaticDir
  122           run port app
  123     else do
  124       let mStaticDir = fmap T.unpack (gcStaticDir cfg)
  125           app = gatewayApp logger deps mStaticDir
  126       run port app

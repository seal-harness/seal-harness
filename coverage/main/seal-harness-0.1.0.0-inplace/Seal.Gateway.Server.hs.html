   46 -- | The assembled WAI application: the REST API routes, then static file
   47 -- serving. When a @static_dir@ is configured, serves from disk (dev path);
   48 -- otherwise falls back to the compile-time-embedded asset map (release
   49 -- default). Each request is wrapped with 'requestTimingMiddleware' so the
   50 -- method, path, status, and duration are logged via Katip for observability.
   51 gatewayApp :: SealLogger -> ApiDeps -> Maybe FilePath -> Application
   52 gatewayApp logger deps mStaticDir =
   53   requestTimingMiddleware logger $ \req respond ->
   54     case pathInfo req of
   55       ("api" : _) -> apiApp deps req respond
   56       _ -> case mStaticDir of
   57         Just staticDir -> serveStatic staticDir req respond
   58         Nothing        -> serveEmbedded req respond
   59 
   60 -- | The assembled application WITH the WebSocket stream endpoint mounted
   61 -- on the same port.
   62 --
   63 -- The WS upgrade is handled by 'streamMiddleware'; every non-WebSocket
   64 -- request falls through to the ordinary gateway app. This is what makes
   65 -- the gateway single-port: one listener, one bind, no second port to
   66 -- collide with.
   67 gatewayAppWithStream
   68   :: SealLogger
   69   -> ApiDeps
   70   -> Maybe FilePath
   71   -> StreamGuard
   72   -> StreamBroker
   73   -> Application
   74 gatewayAppWithStream logger deps mStaticDir guard broker =
   75   streamMiddleware guard broker (gatewayApp logger deps mStaticDir)

   77 -- | WAI middleware that logs each request's HTTP method, path, response
   78 -- status, and total duration (in milliseconds) via the Katip 'SealLogger'.
   79 requestTimingMiddleware :: SealLogger -> Application -> Application
   80 requestTimingMiddleware logger = requestTimingMiddlewareWith (logIO logger InfoS . ls)
   81 
   82 -- | Like 'requestTimingMiddleware' but with an injectable logger (for
   83 -- tests).
   84 requestTimingMiddlewareWith :: (String -> IO ()) -> Application -> Application
   85 requestTimingMiddlewareWith logger innerApp req respond = do
   86   startTime <- getCurrentTime
   87   innerApp req $ \resp -> do
   88     endTime <- getCurrentTime
   89     let durationMs = floor (diffUTCTime endTime startTime * 1000) :: Int
   90         method = BC.unpack (requestMethod req)
   91         path = "/" <> T.unpack (T.intercalate "/" (pathInfo req))
   92         status = HT.statusCode (responseStatus resp)
   93     logger ("HTTP " <> method <> " " <> path <> " -> " <> show status <> " (" <> show durationMs <> "ms)")
   94     respond resp

   96 -- | Serve from the compile-time-embedded asset map. Traversal is
   97 -- structurally impossible (pure Map lookup, no filesystem involved).
   98 -- SPA routes fall back to the embedded @index.html@; misses on
   99 -- asset-like extensions 404 (so a bad asset URL never returns HTML).
  100 -- When the map is empty (fresh checkout / pure nix build), returns a
  101 -- helpful HTML page explaining the situation.
  102 serveEmbedded :: Application
  103 serveEmbedded req respond =
  104   let segs = pathInfo req
  105       key  = T.intercalate "/" segs
  106   in case Map.lookup (T.unpack key) embeddedAssets of
  107        Just bytes -> respond (embeddedResponse (T.unpack key) bytes)
  108        Nothing
  109          | isAssetPath segs -> respond notFoundResponse
  110          | Just idx <- Map.lookup "index.html" embeddedAssets ->
  111              respond (indexResponse idx)
  112          | otherwise -> respond emptyEmbedResponse
  113 
  114 -- | Serve a static file from disk with SPA fallback. Routes through
  115 -- 'mkSafePath' for traversal containment (lexical collapse +
  116 -- canonicalizePath + symlink containment + blocked-name filtering).
  117 -- The root path (@pathInfo = []@) is special-cased to serve @index.html@
  118 -- directly, bypassing @mkSafePath@ (which would return the directory
  119 -- itself). Asset-extension misses 404 instead of returning HTML.
  120 serveStatic :: FilePath -> Application
  121 serveStatic staticDir req respond = do
  122   let segs = pathInfo req
  123   case segs of
  124     [] -> respond (responseFile status200
  125                      (contentTypeHeader ".html" <> cacheHeadersFor "index.html" <> nosniffHeader)
  126                      (staticDir </> "index.html") Nothing)
  127     _  -> do
  128       let key = T.unpack (T.intercalate "/" segs)
  129       eSafe <- mkSafePath (WorkspaceRoot staticDir) key
  130       case eSafe of
  131         Left (PathEscapesWorkspace _) -> respond notFoundResponse
  132         Left (PathIsBlocked _)        -> respond notFoundResponse
  133         Left (PathInsecureMode _)     -> respond notFoundResponse
  134         Left (PathDoesNotExist _)
  135           | isAssetPath segs -> respond notFoundResponse
  136           | otherwise        -> respond (responseFile status200
  137                                           (contentTypeHeader ".html" <> cacheHeadersFor "index.html" <> nosniffHeader)
  138                                           (staticDir </> "index.html") Nothing)
  139         Right safePath -> do
  140           let fp = getSafePath safePath
  141           exists <- doesFileExist fp
  142           if exists
  143             then respond (responseFile status200
  144                            (contentTypeHeader (takeExtension fp) <> cacheHeadersFor key <> nosniffHeader)
  145                            fp Nothing)
  146             else respond notFoundResponse

  148 -- | Build a 200 response for an embedded asset with MIME + cache + nosniff.
  149 embeddedResponse :: FilePath -> BC.ByteString -> Network.Wai.Response
  150 embeddedResponse key bytes =
  151   responseLBS status200
  152     (contentTypeHeader (takeExtension key) <> cacheHeadersFor key <> nosniffHeader)
  153     (LBS.fromStrict bytes)
  154 
  155 -- | Build a 200 response for SPA fallback (index.html) with no-cache.
  156 indexResponse :: BC.ByteString -> Network.Wai.Response
  157 indexResponse bytes =
  158   responseLBS status200
  159     [("Content-Type", "text/html; charset=utf-8")
  160     ,("Cache-Control", "no-cache")
  161     ,("X-Content-Type-Options", "nosniff")]
  162     (LBS.fromStrict bytes)
  163 
  164 -- | 404 for missing asset-like paths (prevents content-type confusion).
  165 notFoundResponse :: Network.Wai.Response
  166 notFoundResponse =
  167   responseLBS status404
  168     [("Content-Type", "text/plain")
  169     ,("Cache-Control", "no-cache")
  170     ,("X-Content-Type-Options", "nosniff")]
  171     "not found"
  172 
  173 -- | Helpful HTML page when no assets are embedded (fresh checkout / nix).
  174 emptyEmbedResponse :: Network.Wai.Response
  175 emptyEmbedResponse =
  176   responseLBS status404
  177     [("Content-Type", "text/html; charset=utf-8")
  178     ,("Cache-Control", "no-cache")
  179     ,("X-Content-Type-Options", "nosniff")]
  180     "<html><body><h1>No UI embedded in this build</h1>\
  181     \<p>Run <code>make build</code> with the frontend built, or set\
  182     \[gateway] static_dir in config.toml.</p></body></html>"

  184 -- | Content-Type header from a file extension (case-insensitive).
  185 contentTypeHeader :: String -> ResponseHeaders
  186 contentTypeHeader ext =
  187   [("Content-Type", contentTypeFor (map toLower ext))]
  188 
  189 -- | Cache-Control headers per §4.4 policy: @assets\/@ prefix → immutable,
  190 -- everything else (index.html, SPA fallback) → no-cache.
  191 cacheHeadersFor :: FilePath -> ResponseHeaders
  192 cacheHeadersFor key
  193   | "assets/" `isPrefixOf` key = [("Cache-Control", "public, max-age=31536000, immutable")]
  194   | otherwise                  = [("Cache-Control", "no-cache")]
  195 
  196 -- | @X-Content-Type-Options: nosniff@ — prevents MIME-sniffing. Present
  197 -- on ALL static responses in M1.
  198 nosniffHeader :: ResponseHeaders
  199 nosniffHeader = [("X-Content-Type-Options", "nosniff")]
  200 
  201 -- | Whether the path looks like a static asset (has a known extension in
  202 -- the final segment). EXCLUDES @.html@ so extensionless SPA routes
  203 -- always fall back to @index.html@. Case-insensitive via 'T.toLower'.
  204 isAssetPath :: [T.Text] -> Bool
  205 isAssetPath segs = case segs of
  206   [] -> False
  207   _  -> let ext = T.toLower (T.pack (takeExtension (T.unpack (last segs))))
  208         in ext `elem` assetExtensions
  209 
  210 -- | Pinned asset extension set (20 extensions, excludes @.html@).
  211 assetExtensions :: [T.Text]
  212 assetExtensions =
  213   [ ".js", ".mjs", ".css", ".map", ".json", ".txt", ".webmanifest"
  214   , ".png", ".jpg", ".jpeg", ".gif", ".webp", ".avif", ".svg", ".ico"
  215   , ".woff", ".woff2", ".ttf", ".otf", ".wasm"
  216   ]
  217 
  218 -- | The content-type for a file extension (case-insensitive — caller
  219 -- should lowercase the input via 'contentTypeHeader' or pre-normalize).
  220 contentTypeFor :: String -> BC.ByteString
  221 contentTypeFor ext = case ext of
  222   ".html"        -> "text/html; charset=utf-8"
  223   ".js"          -> "application/javascript"
  224   ".mjs"         -> "application/javascript"
  225   ".css"         -> "text/css"
  226   ".json"        -> "application/json"
  227   ".txt"         -> "text/plain"
  228   ".webmanifest" -> "application/manifest+json"
  229   ".svg"         -> "image/svg+xml"
  230   ".png"         -> "image/png"
  231   ".jpg"         -> "image/jpeg"
  232   ".jpeg"        -> "image/jpeg"
  233   ".gif"         -> "image/gif"
  234   ".webp"        -> "image/webp"
  235   ".avif"        -> "image/avif"
  236   ".ico"         -> "image/x-icon"
  237   ".woff"        -> "font/woff"
  238   ".woff2"       -> "font/woff2"
  239   ".ttf"         -> "font/ttf"
  240   ".otf"         -> "font/otf"
  241   ".wasm"        -> "application/wasm"
  242   _              -> "application/octet-stream"

  244 -- | Run the gateway. Logs a startup banner with the bind address via the
  245 -- Katip 'SealLogger'. When the host is non-loopback:
  246 --   - If @failClosedOnNonLoopback@ is True (mode=remote), REFUSES to start.
  247 --   - Otherwise, warns (the full slash-command surface is reachable).
  248 --
  249 -- The WebSocket stream endpoint is served by the SAME listener (mounted
  250 -- via 'gatewayAppWithStream'), so there is exactly one port and one bind.
  251 -- A bind failure now aborts the whole server with a clear message, rather
  252 -- than leaving a half-running gateway whose live transcript was dead.
  253 runGateway
  254   :: SealLogger
  255   -> GatewayConfig
  256   -> Bool
  257   -> ApiDeps
  258   -> StreamGuard
  259   -> StreamBroker
  260   -> IO ()
  261 runGateway logger cfg failClosedOnNonLoopback deps guard broker = do
  262   let host = gcHost cfg
  263       port = gcPort cfg
  264   logIO logger InfoS "Seal Harness gateway"
  265   logIO logger InfoS ("  URL:      http://" <> ls host <> ":" <> ls (show port))
  266   logIO logger InfoS ("  Stream:   ws://" <> ls host <> ":" <> ls (show port) <> "/ (same port)")
  267   if host /= "127.0.0.1" && host /= "::1" && host /= "localhost"
  268     then
  269       if failClosedOnNonLoopback
  270         then do
  271           logIO logger ErrorS ("refusing to start gateway on non-loopback address " <> ls host
  272                                <> " in remote-only mode — bind 127.0.0.1 or disable remote-only")
  273           pure ()
  274         else do
  275           logIO logger WarningS ("binding to " <> ls host
  276                                  <> " — the full slash-command surface is reachable by anything that can reach this address")
  277           let mStaticDir = fmap T.unpack (gcStaticDir cfg)
  278               app = gatewayAppWithStream logger deps mStaticDir guard broker
  279           runSettings (setPort port settings) app
  280     else do
  281       let mStaticDir = fmap T.unpack (gcStaticDir cfg)
  282           app = gatewayAppWithStream logger deps mStaticDir guard broker
  283       runSettings (setPort port settings) app

  285 -- | Warp settings for the gateway.
  286 --
  287 -- Warp's default request timeout is 30s, which would tear down an idle
  288 -- WebSocket connection (the stream is idle between transcript events).
  289 -- The WS server sends a ping every 30s (see 'withPingThread'), so the
  290 -- timeout is raised well above that: a connection silent for two minutes
  291 -- is genuinely dead. HTTP requests are unaffected in practice (the
  292 -- gateway answers in milliseconds).
  293 settings :: Settings
  294 settings = setTimeout 120_000 defaultSettings

   54 -- | The assembled WAI application: the REST API routes, then static file
   55 -- serving. When a @static_dir@ is configured, serves from disk (dev path);
   56 -- otherwise falls back to the compile-time-embedded asset map (release
   57 -- default). Each request is wrapped with 'requestTimingMiddleware' so the
   58 -- method, path, status, and duration are logged via Katip for observability.
   59 --
   60 -- The optional 'AccessControlHandle' gates the @\/api\/*@ branch only: the
   61 -- static bundle must always load, because the setup screen is part of it
   62 -- (design §7 "Where the middleware sits").
   63 gatewayApp :: SealLogger -> Maybe AccessControlHandle -> ApiDeps -> Maybe FilePath -> Application
   64 gatewayApp logger mAccess deps mStaticDir =
   65   requestTimingMiddleware logger $ \req respond ->
   66     case pathInfo req of
   67       ("api" : _) -> gatedApi req respond
   68       _ -> case mStaticDir of
   69         Just staticDir -> serveStatic staticDir req respond
   70         Nothing        -> serveEmbedded req respond
   71   where
   72     gatedApi = case mAccess of
   73       Nothing     -> apiApp deps
   74       Just access -> accessControlMiddleware access (apiApp deps)
   75 
   76 -- | The assembled application WITH the WebSocket stream endpoint mounted
   77 -- on the same port.
   78 --
   79 -- The WS upgrade is handled by 'streamMiddleware'; every non-WebSocket
   80 -- request falls through to the ordinary gateway app. This is what makes
   81 -- the gateway single-port: one listener, one bind, no second port to
   82 -- collide with.
   83 gatewayAppWithStream
   84   :: SealLogger
   85   -> Maybe AccessControlHandle
   86   -> ApiDeps
   87   -> Maybe FilePath
   88   -> StreamGuard
   89   -> StreamBroker
   90   -> Application
   91 gatewayAppWithStream logger mAccess deps mStaticDir guard broker =
   92   streamMiddleware guard broker (gatewayApp logger mAccess deps mStaticDir)

   94 -- | WAI middleware that logs each request's HTTP method, path, response
   95 -- status, and total duration (in milliseconds) via the Katip 'SealLogger'.
   96 requestTimingMiddleware :: SealLogger -> Application -> Application
   97 requestTimingMiddleware logger = requestTimingMiddlewareWith (logIO logger InfoS . ls)
   98 
   99 -- | Like 'requestTimingMiddleware' but with an injectable logger (for
  100 -- tests).
  101 requestTimingMiddlewareWith :: (String -> IO ()) -> Application -> Application
  102 requestTimingMiddlewareWith logger innerApp req respond = do
  103   startTime <- getCurrentTime
  104   innerApp req $ \resp -> do
  105     endTime <- getCurrentTime
  106     let durationMs = floor (diffUTCTime endTime startTime * 1000) :: Int
  107         method = BC.unpack (requestMethod req)
  108         path = "/" <> T.unpack (T.intercalate "/" (pathInfo req))
  109         status = HT.statusCode (responseStatus resp)
  110     logger ("HTTP " <> method <> " " <> path <> " -> " <> show status <> " (" <> show durationMs <> "ms)")
  111     respond resp

  113 -- | Serve from the compile-time-embedded asset map. Traversal is
  114 -- structurally impossible (pure Map lookup, no filesystem involved).
  115 -- SPA routes fall back to the embedded @index.html@; misses on
  116 -- asset-like extensions 404 (so a bad asset URL never returns HTML).
  117 -- When the map is empty (fresh checkout / pure nix build), returns a
  118 -- helpful HTML page explaining the situation.
  119 serveEmbedded :: Application
  120 serveEmbedded req respond =
  121   let segs = pathInfo req
  122       key  = T.intercalate "/" segs
  123   in case Map.lookup (T.unpack key) embeddedAssets of
  124        Just bytes -> respond (embeddedResponse (T.unpack key) bytes)
  125        Nothing
  126          | isAssetPath segs -> respond notFoundResponse
  127          | Just idx <- Map.lookup "index.html" embeddedAssets ->
  128              respond (indexResponse idx)
  129          | otherwise -> respond emptyEmbedResponse
  130 
  131 -- | Serve a static file from disk with SPA fallback. Routes through
  132 -- 'mkSafePath' for traversal containment (lexical collapse +
  133 -- canonicalizePath + symlink containment + blocked-name filtering).
  134 -- The root path (@pathInfo = []@) is special-cased to serve @index.html@
  135 -- directly, bypassing @mkSafePath@ (which would return the directory
  136 -- itself). Asset-extension misses 404 instead of returning HTML.
  137 serveStatic :: FilePath -> Application
  138 serveStatic staticDir req respond = do
  139   let segs = pathInfo req
  140   case segs of
  141     [] -> respond (responseFile status200
  142                      (contentTypeHeader ".html" <> cacheHeadersFor "index.html" <> nosniffHeader)
  143                      (staticDir </> "index.html") Nothing)
  144     _  -> do
  145       let key = T.unpack (T.intercalate "/" segs)
  146       eSafe <- mkSafePath (WorkspaceRoot staticDir) key
  147       case eSafe of
  148         Left (PathEscapesWorkspace _) -> respond notFoundResponse
  149         Left (PathIsBlocked _)        -> respond notFoundResponse
  150         Left (PathInsecureMode _)     -> respond notFoundResponse
  151         Left (PathDoesNotExist _)
  152           | isAssetPath segs -> respond notFoundResponse
  153           | otherwise        -> respond (responseFile status200
  154                                           (contentTypeHeader ".html" <> cacheHeadersFor "index.html" <> nosniffHeader)
  155                                           (staticDir </> "index.html") Nothing)
  156         Right safePath -> do
  157           let fp = getSafePath safePath
  158           exists <- doesFileExist fp
  159           if exists
  160             then respond (responseFile status200
  161                            (contentTypeHeader (takeExtension fp) <> cacheHeadersFor key <> nosniffHeader)
  162                            fp Nothing)
  163             else respond notFoundResponse

  165 -- | Build a 200 response for an embedded asset with MIME + cache + nosniff.
  166 embeddedResponse :: FilePath -> BC.ByteString -> Network.Wai.Response
  167 embeddedResponse key bytes =
  168   responseLBS status200
  169     (contentTypeHeader (takeExtension key) <> cacheHeadersFor key <> nosniffHeader)
  170     (LBS.fromStrict bytes)
  171 
  172 -- | Build a 200 response for SPA fallback (index.html) with no-cache.
  173 indexResponse :: BC.ByteString -> Network.Wai.Response
  174 indexResponse bytes =
  175   responseLBS status200
  176     [("Content-Type", "text/html; charset=utf-8")
  177     ,("Cache-Control", "no-cache")
  178     ,("X-Content-Type-Options", "nosniff")]
  179     (LBS.fromStrict bytes)
  180 
  181 -- | 404 for missing asset-like paths (prevents content-type confusion).
  182 notFoundResponse :: Network.Wai.Response
  183 notFoundResponse =
  184   responseLBS status404
  185     [("Content-Type", "text/plain")
  186     ,("Cache-Control", "no-cache")
  187     ,("X-Content-Type-Options", "nosniff")]
  188     "not found"
  189 
  190 -- | Helpful HTML page when no assets are embedded (fresh checkout / nix).
  191 emptyEmbedResponse :: Network.Wai.Response
  192 emptyEmbedResponse =
  193   responseLBS status404
  194     [("Content-Type", "text/html; charset=utf-8")
  195     ,("Cache-Control", "no-cache")
  196     ,("X-Content-Type-Options", "nosniff")]
  197     "<html><body><h1>No UI embedded in this build</h1>\
  198     \<p>Run <code>make build</code> with the frontend built, or set\
  199     \[gateway] static_dir in ~/.seal/security.toml.</p></body></html>"

  201 -- | Content-Type header from a file extension (case-insensitive).
  202 contentTypeHeader :: String -> ResponseHeaders
  203 contentTypeHeader ext =
  204   [("Content-Type", contentTypeFor (map toLower ext))]
  205 
  206 -- | Cache-Control headers per §4.4 policy: @assets\/@ prefix → immutable,
  207 -- everything else (index.html, SPA fallback) → no-cache.
  208 cacheHeadersFor :: FilePath -> ResponseHeaders
  209 cacheHeadersFor key
  210   | "assets/" `isPrefixOf` key = [("Cache-Control", "public, max-age=31536000, immutable")]
  211   | otherwise                  = [("Cache-Control", "no-cache")]
  212 
  213 -- | @X-Content-Type-Options: nosniff@ — prevents MIME-sniffing. Present
  214 -- on ALL static responses in M1.
  215 nosniffHeader :: ResponseHeaders
  216 nosniffHeader = [("X-Content-Type-Options", "nosniff")]

  218 -- | Whether the path looks like a static asset (has a known extension in
  219 -- the final segment). EXCLUDES @.html@ so extensionless SPA routes
  220 -- always fall back to @index.html@. Case-insensitive via 'T.toLower'.
  221 isAssetPath :: [T.Text] -> Bool
  222 isAssetPath segs = case segs of
  223   [] -> False
  224   _  -> let ext = T.toLower (T.pack (takeExtension (T.unpack (last segs))))
  225         in ext `elem` assetExtensions
  226 
  227 -- | Pinned asset extension set (20 extensions, excludes @.html@).
  228 assetExtensions :: [T.Text]
  229 assetExtensions =
  230   [ ".js", ".mjs", ".css", ".map", ".json", ".txt", ".webmanifest"
  231   , ".png", ".jpg", ".jpeg", ".gif", ".webp", ".avif", ".svg", ".ico"
  232   , ".woff", ".woff2", ".ttf", ".otf", ".wasm"
  233   ]
  234 
  235 -- | The content-type for a file extension (case-insensitive — caller
  236 -- should lowercase the input via 'contentTypeHeader' or pre-normalize).
  237 contentTypeFor :: String -> BC.ByteString
  238 contentTypeFor ext = case ext of
  239   ".html"        -> "text/html; charset=utf-8"
  240   ".js"          -> "application/javascript"
  241   ".mjs"         -> "application/javascript"
  242   ".css"         -> "text/css"
  243   ".json"        -> "application/json"
  244   ".txt"         -> "text/plain"
  245   ".webmanifest" -> "application/manifest+json"
  246   ".svg"         -> "image/svg+xml"
  247   ".png"         -> "image/png"
  248   ".jpg"         -> "image/jpeg"
  249   ".jpeg"        -> "image/jpeg"
  250   ".gif"         -> "image/gif"
  251   ".webp"        -> "image/webp"
  252   ".avif"        -> "image/avif"
  253   ".ico"         -> "image/x-icon"
  254   ".woff"        -> "font/woff"
  255   ".woff2"       -> "font/woff2"
  256   ".ttf"         -> "font/ttf"
  257   ".otf"         -> "font/otf"
  258   ".wasm"        -> "application/wasm"
  259   _              -> "application/octet-stream"

  261 -- | Run the gateway. Logs a startup banner with the bind address via the
  262 -- Katip 'SealLogger'.
  263 --
  264 -- A non-loopback host is allowed in **every** configuration, including
  265 -- @untrusted_execution.mode = "remote"@, and warned about. There is
  266 -- deliberately no refusal path.
  267 --
  268 -- It used to refuse in remote mode, as a compensating control for an
  269 -- unauthenticated config write that could flip @remote@ to @local@
  270 -- (design 2026-07-22, V6). That control was retired when the write path's
  271 -- type was narrowed to 'Seal.Config.File.RuntimeConfig', which cannot name
  272 -- the @untrusted_execution@ section at all — the attack became
  273 -- unrepresentable rather than merely refused.
  274 --
  275 -- Keeping the refusal would now be actively harmful. Remote mode is the
  276 -- configuration where the agent cannot reach the harness binary or its
  277 -- environment, so refusing to bind it on a network address pushes operators
  278 -- toward @mode = "local"@, which is strictly weaker. Exposing the port is the
  279 -- operator's decision — behind a VPN, a loopback TLS proxy, or passkey access
  280 -- control it is the normal one — so 'isExposedBind' warns and the gateway
  281 -- serves.
  282 --
  283 -- The WebSocket stream endpoint is served by the SAME listener (mounted
  284 -- via 'gatewayAppWithStream'), so there is exactly one port and one bind.
  285 -- A bind failure now aborts the whole server with a clear message, rather
  286 -- than leaving a half-running gateway whose live transcript was dead.
  287 runGateway
  288   :: SealLogger
  289   -> Maybe AccessControlHandle
  290   -> GatewayConfig
  291   -> ApiDeps
  292   -> StreamGuard
  293   -> StreamBroker
  294   -> IO ()
  295 runGateway logger mAccess cfg deps guard broker = do
  296   let host = gcHost cfg
  297       port = gcPort cfg
  298       mStaticDir = fmap T.unpack (gcStaticDir cfg)
  299       scheme = if tlsEnabled cfg then "https" else "http" :: T.Text
  300       streamScheme = if tlsEnabled cfg then "wss" else "ws" :: T.Text
  301   logIO logger InfoS "Seal Harness gateway"
  302   logIO logger InfoS ("  URL:      " <> ls scheme <> "://" <> ls host <> ":" <> ls (show port))
  303   logIO logger InfoS ("  Stream:   " <> ls streamScheme <> "://" <> ls host <> ":"
  304                        <> ls (show port) <> "/ (same port)")
  305   when (isExposedBind host) $
  306     logIO logger WarningS ("binding to " <> ls host
  307       <> " — the full slash-command surface is reachable by anything that can reach this address")
  308   let app = gatewayAppWithStream logger mAccess deps mStaticDir guard broker
  309       warpSettings = warpSettingsFor cfg
  310   -- TLS in-process when both PEM paths are configured, so passkeys work with
  311   -- no reverse proxy in front. 'tlsConfigError' has already refused the
  312   -- half-configured case at boot.
  313   case (gcTlsCert cfg, gcTlsKey cfg) of
  314     (Just cert, Just key) ->
  315       runTLS (tlsSettings (T.unpack cert) (T.unpack key)) warpSettings app
  316     _ -> runSettings warpSettings app
  317 
  318 -- | The Warp settings for a gateway config: the port to listen on and the
  319 -- host to bind.
  320 --
  321 -- The host has to be passed to Warp explicitly. Warp's default host preference
  322 -- is the IPv4 wildcard, so a gateway configured for @localhost@ would otherwise
  323 -- listen on every interface — the printed URL, the WebAuthn origin and the
  324 -- exposed-bind warning would all describe a loopback listener that does not
  325 -- exist.
  326 warpSettingsFor :: GatewayConfig -> Settings
  327 warpSettingsFor cfg =
  328   setHost (fromString (T.unpack (gcHost cfg))) (setPort (gcPort cfg) settings)
  329 
  330 -- | Does this bind expose the gateway beyond the local machine?
  331 --
  332 -- Only ever chooses a log line. There is intentionally no "refuse" outcome:
  333 -- see the note on 'runGateway'.
  334 isExposedBind :: T.Text -> Bool
  335 isExposedBind host = host `notElem` ["127.0.0.1", "::1", "localhost"]

  337 -- | Warp settings for the gateway.
  338 --
  339 -- Warp's default request timeout is 30s, which would tear down an idle
  340 -- WebSocket connection (the stream is idle between transcript events).
  341 -- The WS server sends keep-alives every 30s — a protocol PING (see
  342 -- 'withPingThread') plus an application-level heartbeat frame — so the
  343 -- timeout is raised well above that: a connection silent for two minutes
  344 -- is genuinely dead. HTTP requests are unaffected in practice (the
  345 -- gateway answers in milliseconds).
  346 settings :: Settings
  347 settings = setTimeout 120_000 defaultSettings

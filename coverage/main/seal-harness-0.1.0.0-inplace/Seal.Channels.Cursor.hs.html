    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The per-conversation tab cursor store. Each conversation (a Telegram
    3 -- chat, a Signal conversation, a TUI session) has a cursor pointing at the
    4 -- tab it's currently focused on. The tab list is shared (one
    5 -- 'TabsHandle' in the gateway); the cursor is per-conversation so
    6 -- @/tab focus N@ on Telegram only affects that Telegram conversation,
    7 -- not other conversations or the TUI.
    8 --
    9 -- The store is a 'TVar' backed 'Map' from 'ConversationKey' to 'TabRef'.
   10 -- 'ConversationKey' is 'ChannelKind' × 'ConversationId' — the
   11 -- server-derived conversation identity (never user-supplied), so a sender
   12 -- cannot forge a cursor key to hijack another conversation's tab.
   13 --
   14 -- == Persistence
   15 --
   16 -- The cursor store survives a @seal serve@ restart via the optional 'csSave'
   17 -- hook (mirroring 'Seal.Tabs.TabsHandle.thSave'). A persisting store
   18 -- ('newPersistingCursorStore') writes the full current map to
   19 -- @\<state\>\/cursors.json@ atomically (0600) after every mutation, and is
   20 -- loaded + seeded at boot so an existing conversation re-resolves to its
   21 -- prior session (carrying the user's @\/model use@ choice) instead of a
   22 -- fresh default session.
   23 module Seal.Channels.Cursor
   24   ( CursorStore
   25   , newPersistingCursorStore
   26   , seedCursorStore
   27   , cursorClearAll
   28   , snapshotCursor
   29   ) where
   30 
   31 import Control.Concurrent.STM (TVar, atomically, newTVarIO, readTVar, readTVarIO, writeTVar)
   32 import Control.Exception (SomeException, catch)
   33 import Data.Map.Strict (Map)
   34 import Data.Map.Strict qualified as Map
   35 import Data.Text (Text)
   36 import Data.Text qualified as T

   38 import Katip (Severity (..), ls)
   39 import Seal.Channels.Cursor.Persist (saveCursorMap)
   40 import Seal.Logging.Global (globalLogIO)

   43 -- | The live cursor store. Backed by a 'TVar' so concurrent reads/writes
   44 -- are race-safe (each STM operation is one transaction). The optional
   45 -- 'csSave' action is invoked after a successful mutation so the map
   46 -- persists to disk; @Just (... saveCursorMap path ...)@ for
   47 -- 'newPersistingCursorStore'. Mirrors 'Seal.Tabs.TabsHandle'.
   48 data CursorStore = CursorStore
   49   { csVar  :: TVar (Map ConversationKey TabRef)
   50   , csSave :: Maybe (IO ())
   51   }
   52 
   53 -- | A conversation identity: 'ChannelKind' × 'ConversationId'.
   54 -- Re-exported here for convenience; the key type is 'ConversationKey'.
   55 type ConversationKey = (Text, Text)
   56   -- ^ (channel-kind-text, conversation-id-text). We use the text forms
   57   -- rather than the structured types so the store doesn't depend on
   58   -- 'ChannelKind' or 'ConversationId' directly (keeping the module
   59   -- lightweight). Callers mint the key from 'MessageSource'.
   60 
   61 -- | Create a 'CursorStore' that persists every mutation to @path@ (atomic
   62 -- write, 0600, MVar-serialized via 'Seal.Util.AtomicJson.saveJsonAtomic').
   63 -- Used by 'Seal.Command.Serve.runServeMain' so the cursor map survives
   64 -- a restart. The save action snapshots the current map inside itself
   65 -- (so the last writer wins with a consistent view, even if mutations
   66 -- interleave the save).
   67 newPersistingCursorStore :: FilePath -> IO CursorStore
   68 newPersistingCursorStore path = do
   69   tv <- newTVarIO Map.empty
   70   let store = CursorStore { csVar = tv, csSave = Just (saveAction store) }
   71       saveAction s = saveCursorMap path =<< snapshotCursor s
   72   pure store
   73 
   74 -- | Replace the handle's map in one STM transaction. Used at boot to
   75 -- seed the in-memory store from the persisted @cursors.json@ (after the
   76 -- caller has dropped stale entries, if desired). Does NOT persist (the
   77 -- caller is loading FROM disk — writing back would be a redundant no-op).
   78 seedCursorStore :: CursorStore -> Map ConversationKey TabRef -> IO ()
   79 seedCursorStore s m = atomically (writeTVar (csVar s) m)
   80 
   81 -- | Snapshot the current map. Used by the save action.
   82 snapshotCursor :: CursorStore -> IO (Map ConversationKey TabRef)
   83 snapshotCursor s = readTVarIO (csVar s)
   84 
   85 -- | Clear every conversation whose cursor points at @ref@. Used when a tab
   86 -- is closed: the closed tab's 'TabRef' is stale, so any conversation still
   87 -- focused on it should drop the cursor (the next message will create a
   88 -- fresh tab). Single STM transaction — race-safe. Persists via 'csSave'.
   89 cursorClearAll :: CursorStore -> TabRef -> IO ()
   90 cursorClearAll s ref = do
   91   atomically $ do
   92     m <- readTVar (csVar s)
   93     writeTVar (csVar s) (Map.filter (/= ref) m)
   94   persistCursor s
   95 
   96 -- | Run the persist action (if any) after a successful mutation. A save
   97 -- failure is logged to stderr (ids + error only — no conversation content)
   98 -- and swallowed — the in-memory store stays authoritative within the
   99 -- session; the next successful mutation will retry the save (writing the
  100 -- full current map, so a missed save self-heals). Mirrors
  101 -- 'Seal.Tabs.persistIf'.
  102 persistCursor :: CursorStore -> IO ()
  103 persistCursor s =
  104   case csSave s of
  105     Nothing  -> pure ()
  106     Just act -> act `catch` \e ->
  107       globalLogIO WarningS ("[persist] cursors.json save failed: " <> ls (T.pack (show (e :: SomeException))))

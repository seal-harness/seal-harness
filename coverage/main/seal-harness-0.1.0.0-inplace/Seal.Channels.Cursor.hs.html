    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The per-conversation tab cursor store. Each conversation (a Telegram
    3 -- chat, a Signal conversation, a TUI session) has a cursor pointing at the
    4 -- tab it's currently focused on. The tab list is shared (one
    5 -- 'TabsHandle' in the gateway); the cursor is per-conversation so
    6 -- @/tab focus N@ on Telegram only affects that Telegram conversation,
    7 -- not other conversations or the TUI.
    8 --
    9 -- The store is a 'TVar' backed 'Map' from 'ConversationKey' to 'TabRef'.
   10 -- 'ConversationKey' is 'ChannelKind' × 'ConversationId' — the
   11 -- server-derived conversation identity (never user-supplied), so a sender
   12 -- cannot forge a cursor key to hijack another conversation's tab.
   13 --
   14 -- == Persistence
   15 --
   16 -- The cursor store survives a @seal serve@ restart via the optional 'csSave'
   17 -- hook (mirroring 'Seal.Tabs.TabsHandle.thSave'). A persisting store
   18 -- ('newPersistingCursorStore') writes the full current map to
   19 -- @\<state\>\/cursors.json@ atomically (0600) after every mutation, and is
   20 -- loaded + seeded at boot so an existing conversation re-resolves to its
   21 -- prior session (carrying the user's @\/model use@ choice) instead of a
   22 -- fresh default session.
   23 module Seal.Channels.Cursor
   24   ( CursorStore
   25   , newPersistingCursorStore
   26   , newPersistingCursorStoreWith
   27   , seedCursorStore
   28   , cursorClearAll
   29   , snapshotCursor
   30   ) where
   31

   45 -- | The live cursor store. Backed by a 'TVar' so concurrent reads/writes
   46 -- are race-safe (each STM operation is one transaction). The optional
   47 -- 'csSave' action is invoked after a successful mutation so the map
   48 -- persists to disk; @Just (... saveCursorMap path ...)@ for
   49 -- 'newPersistingCursorStore'. Mirrors 'Seal.Tabs.TabsHandle'.
   50 data CursorStore = CursorStore
   51   { csVar  :: TVar (Map ConversationKey TabRef)
   52   , csSave :: Maybe (IO ())
   53   , csLock :: MVar ()
   54   -- ^ Serializes the mutation+persist sequence so a stale snapshot from
   55   -- one thread cannot overwrite a newer on-disk write from another.
   56   }
   57 
   58 -- | A conversation identity: 'ChannelKind' × 'ConversationId'.
   59 -- Re-exported here for convenience; the key type is 'ConversationKey'.
   60 type ConversationKey = (Text, Text)
   61   -- ^ (channel-kind-text, conversation-id-text). We use the text forms
   62   -- rather than the structured types so the store doesn't depend on
   63   -- 'ChannelKind' or 'ConversationId' directly (keeping the module
   64   -- lightweight). Callers mint the key from 'MessageSource'.
   65 
   66 -- | Create a 'CursorStore' that persists every mutation to @path@ (atomic
   67 -- write, 0600, MVar-serialized via 'Seal.Util.AtomicJson.saveJsonAtomic').
   68 -- Used by 'Seal.Command.Serve.runServeMain' so the cursor map survives
   69 -- a restart. The save action snapshots the current map inside itself
   70 -- (so the last writer wins with a consistent view, even if mutations
   71 -- interleave the save). The 'csLock' serializes mutation+persist so the
   72 -- snapshot cannot be stale (see 'cursorClearAll').
   73 newPersistingCursorStore :: FilePath -> IO CursorStore
   74 newPersistingCursorStore path = do
   75   tv <- newTVarIO Map.empty
   76   lock <- newMVar ()
   77   let store = CursorStore { csVar = tv, csSave = Just (saveAction store), csLock = lock }
   78       saveAction s = saveCursorMap path =<< snapshotCursor s
   79   pure store
   80 
   81 -- | Create a 'CursorStore' with a custom save function (for testing).
   82 -- The save function receives the full current map. Used by
   83 -- 'Seal.Channels.CursorSpec' to verify persist serialization.
   84 newPersistingCursorStoreWith :: (Map ConversationKey TabRef -> IO ()) -> IO CursorStore
   85 newPersistingCursorStoreWith saveFn = do
   86   tv <- newTVarIO Map.empty
   87   lock <- newMVar ()
   88   let store = CursorStore { csVar = tv, csSave = Just (saveAction store), csLock = lock }
   89       saveAction s = saveFn =<< snapshotCursor s
   90   pure store
   91 
   92 -- | Replace the handle's map in one STM transaction. Used at boot to
   93 -- seed the in-memory store from the persisted @cursors.json@ (after the
   94 -- caller has dropped stale entries, if desired). Does NOT persist (the
   95 -- caller is loading FROM disk — writing back would be a redundant no-op).
   96 seedCursorStore :: CursorStore -> Map ConversationKey TabRef -> IO ()
   97 seedCursorStore s m = atomically (writeTVar (csVar s) m)
   98 
   99 -- | Snapshot the current map. Used by the save action.
  100 snapshotCursor :: CursorStore -> IO (Map ConversationKey TabRef)
  101 snapshotCursor s = readTVarIO (csVar s)
  102 
  103 -- | Clear every conversation whose cursor points at @ref@. Used when a tab
  104 -- is closed: the closed tab's 'TabRef' is stale, so any conversation still
  105 -- focused on it should drop the cursor (the next message will create a
  106 -- fresh tab). Single STM transaction — race-safe. Persists via 'csSave'.
  107 cursorClearAll :: CursorStore -> TabRef -> IO ()
  108 cursorClearAll s ref =
  109   -- Hold the lock through mutation+persist so no concurrent thread can
  110   -- mutate the TVar between our snapshot and our save (which would let a
  111   -- stale snapshot overwrite a newer on-disk write).
  112   withMVar (csLock s) $ \_ -> do
  113     atomically $ do
  114       m <- readTVar (csVar s)
  115       writeTVar (csVar s) (Map.filter (/= ref) m)
  116     persistCursor s
  117 
  118 -- | Run the persist action (if any) after a successful mutation. A save
  119 -- failure is logged to stderr (ids + error only — no conversation content)
  120 -- and swallowed — the in-memory store stays authoritative within the
  121 -- session; the next successful mutation will retry the save (writing the
  122 -- full current map, so a missed save self-heals). Mirrors
  123 -- 'Seal.Tabs.persistIf'.
  124 persistCursor :: CursorStore -> IO ()
  125 persistCursor s =
  126   case csSave s of
  127     Nothing  -> pure ()
  128     Just act -> act `catch` \e ->
  129       globalLogIO WarningS ("[persist] cursors.json save failed: " <> ls (T.pack (show (e :: SomeException))))

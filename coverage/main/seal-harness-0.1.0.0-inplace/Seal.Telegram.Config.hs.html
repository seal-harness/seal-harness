   43 -- ---------------------------------------------------------------------------
   44 -- TelegramToken — the validated bot token for the Telegram Bot API
   45 -- ---------------------------------------------------------------------------
   46 
   47 -- | A Telegram bot token as issued by BotFather (@<digits>:<AA-xx>@). Smart-
   48 -- constructed: non-empty, no leading dash (option-injection defense — a
   49 -- leading-dash value would be interpreted as a flag by a subprocess if the
   50 -- token ever reaches an argv), charset @[A-Za-z0-9+:-_@. The validated type
   51 -- that reaches the Bot API URL path. Not a secret in the vault sense (it is
   52 -- stored in @config.toml@); it is the channel credential, like the Signal
   53 -- account label.
   54 newtype TelegramToken = TelegramToken Text
   55   deriving stock (Eq, Show)
   56 
   57 -- | Smart-construct a 'TelegramToken'. Accepts the BotFather format
   58 -- (@<digits>:<rest>@) and rejects empty / leading-dash / invalid chars.
   59 mkTelegramToken :: Text -> Either Text TelegramToken
   60 mkTelegramToken t
   61   | T.null t        = Left "telegram token is empty"
   62   | T.head t == '-' = Left "telegram token must not start with '-' (option injection)"
   63   | not (T.all validTokenChar t)
   64                      = Left "telegram token has invalid characters"
   65   | otherwise       = Right (TelegramToken t)
   66 
   67 validTokenChar :: Char -> Bool
   68 validTokenChar c = c `Set.member` tokenChars
   69   where
   70     tokenChars :: Set Char
   71     tokenChars = Set.fromList
   72       $ ['A'..'Z'] <> ['a'..'z'] <> ['0'..'9'] <> "+-:_"
   73 
   74 telegramTokenText :: TelegramToken -> Text
   75 telegramTokenText (TelegramToken t) = t

   78 -- TelegramConfig — the [telegram] config section

   81 -- | The @[telegram]@ config section. All fields optional at the file level;
   82 -- @seal telegram@ fails fast via 'resolveTelegramConfig' if 'tcToken' is
   83 -- unset (no config token AND no vault-supplied token).
   84 data TelegramConfig = TelegramConfig
   85   { tcToken         :: Maybe Text       -- ^ BotFather bot token
   86   , tcTextChunkLimit :: Maybe Int        -- ^ default 'defaultTelegramChunkLimit'
   87   , tcAllowFrom     :: AllowList Text   -- ^ sender allow-list (Telegram user id)
   88   } deriving stock (Eq, Show)
   89 
   90 -- | Telegram's text-message character limit. Replies longer than this are
   91 -- chunked via 'Seal.Channels.Chat.Telegram.chunkMessage'. Telegram's
   92 -- hard limit is 4096; we leave headroom for the @model> @ prefix.
   93 defaultTelegramChunkLimit :: Int
   94 defaultTelegramChunkLimit = 3900
   95 
   96 defaultTelegramConfig :: TelegramConfig
   97 defaultTelegramConfig = TelegramConfig
   98   { tcToken = Nothing
   99   , tcTextChunkLimit = Just defaultTelegramChunkLimit
  100   , tcAllowFrom = AllowAll
  101   }

  103 -- ---------------------------------------------------------------------------
  104 -- TOML codec (bidirectional, used by Seal.Config.File)
  105 -- ---------------------------------------------------------------------------
  106 
  107 -- | Bidirectional tomland codec for the @[telegram]@ section. 'tcAllowFrom'
  108 -- is a TOML array of strings (@allow_from = ["12345", "67890"]@); an absent
  109 -- array decodes as 'AllowAll' via the shared 'allowListCodec' from
  110 -- 'Seal.Core.AllowList'.
  111 telegramConfigCodec :: Toml.TomlCodec TelegramConfig
  112 telegramConfigCodec = TelegramConfig
  113   <$> Toml.dioptional (Toml.text "token")            .= tcToken
  114   <*> Toml.dioptional (Toml.int "text_chunk_limit")  .= tcTextChunkLimit
  115   <*> allowListCodec                                  .= tcAllowFrom

  117 -- ---------------------------------------------------------------------------
  118 -- Resolution — config section + vault token -> validated (token, limit, allow)
  119 -- ---------------------------------------------------------------------------
  120 
  121 -- | Resolve the @[telegram]@ section + an optional vault-supplied token
  122 -- into a validated 'TelegramToken', chunk limit, and 'AllowList UserId' for
  123 -- the channel's sender check. The vault token (if supplied) overrides the
  124 -- config token; either way it's smart-constructed here. A missing token (no
  125 -- section, no vault token) is rejected. The @allow_from@ 'AllowList Text' is
  126 -- mapped through 'mkUserId' to an 'AllowList UserId' — a malformed entry
  127 -- fails the whole resolution with a clear error.
  128 resolveTelegramConfig
  129   :: Maybe TelegramConfig -> Maybe Text
  130   -> Either Text (TelegramToken, Int, AllowList UserId)
  131 resolveTelegramConfig mSection mVaultToken =
  132   case mVaultToken <|> (mSection >>= tcToken) of
  133     Nothing -> Left "telegram: no token configured (set [telegram].token or supply via vault)"
  134     Just tokenRaw -> do
  135       token <- mkTelegramToken tokenRaw
  136       let tgCfg = fromMaybe defaultTelegramConfig mSection
  137           limit  = fromMaybe defaultTelegramChunkLimit (tcTextChunkLimit tgCfg)
  138       allowUserId <- mapAllowUserId (tcAllowFrom tgCfg)
  139       Right (token, limit, allowUserId)
  140 
  141 -- | Map an 'AllowList Text' through 'mkUserId' to an 'AllowList UserId'.
  142 -- 'AllowAll' passes through; 'AllowOnly' maps each element (a malformed
  143 -- entry fails the whole resolution).
  144 mapAllowUserId :: AllowList Text -> Either Text (AllowList UserId)
  145 mapAllowUserId AllowAll = Right AllowAll
  146 mapAllowUserId (AllowOnly txts) = do
  147   uids <- traverse mkUserIdText (Set.toList txts)
  148   Right (AllowOnly (Set.fromList uids))
  149   where
  150     mkUserIdText t = case mkUserId t of
  151       Right u  -> Right u
  152       Left err -> Left ("telegram allow_from has invalid user id " <> T.pack (show t) <> ": " <> err)

   34 -- | HARNESS_LIST: return the registry snapshot as a JSON array of
   35 -- {id, label, liveness, flavour} objects.
   36 harnessListOp :: HarnessRegistry -> Opcode
   37 harnessListOp reg = TrustedOpcode
   38   { toName = OpName "HARNESS_LIST"
   39   , toTrust = Trusted
   40   , toDesc = "List the live harnesses registered with the harness backend."
   41   , toInSchema = object
   42       [ "type" .= ("object" :: Text)
   43       , "properties" .= object []
   44       , "required" .= ([] :: [Text])
   45       ]
   46   , toOutSchema = object
   47       [ "type" .= ("array" :: Text)
   48       , "items" .= object
   49           [ "type" .= ("object" :: Text)
   50           , "properties" .= object
   51               [ fromText "id" .= object ["type" .= ("string" :: Text)]
   52               , fromText "label" .= object ["type" .= ("string" :: Text)]
   53               , fromText "liveness" .= object ["type" .= ("string" :: Text)]
   54               ]
   55           ]
   56       ]
   57   , toAuthorize = const (Right ())
   58   , toBlocking = False
   59   , toRun = \_back _input -> liftIO $ do
   60       entries <- snapshot reg
   61       let arr = map entryToJson entries
   62           recorded = object ["harnesses" .= arr]
   63       pure (OpResult [TrpText (T.pack (show arr))] False recorded)
   64   }

   66 -- | HARNESS_START: spawn a harness in a tmux window, stamp the @seal_id
   67 -- marker, insert a HarnessEntry (OriginSpawned), return the HarnessId.
   68 -- The tmux session/window idents + the flavour are supplied by the caller
   69 -- (the 6b wiring resolves them); 'mintId' mints the fresh durable id.
   70 harnessStartOp
   71   :: HarnessRegistry -> TmuxRunner -> TmuxIdent -> TmuxIdent
   72   -> HarnessFlavour -> IO HarnessId -> Opcode
   73 harnessStartOp reg runner session window flavour mintId = TrustedOpcode
   74   { toName = OpName "HARNESS_START"
   75   , toTrust = Trusted
   76   , toDesc = "Start a harness in a tmux window and register it."
   77   , toInSchema = object
   78       [ "type" .= ("object" :: Text)
   79       , "properties" .= object
   80           [ fromText "flavour" .= object
   81               [ "type" .= ("string" :: Text)
   82               , "description" .= ("claude-code | codex | generic | <custom>" :: Text)
   83               ]
   84           ]
   85       , "required" .= (["flavour"] :: [Text])
   86       ]
   87   , toOutSchema = object
   88       [ "type" .= ("object" :: Text)
   89       , "properties" .= object [fromText "id" .= object ["type" .= ("string" :: Text)]]
   90       , "required" .= (["id"] :: [Text])
   91       ]
   92   , toAuthorize = const (Right ())
   93   , toBlocking = False
   94   , toRun = \_back _input -> liftIO $ do
   95       hid <- mintId
   96       r1 <- startTmuxSessionStatus runner session
   97       case r1 of
   98         Left e -> pure (opErr (T.pack (show e)))
   99         Right _ -> do
  100           r2 <- addHarnessWindowNamed runner session window hid
  101           case r2 of
  102             Left e -> pure (opErr (T.pack (show e)))
  103             Right _ -> do
  104               let entry = HarnessEntry
  105                     { heId = hid
  106                     , heLabel = flavourLabel flavour
  107                     , heOrigin = HoSpawned
  108                     , heLiveness = LvIdle
  109                     , heTmuxCoord = Just (tmuxIdentText session <> ":" <> tmuxIdentText window)
  110                     , heFlavour = Just (flavourLabel flavour)
  111                     , heOrphanTicks = 0
  112                     }
  113               atomically (insert reg entry)
  114               let recorded = object ["id" .= harnessIdToText hid, "flavour" .= flavourLabel flavour]
  115               pure (OpResult [TrpText (harnessIdToText hid)] False recorded)
  116   }

  118 -- | HARNESS_STOP: stop a harness (kill the tmux window) + mark the entry
  119 -- LvExited. Takes the harness id in the input.
  120 harnessStopOp :: HarnessRegistry -> TmuxRunner -> Opcode
  121 harnessStopOp reg runner = TrustedOpcode
  122   { toName = OpName "HARNESS_STOP"
  123   , toTrust = Trusted
  124   , toDesc = "Stop a harness (kill its tmux window) and mark it exited."
  125   , toInSchema = object
  126       [ "type" .= ("object" :: Text)
  127       , "properties" .= object [fromText "id" .= object ["type" .= ("string" :: Text)]]
  128       , "required" .= (["id"] :: [Text])
  129       ]
  130   , toOutSchema = object ["type" .= ("object" :: Text), "properties" .= object []]
  131   , toAuthorize = const (Right ())
  132   , toBlocking = False
  133   , toRun = \_back input -> liftIO $ do
  134       case idField input of
  135         Nothing -> pure (opErr "missing id")
  136         Just hidText -> case parseHarnessId hidText of
  137           Left e -> pure (opErr e)
  138           Right hid -> do
  139             mEntry <- atomically (lookupById reg hid)
  140             case mEntry of
  141               Nothing -> pure (opErr "harness not found")
  142               Just _entry -> do
  143                 -- The coord->TmuxIdent split is simplified for 6a (the
  144                 -- real 6b wiring splits "session:window"); use the window
  145                 -- ident the entry was spawned with. For 6a's capstone the
  146                 -- test passes a known window.
  147                 let win = fromMaybe (either (error "ident") id (mkTmuxIdent "win")) Nothing
  148                 r <- stopHarnessWindowNamed runner win
  149                 case r of
  150                   Left e -> pure (opErr (T.pack (show e)))
  151                   Right _ -> do
  152                     atomically (modify reg hid (\e -> e { heLiveness = LvExited }))
  153                     pure (OpResult [TrpText "stopped"] False (object ["id" .= harnessIdToText hid]))
  154   }

  160 opErr :: Text -> OpResult
  161 opErr t = OpResult [TrpText t] True (object ["error" .= t])
  162 
  163 entryToJson :: HarnessEntry -> Value
  164 entryToJson e = object
  165   [ "id" .= harnessIdToText (heId e)
  166   , "label" .= heLabel e
  167   , "liveness" .= livenessText (heLiveness e)
  168   , "flavour" .= heFlavour e
  169   ]
  170 
  171 livenessText :: Liveness -> Text
  172 livenessText LvIdle          = "idle"
  173 livenessText LvThinking      = "thinking"
  174 livenessText LvAwaitingInput = "awaiting_input"
  175 livenessText LvExited        = "exited"
  176 livenessText LvOrphaned      = "orphaned"
  177 
  178 flavourLabel :: HarnessFlavour -> Text
  179 flavourLabel HfClaudeCode = "claude-code"
  180 flavourLabel HfCodex      = "codex"
  181 flavourLabel HfGeneric    = "generic"
  182 flavourLabel (HCustom t)  = t
  183 
  184 idField :: Value -> Maybe Text
  185 idField v = case v of
  186   Data.Aeson.Object o -> case KeyMap.lookup (fromText "id") o of
  187     Just (Data.Aeson.String t) -> Just t
  188     _ -> Nothing
  189   _ -> Nothing

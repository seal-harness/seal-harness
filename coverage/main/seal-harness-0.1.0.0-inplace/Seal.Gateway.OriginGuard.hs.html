    2 -- | The mutation guard's verdict: may this request mutate state?
    3 --
    4 -- The gateway's REST API is reachable from any page in the operator's browser.
    5 -- Passkey sessions do not fix that on their own: a cookie is sent on a
    6 -- cross-origin request, so a hostile page can drive @POST@/@PUT@/@DELETE@
    7 -- against the gateway (CSRF). The passkey layer decides *network* identity; this
    8 -- module decides whether the *request origin* is the gateway's own.
    9 --
   10 -- The canonical origin is resolved once at boot by 'Seal.Config.BootPlan' and is
   11 -- the same value WebAuthn verifies @clientDataJSON.origin@ against. The
   12 -- WebSocket allowlist is derived separately, from @[gateway] host@ + @port@, and
   13 -- is exact: a browser opened at @http:\/\/127.0.0.1:<port>@ can use the REST API
   14 -- here but not the live stream. Open the URL the boot banner prints.
   15 --
   16 -- Pure on purpose: the whole verdict (including the loopback-alias cases that
   17 -- keep @vite dev@ and the e2e harness working) is testable without a socket.
   18 --
   19 -- == Residual threats (stated, not claimed away)
   20 --
   21 -- * Holders of the per-run service token and clients in an explicitly waived
   22 --   @mode = "none"@ install are authorised to mutate; the guard only stops a
   23 --   *browser page from another origin*.
   24 -- * A same-UID local process is admitted: it sends no @Origin@, and it could
   25 --   already write @~\/.seal\/security.toml@ directly. See
   26 --   "Seal.Gateway.Setup"'s module header for the same statement about setup.
   27 -- * @mode = "none"@ on a wildcard bind resolves no canonical origin, so the
   28 --   verdict is @Allowed@ for everything. That is the operator's written
   29 --   waiver, not an oversight.
   30 module Seal.Gateway.OriginGuard
   31   ( GuardHeaders (..)
   32   , GuardVerdict (..)
   33   , sameOriginVerdict
   34   , guardHeadersFrom
   35   ) where

   44 -- | The only two headers the verdict needs.
   45 data GuardHeaders = GuardHeaders
   46   { ghOrigin       :: !(Maybe Text)
   47     -- ^ @Origin@. Absent for curl/HTTP clients and for same-origin GETs.
   48   , ghSecFetchSite :: !(Maybe Text)
   49     -- ^ @Sec-Fetch-Site@. Browsers send it on every request; @cross-site@ on a
   50     -- cross-origin one.
   51   } deriving stock (Eq, Show)
   52 
   53 -- | The verdict. A rejection carries a short reason for the log and the 403
   54 -- body (never request content beyond the fact that it did not match).
   55 data GuardVerdict = Allowed | Rejected !Text
   56   deriving stock (Eq, Show)
   57 
   58 -- | @Sec-Fetch-Site@ has no constant in @http-types@; header names are
   59 -- case-insensitive and @HeaderName@ is @IsString@ over the lowercased form.
   60 hSecFetchSite :: HeaderName
   61 hSecFetchSite = "sec-fetch-site"
   62 
   63 -- | Decide whether a request may mutate state.
   64 --
   65 -- @Nothing@ is the boot plan's explicit "no origin could be resolved" (a
   66 -- wildcard bind with @mode = none@): the operator waived access control and
   67 -- there is no canonical origin to compare against, so everything is allowed.
   68 --
   69 -- A loopback canonical origin accepts any loopback alias on any port. That is
   70 -- load-bearing: the boot plan canonicalises a loopback bind to
   71 -- @http:\/\/localhost:\<port\>@, while the SPA is reached at
   72 -- @http:\/\/127.0.0.1:\<port\>@ (the e2e harness) or
   73 -- @http:\/\/localhost:5173@ (the vite dev proxy). Only code already running on
   74 -- the host can serve from a loopback name.
   75 sameOriginVerdict :: Maybe Text -> GuardHeaders -> GuardVerdict
   76 sameOriginVerdict Nothing _headers = Allowed
   77 sameOriginVerdict (Just canonical) headers =
   78   case ghOrigin headers of
   79     Nothing -> allowWithoutOrigin
   80     Just raw
   81       | T.toLower (T.strip raw) == "null" ->
   82           Rejected "Origin: null is not an acceptable origin"
   83       | otherwise -> case (parseOrigin canonical, parseOrigin raw) of
   84           (Just canon, Just origin)
   85             | identicalOrigin canon origin -> Allowed
   86             | poScheme canon == poScheme origin
   87             , isLoopbackHost (poHost canon)
   88             , isLoopbackHost (poHost origin) -> Allowed
   89             | otherwise ->
   90                 Rejected ("origin does not match the gateway origin (" <> raw <> ")")
   91           _ -> Rejected "Origin header is not a well-formed http(s) origin"
   92   where
   93     -- A browser cannot issue a cross-site request without telling us so. Absent
   94     -- metadata means a non-browser client (curl, the chat channels, tests).
   95     allowWithoutOrigin = case ghSecFetchSite headers of
   96       Just site
   97         | T.toLower (T.strip site) `elem` ["cross-site", "same-site"] ->
   98             Rejected "cross-site fetch metadata without Origin"
   99       _ -> Allowed
  100 
  101 -- | Extract the two headers from a WAI request. Lenient UTF-8 decoding matches
  102 -- the rest of the gateway: a malformed header simply fails to match.
  103 guardHeadersFrom :: Request -> GuardHeaders
  104 guardHeadersFrom req = GuardHeaders
  105   { ghOrigin       = lookupText hOrigin
  106   , ghSecFetchSite = lookupText hSecFetchSite
  107   }
  108   where
  109     lookupText :: HeaderName -> Maybe Text
  110     lookupText name = TE.decodeUtf8Lenient <$> lookup name (requestHeaders req)

  113 -- Origin parsing

  116 -- | A parsed origin: scheme, host (lowercased, brackets kept for IPv6) and the
  117 -- effective port (explicit, else the scheme default). Nothing else is carried —
  118 -- an origin with a path, query or fragment is not an origin.
  119 data ParsedOrigin = ParsedOrigin
  120   { poScheme :: !Text
  121   , poHost   :: !Text
  122   , poPort   :: !Int
  123   } deriving stock (Eq, Show)
  124 
  125 parseOrigin :: Text -> Maybe ParsedOrigin
  126 parseOrigin raw = do
  127   let stripped = T.strip raw
  128       -- Browsers never send a trailing slash, but some tools do; tolerate one.
  129       cleaned = if "/" `T.isSuffixOf` stripped then T.dropEnd 1 stripped else stripped
  130   (scheme, afterScheme) <- splitScheme cleaned
  131   host <- splitHost afterScheme
  132   port <- splitPort scheme afterScheme
  133   pure (ParsedOrigin (T.toLower scheme) host port)
  134 
  135 splitScheme :: Text -> Maybe (Text, Text)
  136 splitScheme t =
  137   case T.breakOn "://" t of
  138     (scheme, rest)
  139       | T.null rest                -> Nothing
  140       | T.null scheme             -> Nothing
  141       | T.toLower scheme `notElem` ["http", "https"] -> Nothing
  142       | otherwise                 -> Just (scheme, T.drop 3 rest)
  143 
  144 -- | The host part of an authority, with a validation that nothing hostile
  145 -- follows it. A path/query/fragment anywhere in the authority is rejected.
  146 splitHost :: Text -> Maybe Text
  147 splitHost authority
  148   | T.null authority = Nothing
  149   | T.any (`elem` ("/?#" :: String)) authority = Nothing
  150   | "[" `T.isPrefixOf` authority =
  151       case T.breakOn "]" authority of
  152         (h, rest)
  153           | T.null rest -> Nothing
  154           | otherwise   ->
  155               let after = T.drop 1 rest
  156               in if T.null after || validPortSuffix after
  157                    then Just (T.toLower h <> "]")
  158                    else Nothing
  159   | otherwise =
  160       case T.breakOn ":" authority of
  161         (h, rest)
  162           | T.null h -> Nothing
  163           | T.null rest -> Just (T.toLower h)
  164           | validPortSuffix rest -> Just (T.toLower h)
  165           | otherwise -> Nothing
  166 
  167 splitPort :: Text -> Text -> Maybe Int
  168 splitPort scheme authority
  169   | "[" `T.isPrefixOf` authority =
  170       case T.breakOn "]" authority of
  171         (_, rest) -> portFrom (T.drop 1 rest) (defaultPort scheme)
  172   | otherwise =
  173       case T.breakOn ":" authority of
  174         (_, rest) -> portFrom rest (defaultPort scheme)
  175 
  176 portFrom :: Text -> Int -> Maybe Int
  177 portFrom rest fallback
  178   | T.null rest = Just fallback
  179   | validPortSuffix rest =
  180       case reads (T.unpack (T.drop 1 rest)) of
  181         [(n, "")] -> Just n
  182         _         -> Nothing
  183   | otherwise = Nothing
  184 
  185 validPortSuffix :: Text -> Bool
  186 validPortSuffix t =
  187   case T.uncons t of
  188     Just (':', digits) -> not (T.null digits) && T.all isDigit digits && T.length digits <= 5
  189     _                  -> False
  190 
  191 defaultPort :: Text -> Int
  192 defaultPort scheme = if T.toLower scheme == "https" then 443 else 80
  193 
  194 identicalOrigin :: ParsedOrigin -> ParsedOrigin -> Bool
  195 identicalOrigin a b =
  196   poScheme a == poScheme b && poHost a == poHost b && poPort a == poPort b
  197 
  198 -- | @localhost@, any @127.0.0.0\/8@ address, and IPv6 @::1@ (with or without
  199 -- brackets). Everything else — including @localhost.evil.com@ and @10.x@ — is
  200 -- not loopback.
  201 isLoopbackHost :: Text -> Bool
  202 isLoopbackHost host
  203   | h == "localhost" = True
  204   | h == "[::1]" || h == "::1" = True
  205   | otherwise = case T.splitOn "." h of
  206       [a, b, c, d] -> a == "127" && all numeric [b, c, d]
  207       _            -> False
  208   where
  209     h = T.toLower host
  210     numeric part = not (T.null part) && T.all isDigit part

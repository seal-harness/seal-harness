   48 -- | Run a process with no stdin, capturing stdout and stderr as strict
   49 -- 'ByteString's. Used by the age-keygen / age-plugin-yubikey setup seams
   50 -- (their output may carry non-ASCII recipient strings).
   51 readProcessNoInput :: FilePath -> [String] -> IO (ExitCode, BS.ByteString, BS.ByteString)
   52 readProcessNoInput cmdPath args =
   53   withCreateProcess
   54     ( (proc cmdPath args)
   55         { std_in = NoStream, std_out = CreatePipe, std_err = CreatePipe }
   56     ) $ \_ mOut mErr ph -> do
   57       (hOut, hErr) <- case (mOut, mErr) of
   58         (Just a, Just b) -> pure (a, b)
   59         _ -> error "readProcessNoInput: pipe creation failed (unreachable)"
   60       out <- BS.hGetContents hOut
   61       err <- BS.hGetContents hErr
   62       ec  <- waitForProcess ph
   63       let !_ = BS.length out
   64           !_ = BS.length err
   65       pure (ec, out, err)

   87 -- | Pure helper: given the filenames in one directory, return the plugin suffixes.
   88 filterPluginNames :: [FilePath] -> [Text]
   89 filterPluginNames names =
   90   [ T.pack (drop prefixLen n)
   91   | n <- names
   92   , pluginPrefix `isPrefixOf` n
   93   ]
   94   where
   95     pluginPrefix = "age-plugin-"
   96     prefixLen    = length pluginPrefix
   97 
   98 detectAgePlugins :: IO [Text]
   99 detectAgePlugins = do
  100   mPath    <- lookupEnv "PATH"
  101   let dirs  = maybe [] splitSearchPath mPath
  102   allNames <- traverse safeList dirs
  103   pure (sort (nub (concatMap filterPluginNames allNames)))
  104   where
  105     safeList d = do
  106       r <- try @IOException (listDirectory d)
  107       pure (fromRight [] r)

  113 -- | First available relative key-file name under the keys directory, of the
  114 -- form @<base><ext>@, @<base>-1<ext>@, @<base>-2<ext>@, … A newly generated key
  115 -- is always written to a name that does not yet exist, so setup NEVER
  116 -- overwrites an identity file that an existing vault still depends on: key
  117 -- rotation must keep the old key intact in order to decrypt the current vault.
  118 freshKeyName :: FilePath -> String -> String -> IO FilePath
  119 freshKeyName keysDir base ext = go (0 :: Int)
  120   where
  121     go n = do
  122       let rel = base <> (if n == 0 then "" else "-" <> show n) <> ext
  123       taken <- doesPathExist (keysDir </> rel)
  124       if taken then go (n + 1) else pure rel

  126 setupLocalAgeKey :: SealPaths -> Text -> IO (Either Text ResolvedKey)
  127 setupLocalAgeKey paths name = do
  128   keysRoot <- ensureKeysRoot (spKeys paths)
  129   rel      <- freshKeyName (spKeys paths) (T.unpack name) ".identity"
  130   pathRes  <- mkSafeKeyPath keysRoot rel
  131   case pathRes of
  132     Left err       -> pure (Left (T.pack (show err)))
  133     Right safePath -> do
  134       let identPath = getSafeKeyPath safePath
  135       (exitCode, _stdout, stderrBs) <-
  136         readProcessNoInput "age-keygen" ["-o", identPath]
  137       case exitCode of
  138         ExitFailure n ->
  139           pure (Left ("age-keygen exited with code " <> T.pack (show n)))
  140         ExitSuccess -> do
  141           let stderrText = TE.decodeUtf8Lenient stderrBs
  142           case parseAgePublicKey stderrText of
  143             Nothing  -> pure (Left "age-keygen: could not parse public key from stderr")
  144             Just pub -> do
  145               setFileMode identPath 0o600
  146               pure (Right ResolvedKey
  147                 { rkRecipient = pub
  148                 , rkIdentity  = T.pack identPath
  149                 , rkKeyType   = "x25519"
  150                 })

  152 -- | Generate (or reuse) a YubiKey-backed age identity.
  153 --
  154 -- @touchRequired@ -> @--touch-policy always|never@.
  155 -- @pinRequired@   -> @--pin-policy   once|never@. Choosing @never@ means
  156 -- decryption needs no PIN (only the token present, plus a touch if required);
  157 -- choosing @once@ (the age-plugin-yubikey default) prompts for the PIN once per
  158 -- decrypt session.
  159 setupYubiKey
  160   :: SealPaths -> Text -> Bool -> Bool -> ChannelCaps
  161   -> IO (Either Text ResolvedKey)
  162 setupYubiKey paths name touchRequired pinRequired caps = do
  163   mPlugin <- findExecutable "age-plugin-yubikey"
  164   case mPlugin of
  165     Nothing ->
  166       pure (Left "age-plugin-yubikey not found on PATH")
  167     Just _ -> do
  168       keysRoot <- ensureKeysRoot (spKeys paths)
  169       rel      <- freshKeyName (spKeys paths) (T.unpack name) ".yubikey.txt"
  170       pathRes  <- mkSafeKeyPath keysRoot rel
  171       case pathRes of
  172         Left err -> pure (Left (T.pack (show err)))
  173         Right safePath -> do
  174           let identPath   = getSafeKeyPath safePath
  175               touchPolicy = if touchRequired then "always" else "never"
  176               pinPolicy   = if pinRequired then "once" else "never"
  177           (exitCode, stdoutBs, _) <-
  178             readProcessNoInput "age-plugin-yubikey"
  179               [ "--generate"
  180               , "--touch-policy", touchPolicy
  181               , "--pin-policy", pinPolicy
  182               ]
  183           let stdoutText = TE.decodeUtf8Lenient stdoutBs
  184           mRecipient <-
  185             if exitCode == ExitSuccess && not (T.null (T.strip stdoutText))
  186               then do
  187                 BS.writeFile identPath (TE.encodeUtf8 stdoutText)
  188                 setFileMode identPath 0o600
  189                 pure (Right (parsePluginRecipient stdoutText))
  190               else do
  191                 -- TTY fallback: instruct user and wait for manual completion.
  192                 ccSend caps
  193                   ("age-plugin-yubikey requires interactive input. Run:\n"
  194                    <> "    age-plugin-yubikey --generate --touch-policy "
  195                    <> T.pack touchPolicy <> " --pin-policy " <> T.pack pinPolicy
  196                    <> " > " <> T.pack identPath)
  197                 _ <- ccPrompt caps (AskPrompt "Press Enter once the command has completed" [])
  198                 rawE <- try @IOException (BS.readFile identPath)
  199                 case rawE of
  200                   Left e ->
  201                     pure (Left ("age-plugin-yubikey: identity file not readable: "
  202                                  <> T.pack (show e)))
  203                   Right raw -> do
  204                     setFileMode identPath 0o600
  205                     pure (Right (parsePluginRecipient (TE.decodeUtf8Lenient raw)))
  206           case mRecipient of
  207             Left ioErr       -> pure (Left ioErr)
  208             Right Nothing    -> pure (Left "age-plugin-yubikey: could not parse recipient line")
  209             Right (Just pub) ->
  210               pure (Right ResolvedKey
  211                 { rkRecipient = pub
  212                 , rkIdentity  = T.pack identPath
  213                 , rkKeyType   = "yubikey"
  214                 })

  216 setupUserSupplied :: ChannelCaps -> IO (Either Text ResolvedKey)
  217 setupUserSupplied caps = do
  218   recipient <- ccPrompt caps (AskPrompt "Recipient (age1\x2026): " [])
  219   identity  <- ccPrompt caps (AskPrompt "Identity file path: " [])
  220   pure (Right ResolvedKey
  221     { rkRecipient = T.strip recipient
  222     , rkIdentity  = T.strip identity
  223     , rkKeyType   = "user"
  224     })

  235 resolveEncryptor :: SecurityConfig -> IO (Either VaultError VaultEncryptor)
  236 resolveEncryptor fc =
  237   case (scVaultRecipient fc, scVaultIdentity fc) of
  238     (Just r, Just i) ->
  239       mkAgeEncryptor (AgeRecipient r) (AgeIdentity i)
  240     (Nothing, _) ->
  241       pure (Left (VaultBackendError "vault not configured: missing vault_recipient"))
  242     (_, Nothing) ->
  243       pure (Left (VaultBackendError "vault not configured: missing vault_identity"))

  249 -- | Parse "Public key: age1..." from age-keygen stderr.
  250 -- age-keygen always emits exactly this line (capital P, no '#' prefix).
  251 parseAgePublicKey :: Text -> Maybe Text
  252 parseAgePublicKey txt =
  253   case filter (T.isPrefixOf "Public key: ") (T.lines txt) of
  254     []       -> Nothing
  255     (line:_) -> Just (T.drop (T.length "Public key: ") line)
  256 
  257 -- | Parse the recipient (public key) from age-plugin-yubikey @--generate@
  258 -- output. The recipient appears on a comment line such as
  259 -- @#    Recipient: age1yubikey1...@. age-plugin-yubikey RIGHT-ALIGNS its
  260 -- comment labels, so the number of spaces between @#@ and the label varies
  261 -- between lines; we therefore strip the leading @#@ and surrounding whitespace
  262 -- before matching the label (case-insensitively per the age plugin spec) and
  263 -- take everything after the first @:@.
  264 parsePluginRecipient :: Text -> Maybe Text
  265 parsePluginRecipient txt =
  266   case filter isRecipientLine (T.lines txt) of
  267     []       -> Nothing
  268     (line:_) ->
  269       let value = T.strip (T.drop 1 (T.dropWhile (/= ':') (labelOf line)))
  270       in if T.null value then Nothing else Just value
  271   where
  272     -- Drop a leading @#@ comment marker and the surrounding alignment spaces,
  273     -- leaving the label, e.g. @"Recipient: age1yubikey1..."@.
  274     labelOf l = T.stripStart (T.dropWhile (== '#') (T.stripStart l))
  275     isRecipientLine l = "recipient:" `T.isPrefixOf` T.toCaseFold (labelOf l)

   57 -- | Run a process with no stdin, capturing stdout and stderr as strict
   58 -- 'ByteString's. Used by the age-keygen / age-plugin-yubikey setup seams
   59 -- (their output may carry non-ASCII recipient strings).
   60 readProcessNoInput :: FilePath -> [String] -> IO (ExitCode, BS.ByteString, BS.ByteString)
   61 readProcessNoInput cmdPath args =
   62   withCreateProcess
   63     ( (proc cmdPath args)
   64         { std_in = NoStream, std_out = CreatePipe, std_err = CreatePipe }
   65     ) $ \_ mOut mErr ph -> do
   66       (hOut, hErr) <- case (mOut, mErr) of
   67         (Just a, Just b) -> pure (a, b)
   68         _ -> error "readProcessNoInput: pipe creation failed (unreachable)"
   69       out <- BS.hGetContents hOut
   70       err <- BS.hGetContents hErr
   71       ec  <- waitForProcess ph
   72       let !_ = BS.length out
   73           !_ = BS.length err
   74       pure (ec, out, err)

   92 -- | Why a non-interactive key setup failed.
   93 --
   94 -- Kept as a sum type rather than a message so the web wizard can map each
   95 -- class to the right response: a missing binary is "install it", a plugin that
   96 -- wanted stdin is "run this command in a terminal", anything else is a 500-ish
   97 -- failure. The 'Text' payloads are for the log; only 'KeySetupNeedsTty'\'s
   98 -- command is ever shown to a user.
   99 data KeySetupFailure
  100   = KeySetupMissing Text
  101     -- ^ The tool is not on PATH (the payload is the binary name).
  102   | KeySetupNeedsTty Text
  103     -- ^ The plugin needs a terminal; the payload is the runnable command.
  104   | KeySetupFailed Text
  105     -- ^ Anything else; the payload is the tool's own output.
  106   deriving stock (Eq, Show)

  112 -- | Pure helper: given the filenames in one directory, return the plugin suffixes.
  113 filterPluginNames :: [FilePath] -> [Text]
  114 filterPluginNames names =
  115   [ T.pack (drop prefixLen n)
  116   | n <- names
  117   , pluginPrefix `isPrefixOf` n
  118   ]
  119   where
  120     pluginPrefix = "age-plugin-"
  121     prefixLen    = length pluginPrefix

  123 detectAgePlugins :: IO [Text]
  124 detectAgePlugins = do
  125   mPath    <- lookupEnv "PATH"
  126   let dirs  = maybe [] splitSearchPath mPath
  127   allNames <- traverse safeList dirs
  128   pure (sort (nub (concatMap filterPluginNames allNames)))
  129   where
  130     safeList d = do
  131       r <- try @IOException (listDirectory d)
  132       pure (fromRight [] r)

  138 -- | First available relative key-file name under the keys directory, of the
  139 -- form @<base><ext>@, @<base>-1<ext>@, @<base>-2<ext>@, … A newly generated key
  140 -- is always written to a name that does not yet exist, so setup NEVER
  141 -- overwrites an identity file that an existing vault still depends on: key
  142 -- rotation must keep the old key intact in order to decrypt the current vault.
  143 freshKeyName :: FilePath -> String -> String -> IO FilePath
  144 freshKeyName keysDir base ext = go (0 :: Int)
  145   where
  146     go n = do
  147       let rel = base <> (if n == 0 then "" else "-" <> show n) <> ext
  148       taken <- doesPathExist (keysDir </> rel)
  149       if taken then go (n + 1) else pure rel

  151 setupLocalAgeKey :: SealPaths -> Text -> IO (Either Text ResolvedKey)
  152 setupLocalAgeKey paths name = do
  153   keysRoot <- ensureKeysRoot (spKeys paths)
  154   rel      <- freshKeyName (spKeys paths) (T.unpack name) ".identity"
  155   pathRes  <- mkSafeKeyPath keysRoot rel
  156   case pathRes of
  157     Left err       -> pure (Left (T.pack (show err)))
  158     Right safePath -> do
  159       let identPath = getSafeKeyPath safePath
  160       (exitCode, _stdout, stderrBs) <-
  161         readProcessNoInput "age-keygen" ["-o", identPath]
  162       case exitCode of
  163         ExitFailure n ->
  164           pure (Left ("age-keygen exited with code " <> T.pack (show n)))
  165         ExitSuccess -> do
  166           let stderrText = TE.decodeUtf8Lenient stderrBs
  167           case parseAgePublicKey stderrText of
  168             Nothing  -> pure (Left "age-keygen: could not parse public key from stderr")
  169             Just pub -> do
  170               setFileMode identPath 0o600
  171               pure (Right ResolvedKey
  172                 { rkRecipient = pub
  173                 , rkIdentity  = T.pack identPath
  174                 , rkKeyType   = "x25519"
  175                 })

  177 -- | Generate (or reuse) a YubiKey-backed age identity.
  178 --
  179 -- @touchRequired@ -> @--touch-policy always|never@.
  180 -- @pinRequired@   -> @--pin-policy   once|never@. Choosing @never@ means
  181 -- decryption needs no PIN (only the token present, plus a touch if required);
  182 -- choosing @once@ (the age-plugin-yubikey default) prompts for the PIN once per
  183 -- decrypt session.
  184 setupYubiKey
  185   :: SealPaths -> Text -> Bool -> Bool -> ChannelCaps
  186   -> IO (Either Text ResolvedKey)
  187 setupYubiKey paths name touchRequired pinRequired caps = do
  188   mPlugin <- findExecutable "age-plugin-yubikey"
  189   case mPlugin of
  190     Nothing ->
  191       pure (Left "age-plugin-yubikey not found on PATH")
  192     Just _ -> do
  193       keysRoot <- ensureKeysRoot (spKeys paths)
  194       rel      <- freshKeyName (spKeys paths) (T.unpack name) ".yubikey.txt"
  195       pathRes  <- mkSafeKeyPath keysRoot rel
  196       case pathRes of
  197         Left err -> pure (Left (T.pack (show err)))
  198         Right safePath -> do
  199           let identPath   = getSafeKeyPath safePath
  200               touchPolicy = if touchRequired then "always" else "never"
  201               pinPolicy   = if pinRequired then "once" else "never"
  202           (exitCode, stdoutBs, _) <-
  203             readProcessNoInput "age-plugin-yubikey"
  204               [ "--generate"
  205               , "--touch-policy", touchPolicy
  206               , "--pin-policy", pinPolicy
  207               ]
  208           let stdoutText = TE.decodeUtf8Lenient stdoutBs
  209           mRecipient <-
  210             if exitCode == ExitSuccess && not (T.null (T.strip stdoutText))
  211               then do
  212                 BS.writeFile identPath (TE.encodeUtf8 stdoutText)
  213                 setFileMode identPath 0o600
  214                 pure (Right (parsePluginRecipient stdoutText))
  215               else do
  216                 -- TTY fallback: instruct user and wait for manual completion.
  217                 ccSend caps
  218                   ("age-plugin-yubikey requires interactive input. Run:\n"
  219                    <> "    age-plugin-yubikey --generate --touch-policy "
  220                    <> T.pack touchPolicy <> " --pin-policy " <> T.pack pinPolicy
  221                    <> " > " <> T.pack identPath)
  222                 _ <- ccPrompt caps (AskPrompt "Press Enter once the command has completed" [])
  223                 rawE <- try @IOException (BS.readFile identPath)
  224                 case rawE of
  225                   Left e ->
  226                     pure (Left ("age-plugin-yubikey: identity file not readable: "
  227                                  <> T.pack (show e)))
  228                   Right raw -> do
  229                     setFileMode identPath 0o600
  230                     pure (Right (parsePluginRecipient (TE.decodeUtf8Lenient raw)))
  231           case mRecipient of
  232             Left ioErr       -> pure (Left ioErr)
  233             Right Nothing    -> pure (Left "age-plugin-yubikey: could not parse recipient line")
  234             Right (Just pub) ->
  235               pure (Right ResolvedKey
  236                 { rkRecipient = pub
  237                 , rkIdentity  = T.pack identPath
  238                 , rkKeyType   = "yubikey"
  239                 })

  287 -- | Parse "Public key: age1..." from age-keygen stderr.
  288 -- age-keygen always emits exactly this line (capital P, no '#' prefix).
  289 parseAgePublicKey :: Text -> Maybe Text
  290 parseAgePublicKey txt =
  291   case filter (T.isPrefixOf "Public key: ") (T.lines txt) of
  292     []       -> Nothing
  293     (line:_) -> Just (T.drop (T.length "Public key: ") line)

  295 -- | The non-interactive core of 'setupYubiKey': no 'ChannelCaps', no prompt,
  296 -- no waiting. The web wizard cannot answer a terminal prompt, so a plugin that
  297 -- wants one is reported as 'KeySetupNeedsTty' together with the exact command
  298 -- the operator should run.
  299 --
  300 -- A key file left behind by that terminal command is adopted on the next
  301 -- attempt: the wizard's Re-check re-sends this backend, the file under the keys
  302 -- directory parses, and setup finishes without touching the token again. An
  303 -- adoptable file is checked before 'findExecutable', so the completion path
  304 -- does not depend on the plugin staying on the gateway's PATH.
  305 --
  306 -- The identity file it writes is @0600@ under the keys directory (mode 0700).
  307 setupYubiKeyCore
  308   :: SealPaths -> Text -> Bool -> Bool -> IO (Either KeySetupFailure ResolvedKey)
  309 setupYubiKeyCore paths name touchRequired pinRequired = do
  310   keysRoot <- ensureKeysRoot (spKeys paths)
  311   adopted <- adoptExistingYubiKey keysRoot (spKeys paths) (T.unpack name)
  312   case adopted of
  313     Just key -> pure (Right key)
  314     Nothing -> do
  315       mPlugin <- findExecutable "age-plugin-yubikey"
  316       case mPlugin of
  317         Nothing -> pure (Left (KeySetupMissing "age-plugin-yubikey"))
  318         Just _ -> do
  319           rel      <- freshKeyName (spKeys paths) (T.unpack name) ".yubikey.txt"
  320           pathRes  <- mkSafeKeyPath keysRoot rel
  321           case pathRes of
  322             Left err -> pure (Left (KeySetupFailed (T.pack (show err))))
  323             Right safePath -> do
  324               let identPath   = getSafeKeyPath safePath
  325                   touchPolicy = if touchRequired then "always" else "never"
  326                   pinPolicy   = if pinRequired then "once" else "never"
  327               (exitCode, stdoutBs, stderrBs) <-
  328                 readProcessNoInput "age-plugin-yubikey"
  329                   [ "--generate"
  330                   , "--touch-policy", touchPolicy
  331                   , "--pin-policy", pinPolicy
  332                   ]
  333               let stdoutText = TE.decodeUtf8Lenient stdoutBs
  334               if exitCode == ExitSuccess && not (T.null (T.strip stdoutText))
  335                 then case parsePluginRecipient stdoutText of
  336                   Nothing -> pure (Left (KeySetupFailed
  337                     ("age-plugin-yubikey: could not parse a recipient line from its output; "
  338                      <> TE.decodeUtf8Lenient stderrBs)))
  339                   Just pub -> do
  340                     BS.writeFile identPath (TE.encodeUtf8 stdoutText)
  341                     setFileMode identPath 0o600
  342                     pure (Right ResolvedKey
  343                       { rkRecipient = pub
  344                       , rkIdentity  = T.pack identPath
  345                       , rkKeyType   = "yubikey"
  346                       })
  347                 else do
  348                   home <- getHomeDirectory
  349                   pure (Left (KeySetupNeedsTty
  350                     (yubiCommandFor home identPath touchPolicy pinPolicy)))

  352 -- | Adopt the identity file the TTY fallback command wrote. Only files named
  353 -- like 'freshKeyName''s output for this backend (@\<name\>.yubikey.txt@,
  354 -- @\<name\>-1.yubikey.txt@, …) and containing a parseable @Recipient:@ line
  355 -- qualify, so unrelated key material is never picked up. Names are sorted, so
  356 -- the choice is deterministic when several attempts left several files.
  357 adoptExistingYubiKey :: KeysRoot -> FilePath -> FilePath -> IO (Maybe ResolvedKey)
  358 adoptExistingYubiKey keysRoot keysDir name = do
  359   listed <- try @IOException (listDirectory keysDir)
  360   let candidates = sort (filter (isYubiCandidate name) (fromRight [] listed))
  361   firstAdoptable candidates
  362   where
  363     firstAdoptable [] = pure Nothing
  364     firstAdoptable (candidate : rest) = do
  365       accepted <- acceptCandidate candidate
  366       case accepted of
  367         Nothing -> firstAdoptable rest
  368         Just safePath -> do
  369           let identPath = getSafeKeyPath safePath
  370           parsed <- readRecipient identPath
  371           case parsed of
  372             Nothing -> firstAdoptable rest
  373             Just pub -> pure (Just ResolvedKey
  374               { rkRecipient = pub
  375               , rkIdentity  = T.pack identPath
  376               , rkKeyType   = "yubikey"
  377               })
  378     -- The shell command writes the file under the operator's umask, usually
  379     -- 0644. A confined-but-loose file is tightened to 0600 and re-validated;
  380     -- anything that escapes the keys root is never touched.
  381     acceptCandidate candidate = do
  382       pathRes <- mkSafeKeyPath keysRoot candidate
  383       case pathRes of
  384         Right safePath -> pure (Just safePath)
  385         Left (PathInsecureMode p) -> do
  386           tightened <- try @IOException (setFileMode p 0o600)
  387           case tightened of
  388             Left _ -> pure Nothing
  389             Right () -> either (const Nothing) Just <$> mkSafeKeyPath keysRoot candidate
  390         Left _ -> pure Nothing
  391     readRecipient path = do
  392       raw <- try @IOException (BS.readFile path)
  393       pure $ either (const Nothing)
  394         (parsePluginRecipient . TE.decodeUtf8Lenient) raw
  395 
  396 isYubiCandidate :: FilePath -> FilePath -> Bool
  397 isYubiCandidate name candidate =
  398   case T.stripSuffix ".yubikey.txt" (T.pack candidate) of
  399     Nothing -> False
  400     Just stemText ->
  401       let stem = T.unpack stemText
  402       in stem == name || case stripPrefix (name <> "-") stem of
  403           Just suffix -> not (null suffix) && all isDigit suffix
  404           Nothing -> False

  406 -- | The command the operator runs by hand when the plugin needs a TTY. The
  407 -- path is rendered in tilde form when it is under the user's home, because
  408 -- that is the form that stays readable in an error body and is still runnable
  409 -- from the operator's own shell.
  410 --
  411 -- A tilde only expands when the shell sees it unquoted, so the default
  412 -- @~/.seal/keys/…@ form is emitted bare. A path that needs quoting (a
  413 -- @SEAL_HOME@ with a space) uses the double-quoted @\"\$HOME/…\"@ form, which
  414 -- keeps expansion while still being one word; a path outside @\$HOME@ is
  415 -- single-quoted absolute.
  416 yubiCommandFor :: FilePath -> FilePath -> String -> String -> Text
  417 yubiCommandFor home identPath touchPolicy pinPolicy =
  418   T.unwords
  419     [ "age-plugin-yubikey --generate"
  420     , "--touch-policy", T.pack touchPolicy
  421     , "--pin-policy", T.pack pinPolicy
  422     , ">", renderShellTarget home identPath
  423     ]
  424 
  425 renderShellTarget :: FilePath -> FilePath -> Text
  426 renderShellTarget home path = case stripPrefix (home <> "/") path of
  427   Just rest
  428     | isShellSafeWord rest -> "~/" <> T.pack rest
  429     | otherwise -> "\"$HOME/" <> T.pack (escapeDouble rest) <> "\""
  430   Nothing -> "'" <> T.pack (escapeSingle path) <> "'"
  431 
  432 isShellSafeWord :: FilePath -> Bool
  433 isShellSafeWord = all (`elem` safeChars)
  434   where
  435     safeChars = ['a' .. 'z'] ++ ['A' .. 'Z'] ++ ['0' .. '9'] ++ "/._-+"
  436 
  437 escapeDouble :: String -> String
  438 escapeDouble = concatMap escapeChar
  439   where
  440     escapeChar c
  441       | c `elem` ['\\', '"', '$', '`'] = ['\\', c]
  442       | otherwise = [c]
  443 
  444 escapeSingle :: String -> String
  445 escapeSingle = concatMap (\c -> if c == '\'' then "'\\''" else [c])

  447 -- | Parse the recipient (public key) from age-plugin-yubikey @--generate@
  448 -- output. The recipient appears on a comment line such as
  449 -- @#    Recipient: age1yubikey1...@. age-plugin-yubikey RIGHT-ALIGNS its
  450 -- comment labels, so the number of spaces between @#@ and the label varies
  451 -- between lines; we therefore strip the leading @#@ and surrounding whitespace
  452 -- before matching the label (case-insensitively per the age plugin spec) and
  453 -- take everything after the first @:@.
  454 parsePluginRecipient :: Text -> Maybe Text
  455 parsePluginRecipient txt =
  456   case filter isRecipientLine (T.lines txt) of
  457     []       -> Nothing
  458     (line:_) ->
  459       let value = T.strip (T.drop 1 (T.dropWhile (/= ':') (labelOf line)))
  460       in if T.null value then Nothing else Just value
  461   where
  462     -- Drop a leading @#@ comment marker and the surrounding alignment spaces,
  463     -- leaving the label, e.g. @"Recipient: age1yubikey1..."@.
  464     labelOf l = T.stripStart (T.dropWhile (== '#') (T.stripStart l))
  465     isRecipientLine l = "recipient:" `T.isPrefixOf` T.toCaseFold (labelOf l)

   48 -- | Resolve @origin@ and @rp_id@ for an already-decided mode.
   49 --
   50 -- Split out of 'resolveAccessControl' so the boot plan can resolve the pair
   51 -- without also deciding the mode and decoding credentials: those two now live
   52 -- in 'Seal.Config.BootPlan.bootPlan', and having a second copy here is how the
   53 -- two would drift apart.
   54 resolveCoordinates
   55   :: Text                       -- ^ bind host
   56   -> Int                        -- ^ bind port
   57   -> Maybe AccessControlFileConfig
   58   -> AccessMode
   59   -> Either Text (Maybe Text, Maybe Text)
   60 resolveCoordinates bindHost bindPort mSection mode = do
   61   origin <- resolveOrigin bindHost bindPort (isWildcardHost bindHost)
   62                           (mSection >>= acfcOrigin) mode
   63   rpId <- resolveRpId origin (mSection >>= acfcRpId)
   64   pure (origin, rpId)
   65 
   66 resolveOrigin
   67   :: Text -> Int -> Bool -> Maybe Text -> AccessMode -> Either Text (Maybe Text)
   68 resolveOrigin bindHost bindPort wildcard mOrigin mode = case mOrigin of
   69   Just raw -> case normalizeOrigin raw of
   70     Just o  -> Right (Just o)
   71     Nothing -> Left ("[access_control] origin \"" <> raw
   72                       <> "\" is not an absolute URL of the form scheme://host[:port]")
   73   Nothing
   74     | wildcard && mode == AccessOpen -> Right Nothing
   75     | wildcard ->
   76         Left "[access_control] origin must be set when the gateway binds a wildcard \
   77              \address and access control is not \"none\""
   78     | otherwise -> Right (Just (canonicalOriginFor bindHost bindPort))
   79 
   80 -- | @http:\/\/localhost:\<port\>@ for a loopback bind, @http:\/\/\<host\>:\<port\>@
   81 -- otherwise.
   82 --
   83 -- Loopback canonicalises to the literal @localhost@ because that is the name
   84 -- every browser treats as a secure context, it is the value the gateway's
   85 -- default origin already uses, and it makes the banner URL and the verified
   86 -- origin the same string. @127.0.0.1@ is deliberately not used: rpIds match by
   87 -- domain suffix, so mixing the two spellings yields credentials that silently
   88 -- fail depending on which URL the operator typed.
   89 canonicalOriginFor :: Text -> Int -> Text
   90 canonicalOriginFor bindHost bindPort
   91   | isLoopbackHost bindHost = "http://localhost:" <> tshow bindPort
   92   | otherwise               = "http://" <> T.toLower bindHost <> ":" <> tshow bindPort
   93 
   94 resolveRpId :: Maybe Text -> Maybe Text -> Either Text (Maybe Text)
   95 resolveRpId mOrigin mRpId = do
   96   resolved <- resolveRpIdUnchecked mOrigin mRpId
   97   case resolved of
   98     Just rp | isIpLiteral rp -> Left (ipRpIdMessage rp)
   99     _ -> Right resolved
  100 
  101 -- | An IPv4 or IPv6 literal, which can never be an RP ID.
  102 --
  103 -- WebAuthn requires the relying-party id to be a valid domain string, and
  104 -- browsers reject an IP origin with @SecurityError@ before any ceremony runs.
  105 -- Catching it here turns a baffling in-browser failure into a boot refusal.
  106 isIpLiteral :: Text -> Bool
  107 isIpLiteral raw =
  108   let h = T.toLower (T.dropWhile (== '[') (T.dropWhileEnd (== ']') raw))
  109   in (not (T.null h) && T.isInfixOf ":" h) || isIpv4 h
  110   where
  111     isIpv4 h = T.count "." h == 3
  112                  && T.all (\c -> isDigit c || c == '.') h
  113                  && not (any T.null (T.splitOn "." h))
  114 
  115 ipRpIdMessage :: Text -> Text
  116 ipRpIdMessage rp = T.intercalate "\n"
  117   [ "[access_control] rp_id resolves to \"" <> rp <> "\", which is an IP address."
  118   , "  A WebAuthn relying-party id must be a domain name. Browsers reject an IP"
  119   , "  with `SecurityError` before any passkey ceremony runs, however valid the"
  120   , "  certificate is. Give this gateway a hostname — a DNS record, a tailnet"
  121   , "  name, or an entry in every client's hosts file — and set origin and rp_id"
  122   , "  to that name (not to the address it resolves to)."
  123   ]
  124 
  125 resolveRpIdUnchecked :: Maybe Text -> Maybe Text -> Either Text (Maybe Text)
  126 resolveRpIdUnchecked mOrigin mRpId = case (mOrigin, mRpId) of
  127   (Nothing, Nothing) -> Right Nothing
  128   (Just o, Nothing)  -> Right (hostOf o)
  129   (Nothing, Just _)  ->
  130     Left "[access_control] rp_id is set but no origin could be resolved"
  131   (Just o, Just raw) ->
  132     case hostOf o of
  133       Nothing -> Left "[access_control] origin has no host"
  134       Just host
  135         | raw == host
  136             || ("." <> host) `T.isSuffixOf` raw
  137             || ("." <> raw) `T.isSuffixOf` host
  138           -> Right (Just raw)
  139         | otherwise ->
  140             Left ("[access_control] rp_id \"" <> raw <> "\" is not a suffix of the "
  141                   <> "origin host \"" <> host <> "\"")
  142   where
  143     hostOf = originHostOf
  144 
  145 -- | The host of an absolute @scheme://host[:port]@ URL, lower-cased.
  146 --
  147 -- Shared with the certificate code, which has to put the origin's name in the
  148 -- SAN list: a certificate that omits it makes every WebAuthn ceremony fail
  149 -- with a message about origins rather than about certificates.
  150 originHostOf :: Text -> Maybe Text
  151 originHostOf o = case T.breakOn "://" o of
  152   (_, rest) | T.null rest -> Nothing
  153   (_, rest) ->
  154     let authority = T.takeWhile (/= '/') (T.drop 3 rest)
  155     in Just (T.toLower (fst (splitHostPort' authority)))
  156 
  157 splitHostPort' :: Text -> (Text, Maybe Text)
  158 splitHostPort' authority =
  159   case T.breakOnEnd ":" authority of
  160     (withColon, port)
  161       | T.null withColon -> (authority, Nothing)
  162       | T.any (== ':') (T.dropEnd 1 withColon) -> (authority, Nothing)
  163       | T.null port -> (authority, Nothing)
  164       | T.all isDigit port -> (T.dropEnd 1 withColon, Just port)
  165       | otherwise -> (authority, Nothing)

  168 -- Credential availability
  169 -- ---------------------------------------------------------------------------
  170 
  171 -- | What the enrolled credentials can actually do on /this/ boot.
  172 --
  173 -- Three values, not a boolean, because the three states call for three
  174 -- different responses and conflating them is the bug this type removes:
  175 -- "some credential differs" never clears, so it mints a grant on every boot
  176 -- and never stops.
  177 data CredentialAvailability
  178   = NoCredentials
  179     -- ^ Nothing enrolled: the bootstrap state. A CLI-minted grant is the only
  180     -- authority (design §5 invariant 3).
  181   | UnusableOnly
  182     -- ^ Credentials exist, but none was created for this rpId — the operator
  183     -- changed @origin@/@rp_id@, or the gateway is reached under a different
  184     -- name. This is @rp_mismatch@. A CLI-minted grant is the only authority:
  185     -- deliberately not the loopback window, since this is the state an
  186     -- attacker who can change @origin@/@rp_id@ would aim to create (§7.2).
  187   | Usable
  188     -- ^ At least one credential can authenticate here.
  189   deriving stock (Eq, Show)
  190 
  191 -- | Classify a credential map against the boot-resolved rpId.
  192 --
  193 -- A 'Nothing' rpId means no origin could be resolved (a wildcard bind with no
  194 -- explicit @origin@), so there is no name to call a credential unusable — every
  195 -- credential counts as usable, which is also what the last-usable-credential
  196 -- deletion guard relies on (design §9.4).
  197 credentialAvailability :: Maybe Text -> Map Text Credential -> CredentialAvailability
  198 credentialAvailability _        creds | Map.null creds = NoCredentials
  199 credentialAvailability Nothing  _                      = Usable
  200 credentialAvailability (Just rp) creds
  201   | any ((== rp) . credRpId) (Map.elems creds) = Usable
  202   | otherwise                                  = UnusableOnly
  203 
  204 -- | @rp_mismatch@: every enrolled credential was created for a different rpId,
  205 -- so none of them can authenticate here.
  206 --
  207 -- Derived, never stored. The stored-boolean version used the quantifier "some
  208 -- credential differs", which is vacuous on an empty store and permanently true
  209 -- once the operator changes @origin@ — so it never cleared, and every boot
  210 -- minted another grant (design §1.6). Deriving it from the map makes "stale"
  211 -- unrepresentable: there is nothing to forget to update.
  212 rpMismatch :: Maybe Text -> Map Text Credential -> Bool
  213 rpMismatch rpId creds = credentialAvailability rpId creds == UnusableOnly

  219 -- | Decode one @[access_control.credentials.\<id\>]@ table. The map key is the
  220 -- base64url credential id.
  221 credentialFromFile :: (Text, CredentialFileConfig) -> Either Text (Text, Credential)
  222 credentialFromFile (rawId, cfc) = do
  223   rawIdBytes <- case B64Url.decodeUnpadded (TE.encodeUtf8 rawId) of
  224     Left e  -> Left (bad <> "the credential id is not base64url (" <> T.pack e <> ")")
  225     Right b
  226       | BS.null b || BS.length b > 1023 -> Left (bad <> "the credential id length is out of range")
  227       | otherwise -> Right b
  228   key <- case B64Url.decodeUnpadded (TE.encodeUtf8 (cfcPublicKey cfc)) of
  229     Left e -> Left (bad <> "public_key is not base64url (" <> T.pack e <> ")")
  230     Right b
  231       | BS.length b == 64 -> Right (CosePublicKey b)
  232       | otherwise -> Left (bad <> "public_key must be 64 bytes (x || y)")
  233   createdAt <- case parseIsoTime (cfcCreatedAt cfc) of
  234     Nothing -> Left (bad <> "created_at is not an ISO-8601 timestamp")
  235     Just t  -> Right t
  236   signCount <- if cfcSignCount cfc < 0
  237                  then Left (bad <> "sign_count is negative")
  238                  else Right (cfcSignCount cfc)
  239   pure (rawId, Credential
  240     { credId = rawIdBytes
  241     , credPublicKey = key
  242     , credSignCount = signCount
  243     , credLabel = cfcLabel cfc
  244     , credCreatedAt = createdAt
  245     , credRpId = cfcRpId cfc
  246     , credAddedVia = cfcAddedVia cfc
  247     })
  248   where
  249     bad = "credential \"" <> rawId <> "\": "
  250 
  251 -- | The inverse, for the writer.
  252 credentialToFile :: Credential -> CredentialFileConfig
  253 credentialToFile cred = CredentialFileConfig
  254   { cfcPublicKey = TE.decodeUtf8 (B64Url.encodeUnpadded (unCose (credPublicKey cred)))
  255   , cfcSignCount = credSignCount cred
  256   , cfcLabel     = credLabel cred
  257   , cfcCreatedAt = formatIsoTime (credCreatedAt cred)
  258   , cfcRpId      = credRpId cred
  259   , cfcAddedVia  = credAddedVia cred
  260   }
  261   where
  262     unCose (CosePublicKey b) = b

  264 -- | Longest accepted label.
  265 --
  266 -- The audit log caps a rendered label at the same length
  267 -- ("Seal.Gateway.Audit.auditLine"), so a stored label always fits there whole.
  268 credentialLabelMaxLength :: Int
  269 credentialLabelMaxLength = 64
  270 
  271 -- | Normalise and validate an operator-chosen credential label.
  272 --
  273 -- The label is display-only — it never takes part in authentication — but it is
  274 -- stored in @credentials.toml@, rendered in the web UI, and echoed into the
  275 -- audit log, so it is normalised once here rather than at each of those three
  276 -- readers: trimmed, non-empty, bounded, and free of control characters (which
  277 -- would corrupt both a list row and a log line). Uniqueness is deliberately
  278 -- not required: two devices may honestly be called the same thing, and the
  279 -- credential id still tells them apart.
  280 mkCredentialLabel :: Text -> Either Text Text
  281 mkCredentialLabel raw
  282   | T.null label = Left "the label must not be empty"
  283   | T.length label > credentialLabelMaxLength =
  284       Left ("the label must be at most " <> tshow credentialLabelMaxLength <> " characters")
  285   | T.any isControl label = Left "the label must not contain control characters"
  286   | otherwise = Right label
  287   where
  288     label = T.strip raw
  289     isControl c = c < ' ' || c == '\DEL'

  295 isLoopbackHost :: Text -> Bool
  296 isLoopbackHost h = T.toLower h `elem` ["127.0.0.1", "::1", "localhost"]
  297 
  298 isWildcardHost :: Text -> Bool
  299 isWildcardHost h = h `elem` ["0.0.0.0", "::", "*"]
  300 
  301 parseIsoTime :: Text -> Maybe UTCTime
  302 parseIsoTime t =
  303   parseTimeM True defaultTimeLocale "%Y-%m-%dT%H:%M:%S%QZ" (T.unpack t)
  304     `orElse` parseTimeM True defaultTimeLocale "%Y-%m-%dT%H:%M:%SZ" (T.unpack t)
  305 
  306 orElse :: Maybe a -> Maybe a -> Maybe a
  307 orElse (Just a) _ = Just a
  308 orElse Nothing  b = b
  309 
  310 formatIsoTime :: UTCTime -> Text
  311 formatIsoTime = T.pack . formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%SZ"
  312 
  313 tshow :: Show a => a -> Text
  314 tshow = T.pack . show

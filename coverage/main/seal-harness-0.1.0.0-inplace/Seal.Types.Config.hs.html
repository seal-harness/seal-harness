    1 {-# LANGUAGE OverloadedStrings #-}
    2 {-# LANGUAGE TemplateHaskell #-}
    3 module Seal.Types.Config
    4   ( Config(..)
    5   , defaultConfig
    6   , pConfig
    7   -- * Lenses
    8   , config_command, config_logLevel
    9   ) where
   10 
   11 import Data.Text (Text)
   12 
   13 import Configuration.Utils
   14 import Control.Lens hiding ((.=))
   15 
   16 import Seal.Types.Command
   17 
   18 -- | The resolved program configuration. The '_config_command' field is
   19 -- CLI-only (it is not part of the 'FromJSON'/'ToJSON' instances) and holds
   20 -- the selected subcommand plus its per-command arguments.
   21 --
   22 -- The gateway's bind host/port live in @~/.seal/security.toml@ (see
   23 -- 'Seal.Gateway.Config'), not here: they decide who can reach the agent and
   24 -- are agent-immutable. The old @[server]@/@--server-host@ scaffold was never
   25 -- wired to the listener and has been removed.
   26 data Config = Config
   27   { _config_command :: !Command
   28   , _config_logLevel :: !Text
   29   } deriving (Eq, Show)
   30 
   31 makeLenses ''Config
   32 
   33 defaultConfig :: Config
   34 defaultConfig = Config
   35   { _config_command = CommandNoOp
   36   , _config_logLevel = "Info"
   37   }
   38 
   39 -- | Config-file 'FromJSON' instance (update-function style). The '_config_command'
   40 -- field is intentionally excluded: commands are not config-file data.
   41 instance FromJSON (Config -> Config) where
   42   parseJSON = withObject "Config" $ \o -> id
   43     <$< config_logLevel ..: "log-level" % o
   44 
   45 -- | Hand-written 'ToJSON' for '--print-config'. Excludes '_config_command'.
   46 instance ToJSON Config where
   47   toJSON c = object
   48     [ "log-level" .= _config_logLevel c
   49     ]
   50 
   51 -- | CLI parser. The command is a flat field parsed with '.::' from an
   52 -- 'hsubparser'; everything else mirrors the config-file structure.
   53 pConfig :: MParser Config
   54 pConfig = id
   55   <$< config_command .:: pCommand
   56   <*< config_logLevel .:: strOption
   57       ( long "log-level"
   58       <> help "Minimum log severity (Debug|Info|Notice|Warning|Error)" )

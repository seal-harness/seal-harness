   44 -- | Opaque session label. No parse invariant on construction history, but a
   45 -- single strict predicate guards every path-join / network boundary.
   46 newtype SessionId = SessionId Text
   47   deriving stock (Eq, Ord, Show)
   48   deriving newtype (ToJSON, FromJSON)
   49 
   50 isValidSessionId :: Text -> Bool
   51 isValidSessionId t =
   52   not (T.null t)
   53     && T.head t /= '.'
   54     && T.all (`elem` chars) t
   55   where
   56     chars = ['A' .. 'Z'] <> ['a' .. 'z'] <> ['0' .. '9'] <> "_-"
   57 
   58 mkSessionId :: Text -> Either Text SessionId
   59 mkSessionId t
   60   | isValidSessionId t = Right (SessionId t)
   61   | otherwise          = Left ("invalid session id: " <> T.pack (show t))
   62 
   63 -- | Total constructor for known-safe system strings (e.g. @"web"@,
   64 -- @"manual"@). Calls 'isValidSessionId' internally and errors on
   65 -- failure — it does NOT bypass validation. Use only for compile-time-
   66 -- known literals; runtime-derived strings should use 'mkSessionId' (the
   67 -- 'Either'-returning variant).
   68 mkSystemSessionId :: Text -> SessionId
   69 mkSystemSessionId t
   70   | isValidSessionId t = SessionId t
   71   | otherwise          = error ("mkSystemSessionId: invalid session id: " <> T.unpack t)

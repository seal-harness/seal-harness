   51 -- | The output-token cap sent to the provider on every completion request.
   52 -- Raised from 4096 to 8192 so tool-using agents (which often emit large
   53 -- tool-call inputs, e.g. embedding file contents into @task@ prompts) have
   54 -- headroom and are less likely to hit 'StopMaxTokens'. This alone does not
   55 -- fix truncation handling — the loop must still handle 'StopMaxTokens'
   56 -- explicitly (see 'go') — but it reduces the frequency.
   57 defaultMaxTokens :: Int
   58 defaultMaxTokens = 8192
   59 
   60 -- | Hard timeout for a single provider streaming call, in microseconds.
   61 -- 5 minutes gives generous headroom for large contexts on remote models
   62 -- (the Ollama HTTP client has its own 5-minute response timeout; this is
   63 -- a safety net for a stalled connection that never closes). If the stream
   64 -- does not complete within this window, the turn aborts with a provider
   65 -- error so the session does not hang forever in "thinking" with no log
   66 -- output. The error is logged to seal.log and surfaced to the user as a
   67 -- normal provider error (not an exception), so the bracket cleanup in
   68 -- the send path still fires the idle broadcast.
   69 streamTimeoutUs :: Int
   70 streamTimeoutUs = 5 * 60 * 1_000_000

   72 -- | The synthetic user message appended after a 'StopMaxTokens' truncation
   73 -- with no tool calls, asking the model to continue exactly where it left off
   74 -- without repeating prior text. Mirrors Hermes' @_get_continuation_prompt@.
   75 continuationPrompt :: Text
   76 continuationPrompt =
   77   "[System: Your previous response was truncated by the output length \
   78   \limit. Continue exactly where you left off. Do not restart or repeat \
   79   \prior text. Finish the answer directly.]"

   81 -- | Build the request @erMeta@: a @channel@ key (always present, sourced
   82 -- from 'aeChannel' so every channel — including web/CLI with no
   83 -- 'MessageSource' — is attributed) plus a @conversationId@ key when
   84 -- 'aeMessageSource' is present (carrying the server-derived conversation
   85 -- id for channels that have one). This is the single place channel
   86 -- provenance is stamped into the transcript, so every turn is attributed
   87 -- regardless of the source channel.
   88 requestMeta :: Text -> Maybe MessageSource -> Map.Map Text Value
   89 requestMeta channel mSrc =
   90   Map.fromList
   91     ([ ("channel", String channel)
   92      ] <> conversationIdField)
   93   where
   94     conversationIdField = case mSrc of
   95       Nothing -> []
   96       Just ms -> [("conversationId", String (conversationIdText (msConversationId ms)))]

   98 runTurn :: AgentEnv -> Text -> App ()
   99 runTurn env userText = do
  100   -- Clear the per-turn abort flag at turn entry (design: clearAbort fires
  101   -- once at runTurn entry, before any tool call; a mid-turn abort keeps the
  102   -- flag set until the next turn begins).
  103   liftIO (clearAbort (aeAbortFlag env))
  104   -- Load the prior conversation from disk so the model sees the full history
  105   -- (not just this turn's new message). The two-file writer's diff-based
  106   -- appender requires the incoming message list to be a prefix-extension of
  107   -- the on-disk conversation; without the prior messages, the diff falls back
  108   -- to re-appending the whole list every iteration, corrupting
  109   -- @conversation.jsonl@ with duplicate user + assistant lines.
  110   prior <- liftIO (tfwReadConversation (aeTranscript env))
  111   let userMsg = textMsg User userText
  112       turn0   = prior <> [userMsg]
  113   -- Record the initial user message as a Request entry. The envelope delta
  114   -- carries the full envelope in effect for this turn (model / system / tools
  115   -- / maxTokens), so reconstruction can rebuild the exact CompletionRequest.
  116   -- The request's @erMeta@ carries the channel + conversation id when
  117   -- 'aeMessageSource' is present (Signal), so the transcript records which
  118   -- channel + conversation this turn served. @erConvLen@ is the full
  119   -- conversation length in effect at this request (prior + new user message),
  120   -- so reconstruction can slice the right prefix from @conversation.jsonl@.
  121   liftIO $ do
  122     now <- getCurrentTime
  123     let env0 = EnvelopeDelta
  124           { edModel = Just (aeModel env)
  125           , edSystem = Just (aeSystem env)
  126           , edTools = Just (registryToolDefs' (aeOnDemandSchemas env) (aeRegistry env))
  127           , edToolChoice = Just ToolAuto
  128           , edMaxTokens = Just defaultMaxTokens
  129           }
  130         entry = EntryRecord
  131           { erId = ""
  132           , erTimestamp = now
  133           , erKind = EKRequest
  134           , erConvLen = length turn0
  135           , erEnvelope = Just env0
  136           , erUsage = Nothing
  137           , erStop = Nothing
  138           , erDurationMs = Nothing
  139           , erHarness = Nothing
  140           , erCorrelation = Nothing
  141           , erMeta = requestMeta (aeChannel env) (aeMessageSource env)
  142           }
  143     -- When a caller wants to react to the user message being durable
  144     -- (e.g. the /bg path broadcasts a lists snapshot so the sidebar shows
  145     -- the session name immediately), record it with tfwRecordAndAck
  146     -- (synchronously fsync'd) and run the hook. Otherwise keep the async
  147     -- write (no fsync latency at turn start). In the fsync path, also fire
  148     -- the per-entry broadcast hook (aeOnEntry) so the web frontend sees the
  149     -- user's message appear live — without this, the user message is
  150     -- written but never broadcast, so the transcript area stays blank
  151     -- until the first response entry lands (after the full LLM stream
  152     -- completes). The async path skips aeOnEntry because the entry may
  153     -- not be on disk yet (the read would return stale data).
  154     case aeOnUserMessage env of
  155       Just after -> do
  156         tfwRecordAndAck (aeTranscript env) (TwoFileWrite turn0 entry)
  157         liftIO after
  158         liftIO (aeOnEntry env)
  159       Nothing ->
  160         tfwRecordAsync (aeTranscript env) (TwoFileWrite turn0 entry)
  161   go (aeMaxTurns env) 0 turn0
  162   where
  163     -- | Fan out the final user-visible text to every chat channel subscribed
  164     -- to the session ('aeOnStop'). This does NOT call 'ccSend' (the arrival
  165     -- channel); the stop branches handle that separately, guarded by
  166     -- 'alreadySentText' so the arrival channel doesn't get double-delivered
  167     -- when text was already streamed.
  168     notifyStop :: Text -> IO ()
  169     notifyStop text =
  170       case aeOnStop env of
  171         Just fanout -> fanout text
  172         Nothing     -> pure ()
  173     go :: Int -> Int -> [Message] -> App ()
  174     go 0 _ msgs = liftIO $ do
  175       logMaxTurns (aeLogPath env)
  176       let stopMsg = "(stopped: reached the " <> T.pack (show (aeMaxTurns env))
  177             <> "-turn limit for this message. Ask again to continue, "
  178             <> "or raise `max_turns` in config.toml.)"
  179           assistantMsg = Message Assistant [CbText stopMsg]
  180           conv = msgs <> [assistantMsg]
  181       now <- getCurrentTime
  182       let entry = EntryRecord
  183             { erId = ""
  184             , erTimestamp = now
  185             , erKind = EKResponse
  186             , erConvLen = length conv
  187             , erEnvelope = Nothing
  188             , erUsage = Nothing
  189             , erStop = Nothing
  190             , erDurationMs = Nothing
  191             , erHarness = Nothing
  192             , erCorrelation = Nothing
  193             , erMeta = Map.empty
  194             }
  195       tfwRecordAndAck (aeTranscript env) (TwoFileWrite conv entry)
  196       aeOnEntry env
  197       ccSend (aeCaps env) stopMsg
  198       notifyStop stopMsg
  199     go n lenContinue msgs = do
  200       liftIO (logTurnStart (aeLogPath env) n)
  201       tStart <- liftIO getCurrentTime
  202       let req = CompletionRequest
  203                   { crModel = aeModel env
  204                   , crSystem = aeSystem env
  205                   , crMessages = msgs
  206                   , crTools = registryToolDefs' (aeOnDemandSchemas env) (aeRegistry env)
  207                   , crToolChoice = ToolAuto
  208                   , crMaxTokens = defaultMaxTokens
  209                   }
  210       liftIO (appendDebugRequest (aeDebugRequestsPath env) req)
  211       -- Stream the completion, aggregating events into ContentBlocks. Text
  212       -- deltas are broadcast live via ccSend (CLI/Telegram/Signal) so the
  213       -- user sees incremental output. This replaces the blocking
  214       -- providerCompleteWithRetry call — the first token arrives in ~2s
  215       -- instead of waiting for the entire response (which could take 74+s
  216       -- for large contexts on remote models, hitting the 90s timeout).
  217       -- Events are collected in an IORef so we can reconstruct the final
  218       -- ContentBlocks for the transcript entry after the stream completes.
  219       collectedRef <- liftIO (newIORef ([] :: [StreamEvent]))
  220       let streamSends = ccStreaming (aeCaps env)
  221       let isTransient = isTransientError
  222       -- Wrap the stream in a hard timeout so a stalled provider
  223       -- connection (TCP open but no data) cannot hang the session
  224       -- forever in "thinking". On timeout, the turn surfaces a provider
  225       -- error — logged to seal.log + the transcript — and the bracket@.
  226       -- cleanup in the send path fires the idle broadcast.
  227       -- Race the provider stream against the abort-poll so a /stop
  228       -- (web POST /api/sessions/:id/stop, CLI Ctrl+C, channel /stop)
  229       -- interrupts the in-flight LLM call, not just tool calls. When
  230       -- abort wins the race, the stream is cancelled and a "(stopped)"
  231       -- message is recorded — the user can immediately send a new
  232       -- prompt. The abort poll interval matches the tool-call poll
  233       -- (100ms, from ttcAbortPollMicros).
  234       let abortPoll = do
  235             aborted <- isAborted (aeAbortFlag env)
  236             if aborted then pure () else do
  237               threadDelay 100_000
  238               abortPoll
  239       mResult <- liftIO (timeout streamTimeoutUs $
  240         race abortPoll
  241              (providerStreamWithRetry (aeProvider env) isTransient req (\ev -> do
  242                case ev of
  243                  StreamTextChunk delta
  244                    | streamSends -> ccSend (aeCaps env) delta
  245                  _ -> pure ()
  246                modifyIORef' collectedRef (++ [ev])
  247                pure True)))
  248       let estream = case mResult of
  249             Nothing -> Left "stream timed out (no data for 5 minutes)"
  250             Just (Left ()) -> Left "(stopped)"
  251             Just (Right r) -> r
  252       case estream of
  253         Left err -> liftIO $ do
  254           -- Distinguish a user-initiated stop from a genuine provider
  255           -- error. "(stopped)" is the abort-signal message; everything
  256           -- else is a provider failure.
  257           let isStop = err == "(stopped)"
  258           unless isStop $ logProviderError (aeLogPath env) err
  259           let errMsg = if isStop then err else "provider error: " <> err
  260               assistantMsg = Message Assistant [CbText errMsg]
  261               conv = msgs <> [assistantMsg]
  262           now <- getCurrentTime
  263           let entry = EntryRecord
  264                 { erId = ""
  265                 , erTimestamp = now
  266                 , erKind = EKResponse
  267                 , erConvLen = length conv
  268                 , erEnvelope = Nothing
  269                 , erUsage = Nothing
  270                 , erStop = Nothing
  271                 , erDurationMs = Nothing
  272                 , erHarness = Nothing
  273                 , erCorrelation = Nothing
  274                 , erMeta = Map.empty
  275                 }
  276           tfwRecordAndAck (aeTranscript env) (TwoFileWrite conv entry)
  277           aeOnEntry env
  278           ccSend (aeCaps env) errMsg
  279           notifyStop errMsg
  280         Right outcome -> do
  281           events <- liftIO (readIORef collectedRef)
  282           let resp = aggregateStreamEvents events outcome
  283           handleResponse env tStart n lenContinue msgs resp streamSends
  284 
  285     -- | Handle a completed provider response: record it to the transcript,
  286     -- then branch on tool calls vs final text vs StopMaxTokens. When
  287     -- @alreadySentText@ is True, the text was already streamed live via
  288     -- ccSend (streaming path) — don't re-send the full text, only send the
  289     -- prefix / truncation notice if applicable.
  290     handleResponse :: AgentEnv -> UTCTime -> Int -> Int -> [Message] -> CompletionResponse -> Bool -> App ()
  291     handleResponse env' tStart n lenContinue msgs resp _alreadySentText = do
  292       -- Record the provider response.
  293       liftIO $ do
  294         now <- getCurrentTime
  295         let assistantMsg = Message Assistant (rsContent resp)
  296             conv = msgs <> [assistantMsg]
  297             entry = EntryRecord
  298               { erId = ""
  299               , erTimestamp = now
  300               , erKind = EKResponse
  301               , erConvLen = length conv
  302               , erEnvelope = Nothing
  303               , erUsage = Just (rsUsage resp)
  304               , erStop = Just (rsStop resp)
  305               , erDurationMs = Nothing
  306               , erHarness = Nothing
  307               , erCorrelation = Nothing
  308               , erMeta = Map.empty
  309               }
  310         tfwRecordAndAck (aeTranscript env') (TwoFileWrite conv entry)
  311         aeOnEntry env'
  312       let toolUses = [b | b@CbToolUse{} <- rsContent resp]
  313       if null toolUses
  314         then
  315           if rsStop resp == StopMaxTokens && lenContinue < maxLengthContinueRetries
  316             then do
  317               liftIO (logTruncation (aeLogPath env') (lenContinue + 1))
  318               let assistantMsg = Message Assistant (rsContent resp)
  319                   contMsg = textMsg User continuationPrompt
  320                   conv2 = msgs <> [assistantMsg, contMsg]
  321               liftIO $ do
  322                 now2 <- getCurrentTime
  323                 let entry2 = EntryRecord
  324                       { erId = ""
  325                       , erTimestamp = now2
  326                       , erKind = EKRequest
  327                       , erConvLen = length conv2
  328                       , erEnvelope = Nothing
  329                       , erUsage = Nothing
  330                       , erStop = Nothing
  331                       , erDurationMs = Nothing
  332                       , erHarness = Nothing
  333                       , erCorrelation = Nothing
  334                       , erMeta = Map.empty
  335                       }
  336                 tfwRecordAndAck (aeTranscript env') (TwoFileWrite conv2 entry2)
  337                 aeOnEntry env'
  338               tEnd <- liftIO getCurrentTime
  339               liftIO (logTurnEnd (aeLogPath env') (n - 1) (msDiff tStart tEnd))
  340               go (n - 1) (lenContinue + 1) conv2
  341             else liftIO $ do
  342               tEnd <- getCurrentTime
  343               logTurnEnd (aeLogPath env') (n - 1) (msDiff tStart tEnd)
  344               let texts = [t | CbText t <- rsContent resp]
  345                   ModelId m = aeModel env'
  346                   prefix = aeProviderLabel env' <> "/" <> m <> "> "
  347               if rsStop resp == StopMaxTokens
  348                 then do
  349                   logTruncationGiveUp (aeLogPath env')
  350                   let notice = "(stopped: response truncated at the "
  351                         <> T.pack (show defaultMaxTokens)
  352                         <> "-token output limit after "
  353                         <> T.pack (show maxLengthContinueRetries)
  354                         <> " continuation attempts. Send any message to \
  355                         \continue.)"
  356                       combined = case texts of
  357                         [] -> notice
  358                         ts -> T.intercalate "\n" ts <> "\n\n" <> notice
  359                   -- The truncation notice is always sent to the arrival
  360                   -- channel (it's new information — the user needs to know
  361                   -- the turn was truncated), even if the partial text was
  362                   -- already streamed.
  363                   ccSend (aeCaps env') (prefix <> combined)
  364                   notifyStop (prefix <> combined)
  365                 else do
  366                   -- Normal final answer: text was already streamed live;
  367                   -- don't re-send via ccSend (would double-deliver). Only
  368                   -- fan out to other chat channels via notifyStop.
  369                   -- For non-streaming channels (Telegram, Signal — chStreaming
  370                   -- = False), alreadySentText is False, so ccSend would fire.
  371                   -- But notifyStop (replyFanout) already sends to ALL
  372                   -- subscribed channels INCLUDING the arrival channel. So
  373                   -- ccSend would double-deliver. Fix: skip ccSend entirely;
  374                   -- replyFanout handles all delivery (it sends to every
  375                   -- subscribed channel, which includes the arrival one).
  376                   notifyStop (prefix <> T.intercalate "\n" texts)
  377         else do
  378           results <- mapM dispatchOne toolUses
  379           -- Abort guard: if a tool call was aborted mid-flight (the user
  380           -- clicked stop, set the abort flag), terminate the turn
  381           -- immediately instead of feeding the aborted tool results back
  382           -- to the LLM and continuing. Without this guard the loop would
  383           -- recurse into `go`, the LLM would emit a follow-up response,
  384           -- and the tab would stay "thinking" until that response
  385           -- completes — the user-visible stop button would appear to do
  386           -- nothing. Mirrors the stream-abort `Left "(stopped)"` path
  387           -- above so the bracket cleanup fires the idle broadcast.
  388           aborted <- liftIO (isAborted (aeAbortFlag env'))
  389           if aborted
  390             then liftIO $ do
  391               let stopMsg = "(stopped)"
  392                   assistantMsg = Message Assistant (rsContent resp)
  393                   resultMsg = Message User results
  394                   conv = msgs <> [assistantMsg, resultMsg]
  395                           <> [Message Assistant [CbText stopMsg]]
  396               now <- getCurrentTime
  397               let entry = EntryRecord
  398                     { erId = ""
  399                     , erTimestamp = now
  400                     , erKind = EKResponse
  401                     , erConvLen = length conv
  402                     , erEnvelope = Nothing
  403                     , erUsage = Nothing
  404                     , erStop = Nothing
  405                     , erDurationMs = Nothing
  406                     , erHarness = Nothing
  407                     , erCorrelation = Nothing
  408                     , erMeta = Map.empty
  409                     }
  410               tfwRecordAndAck (aeTranscript env') (TwoFileWrite conv entry)
  411               aeOnEntry env'
  412               ccSend (aeCaps env') stopMsg
  413               notifyStop stopMsg
  414             else do
  415               let assistantMsg = Message Assistant (rsContent resp)
  416                   resultMsg = Message User results
  417               liftIO $ do
  418                 now2 <- getCurrentTime
  419                 let conv2 = msgs <> [assistantMsg, resultMsg]
  420                     entry2 = EntryRecord
  421                       { erId = ""
  422                       , erTimestamp = now2
  423                       , erKind = EKRequest
  424                       , erConvLen = length conv2
  425                       , erEnvelope = Nothing
  426                       , erUsage = Nothing
  427                       , erStop = Nothing
  428                       , erDurationMs = Nothing
  429                       , erHarness = Nothing
  430                       , erCorrelation = Nothing
  431                       , erMeta = Map.empty
  432                       }
  433                 tfwRecordAndAck (aeTranscript env') (TwoFileWrite conv2 entry2)
  434                 aeOnEntry env'
  435               tEnd <- liftIO getCurrentTime
  436               liftIO (logTurnEnd (aeLogPath env') (n - 1) (msDiff tStart tEnd))
  437               go (n - 1) 0 (msgs <> [assistantMsg, resultMsg])

  439     dispatchOne :: ContentBlock -> App ContentBlock
  440     dispatchOne (CbToolUse tcid name input) = do
  441       let mOp = lookupOp (aeRegistry env) name
  442       mConfirmed <- checkConfirmation name mOp input
  443       res <- case mConfirmed of
  444         Left denyMsg -> pure (Left (Denied denyMsg))
  445         Right () -> dispatch (aeRegistry env) (aeTranscript env) (aeBackend env) (aeUIOEnv env) (aeToolTimeout env) (aeAbortFlag env) name input
  446       pure $ case res of
  447         Left e -> CbToolResult tcid [TrpText (T.pack (show e))] True
  448         Right r -> CbToolResult tcid (orParts r) (orIsError r)
  449       where
  450         -- | The human-confirmation gate. 'Full' autonomy (@--yolo@) bypasses
  451         -- the gate. 'Supervised' checks the approval cache first; on a miss,
  452         -- prompts the human via 'ccPrompt' (the reply text is the approval
  453         -- scope's wire form: @"once"@, @"for_session"@, @"always"@, or
  454         -- @"rejected"@ on the web; on the CLI, @"y"/@"yes"@ → 'ScopeOnce',
  455         -- anything else → 'ScopeRejected'). On a hit, records the approval
  456         -- scope in the transcript, then proceeds (or denies for 'ScopeRejected').
  457         -- Trusted opcodes skip the gate.
  458         checkConfirmation :: OpName -> Maybe Opcode -> Value -> App (Either Text ())
  459         checkConfirmation opName' mOp input' =
  460           case aeAutonomy env of
  461             Full -> pure (Right ())
  462             _ ->
  463               case mOp of
  464                 Nothing -> pure (Right ())
  465                 Just op ->
  466                   case opTrust op of
  467                     Untrusted -> do
  468                       mCached <- liftIO (checkApproval (aeApprovals env) (aeSession env) opName')
  469                       case mCached of
  470                         Just ScopeRejected -> do
  471                           recordApprovalEvidence opName' input' ScopeRejected
  472                           pure (Left ("SHELL_EXEC denied by human" <> suffixFor opName'))
  473                         Just ScopeForSession -> do
  474                           recordApprovalEvidence opName' input' ScopeForSession
  475                           pure (Right ())
  476                         Just ScopeAlways -> do
  477                           recordApprovalEvidence opName' input' ScopeAlways
  478                           pure (Right ())
  479                         Just ScopeOnce -> do
  480                           -- ScopeOnce is never cached (recordApproval is a
  481                           -- no-op for it), so this branch is unreachable.
  482                           -- If it somehow appears, treat it as a cache miss.
  483                           recordApprovalEvidence opName' input' ScopeOnce
  484                           pure (Right ())
  485                         Nothing -> do
  486                           let prompt = buildConfirmationPrompt opName' input'
  487                           -- The confirmation gate's prompt (Allow <NAME>
  488                           -- <JSON>? [y/N]) is NOT model text, so it must be
  489                           -- sent to the channel. ccPrompt no longer sends
  490                           -- (the model already streamed ASK_HUMAN text), so
  491                           -- we send here.
  492                           liftIO (ccSend (aeCaps env) prompt)
  493                           reply <- liftIO (ccPrompt (aeCaps env) (AskPrompt prompt []))
  494                           let scope = parseScopeReply reply
  495                           liftIO (recordApproval (aeApprovals env) (aeSession env) opName' scope)
  496                           recordApprovalEvidence opName' input' scope
  497                           case scope of
  498                             ScopeRejected -> pure (Left ("SHELL_EXEC denied by human" <> suffixFor opName'))
  499                             _ -> pure (Right ())
  500                     _ -> pure (Right ())
  501         -- | Parse the 'ccPrompt' reply into an 'ApprovalScope'. The web
  502         -- returns the scope's wire form (@once@, @for_session@, @always@,
  503         -- @rejected@). The CLI returns free text (@y@/@yes@ → 'ScopeOnce',
  504         -- anything else → 'ScopeRejected').
  505         parseScopeReply :: Text -> ApprovalScope
  506         parseScopeReply reply =
  507           case parseApprovalScope reply of
  508             Right scope -> scope
  509             Left _ ->
  510               let lower = T.toLower (T.strip reply) in
  511               if lower == "y" || lower == "yes"
  512                 then ScopeOnce
  513                 else ScopeRejected
  514         buildConfirmationPrompt :: OpName -> Value -> Text
  515         buildConfirmationPrompt (OpName n) inp =
  516           "Allow " <> n <> " " <> T.pack (BLC.unpack (A.encode inp)) <> "? [y/N] "
  517         suffixFor :: OpName -> Text
  518         suffixFor (OpName n) = " (" <> n <> ")"
  519         -- | Record an EKHarness entry in the transcript carrying the opcode
  520         -- name, input, and approval scope as evidence of the human's
  521         -- decision. This is separate from the dispatcher's own entry (which
  522         -- records the invocation, not the approval).
  523         recordApprovalEvidence :: OpName -> Value -> ApprovalScope -> App ()
  524         recordApprovalEvidence opName' input' scope = liftIO $ do
  525           now <- getCurrentTime
  526           let entry = EntryRecord
  527                 { erId = ""
  528                 , erTimestamp = now
  529                 , erKind = EKHarness
  530                 , erConvLen = 0
  531                 , erEnvelope = Nothing
  532                 , erUsage = Nothing
  533                 , erStop = Nothing

  547 -- | Is a provider error text transient (worth retrying)? Transport failures
  548 -- (the common "Ollama not running" case), rate limits (HTTP 429), and server
  549 -- errors (HTTP 5xx) are transient — the daemon might come back, a rate limit
  550 -- might clear, a 5xx might be a transient upstream fault. Auth errors (401),
  551 -- bad requests (400), and decode failures are permanent — retrying wastes
  552 -- time and delays surfacing the real problem to the user.
  553 isTransientError :: Text -> Bool
  554 isTransientError err =
  555   any (`T.isInfixOf` err) transientMarkers
  556   where
  557     transientMarkers =
  558       [ "could not reach"           -- Ollama transport failure
  559       , "connection or transport error"  -- Anthropic transport failure
  560       , "HTTP 429"                  -- rate limit (Anthropic + Ollama)
  561       , "HTTP 500"                  -- server error
  562       , "HTTP 502"                  -- bad gateway
  563       , "HTTP 503"                  -- service unavailable
  564       , "HTTP 504"                  -- gateway timeout
  565       ]

  585 -- | Aggregate a list of 'StreamEvent's (collected during streaming) into a
  586 -- 'CompletionResponse' for the transcript. Text chunks are concatenated into
  587 -- a single 'CbText' block (with stray tool-call XML stripped — see
  588 -- 'stripToolCallXml'); tool-call start/end pairs become 'CbToolUse'
  589 -- blocks. The stop reason + usage come from the 'StreamOutcome'.
  590 aggregateStreamEvents :: [StreamEvent] -> StreamOutcome -> CompletionResponse
  591 aggregateStreamEvents events outcome =
  592   CompletionResponse blocks (soStop outcome) (soUsage outcome)
  593   where
  594     textChunks = [t | StreamTextChunk t <- events]
  595     textBlocks = [CbText (stripToolCallXml (T.intercalate "" textChunks))
  596                  | not (null textChunks)]
  597     toolBlocks = [CbToolUse tcid name args
  598                 | StreamToolEnd tcid name args <- events]
  599     blocks = textBlocks <> toolBlocks

  601 -- | Upper bound on how far back a truncation-tail scan looks (chars). The
  602 -- tail of a StopMaxTokens response can contain a partial tool-call XML
  603 -- opening tag (e.g. @\<invoke name="FILE_RE@); if the continuation prepends
  604 -- the last chunk of the prior text (which carries that partial tag) to the
  605 -- new stream's first chunk, the strip below would otherwise have to rescan
  606 -- the entire continuation text. Cap the re-scan window so stripping stays
  607 -- O(1) amortized per continuation.
  608 maxPartialTagScan :: Int
  609 maxPartialTagScan = 512

  611 -- | Strip leaked tool-call XML from streamed assistant text.
  612 --
  613 -- Some provider stacks (observed with GLM via Ollama's cloud endpoint, in
  614 -- sessions 20260831-162314-870 / 20260831-162806-923) occasionally emit a
  615 -- tool call as XML in the text stream instead of a structured tool_calls
  616 -- chunk — e.g. @\<invoke name="FILE_READ">\<^arg_key>path\<^\/arg_key>...\<\/invoke>@.
  617 -- If that text lands in the transcript as @CbText@ it (a) confuses the
  618 -- model on subsequent turns (it reads its own half-parsed tool call as
  619 -- prose) and (b) renders as garbage in the frontend. When the provider
  620 -- ALSO surfaces the structured @tool_calls@ for the same call (the normal
  621 -- case), the XML is redundant; when it does not, the call is unactionable
  622 -- markup either way. Strip it, but leave normal prose (including literal
  623 -- XML in fenced code contexts that doesn't match the tool-call shape) alone.
  624 --
  625 -- Stripped shapes (case-insensitive, allow whitespace around tags):
  626 --
  627 --   * a complete @\<invoke name="..."\>...\<\/invoke>@ block
  628 --   * orphan @\<\/invoke>@ / @\<function_calls>@ / @\<\/function_calls>@ /
  629 --     @\<tool_call>@ / @\<\/tool_call>@ tags
  630 stripToolCallXml :: Text -> Text
  631 stripToolCallXml t0 =
  632   let -- 0. Remove complete blocks FIRST: an intact closer at the tail
  633       -- must not be mistaken for a truncation cut by the partial-tail
  634       -- scan below (which would otherwise eat real content).
  635       t2 = dropBlocks "<invoke" "</invoke>"
  636                   (dropToolCallBlock t0)
  637       -- 1. Drop a trailing partial opening tag (a StopMaxTokens cut can
  638       --    leave e.g. an unfinished '<invoke na' at the end of the text).
  639       t1b = dropTrailingPartialTag t2
  640       -- 2. Remove orphan tags (unterminated blocks whose opener was cut,
  641       --    or a stray close tag the provider emitted).
  642       t3 = T.replace oInvokeClose "" t1b
  643       t4 = T.replace oFcOpen "" t3
  644       t5 = T.replace oFcClose "" t4
  645       t6 = T.replace oToolOpen "" t5
  646       t7 = T.replace oToolClose "" t6
  647       -- 3. Remove orphan param tags (leaked args wrap key/value in
  648       --    \<arg_key\>/\<arg_value\> tags; when the surrounding
  649       --    \<invoke\> block was already removed these are strays).
  650       t8 = T.replace paramKeyOpen "" t7
  651       t9 = T.replace paramKeyClose "" t8
  652       t10 = T.replace paramValOpen "" t9
  653       t11 = T.replace paramValClose "" t10
  654       -- 5. Remove a trailing unterminated opener (its closer was never
  655       --    emitted, e.g. a truncation cut mid-block): if an opener
  656       --    appears in the last maxPartialTagScan chars and no closer
  657       --    follows it, drop from the opener to the end of the text.
  658       t12 = dropTrailingUnterminated t11
  659   in t12
  660   where
  661     oInvokeClose = "</invoke>"
  662     oFcOpen      = "<function_calls>"
  663     oFcClose     = "</function_calls>"
  664     oToolOpen    = "<tool_call>"
  665     oToolClose   = "</tool_call>"
  666     paramKeyOpen  = "<arg_key>"
  667     paramKeyClose = "</arg_key>"
  668     paramValOpen  = "<arg_value>"
  669     paramValClose = "</arg_value>"
  670 
  671 -- | Remove a complete @\<tool_call>...\<tool_call>\/tool_call\>@ span. The opener
  672 -- carries no attributes, so this is a plain two-marker scan (unlike
  673 -- 'dropBlocks', whose attribute handling misfires on JSON bodies that
  674 -- contain no '@<@' but do contain a close tag later in the text).
  675 dropToolCallBlock :: Text -> Text
  676 dropToolCallBlock = go
  677   where
  678     open = "<tool_call>"
  679     close = "</tool_call>"
  680     go acc = case T.breakOn open acc of
  681       (before, rest)
  682         | T.null rest -> acc
  683         | otherwise   ->
  684             let afterOpen = T.drop (T.length open) rest
  685             in case T.breakOn close afterOpen of
  686                  (_, afterCloseRest)
  687                    | T.null afterCloseRest -> acc  -- no close: leave
  688                  (_, afterClose)           -> go (before <> T.drop (T.length close) afterClose)
  689 
  690 -- | Remove every complete @open ... close@ span (non-greedy: up to the
  691 -- FIRST closing tag), including the tags themselves. Unterminated blocks
  692 -- are left alone (the orphan-tag pass handles their remnants).
  693 dropBlocks :: Text -> Text -> Text -> Text
  694 dropBlocks open close = go
  695   where
  696     go acc = case breakOnTag acc of
  697       Nothing -> acc
  698       Just (before, afterOpenTag) ->
  699         case T.breakOn close afterOpenTag of
  700           (_, afterCloseRest)
  701             | T.null afterCloseRest -> acc  -- no close: leave as-is
  702           (_, afterClose)           -> go (before <> T.drop (T.length close) afterClose)
  703     -- Find the open tag (with any attributes up to its closing '>') and
  704     -- return (text before it, text after the tag's '>').
  705     breakOnTag acc = case T.breakOn open acc of
  706       (before, rest)
  707         | T.null rest -> Nothing
  708         | otherwise   ->
  709             let afterName = T.drop (T.length open) rest
  710             in case T.breakOn ">" afterName of
  711                  (attrs, gtRest)
  712                    -- attrs is everything up to the first '>' — treat it
  713                    -- as the tag's attribute list only when it contains
  714                    -- no other '<' (a nested '<' means this '>' closes a
  715                    -- LATER tag, not this opener; the match was a false
  716                    -- positive on e.g. '<tool_call>{...json...}'.
  717                    | T.isInfixOf "<" attrs -> Nothing
  718                    | T.null gtRest          -> Nothing  -- no '>' at all
  719                    | otherwise              -> Just (before, T.drop 1 gtRest)
  720 
  721 -- | If a tool-call opener (@<invoke@, @<function_calls@, @<tool_call@)
  722 -- appears within the last 'maxPartialTagScan' chars and NO closer follows
  723 -- it, the tail is an unterminated leaked tool call — drop from the opener
  724 -- to the end of the text. Catches blocks the provider broke across a
  725 -- truncation boundary (opener present, close never emitted).
  726 dropTrailingUnterminated :: Text -> Text
  727 dropTrailingUnterminated t =
  728   case openIdx of
  729     Nothing -> t
  730     Just i ->
  731       let afterOpen = T.drop i t
  732       in if any (`T.isInfixOf` afterOpen) closers then t else T.take i t
  733   where
  734     closers = ["</invoke>", "</function_calls>", "</tool_call>"]
  735     openers = ["invoke", "function_calls", "tool_call"]
  736     -- the LAST opener occurrence inside the trailing window (global
  737     -- index, with a tag-ish boundary after the name — prose like
  738     -- "<invoke-ish" must not match)
  739     openIdx =
  740       let window = T.takeEnd maxPartialTagScan t
  741           base   = T.length t - T.length window
  742           hits   = [ (base + i, o)
  743                    | o <- openers
  744                    , i <- reverse [0 .. T.length window - (T.length o + 1)]
  745                    , T.index window i == '<'
  746                    , Just rest <- [T.stripPrefix o (T.drop (i + 1) window)]
  747                    , tagBoundary rest ]
  748       in case hits of
  749            []           -> Nothing
  750            ((i, _) : _) -> Just i
  751     tagBoundary rest = case T.uncons rest of
  752       Nothing     -> True
  753       Just (c, _) -> c == ' ' || c == '>' || c == '/' || c == '\t' || c == '\n' || c == '\r'
  754 
  755 -- | Drop a truncated tool-call XML tag at the very end of the text (a
  756 -- StopMaxTokens cut can leave an unfinished tag as the final characters).
  757 -- Only the last 'maxPartialTagScan' chars are scanned: a final unmatched
  758 -- '<' that would begin one of the known tool-call tag names (case-
  759 -- insensitive) is cut, along with everything after it. A '<' in prose
  760 -- ("5 < 6") never matches a known tag-name prefix, so ordinary text is
  761 -- untouched.
  762 dropTrailingPartialTag :: Text -> Text
  763 dropTrailingPartialTag t =
  764   case ltIdx of
  765     Nothing -> t
  766     Just i ->
  767       let afterLt = T.toLower (T.drop (i + 1) t)
  768       in if any (tagStart afterLt) tagPrefixes
  769            then T.take i t
  770            else t
  771   where
  772     tagPrefixes = ["invoke", "function_calls", "tool_call", "/invoke",
  773                    "/function_calls", "/tool_call"]
  774     -- The text right after '<' begins a known tag name AND the name is
  775     -- followed by a tag-ish boundary (whitespace, '>', '/', or end of
  776     -- text). A hyphen (as in prose like "<invoke-ish>") disqualifies it —
  777     -- that's an ordinary word, not a tag.
  778     tagStart after tag =
  779       case T.stripPrefix tag after of
  780         Nothing   -> False
  781         Just rest -> case T.uncons rest of
  782           Nothing     -> True  -- cut exactly after the tag name
  783           Just (c, _) -> c == ' ' || c == '>' || c == '/' || c == '\t' || c == '\n' || c == '\r'
  784     -- the last '<' within the trailing scan window (global index), if any
  785     ltIdx =
  786       let window = T.takeEnd maxPartialTagScan t
  787           base   = T.length t - T.length window
  788       in case [base + i | i <- reverse [0 .. T.length window - 1], T.index window i == '<'] of
  789            (i : _) -> Just i
  790            []      -> Nothing

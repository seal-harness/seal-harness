    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Unified exception handling helper. Replaces the 6 duplicated @catch@
    3 -- patterns across 'Seal.Gateway.Send', and
    4 -- 'Seal.Channel.Cli' with a single function that:
    5 --
    6 -- 1. Catches /synchronous/ exceptions only. 'AsyncException' (including
    7 --    'ThreadKilled') is rethrown after logging at 'InfoS' — shutdown must
    8 --    propagate to the bracket, not be swallowed.
    9 -- 2. Logs the full exception to the operator log via 'logIO' at 'ErrorS',
   10 --    including the 'where' label and the logger's 'ChannelContext'.
   11 -- 3. Returns a /sanitized/ 'Text' for the user (via 'Left'): a generic
   12 --    message (@\"internal error (\<where\>) [ref: \<correlationId\>]"@)
   13 --    with a short correlation id so the operator can correlate the user's
   14 --    report to the log entry. 'TranscriptError' text passes through
   15 --    unchanged (it's already user-facing).
   16 -- 4. Calls the optional session-log path ('logTurnError') alongside the
   17 --    katip emission, so the per-session @seal.log@ audit trail is kept.
   18 -- 5. Escapes newlines in the exception text before logging (log-injection
   19 --    defense).
   20 module Seal.Logging.Exceptions
   21   ( withExceptionLogging
   22   , withExceptionLoggingRef
   23   , forkIOLogged
   24   , mkCorrelationId
   25   ) where

   40 -- | Run an 'IO' action, catching synchronous exceptions. Logs the exception
   41 -- with the logger's context + the given 'where' label, calls the optional
   42 -- session-log fallback, and returns either a sanitized error text (for the
   43 -- caller to send to the user) or the original result.
   44 --
   45 -- 'AsyncException' (including 'ThreadKilled') is NOT caught — it is
   46 -- rethrown after logging at 'InfoS', so shutdown propagates to the
   47 -- bracket.
   48 withExceptionLogging
   49   :: SealLogger
   50   -> Maybe FilePath          -- ^ session log path (for seal.log fallback)
   51   -> Text                    -- ^ 'where' label (e.g. "slash command", "turn")
   52   -> IO a
   53   -> IO (Either Text a)
   54 withExceptionLogging logger mLogPath whereLabel action =
   55   (Right <$> action) `catch` \e -> do
   56     case (fromException e :: Maybe AsyncException) of
   57       Just asyncEx -> do
   58         logIO logger InfoS (escapeNewlines (ls $
   59           "shutdown signal in " <> whereLabel <> ": " <> showLs asyncEx))
   60         -- Also record the shutdown in the per-session log so an operator
   61         -- inspecting seal.log after a hung session can see WHY the turn
   62         -- died (e.g. ThreadKilled) rather than just a dangling [TURN]
   63         -- start with no matching end. Best-effort (logTurnError swallows
   64         -- IO errors) and rethrown after logging so the bracket cleanup
   65         -- (idle broadcast) still fires.
   66         logTurnError mLogPath ("shutdown signal in " <> whereLabel <> ": " <> showLs asyncEx)
   67         throwIO e
   68       Nothing -> do
   69         correlationId <- mkCorrelationId
   70         let (userMsg, fullDetail) = sanitizeException whereLabel e correlationId
   71         logIO logger ErrorS (escapeNewlines (ls fullDetail))
   72         logTurnError mLogPath fullDetail
   73         pure (Left userMsg)
   74 
   75 -- | Like 'withExceptionLogging', but also hands the caller the exception and
   76 -- the correlation id it logged.
   77 --
   78 -- The three-part result exists because a failure report needs all three, and
   79 -- only one of them survives the plain variant:
   80 --
   81 -- * the /exception/ — so the report can classify the real cause. The plain
   82 --   variant consumes it into a sanitized 'Text', which is not recoverable.
   83 -- * the /ref/ — so the report's reference id is the SAME id as the log line.
   84 --   The plain variant mints it internally and only embeds it inside the
   85 --   returned 'Text', so a caller cannot use it to join to @seal.log@.
   86 -- * the /result/ — 'Right' on success, so the caller’s happy path is unchanged.
   87 --
   88 -- 'AsyncException' is rethrown after logging, exactly as in the plain variant:
   89 -- a shutdown is not a failure.
   90 withExceptionLoggingRef
   91   :: SealLogger
   92   -> Maybe FilePath
   93   -> Text
   94   -> IO a
   95   -> IO (Either (SomeException, Text) a)
   96 withExceptionLoggingRef logger mLogPath whereLabel action =
   97   (Right <$> action) `catch` \e -> do
   98     case (fromException e :: Maybe AsyncException) of
   99       Just asyncEx -> do
  100         logIO logger InfoS (escapeNewlines (ls $
  101           "shutdown signal in " <> whereLabel <> ": " <> showLs asyncEx))
  102         logTurnError mLogPath ("shutdown signal in " <> whereLabel <> ": " <> showLs asyncEx)
  103         throwIO e
  104       Nothing -> do
  105         correlationId <- mkCorrelationId
  106         let (_userMsg, fullDetail) = sanitizeException whereLabel e correlationId
  107         logIO logger ErrorS (escapeNewlines (ls fullDetail))
  108         logTurnError mLogPath fullDetail
  109         pure (Left (e, correlationId))
  110 
  111 -- | Generate a short correlation id (8 hex chars from a random nonce).
  112 mkCorrelationId :: IO Text
  113 mkCorrelationId = do
  114   n <- abs <$> randomIO :: IO Int
  115   pure (T.justifyRight 8 '0' (T.pack (show n)))

  117 -- | Fork a thread whose synchronous exceptions are LOGGED rather than left to
  118 -- the RTS default handler.
  119 --
  120 -- Every bare @forkIO@ site in the harness previously had this hole: an
  121 -- exception in the forked thread printed to stderr and left no trace in the
  122 -- session log or anywhere a user would look. In the failure that motivated
  123 -- this work, that is one of the ways a session could appear to simply stop.
  124 --
  125 -- 'AsyncException' is deliberately RETHROWN and NOT logged. This repo uses
  126 -- 'killThread' for ordinary shutdown (@Agent.Runtime.Registry@,
  127 -- @Chat.Telegram@, @Chat.Signal@), so logging it would emit a spurious ERROR
  128 -- on every normal teardown. This mirrors 'withExceptionLogging', which
  129 -- rethrows for the same reason, and the rule stated in the design: a shutdown
  130 -- is not a failure.
  131 --
  132 -- The log line carries the exception's @show@ at @ErrorS@ with a caller
  133 -- supplied label. It is an /operator/ log line, not a user-facing surface --
  134 -- the closed-vocabulary rule that governs 'Seal.Transcript.TurnFailure' is
  135 -- about what crosses to the model and to chat channels, neither of which this
  136 -- touches.
  137 forkIOLogged :: SealLogger -> Text -> IO () -> IO ThreadId
  138 forkIOLogged logger label action = forkIO $
  139   action `catch` \e -> case fromException e :: Maybe AsyncException of
  140     Just _  -> throwIO e
  141     Nothing -> logIO logger ErrorS
  142       (escapeNewlines (ls (label <> " thread crashed: " <> showLs e)))
  143 
  144 -- | Sanitize an exception for user-facing text. Returns:
  145 -- * (userMsg, fullDetail) where userMsg is safe to send to the user
  146 --   and fullDetail is the complete exception text for the operator log.
  147 sanitizeException :: Text -> SomeException -> Text -> (Text, Text)
  148 sanitizeException whereLabel e correlationId =
  149   case (fromException e :: Maybe TranscriptError) of
  150     Just (TranscriptError te _) ->
  151       ("transcript error: " <> te, "transcript error: " <> te <> " (" <> whereLabel <> ")")
  152     Nothing ->
  153       ( "internal error (" <> whereLabel <> ") [ref: " <> correlationId <> "]"
  154       , whereLabel <> " failed: " <> T.pack (show e) <> " [ref: " <> correlationId <> "]"
  155       )

  158 showLs :: Show a => a -> Text
  159 showLs = T.pack . show

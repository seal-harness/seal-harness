   63 -- | Read RSS (resident set size) for the current process. On Linux,
   64 -- reads @VmRSS@ from @/proc/self/status@. On macOS (no @/proc@), shells
   65 -- out to @ps -o rss= -p <pid>@. Returns "N/A" if both fail. Used for
   66 -- debug logging to track memory growth at key transcript-read phases.
   67 vmRSS :: IO Text
   68 vmRSS = do
   69   exists <- doesFileExist "/proc/self/status"
   70   if exists
   71     then do
   72       raw <- readFileTextStrict "/proc/self/status"
   73       pure $ case [l | l <- T.lines raw, T.isPrefixOf "VmRSS:" l] of
   74         (l : _) -> T.strip l
   75         []      -> "N/A"
   76     else vmRSSMacOS `catch` \(_ :: SomeException) -> pure "N/A"
   77 
   78 -- | macOS fallback: use @ps@ to get RSS for the current PID.
   79 vmRSSMacOS :: IO Text
   80 vmRSSMacOS = do
   81   pid <- getProcessID
   82   rss <- readProcess "ps" ["-o", "rss=", "-p", show pid] ""
   83   pure (T.strip (T.pack rss))

   85 -- | How much of the transcript to read. Replaces the old @Maybe Int@
   86 -- parameter (where @Just n@ meant "last N" and @Nothing@ meant "all").
   87 data ReadLimit
   88   = ReadAll    -- ^ read every entry (no limit)
   89   | ReadLastN Int  -- ^ read only the last N entries
   90   | ReadFirstN Int -- ^ read only the first N entries (for scroll-to-top)
   91   deriving (Eq, Show)

   93 -- | Convert a legacy @Maybe Int@ (where @Just n@ = last N, @Nothing@ = all)
   94 -- to a 'ReadLimit'. Callers that previously passed @Maybe Int@ can use this
   95 -- for a minimal diff.
   96 limitFromMaybe :: Maybe Int -> ReadLimit
   97 limitFromMaybe Nothing  = ReadAll
   98 limitFromMaybe (Just n) = ReadLastN n

  100 -- | Which on-disk source produced the entries. Surfaced in 'TranscriptTimings'
  101 -- so the @Server-Timing@ header can name the path the request took.
  102 data TranscriptSource
  103   = TSMissing      -- ^ no transcript file exists
  104   | TSLegacy       -- ^ legacy @transcript.jsonl@
  105   | TSConvEntries  -- @conversation.jsonl@ + @entries.jsonl@ (reconstructed)
  106   | TSConvOnly     -- @conversation.jsonl@ only (no entries sidecar)
  107   deriving (Eq, Show)

  109 -- | Per-phase timings for a single 'readTranscriptEntries' call, in
  110 -- milliseconds. Phase durations are measured with 'getCurrentTime' (wall
  111 -- clock); for the small file sizes typical of a transcript, GHC's wall
  112 -- clock has enough resolution. Phases are exclusive of each other EXCEPT
  113 -- @ttTotal@, which spans the entire call. A phase that didn't run (e.g.
  114 -- @ttReconstruct@ when the legacy path was taken) is 0.
  115 data TranscriptTimings = TranscriptTimings
  116   { ttSource       :: TranscriptSource
  117   , ttEntryCount   :: Int           -- ^ number of frontend entries returned
  118   , ttFileReadMs   :: Integer       -- ^ time spent in 'TIO.readFile' (conv + entries files combined)
  119   , ttParseMs      :: Integer       -- ^ 'A.decode' of every line (both files combined)
  120   , ttReconstructMs :: Integer      -- @'reconstruct' (two-file path only)
  121   , ttRewriteMs    :: Integer        -- 'teLineToFrontend' / 'reconEntryToFrontend' / 'convLineToFrontend'
  122   , ttEncodeMs     :: Integer        -- ^ 'A.encode' of the frontend [Value] (set by 'handleTranscript', 0 for the bare reader)
  123   , ttTotalMs      :: Integer       -- ^ wall-clock total of the whole call
  124   }
  125 
  126 -- | Format a 'TranscriptTimings' as a @Server-Timing@ header value per
  127 -- <https://www.w3.org/TR/server-timing/>. Each phase is a separate
  128 -- comma-separated token with a short name, a numeric @dur@ (milliseconds,
  129 -- 3-decimal precision in the spec but we emit integers), and a @desc@
  130 -- naming the phase. The frontend parses these via @performance.getEntriesByName@
  131 -- or by reading the @Server-Timing@ response header directly.
  132 --
  133 -- Example output:
  134 --   @tt;dur=42;desc="total", fr;dur=3;desc="file-read", pr;dur=18;desc="parse", rc;dur=12;desc="reconstruct", rw;dur=4;desc="rewrite", en;dur=8;desc="encode", src;desc="conv+entries", n;desc="127"@
  135 renderServerTiming :: TranscriptTimings -> BC.ByteString
  136 renderServerTiming t =
  137   let phase name ms desc =
  138         BC.pack name <> ";dur=" <> BC.pack (show ms) <> ";desc=\"" <> BC.pack desc <> "\""
  139       tag name desc = BC.pack name <> ";desc=\"" <> BC.pack desc <> "\""
  140       srcDesc = case ttSource t of
  141         TSMissing      -> "missing"
  142         TSLegacy       -> "legacy"
  143         TSConvEntries  -> "conv+entries"
  144         TSConvOnly     -> "conv-only"
  145       parts =
  146         [ phase "tt" (ttTotalMs t)       "total"
  147         , phase "fr" (ttFileReadMs t)    "file-read"
  148         , phase "pr" (ttParseMs t)        "parse"
  149         , phase "rc" (ttReconstructMs t) "reconstruct"
  150         , phase "rw" (ttRewriteMs t)      "rewrite"
  151         , phase "en" (ttEncodeMs t)       "encode"
  152         , tag  "src" srcDesc
  153         , tag  "n"   (show (ttEntryCount t))
  154         ]
  155   in BC.intercalate ", " parts

  157 -- | Set the encode-phase duration + bump the total to span read+encode.
  158 -- 'handleTranscript' uses this after timing 'A.encode' so the @en@ token
  159 -- reflects the JSON-serialization cost (which dominates for transcripts
  160 -- carrying large web_fetch / shell-output tool_result blocks) and @tt@
  161 -- reflects the full read+encode cost.
  162 setEncodeMs :: Integer -> Integer -> TranscriptTimings -> TranscriptTimings
  163 setEncodeMs encodeMs totalMs tt = tt { ttEncodeMs = encodeMs, ttTotalMs = totalMs }

  165 -- | 'zipWith' + 'mapMaybe': apply a partial function across two lists
  166 -- in lockstep, dropping the elements for which the function returns
  167 -- 'Nothing'.
  168 zipWithMaybe :: (a -> b -> Maybe c) -> [a] -> [b] -> [c]
  169 zipWithMaybe _ [] _ = []
  170 zipWithMaybe _ _ [] = []
  171 zipWithMaybe f (a : as) (b : bs) = case f a b of
  172   Just c  -> c : zipWithMaybe f as bs
  173   Nothing -> zipWithMaybe f as bs

  191 -- | Instrumented variant of 'readTranscriptEntries': returns the frontend
  192 -- values alongside per-phase wall-clock timings. Used by the @/transcript@
  193 -- HTTP handler to emit a @Server-Timing@ header; other callers (broadcast,
  194 -- snippet helpers) can use the plain 'readTranscriptEntries' and ignore the
  195 -- overhead of timing capture.
  196 --
  197 -- When the limit is @'Just' n@, only the last @n@ entries are processed
  198 -- (reconstructed + rewritten), avoiding O(all entries) work when the
  199 -- caller only needs a page. The total count is still computed from the
  200 -- entries file so the caller can report @hasMore@.
  201 readTranscriptEntriesTimed
  202   :: SealPaths -> Text -> String -> SessionId -> ReadLimit -> IO ([Value], TranscriptTimings)
  203 readTranscriptEntriesTimed paths model fallbackTs sid rlim = do
  204   tStart <- getCurrentTime
  205   let legacyPath = sessionTranscriptPath paths sid
  206       convPath   = sessionConversationPath paths sid
  207   legacyExists <- doesFileExist legacyPath
  208   convExists   <- doesFileExist convPath
  209   if legacyExists
  210     then do
  211       tFr0 <- getCurrentTime
  212       raw <- readFileTextStrict legacyPath
  213       tFr1 <- getCurrentTime
  214       tPr0 <- getCurrentTime
  215       let vals = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  216                           (filter (not . T.null) (T.lines raw))
  217       tPr1 <- getCurrentTime
  218       tRw0 <- getCurrentTime
  219       let frontend = map teLineToFrontend vals
  220           limited = case rlim of
  221             ReadAll     -> frontend
  222             ReadLastN n -> drop (max 0 (length frontend - n)) frontend
  223             ReadFirstN n -> take n frontend
  224       tRw1 <- getCurrentTime
  225       tEnd <- getCurrentTime
  226       let tt = TranscriptTimings
  227             { ttSource        = TSLegacy
  228             , ttEntryCount    = length frontend
  229             , ttFileReadMs    = ms tFr0 tFr1
  230             , ttParseMs       = ms tPr0 tPr1
  231             , ttReconstructMs = 0
  232             , ttRewriteMs     = ms tRw0 tRw1
  233             , ttEncodeMs      = 0
  234             , ttTotalMs       = ms tStart tEnd
  235             }
  236       pure (limited, tt)
  237     else if convExists
  238       then do
  239         entriesExist <- doesFileExist (sessionEntriesPath paths sid)
  240         if entriesExist
  241           then do
  242             -- Indexed path: read entries.jsonl fully (small file), then
  243             -- read conversation lines on demand via the index.
  244             tFr2 <- getCurrentTime
  245             eraw <- readFileTextStrict (sessionEntriesPath paths sid)
  246             tFr3 <- getCurrentTime
  247             tPr2 <- getCurrentTime
  248             let evs = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  249                                (filter (not . T.null) (T.lines eraw)) :: [EntryRecord]
  250             tPr3 <- getCurrentTime
  251             -- Ensure the conversation index exists and is current
  252             let idxPath = sessionConversationIndexPath paths sid
  253             rssIdx0 <- vmRSS
  254             tIdx0 <- getCurrentTime
  255             idxResult <- ensureIndex convPath idxPath
  256             tIdx1 <- getCurrentTime
  257             rssIdx1 <- vmRSS
  258             putStrLn $ "[transcript] ensureIndex " <> show idxResult
  259                      <> " (" <> show (ms tIdx0 tIdx1 :: Integer) <> "ms) rss=" <> T.unpack rssIdx0 <> " -> " <> T.unpack rssIdx1
  260             -- Trailing conv entries: read lines [maxConvLen..totalLines) via index
  261             totalLines <- convLineCount idxPath
  262             let maxConvLen = if null evs then 0 else maximum (map erConvLen evs)
  263                 trailingCount = totalLines - maxConvLen
  264             -- Split the limit between reconstructed entries and trailing
  265             -- conv entries.
  266             --
  267             -- 'ReadLastN': trailing entries are the most recent activity
  268             -- (they have no entry records), so they get priority: allocate
  269             -- up to half the limit to trailing, the rest to reconstruction.
  270             -- Without this, a limit fully consumed by reconstructed entries
  271             -- starves the trailing range — the user sees old entries instead
  272             -- of the end of the session.
  273             --
  274             -- 'ReadFirstN': prioritize reconstructed entries from the
  275             -- beginning. Only read trailing lines if there aren't enough
  276             -- entry records to fill the limit. Trailing lines are read from
  277             -- the START of the trailing range (the earliest ones), not the
  278             -- end.
  279             let (reconLimit, trailingLimit, fromBeginning) = case rlim of
  280                   ReadLastN n | n > 0 ->
  281                     let trailingBudget = min n (min trailingCount (n `div` 2))
  282                         reconBudget = n - trailingBudget
  283                     in (Just reconBudget, trailingBudget, False)
  284                   ReadFirstN n | n > 0 ->
  285                     let reconBudget = min n (length evs)
  286                         trailingBudget = max 0 (n - reconBudget)
  287                     in (Just reconBudget, min trailingBudget trailingCount, True)
  288                   _ -> (Nothing, trailingCount, False)
  289             -- Stream-reconstruct: use paginated variant when a limit is
  290             -- specified (only process the last N entries' conversation lines)
  291             rssRecon0 <- vmRSS
  292             tRc0 <- getCurrentTime
  293             (reconstructed, reconStartIdx) <- case reconLimit of
  294               Just n | n > 0 && n < length evs ->
  295                 let start = if fromBeginning then 0 else length evs - n
  296                 in (, start) <$> reconstructStreamingPage convPath idxPath evs start n
  297               _ -> (, 0) <$> reconstructStreaming convPath idxPath evs
  298             let reconFrontend = zipWithMaybe reconEntryToFrontend [reconStartIdx..] reconstructed
  299             -- Read trailing conv lines. For ReadLastN, read the LAST
  300             -- trailingLimit lines (most recent). For ReadFirstN, read the
  301             -- FIRST trailingLimit lines (earliest, right after the entries).
  302             let trailingStart = if fromBeginning
  303                                 then 0
  304                                 else max 0 (trailingCount - trailingLimit)
  305             trailing <- if trailingCount > 0 && trailingLimit > 0
  306               then do
  307                 let readStart = maxConvLen + trailingStart
  308                     readEnd = maxConvLen + min trailingCount (trailingStart + trailingLimit)
  309                 putStrLn $ "[transcript] trailing-read readStart=" <> show readStart <> " readEnd=" <> show readEnd <> " (convLines " <> show readStart <> ".." <> show (readEnd - 1) <> ")"
  310                 eTrailingVals <- readConvLinesRaw convPath idxPath readStart readEnd
  311                 let trailingVals = case eTrailingVals of
  312                       Right vs -> vs
  313                       Left err -> trace ("[transcript] trailing-read ERROR: " <> T.unpack err) []
  314                 putStrLn $ "[transcript] trailing-read got " <> show (length trailingVals) <> " values"
  315                 pure (trailingConvEntries model fallbackTs (maxConvLen + trailingStart) trailingVals)
  316               else do
  317                 putStrLn $ "[transcript] trailing-read SKIPPED trailingCount=" <> show trailingCount <> " trailingLimit=" <> show trailingLimit
  318                 pure []
  319             let frontend = reconFrontend <> trailing
  320             rssRecon1 <- vmRSS
  321             tRc1 <- getCurrentTime
  322             tEnd <- getCurrentTime
  323             putStrLn $ "[transcript] conv+entries sid=" <> show sid
  324                      <> " evs=" <> show (length evs)
  325                      <> " totalLines=" <> show totalLines
  326                      <> " maxConvLen=" <> show maxConvLen
  327                      <> " trailingCount=" <> show trailingCount
  328                     <> " rlim=" <> show rlim
  329                      <> " reconLimit=" <> show reconLimit
  330                      <> " trailingLimit=" <> show trailingLimit
  331                      <> " reconFrontend=" <> show (length reconFrontend)
  332                      <> " trailing=" <> show (length trailing)
  333                      <> " frontend=" <> show (length frontend)
  334                      <> " rss=" <> T.unpack rssRecon0 <> " -> " <> T.unpack rssRecon1
  335                      <> " idxMs=" <> show (ms tIdx0 tIdx1 :: Integer)
  336             let tt = TranscriptTimings
  337                   { ttSource        = TSConvEntries
  338                   , ttEntryCount    = length evs + trailingCount
  339                   , ttFileReadMs    = ms tFr2 tFr3
  340                   , ttParseMs       = ms tPr2 tPr3
  341                   , ttReconstructMs = ms tRc0 tRc1
  342                   , ttRewriteMs     = 0
  343                   , ttEncodeMs      = 0
  344                   , ttTotalMs       = ms tStart tEnd
  345                   }
  346             pure (frontend, tt)
  347           else do
  348             -- Conv-only path: no entries.jsonl. Use the index to read
  349             -- conversation lines in batches instead of the full file.
  350             let idxPath = sessionConversationIndexPath paths sid
  351             _ <- ensureIndex convPath idxPath
  352             totalLines <- convLineCount idxPath
  353             -- Apply limit: read only the needed lines.
  354             let (readStart, readCount) = case rlim of
  355                   ReadLastN n | n >= 0 && n < totalLines -> (totalLines - n, n)
  356                   ReadFirstN n | n >= 0 && n < totalLines -> (0, n)
  357                   _ -> (0, totalLines)
  358             tRw0 <- getCurrentTime
  359             eMsgs <- readConvLinesRaw convPath idxPath readStart (readStart + readCount)
  360             let msgVals = case eMsgs of
  361                   Right vs -> vs
  362                   Left _ -> []
  363                 frontend = zipWith (convLineToFrontend model [] fallbackTs) [readStart..] msgVals
  364             tRw1 <- getCurrentTime
  365             tEnd <- getCurrentTime
  366             let tt = TranscriptTimings
  367                   { ttSource        = TSConvOnly
  368                   , ttEntryCount    = totalLines
  369                   , ttFileReadMs    = 0
  370                   , ttParseMs       = 0
  371                   , ttReconstructMs = 0
  372                   , ttRewriteMs     = ms tRw0 tRw1
  373                   , ttEncodeMs      = 0
  374                   , ttTotalMs       = ms tStart tEnd
  375                   }
  376             pure (frontend, tt)
  377       else do
  378         tEnd <- getCurrentTime
  379         let tt = TranscriptTimings
  380               { ttSource        = TSMissing
  381               , ttEntryCount    = 0
  382               , ttFileReadMs    = 0
  383               , ttParseMs       = 0
  384               , ttReconstructMs = 0
  385               , ttRewriteMs     = 0
  386               , ttEncodeMs      = 0
  387               , ttTotalMs       = ms tStart tEnd
  388               }
  389         pure ([], tt)
  390   where
  391     ms a b = round (realToFrac (b `diffUTCTime` a) * 1000 :: Double)

  393 -- | Extract a short snippet of the first user message in a session's
  394 -- transcript, for use as the default session title when the user has not
  395 -- set an explicit description. Reads the indexed transcript format
  396 -- (@conversation.jsonl@) first, falling back to the legacy
  397 -- @transcript.jsonl@. Returns 'Nothing' when the session has no transcript
  398 -- or no user message with text content. The snippet is truncated to 80
  399 -- characters (on a codepoint boundary) to keep the sessions list compact.
  400 firstUserMessageSnippet :: SealPaths -> SessionId -> IO (Maybe Text)
  401 firstUserMessageSnippet paths sid = do
  402   let legacyPath = sessionTranscriptPath paths sid
  403       convPath   = sessionConversationPath paths sid
  404   legacyExists <- doesFileExist legacyPath
  405   convExists   <- doesFileExist convPath
  406   if convExists
  407     then do
  408       raw <- readFileTextStrict convPath
  409       let msgs = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  410                           (filter (not . T.null) (T.lines raw)) :: [Message]
  411       pure (snippetFromMessages msgs)
  412     else if legacyExists
  413       then do
  414         raw <- readFileTextStrict legacyPath
  415         let entries = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  416                                (filter (not . T.null) (T.lines raw)) :: [TranscriptEntry]
  417         pure (snippetFromMessages (legacyRequestMessages entries))
  418       else pure Nothing

  420 -- | Extract the first user message's text from a list of messages, truncated
  421 -- to 80 characters. Tool-use and tool-result blocks are skipped (only the
  422 -- first 'CbText' block of the first 'User' message is used).
  423 snippetFromMessages :: [Message] -> Maybe Text
  424 snippetFromMessages msgs =
  425   case filter (\m -> msgRole m == User) msgs of
  426     (m : _) -> case [t | CbText t <- msgContent m] of
  427       (t : _) -> Just (truncateSnippet 80 t)
  428       []      -> Nothing
  429     [] -> Nothing

  431 -- | Truncate a snippet to at most @n@ characters, appending an ellipsis when
  432 -- truncation occurs. 'T.take' is codepoint-safe for UTF-8 'Text'.
  433 truncateSnippet :: Int -> Text -> Text
  434 truncateSnippet n t
  435   | T.length t <= n = t
  436   | otherwise       = T.take n t <> "…"

  438 -- | Extract the request messages from legacy @transcript.jsonl@ entries.
  439 -- Only 'Request'-direction entries carry a payload with a @messages@ array;
  440 -- we collect messages from every request entry (the first user message is
  441 -- always in the first request).
  442 legacyRequestMessages :: [TranscriptEntry] -> [Message]
  443 legacyRequestMessages = concatMap go
  444   where
  445     go te
  446       | teDirection te /= Request = []
  447       | otherwise = case tePayload te of
  448           A.Object o -> case KeyMap.lookup (Key.fromText "messages") o of
  449             Just (A.Array arr) -> mapMaybe decodeMsg (V.toList arr)
  450             _ -> []
  451           _ -> []
  452     decodeMsg v = case A.fromJSON v :: A.Result Message of
  453       A.Success m -> Just m
  454       A.Error _   -> Nothing

  456 -- | Map a legacy on-disk transcript line (Haskell TranscriptEntry JSON
  457 -- with @te*@-prefixed fields) to the frontend's TranscriptEntry shape.
  458 teLineToFrontend :: A.Value -> A.Value
  459 teLineToFrontend rawLine =
  460   let o = case rawLine of
  461         A.Object m -> m
  462         _          -> KeyMap.empty
  463       k = Key.fromText
  464       lookupT key = case KeyMap.lookup (k key) o of
  465         Just (A.String t) -> Just t
  466         _                 -> Nothing
  467       -- The on-disk `payload` field is a JSON string; decode it to a Value
  468       -- so we can send it as a JSON object (not a string) in the wire
  469       -- response. Falls back to the raw string when it's not valid JSON
  470       -- (defensive against a malformed on-disk line).
  471       payloadVal = case KeyMap.lookup (k "payload") o of
  472         Just (A.String s) -> case A.decode (BL.fromStrict (TE.encodeUtf8 s)) of
  473           Just v  -> v
  474           Nothing -> A.String s
  475         other -> fromMaybe A.Null other
  476       -- The legacy transcript.jsonl entry's `meta` field carries the
  477       -- channel label (stamped by runTurn's requestMeta). Extract it so
  478       -- the frontend can attribute user messages to their channel on the
  479       -- legacy path too. Null when absent (response entries, old entries).
  480       mChannel = case KeyMap.lookup (k "meta") o of
  481         Just (A.Object mo) -> case KeyMap.lookup (k "channel") mo of
  482           Just (A.String ch) -> Just ch
  483           _                  -> Nothing
  484         _ -> Nothing
  485       mInternal = case KeyMap.lookup (k "meta") o of
  486         Just (A.Object mo) -> case KeyMap.lookup (k "internal") mo of
  487           Just (A.Bool b) -> Just b
  488           _               -> Nothing
  489         _ -> Nothing
  490   in object
  491      [ "id"        .= lookupT "id"
  492      , "timestamp" .= lookupT "timestamp"
  493      , "direction" .= lookupT "direction"
  494      , "payload"   .= payloadVal
  495      , "harness"   .= lookupT "correlation"
  496      , "model"     .= lookupT "model"
  497      , "channel"   .= mChannel
  498     , "internal"  .= mInternal
  499      , "raw"       .= TE.decodeUtf8 (BL.toStrict (A.encode rawLine))
  500      ]

  502 -- | Synthesize frontend TranscriptEntry JSON values for conversation
  503 -- messages that are NOT covered by any 'EntryRecord' (i.e. messages beyond
  504 -- the maximum @convLen@ across all entries). This handles incomplete
  505 -- @entries.jsonl@ files (e.g. due to truncation or a crash during
  506 -- recording) — without this, any conversation messages beyond the last
  507 -- entry's @convLen@ are silently dropped by 'reconstruct', making them
  508 -- invisible in the web UI. Each trailing message is converted via
  509 -- 'convLineToFrontend' (the same function used when @entries.jsonl@ does
  510 -- not exist at all), so User messages become @request@ entries and
  511 -- Assistant messages become @response@ entries with the same shape the
  512 -- frontend's @transcriptToMessages@ expects.
  513 trailingConvEntries
  514   :: Text     -- ^ model label (for the @model@ field in the frontend entry)
  515   -> String   -- ^ fallback timestamp (conversation.jsonl carries no timestamps)
  516   -> Int      -- ^ maxConvLen — the highest @convLen@ across all entry records
  517   -> [A.Value] -- ^ all conversation.jsonl lines (parsed as Aeson values)
  518   -> [A.Value] -- ^ frontend TranscriptEntry JSON for the trailing messages
  519 trailingConvEntries model fallbackTs maxConvLen msgVals =
  520   let convCount = length msgVals
  521   in if convCount > maxConvLen
  522        then zipWith (convLineToFrontend model [] fallbackTs)
  523                     [maxConvLen ..]
  524                     (drop maxConvLen msgVals)
  525        else []

  527 -- | Synthesize a frontend TranscriptEntry from a conversation.jsonl line
  528 -- (@role@/@content@). User → request; Assistant → response. The
  529 -- line index (0-based) is used as the entry id — conversation.jsonl
  530 -- carries no per-entry id, and the line index is stable across reads of
  531 -- the same file, so the frontend's dedup-by-id works.
  532 convLineToFrontend :: Text -> [String] -> String -> Int -> A.Value -> A.Value
  533 convLineToFrontend model entryTimestamps fallbackTs idx rawLine =
  534   let o = case rawLine of
  535         A.Object m -> m
  536         _          -> KeyMap.empty
  537       k = Key.fromText
  538       role = case KeyMap.lookup (k "role") o of
  539         Just (A.String t) -> t
  540         _                  -> "user"
  541       rawContent = case KeyMap.lookup (k "content") o of
  542         Just (A.Array arr) -> arr
  543         _                  -> mempty
  544       contentBlocks = map cbToFrontend (V.toList rawContent)
  545       content = A.Array (V.fromList contentBlocks)
  546       direction :: Text
  547       direction = if T.toCaseFold role == "user" then "request" else "response"
  548       payloadJson = if direction == "request"
  549         then A.object ["messages" A..= [A.object ["role" A..= ("user" :: Text), "content" A..= content]]]
  550         else A.object ["content" A..= content]
  551       entryId = T.pack (show idx)
  552       ts = fromMaybe fallbackTs (lookup idx (zip [0..] entryTimestamps))
  553     in object
  554        [ "id"        .= entryId
  555        , "timestamp" .= T.pack ts
  556        , "direction" .= direction
  557        , "payload"   .= payloadJson
  558        , "harness"   .= (Nothing :: Maybe Text)
  559        , "model"     .= model
  560        , "channel"   .= (Nothing :: Maybe Text)
  561       , "internal"  .= (Nothing :: Maybe Bool)
  562       , "raw"       .= TE.decodeUtf8 (BL.toStrict (A.encode rawLine))
  563       ]

  565 -- | Rewrite one on-disk 'ContentBlock' (GHC-Generics 'TaggedObject' shape)
  566 -- into the Anthropic-style block the frontend parses.
  567 cbToFrontend :: A.Value -> A.Value
  568 cbToFrontend blk =
  569   let bo = case blk of
  570         A.Object m -> m
  571         _          -> KeyMap.empty
  572       k = Key.fromText
  573       tag = case KeyMap.lookup (k "tag") bo of
  574         Just (A.String t) -> Just t
  575         _                  -> Nothing
  576       contents = KeyMap.lookup (k "contents") bo
  577       lookupT key m = case KeyMap.lookup (k key) m of
  578         Just (A.String t) -> Just t
  579         _                 -> Nothing
  580       lookupB key m = case KeyMap.lookup (k key) m of
  581         Just (A.Bool b) -> Just b
  582         _               -> Nothing
  583   in case tag of
  584        Just "CbText" -> case contents of
  585          Just (A.String t) -> object ["type" .= ("text" :: Text), "text" .= t]
  586          _                 -> fallback
  587        Just "CbThinking" -> case contents of
  588          Just (A.String t) -> object ["type" .= ("thinking" :: Text), "thinking" .= t]
  589          _                 -> fallback
  590        Just "CbToolUse" -> object
  591          [ "type"  .= ("tool_use" :: Text)
  592          , "id"    .= fromMaybe "" (lookupT "id" bo)
  593          , "name"  .= fromMaybe "" (lookupT "name" bo)
  594          , "input" .= fromMaybe A.Null (KeyMap.lookup (k "input") bo)
  595          ]
  596        Just "CbToolResult" -> object
  597          [ "type"        .= ("tool_result" :: Text)
  598          , "tool_use_id" .= fromMaybe "" (lookupT "forId" bo)
  599          , "content"     .= toolResultPartsToFrontend (KeyMap.lookup (k "parts") bo)
  600          , "is_error"    .= fromMaybe False (lookupB "isError" bo)
  601          ]
  602        _ -> fallback
  603   where
  604      fallback = object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]
  605 
  606 -- | Rewrite the on-disk 'parts' value into the Anthropic-style array
  607 -- the frontend parses (@[{type:"text", text:"..."}]@). 'ToolResultPart'
  608 -- is a @newtype TrpText Text@, so aeson's derived 'ToJSON' serializes it
  609 -- as a bare JSON string (not a 'TaggedObject'). The on-disk 'parts' is
  610 -- thus @["text"]@, not @[{tag:"TrpText", contents:"text"}]@. Handles both
  611 -- the bare-string shape and the object shape (defensive fallback).
  612 toolResultPartsToFrontend :: Maybe A.Value -> A.Value
  613 toolResultPartsToFrontend mparts =
  614   case mparts of
  615     Just (A.Array arr) ->
  616       A.Array (V.fromList (map trpToFrontend (V.toList arr)))
  617     other -> fromMaybe A.Null other
  618 
  619 trpToFrontend :: A.Value -> A.Value
  620 trpToFrontend blk =
  621   case blk of
  622     A.String t -> object ["type" .= ("text" :: Text), "text" .= t]
  623     A.Object m ->
  624       let tag = case KeyMap.lookup (Key.fromString "tag") m of
  625             Just (A.String t) -> Just t
  626             _                 -> Nothing
  627           contents = KeyMap.lookup (Key.fromString "contents") m
  628       in case tag of
  629            Just "TrpText" -> case contents of
  630              Just (A.String t) -> object ["type" .= ("text" :: Text), "text" .= t]
  631              _                 -> object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]
  632            _ -> object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]
  633     _ -> object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]

  635 -- | The set of opcode names whose 'EKHarness' entries should surface to
  636 -- the web frontend SPA as distinct harness entries (not be dropped by
  637 -- 'reconEntryToFrontend'). v1: @SKILL_LOAD@ so /skill load invocations
  638 -- appear in the transcript as identifiable skill-load events (per the
  639 -- user's "properly identified as a skill load operation" requirement);
  640 -- @SETUP_REPO@ so the clone/no-op/conflict/failure outcome of a
  641 -- repo-attached session creation is visible in the chat (the frontend's
  642 -- 'transcriptToMessages' renders it as a tool-call box; without this
  643 -- whitelist entry, the SETUP_REPO result recorded by
  644 -- 'recordSetupRepoResult' is dropped before reaching either the WS
  645 -- broadcast or the HTTP seed, so the user sees nothing for the repo
  646 -- setup).
  647 -- @ASK_HUMAN@ so pending questions from channel-originated turns
  648 -- (Telegram, Signal) surface in the web transcript. Without this, a
  649 -- question sent to a chat channel is invisible in the web UI — a
  650 -- "phantom message" that exists in the audit log but not the frontend.
  651 -- Approval-bearing entries always surface regardless of this whitelist.
  652 -- A 'Set' rather than a list to make the shared-state surface
  653 -- discoverable (see the design doc's §8 risk 2).
  654 userSurfacingOps :: Set.Set Text
  655 userSurfacingOps = Set.fromList ["SKILL_LOAD", "SKILL_MANAGE", "SETUP_REPO", "ASK_HUMAN"]

  657 -- | Predicate: does a harness payload's @op.name@ fall in
  658 -- 'userSurfacingOps'? Returns 'False' for payloads with no @op@ key, a
  659 -- non-object @op@, or no @name@ field — those are dropped by the filter
  660 -- (matching the pre-v1 behavior for non-approval harness entries).
  661 -- For @SKILL_MANAGE@, additionally requires @input.action == "load"@
  662 -- (other SKILL_MANAGE actions like write/list/delete are not user-surfacing).
  663 isUserSurfacingOp :: KeyMap.KeyMap Value -> Bool
  664 isUserSurfacingOp o =
  665   case KeyMap.lookup (Key.fromText "op") o of
  666     Just (Object opObj) ->
  667       case KeyMap.lookup (Key.fromText "name") opObj of
  668         Just (String n)
  669           | n == "SKILL_MANAGE" -> isSkillLoadAction o
  670           | otherwise -> n `Set.member` userSurfacingOps
  671         _               -> False
  672     _ -> False
  673 
  674 -- | Predicate: does a harness payload's @input.action@ equal @"load"@?
  675 -- Used to distinguish the user-surfacing skill-load action of
  676 -- @SKILL_MANAGE@ from its other actions (write/list/delete).
  677 isSkillLoadAction :: KeyMap.KeyMap Value -> Bool
  678 isSkillLoadAction o =
  679   case KeyMap.lookup (Key.fromText "input") o of
  680     Just (Object inputObj) ->
  681       case KeyMap.lookup (Key.fromText "action") inputObj of
  682         Just (String a) -> a == "load"
  683         _               -> False
  684     _ -> False

  686 -- | Map a reconstructed 'TranscriptEntry' (from 'reconstruct') to the
  687 -- frontend's TranscriptEntry JSON shape.
  688 reconEntryToFrontend :: Int -> TranscriptEntry -> Maybe A.Value
  689 reconEntryToFrontend idx te =
  690   case tePayload te of
  691     A.Null -> Nothing
  692     -- Drop harness entries (opcode invocations from the dispatcher) that
  693     -- carry a "harness" key in the payload — UNLESS they also carry an
  694     -- "approval" key (which marks them as approval-evidence entries that
  695     -- should surface so the user sees the confirmation decision in the
  696     -- transcript), OR their op.name is in 'userSurfacingOps' (which marks
  697     -- them as user-surfacing opcode invocations like /skill load that
  698     -- should appear as distinct harness entries, not be folded into the
  699     -- surrounding user/assistant bubbles).
  700     A.Object o
  701       | KeyMap.member (Key.fromText "harness") o,
  702         not (KeyMap.member (Key.fromText "approval") o),
  703         not (isUserSurfacingOp o) -> Nothing
  704     payloadVal -> Just $
  705       let dirStr = case teDirection te of
  706             Request  -> "request" :: Text
  707             Response -> "response"
  708           payloadJson = rewritePayload False payloadVal (teDirection te)
  709           -- The `raw` view preserves full tool input_schemas so the
  710           -- "View raw JSON" modal shows the verbatim schemas the LLM was
  711           -- sent (the `payload` field strips them to keep the wire small).
  712           rawJson = rewritePayload True payloadVal (teDirection te)
  713           entryId = let raw = teId te in if T.null raw then T.pack (show idx) else raw
  714           -- The originating channel label (e.g. "telegram", "web", "cli"),
  715           -- stamped into the request entry's erMeta by runTurn's
  716           -- requestMeta (and into SKILL_LOAD entries by
  717           -- recordSkillLoadResult). Surfaced as a top-level `channel` field
  718           -- so the frontend can attribute user messages — and skill loads —
  719           -- to the channel they came from. Null when the entry carries no
  720           -- channel (e.g. response entries, CLI TUI turns with no
  721           -- MessageSource).
  722           mChannel = case Map.lookup "channel" (teMeta te) of
  723             Just (A.String ch) -> Just ch
  724             _                  -> Nothing
  725           mInternal = case Map.lookup "internal" (teMeta te) of
  726             Just (A.Bool b) -> Just b
  727             _               -> Nothing
  728       in object
  729          [ "id"        .= entryId
  730          , "timestamp" .= T.pack (showIso (teTimestamp te))
  731          , "direction" .= dirStr
  732          -- `payload` is a JSON OBJECT (not a string) so the frontend can
  733          -- access its fields directly without a second JSON.parse pass.
  734          -- The old format encoded payload as a JSON string
  735          -- (TE.decodeUtf8 (A.encode payloadJson)), which forced the browser
  736          -- to parse the outer JSON (un-escaping every `\"` in the payload
  737          -- string), then call JSON.parse(payload) AGAIN to parse the inner
  738          -- JSON — two full parse passes with O(n) escape processing in
  739          -- between. For a 279KB body this took ~889ms; as an object it's a
  740          -- single pass and should be <10ms.
  741          , "payload"   .= payloadJson
  742          , "harness"   .= (Nothing :: Maybe Text)
  743          , "model"     .= (Nothing :: Maybe Text)
  744          , "channel"   .= mChannel
  745         , "internal"  .= mInternal
  746          -- The `raw` field carries the full rewritten payload with tool
  747          -- input_schemas preserved, so the frontend's "View raw JSON"
  748          -- modal shows the verbatim schemas the LLM was sent. The legacy
  749          -- path (teLineToFrontend) still ships the verbatim on-disk line.
  750          , "raw"       .= TE.decodeUtf8 (BL.toStrict (A.encode rawJson))
  751          ]

  753 -- | Rewrite a reconstructed payload 'Value' from GHC-Generics
  754 -- 'TaggedObject' encoding to the Anthropic-style JSON the frontend parses.
  755 -- When @preserveSchemas@ is 'True', the @tools@ array keeps its full
  756 -- @input_schema@/parameters fields — used to build the @raw@ view so the
  757 -- "View raw JSON" modal shows the verbatim schemas the LLM was sent. When
  758 -- 'False', 'toolsWithDescriptions' strips them to keep the wire payload
  759 -- small (the frontend's collapsed Tools row only needs names + descriptions).
  760 rewritePayload :: Bool -> A.Value -> Direction -> A.Value
  761 rewritePayload preserveSchemas val dir =
  762   case val of
  763     A.Object o ->
  764       let k = Key.fromText
  765           rewriteMsgs = case KeyMap.lookup (k "messages") o of
  766             Just (A.Array arr) ->
  767               let msgs = map rewriteMessage (V.toList arr)
  768               in ["messages" .= A.Array (V.fromList msgs)]
  769             _ -> []
  770           rewriteContent = case KeyMap.lookup (k "content") o of
  771             Just (A.Array arr) ->
  772               let blocks = map cbToFrontend (V.toList arr)
  773               in ["content" .= A.Array (V.fromList blocks)]
  774             _ -> []
  775           passthrough key = case KeyMap.lookup key o of
  776             Just v  -> [key .= v]
  777             Nothing -> []
  778           usageFields = case KeyMap.lookup (k "usage") o of
  779             Just (A.Object uo) ->
  780               let uIn  = case KeyMap.lookup (k "input")  uo of
  781                     Just n -> Just n; _ -> Nothing
  782                   uOut = case KeyMap.lookup (k "output") uo of
  783                     Just n -> Just n; _ -> Nothing
  784               in case (uIn, uOut) of
  785                    (Just _, Just _) ->
  786                      [ "usage" .= object
  787                        [ "input_tokens"  .= uIn
  788                        , "output_tokens" .= uOut
  789                        ]
  790                      ]
  791                    _ -> []
  792             _ -> []
  793           toolsField = if preserveSchemas
  794             then passthrough (k "tools")
  795             else toolsWithDescriptions o
  796           fields = case dir of
  797             Request ->
  798               passthrough (k "system")
  799                <> passthrough (k "model")
  800                -- Replace the full tool definitions with name + description,
  801                -- stripping input_schema/parameters to keep the wire payload
  802                -- small. The full definitions (with input_schema) are the
  803                -- same across every request turn and can be 11KB+ each;
  804                -- shipping them in all 73 request entries of a 146-turn
  805                -- session wastes ~840KB. The frontend's "Tools" collapsed
  806                -- row shows the names + descriptions, and the expanded
  807                -- view's JSON carries the name + description array (useful
  808                -- for seeing WHAT tools were available + what they do, just
  809                -- not the full schemas). The frontend already deduplicates
  810                -- tools via `seenTools` so only the first occurrence renders.
  811                -- The @raw@ view (built with preserveSchemas=True) ships the
  812                -- full tools array so the "View raw JSON" modal shows the
  813                -- verbatim schemas the LLM was sent.
  814                <> toolsField
  815                <> passthrough (k "toolChoice")
  816                <> passthrough (k "maxTokens")
  817                <> passthrough (k "approval")
  818                <> passthrough (k "op")
  819                -- Harness/opcode-invocation entries (EKHarness) carry the
  820                -- opcode's input + result under these keys (see
  821                -- Seal.Transcript.Reconstruct.harnessPayload). The SKILL_LOAD
  822                -- result entry is a Request-direction harness entry; without
  823                -- passing @input@ + @result@ through, the frontend's
  824                -- @transcriptToMessages@ sees @op.name = "SKILL_LOAD"@ but no
  825                -- @result@, so the skill-load tool-call box never renders.
  826                 <> passthrough (k "input")
  827                 <> passthrough (k "result")
  828                 <> passthrough (k "channel")
  829                 <> rewriteMsgs
  830             Response ->
  831               passthrough (k "model")
  832                <> rewriteContent
  833                <> usageFields
  834                <> passthrough (k "stop")
  835                <> passthrough (k "durationMs")
  836       in A.object fields
  837     _ -> val
  838 
  839 -- | Extract name + description from a tools array, dropping input_schema /
  840 -- parameters to keep the wire payload small. Returns
  841 -- @["tools" .= [...]]@ where each element is
  842 -- @{"name": "<toolName>", "description": "<toolDesc>"}@ — the shape the
  843 -- frontend's 'extractToolDefNames' parses. Returns @[]@ when the tools field
  844 -- is absent or not an array.
  845 toolsWithDescriptions :: KeyMap.KeyMap Value -> [(Key, Value)]
  846 toolsWithDescriptions o =
  847   case KeyMap.lookup (Key.fromText "tools") o of
  848     Just (A.Array arr) ->
  849       let defs = map toolNameDesc (V.toList arr)
  850       in ["tools" .= A.Array (V.fromList defs)]
  851     _ -> []
  852   where
  853     -- Extract {name, description} from a tool definition Value, dropping
  854     -- input_schema / parameters. Handles both the Anthropic shape
  855     -- ({name, description, input_schema}) and the Ollama shape
  856     -- ({type:"function", function:{name, description, parameters}}).
  857     toolNameDesc v = case v of
  858       A.Object td ->
  859         case ( KeyMap.lookup (Key.fromText "name") td
  860              , KeyMap.lookup (Key.fromText "description") td ) of
  861           (Just n, Just d) -> A.object ["name" .= n, "description" .= d]
  862           (Just n, Nothing) -> A.object ["name" .= n]
  863           (Nothing, _) -> case KeyMap.lookup (Key.fromText "function") td of
  864             Just (A.Object fn) -> case KeyMap.lookup (Key.fromText "name") fn of
  865               Just n -> case KeyMap.lookup (Key.fromText "description") fn of
  866                 Just d -> A.object ["name" .= n, "description" .= d]
  867                 Nothing -> A.object ["name" .= n]
  868               Nothing -> A.object ["name" .= A.Null]
  869             _ -> A.object ["name" .= A.Null]
  870       _ -> A.object ["name" .= A.Null]

  872 -- | Rewrite one 'Message' (@{role, content: [...]}@, strip-prefix camelCase
  873 -- JSON) to the Anthropic-style shape (@{role, content: [...]}@) the frontend
  874 -- parses, with content blocks rewritten by 'cbToFrontend'. The role is
  875 -- lowercased because GHC-Generics encodes 'User'/'Assistant' as
  876 -- @"User"@/@"Assistant"@ but the frontend checks @msg.role === "user"@.
  877 rewriteMessage :: A.Value -> A.Value
  878 rewriteMessage msg =
  879   case msg of
  880     A.Object mo ->
  881       let k = Key.fromText
  882           role = case KeyMap.lookup (k "role") mo of
  883             Just (A.String t) -> T.toLower t
  884             _                 -> "user" :: Text
  885           rawContent = case KeyMap.lookup (k "content") mo of
  886             Just (A.Array arr) -> arr
  887             _                  -> mempty
  888           blocks = map cbToFrontend (V.toList rawContent)
  889       in A.object
  890          [ "role"    .= role
  891          , "content" .= A.Array (V.fromList blocks)
  892          ]
  893     _ -> msg

  895 -- | ISO-8601 with milliseconds + Z (the frontend's expected timestamp shape).
  896 showIso :: UTCTime -> String
  897 showIso = formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%S%3QZ"

  899 -- | The ISO timestamp of the most recent @request@-direction entry in a
  900 -- session's transcript (the last time the USER sent a message), or 'Nothing'
  901 -- when the session has no transcript or no user message. Used by the sidebar
  902 -- to sort tabs by last-user-message time (oldest first within each status
  903 -- bucket). Reads the same transcript paths as 'firstUserMessageSnippet';
  904 -- returns the timestamp as an ISO string (matching the frontend's
  905 -- TranscriptEntry.timestamp shape).
  906 lastUserMessageAt :: SealPaths -> SessionId -> IO (Maybe Text)
  907 lastUserMessageAt paths sid = do
  908   -- model + fallbackTs are only used for synthesized conversation.jsonl
  909   -- entries that lack an entries.jsonl sidecar; the empty model + a
  910   -- neutral fallback keep that path consistent without affecting the
  911   -- entries.jsonl path (which carries real timestamps).
  912   entries <- readTranscriptEntries paths "" "" sid
  913   pure (lastRequestTimestamp entries)
  914   where
  915     -- Walk the frontend-shaped entries in reverse, return the first
  916     -- (i.e. most recent) @request@-direction entry's timestamp.
  917     lastRequestTimestamp :: [A.Value] -> Maybe Text
  918     lastRequestTimestamp vals =
  919       case reverse vals of
  920         [] -> Nothing
  921         xs -> go xs
  922       where
  923         k = Key.fromText
  924         isBlank (A.String t) = T.null t
  925         isBlank _ = True
  926         go [] = Nothing
  927         go (v : rest) =
  928           case v of
  929             A.Object o ->
  930               case ( KeyMap.lookup (k "direction") o
  931                    , KeyMap.lookup (k "timestamp") o ) of
  932                 (Just (A.String d), Just (A.String t))
  933                   | d == "request" && not (isBlank (A.String t)) -> Just t
  934                 _ -> go rest
  935             _ -> go rest

  938 -- ── Fast variants for buildListsSnapshot ───────────────────────────────
  939 --
  940 -- These functions avoid the full transcript read + parse + reconstruction
  941 -- that 'firstUserMessageSnippet' and 'lastUserMessageAt' perform. They are
  942 -- used by 'buildListsSnapshot' which calls them for EVERY session (704+
  943 -- sessions × 2.7MB avg transcript = ~2GB of I/O on every call with the
  944 -- slow versions). 'firstUserMessageSnippetFast' goes further: it streams
  945 -- the file line-by-line via conduit and stops as soon as the first user
  946 -- message is found, avoiding reading the entire file.

  948 -- | Fast variant of 'firstUserMessageSnippet': streams the conversation
  949 -- file line-by-line and stops after the first user message, instead of
  950 -- parsing the entire file. For 'conversation.jsonl' (the common case), each
  951 -- line is a JSON 'Message'; we decode lines one at a time and return the
  952 -- first @User@ message's text (truncated to 80 chars). For legacy
  953 -- 'transcript.jsonl', each line is a 'TranscriptEntry'; we extract request
  954 -- messages and return the first user text. Returns 'Nothing' when no
  955 -- transcript file exists or no user message is found.
  956 firstUserMessageSnippetFast :: SealPaths -> SessionId -> IO (Maybe Text)
  957 firstUserMessageSnippetFast paths sid = do
  958   let legacyPath = sessionTranscriptPath paths sid
  959       convPath   = sessionConversationPath paths sid
  960   convExists <- doesFileExist convPath
  961   if convExists
  962     then streamFirstMatch matchFirstUserMsg convPath
  963     else do
  964       legacyExists <- doesFileExist legacyPath
  965       if legacyExists
  966         then streamFirstMatch matchFirstLegacyUserMsg legacyPath
  967         else pure Nothing
  968   where
  969     matchFirstUserMsg bs =
  970       case A.decodeStrict bs :: Maybe Message of
  971         Just m | msgRole m == User -> snippetFromMessages [m]
  972         _ -> Nothing
  973     matchFirstLegacyUserMsg bs =
  974       case A.decodeStrict bs :: Maybe TranscriptEntry of
  975         Just te | teDirection te == Request ->
  976           case legacyRequestMessages [te] of
  977             (m : _) | msgRole m == User -> snippetFromMessages [m]
  978             _ -> Nothing
  979         _ -> Nothing

  981 -- | Fast variant of 'lastUserMessageAt': reads only 'entries.jsonl' (avg
  982 -- ~128KB, vs 2.7MB for conversation.jsonl) and finds the last
  983 -- 'EKRequest' entry's timestamp. Returns 'Nothing' when 'entries.jsonl'
  984 -- doesn't exist (the frontend falls back to @smLastActive@ for sorting).
  985 -- This avoids the full 'readTranscriptEntries' call which reads BOTH
  986 -- files, parses everything, and reconstructs entries.
  987 lastUserMessageAtFast :: SealPaths -> SessionId -> IO (Maybe Text)
  988 lastUserMessageAtFast paths sid = do
  989   let entriesPath = sessionEntriesPath paths sid
  990   entriesExist <- doesFileExist entriesPath
  991   if not entriesExist
  992     then pure Nothing
  993     else do
  994       raw <- readFileTextStrict entriesPath
  995       pure (lastRequestTs (parseEntryRecordsLazy raw))

  999 -- | Lazily parse non-empty lines as 'EntryRecord' values.
 1000 parseEntryRecordsLazy :: Text -> [EntryRecord]
 1001 parseEntryRecordsLazy raw =
 1002   mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
 1003            (filter (not . T.null) (T.lines raw))

 1005 -- | Find the timestamp of the last 'EKRequest' entry, formatted as ISO.
 1006 lastRequestTs :: [EntryRecord] -> Maybe Text
 1007 lastRequestTs = go Nothing
 1008   where
 1009     go acc [] = fmap (T.pack . showIso) acc
 1010     go acc (e : es) = case erKind e of
 1011       EKRequest -> go (Just (erTimestamp e)) es
 1012       _         -> go acc es

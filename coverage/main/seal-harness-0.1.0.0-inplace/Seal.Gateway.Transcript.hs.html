   53 -- | Which on-disk source produced the entries. Surfaced in 'TranscriptTimings'
   54 -- so the @Server-Timing@ header can name the path the request took.
   55 data TranscriptSource
   56   = TSMissing      -- ^ no transcript file exists
   57   | TSLegacy       -- ^ legacy @transcript.jsonl@
   58   | TSConvEntries  -- @conversation.jsonl@ + @entries.jsonl@ (reconstructed)
   59   | TSConvOnly     -- @conversation.jsonl@ only (no entries sidecar)
   60   deriving (Eq, Show)

   62 -- | Per-phase timings for a single 'readTranscriptEntries' call, in
   63 -- milliseconds. Phase durations are measured with 'getCurrentTime' (wall
   64 -- clock); for the small file sizes typical of a transcript, GHC's wall
   65 -- clock has enough resolution. Phases are exclusive of each other EXCEPT
   66 -- @ttTotal@, which spans the entire call. A phase that didn't run (e.g.
   67 -- @ttReconstruct@ when the legacy path was taken) is 0.
   68 data TranscriptTimings = TranscriptTimings
   69   { ttSource       :: TranscriptSource
   70   , ttEntryCount   :: Int           -- ^ number of frontend entries returned
   71   , ttFileReadMs   :: Integer       -- ^ time spent in 'TIO.readFile' (conv + entries files combined)
   72   , ttParseMs      :: Integer       -- ^ 'A.decode' of every line (both files combined)
   73   , ttReconstructMs :: Integer      -- @'reconstruct' (two-file path only)
   74   , ttRewriteMs    :: Integer        -- 'teLineToFrontend' / 'reconEntryToFrontend' / 'convLineToFrontend'
   75   , ttEncodeMs     :: Integer        -- ^ 'A.encode' of the frontend [Value] (set by 'handleTranscript', 0 for the bare reader)
   76   , ttTotalMs      :: Integer       -- ^ wall-clock total of the whole call
   77   }

   79 -- | Format a 'TranscriptTimings' as a @Server-Timing@ header value per
   80 -- <https://www.w3.org/TR/server-timing/>. Each phase is a separate
   81 -- comma-separated token with a short name, a numeric @dur@ (milliseconds,
   82 -- 3-decimal precision in the spec but we emit integers), and a @desc@
   83 -- naming the phase. The frontend parses these via @performance.getEntriesByName@
   84 -- or by reading the @Server-Timing@ response header directly.
   85 --
   86 -- Example output:
   87 --   @tt;dur=42;desc="total", fr;dur=3;desc="file-read", pr;dur=18;desc="parse", rc;dur=12;desc="reconstruct", rw;dur=4;desc="rewrite", en;dur=8;desc="encode", src;desc="conv+entries", n;desc="127"@
   88 renderServerTiming :: TranscriptTimings -> BC.ByteString
   89 renderServerTiming t =
   90   let phase name ms desc =
   91         BC.pack name <> ";dur=" <> BC.pack (show ms) <> ";desc=\"" <> BC.pack desc <> "\""
   92       tag name desc = BC.pack name <> ";desc=\"" <> BC.pack desc <> "\""
   93       srcDesc = case ttSource t of
   94         TSMissing      -> "missing"
   95         TSLegacy       -> "legacy"
   96         TSConvEntries  -> "conv+entries"
   97         TSConvOnly     -> "conv-only"
   98       parts =
   99         [ phase "tt" (ttTotalMs t)       "total"
  100         , phase "fr" (ttFileReadMs t)    "file-read"
  101         , phase "pr" (ttParseMs t)        "parse"
  102         , phase "rc" (ttReconstructMs t) "reconstruct"
  103         , phase "rw" (ttRewriteMs t)      "rewrite"
  104         , phase "en" (ttEncodeMs t)       "encode"
  105         , tag  "src" srcDesc
  106         , tag  "n"   (show (ttEntryCount t))
  107         ]
  108   in BC.intercalate ", " parts

  110 -- | Set the encode-phase duration + bump the total to span read+encode.
  111 -- 'handleTranscript' uses this after timing 'A.encode' so the @en@ token
  112 -- reflects the JSON-serialization cost (which dominates for transcripts
  113 -- carrying large web_fetch / shell-output tool_result blocks) and @tt@
  114 -- reflects the full read+encode cost.
  115 setEncodeMs :: Integer -> Integer -> TranscriptTimings -> TranscriptTimings
  116 setEncodeMs encodeMs totalMs tt = tt { ttEncodeMs = encodeMs, ttTotalMs = totalMs }

  118 -- | 'zipWith' + 'mapMaybe': apply a partial function across two lists
  119 -- in lockstep, dropping the elements for which the function returns
  120 -- 'Nothing'.
  121 zipWithMaybe :: (a -> b -> Maybe c) -> [a] -> [b] -> [c]
  122 zipWithMaybe _ [] _ = []
  123 zipWithMaybe _ _ [] = []
  124 zipWithMaybe f (a : as) (b : bs) = case f a b of
  125   Just c  -> c : zipWithMaybe f as bs
  126   Nothing -> zipWithMaybe f as bs

  136 -- | Instrumented variant of 'readTranscriptEntries': returns the frontend
  137 -- values alongside per-phase wall-clock timings. Used by the @/transcript@
  138 -- HTTP handler to emit a @Server-Timing@ header; other callers (broadcast,
  139 -- snippet helpers) can use the plain 'readTranscriptEntries' and ignore the
  140 -- overhead of timing capture.
  141 readTranscriptEntriesTimed
  142   :: SealPaths -> Text -> String -> SessionId -> IO ([Value], TranscriptTimings)
  143 readTranscriptEntriesTimed paths model fallbackTs sid = do
  144   tStart <- getCurrentTime
  145   let legacyPath = sessionTranscriptPath paths sid
  146       convPath   = sessionConversationPath paths sid
  147   legacyExists <- doesFileExist legacyPath
  148   convExists   <- doesFileExist convPath
  149   if legacyExists
  150     then do
  151       tFr0 <- getCurrentTime
  152       raw <- readFileTextStrict legacyPath
  153       tFr1 <- getCurrentTime
  154       tPr0 <- getCurrentTime
  155       let vals = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  156                           (filter (not . T.null) (T.lines raw))
  157       tPr1 <- getCurrentTime
  158       tRw0 <- getCurrentTime
  159       let frontend = map teLineToFrontend vals
  160       tRw1 <- getCurrentTime
  161       tEnd <- getCurrentTime
  162       let tt = TranscriptTimings
  163             { ttSource        = TSLegacy
  164             , ttEntryCount    = length frontend
  165             , ttFileReadMs    = ms tFr0 tFr1
  166             , ttParseMs       = ms tPr0 tPr1
  167             , ttReconstructMs = 0
  168             , ttRewriteMs     = ms tRw0 tRw1
  169             , ttEncodeMs      = 0
  170             , ttTotalMs       = ms tStart tEnd
  171             }
  172       pure (frontend, tt)
  173     else if convExists
  174       then do
  175         tFr0 <- getCurrentTime
  176         raw <- readFileTextStrict convPath
  177         tFr1 <- getCurrentTime
  178         tPr0 <- getCurrentTime
  179         let msgVals = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  180                                (filter (not . T.null) (T.lines raw)) :: [A.Value]
  181             msgs    = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  182                                (filter (not . T.null) (T.lines raw)) :: [Message]
  183         tPr1 <- getCurrentTime
  184         entriesExist <- doesFileExist (sessionEntriesPath paths sid)
  185         if entriesExist
  186           then do
  187             tFr2 <- getCurrentTime
  188             eraw <- readFileTextStrict (sessionEntriesPath paths sid)
  189             tFr3 <- getCurrentTime
  190             tPr2 <- getCurrentTime
  191             let evs = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  192                                (filter (not . T.null) (T.lines eraw)) :: [EntryRecord]
  193             tPr3 <- getCurrentTime
  194             tRc0 <- getCurrentTime
  195             let reconstructed = reconstruct msgs evs
  196                 reconFrontend = zipWithMaybe reconEntryToFrontend [0..] reconstructed
  197                 -- If entries.jsonl doesn't cover all conversation messages
  198                 -- (e.g. due to truncation or a crash during recording),
  199                 -- synthesize frontend entries for the remaining messages so
  200                 -- they're visible in the web UI instead of silently dropped.
  201                 maxConvLen = if null evs then 0 else maximum (map erConvLen evs)
  202                 trailing = trailingConvEntries model fallbackTs maxConvLen msgVals
  203                 frontend = reconFrontend <> trailing
  204             tRc1 <- getCurrentTime
  205             tEnd <- getCurrentTime
  206             let tt = TranscriptTimings
  207                   { ttSource        = TSConvEntries
  208                   , ttEntryCount    = length frontend
  209                   , ttFileReadMs    = ms tFr0 tFr1 + ms tFr2 tFr3
  210                   , ttParseMs       = ms tPr0 tPr1 + ms tPr2 tPr3
  211                   , ttReconstructMs = ms tRc0 tRc1
  212                   , ttRewriteMs     = 0  -- reconstruction includes rewrite
  213                   , ttEncodeMs      = 0
  214                   , ttTotalMs       = ms tStart tEnd
  215                   }
  216             pure (frontend, tt)
  217           else do
  218             tRw0 <- getCurrentTime
  219             let frontend = zipWith (convLineToFrontend model [] fallbackTs) [0..] msgVals
  220             tRw1 <- getCurrentTime
  221             tEnd <- getCurrentTime
  222             let tt = TranscriptTimings
  223                   { ttSource        = TSConvOnly
  224                   , ttEntryCount    = length frontend
  225                   , ttFileReadMs    = ms tFr0 tFr1
  226                   , ttParseMs       = ms tPr0 tPr1
  227                   , ttReconstructMs = 0
  228                   , ttRewriteMs     = ms tRw0 tRw1
  229                   , ttEncodeMs      = 0
  230                   , ttTotalMs       = ms tStart tEnd
  231                   }
  232             pure (frontend, tt)
  233       else do
  234         tEnd <- getCurrentTime
  235         let tt = TranscriptTimings
  236               { ttSource        = TSMissing
  237               , ttEntryCount    = 0
  238               , ttFileReadMs    = 0
  239               , ttParseMs       = 0
  240               , ttReconstructMs = 0
  241               , ttRewriteMs     = 0
  242               , ttEncodeMs      = 0
  243               , ttTotalMs       = ms tStart tEnd
  244               }
  245         pure ([], tt)
  246   where
  247     ms a b = round (realToFrac (b `diffUTCTime` a) * 1000 :: Double)

  249 -- | Extract a short snippet of the first user message in a session's
  250 -- transcript, for use as the default session title when the user has not
  251 -- set an explicit description. Reads the two-file format
  252 -- (@conversation.jsonl@) first, falling back to the legacy
  253 -- @transcript.jsonl@. Returns 'Nothing' when the session has no transcript
  254 -- or no user message with text content. The snippet is truncated to 80
  255 -- characters (on a codepoint boundary) to keep the sessions list compact.
  256 firstUserMessageSnippet :: SealPaths -> SessionId -> IO (Maybe Text)
  257 firstUserMessageSnippet paths sid = do
  258   let legacyPath = sessionTranscriptPath paths sid
  259       convPath   = sessionConversationPath paths sid
  260   legacyExists <- doesFileExist legacyPath
  261   convExists   <- doesFileExist convPath
  262   if convExists
  263     then do
  264       raw <- readFileTextStrict convPath
  265       let msgs = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  266                           (filter (not . T.null) (T.lines raw)) :: [Message]
  267       pure (snippetFromMessages msgs)
  268     else if legacyExists
  269       then do
  270         raw <- readFileTextStrict legacyPath
  271         let entries = mapMaybe (A.decode . BL.fromStrict . TE.encodeUtf8)
  272                                (filter (not . T.null) (T.lines raw)) :: [TranscriptEntry]
  273         pure (snippetFromMessages (legacyRequestMessages entries))
  274       else pure Nothing

  276 -- | Extract the first user message's text from a list of messages, truncated
  277 -- to 80 characters. Tool-use and tool-result blocks are skipped (only the
  278 -- first 'CbText' block of the first 'User' message is used).
  279 snippetFromMessages :: [Message] -> Maybe Text
  280 snippetFromMessages msgs =
  281   case filter (\m -> msgRole m == User) msgs of
  282     (m : _) -> case [t | CbText t <- msgContent m] of
  283       (t : _) -> Just (truncateSnippet 80 t)
  284       []      -> Nothing
  285     [] -> Nothing

  287 -- | Truncate a snippet to at most @n@ characters, appending an ellipsis when
  288 -- truncation occurs. 'T.take' is codepoint-safe for UTF-8 'Text'.
  289 truncateSnippet :: Int -> Text -> Text
  290 truncateSnippet n t
  291   | T.length t <= n = t
  292   | otherwise       = T.take n t <> "…"

  294 -- | Extract the request messages from legacy @transcript.jsonl@ entries.
  295 -- Only 'Request'-direction entries carry a payload with a @messages@ array;
  296 -- we collect messages from every request entry (the first user message is
  297 -- always in the first request).
  298 legacyRequestMessages :: [TranscriptEntry] -> [Message]
  299 legacyRequestMessages = concatMap go
  300   where
  301     go te
  302       | teDirection te /= Request = []
  303       | otherwise = case tePayload te of
  304           A.Object o -> case KeyMap.lookup (Key.fromText "messages") o of
  305             Just (A.Array arr) -> mapMaybe decodeMsg (V.toList arr)
  306             _ -> []
  307           _ -> []
  308     decodeMsg v = case A.fromJSON v :: A.Result Message of
  309       A.Success m -> Just m
  310       A.Error _   -> Nothing

  312 -- | Map a legacy on-disk transcript line (Haskell TranscriptEntry JSON
  313 -- with @te*@-prefixed fields) to the frontend's TranscriptEntry shape.
  314 teLineToFrontend :: A.Value -> A.Value
  315 teLineToFrontend rawLine =
  316   let o = case rawLine of
  317         A.Object m -> m
  318         _          -> KeyMap.empty
  319       k = Key.fromText
  320       lookupT key = case KeyMap.lookup (k key) o of
  321         Just (A.String t) -> Just t
  322         _                 -> Nothing
  323       -- The on-disk `payload` field is a JSON string; decode it to a Value
  324       -- so we can send it as a JSON object (not a string) in the wire
  325       -- response. Falls back to the raw string when it's not valid JSON
  326       -- (defensive against a malformed on-disk line).
  327       payloadVal = case KeyMap.lookup (k "payload") o of
  328         Just (A.String s) -> case A.decode (BL.fromStrict (TE.encodeUtf8 s)) of
  329           Just v  -> v
  330           Nothing -> A.String s
  331         other -> fromMaybe A.Null other
  332       -- The legacy transcript.jsonl entry's `meta` field carries the
  333       -- channel label (stamped by runTurn's requestMeta). Extract it so
  334       -- the frontend can attribute user messages to their channel on the
  335       -- legacy path too. Null when absent (response entries, old entries).
  336       mChannel = case KeyMap.lookup (k "meta") o of
  337         Just (A.Object mo) -> case KeyMap.lookup (k "channel") mo of
  338           Just (A.String ch) -> Just ch
  339           _                  -> Nothing
  340         _ -> Nothing
  341       mInternal = case KeyMap.lookup (k "meta") o of
  342         Just (A.Object mo) -> case KeyMap.lookup (k "internal") mo of
  343           Just (A.Bool b) -> Just b
  344           _               -> Nothing
  345         _ -> Nothing
  346   in object
  347      [ "id"        .= lookupT "id"
  348      , "timestamp" .= lookupT "timestamp"
  349      , "direction" .= lookupT "direction"
  350      , "payload"   .= payloadVal
  351      , "harness"   .= lookupT "correlation"
  352      , "model"     .= lookupT "model"
  353      , "channel"   .= mChannel
  354     , "internal"  .= mInternal
  355      , "raw"       .= TE.decodeUtf8 (BL.toStrict (A.encode rawLine))
  356      ]

  358 -- | Synthesize frontend TranscriptEntry JSON values for conversation
  359 -- messages that are NOT covered by any 'EntryRecord' (i.e. messages beyond
  360 -- the maximum @convLen@ across all entries). This handles incomplete
  361 -- @entries.jsonl@ files (e.g. due to truncation or a crash during
  362 -- recording) — without this, any conversation messages beyond the last
  363 -- entry's @convLen@ are silently dropped by 'reconstruct', making them
  364 -- invisible in the web UI. Each trailing message is converted via
  365 -- 'convLineToFrontend' (the same function used when @entries.jsonl@ does
  366 -- not exist at all), so User messages become @request@ entries and
  367 -- Assistant messages become @response@ entries with the same shape the
  368 -- frontend's @transcriptToMessages@ expects.
  369 trailingConvEntries
  370   :: Text     -- ^ model label (for the @model@ field in the frontend entry)
  371   -> String   -- ^ fallback timestamp (conversation.jsonl carries no timestamps)
  372   -> Int      -- ^ maxConvLen — the highest @convLen@ across all entry records
  373   -> [A.Value] -- ^ all conversation.jsonl lines (parsed as Aeson values)
  374   -> [A.Value] -- ^ frontend TranscriptEntry JSON for the trailing messages
  375 trailingConvEntries model fallbackTs maxConvLen msgVals =
  376   let convCount = length msgVals
  377   in if convCount > maxConvLen
  378        then zipWith (convLineToFrontend model [] fallbackTs)
  379                     [maxConvLen ..]
  380                     (drop maxConvLen msgVals)
  381        else []

  383 -- | Synthesize a frontend TranscriptEntry from a conversation.jsonl line
  384 -- (@role@/@content@). User → request; Assistant → response. The
  385 -- line index (0-based) is used as the entry id — conversation.jsonl
  386 -- carries no per-entry id, and the line index is stable across reads of
  387 -- the same file, so the frontend's dedup-by-id works.
  388 convLineToFrontend :: Text -> [String] -> String -> Int -> A.Value -> A.Value
  389 convLineToFrontend model entryTimestamps fallbackTs idx rawLine =
  390   let o = case rawLine of
  391         A.Object m -> m
  392         _          -> KeyMap.empty
  393       k = Key.fromText
  394       role = case KeyMap.lookup (k "role") o of
  395         Just (A.String t) -> t
  396         _                  -> "user"
  397       rawContent = case KeyMap.lookup (k "content") o of
  398         Just (A.Array arr) -> arr
  399         _                  -> mempty
  400       contentBlocks = map cbToFrontend (V.toList rawContent)
  401       content = A.Array (V.fromList contentBlocks)
  402       direction :: Text
  403       direction = if T.toCaseFold role == "user" then "request" else "response"
  404       payloadJson = if direction == "request"
  405         then A.object ["messages" A..= [A.object ["role" A..= ("user" :: Text), "content" A..= content]]]
  406         else A.object ["content" A..= content]
  407       entryId = T.pack (show idx)
  408       ts = fromMaybe fallbackTs (lookup idx (zip [0..] entryTimestamps))
  409     in object
  410        [ "id"        .= entryId
  411        , "timestamp" .= T.pack ts
  412        , "direction" .= direction
  413        , "payload"   .= payloadJson
  414        , "harness"   .= (Nothing :: Maybe Text)
  415        , "model"     .= model
  416        , "channel"   .= (Nothing :: Maybe Text)
  417       , "internal"  .= (Nothing :: Maybe Bool)
  418       , "raw"       .= TE.decodeUtf8 (BL.toStrict (A.encode rawLine))
  419       ]

  421 -- | Rewrite one on-disk 'ContentBlock' (GHC-Generics 'TaggedObject' shape)
  422 -- into the Anthropic-style block the frontend parses.
  423 cbToFrontend :: A.Value -> A.Value
  424 cbToFrontend blk =
  425   let bo = case blk of
  426         A.Object m -> m
  427         _          -> KeyMap.empty
  428       k = Key.fromText
  429       tag = case KeyMap.lookup (k "tag") bo of
  430         Just (A.String t) -> Just t
  431         _                  -> Nothing
  432       contents = KeyMap.lookup (k "contents") bo
  433       lookupT key m = case KeyMap.lookup (k key) m of
  434         Just (A.String t) -> Just t
  435         _                 -> Nothing
  436       lookupB key m = case KeyMap.lookup (k key) m of
  437         Just (A.Bool b) -> Just b
  438         _               -> Nothing
  439   in case tag of
  440        Just "CbText" -> case contents of
  441          Just (A.String t) -> object ["type" .= ("text" :: Text), "text" .= t]
  442          _                 -> fallback
  443        Just "CbThinking" -> case contents of
  444          Just (A.String t) -> object ["type" .= ("thinking" :: Text), "thinking" .= t]
  445          _                 -> fallback
  446        Just "CbToolUse" -> object
  447          [ "type"  .= ("tool_use" :: Text)
  448          , "id"    .= fromMaybe "" (lookupT "id" bo)
  449          , "name"  .= fromMaybe "" (lookupT "name" bo)
  450          , "input" .= fromMaybe A.Null (KeyMap.lookup (k "input") bo)
  451          ]
  452        Just "CbToolResult" -> object
  453          [ "type"        .= ("tool_result" :: Text)
  454          , "tool_use_id" .= fromMaybe "" (lookupT "forId" bo)
  455          , "content"     .= toolResultPartsToFrontend (KeyMap.lookup (k "parts") bo)
  456          , "is_error"    .= fromMaybe False (lookupB "isError" bo)
  457          ]
  458        _ -> fallback
  459   where
  460      fallback = object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]

  462 -- | Rewrite the on-disk 'parts' value into the Anthropic-style array
  463 -- the frontend parses (@[{type:"text", text:"..."}]@). 'ToolResultPart'
  464 -- is a @newtype TrpText Text@, so aeson's derived 'ToJSON' serializes it
  465 -- as a bare JSON string (not a 'TaggedObject'). The on-disk 'parts' is
  466 -- thus @["text"]@, not @[{tag:"TrpText", contents:"text"}]@. Handles both
  467 -- the bare-string shape and the object shape (defensive fallback).
  468 toolResultPartsToFrontend :: Maybe A.Value -> A.Value
  469 toolResultPartsToFrontend mparts =
  470   case mparts of
  471     Just (A.Array arr) ->
  472       A.Array (V.fromList (map trpToFrontend (V.toList arr)))
  473     other -> fromMaybe A.Null other

  475 trpToFrontend :: A.Value -> A.Value
  476 trpToFrontend blk =
  477   case blk of
  478     A.String t -> object ["type" .= ("text" :: Text), "text" .= t]
  479     A.Object m ->
  480       let tag = case KeyMap.lookup (Key.fromString "tag") m of
  481             Just (A.String t) -> Just t
  482             _                 -> Nothing
  483           contents = KeyMap.lookup (Key.fromString "contents") m
  484       in case tag of
  485            Just "TrpText" -> case contents of
  486              Just (A.String t) -> object ["type" .= ("text" :: Text), "text" .= t]
  487              _                 -> object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]
  488            _ -> object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]
  489     _ -> object ["type" .= ("text" :: Text), "text" .= TE.decodeUtf8 (BL.toStrict (A.encode blk))]

  491 -- | The set of opcode names whose 'EKHarness' entries should surface to
  492 -- the web frontend SPA as distinct harness entries (not be dropped by
  493 -- 'reconEntryToFrontend'). v1: @SKILL_LOAD@ so /skill load invocations
  494 -- appear in the transcript as identifiable skill-load events (per the
  495 -- user's "properly identified as a skill load operation" requirement);
  496 -- @SETUP_REPO@ so the clone/no-op/conflict/failure outcome of a
  497 -- repo-attached session creation is visible in the chat (the frontend's
  498 -- 'transcriptToMessages' renders it as a tool-call box; without this
  499 -- whitelist entry, the SETUP_REPO result recorded by
  500 -- 'recordSetupRepoResult' is dropped before reaching either the WS
  501 -- broadcast or the HTTP seed, so the user sees nothing for the repo
  502 -- setup).
  503 -- @ASK_HUMAN@ so pending questions from channel-originated turns
  504 -- (Telegram, Signal) surface in the web transcript. Without this, a
  505 -- question sent to a chat channel is invisible in the web UI — a
  506 -- "phantom message" that exists in the audit log but not the frontend.
  507 -- Approval-bearing entries always surface regardless of this whitelist.
  508 -- A 'Set' rather than a list to make the shared-state surface
  509 -- discoverable (see the design doc's §8 risk 2).
  510 userSurfacingOps :: Set.Set Text
  511 userSurfacingOps = Set.fromList ["SKILL_LOAD", "SKILL_MANAGE", "SETUP_REPO", "ASK_HUMAN"]
  512 
  513 -- | Predicate: does a harness payload's @op.name@ fall in
  514 -- 'userSurfacingOps'? Returns 'False' for payloads with no @op@ key, a
  515 -- non-object @op@, or no @name@ field — those are dropped by the filter
  516 -- (matching the pre-v1 behavior for non-approval harness entries).
  517 -- For @SKILL_MANAGE@, additionally requires @input.action == "load"@
  518 -- (other SKILL_MANAGE actions like write/list/delete are not user-surfacing).
  519 isUserSurfacingOp :: KeyMap.KeyMap Value -> Bool
  520 isUserSurfacingOp o =
  521   case KeyMap.lookup (Key.fromText "op") o of
  522     Just (Object opObj) ->
  523       case KeyMap.lookup (Key.fromText "name") opObj of
  524         Just (String n)
  525           | n == "SKILL_MANAGE" -> isSkillLoadAction o
  526           | otherwise -> n `Set.member` userSurfacingOps
  527         _               -> False
  528     _ -> False
  529 
  530 -- | Predicate: does a harness payload's @input.action@ equal @"load"@?
  531 -- Used to distinguish the user-surfacing skill-load action of
  532 -- @SKILL_MANAGE@ from its other actions (write/list/delete).
  533 isSkillLoadAction :: KeyMap.KeyMap Value -> Bool
  534 isSkillLoadAction o =
  535   case KeyMap.lookup (Key.fromText "input") o of
  536     Just (Object inputObj) ->
  537       case KeyMap.lookup (Key.fromText "action") inputObj of
  538         Just (String a) -> a == "load"
  539         _               -> False
  540     _ -> False

  542 -- | Map a reconstructed 'TranscriptEntry' (from 'reconstruct') to the
  543 -- frontend's TranscriptEntry JSON shape.
  544 reconEntryToFrontend :: Int -> TranscriptEntry -> Maybe A.Value
  545 reconEntryToFrontend idx te =
  546   case tePayload te of
  547     A.Null -> Nothing
  548     -- Drop harness entries (opcode invocations from the dispatcher) that
  549     -- carry a "harness" key in the payload — UNLESS they also carry an
  550     -- "approval" key (which marks them as approval-evidence entries that
  551     -- should surface so the user sees the confirmation decision in the
  552     -- transcript), OR their op.name is in 'userSurfacingOps' (which marks
  553     -- them as user-surfacing opcode invocations like /skill load that
  554     -- should appear as distinct harness entries, not be folded into the
  555     -- surrounding user/assistant bubbles).
  556     A.Object o
  557       | KeyMap.member (Key.fromText "harness") o,
  558         not (KeyMap.member (Key.fromText "approval") o),
  559         not (isUserSurfacingOp o) -> Nothing
  560     payloadVal -> Just $
  561       let dirStr = case teDirection te of
  562             Request  -> "request" :: Text
  563             Response -> "response"
  564           payloadJson = rewritePayload False payloadVal (teDirection te)
  565           -- The `raw` view preserves full tool input_schemas so the
  566           -- "View raw JSON" modal shows the verbatim schemas the LLM was
  567           -- sent (the `payload` field strips them to keep the wire small).
  568           rawJson = rewritePayload True payloadVal (teDirection te)
  569           entryId = let raw = teId te in if T.null raw then T.pack (show idx) else raw
  570           -- The originating channel label (e.g. "telegram", "web", "cli"),
  571           -- stamped into the request entry's erMeta by runTurn's
  572           -- requestMeta (and into SKILL_LOAD entries by
  573           -- recordSkillLoadResult). Surfaced as a top-level `channel` field
  574           -- so the frontend can attribute user messages — and skill loads —
  575           -- to the channel they came from. Null when the entry carries no
  576           -- channel (e.g. response entries, CLI TUI turns with no
  577           -- MessageSource).
  578           mChannel = case Map.lookup "channel" (teMeta te) of
  579             Just (A.String ch) -> Just ch
  580             _                  -> Nothing
  581           mInternal = case Map.lookup "internal" (teMeta te) of
  582             Just (A.Bool b) -> Just b
  583             _               -> Nothing
  584       in object
  585          [ "id"        .= entryId
  586          , "timestamp" .= T.pack (showIso (teTimestamp te))
  587          , "direction" .= dirStr
  588          -- `payload` is a JSON OBJECT (not a string) so the frontend can
  589          -- access its fields directly without a second JSON.parse pass.
  590          -- The old format encoded payload as a JSON string
  591          -- (TE.decodeUtf8 (A.encode payloadJson)), which forced the browser
  592          -- to parse the outer JSON (un-escaping every `\"` in the payload
  593          -- string), then call JSON.parse(payload) AGAIN to parse the inner
  594          -- JSON — two full parse passes with O(n) escape processing in
  595          -- between. For a 279KB body this took ~889ms; as an object it's a
  596          -- single pass and should be <10ms.
  597          , "payload"   .= payloadJson
  598          , "harness"   .= (Nothing :: Maybe Text)
  599          , "model"     .= (Nothing :: Maybe Text)
  600          , "channel"   .= mChannel
  601         , "internal"  .= mInternal
  602          -- The `raw` field carries the full rewritten payload with tool
  603          -- input_schemas preserved, so the frontend's "View raw JSON"
  604          -- modal shows the verbatim schemas the LLM was sent. The legacy
  605          -- path (teLineToFrontend) still ships the verbatim on-disk line.
  606          , "raw"       .= TE.decodeUtf8 (BL.toStrict (A.encode rawJson))
  607          ]
  608 
  609 -- | Rewrite a reconstructed payload 'Value' from GHC-Generics
  610 -- 'TaggedObject' encoding to the Anthropic-style JSON the frontend parses.
  611 -- When @preserveSchemas@ is 'True', the @tools@ array keeps its full
  612 -- @input_schema@/parameters fields — used to build the @raw@ view so the
  613 -- "View raw JSON" modal shows the verbatim schemas the LLM was sent. When
  614 -- 'False', 'toolsWithDescriptions' strips them to keep the wire payload
  615 -- small (the frontend's collapsed Tools row only needs names + descriptions).
  616 rewritePayload :: Bool -> A.Value -> Direction -> A.Value
  617 rewritePayload preserveSchemas val dir =
  618   case val of
  619     A.Object o ->
  620       let k = Key.fromText
  621           rewriteMsgs = case KeyMap.lookup (k "messages") o of
  622             Just (A.Array arr) ->
  623               let msgs = map rewriteMessage (V.toList arr)
  624               in ["messages" .= A.Array (V.fromList msgs)]
  625             _ -> []
  626           rewriteContent = case KeyMap.lookup (k "content") o of
  627             Just (A.Array arr) ->
  628               let blocks = map cbToFrontend (V.toList arr)
  629               in ["content" .= A.Array (V.fromList blocks)]
  630             _ -> []
  631           passthrough key = case KeyMap.lookup key o of
  632             Just v  -> [key .= v]
  633             Nothing -> []
  634           usageFields = case KeyMap.lookup (k "usage") o of
  635             Just (A.Object uo) ->
  636               let uIn  = case KeyMap.lookup (k "input")  uo of
  637                     Just n -> Just n; _ -> Nothing
  638                   uOut = case KeyMap.lookup (k "output") uo of
  639                     Just n -> Just n; _ -> Nothing
  640               in case (uIn, uOut) of
  641                    (Just _, Just _) ->
  642                      [ "usage" .= object
  643                        [ "input_tokens"  .= uIn
  644                        , "output_tokens" .= uOut
  645                        ]
  646                      ]
  647                    _ -> []
  648             _ -> []
  649           toolsField = if preserveSchemas
  650             then passthrough (k "tools")
  651             else toolsWithDescriptions o
  652           fields = case dir of
  653             Request ->
  654               passthrough (k "system")
  655                <> passthrough (k "model")
  656                -- Replace the full tool definitions with name + description,
  657                -- stripping input_schema/parameters to keep the wire payload
  658                -- small. The full definitions (with input_schema) are the
  659                -- same across every request turn and can be 11KB+ each;
  660                -- shipping them in all 73 request entries of a 146-turn
  661                -- session wastes ~840KB. The frontend's "Tools" collapsed
  662                -- row shows the names + descriptions, and the expanded
  663                -- view's JSON carries the name + description array (useful
  664                -- for seeing WHAT tools were available + what they do, just
  665                -- not the full schemas). The frontend already deduplicates
  666                -- tools via `seenTools` so only the first occurrence renders.
  667                -- The @raw@ view (built with preserveSchemas=True) ships the
  668                -- full tools array so the "View raw JSON" modal shows the
  669                -- verbatim schemas the LLM was sent.
  670                <> toolsField
  671                <> passthrough (k "toolChoice")
  672                <> passthrough (k "maxTokens")
  673                <> passthrough (k "approval")
  674                <> passthrough (k "op")
  675                -- Harness/opcode-invocation entries (EKHarness) carry the
  676                -- opcode's input + result under these keys (see
  677                -- Seal.Transcript.Reconstruct.harnessPayload). The SKILL_LOAD
  678                -- result entry is a Request-direction harness entry; without
  679                -- passing @input@ + @result@ through, the frontend's
  680                -- @transcriptToMessages@ sees @op.name = "SKILL_LOAD"@ but no
  681                -- @result@, so the skill-load tool-call box never renders.
  682                 <> passthrough (k "input")
  683                 <> passthrough (k "result")
  684                 <> passthrough (k "channel")
  685                 <> rewriteMsgs
  686             Response ->
  687               passthrough (k "model")
  688                <> rewriteContent
  689                <> usageFields
  690                <> passthrough (k "stop")
  691                <> passthrough (k "durationMs")
  692       in A.object fields
  693     _ -> val
  694 
  695 -- | Extract name + description from a tools array, dropping input_schema /
  696 -- parameters to keep the wire payload small. Returns
  697 -- @["tools" .= [...]]@ where each element is
  698 -- @{"name": "<toolName>", "description": "<toolDesc>"}@ — the shape the
  699 -- frontend's 'extractToolDefNames' parses. Returns @[]@ when the tools field
  700 -- is absent or not an array.
  701 toolsWithDescriptions :: KeyMap.KeyMap Value -> [(Key, Value)]
  702 toolsWithDescriptions o =
  703   case KeyMap.lookup (Key.fromText "tools") o of
  704     Just (A.Array arr) ->
  705       let defs = map toolNameDesc (V.toList arr)
  706       in ["tools" .= A.Array (V.fromList defs)]
  707     _ -> []
  708   where
  709     -- Extract {name, description} from a tool definition Value, dropping
  710     -- input_schema / parameters. Handles both the Anthropic shape
  711     -- ({name, description, input_schema}) and the Ollama shape
  712     -- ({type:"function", function:{name, description, parameters}}).
  713     toolNameDesc v = case v of
  714       A.Object td ->
  715         case ( KeyMap.lookup (Key.fromText "name") td
  716              , KeyMap.lookup (Key.fromText "description") td ) of
  717           (Just n, Just d) -> A.object ["name" .= n, "description" .= d]
  718           (Just n, Nothing) -> A.object ["name" .= n]
  719           (Nothing, _) -> case KeyMap.lookup (Key.fromText "function") td of
  720             Just (A.Object fn) -> case KeyMap.lookup (Key.fromText "name") fn of
  721               Just n -> case KeyMap.lookup (Key.fromText "description") fn of
  722                 Just d -> A.object ["name" .= n, "description" .= d]
  723                 Nothing -> A.object ["name" .= n]
  724               Nothing -> A.object ["name" .= A.Null]
  725             _ -> A.object ["name" .= A.Null]
  726       _ -> A.object ["name" .= A.Null]

  728 -- | Rewrite one 'Message' (@{role, content: [...]}@, strip-prefix camelCase
  729 -- JSON) to the Anthropic-style shape (@{role, content: [...]}@) the frontend
  730 -- parses, with content blocks rewritten by 'cbToFrontend'. The role is
  731 -- lowercased because GHC-Generics encodes 'User'/'Assistant' as
  732 -- @"User"@/@"Assistant"@ but the frontend checks @msg.role === "user"@.
  733 rewriteMessage :: A.Value -> A.Value
  734 rewriteMessage msg =
  735   case msg of
  736     A.Object mo ->
  737       let k = Key.fromText
  738           role = case KeyMap.lookup (k "role") mo of
  739             Just (A.String t) -> T.toLower t
  740             _                 -> "user" :: Text
  741           rawContent = case KeyMap.lookup (k "content") mo of
  742             Just (A.Array arr) -> arr
  743             _                  -> mempty
  744           blocks = map cbToFrontend (V.toList rawContent)
  745       in A.object
  746          [ "role"    .= role
  747          , "content" .= A.Array (V.fromList blocks)
  748          ]
  749     _ -> msg

  751 -- | ISO-8601 with milliseconds + Z (the frontend's expected timestamp shape).
  752 showIso :: UTCTime -> String
  753 showIso = formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%S%3QZ"

  755 -- | The ISO timestamp of the most recent @request@-direction entry in a
  756 -- session's transcript (the last time the USER sent a message), or 'Nothing'
  757 -- when the session has no transcript or no user message. Used by the sidebar
  758 -- to sort tabs by last-user-message time (oldest first within each status
  759 -- bucket). Reads the same transcript paths as 'firstUserMessageSnippet';
  760 -- returns the timestamp as an ISO string (matching the frontend's
  761 -- TranscriptEntry.timestamp shape).
  762 lastUserMessageAt :: SealPaths -> SessionId -> IO (Maybe Text)
  763 lastUserMessageAt paths sid = do
  764   -- model + fallbackTs are only used for synthesized conversation.jsonl
  765   -- entries that lack an entries.jsonl sidecar; the empty model + a
  766   -- neutral fallback keep that path consistent without affecting the
  767   -- entries.jsonl path (which carries real timestamps).
  768   entries <- readTranscriptEntries paths "" "" sid
  769   pure (lastRequestTimestamp entries)
  770   where
  771     -- Walk the frontend-shaped entries in reverse, return the first
  772     -- (i.e. most recent) @request@-direction entry's timestamp.
  773     lastRequestTimestamp :: [A.Value] -> Maybe Text
  774     lastRequestTimestamp vals =
  775       case reverse vals of
  776         [] -> Nothing
  777         xs -> go xs
  778       where
  779         k = Key.fromText
  780         isBlank (A.String t) = T.null t
  781         isBlank _ = True
  782         go [] = Nothing
  783         go (v : rest) =
  784           case v of
  785             A.Object o ->
  786               case ( KeyMap.lookup (k "direction") o
  787                    , KeyMap.lookup (k "timestamp") o ) of
  788                 (Just (A.String d), Just (A.String t))
  789                   | d == "request" && not (isBlank (A.String t)) -> Just t
  790                 _ -> go rest
  791             _ -> go rest

  794 -- ── Fast variants for buildListsSnapshot ───────────────────────────────
  795 --
  796 -- These functions avoid the full transcript read + parse + reconstruction
  797 -- that 'firstUserMessageSnippet' and 'lastUserMessageAt' perform. They are
  798 -- used by 'buildListsSnapshot' which calls them for EVERY session (704+
  799 -- sessions × 2.7MB avg transcript = ~2GB of I/O on every call with the
  800 -- slow versions). 'firstUserMessageSnippetFast' goes further: it streams
  801 -- the file line-by-line via conduit and stops as soon as the first user
  802 -- message is found, avoiding reading the entire file.
  803 
  804 -- | Fast variant of 'firstUserMessageSnippet': streams the conversation
  805 -- file line-by-line and stops after the first user message, instead of
  806 -- parsing the entire file. For 'conversation.jsonl' (the common case), each
  807 -- line is a JSON 'Message'; we decode lines one at a time and return the
  808 -- first @User@ message's text (truncated to 80 chars). For legacy
  809 -- 'transcript.jsonl', each line is a 'TranscriptEntry'; we extract request
  810 -- messages and return the first user text. Returns 'Nothing' when no
  811 -- transcript file exists or no user message is found.
  812 firstUserMessageSnippetFast :: SealPaths -> SessionId -> IO (Maybe Text)
  813 firstUserMessageSnippetFast paths sid = do
  814   let legacyPath = sessionTranscriptPath paths sid
  815       convPath   = sessionConversationPath paths sid
  816   convExists <- doesFileExist convPath
  817   if convExists
  818     then streamFirstMatch matchFirstUserMsg convPath
  819     else do
  820       legacyExists <- doesFileExist legacyPath
  821       if legacyExists
  822         then streamFirstMatch matchFirstLegacyUserMsg legacyPath
  823         else pure Nothing
  824   where
  825     matchFirstUserMsg bs =
  826       case A.decodeStrict bs :: Maybe Message of
  827         Just m | msgRole m == User -> snippetFromMessages [m]
  828         _ -> Nothing
  829     matchFirstLegacyUserMsg bs =
  830       case A.decodeStrict bs :: Maybe TranscriptEntry of
  831         Just te | teDirection te == Request ->
  832           case legacyRequestMessages [te] of
  833             (m : _) | msgRole m == User -> snippetFromMessages [m]
  834             _ -> Nothing
  835         _ -> Nothing

  837 -- | Fast variant of 'lastUserMessageAt': reads only 'entries.jsonl' (avg
  838 -- ~128KB, vs 2.7MB for conversation.jsonl) and finds the last
  839 -- 'EKRequest' entry's timestamp. Returns 'Nothing' when 'entries.jsonl'
  840 -- doesn't exist (the frontend falls back to @smLastActive@ for sorting).
  841 -- This avoids the full 'readTranscriptEntries' call which reads BOTH
  842 -- files, parses everything, and reconstructs entries.
  843 lastUserMessageAtFast :: SealPaths -> SessionId -> IO (Maybe Text)
  844 lastUserMessageAtFast paths sid = do
  845   let entriesPath = sessionEntriesPath paths sid
  846   entriesExist <- doesFileExist entriesPath
  847   if not entriesExist
  848     then pure Nothing
  849     else do
  850       raw <- readFileTextStrict entriesPath
  851       pure (lastRequestTs (parseEntryRecordsLazy raw))

  865     go acc [] = fmap (T.pack . showIso) acc

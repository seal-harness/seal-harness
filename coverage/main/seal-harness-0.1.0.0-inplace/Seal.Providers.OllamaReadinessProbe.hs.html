    2 -- | The I/O half of Ollama readiness: the HTTP probe, the TTL cache, the
    3 -- single-flight gate, the binary lookup and the RAM reader.
    4 --
    5 -- == The rules this module exists to enforce
    6 --
    7 -- 1. /Keyless./ Every request comes from
    8 --    'Seal.Providers.OllamaReadiness.probeTargetFor', never through
    9 --    @withHeaders@. The vault key cannot reach an operator-set @base_url@
   10 --    because no code path here can read the vault at all — there is no
   11 --    'Seal.Security.Vault.VaultHandle' anywhere in this module's interface.
   12 --
   13 -- 2. /The provider manager, never the gateway manager./ The manager is a
   14 --    parameter ('pcManager'). @gatewayManager@ installs a
   15 --    @managerModifyRequest@ that stamps @Authorization: Bearer \<service
   16 --    token\>@ onto every request; building the probe on it would ship that
   17 --    token to an operator-set URL. The wiring site passes the provider
   18 --    manager, and a test constructs the factory over a manager that stamps a
   19 --    sentinel header to prove the factory honours whatever it is given.
   20 --
   21 -- 3. /Bounded, twice./ A short per-request timeout (default 2 s) and a 256 KiB
   22 --    accumulating body cap. The timeout is a liveness requirement, not a
   23 --    nicety: the cache holds at most one in-flight probe /globally/, so a probe
   24 --    that never returns would hold that slot forever and silently starve every
   25 --    later refresh.
   26 --
   27 -- 4. /Never inline./ A request handler reads the cached 'IORef' and returns; it
   28 --    never performs the outbound call. Refreshes happen out of band.
   29 --
   30 -- All seams are fields of 'ProbesConfig' rather than positional arguments:
   31 -- production calls 'defaultProbesConfig' and tests override only the field
   32 -- under test. That is what makes the timeout, TTL, clock, PATH lookup and RAM
   33 -- reader hermetically testable — no real network, no real PATH, no 30 s sleeps.

   80 -- | The seam record. Every field is a seam so each rule above is testable in
   81 -- isolation; 'defaultProbesConfig' fills in production values.
   82 data ProbesConfig = ProbesConfig
   83   { pcManager      :: Manager
   84     -- ^ The PROVIDER manager (the @mgr@ from @Serve.hs@), never @gatewayManager@.
   85   , pcReadConfig   :: IO RuntimeConfig
   86     -- ^ Re-read on every refresh (D2.10), so an edited @base_url@ takes effect
   87     --   without a restart.
   88   , pcTimeoutMicro :: Int
   89     -- ^ Per-request timeout (D2.7), default two seconds.
   90   , pcTtlMicro     :: Int
   91     -- ^ Cache TTL (D2.5), default 30 s. Injectable so a test never sleeps.
   92   , pcNow          :: IO UTCTime
   93     -- ^ Injectable clock, so TTL expiry is a unit under test, not a wait.
   94   , pcIsExecutable :: Text -> IO (Maybe FilePath)
   95     -- ^ @findExecutable@ seam: tests must not consult the real PATH.
   96   , pcTotalRam     :: IO (Maybe Word64)
   97     -- ^ RAM reader seam (WU-B's fit predicate consumes it; WU-A stores it).
   98   }
   99 
  100 -- | Production seams: the only place a real manager, a real clock and a real
  101 -- PATH lookup are wired in.
  102 defaultProbesConfig :: Manager -> IO RuntimeConfig -> ProbesConfig
  103 defaultProbesConfig mgr loadCfg = ProbesConfig
  104   { pcManager      = mgr
  105   , pcReadConfig   = loadCfg
  106   , pcTimeoutMicro = probeTimeoutMicro
  107   , pcTtlMicro     = defaultTtlMicro
  108   , pcNow          = getCurrentTime
  109   , pcIsExecutable = findExecutable . T.unpack
  110   , pcTotalRam     = readTotalRam
  111   }

  113 -- | The cache TTL: 30 s (design D2.5).
  114 defaultTtlMicro :: Int
  115 defaultTtlMicro = 30_000_000

  117 -- | The read side of the seam, plus the actions behind it.
  118 data OllamaProbes = OllamaProbes
  119   { oprReadiness :: IO OllamaReadiness
  120     -- ^ Returns the cached value /immediately/, kicking an async refresh when
  121     --   the entry is older than the TTL. Never blocks on the network.
  122   , oprRefresh   :: IO ()
  123     -- ^ Forks a probe and returns immediately. The startup warm only.
  124   , oprTotalRam  :: IO (Maybe Word64)
  125     -- ^ Memoised on success (WU-B).
  126   }

  128 -- | The cache's mutable state. Every transition goes through
  129 -- 'atomicModifyIORef'' — never read-then-write, the race this repo has already
  130 -- been bitten by once (@claimToolCallIds@).
  131 data CacheState = CacheState
  132   { csReadiness :: !OllamaReadiness
  133   , csFetchedAt :: !(Maybe UTCTime)
  134   , csInFlight  :: !(Maybe Text)  -- ^ the single-flight key currently probing
  135   }
  136 
  137 -- | Build the seam. The returned actions close over one cache, so two sessions
  138 -- observing one base URL cannot disagree, and the startup warm and the handlers
  139 -- share it.
  140 mkOllamaProbes :: ProbesConfig -> IO OllamaProbes
  141 mkOllamaProbes cfg = do
  142   cache    <- newIORef CacheState
  143     { csReadiness = RdPending, csFetchedAt = Nothing, csInFlight = Nothing }
  144   ramCache <- newIORef Nothing
  145   pure OllamaProbes
  146     { oprReadiness = readWithRefresh cfg cache
  147     , oprRefresh   = void (forkIO (refresh cfg cache))
  148     , oprTotalRam  = memoisedRam cfg ramCache
  149     }
  150 
  151 -- | Read the cache; kick a refresh when stale or never probed. The value
  152 -- returned is whatever the cache holds /now/, so the caller never waits.
  153 readWithRefresh :: ProbesConfig -> IORef CacheState -> IO OllamaReadiness
  154 readWithRefresh cfg cache = do
  155   st  <- readIORef cache
  156   now <- pcNow cfg
  157   let ageMicro = fmap (micros . diffUTCTime now) (csFetchedAt st)
  158       stale = case ageMicro of
  159         Nothing   -> True
  160         Just age  -> age >= fromIntegral (pcTtlMicro cfg)
  161   if stale
  162     then do
  163       void (forkIO (refresh cfg cache))
  164       pure (csReadiness st)
  165     else pure (csReadiness st)
  166 
  167 -- | Microseconds in a 'Data.Time.Clock.NominalDiffTime' (which is a
  168 -- 'Data.Fixed.Pico'). Exact, no floating point.
  169 micros :: RealFrac a => a -> Integer
  170 micros d = round (d * 1_000_000)

  172 -- | One probe pass: re-read config, take the single-flight slot atomically,
  173 -- probe, store, release. The slot is released in both the success and the
  174 -- failure path — a leaked slot would starve every later refresh permanently.
  175 refresh :: ProbesConfig -> IORef CacheState -> IO ()
  176 refresh cfg cache = do
  177   runtimeCfg <- pcReadConfig cfg
  178   let baseUrl = fromMaybe defaultOllamaBaseUrl (providerBaseUrl runtimeCfg "ollama")
  179       key     = singleFlightKey baseUrl
  180   claimed <- claimSlot cache key
  181   when claimed $ do
  182     result <- try (probeOnce cfg baseUrl) :: IO (Either SomeException OllamaReadiness)
  183     now <- pcNow cfg
  184     atomicModifyIORef' cache $ const ( CacheState
  185       { csReadiness = fromRight RdUnknown result
  186       , csFetchedAt = Just now
  187       , csInFlight  = Nothing
  188       }, () )
  189 
  190 -- | Atomically take the single-flight slot, but only when nothing else holds
  191 -- it. A global cap of one (D2.6) means a second URL waits rather than running
  192 -- concurrently; a re-entry while a probe is in flight is a no-op.
  193 claimSlot :: IORef CacheState -> Text -> IO Bool
  194 claimSlot cache key = atomicModifyIORef' cache $ \st -> case csInFlight st of
  195   Just _  -> (st, False)
  196   Nothing -> (st { csInFlight = Just key }, True)
  197 
  198 -- | One probe: URL-shape/floor validation, then the HTTP round-trip.
  199 probeOnce :: ProbesConfig -> Text -> IO OllamaReadiness
  200 probeOnce cfg baseUrl = case hostKindFor baseUrl of
  201   -- ollama.com is never probed (D3.2): there is nothing local to observe.
  202   OhCloud -> pure RdCloud
  203   kind -> do
  204     binary <- binaryPresent cfg kind
  205     ram    <- pcTotalRam cfg
  206     tags   <- case probeTargetFor baseUrl of
  207       Left _    -> pure (Left OpeBlocked)
  208       Right req -> probeTagsWith cfg req
  209     pure (classifyOllamaReadiness OllamaProbe
  210       { opHostKind      = kind
  211       , opTags          = tags
  212       , opBinaryPresent = binary
  213       , opTotalRam      = ram
  214       })

  216 -- | D3.3: PATH is only meaningful for a loopback URL. Consulting it for a LAN
  217 -- or remote host would label a Docker-hosted daemon on another machine "not
  218 -- installed" on the strength of /our/ PATH.
  219 binaryPresent :: ProbesConfig -> OllamaHostKind -> IO Bool
  220 binaryPresent cfg kind = case kind of
  221   OhDefaultLoopback -> isJustPath <$> pcIsExecutable cfg "ollama"
  222   OhLoopback        -> isJustPath <$> pcIsExecutable cfg "ollama"
  223   _                 -> pure False
  224   where
  225     isJustPath (Just _) = True
  226     isJustPath Nothing  = False

  228 -- | Run the validated request and decode the tag list. The timeout is applied
  229 -- here from 'pcTimeoutMicro' so a test can shorten it without rebuilding the
  230 -- request.
  231 probeTagsWith :: ProbesConfig -> Request -> IO (Either OllamaProbeError [OllamaTag])
  232 probeTagsWith cfg req0 = do
  233   let req = req0 { responseTimeout = responseTimeoutMicro (pcTimeoutMicro cfg) }
  234   result <- try (readCapped (pcManager cfg) req) :: IO (Either HttpException (Int, ByteString))
  235   pure $ case result of
  236     Left e                    -> Left (classifyProbeFailure e)
  237     Right (code, body)        -> decodeStatus code body
  238 
  239 -- | One GET, with the body read as an /accumulating/ loop until 'BS.empty' or
  240 -- 'bodyCapBytes'. A single 'brReadSome' may return short and would truncate a
  241 -- healthy remote daemon into a permanent @unknown@; reading past the cap is
  242 -- 'HttpException'→'OpeBadResponse' territory, so the cap is signalled by
  243 -- returning the over-long body and letting 'decodeStatus' reject it.
  244 --
  245 -- http-client's 'withResponse' plus a manual loop is what makes the cap
  246 -- enforceable; 'httpLbs' would read the whole body first.
  247 readCapped :: Manager -> Request -> IO (Int, ByteString)
  248 readCapped mgr req = withResponse req mgr $ \resp -> do
  249   let code = statusCode (responseStatus resp)
  250   if code < 200 || code > 299
  251     then do
  252       -- A non-2xx body is still bounded, but only its presence matters: the
  253       -- status is the payload.
  254       _ <- drainCapped (responseBody resp)
  255       pure (code, BS.empty)
  256     else do
  257       body <- drainCapped (responseBody resp)
  258       pure (code, BL.toStrict body)
  259 
  260 -- | Accumulate until end-of-body or the cap, whichever comes first. Stops one
  261 -- chunk past the cap so 'decodeStatus' can tell "exactly at the cap" (fine)
  262 -- from "over it" (reject).
  263 drainCapped :: BodyReader -> IO BL.ByteString
  264 drainCapped br = go BL.empty
  265   where
  266     go acc
  267       | BL.length acc > fromIntegral bodyCapBytes = pure acc
  268       | otherwise = do
  269           chunk <- brReadSome br 32768
  270           if BL.null chunk
  271             then pure acc
  272             else go (acc <> chunk)
  273 
  274 -- | Decide from the status and the (already bounded) body.
  275 decodeStatus :: Int -> ByteString -> Either OllamaProbeError [OllamaTag]
  276 decodeStatus code body
  277   | code < 200 || code > 299  = Left (OpeHttp code)
  278   | BS.length body > bodyCapBytes = Left OpeBadResponse
  279   | otherwise = case parseTagsBody body of
  280       Left _  -> Left OpeBadResponse
  281       Right t -> Right t
  282 
  283 -- | The 256 KiB body cap (D2.3). A hostile or broken daemon cannot make the
  284 -- gateway allocate without bound.
  285 bodyCapBytes :: Int
  286 bodyCapBytes = 256 * 1024

  288 -- | Decode @{"models":[{"name":"..."}]}@. Pure, so the wire shape is testable
  289 -- without a listener. An absent @models@ key is an empty list (a daemon with
  290 -- nothing pulled is a real, reportable state); an entry without a @name@ makes
  291 -- the whole body malformed, because a tag list we cannot fully trust is worse
  292 -- than "unknown".
  293 parseTagsBody :: ByteString -> Either Text [OllamaTag]
  294 parseTagsBody raw = do
  295   value <- either (Left . T.pack) Right (eitherDecodeStrict' raw)
  296   either (Left . T.pack) Right (parseEither parseTags value)
  297 
  298 parseTags :: Value -> Parser [OllamaTag]
  299 parseTags = withObject "tags" $ \o -> do
  300   models <- o .:? "models" .!= []
  301   traverse parseOne models
  302   where
  303     -- `size` and `remote_host` are both optional: a daemon build may omit
  304     -- either, and an absent field must become `Nothing` (no evidence) rather
  305     -- than a decode failure, or a single odd model would collapse the whole
  306     -- list into "unknown" (design D5.3).
  307     parseOne = withObject "model" $ \m ->
  308       OllamaTag <$> m .: "name" <*> m .:? "size" <*> m .:? "remote_host"

  314 -- | Memoise the RAM reader, but /only/ on success: a transient PATH miss must
  315 -- not permanently disable the judgement (WU-B DoD 9).
  316 --
  317 -- The "only on success" half is guaranteed by the representation, which is
  318 -- worth stating so the missing guard does not read as an oversight: the cache
  319 -- is a @Maybe Word64@, so there is no way to record "I fetched this and got
  320 -- Nothing". A miss simply leaves the ref empty and the next call retries. The
  321 -- test below therefore pins the /observed/ behaviour (a miss does not
  322 -- short-circuit) rather than an unreachable branch.
  323 memoisedRam :: ProbesConfig -> IORef (Maybe Word64) -> IO (Maybe Word64)
  324 memoisedRam cfg ref = do
  325   cached <- readIORef ref
  326   case cached of
  327     Just r  -> pure (Just r)
  328     Nothing -> do
  329       r <- pcTotalRam cfg
  330       case r of
  331         Just v  -> writeIORef ref (Just v)
  332         Nothing -> pure ()
  333       pure r

  335 -- | Total RAM in bytes. macOS reads @sysctl -n hw.memsize@ (absolute path
  336 -- first, because @\/usr\/sbin@ is often absent from a launchd or GUI gateway's
  337 -- PATH; then a plain-PATH retry); Linux reads @MemTotal@ from @\/proc\/meminfo@.
  338 -- Any failure is 'Nothing', which degrades safely to "assume usable".
  339 readTotalRam :: IO (Maybe Word64)
  340 readTotalRam
  341   | os == "darwin" = readSysctlMem
  342   | otherwise = readProcMemTotal
  343 
  344 -- | @\/usr\/sbin\/sysctl -n hw.memsize@, falling back to a plain-PATH
  345 -- @sysctl@. 'readProcess' is a fixed-argv invocation of a specific trusted
  346 -- binary (no shell, no operator-supplied argument), matching the precedent of
  347 -- 'Seal.Tls.Cert' and friends.
  348 readSysctlMem :: IO (Maybe Word64)
  349 readSysctlMem = do
  350   r <- try (readProcess "/usr/sbin/sysctl" ["-n", "hw.memsize"] "")
  351   case r of
  352     Right out -> pure (parseWord64 out)
  353     Left (_ :: SomeException) -> do
  354       r2 <- try (readProcess "sysctl" ["-n", "hw.memsize"] "")
  355       case r2 of
  356         Right out -> pure (parseWord64 out)
  357         Left (_ :: SomeException) -> pure Nothing
  358 
  359 parseWord64 :: String -> Maybe Word64
  360 parseWord64 s = case reads (filter (`notElem` ("\r\n " :: String)) s) of
  361   [(n, "")] -> Just n
  362   _         -> Nothing
  363 
  364 -- | Linux: @MemTotal@ from @\/proc\/meminfo@.
  365 readProcMemTotal :: IO (Maybe Word64)
  366 readProcMemTotal = do
  367   r <- try (readFile memTotalPath)
  368   pure $ case r of
  369     Left (_ :: SomeException) -> Nothing
  370     Right contents -> firstMatch (map (parseMemTotal . T.pack) (lines contents))
  371 
  372 firstMatch :: [Maybe Word64] -> Maybe Word64
  373 firstMatch [] = Nothing
  374 firstMatch (Just v : _) = Just v
  375 firstMatch (Nothing : rest) = firstMatch rest
  376 
  377 memTotalPath :: FilePath
  378 memTotalPath = "/proc/meminfo"

   48 -- | The testability seam over the Telegram Bot API. The channel layer calls
   49 -- 'tgReceive' to pull the next parsed update (blocks until one arrives or
   50 -- the transport closes), 'tgSend' to send a message (chat id + body),
   51 -- 'tgSetCommands' to register the bot's slash-command menu with BotFather
   52 -- (for auto-completion), and 'tgClose' to clean up.

   56     -- close/failure (the reader thread stops).

   62     -- @parse_mode@ (gate: Security #3 — plain text).

  100 -- | A parsed inbound Telegram update: the conversation id (from chat.id),
  101 -- the sender's user id (from from.id), and the message body. The
  102 -- conversation id is server-derived from authenticated transport metadata
  103 -- (the Telegram @chat.id@ field), never read from the message body. The
  104 -- raw @chatId@ is also carried so the channel can address replies without
  105 -- stripping the @tg:@ conversation-id prefix.
  106 --
  107 -- When the update is a @callback_query@ (a button tap), 'tuCallbackData'
  108 -- is @Just data@ + 'tuCallbackId' is @Just id@; 'tuBody' is the
  109 -- callback_data (the loop uses it to route). When the update is a regular
  110 -- @message@, both are @Nothing@ + 'tuBody' is the message text.
  111 data TelegramUpdate = TelegramUpdate
  112   { tuConversationId :: ConversationId
  113   , tuChatId          :: Text
  114   , tuSender          :: UserId
  115   , tuBody            :: Text
  116   , tuCallbackData    :: Maybe Text  -- ^ @Just data@ for callback_query; @Nothing@ for message
  117   , tuCallbackId      :: Maybe Text  -- ^ @Just id@ for callback_query (for answerCallbackQuery); @Nothing@ for message
  118   , tuCallbackMessageId :: Maybe Text
  119   -- ^ @Just msgId@ for callback_query (the message holding the keyboard,
  120   -- so we can remove it after a tap); @Nothing@ for message.
  121   } deriving stock (Eq, Show)

  127 -- | A mock transport backed by a 'TQueue' of inbound 'TelegramUpdate's and an
  128 -- 'IORef' of captured sends. 'tgReceive' pops the next inbound (or returns
  129 -- @Left "inbox empty"@); 'tgSend' appends @(chatId, body)@ to the capture;
  130 -- 'tgSetCommands' captures the last registered commands; 'tgClose' is a
  131 -- no-op (idempotent). 'tgSendWithKeyboard' + 'tgAnswerCallback' capture to
  132 -- separate IORefs (for test assertions). Returns the transport + an action
  133 -- to read captured sends + an action to read captured commands + an action
  134 -- to read captured callback acknowledgements + an action to read captured
  135 -- keyboard sends (each @(chatId, body, keyboard)@).
  136 mkMockTelegramTransport
  137   :: [TelegramUpdate]
  138   -> IO ( TelegramTransport, IO [(Text, Text)], IO [BotCommand]
  139         , IO [Text], IO [(Text, Text, [[TelegramButton]])] )
  140 mkMockTelegramTransport scripted = do
  141   q <- newTQueueIO
  142   mapM_ (atomically . writeTQueue q) scripted
  143   capRef <- newIORef []
  144   cmdRef <- newIORef []
  145   kbRef <- newIORef []
  146   cbRef <- newIORef []
  147   let transport = TelegramTransport
  148         { tgReceive = do
  149             m <- atomically (tryReadTQueue q)
  150             case m of
  151               Just u  -> pure (Right u)
  152               Nothing -> pure (Left "telegram inbox empty")
  153         , tgSend = \c b -> modifyIORef' capRef ((c, b) :)
  154         , tgSendWithKeyboard = \c b kb -> modifyIORef' kbRef ((c, b, kb) :)
  155         , tgSetCommands = writeIORef cmdRef
  156         , tgAnswerCallback = \cbId -> modifyIORef' cbRef (cbId :)
  157         , tgEditReplyMarkup = \_c _m -> pure ()  -- mock: no-op
  158         , tgClose = pure ()
  159         }
  160       getCaptured = reverse <$> readIORef capRef
  161       getCommands = readIORef cmdRef
  162       getCallbacks = reverse <$> readIORef cbRef
  163       getKeyboards = reverse <$> readIORef kbRef
  164   pure (transport, getCaptured, getCommands, getCallbacks, getKeyboards)

  174 -- | Spawn the real Telegram transport: long-polls @getUpdates@ and sends via
  175 -- @sendMessage@. 'tgReceive' blocks on @getUpdates@ (30s long-poll), parses
  176 -- each update into a 'TelegramUpdate', and advances the @offset@ so
  177 -- acknowledged updates are not re-delivered. 'tgSend' calls @sendMessage@.
  178 -- 'tgSetCommands' calls @setMyCommands@ to register the bot's slash-command
  179 -- menu for auto-completion. 'tgClose' is a no-op (long-polling is stateless;
  180 -- no child process to kill). The transport maintains an internal buffer
  181 -- ('TQueue') of parsed updates from the last @getUpdates@ call, refilling
  182 -- when it drains.
  183 mkRealTelegramTransport :: Text -> Manager -> IO TelegramTransport
  184 mkRealTelegramTransport token mgr = do
  185   buffer <- newTQueueIO
  186   offsetRef <- newIORef (0 :: Int)
  187   pure TelegramTransport
  188     { tgReceive = fillAndReceive buffer offsetRef
  189     , tgSend = sendViaApi mgr token
  190     , tgSendWithKeyboard = tgSendWithKeyboardViaApi mgr token
  191     , tgSetCommands = \cmds -> do
  192         eRes <- setMyCommandsViaApi mgr token cmds
  193         case eRes of
  194           Left err -> globalLogIO ErrorS ("telegram setMyCommands failed: " <> ls err)
  195           Right _  -> pure ()
  196     , tgAnswerCallback = answerCallbackQueryViaApi mgr token
  197     , tgEditReplyMarkup = editReplyMarkupViaApi mgr token
  198     , tgClose = pure ()
  199     }
  200   where
  201     -- If the buffer is empty, call getUpdates to refill it, then pop one.
  202     fillAndReceive buffer offsetRef = do
  203       m <- atomically (tryReadTQueue buffer)
  204       case m of
  205         Just u  -> pure (Right u)
  206         Nothing -> do
  207           offset <- readIORef offsetRef
  208           eResult <- getUpdates mgr token offset
  209           case eResult of
  210             Left err -> pure (Left err)
  211             Right (updates, rawIds) ->
  212               if null updates
  213                 then
  214                   -- No parseable updates. Still advance the offset past
  215                   -- any raw update_ids so we don't loop forever on
  216                   -- unparseable updates.
  217                   if null rawIds
  218                     then fillAndReceive buffer offsetRef
  219                     else do
  220                       let lastId = maximum rawIds
  221                       modifyIORef' offsetRef (const (lastId + 1))
  222                       fillAndReceive buffer offsetRef
  223                 else do
  224                   let lastId = maximum (map fst updates)
  225                   modifyIORef' offsetRef (const (lastId + 1))
  226                   mapM_ ((atomically . writeTQueue buffer) . snd) updates
  227                   fillAndReceive buffer offsetRef

  229 -- | Call @getUpdates@ with long-polling (30s timeout). Returns the parsed
  230 -- updates as @(update_id, TelegramUpdate)@ pairs — @message@ and
  231 -- @callback_query@ updates are parsed; other update types are skipped.
  232 -- Explicitly requests @message@ + @callback_query@ via @allowed_updates@
  233 -- so button taps are delivered (without this, Telegram uses the previous
  234 -- setting, which may exclude @callback_query@).
  235 getUpdates :: Manager -> Text -> Int -> IO (Either Text ([(Int, TelegramUpdate)], [Int]))
  236 -- ^ Returns @(parsedUpdates, allRawUpdateIds)@. The raw ids let the caller
  237 -- advance the offset even when some updates fail to parse (e.g. an
  238 -- unparseable callback_query), avoiding an infinite loop re-receiving
  239 -- them.
  240 getUpdates mgr token offset = do
  241   let url = T.unpack (telegramApiBase <> token <> "/getUpdates")
  242              <> "?offset=" <> show offset <> "&timeout=30"
  243   eReq <- try @SomeException (parseRequest url)
  244   case eReq of
  245     Left ex -> pure (Left ("getUpdates request error: " <> T.pack (show ex)))
  246     Right req0 -> do
  247       let req = req0 { responseTimeout = responseTimeoutMicro 60000000 }
  248       eResp <- try @SomeException (httpLbs req mgr)
  249       case eResp of
  250         Left ex -> pure (Left ("getUpdates network error: " <> T.pack (show ex)))
  251         Right resp ->
  252           let code = statusCode (responseStatus resp)
  253               body = responseBody resp
  254           in if code == 200
  255                then pure (parseGetUpdatesResponse body)
  256                else pure (Left ("getUpdates returned HTTP " <> T.pack (show code)
  257                               <> " — " <> T.pack (show body)))
  258 
  259 -- | Parse the JSON response from getUpdates. Returns @(parsedUpdates,
  260 -- allRawUpdateIds)@. The raw ids let the caller advance the offset even
  261 -- when some updates fail to parse (e.g. an unparseable callback_query).
  262 parseGetUpdatesResponse :: BL.ByteString -> Either Text ([(Int, TelegramUpdate)], [Int])
  263 parseGetUpdatesResponse body =
  264   case A.decode body of
  265     Nothing -> Left "getUpdates: malformed JSON response"
  266     Just (A.Object o) -> case KeyMap.lookup (Key.fromString "result") o of
  267       Just (A.Array arr) ->
  268         let raws = V.toList arr
  269             allIds = [ uid | v <- raws, Just uid <- [rawUpdateId v] ]
  270             parsed = [ parseOneUpdate v | v <- raws ]
  271             okUpdates = [ (uid, u) | Right (uid, u) <- parsed ]
  272         in Right (okUpdates, allIds)
  273       _ -> Right ([], [])
  274     Just _ -> Left "getUpdates: response not an object"
  275   where
  276     parseOneUpdate v = do
  277       uid <- updateId v
  278       case parseTelegramUpdate v of
  279         Left err -> Left err
  280         Right u  -> Right (uid, u)
  281 
  282 -- | Extract the raw @update_id@ from an update object (for offset
  283 -- advancement even when the update body fails to parse).
  284 rawUpdateId :: Value -> Maybe Int
  285 rawUpdateId v =
  286   case v of
  287     A.Object o -> case KeyMap.lookup (Key.fromString "update_id") o of
  288       Just (A.Number n) -> Just (round n)
  289       _ -> Nothing
  290     _ -> Nothing

  292 -- | Extract the numeric @update_id@ from a Telegram update object.
  293 updateId :: Value -> Either Text Int
  294 updateId v =
  295   case v of
  296     A.Object o -> case KeyMap.lookup (Key.fromString "update_id") o of
  297       Just (A.Number n) -> Right (round n)
  298       _ -> Left "update missing update_id"
  299     _ -> Left "update not an object"

  301 -- | Parse a raw Telegram update into a 'TelegramUpdate'. Handles two shapes:
  302 --
  303 -- 1. @message@ — a regular text message. 'tuCallbackData'/'tuCallbackId' are
  304 --    @Nothing@; 'tuBody' is the message text.
  305 -- 2. @callback_query@ — a button tap on an inline keyboard.
  306 --    'tuCallbackData' is @Just data@ (the callback_data); 'tuCallbackId' is
  307 --    @Just id@ (the callback_query id, for answerCallbackQuery);
  308 --    'tuBody' is the callback_data (the loop routes by it); the chat id +
  309 --    sender come from @callback_query.message.chat@ / @callback_query.from@.
  310 --
  311 -- Skips non-message/non-callback updates with a 'Left'.
  312 parseTelegramUpdate :: Value -> Either Text TelegramUpdate
  313 parseTelegramUpdate v =
  314   case v of
  315     A.Object o ->
  316       case KeyMap.lookup (Key.fromString "callback_query") o of
  317         Just cq -> parseCallbackQuery cq
  318         Nothing -> case KeyMap.lookup (Key.fromString "message") o of
  319           Just m  -> parseMessage m
  320           Nothing -> Left "update has no message or callback_query field"
  321     _ -> Left "update not an object"
  322   where
  323     parseMessage msg =
  324       case msg of
  325         A.Object mo -> do
  326           chatId <- requireChatId mo
  327           cid <- case mkConversationId ("tg:" <> chatId) of
  328             Right c -> Right c
  329             Left err -> Left ("conversation id construction failed: " <> err)
  330           sender <- requireSender mo
  331           let body = extractText mo
  332           Right TelegramUpdate
  333             { tuConversationId = cid
  334             , tuChatId          = chatId
  335             , tuSender          = sender
  336             , tuBody            = body
  337             , tuCallbackData    = Nothing
  338             , tuCallbackId      = Nothing
  339             , tuCallbackMessageId = Nothing
  340             }
  341         _ -> Left "message not an object"
  342     parseCallbackQuery cq =
  343       case cq of
  344         A.Object cqo -> do
  345           -- callback_query has: id (a String per the Bot API), from,
  346           -- message (with chat), data.
  347           callbackId <- case KeyMap.lookup (Key.fromString "id") cqo of
  348             Just (A.String t) -> Right t
  349             Just (A.Number n) -> Right (T.pack (show (round n :: Int)))
  350             _ -> Left "callback_query.id missing"
  351           callbackData <- case KeyMap.lookup (Key.fromString "data") cqo of
  352             Just (A.String t) -> Right t
  353             _ -> Left "callback_query.data missing"
  354           -- chat.id comes from callback_query.message.chat.id
  355           msg <- case KeyMap.lookup (Key.fromString "message") cqo of
  356             Just m -> Right m
  357             Nothing -> Left "callback_query has no message field"
  358           case msg of
  359             A.Object mo -> do
  360               chatId <- requireChatId mo
  361               cid <- case mkConversationId ("tg:" <> chatId) of
  362                 Right c -> Right c
  363                 Left err -> Left ("conversation id construction failed: " <> err)
  364               sender <- case KeyMap.lookup (Key.fromString "from") cqo of
  365                 Just (A.Object fo) -> case KeyMap.lookup (Key.fromString "id") fo of
  366                   Just (A.Number n) -> case mkUserId (T.pack (show (round n :: Int))) of
  367                     Right u  -> Right u
  368                     Left err -> Left ("telegram callback sender not a valid UserId: " <> err)
  369                   _ -> Left "callback_query.from.id missing"
  370                 _ -> Left "callback_query.from field missing"
  371               -- The message_id of the message holding the keyboard (so the
  372               -- reader can remove the keyboard after a tap).
  373               mMsgId <- case KeyMap.lookup (Key.fromString "message_id") mo of
  374                 Just (A.Number n) -> Right (T.pack (show (round n :: Int)))
  375                 _ -> Left "callback_query.message.message_id missing"
  376               Right TelegramUpdate
  377                 { tuConversationId = cid
  378                 , tuChatId          = chatId
  379                 , tuSender          = sender
  380                 , tuBody            = callbackData
  381                 , tuCallbackData    = Just callbackData
  382                 , tuCallbackId      = Just callbackId
  383                 , tuCallbackMessageId = Just mMsgId
  384                 }
  385             _ -> Left "callback_query.message not an object"
  386         _ -> Left "callback_query not an object"
  387 
  388 -- | Extract @chat.id@ from a message object. Telegram chat ids are integers;
  389 -- we stringify them for the conversation id prefix.
  390 requireChatId :: A.Object -> Either Text Text
  391 requireChatId mo =
  392   case KeyMap.lookup (Key.fromString "chat") mo of
  393     Just (A.Object co) -> case KeyMap.lookup (Key.fromString "id") co of
  394       Just (A.Number n) -> Right (T.pack (show (round n :: Int)))
  395       _ -> Left "chat.id missing"
  396     _ -> Left "chat field missing"

  398 -- | Extract @from.id@ from a message object.
  399 requireSender :: A.Object -> Either Text UserId
  400 requireSender mo =
  401   case KeyMap.lookup (Key.fromString "from") mo of
  402     Just (A.Object fo) -> case KeyMap.lookup (Key.fromString "id") fo of
  403       Just (A.Number n) -> case mkUserId (T.pack (show (round n :: Int))) of
  404         Right u  -> Right u
  405         Left err -> Left ("telegram sender not a valid UserId: " <> err)
  406       _ -> Left "from.id missing"
  407     _ -> Left "from field missing"

  417 -- | Send a message via the Bot API @sendMessage@.
  418 sendViaApi :: Manager -> Text -> Text -> Text -> IO ()
  419 sendViaApi mgr token chatId body = do
  420   eReq <- try @SomeException
  421     (parseRequest (T.unpack (telegramApiBase <> token <> "/sendMessage")))
  422   case eReq of
  423     Left ex -> globalLogIO WarningS ("telegram send: request error: " <> ls (T.pack (show ex)))
  424     Right req0 -> do
  425       let payload = A.object
  426             [ "chat_id" A..= chatId
  427             , "text"   A..= body
  428             ]
  429           req = req0 { method = methodPost
  430                      , requestBody = RequestBodyLBS (A.encode payload)
  431                      , requestHeaders = [("Content-Type", "application/json")]
  432                      }
  433       _ <- try @SomeException (httpLbs req mgr)
  434       pure ()

  508 -- | Register the bot's command menu via @setMyCommands@ so Telegram shows
  509 -- auto-completion for the bot's slash commands. Calls the Bot API with a
  510 -- JSON array of @{command, description}@ objects. Returns 'Left' with a
  511 -- diagnostic on failure (the bot still works without auto-completion); the
  512 -- caller logs the error.
  513 setMyCommandsViaApi :: Manager -> Text -> [BotCommand] -> IO (Either Text ())
  514 setMyCommandsViaApi mgr token commands = do
  515   eReq <- try @SomeException
  516     (parseRequest (T.unpack (telegramApiBase <> token <> "/setMyCommands")))
  517   case eReq of
  518     Left ex -> pure (Left ("request error: " <> T.pack (show ex)))
  519     Right req0 -> do
  520       let cmds = [ A.object [ "command" A..= bcName bc
  521                             , "description" A..= bcDescription bc
  522                             ]
  523                  | bc <- commands
  524                  ]
  525           payload = A.object [ "commands" A..= cmds ]
  526           req = req0 { method = methodPost
  527                      , requestBody = RequestBodyLBS (A.encode payload)
  528                      , requestHeaders = [("Content-Type", "application/json")]
  529                      }
  530       eResp <- try @SomeException (httpLbs req mgr)
  531       case eResp of
  532         Left ex -> pure (Left ("network error: " <> T.pack (show ex)))
  533         Right resp -> do
  534           let code = statusCode (responseStatus resp)
  535           if code == 200
  536             then pure (Right ())
  537             else pure (Left ("HTTP " <> T.pack (show code) <> " — " <> T.pack (show (responseBody resp))))

  539 -- ---------------------------------------------------------------------------
  540 -- chunkMessage — split long messages for Telegram's 4096-char limit
  541 -- ---------------------------------------------------------------------------
  542 
  543 -- | Split a message into chunks of at most 'limit' characters, preferring
  544 -- paragraph boundaries (@\\n\\n@), then line boundaries (@\\n@), hard-cut
  545 -- as a last resort. Chunks carry their trailing separator (except the last),
  546 -- so 'T.concat' of the chunks is identity. Mirrors
  547 -- 'Seal.Channels.Signal.Transport.chunkMessage'.
  548 chunkMessage :: Int -> Text -> [Text]
  549 chunkMessage limit t
  550   | limit < 1 = error "chunkMessage: limit must be >= 1"
  551   | T.null t  = []
  552   | otherwise = go t
  553   where
  554     go s
  555       | T.null s       = []
  556       | T.length s <= limit = [s]
  557       | otherwise =
  558           let (chunk, rest) = nextChunk limit s
  559           in chunk : if T.null rest then [] else go rest
  560 
  561 nextChunk :: Int -> Text -> (Text, Text)
  562 nextChunk limit s =
  563   case findParagraphBreak limit s of
  564     Just n -> T.splitAt n s
  565     Nothing -> case findLineBreak limit s of
  566       Just n -> T.splitAt n s
  567       Nothing -> T.splitAt limit s
  568 
  569 findParagraphBreak :: Int -> Text -> Maybe Int
  570 findParagraphBreak limit s =
  571   let window = T.take limit s
  572   in case T.breakOnEnd "\n\n" window of
  573        (pre, _post) | not (T.null pre) -> Just (T.length pre)
  574        _ -> Nothing
  575 
  576 findLineBreak :: Int -> Text -> Maybe Int
  577 findLineBreak limit s =
  578   let window = T.take limit s
  579   in case T.breakOnEnd "\n" window of
  580        (pre, _post) | not (T.null pre) -> Just (T.length pre)
  581        _ -> Nothing

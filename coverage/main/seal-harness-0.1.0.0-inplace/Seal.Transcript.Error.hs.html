    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The transcript writer's exception type, in a leaf module.
    3 --
    4 -- Lives here rather than in 'Seal.Handles.Transcript' so that
    5 -- 'Seal.Transcript.TurnFailure.classify' can probe it without the pure
    6 -- classification module depending on @Seal.Handles.*@. That keeps
    7 -- @Seal.Transcript.*@ a leaf layer (design §6, CTO suggestion 1): classifying
    8 -- from @Seal.Session.TurnFailure@ instead would have imported
    9 -- @Handles.Transcript@ into @Session.@, and @Handles.Transcript@ already
   10 -- imports @Session.Log@, so the next change wanting to log a failure would
   11 -- have closed the cycle.
   12 --
   13 -- The type is re-exported from 'Seal.Handles.Transcript' so every existing
   14 -- import site keeps working unchanged.
   15 module Seal.Transcript.Error
   16   ( TranscriptError (..)
   17   ) where
   18 
   19 import Control.Exception (Exception, SomeException)
   20 import Data.Text (Text)
   21 import Data.Text qualified as T
   22 
   23 -- | Raised by @itwRecordAndAck@ when the single-writer daemon has died (an I/O
   24 -- error in @writeOne@ killed it), or when a write cannot be acknowledged.
   25 --
   26 -- The optional inner exception is the daemon's own fatal cause, carried up so
   27 -- the user-facing failure report can name it. Without it the caller only ever
   28 -- sees "the transcript became unavailable", which says nothing about /why/ —
   29 -- the exact problem that made session @20261127-133020-166@ look like a silent
   30 -- hang. See 'Seal.Handles.Transcript.itwLastFailure' for the read side.
   31 --
   32 -- 'Eq' and 'Show' are written by hand because 'SomeException' has neither, and
   33 -- the type must stay comparable for tests and for 'Control.Exception' idioms.
   34 -- Both instances deliberately look only at the /text/: two errors are equal
   35 -- when their messages are, and 'Show' renders the message plus the inner
   36 -- exception's own rendering (never a constructor dump, which for an arbitrary
   37 -- exception can be enormous).
   38 data TranscriptError = TranscriptError Text (Maybe SomeException)
   39 
   40 instance Eq TranscriptError where
   41   TranscriptError a ma == TranscriptError b mb = a == b && fmap displayEx ma == fmap displayEx mb
   42 
   43 instance Show TranscriptError where
   44   show (TranscriptError msg Nothing)  = "TranscriptError " <> show msg
   45   show (TranscriptError msg (Just e)) =
   46     "TranscriptError " <> show msg <> " (caused by: " <> T.unpack (displayEx e) <> ")"
   47 
   48 -- | Render an exception the way @displayException@ does, so the 'Show' output
   49 -- is a readable message rather than a nested constructor tree.
   50 displayEx :: SomeException -> Text
   51 displayEx = T.pack . show
   52 
   53 instance Exception TranscriptError

    2 -- | The broadcast helper for the WS @lists@ frame. Builds the partitioned
    3 -- snapshot, wraps it with the WS @{"type": "lists", ...}@ envelope, and
    4 -- pushes it to every WS subscriber via 'broadcastLists'. Called after
    5 -- every state change that affects the sidebar partition (tab
    6 -- insert/remove/rebind/rename/acknowledge/release, archive/unarchive,
    7 -- session create, auto-tab insert).
    8 --
    9 -- The REST @GET /api/lists@ returns the bare 'ListsSnapshotWire' (no
   10 -- @type@ field); this wrapper adds the WS envelope for the broadcast path
   11 -- (the frontend's @useListsStream@ discriminates on @type: "lists"@).
   12 --
   13 -- NOTE: the design's 50ms debounce/coalesce is deferred to a follow-up —
   14 -- at the expected single-user scale, bursts are rare and the direct
   15 -- broadcast is correct (if occasionally redundant). A debounce helper
   16 -- can be layered in here without changing the call sites.

   46 -- | Build the WS @lists@ envelope (@{"type": "lists", ...}@) from the
   47 -- current state and push it to every WS subscriber via 'broadcastLists'.
   48 -- Includes the broker's current thinking-session set so a freshly-connected
   49 -- web client hydrates its sidebar (mid-turn tabs show Thinking immediately).
   50 broadcastListsSnapshot :: StreamBroker -> TabsHandle -> SealPaths -> IO ()
   51 broadcastListsSnapshot broker tabsH paths = do
   52   thinkingSids <- thinkingSessions broker
   53   snap <- buildListsSnapshot tabsH paths thinkingSids
   54   -- Merge the "type": "lists" tag into the snapshot object so the WS frame
   55   -- carries the discriminator the frontend's useListsStream dispatches on.
   56   let snapObj = case A.toJSON snap of
   57         A.Object o -> o
   58         other      -> KeyMap.singleton (Key.fromText "snapshot") other
   59       envelope = A.Object (KeyMap.insert (Key.fromText "type") (A.String "lists") snapObj)
   60   broadcastLists broker envelope

   62 -- | Push a per-session @harness-status@ activity signal to every WS
   63 -- subscriber focused on the session. The frontend's
   64 -- @useSessionActivityStream@ consumes the @activity@ envelope to drive the
   65 -- tab status indicator (Thinking while the LLM is actively processing,
   66 -- Idle otherwise). 'Nothing' broker (tests) is a no-op. @status@ is
   67 -- @"thinking"@ at turn start and @"idle"@ at turn end (including failures).
   68 -- Also updates the broker's in-memory thinking set so a freshly-connected
   69 -- web client can hydrate its sidebar from the lists snapshot without
   70 -- waiting for the next harness-status event.
   71 broadcastHarnessStatus :: Maybe StreamBroker -> SessionId -> Text -> IO ()
   72 broadcastHarnessStatus mBroker sid status =
   73   case mBroker of
   74     Nothing -> pure ()
   75     Just broker -> do
   76       setThinking broker sid (status == "thinking")
   77       broadcast broker (BeActivity sid (object
   78         [ "kind" .= ("harness-status" :: Text)
   79         , "status" .= status
   80         ]))

   82 -- | Push a per-session @reply-delivered@ activity signal to every WS
   83 -- subscriber focused on the session. Marks the last assistant reply as
   84 -- "seen" because it was delivered to ≥1 subscribed chat channel
   85 -- (Signal/Telegram/CLI), so the frontend transitions the tab to Idle Read.
   86 -- 'Nothing' broker (tests) is a no-op. The timestamp defaults to now.
   87 broadcastReplyDelivered :: Maybe StreamBroker -> SessionId -> IO ()
   88 broadcastReplyDelivered mBroker sid =
   89   case mBroker of
   90     Nothing -> pure ()
   91     Just broker -> do
   92       now <- getCurrentTime
   93       broadcast broker (BeActivity sid (object
   94         [ "kind" .= ("reply-delivered" :: Text)
   95         , "timestamp" .= T.pack (showIso now)
   96         ]))

  120 -- | Wrap a 'ChannelCaps' so that 'ccPrompt' (the blocking ask primitive
  121 -- used by @ASK_HUMAN@ and the Untrusted confirmation gate) broadcasts a
  122 -- harness-status transition: @idle@ /before/ blocking on the human's
  123 -- reply, and @thinking@ /after/ the reply arrives (the turn is resuming).
  124 -- This ensures the web frontend shows the session as idle (not thinking)
  125 -- while it is waiting for human input — the core fix for the recurring
  126 -- \"stuck on thinking during ASK_HUMAN\" bug.
  127 --
  128 -- The wrapper uses 'onException' so the @thinking@ broadcast fires even
  129 -- if the inner 'ccPrompt' throws (e.g. a timeout/cancel from the
  130 -- AskReplyStore). Without this, an exception in the inner prompt would
  131 -- leave the session permanently idle in the broker's in-memory state
  132 -- (the turn's bracket cleanup would still broadcast idle at turn end,
  133 -- but the broker's thinking set would be inconsistent until then).
  134 --
  135 -- 'Nothing' broker (tests, standalone CLI) is a passthrough: the caps
  136 -- are returned unchanged except for the 'ccPrompt' wrapper which is a
  137 -- direct delegate (no broadcast, no overhead).
  138 --
  139 -- All other 'ChannelCaps' fields ('ccSend', 'ccShowHuman',
  140 -- 'ccPromptSecret', 'ccStreaming') are passed through unchanged.
  141 wrapCapsForAskStatus :: Maybe StreamBroker -> SessionId -> ChannelCaps -> ChannelCaps
  142 wrapCapsForAskStatus mBroker sid caps =
  143   caps { ccPrompt = \prompt -> do
  144            broadcastHarnessStatus mBroker sid "idle"
  145            ans <- ccPrompt caps prompt `onException` broadcastHarnessStatus mBroker sid "thinking"
  146            broadcastHarnessStatus mBroker sid "thinking"
  147            pure ans
  148        }

    2 -- | The broadcast helper for the WS @lists@ frame. Builds the partitioned
    3 -- snapshot, wraps it with the WS @{"type": "lists", ...}@ envelope, and
    4 -- pushes it to every WS subscriber via 'broadcastLists'. Called after
    5 -- every state change that affects the sidebar partition (tab
    6 -- insert/remove/rebind/rename/acknowledge/release, archive/unarchive,
    7 -- session create, auto-tab insert).
    8 --
    9 -- The REST @GET /api/lists@ returns the bare 'ListsSnapshotWire' (no
   10 -- @type@ field); this wrapper adds the WS envelope for the broadcast path
   11 -- (the frontend's @useListsStream@ discriminates on @type: "lists"@).
   12 --
   13 -- NOTE: the design's 50ms debounce/coalesce is deferred to a follow-up —
   14 -- at the expected single-user scale, bursts are rare and the direct
   15 -- broadcast is correct (if occasionally redundant). A debounce helper

   53 -- | Build the WS @lists@ envelope (@{"type": "lists", ...}@) from the
   54 -- current state and push it to every WS subscriber via 'broadcastLists'.
   55 -- Includes the broker's current thinking-session set so a freshly-connected
   56 -- web client hydrates its sidebar (mid-turn tabs show Thinking immediately).
   57 broadcastListsSnapshot :: StreamBroker -> TabsHandle -> SealPaths -> IO ()
   58 broadcastListsSnapshot broker tabsH paths = do
   59   thinkingSids <- thinkingSessions broker
   60   snap <- buildListsSnapshot tabsH paths thinkingSids
   61   -- Merge the "type": "lists" tag into the snapshot object so the WS frame
   62   -- carries the discriminator the frontend's useListsStream dispatches on.
   63   let snapObj = case A.toJSON snap of
   64         A.Object o -> o
   65         other      -> KeyMap.singleton (Key.fromText "snapshot") other
   66       envelope = A.Object (KeyMap.insert (Key.fromText "type") (A.String "lists") snapObj)
   67   broadcastLists broker envelope

   69 -- | Push a per-session @harness-status@ activity signal to every WS
   70 -- subscriber focused on the session. The frontend's
   71 -- @useSessionActivityStream@ consumes the @activity@ envelope to drive the
   72 -- tab status indicator (Thinking while the LLM is actively processing,
   73 -- Idle otherwise). 'Nothing' broker (tests) is a no-op. @status@ is
   74 -- @"thinking"@ at turn start and @"idle"@ at turn end (including failures).
   75 -- Also updates the broker's in-memory thinking set so a freshly-connected
   76 -- web client can hydrate its sidebar from the lists snapshot without
   77 -- waiting for the next harness-status event.
   78 broadcastHarnessStatus :: Maybe StreamBroker -> SessionId -> Text -> IO ()
   79 broadcastHarnessStatus mBroker sid status =
   80   case mBroker of
   81     Nothing -> pure ()
   82     Just broker -> do
   83       globalLogIO DebugS (ls ("[idle-diag] broadcastHarnessStatus: sid=" <> sessionIdText sid <> " status=" <> status))
   84       setThinking broker sid (status == "thinking")
   85       broadcast broker (BeActivity sid (object
   86         [ "kind" .= ("harness-status" :: Text)
   87         , "status" .= status
   88         ]))

   90 -- | Push a per-session @reply-delivered@ activity signal to every WS
   91 -- subscriber focused on the session. Marks the last assistant reply as
   92 -- "seen" because it was delivered to ≥1 subscribed chat channel
   93 -- (Signal/Telegram/CLI), so the frontend transitions the tab to Idle Read.
   94 -- 'Nothing' broker (tests) is a no-op. The timestamp defaults to now.
   95 broadcastReplyDelivered :: Maybe StreamBroker -> SessionId -> IO ()
   96 broadcastReplyDelivered mBroker sid =
   97   case mBroker of
   98     Nothing -> pure ()
   99     Just broker -> do
  100       now <- getCurrentTime
  101       broadcast broker (BeActivity sid (object
  102         [ "kind" .= ("reply-delivered" :: Text)
  103         , "timestamp" .= T.pack (showIso now)
  104         ]))
  105 
  106 -- | Push a per-session @turn-error@ activity signal to every WS subscriber
  107 -- focused on the session, so a client learns the turn failed /without/ a
  108 -- reload (requirement 3).
  109 --
  110 -- Carries the headline and reference only -- never the cause. The cause is
  111 -- deliberately withheld from any surface that could leave the machine, and the
  112 -- banner only needs enough to point at the durable transcript entry.
  113 --
  114 -- 'Nothing' broker (tests) is a no-op.
  115 broadcastTurnFailure :: Maybe StreamBroker -> SessionId -> TurnFailure -> IO ()
  116 broadcastTurnFailure mBroker sid failure =
  117   case mBroker of
  118     Nothing -> pure ()
  119     Just broker -> do
  120       now <- getCurrentTime
  121       broadcast broker (BeActivity sid (object
  122         [ "kind" .= ("turn-error" :: Text)
  123         , "headline" .= tfHeadline failure
  124         , "ref" .= tfRef failure
  125         , "timestamp" .= T.pack (showIso now)
  126         ]))

  128 -- | Push a per-session @tool-call@ activity signal to every WS subscriber.
  129 -- The chat-channel WS client receives this as an @activity@ event and
  130 -- renders a tool-progress bubble. The tool input is redacted for opcodes
  131 -- in the @secretOps@ set (e.g. @SECRET_MANAGE@) and truncated to 120
  132 -- characters for non-secret opcodes. 'Nothing' broker (tests) is a no-op.
  133 -- Pure input formatting (no IO beyond the broadcast itself).
  134 broadcastToolCall
  135   :: Maybe StreamBroker
  136   -> SessionId
  137   -> OpName
  138   -> Text
  139   -> Set OpName
  140   -> IO ()
  141 broadcastToolCall mBroker sid opName input secretOps =
  142   for_ mBroker $ \broker ->
  143     broadcast broker (BeActivity sid (object
  144       [ "kind" .= ("tool-call" :: Text)
  145       , "tool" .= opNameText
  146       , "input" .= inputDisplay
  147       ]))
  148   where
  149     OpName opNameText = opName
  150     inputDisplay
  151       | opName `Set.member` secretOps = "<redacted>" :: Text
  152       | otherwise = truncateInput input
  153     truncateInput t =
  154       if T.length t > 120
  155         then T.take 120 t <> "..."
  156         else t

  180 -- | Wrap a 'ChannelCaps' so that 'ccPrompt' (the blocking ask primitive
  181 -- used by @ASK_HUMAN@ and the Untrusted confirmation gate) broadcasts a
  182 -- harness-status transition: @idle@ /before/ blocking on the human's
  183 -- reply, and @thinking@ /after/ the reply arrives (the turn is resuming).
  184 -- This ensures the web frontend shows the session as idle (not thinking)
  185 -- while it is waiting for human input — the core fix for the recurring
  186 -- \"stuck on thinking during ASK_HUMAN\" bug.
  187 --
  188 -- The wrapper uses 'onException' so the @thinking@ broadcast fires even
  189 -- if the inner 'ccPrompt' throws (e.g. a timeout/cancel from the
  190 -- AskReplyStore). Without this, an exception in the inner prompt would
  191 -- leave the session permanently idle in the broker's in-memory state
  192 -- (the turn's bracket cleanup would still broadcast idle at turn end,
  193 -- but the broker's thinking set would be inconsistent until then).
  194 --
  195 -- 'Nothing' broker (tests, standalone CLI) is a passthrough: the caps
  196 -- are returned unchanged except for the 'ccPrompt' wrapper which is a
  197 -- direct delegate (no broadcast, no overhead).
  198 --
  199 -- All other 'ChannelCaps' fields ('ccSend', 'ccShowHuman',
  200 -- 'ccPromptSecret', 'ccStreaming') are passed through unchanged.
  201 wrapCapsForAskStatus :: Maybe StreamBroker -> SessionId -> ChannelCaps -> ChannelCaps
  202 wrapCapsForAskStatus mBroker sid caps =
  203   caps { ccPrompt = \prompt -> do
  204            broadcastHarnessStatus mBroker sid "idle"
  205            ans <- ccPrompt caps prompt `onException` broadcastHarnessStatus mBroker sid "thinking"
  206            broadcastHarnessStatus mBroker sid "thinking"
  207            pure ans
  208        }

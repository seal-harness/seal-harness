    2 -- | The broadcast helper for the WS @lists@ frame. Builds the partitioned
    3 -- snapshot, wraps it with the WS @{"type": "lists", ...}@ envelope, and
    4 -- pushes it to every WS subscriber via 'broadcastLists'. Called after
    5 -- every state change that affects the sidebar partition (tab
    6 -- insert/remove/rebind/rename/acknowledge/release, archive/unarchive,
    7 -- session create, auto-tab insert).
    8 --
    9 -- The REST @GET /api/lists@ returns the bare 'ListsSnapshotWire' (no
   10 -- @type@ field); this wrapper adds the WS envelope for the broadcast path
   11 -- (the frontend's @useListsStream@ discriminates on @type: "lists"@).
   12 --
   13 -- NOTE: the design's 50ms debounce/coalesce is deferred to a follow-up —
   14 -- at the expected single-user scale, bursts are rare and the direct
   15 -- broadcast is correct (if occasionally redundant). A debounce helper
   16 -- can be layered in here without changing the call sites.

   18   ( broadcastListsSnapshot
   19   , broadcastHarnessStatus
   20   , broadcastReplyDelivered
   21   , broadcastAgentDefsChanged
   22   , broadcastSkillsChanged

   49 -- | Build the WS @lists@ envelope (@{"type": "lists", ...}@) from the
   50 -- current state and push it to every WS subscriber via 'broadcastLists'.
   51 -- Includes the broker's current thinking-session set so a freshly-connected
   52 -- web client hydrates its sidebar (mid-turn tabs show Thinking immediately).
   53 broadcastListsSnapshot :: StreamBroker -> TabsHandle -> SealPaths -> IO ()
   54 broadcastListsSnapshot broker tabsH paths = do
   55   thinkingSids <- thinkingSessions broker
   56   snap <- buildListsSnapshot tabsH paths thinkingSids
   57   -- Merge the "type": "lists" tag into the snapshot object so the WS frame
   58   -- carries the discriminator the frontend's useListsStream dispatches on.
   59   let snapObj = case A.toJSON snap of
   60         A.Object o -> o
   61         other      -> KeyMap.singleton (Key.fromText "snapshot") other
   62       envelope = A.Object (KeyMap.insert (Key.fromText "type") (A.String "lists") snapObj)
   63   broadcastLists broker envelope

   65 -- | Push a per-session @harness-status@ activity signal to every WS
   66 -- subscriber focused on the session. The frontend's
   67 -- @useSessionActivityStream@ consumes the @activity@ envelope to drive the
   68 -- tab status indicator (Thinking while the LLM is actively processing,
   69 -- Idle otherwise). 'Nothing' broker (tests) is a no-op. @status@ is
   70 -- @"thinking"@ at turn start and @"idle"@ at turn end (including failures).
   71 -- Also updates the broker's in-memory thinking set so a freshly-connected
   72 -- web client can hydrate its sidebar from the lists snapshot without
   73 -- waiting for the next harness-status event.
   74 broadcastHarnessStatus :: Maybe StreamBroker -> SessionId -> Text -> IO ()
   75 broadcastHarnessStatus mBroker sid status =
   76   case mBroker of
   77     Nothing -> pure ()
   78     Just broker -> do
   79       setThinking broker sid (status == "thinking")
   80       broadcast broker (BeActivity sid (object
   81         [ "kind" .= ("harness-status" :: Text)
   82         , "status" .= status
   83         ]))
   84 
   85 -- | Push a per-session @reply-delivered@ activity signal to every WS
   86 -- subscriber focused on the session. Marks the last assistant reply as
   87 -- "seen" because it was delivered to ≥1 subscribed chat channel
   88 -- (Signal/Telegram/CLI), so the frontend transitions the tab to Idle Read.
   89 -- 'Nothing' broker (tests) is a no-op. The timestamp defaults to now.
   90 broadcastReplyDelivered :: Maybe StreamBroker -> SessionId -> IO ()
   91 broadcastReplyDelivered mBroker sid =
   92   case mBroker of
   93     Nothing -> pure ()
   94     Just broker -> do
   95       now <- getCurrentTime
   96       broadcast broker (BeActivity sid (object
   97         [ "kind" .= ("reply-delivered" :: Text)
   98         , "timestamp" .= T.pack (showIso now)
   99         ]))

  101 -- | Push a per-session @tool-call@ activity signal to every WS subscriber.
  102 -- The chat-channel WS client receives this as an @activity@ event and
  103 -- renders a tool-progress bubble. The tool input is redacted for opcodes
  104 -- in the @secretOps@ set (e.g. @SECRET_GET@) and truncated to 120
  105 -- characters for non-secret opcodes. 'Nothing' broker (tests) is a no-op.
  106 -- Pure input formatting (no IO beyond the broadcast itself).
  107 broadcastToolCall
  108   :: Maybe StreamBroker
  109   -> SessionId
  110   -> OpName
  111   -> Text
  112   -> Set OpName
  113   -> IO ()
  114 broadcastToolCall mBroker sid opName input secretOps =
  115   for_ mBroker $ \broker ->
  116     broadcast broker (BeActivity sid (object
  117       [ "kind" .= ("tool-call" :: Text)
  118       , "tool" .= opNameText
  119       , "input" .= inputDisplay
  120       ]))
  121   where
  122     OpName opNameText = opName
  123     inputDisplay
  124       | opName `Set.member` secretOps = "<redacted>" :: Text
  125       | otherwise = truncateInput input
  126     truncateInput t =
  127       if T.length t > 120
  128         then T.take 120 t <> "..."
  129         else t

  153 -- | Wrap a 'ChannelCaps' so that 'ccPrompt' (the blocking ask primitive
  154 -- used by @ASK_HUMAN@ and the Untrusted confirmation gate) broadcasts a
  155 -- harness-status transition: @idle@ /before/ blocking on the human's
  156 -- reply, and @thinking@ /after/ the reply arrives (the turn is resuming).
  157 -- This ensures the web frontend shows the session as idle (not thinking)
  158 -- while it is waiting for human input — the core fix for the recurring
  159 -- \"stuck on thinking during ASK_HUMAN\" bug.
  160 --
  161 -- The wrapper uses 'onException' so the @thinking@ broadcast fires even
  162 -- if the inner 'ccPrompt' throws (e.g. a timeout/cancel from the
  163 -- AskReplyStore). Without this, an exception in the inner prompt would
  164 -- leave the session permanently idle in the broker's in-memory state
  165 -- (the turn's bracket cleanup would still broadcast idle at turn end,
  166 -- but the broker's thinking set would be inconsistent until then).
  167 --
  168 -- 'Nothing' broker (tests, standalone CLI) is a passthrough: the caps
  169 -- are returned unchanged except for the 'ccPrompt' wrapper which is a
  170 -- direct delegate (no broadcast, no overhead).
  171 --
  172 -- All other 'ChannelCaps' fields ('ccSend', 'ccShowHuman',
  173 -- 'ccPromptSecret', 'ccStreaming') are passed through unchanged.
  174 wrapCapsForAskStatus :: Maybe StreamBroker -> SessionId -> ChannelCaps -> ChannelCaps
  175 wrapCapsForAskStatus mBroker sid caps =
  176   caps { ccPrompt = \prompt -> do
  177            broadcastHarnessStatus mBroker sid "idle"
  178            ans <- ccPrompt caps prompt `onException` broadcastHarnessStatus mBroker sid "thinking"
  179            broadcastHarnessStatus mBroker sid "thinking"
  180            pure ans
  181        }

    2 -- | The broadcast helper for the WS @lists@ frame. Builds the partitioned
    3 -- snapshot, wraps it with the WS @{"type": "lists", ...}@ envelope, and
    4 -- pushes it to every WS subscriber via 'broadcastLists'. Called after
    5 -- every state change that affects the sidebar partition (tab
    6 -- insert/remove/rebind/rename/acknowledge/release, archive/unarchive,
    7 -- session create, auto-tab insert).
    8 --
    9 -- The REST @GET /api/lists@ returns the bare 'ListsSnapshotWire' (no
   10 -- @type@ field); this wrapper adds the WS envelope for the broadcast path
   11 -- (the frontend's @useListsStream@ discriminates on @type: "lists"@).
   12 --
   13 -- NOTE: the design's 50ms debounce/coalesce is deferred to a follow-up —
   14 -- at the expected single-user scale, bursts are rare and the direct
   15 -- broadcast is correct (if occasionally redundant). A debounce helper
   16 -- can be layered in here without changing the call sites.

   19   , broadcastHarnessStatus
   20   , broadcastReplyDelivered

   43 -- | Build the WS @lists@ envelope (@{"type": "lists", ...}@) from the
   44 -- current state and push it to every WS subscriber via 'broadcastLists'.
   45 -- Includes the broker's current thinking-session set so a freshly-connected
   46 -- web client hydrates its sidebar (mid-turn tabs show Thinking immediately).
   47 broadcastListsSnapshot :: StreamBroker -> TabsHandle -> SealPaths -> IO ()
   48 broadcastListsSnapshot broker tabsH paths = do
   49   thinkingSids <- thinkingSessions broker
   50   snap <- buildListsSnapshot tabsH paths thinkingSids
   51   -- Merge the "type": "lists" tag into the snapshot object so the WS frame
   52   -- carries the discriminator the frontend's useListsStream dispatches on.
   53   let snapObj = case A.toJSON snap of
   54         A.Object o -> o
   55         other      -> KeyMap.singleton (Key.fromText "snapshot") other
   56       envelope = A.Object (KeyMap.insert (Key.fromText "type") (A.String "lists") snapObj)
   57   broadcastLists broker envelope

   59 -- | Push a per-session @harness-status@ activity signal to every WS
   60 -- subscriber focused on the session. The frontend's
   61 -- @useSessionActivityStream@ consumes the @activity@ envelope to drive the
   62 -- tab status indicator (Thinking while the LLM is actively processing,
   63 -- Idle otherwise). 'Nothing' broker (tests) is a no-op. @status@ is
   64 -- @"thinking"@ at turn start and @"idle"@ at turn end (including failures).
   65 -- Also updates the broker's in-memory thinking set so a freshly-connected
   66 -- web client can hydrate its sidebar from the lists snapshot without
   67 -- waiting for the next harness-status event.
   68 broadcastHarnessStatus :: Maybe StreamBroker -> SessionId -> Text -> IO ()
   69 broadcastHarnessStatus mBroker sid status =
   70   case mBroker of
   71     Nothing -> pure ()
   72     Just broker -> do
   73       setThinking broker sid (status == "thinking")
   74       broadcast broker (BeActivity sid (object
   75         [ "kind" .= ("harness-status" :: Text)
   76         , "status" .= status
   77         ]))
   78 
   79 -- | Push a per-session @reply-delivered@ activity signal to every WS
   80 -- subscriber focused on the session. Marks the last assistant reply as
   81 -- "seen" because it was delivered to ≥1 subscribed chat channel
   82 -- (Signal/Telegram/CLI), so the frontend transitions the tab to Idle Read.
   83 -- 'Nothing' broker (tests) is a no-op. The timestamp defaults to now.
   84 broadcastReplyDelivered :: Maybe StreamBroker -> SessionId -> IO ()
   85 broadcastReplyDelivered mBroker sid =
   86   case mBroker of
   87     Nothing -> pure ()
   88     Just broker -> do
   89       now <- getCurrentTime
   90       broadcast broker (BeActivity sid (object
   91         [ "kind" .= ("reply-delivered" :: Text)
   92         , "timestamp" .= T.pack (showIso now)
   93         ]))

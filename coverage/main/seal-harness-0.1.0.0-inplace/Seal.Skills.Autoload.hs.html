    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Skill auto-injection at session start. The harness reads a configured
    3 -- skill id (default @"seal-usage"@, the fresh-workdir contract), loads its
    4 -- body from the skill backend, and appends it to the resolved system prompt
    5 -- so the model is oriented to its per-session workspace from turn one.
    6 --
    7 -- The injected skill is /teaching/, not enforcement: it tells the model the
    8 -- workdir contract so it cooperates. The operator can disable auto-injection
    9 -- by setting @[skills] autoload = ""@ in @config.toml@, or override the
   10 -- injected skill id with any other value. The codegraph skill body (when
   11 -- a repo with @.codegraph/@ is in the workdir) is injected the same way,
   12 -- after the autoload skill, by 'injectCodegraphSkill'.
   13 module Seal.Skills.Autoload
   14   ( injectAutoloadSkill
   15   , injectCodegraphSkill
   16   , renderSkillForPrompt
   17   ) where
   18 
   19 import Data.Text (Text)
   20 
   21 import Seal.Skills.Backend (SkillBackend (..))
   22 import Seal.Skills.Types (Skill (..), mkSkillId, skillIdText)
   23 
   24 -- | The skill id for the codegraph built-in. Used by the caller to
   25 -- look up the skill body before calling 'injectCodegraphSkill'.
   26 codegraphSkillId :: Text
   27 codegraphSkillId = "codegraph"
   28 
   29 -- | Append the auto-loaded skill body to the resolved system prompt.
   30 --
   31 -- * @mSkillId = Nothing@ → no auto-injection; return the prompt unchanged.
   32 -- * @mSkillId = Just sid@ → load the skill body from @backend@; if found,
   33 --   append it under an @## Auto-loaded skill: <id>@ header. If the skill is
   34 --   missing from the backend, fall back to the prompt unchanged (a missing
   35 --   autoload skill is a soft failure — the session proceeds, the model
   36 --   just doesn't get the orientation note).
   37 --
   38 -- The skill body is appended /after/ any existing system prompt (the bound
   39 -- agent's @adSystem@). The header separates the agent's identity/prompt
   40 -- from the harness-injected orientation.
   41 injectAutoloadSkill
   42   :: SkillBackend
   43   -- ^ The skill store (reads from @~/.seal/config/skills@).
   44   -> Maybe Text
   45   -- ^ The skill id to auto-inject ('Nothing' disables).
   46   -> Maybe Text
   47   -- ^ The resolved system prompt (the bound agent's @adSystem@, or
   48   -- 'Nothing' when no agent is bound).
   49   -> IO (Maybe Text)
   50 injectAutoloadSkill _ Nothing prompt = pure prompt
   51 injectAutoloadSkill backend (Just skillIdText_) prompt = do
   52   case mkSkillId skillIdText_ of
   53     Left _err -> pure prompt  -- invalid id → soft-fail (no injection)
   54     Right sid -> do
   55       mSkill <- sbRead backend sid
   56       case mSkill of
   57         Nothing    -> pure prompt  -- skill missing → soft-fail
   58         Just skill -> pure (Just (renderSkillForPrompt prompt skill))
   59 
   60 -- | Render the system prompt + auto-loaded skill body as a single text.
   61 -- The skill body is appended under a header; if the existing prompt is
   62 -- 'Nothing', the skill body becomes the entire prompt.
   63 renderSkillForPrompt :: Maybe Text -> Skill -> Text
   64 renderSkillForPrompt mPrompt skill =
   65   case mPrompt of
   66     Nothing    -> rendered
   67     Just base  -> base <> "\n\n" <> rendered
   68   where
   69     rendered =
   70       "# Auto-loaded skill: " <> skillIdText (skId skill) <> "\n\n"
   71       <> skBody skill
   72 
   73 -- | Conditionally inject the codegraph skill body into the system prompt.
   74 -- Called by the turn engine after the autoload skill is injected, but
   75 -- only when a @.codegraph/@ directory exists in one of the repos cloned
   76 -- into the workdir. The caller (TurnEngine) checks for @.codegraph/@ and,
   77 -- if present, looks up the codegraph skill body from the backend and
   78 -- passes @Just body@ here. When @Nothing@, the prompt is returned unchanged.
   79 --
   80 -- The skill body is appended under the same @## Auto-loaded skill: <id>@
   81 -- header as the autoload skill, so the model sees it as orientation, not
   82 -- as part of the agent's identity prompt.
   83 injectCodegraphSkill
   84   :: Maybe Text
   85   -- ^ The codegraph skill body ('Nothing' when no @.codegraph/@ is
   86   -- present; the prompt is unchanged in that case).
   87   -> Maybe Text
   88   -- ^ The resolved system prompt so far (base + autoload body, or
   89   -- 'Nothing' when no agent is bound and no autoload skill injected).
   90   -> Maybe Text
   91 injectCodegraphSkill Nothing prompt = prompt
   92 injectCodegraphSkill (Just body) prompt =
   93   let rendered = "# Auto-loaded skill: " <> codegraphSkillId <> "\n\n" <> body
   94   in case prompt of
   95        Nothing   -> Just rendered
   96        Just base -> Just (base <> "\n\n" <> rendered)

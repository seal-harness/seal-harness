  232 -- | Build the web 'TurnAdapter' for a given 'ChannelCaps'. The web adapter:
  233 -- * @taPreTurn@ — fans out the user's message to subscribed chat channels
  234 --   (cross-channel mirroring, @"web"@ label). No 'replySubscribe' (the web
  235 --   isn't an inbox-driven channel handle).
  236 -- * @taChannelLabel@ — constant @"web"@.
  237 -- * @taOnStop@ — 'Just' the reply fan-out so chat channels subscribed to
  238 --   this session receive the final reply.
  239 -- * @taOnUserMessage@ — 'Nothing' (web turns aren't /bg).
  240 -- * @taPostTurn@ — @pure ()@ (the web's @lists@ snapshot is refreshed by
  241 --   'triggerBroadcast' in 'handleSend', not here).
  242 -- * @taStartWiring@ — the engine-owned 'TurnEngine.buildStartWiring' (W4
  243 --   collapsed the per-surface builders into this single one).
  244 mkWebTurnAdapter :: SendDeps -> TurnDeps -> ChannelCaps -> TurnAdapter
  245 mkWebTurnAdapter deps td caps = TurnAdapter
  246   { taCaps          = caps
  247   , taPreTurn       = \sid _meta t -> replyFanoutMessage (sdReplies deps) sid "web" t
  248   , taChannelLabel  = const "web"
  249   , taOnStop        = Just . replyFanout (sdReplies deps)
  250   , taOnUserMessage = const (Just (pure ()))
  251   , taPostTurn      = \_ _ -> pure ()
  252   , taStartWiring   = \sessionBackends sid appEnv eCfg operatorCeiling _meta ->
  253       TurnEngine.buildStartWiring td sessionBackends sid appEnv eCfg operatorCeiling "web"

  256 -- | The outcome of a send request. The HTTP layer ('Seal.Gateway.API') turns
  257 -- this into the JSON response body the frontend's @SendResult@ parses.
  258 data SendOutcome
  259   = SendSlash Text (Maybe SessionId)
  260     -- ^ transient slash-command output (no transcript entry). The optional
  261     -- 'SessionId' is set by slash commands that mint+focus a new session
  262     -- (e.g. @\/new@) so the frontend can navigate to it. 'Nothing' for
  263     -- ordinary slash commands.
  264   | SendAssistant       -- ^ plain turn; reply lands in the transcript
  265   | SendError Int Text  -- ^ HTTP status code + message (400/404/500)
  266   deriving stock (Eq, Show)

  268 -- | Encode a 'SendOutcome' as the JSON the frontend's @SendResult@ parses.
  269 -- Errors carry an @error@ field (the frontend logs them); success carries
  270 -- @kind@ + @response@; @\/new@ also carries @session_id@ so the SPA can
  271 -- navigate to the freshly-minted session.
  272 sendOutcomeJson :: SendOutcome -> (Int, Value)
  273 sendOutcomeJson = \case
  274   SendSlash t mSid ->
  275     (200, object
  276       [ "kind" .= ("slash" :: Text)
  277       , "response" .= t
  278       , "session_id" .= (sessionIdText <$> mSid)
  279       ])
  280   SendAssistant  -> (200, object [ "kind" .= ("assistant" :: Text), "response" .= ("" :: Text) ])
  281   SendError c m  -> (c, object [ "error" .= m ])

  283 -- | Handle POST /api/sessions/:id/send. Loads the session meta by id, routes
  284 -- the text, runs the turn, and returns the 'SendOutcome'. A missing session
  285 -- -> 404; an unknown provider / vault error -> 400; an internal failure ->
  286 -- 500 (logged to stderr).
  287 --
  288 -- After a successful 'Plain'/'SlashCommand'/'NewSession' turn (any outcome
  289 -- that is NOT 'SendError'), auto-tabs the session via
  290 -- 'ensureTabForSession' (W2 invariant 2). The 'TabCommand'/'Focus'/'Inject'
  291 -- routes return 'SendSlash' but no message was sent — they are explicitly
  292 -- excluded from auto-tabbing.
  293 handleSend :: SendDeps -> SessionId -> Text -> IO SendOutcome
  294 handleSend deps sid rawText = do
  295   mMeta <- loadSessionMeta (sdPaths deps) sid
  296   case mMeta of
  297     Nothing -> pure (SendError 404 "session not found")
  298     Just meta -> case route rawText of
  299       Left (ParseError e) -> pure (SendSlash e Nothing)
  300       Right (Plain t) -> do
  301         er <- withExceptionLogging (sdLogger deps) (Just (sessionLogPath (sdPaths deps) (smId meta))) "plainTurn" $
  302           plainTurn deps meta t
  303         case er of
  304           Left err -> pure (SendError 500 err)
  305           Right (Left err) -> pure (SendError 500 err)
  306           Right (Right ()) -> do
  307             ensureTabForSession (sdTabsHandle deps) KindProvider (smId meta)
  308             triggerBroadcast deps
  309             pure SendAssistant
  310       Right (SlashCommand cmdName) -> do
  311         -- W5: no srActive swap bracket. The per-request call/skill
  312         -- dispatcher closes over the request's explicit sid (see
  313         -- 'runSlash'). /new still swaps srActive via its ndRebind (the
  314         -- swap is how /new communicates the new session to
  315         -- 'newSessionIdIfChangedFrom').
  316         r <- runSlash deps meta rawText
  317         case r of
  318           SendError _ _ -> pure r
  319           _            -> do
  320             -- /tab owns its tab lifecycle (close/resume mutate the list);
  321             -- auto-tabbing the request's session afterward would resurrect
  322             -- a tab that /tab close just removed. Skip the auto-tab for
  323             -- /tab; still broadcast so the frontend reflects any mutation.
  324             unless (isTabCommand cmdName) $
  325               ensureTabForSession (sdTabsHandle deps) KindProvider (smId meta)
  326             triggerBroadcast deps
  327             pure r
  328       Right (NewSession _args) -> do
  329         r <- runSlash deps meta rawText
  330         case r of
  331           SendError _ _ -> pure r
  332           _            -> do
  333             ensureTabForSession (sdTabsHandle deps) KindProvider (smId meta)
  334             triggerBroadcast deps
  335             pure r
  336       Right (TabCommand _)   -> pure (SendSlash "(tab commands are not supported over the web send endpoint)" Nothing)
  337       Right CurrentTab        -> do
  338         tl <- snapshotTabs (sdTabsHandle deps)
  339         pure $ case lookupByRef tl (BoundSession (smId meta)) of
  340           Just t  -> SendSlash (renderWebTab t) Nothing
  341           Nothing -> SendSlash "no current tab" Nothing
  342       Right (Focus _)        -> pure (SendSlash "(focus is a tab-level operation; use the sidebar)" Nothing)
  343       Right (Inject _ _)    -> pure (SendSlash "(inject is a tab-level operation; use the sidebar)" Nothing)
  344 
  345 -- | Render the current tab for the web send endpoint: @<index>  <kind>  [label]@.
  346 renderWebTab :: Tab -> Text
  347 renderWebTab t =
  348   T.singleton (tabIndexToChar (tIndex t)) <> "  " <> T.pack (show (tKind t))
  349     <> maybe "" ("  " <>) (tLabel t)

  351 -- | True for slash commands that own their tab lifecycle (@/tab ...@).
  352 -- Used to skip 'ensureTabForSession' on the web send path so a
  353 -- @/tab close N@ that just removed a tab isn't immediately undone by the
  354 -- auto-tab. The head word is checked case-insensitively (the registry
  355 -- lookup is case-insensitive too).
  356 isTabCommand :: Text -> Bool
  357 isTabCommand cmdName =
  358   let headWord = T.takeWhile (/= ' ') (T.strip cmdName)
  359   in T.toCaseFold headWord == "tab"

  384 -- | Resolve the system prompt for a web turn. An ad-hoc
  385 -- 'smSystemOverride' (set via PUT /api/sessions/:id/prompt from the
  386 -- Session setup screen's "Use a one-off agent file" upload) takes
  387 -- precedence over the bound agent's 'adSystem'. Returns 'Nothing' when
  388 -- neither is set. The auto-loaded skill (default @seal-usage@, the
  389 -- fresh-workdir contract) is appended so the model is oriented to its
  390 -- per-session workspace from turn one. Disabled by setting
  391 -- | Run a plain (non-slash) turn through the agent loop. Mirrors
  392 -- 'Seal.Channel.Cli.runCliTui's @plainHandler@ but pulls the session by id
  393 -- and uses the ask/reply-backed 'ChannelCaps' ('webAskCaps') so ASK_HUMAN
  394 -- surfaces the question to the frontend and blocks until the human answers
  395 -- (the web frontend reads replies + questions from the WS stream, not from
  396 -- ccSend).
  397 --
  398 -- After the turn, the reply is fanned out to any chat channels (Telegram,
  399 -- Signal) subscribed to this session via the 'ReplyRegistry', so a web
  400 -- send on a channel-origin tab also delivers the reply to the channel.
  401 plainTurn :: SendDeps -> SessionMeta -> Text -> IO (Either Text ())
  402 plainTurn deps meta t = do
  403   let td = mkWebTurnDeps deps
  404   caps <- webAskCaps (sdBroker deps) (sdAskReply deps) (smId meta)
  405   let adapter = mkWebTurnAdapter deps td caps
  406   outcome <- runSessionTurn td adapter meta Nothing t
  407   pure (case toError outcome of
  408           Just err -> Left err
  409           Nothing  -> Right ())

  411 -- | Run a slash command. The output is collected via a 'ChannelCaps' whose
  412 -- 'ccSend' appends to an MVar-backed list, then returned as the @response@.
  413 -- 'ccPrompt' routes through the ask/reply store so ASK_HUMAN (e.g. from a
  414 -- slash command that delegates to the agent) surfaces to the frontend.
  415 --
  416 -- For @\/new@ specifically: the command's 'ndRebind' swaps the active-session
  417 -- ref ('sdSession') to the freshly-minted session. After the action runs, we
  418 -- re-read the active session and, if its id differs from the one we entered
  419 -- with, include the new id in the 'SendSlash' outcome so the frontend can
  420 -- navigate to it. (This avoids widening the 'CommandAction' contract or
  421 -- threading a per-call IORef through the registry.)
  422 runSlash :: SendDeps -> SessionMeta -> Text -> IO SendOutcome
  423 runSlash deps meta fullLine = do
  424   outVar <- newMVar ([] :: [Text])
  425   askCaps <- webAskCaps (sdBroker deps) (sdAskReply deps) (smId meta)
  426   let caps = askCaps { ccSend = \t' -> modifyMVar_ outVar (\acc -> pure (acc <> [t'])) }
  427       sid = smId meta
  428       td = mkWebTurnDeps deps
  429       -- W5: build a per-request call dispatcher that closes over the
  430       -- request's explicit sid (no srActive swap). The sdRegistry is
  431       -- rebuilt per-request by replacing the call/skill specs with
  432       -- per-request versions; the rest of the specs are reused as-is.
  433       perRequestCallDispatcher = webCallDispatcher deps td sid
  434       perRequestRegistry = replaceCallSkillSpecs (sdRegistry deps)
  435         (skillCommandSpec (bSkills (sdBackends deps)) perRequestCallDispatcher)
  436         (callCommandSpec perRequestCallDispatcher)
  437         (stopCommandSpecForSession (sdAbortReg deps) sid
  438            (mkStopTranscriptWriter (sdPaths deps) (sdBroker deps)))
  439         (modelCommandSpecForSession (sdProvider deps) (sdPaths deps) (pure sid)
  440            (mkModelTranscriptWriter (sdPaths deps) (sdBroker deps)))
  441   -- Snapshot the active-session ref BEFORE the action runs. The web
  442   -- gateway is multi-session: @srActive@ is a process-global ref that
  443   -- points at whatever session the last @\/new@ (or session creation)
  444   -- left it at — which may be a DIFFERENT session than the one this
  445   -- request is operating on. Comparing the post-action @srActive@ to
  446   -- the request's sid would falsely report a change for every session
  447   -- that isn't currently "active", causing the frontend to navigate
  448   -- away on benign slash commands like @\/skill list@. Instead we
  449   -- compare before vs after: only a slash command that actually
  450   -- swapped @srActive@ during THIS call (e.g. @\/new@) reports a
  451   -- change.
  452   activeBefore <- readIORef (srActive (sdSession deps))
  453   d <- ingest perRequestRegistry (sdPreprocess deps) (RawInbound fullLine)
  454   case d of
  455     DispatchAction (CommandAction act) -> do
  456       mFollowUp <- act caps
  457       chunks <- readMVar outVar
  458       -- If the action swapped the active session (e.g. /new), thread the
  459       -- new sid into the outcome so the frontend navigates to it.
  460       mNewSid <- newSessionIdIfChangedFrom deps (smId activeBefore)
  461       -- When the command signals a follow-up turn (e.g. /skill load),
  462       -- run the turn so the LLM runs immediately. The follow-up text
  463       -- becomes the user message 'runTurn' writes to conversation.jsonl;
  464       -- the skill body is already there from 'recordSkillLoadResult'.
  465       case mFollowUp of
  466         Just t -> do
  467           _ <- plainTurnWithCaps deps meta caps t
  468           pure (SendSlash (T.intercalate "\n" chunks) mNewSid)
  469         Nothing -> pure (SendSlash (T.intercalate "\n" chunks) mNewSid)
  470     ShowText t -> pure (SendSlash t Nothing)
  471     Rejected t -> pure (SendError 400 t)
  472     PlainMessage t -> do
  473       er <- plainTurnWithCaps deps meta caps t
  474       case er of
  475         Left err -> pure (SendError 400 err)
  476         Right () -> do
  477           chunks <- readMVar outVar
  478           pure (SendSlash (T.intercalate "\n" chunks) Nothing)

  480 -- | After a slash action runs, check whether the active session changed
  481 -- DURING this call. Returns the new 'SessionId' if 'sdSession' now points
  482 -- at a different session than @beforeSid@ (the snapshot taken before the
  483 -- action ran); 'Nothing' otherwise. Used by @\/new@ to thread the
  484 -- freshly-minted sid into the 'SendSlash' outcome. Comparing to the
  485 -- pre-action snapshot (rather than the request's sid) avoids false
  486 -- positives on multi-session web gateways where @srActive@ may already
  487 -- point at a different session than the one this request targets.
  488 newSessionIdIfChangedFrom :: SendDeps -> SessionId -> IO (Maybe SessionId)
  489 newSessionIdIfChangedFrom deps beforeSid = do
  490   active <- readIORef (srActive (sdSession deps))
  491   if smId active == beforeSid
  492     then pure Nothing
  493     else pure (Just (smId active))

  507 -- | Build a 'CallDispatcher' for the web channel. Resolves the active
  508 -- session at call time, opens its transcript, builds the session's ISA
  509 -- registry, and dispatches the opcode via 'Seal.ISA.Dispatch.dispatch'
  510 -- under 'Full' autonomy semantics (the operator is the approver by
  511 -- typing @/call@). Mirrors 'plainTurnWithCaps' but invokes 'dispatch'
  512 -- directly instead of 'runTurn'. Returns the structured result for
  513 -- 'Seal.Command.Call.renderOpResult' to render.
  514 -- | Build a 'CallDispatcher' for the web channel. Closes over the request's
  515 -- explicit 'SessionId' (W5: no longer reads the process-global @srActive@).
  516 -- Delegates to the unified 'TurnEngine.callDispatcher'.
  517 webCallDispatcher :: SendDeps -> TurnDeps -> SessionId -> CallDispatcher
  518 webCallDispatcher deps td sid callOpName val = do
  519   caps <- webAskCaps (sdBroker deps) (sdAskReply deps) sid
  520   TurnEngine.callDispatcher td caps sid "web" callOpName val

  522 -- | Build the web 'ChannelCaps' for a per-turn 'AskReplyStore'. 'ccSend'
  523 -- accumulates streaming text deltas and broadcasts an @entry-update@ WS event
  524 -- so the web frontend sees the response grow live (token-by-token) instead of
  525 -- appearing only after the full LLM stream completes. 'ccPrompt' drives the
  526 -- full ask/reply primitive: it mints a pending question (carrying the opcode
  527 -- metadata when provided), broadcasts a 'BeAsk' event so the frontend renders
  528 -- it, and blocks on the store until the human answers via POST
  529 -- /api/sessions/:id/questions/:qid/answer (or the question is
  530 -- cancelled/timed out). The returned 'Text' is the approval scope's wire form
  531 -- (e.g. @"once"@, @"for_session"@, @"always"@, @"rejected"@) for the confirmation
  532 -- gate, or the human's typed reply for @ASK_HUMAN@. 'ccPromptSecret' is
  533 -- fail-closed.
  534 webAskCaps
  535   :: Maybe StreamBroker -> AskReplyStore -> SessionId -> IO ChannelCaps
  536 webAskCaps mBroker store sid = do
  537   accRef <- newIORef ("" :: Text)
  538   let caps = def
  539         { ccSend = \delta -> case mBroker of
  540             Nothing -> pure ()
  541             Just broker -> do
  542               acc <- atomicModifyIORef' accRef (\a -> (a <> delta, a <> delta))
  543               broadcast broker (BeEntryUpdate sid (streamingEntryJson acc))
  544         , ccPrompt = \(AskPrompt q opts) -> do
  545             outcome <- askHumanWithOptions store sid q opts (\qid ->
  546               case mBroker of
  547                 Nothing -> pure ()
  548                 Just broker ->
  549                   broadcast broker (BeAsk sid (object
  550                     [ "id" .= askIdText qid
  551                     , "question" .= q
  552                     , "options" .= opts
  553                     ])))
  554             pure (case outcome of
  555               Left _  -> "rejected"
  556               Right t -> t)
  557         }
  558   pure caps
  559 
  560 -- | Build a synthetic streaming @entry-update@ JSON envelope for the
  561 -- accumulated text so far. The entry id is the fixed sentinel @"streaming"@
  562 -- so the frontend's 'reconcileEntries' replaces the same placeholder in place
  563 -- as text grows. When the final @entry@ event arrives (after the response is
  564 -- recorded to the transcript), the frontend evicts the streaming placeholder
  565 -- and inserts the real entry. The payload mirrors a response entry's shape so
  566 -- the frontend's 'transcriptToMessages' renders it as an assistant message.
  567 streamingEntryJson :: Text -> Value
  568 streamingEntryJson text = object
  569   [ "id"        .= ("streaming" :: Text)
  570   , "timestamp" .= ("" :: Text)
  571   , "direction" .= ("response" :: Text)
  572   , "payload"   .= object
  573       [ "content" .= [object ["type" .= ("text" :: Text), "text" .= text]]
  574       ]
  575   , "harness"   .= (Nothing :: Maybe Text)
  576   , "model"     .= (Nothing :: Maybe Text)
  577   , "channel"   .= (Nothing :: Maybe Text)
  578   , "raw"       .= ("" :: Text)
  579   ]

  581 -- | Notify the broker that a pending question was resolved (answered or
  582 -- cancelled) so the frontend dismisses it. 'Nothing' broker (tests) is a
  583 -- no-op.
  584 broadcastAskResolved
  585   :: Maybe StreamBroker -> SessionId -> AskId -> Text -> IO ()
  586 broadcastAskResolved mBroker sid qid resolution =
  587   case mBroker of
  588     Nothing -> pure ()
  589     Just broker ->
  590       broadcast broker (BeAskResolved sid (object
  591         [ "id" .= askIdText qid
  592         , "resolution" .= resolution
  593         ]))

  595 -- | Handle @POST /api/sessions/:id/setup-repo@: clone a repo into the
  596 -- session's workdir before the first turn. The web "set up repo" combo box
  597 -- calls this. Rather than calling 'cloneRepoIO' directly (which is
  598 -- unaudited — a side channel that never appears in the transcript), this
  599 -- dispatches the real 'SETUP_REPO' opcode via 'webCallDispatcher'. The
  600 -- clone (and any failure) is therefore recorded in the session's
  601 -- transcript exactly like a model-invoked SETUP_REPO, so the user sees it
  602 -- in the chat and any error is visible there — not silent.
  603 --
  604 -- The 'srActive' ref is scoped to the target session for the duration of
  605 -- the dispatch (the web gateway is multi-session; 'webCallDispatcher'
  606 -- reads 'srActive' to pick the transcript), then restored.
  607 --
  608 -- Returns @Left err@ for an invalid url or a dispatch failure; @Right msg@
  609 -- with the opcode's text result (the clone/no-op/conflict/failure message)
  610 -- for the API layer to pass through to the frontend.
  611 handleSetupRepo :: SendDeps -> SessionId -> Text -> IO (Either Text Text)
  612 handleSetupRepo deps sid url =
  613   case validateRepoUrl url of
  614     Left err -> pure (Left ("invalid url: " <> err))
  615     Right cleanUrl -> do
  616       mMeta <- loadSessionMeta (sdPaths deps) sid
  617       case mMeta of
  618         Nothing -> pure (Left "session not found")
  619         Just _targetMeta -> do
  620           -- Dispatch SETUP_REPO via the audited path (records into the
  621           -- transcript + broadcasts the entry so the frontend sees it).
  622           -- W5: the dispatcher takes the explicit sid (no srActive swap).
  623           let td = mkWebTurnDeps deps
  624               dispatcher = webCallDispatcher deps td sid
  625           res <- dispatcher (OpName "SETUP_REPO") (object ["url" .= cleanUrl])
  626           case res of
  627             Left dErr -> pure (Left ("SETUP_REPO dispatch failed: " <> renderDispatchError dErr))
  628             Right opRes ->
  629               if orIsError opRes
  630                 then pure (Left (opResultText opRes))
  631                 else pure (Right (opResultText opRes))

  633 -- | Join the text parts of an 'OpResult' into a single message (the
  634 -- clone/no-op/conflict/failure text from SETUP_REPO).
  635 opResultText :: OpResult -> Text
  636 opResultText r = T.intercalate "\n" [ t | TrpText t <- orParts r ]

  638 -- | Deliver an answer to a pending question for a session. Returns 'True'
  639 -- if the answer was accepted (the question was pending and not yet
  640 -- answered). Also broadcasts 'BeAskResolved' so the frontend dismisses the
  641 -- question. A 'Left' parse error is returned for a malformed ask id or
  642 -- approval scope.
  643 handleAnswerDelivery
  644   :: SendDeps -> SessionId -> Text -> ApprovalScope -> IO (Either Text Bool)
  645 handleAnswerDelivery deps sid qidTxt scope =
  646   case parseAskId qidTxt of
  647     Left e -> pure (Left e)
  648     Right qid -> do
  649       let reply = AskReply scope (approvalScopeText scope)
  650       accepted <- deliverAnswer (sdAskReply deps) qid reply
  651       when accepted $
  652         broadcastAskResolved (sdBroker deps) sid qid "answered"
  653       pure (Right accepted)

  655 -- | Deliver a free-text answer (a chosen option's label or a typed "Other"
  656 -- reply) to a pending ASK_HUMAN question. The 'ApprovalScope' is always
  657 -- 'ScopeOnce' (ASK_HUMAN replies are never cached). Returns 'True' if the
  658 -- answer was accepted. Broadcasts 'BeAskResolved' so the frontend dismisses
  659 -- the question. A 'Left' parse error is returned for a malformed ask id.
  660 handleAnswerTextDelivery
  661   :: SendDeps -> SessionId -> Text -> Text -> IO (Either Text Bool)
  662 handleAnswerTextDelivery deps sid qidTxt answerText =
  663   case parseAskId qidTxt of
  664     Left e -> pure (Left e)
  665     Right qid -> do
  666       let reply = AskReply ScopeOnce answerText
  667       accepted <- deliverAnswer (sdAskReply deps) qid reply
  668       when accepted $
  669         broadcastAskResolved (sdBroker deps) sid qid "answered"
  670       pure (Right accepted)

  672 -- | Parse the POST .../questions/:qid/answer body. Accepts EITHER
  673 -- @{scope: "once|for_session|always|rejected"}@ (the confirmation gate) OR
  674 -- @{answer: "<text>"}@ (ASK_HUMAN). Returns 'Right (Left scope)' for the
  675 -- scope path, 'Right (Right answerText)' for the answer path. 'Left' with
  676 -- an error message when both fields are present (ambiguous), neither is
  677 -- present, or a value is malformed. This is the explicit both-reject parser
  678 -- (gate: Security #6) — a body with both @scope@ and @answer@ is rejected
  679 -- with 400, NOT silently delivering @answer@ with @scope@ semantics.
  680 parseAnswerBody :: BL.ByteString -> Either Text (Either ApprovalScope Text)
  681 parseAnswerBody body =
  682   case A.decode body :: Maybe A.Value of
  683     Just (A.Object o) ->
  684       let mScope  = KeyMap.lookup (Key.fromText "scope") o
  685           mAnswer = KeyMap.lookup (Key.fromText "answer") o
  686       in case (mScope, mAnswer) of
  687         (Just (A.String t), Nothing) ->
  688           case parseApprovalScope t of
  689             Right scope -> Right (Left scope)
  690             Left e -> Left e
  691         (Nothing, Just (A.String t)) -> Right (Right t)
  692         (Just _, Just _) -> Left "ambiguous: send either {scope} or {answer}, not both"
  693         (Nothing, Nothing) -> Left "missing 'scope' or 'answer' field"
  694         (Just _, Nothing) -> Left "invalid 'scope' field (expected a string)"
  695         (Nothing, Just _) -> Left "invalid 'answer' field (expected a string)"
  696     _ -> Left "invalid JSON body"

  698 -- | Cancel a pending question for a session. Returns 'True' if the question
  699 -- was pending and is now cancelled. Broadcasts 'BeAskResolved' so the
  700 -- frontend dismisses the question.
  701 handleAskCancel

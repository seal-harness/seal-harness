   66 parseSessionAction :: Value -> Either Text SessionAction
   67 parseSessionAction v =
   68   case parseMaybe (withObject "in" (.: "action")) v of
   69     Just t -> case t :: Text of
   70       "new"    -> Right SessNew
   71       "list"   -> Right SessList
   72       "search" -> Right SessSearch
   73       "get"    -> Right SessGet
   74       other    -> Left ("unknown session action: " <> other)
   75     Nothing -> Left "missing or invalid action field"

   90 handleNew :: SealPaths -> Value -> App OpResult
   91 handleNew paths v = do
   92   let provider = fromMaybe "anthropic" (textField v "provider")
   93       model = fromMaybe "claude-sonnet-4-20250514" (textField v "model")
   94       channel = fromMaybe "api" (textField v "channel")
   95       mDesc = textField v "description"
   96   meta <- liftIO (newSession paths provider model channel Nothing)
   97   let meta' = case mDesc of
   98         Just d | not (T.null (T.strip d)) -> meta { smDescription = Just d }
   99         _ -> meta
  100   case mDesc of
  101     Just d | not (T.null (T.strip d)) -> liftIO (saveSessionMeta paths meta')
  102     _ -> pure ()
  103   let rendered = renderSessionMeta meta'
  104       recorded = object
  105         [ "session_id" .= sessionIdText (smId meta')
  106         , "provider" .= smProvider meta'
  107         , "model" .= smModel meta'
  108         , "channel" .= smChannel meta'
  109         , "description" .= smDescription meta'
  110         ]
  111   pure (OpResult [TrpText rendered] False recorded)
  112 
  113 handleList :: SealPaths -> Value -> App OpResult
  114 handleList paths v = do
  115   let archived = fromMaybe False (boolField v "archived")
  116       includeChildren = fromMaybe False (boolField v "include_children")
  117   metas <- liftIO (if archived then listArchivedSessions paths else listSessions paths)
  118   children <- if includeChildren
  119                 then liftIO (listChildSessions paths)
  120                 else pure []
  121   let parentLines = map renderSessionMeta metas
  122       childLines = [ renderChildSummary parentSid childSid
  123                    | (parentSid, childSid) <- children ]
  124       allLines = parentLines <> childLines
  125       rendered = case allLines of
  126         [] -> "(no sessions found)"
  127         _ -> T.intercalate "\n" allLines
  128       recorded = object
  129         [ "count" .= length metas
  130         , "archived" .= archived
  131         , "include_children" .= includeChildren
  132         , "child_count" .= length children
  133         , "session_ids" .= fmap (sessionIdText . smId) metas
  134         ]
  135   pure (OpResult [TrpText rendered] False recorded)
  136 
  137 handleSearch :: SealPaths -> Value -> App OpResult
  138 handleSearch paths v = do
  139   let q = fromMaybe "" (textField v "query")
  140   if T.null (T.strip q)
  141     then pure (OpResult [TrpText "search: query is empty"] True (object []))
  142     else do
  143       let qLower = T.toCaseFold (T.strip q)
  144           archived = fromMaybe False (boolField v "archived")
  145       metas <- liftIO (if archived then listArchivedSessions paths else listSessions paths)
  146       results <- liftIO (mapM (searchSession paths qLower) metas)
  147       let matched = [ (m, snip) | (m, Just snip) <- results ]
  148       children <- liftIO (listChildSessions paths)
  149       childResults <- liftIO (mapM (searchChildSession paths qLower) children)
  150       let childMatched = [ (p, c, snip) | (p, c, Just snip) <- childResults ]
  151           rendered
  152             | null matched && null childMatched = "(no sessions found)"
  153             | otherwise = T.intercalate "\n\n" $
  154                  [ renderSessionMeta m <> "\n  snippet: " <> snip
  155                  | (m, snip) <- matched ]
  156               <> [ renderChildSummary p c <> "\n  snippet: " <> snip
  157                  | (p, c, snip) <- childMatched ]
  158           recorded = object
  159             [ "query" .= q
  160             , "archived" .= archived
  161             , "match_count" .= (length matched + length childMatched)
  162             , "session_ids" .= fmap (sessionIdText . smId . fst) matched
  163             ]
  164       pure (OpResult [TrpText rendered] False recorded)

  166 handleGet :: SealPaths -> Value -> App OpResult
  167 handleGet paths v = do
  168   let mSidText = textField v "session_id"
  169       offset = fromMaybe 0 (intField v "offset")
  170       mLimit = intField v "limit"
  171   case mSidText >>= either (const Nothing) Just . mkSessionId . T.strip of
  172     Nothing -> pure (OpResult [TrpText "invalid session id"] True (object []))
  173     Just sid -> do
  174       metaExists <- liftIO (doesFileExist (sessionMetaPath paths sid))
  175       mMeta <- if not metaExists
  176                  then pure Nothing
  177                  else liftIO (decodeFileStrict (sessionMetaPath paths sid))
  178       case mMeta of
  179         Just meta -> do
  180           msgs <- liftIO (readSessionMessages paths sid)
  181           let total = length msgs
  182               page = paginate sessionPageParams offset mLimit msgs
  183               windowMsgs = pgItems page
  184               header = renderSessionMeta meta
  185               body = T.intercalate "\n\n"
  186                 (zipWith (renderMessage (pgOffset page)) [0..] windowMsgs)
  187               footer = renderPageFooter (pgOffset page) (length windowMsgs) total (pgHasMore page)
  188               rendered = T.intercalate "\n\n" (filter (not . T.null) [header, body, footer])
  189               recorded = object
  190                 [ "session_id" .= sessionIdText sid
  191                 , "provider" .= smProvider meta
  192                 , "model" .= smModel meta
  193                 , "offset" .= pgOffset page
  194                 , "limit" .= length windowMsgs
  195                 , "total_messages" .= total
  196                 , "has_more" .= pgHasMore page
  197                 ]
  198           pure (OpResult [TrpText rendered] False recorded)
  199         Nothing -> do
  200           mChild <- liftIO (resolveChildSessionPath paths sid)
  201           case mChild of
  202             Nothing -> pure (OpResult
  203               [TrpText ("session not found: " <> sessionIdText sid)] True
  204               (object ["session_id" .= sessionIdText sid]))
  205             Just (parentSid, childDir) -> do
  206               msgs <- liftIO (readMessagesFromDir childDir)
  207               let total = length msgs
  208                   page = paginate sessionPageParams offset mLimit msgs
  209                   windowMsgs = pgItems page
  210                   header = "[child session of " <> sessionIdText parentSid <> "] " <> sessionIdText sid
  211                   body = T.intercalate "\n\n"
  212                     (zipWith (renderMessage (pgOffset page)) [0..] windowMsgs)
  213                   footer = renderPageFooter (pgOffset page) (length windowMsgs) total (pgHasMore page)
  214                   rendered = T.intercalate "\n\n" (filter (not . T.null) [header, body, footer])
  215                   recorded = object
  216                     [ "session_id" .= sessionIdText sid
  217                     , "child_of" .= sessionIdText parentSid
  218                     , "offset" .= pgOffset page
  219                     , "limit" .= length windowMsgs
  220                     , "total_messages" .= total
  221                     , "has_more" .= pgHasMore page
  222                     ]
  223               pure (OpResult [TrpText rendered] False recorded)

  465 -- | Render a SessionMeta as a compact one-line summary.
  466 renderSessionMeta :: SessionMeta -> Text
  467 renderSessionMeta m =
  468   let sid = sessionIdText (smId m)
  469       desc = fromMaybe "(no description)" (smDescription m)
  470       created = T.pack (formatTime defaultTimeLocale "%Y-%m-%d %H:%M" (smCreatedAt m))
  471       active = T.pack (formatTime defaultTimeLocale "%Y-%m-%d %H:%M" (smLastActive m))
  472   in sid <> " | " <> smProvider m <> "/" <> smModel m
  473        <> " | " <> desc
  474        <> " | created: " <> created <> " | last active: " <> active

  476 -- | Search a single session for the query string. Returns 'Just snippet'
  477 -- if the session matches (description or first user message), 'Nothing'
  478 -- otherwise. The snippet is the matched text for context.
  479 searchSession :: SealPaths -> Text -> SessionMeta -> IO (SessionMeta, Maybe Text)
  480 searchSession paths qLower meta = do
  481   let descLower = maybe "" T.toCaseFold (smDescription meta)
  482       descMatch = qLower `T.isInfixOf` descLower
  483   mSnippet <- firstUserSnippet paths (smId meta)
  484   let snippetLower = maybe "" T.toCaseFold mSnippet
  485       snippetMatch = qLower `T.isInfixOf` snippetLower
  486   if descMatch || snippetMatch
  487     then pure (meta, Just (fromMaybe (fromMaybe "(no description)" (smDescription meta)) mSnippet))
  488     else pure (meta, Nothing)
  489 
  490 -- | Search a child (sub-agent) session's transcript for the query string.
  491 -- Returns 'Just snippet' when the first user message matches, 'Nothing'
  492 -- otherwise. The child is identified by its parent + child SessionId pair.
  493 searchChildSession
  494   :: SealPaths -> Text -> (SessionId, SessionId) -> IO (SessionId, SessionId, Maybe Text)
  495 searchChildSession paths qLower (parentSid, childSid) = do
  496   mSnippet <- firstUserSnippet paths childSid
  497   let snippetLower = maybe "" T.toCaseFold mSnippet
  498       snippetMatch = qLower `T.isInfixOf` snippetLower
  499   pure (parentSid, childSid, if snippetMatch then mSnippet else Nothing)
  500 
  501 -- | Render a one-line summary for a child (sub-agent) session, attributing it
  502 -- to its parent. The marker @[child of \<parent\>]@ distinguishes child
  503 -- sessions from top-level ones in SESSION_LIST / SESSION_SEARCH output.
  504 renderChildSummary :: SessionId -> SessionId -> Text
  505 renderChildSummary parentSid childSid =
  506   "[child of " <> sessionIdText parentSid <> "] " <> sessionIdText childSid

  508 -- | Extract the first user message's text from a session's
  509 -- @conversation.jsonl@. Returns 'Nothing' when the session has no
  510 -- conversation or no user message with text content. Falls back to the
  511 -- child (sub-agent) transcript path when the top-level conversation file
  512 -- does not exist.
  513 firstUserSnippet :: SealPaths -> SessionId -> IO (Maybe Text)
  514 firstUserSnippet paths sid = do
  515   let convPath = sessionConversationPath paths sid
  516   exists <- doesFileExist convPath
  517   if exists
  518     then firstUserSnippetFromDir (sessionDir paths sid)
  519     else do
  520       mChild <- resolveChildSessionPath paths sid
  521       case mChild of
  522         Just (_, childDir) -> firstUserSnippetFromDir childDir
  523         Nothing            -> pure Nothing
  524 
  525 -- | Extract the first user message's text from a @conversation.jsonl@ in the
  526 -- given directory. Returns 'Nothing' when the file is absent or has no user
  527 -- message with text content.
  528 firstUserSnippetFromDir :: FilePath -> IO (Maybe Text)
  529 firstUserSnippetFromDir dir = do
  530   let convPath = dir </> "conversation.jsonl"
  531   exists <- doesFileExist convPath
  532   if not exists
  533     then pure Nothing
  534     else do
  535       raw <- readFileTextStrict convPath
  536       let msgs = mapMaybe decodeMsg (T.lines raw)
  537       pure (snippetFromMessages msgs)
  538 
  539 -- | Extract the first user message's text from a list of messages.
  540 snippetFromMessages :: [Message] -> Maybe Text
  541 snippetFromMessages msgs =
  542   case filter (\m -> msgRole m == User) msgs of
  543     (m : _) -> case [t | CbText t <- msgContent m] of
  544       (t : _) -> Just (truncateSnippet 120 t)
  545       []      -> Nothing
  546     [] -> Nothing
  547 
  548 -- | Truncate a snippet to at most @n@ characters, appending an ellipsis.
  549 truncateSnippet :: Int -> Text -> Text
  550 truncateSnippet n t
  551   | T.length t <= n = t
  552   | otherwise       = T.take n t <> "…"
  553 
  554 -- | Read all messages from a session's @conversation.jsonl@. Returns @[]@
  555 -- when the session has no conversation file. Falls back to the child
  556 -- (sub-agent) transcript path when the top-level conversation file does not
  557 -- exist.
  558 readSessionMessages :: SealPaths -> SessionId -> IO [Message]
  559 readSessionMessages paths sid = do
  560   let convPath = sessionConversationPath paths sid
  561   exists <- doesFileExist convPath
  562   if exists
  563     then readMessagesFromDir (sessionDir paths sid)
  564     else do
  565       mChild <- resolveChildSessionPath paths sid
  566       case mChild of
  567         Just (_, childDir) -> readMessagesFromDir childDir
  568         Nothing            -> pure []
  569 
  570 -- | Read all messages from a @conversation.jsonl@ in the given directory.
  571 -- Returns @[]@ when the file is absent.
  572 readMessagesFromDir :: FilePath -> IO [Message]
  573 readMessagesFromDir dir = do
  574   let convPath = dir </> "conversation.jsonl"
  575   exists <- doesFileExist convPath
  576   if not exists
  577     then pure []
  578     else do
  579       raw <- readFileTextStrict convPath
  580       pure (mapMaybe decodeMsg (T.lines raw))
  581 
  582 -- | Decode a JSON line into a Message.
  583 decodeMsg :: Text -> Maybe Message
  584 decodeMsg line = A.decode (BL.fromStrict (TE.encodeUtf8 line))
  585 
  586 -- | Render a single message as readable text. The index is the 0-based
  587 -- position within the full transcript (offset + relative index), shown
  588 -- so the model can reference specific messages.
  589 renderMessage :: Int -> Int -> Message -> Text
  590 renderMessage offset relIdx msg =
  591   let globalIdx = offset + relIdx
  592       roleStr = case msgRole msg of
  593         User      -> "User"
  594         Assistant -> "Assistant"
  595       contentText = T.intercalate "\n" (map renderBlock (msgContent msg))
  596   in "## [" <> roleStr <> "] (#" <> T.pack (show globalIdx) <> ")\n" <> contentText
  597 
  598 -- | Render a content block as readable text.
  599 renderBlock :: ContentBlock -> Text
  600 renderBlock (CbText t) = t
  601 renderBlock (CbThinking t) = "[thinking] " <> t
  602 renderBlock (CbToolUse{cbName = OpName n}) = "[tool call: " <> n <> "]"
  603 renderBlock (CbToolResult{cbParts = parts, cbIsError = isErr}) =
  604   let label = if isErr then "[tool error]" else "[tool result]"
  605       content = T.intercalate "\n" [t | TrpText t <- parts]
  606   in if T.null content then label else label <> " " <> content
  607 
  608 -- | Render the pagination footer, telling the model how to page forward.
  609 renderPageFooter :: Int -> Int -> Int -> Bool -> Text
  610 renderPageFooter offset count total hasMore
  611   | total == 0 = "(0 messages in transcript)"
  612   | offset >= total && count == 0
  613   = "[offset " <> T.pack (show offset) <> " is past end of transcript ("
  614        <> T.pack (show total) <> " messages); read with offset=0 to start over]"
  615   | hasMore
  616   = "[messages " <> T.pack (show (offset + 1)) <> "-" <> T.pack (show (offset + count))
  617        <> " of " <> T.pack (show total) <> "; " <> T.pack (show (total - offset - count))
  618        <> " more - read with offset=" <> T.pack (show (offset + count)) <> " for the next window]"
  619   | otherwise
  620   = "[messages " <> T.pack (show (offset + 1)) <> "-" <> T.pack (show (offset + count))
  621        <> " of " <> T.pack (show total) <> " (end of transcript)]"

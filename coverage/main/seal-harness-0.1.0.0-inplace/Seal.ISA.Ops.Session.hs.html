   88 -- | Controls whether transcript pagination starts from the beginning
   89 -- (front-to-back) or the end (back-to-front) of the session.
   90 data ReadOrder = OrderAsc | OrderDesc
   91   deriving stock (Eq, Show)
   92 
   93 -- | Parse the @order@ field from the input JSON. Absent → 'OrderDesc'
   94 -- (back-to-front is the default, since the end of the session is usually
   95 -- what the caller wants). Unrecognized values also default to 'OrderDesc'.
   96 parseReadOrder :: Value -> ReadOrder
   97 parseReadOrder v =
   98   case textField v "order" of
   99     Just "front-to-back" -> OrderAsc
  100     Just "back-to-front" -> OrderDesc
  101     _                    -> OrderDesc

  103 -- | Dispatch to 'paginate' (front-to-back) or 'paginateDesc' (back-to-front).
  104 paginateBy :: ReadOrder -> PageParams -> Int -> Maybe Int -> [a] -> Page a
  105 paginateBy OrderAsc = paginate
  106 paginateBy OrderDesc = paginateDesc

  133 -- | Total rendered-output budget for a single @get@ page: the rendered
  134 -- text (header, message body, footer) never exceeds this many UTF-8
  135 -- bytes, so no foreign-session read can flood the caller's context.
  136 -- Oversized windows are cut with a truncation notice instead.
  137 -- 96 KiB (98304 bytes).
  138 getRenderBudgetBytes :: Int
  139 getRenderBudgetBytes = 98304
  140 
  141 -- | Per-message render cap: a single message whose rendered text exceeds
  142 -- this many bytes is cut at the cap and marked @[content truncated]@, so
  143 -- one huge tool result cannot consume the whole page budget. 8 KiB
  144 -- (8192 bytes).
  145 getRenderMessageBytes :: Int
  146 getRenderMessageBytes = 8192

  148 -- | Rendered-header cap: the rendered session-meta header line is capped
  149 -- to this many UTF-8 bytes. The header embeds the session description
  150 -- verbatim, and a description is unbounded agent-controlled input —
  151 -- without this cap a single oversized description would blow the page
  152 -- budget ('getRenderBudgetBytes') on its own. 2 KiB (2048 bytes).
  153 getRenderHeaderBytes :: Int
  154 getRenderHeaderBytes = 2048

  171 -- | A count-selected transcript window plus the pagination metadata the
  172 -- page footer and recorded fields need.
  173 data WindowedRead = WindowedRead
  174   { wrMessages :: [Message]  -- ^ the window's messages, in transcript order
  175   , wrMargin :: [Message]    -- ^ the messages just before the window start
  176                              -- (up to 'redactionScanMargin'), not rendered;
  177                              -- widens the secret-opcode scan of
  178                              -- 'redactWindowedRead'
  179   , wrWindowCount :: !Int   -- ^ count-based window size (line count on the idx path)
  180   , wrTotal :: !Int         -- ^ total messages (line count on the idx path)
  181   , wrOffset :: !Int        -- ^ 0-based start index of the window
  182   , wrHasMore :: !Bool      -- ^ more messages outside the window
  183   }
  184 
  185 -- | Number of messages before the window start that the read also loads
  186 -- ('wrMargin') for the secret-opcode scan: a tool-use block typically
  187 -- precedes its tool-result block in the same or the immediately previous
  188 -- message, so a 2-message margin is enough to associate a window-edge
  189 -- tool result with the tool-use that produced it.
  190 redactionScanMargin :: Int
  191 redactionScanMargin = 2
  192 
  193 -- | Resolve the transcript directory a @get@ read should use: the
  194 -- session's own dir when it has a conversation file, otherwise the
  195 -- child (sub-agent) dir when one exists (mirrors the whole-file read's
  196 -- directory choice).
  197 sessionReadDir :: SealPaths -> SessionId -> IO FilePath
  198 sessionReadDir paths sid = do
  199   let convPath = sessionConversationPath paths sid
  200   exists <- doesFileExist convPath
  201   if exists
  202     then pure (sessionDir paths sid)
  203     else do
  204       mChild <- resolveChildSessionPath paths sid
  205       pure (maybe (sessionDir paths sid) snd mChild)

  207 -- | Resolve the effective window parameters for a transcript read. An
  208 -- around-anchor (@around_index@), when present, overrides the caller's
  209 -- offset: the window becomes @limit@ messages starting at
  210 -- @max 0 (around_index - limit div 2)@, read front-to-back so the
  211 -- anchored message sits at the window's center. The caller's explicit
  212 -- limit is honored; without one the around-mode default is
  213 -- 'getAroundDefaultLimit' (resolved through 'windowSize', so it is
  214 -- passed as an explicit limit — the pagination floor cannot resize it).
  215 -- The start clamps to @[0, max 0 (total - limit)]@, so an anchor before
  216 -- the transcript reads from message 0 and an anchor beyond the end
  217 -- yields the window ending at the tail — out-of-range anchors clamp,
  218 -- never error. Without an anchor the caller's order, offset, and limit
  219 -- pass through unchanged.
  220 effectiveWindowParams
  221   :: ReadOrder -> PageParams -> Int -> Maybe Int -> Maybe Int -> Int
  222   -> (ReadOrder, Int, Maybe Int)
  223 effectiveWindowParams order params offset mLimit mAround total =
  224   case mAround of
  225     Nothing -> (order, offset, mLimit)
  226     Just around ->
  227       let limit = windowSize params total
  228             (Just (fromMaybe getAroundDefaultLimit mLimit))
  229           start = clamp 0 (max 0 (total - limit))
  230             (max 0 (around - limit `div` 2))
  231       in (OrderAsc, start, Just limit)
  232 
  233 -- | Read the count-selected message window from a transcript directory.
  234 -- Window selection is count-based ('paginateBy') on both paths; when
  235 -- @conversation.idx@ exists only the window's byte range is read
  236 -- ('readConvLines'), otherwise the whole file is read and sliced. An
  237 -- around-anchor ('effectiveWindowParams') is translated here — where the
  238 -- message total is known — into an equivalent front-to-back window, so
  239 -- the anchored read runs through the same windowing machinery. On
  240 -- the idx path the counts are the index's line counts — equal to the
  241 -- decoded-message counts for well-formed files, diverging only on
  242 -- malformed lines (accepted: guard behavior is identical, only the
  243 -- count-based totals can differ). Any index read failure (missing
  244 -- conversation file, corrupt index) falls back to the whole-file read.
  245 -- Both paths also load a 'redactionScanMargin'-sized margin of messages
  246 -- before the window start ('wrMargin').
  247 readTranscriptWindow
  248   :: ReadOrder -> PageParams -> Int -> Maybe Int -> Maybe Int -> FilePath -> IO WindowedRead
  249 readTranscriptWindow order params offset mLimit mAround dir = do
  250   let convPath = dir </> "conversation.jsonl"
  251       idxPath = dir </> "conversation.idx"
  252   idxExists <- doesFileExist idxPath
  253   if not idxExists
  254     then viaWholeFile
  255     else do
  256       lineTotal <- convLineCount idxPath
  257       if lineTotal <= 0
  258         then viaWholeFile
  259         else do
  260           -- Window selection is pure count math; the unit list reuses
  261           -- 'paginateBy' itself so the idx path cannot drift from the
  262           -- whole-file path's windowing.
  263           let (order', offset', mLimit') =
  264                 effectiveWindowParams order params offset mLimit mAround lineTotal
  265               page = paginateBy order' params offset' mLimit' (replicate lineTotal ())
  266               start = pgOffset page
  267               count = length (pgItems page)
  268               marginStart = max 0 (start - redactionScanMargin)
  269           eMsgs <- readConvLines convPath idxPath marginStart (start + count)
  270           case eMsgs of
  271             Right msgs ->
  272               let margin = take (start - marginStart) msgs
  273                   window = drop (start - marginStart) msgs
  274               in pure WindowedRead
  275                 { wrMessages = window
  276                 , wrMargin = margin
  277                 , wrWindowCount = count
  278                 , wrTotal = lineTotal
  279                 , wrOffset = start
  280                 , wrHasMore = pgHasMore page
  281                 }
  282             Left _ -> viaWholeFile
  283   where
  284     viaWholeFile = do
  285       msgs <- readMessagesFromDir dir
  286       let (order', offset', mLimit') =
  287             effectiveWindowParams order params offset mLimit mAround (length msgs)
  288           page = paginateBy order' params offset' mLimit' msgs
  289           start = pgOffset page
  290           marginStart = max 0 (start - redactionScanMargin)
  291       pure WindowedRead
  292         { wrMessages = pgItems page
  293         , wrMargin = take (start - marginStart) (drop marginStart msgs)
  294         , wrWindowCount = length (pgItems page)
  295         , wrTotal = pgTotal page
  296         , wrOffset = start
  297         , wrHasMore = pgHasMore page
  298         }

  300 -- | Redact secret-opcode tool results from a windowed read's messages
  301 -- before rendering: 'redactMessages' scans the window plus its pre-window
  302 -- margin ('wrMargin') for 'CbToolUse' blocks of secret opcodes and
  303 -- replaces the matching 'CbToolResult' parts inside the window with the
  304 -- transcript placeholder. Foreign transcripts written before their
  305 -- handle had secret ops configured would otherwise leak raw secret-op
  306 -- results to any reader. The margin is only scanned, never rendered.
  307 redactWindowedRead :: WindowedRead -> WindowedRead
  308 redactWindowedRead wr = wr
  309   { wrMessages =
  310       redactMessages secretOpcodes (wrMargin wr <> wrMessages wr) (wrMessages wr)
  311   }

  336 handleNew :: SealPaths -> Value -> App OpResult
  337 handleNew paths v = do
  338   let provider = fromMaybe "anthropic" (textField v "provider")
  339       model = fromMaybe "claude-sonnet-4-20250514" (textField v "model")
  340       channel = fromMaybe "api" (textField v "channel")
  341       mDesc = textField v "description"

  343   let meta' = case mDesc of
  344         Just d | not (T.null (T.strip d)) -> meta { smDescription = Just d }

  347     Just d | not (T.null (T.strip d)) -> liftIO (saveSessionMeta paths meta')
  348     _ -> pure ()
  349   let rendered = renderSessionMeta meta'
  350       recorded = object
  351         [ "session_id" .= sessionIdText (smId meta')
  352         , "provider" .= smProvider meta'
  353         , "model" .= smModel meta'
  354         , "channel" .= smChannel meta'
  355         , "description" .= smDescription meta'
  356         ]
  357   pure (OpResult [TrpText rendered] False recorded)

  359 handleList :: SealPaths -> Value -> App OpResult
  360 handleList paths v = do
  361   let archived = fromMaybe False (boolField v "archived")
  362       includeChildren = fromMaybe False (boolField v "include_children")
  363   metas <- liftIO (if archived then listArchivedSessions paths else listSessions paths)
  364   children <- if includeChildren
  365                 then liftIO (listChildSessions paths)
  366                 else pure []
  367   let parentLines = map renderSessionMeta metas
  368       childLines = [ renderChildSummary parentSid childSid
  369                    | (parentSid, childSid) <- children ]
  370       allLines = parentLines <> childLines
  371       rendered = case allLines of
  372         [] -> "(no sessions found)"
  373         _ -> T.intercalate "\n" allLines
  374       recorded = object
  375         [ "count" .= length metas
  376         , "archived" .= archived
  377         , "include_children" .= includeChildren
  378         , "child_count" .= length children
  379         , "session_ids" .= fmap (sessionIdText . smId) metas
  380         ]
  381   pure (OpResult [TrpText rendered] False recorded)

  383 -- | Search sessions by query. The sessions in the current session's
  384 -- lineage are excluded from both transcript matches and child (sub-agent)
  385 -- matches — a session never sees its own contents, nor its ancestors',
  386 -- in search results. The lineage is ordered self-first (the current
  387 -- session is the head); an empty lineage excludes nothing. Results are
  388 -- capped at 'searchResultCap' rendered rows (backend matches first, then
  389 -- child matches) with a footer when matches were cut; every rendered
  390 -- snippet is clamped to 'searchSnippetCap' characters of content.
  391 handleSearch :: SessionSearchBackend -> SealPaths -> [SessionId] -> Value -> App OpResult
  392 handleSearch searchBackend paths lineage v = do
  393   let q = fromMaybe "" (textField v "query")
  394   if T.null (T.strip q)
  395     then pure (OpResult [TrpText "search: query is empty"] True (object []))
  396     else do
  397       let archived = fromMaybe False (boolField v "archived")
  398           qLower = T.toCaseFold (T.strip q)
  399       rawMatched <- liftIO (ssbSearch searchBackend paths q archived)
  400       children <- liftIO (listChildSessions paths)
  401       childResults <- liftIO (mapM (searchChildSession paths qLower) children)
  402       let excluded = lineage
  403           matched = [ (m, snip) | (m, snip) <- rawMatched, smId m `notElem` excluded ]
  404           childMatched = [ (p, c, snip) | (p, c, Just snip) <- childResults, c `notElem` excluded ]
  405           matchTotal = length matched + length childMatched
  406           shownMatched = take searchResultCap matched
  407           shownChildMatched =
  408             take (max 0 (searchResultCap - length shownMatched)) childMatched
  409           shownCount = length shownMatched + length shownChildMatched
  410           capFooter
  411             | matchTotal > searchResultCap =
  412                 "\n\n(showing " <> T.pack (show shownCount) <> " of "
  413                   <> T.pack (show matchTotal) <> " matches — refine your query)"
  414             | otherwise = ""
  415           rendered
  416             | null matched && null childMatched = "(no sessions found)"
  417             | otherwise = T.intercalate "\n\n"
  418                 ([ renderSessionMeta m <> "\n  snippet: "
  419                      <> truncateSnippet searchSnippetCap snip
  420                  | (m, snip) <- shownMatched ]
  421                 <> [ renderChildSummary p c <> "\n  snippet: "
  422                        <> truncateSnippet searchSnippetCap snip
  423                    | (p, c, snip) <- shownChildMatched ])
  424                 <> capFooter
  425           recorded = object
  426             [ "query" .= q
  427             , "archived" .= archived
  428             , "match_count" .= shownCount
  429             , "match_total" .= matchTotal
  430             , "session_ids" .= fmap (sessionIdText . smId . fst) shownMatched
  431             ]
  432       pure (OpResult [TrpText rendered] False recorded)

  434 -- | Refusal message for a session attempting to read its own transcript.
  435 -- Reading own transcript content appends the read result to that same
  436 -- transcript, so each successive read includes the outputs of every
  437 -- previous read — an unbounded feedback loop that can exhaust the context
  438 -- window in minutes.
  439 selfReadRefusal :: Text
  440 selfReadRefusal =
  441   T.concat
  442     [ "cannot read the transcript of the current session: a session reading "
  443     , "its own transcript appends the result to that same transcript, "
  444     , "creating an unbounded feedback loop. Use SESSION_MANAGE action=list "
  445     , "or search to find other sessions, or AGENT_STATUS for sub-agent progress."
  446     ]

  448 -- | Refusal message for a session attempting to read the transcript of an
  449 -- ancestor session (its spawner, or any earlier generation). The ancestor
  450 -- transcript can contain reads of this session's own work; reading it
  451 -- back appends those reads to this session's transcript in the same
  452 -- feedback loop the self-read guard exists to prevent.
  453 ancestorReadRefusal :: SessionId -> Text
  454 ancestorReadRefusal sid =
  455   T.concat
  456     [ "cannot read the transcript of an ancestor session ("
  457     , sessionIdText sid
  458     , " is in this session's lineage): the ancestor transcript can contain "
  459     , "reads of your own work, and reading it back appends those reads to "
  460     , "your transcript in a loop. Use SESSION_MANAGE action=list or search "
  461     , "for unrelated sessions, or AGENT_STATUS for sub-agent progress."

  464 handleGet :: SealPaths -> [SessionId] -> Value -> App OpResult
  465 handleGet paths lineage v = do
  466   let mSidText = textField v "session_id"
  467       offset = fromMaybe 0 (intField v "offset")
  468       mLimit = intField v "limit"
  469       mAround = intField v "around_index"
  470       order = parseReadOrder v
  471   case mSidText >>= either (const Nothing) Just . mkSessionId . T.strip of
  472     Nothing -> pure (OpResult [TrpText "invalid session id"] True (object []))
  473     Just sid
  474       | self : _ <- lineage, sid == self -> pure (OpResult [TrpText selfReadRefusal] True (object
  475           [ "session_id" .= sessionIdText sid
  476           , "self_read" .= True
  477           ]))
  478       | sid `elem` lineage -> pure (OpResult
  479           [TrpText (ancestorReadRefusal sid)] True (object
  480           [ "session_id" .= sessionIdText sid
  481           , "ancestor_read" .= True
  482           ]))
  483       | otherwise -> do
  484       metaExists <- liftIO (doesFileExist (sessionMetaPath paths sid))
  485       mMeta <- if not metaExists
  486                  then pure Nothing
  487                  else liftIO (decodeFileStrict (sessionMetaPath paths sid))
  488       case mMeta of
  489         Just meta -> do
  490           dir <- liftIO (sessionReadDir paths sid)
  491           wr <- liftIO (readTranscriptWindow order sessionPageParams offset mLimit mAround dir)
  492           pure (renderGetPage (renderSessionMeta meta) order offset mAround
  493             [ "session_id" .= sessionIdText sid
  494             , "provider" .= smProvider meta
  495             , "model" .= smModel meta
  496             ] (redactWindowedRead wr))
  497         Nothing -> do
  498           mChild <- liftIO (resolveChildSessionPath paths sid)
  499           case mChild of
  500             Nothing -> pure (OpResult
  501               [TrpText ("session not found: " <> sessionIdText sid)] True
  502               (object ["session_id" .= sessionIdText sid]))
  503             Just (parentSid, childDir) -> do
  504               wr <- liftIO (readTranscriptWindow order sessionPageParams offset mLimit mAround childDir)
  505               pure (renderGetPage
  506                 ("[child session of " <> sessionIdText parentSid <> "] " <> sessionIdText sid)
  507                 order offset mAround
  508                 [ "session_id" .= sessionIdText sid
  509                 , "child_of" .= sessionIdText parentSid
  510                 ] (redactWindowedRead wr))

  512 -- | Assemble the @get@ page from a windowed read: header line (capped at
  513 -- 'getRenderHeaderBytes' UTF-8 bytes), byte-budgeted message body, and
  514 -- footer (the count-based paging footer plus a truncation notice when the
  515 -- byte budget cut the body). The @userOffset@ is the caller-supplied
  516 -- offset, used for the footer's paging guidance; @mAround@ is the
  517 -- caller-supplied @around_index@ anchor — when present the window was
  518 -- read front-to-back around that anchor, the footer points at
  519 -- @around_index@ for the adjacent windows, and the recorded metadata
  520 -- reports the effective window start plus the echoed anchor. The
  521 -- recorded metadata stays content-free; @truncated@ is true when any
  522 -- output was cut — the header, a single message, or the page budget.
  523 renderGetPage :: Text -> ReadOrder -> Int -> Maybe Int -> [Pair] -> WindowedRead -> OpResult
  524 renderGetPage rawHeader order userOffset mAround extraRecorded wr =
  525   let (header, headerTruncated) = capHeader getRenderHeaderBytes rawHeader
  526       bodyBudget = max minBodyBudget
  527         (getRenderBudgetBytes - utf8Bytes header - getRenderFooterReserveBytes)
  528       minBodyBudget = getRenderMessageBytes + utf8Bytes contentTruncatedMarker + 2
  529       br = renderBudgetedWindow bodyBudget getRenderMessageBytes (wrOffset wr) (wrMessages wr)
  530       pageTruncated = headerTruncated || brAnyTruncated br || brBudgetHit br
  531       body = T.intercalate "\n\n" (brParts br)
  532       effOrder = maybe order (const OrderAsc) mAround
  533       countFooter = case mAround of
  534         Just _  -> renderAroundPageFooter (wrOffset wr) (wrWindowCount wr) (wrTotal wr)
  535         Nothing -> renderPageFooter order (wrOffset wr) userOffset
  536           (wrWindowCount wr) (wrTotal wr) (wrHasMore wr)
  537       footer
  538         | brBudgetHit br =
  539             countFooter <> "\n[output truncated at " <> T.pack (show (brRenderedBytes br))
  540               <> " bytes — narrow the window (limit / offset / around_index) for the part you need]"
  541         | otherwise = countFooter
  542       rendered = T.intercalate "\n\n" (filter (not . T.null) [header, body, footer])
  543       recorded = object
  544         ( extraRecorded <>
  545           [ "order" .= renderOrderJson effOrder
  546           , "offset" .= wrOffset wr
  547           ] <>
  548           maybe [] (\a -> ["around_index" .= a]) mAround <>
  549           [ "limit" .= wrWindowCount wr
  550           , "total_messages" .= wrTotal wr
  551           , "has_more" .= wrHasMore wr
  552           , "truncated" .= pageTruncated
  553           ] )
  554   in OpResult [TrpText rendered] False recorded

  893 -- | The outcome of rendering a message window under the byte budget.
  894 data BudgetedRender = BudgetedRender
  895   { brParts :: [Text]        -- ^ rendered messages, in window order
  896   , brAnyTruncated :: !Bool  -- ^ any content was cut (per message or budget)
  897   , brBudgetHit :: !Bool     -- ^ the budget stopped further messages
  898   , brRenderedBytes :: !Int  -- ^ UTF-8 bytes of the rendered body
  899   }
  900 
  901 -- | Render window messages in order under the byte budget: each message
  902 -- is capped at @perMsgCap@ bytes (marked 'contentTruncatedMarker' when
  903 -- cut), and rendering stops before the first message that would push the
  904 -- accumulated body past @bodyBudget@. Callers floor the budget so at
  905 -- least one (capped) message always renders.
  906 renderBudgetedWindow :: Int -> Int -> Int -> [Message] -> BudgetedRender
  907 renderBudgetedWindow bodyBudget perMsgCap displayOffset = go 0 0 False []
  908   where
  909     go _ acc anyTrunc out [] =
  910       BudgetedRender (reverse out) anyTrunc False acc
  911     go relIdx acc anyTrunc out (m : rest) =
  912       let rendered = renderMessage displayOffset relIdx m
  913           (capped, msgTrunc) = capMessage perMsgCap rendered
  914           separator = if null out then 0 else 2
  915           acc' = acc + separator + utf8Bytes capped
  916       in if acc' > bodyBudget
  917            then BudgetedRender (reverse out) anyTrunc True acc
  918            else go (relIdx + 1) acc' (anyTrunc || msgTrunc) (capped : out) rest
  919 
  920 -- | Cap a single rendered message at @perMsgCap@ UTF-8 bytes, appending
  921 -- the truncation marker when the text was cut.
  922 capMessage :: Int -> Text -> (Text, Bool)
  923 capMessage perMsgCap rendered
  924   | utf8Bytes rendered > perMsgCap =
  925       (truncateToBytes perMsgCap rendered <> contentTruncatedMarker, True)
  926   | otherwise = (rendered, False)
  927 
  928 -- | Cap the rendered header at @cap@ UTF-8 bytes, marker included: an
  929 -- oversized header is cut to @cap@ minus the marker's bytes and
  930 -- 'contentTruncatedMarker' is appended, so the capped header never
  931 -- exceeds @cap@ even with the marker (same technique as 'capMessage').
  932 capHeader :: Int -> Text -> (Text, Bool)
  933 capHeader cap raw
  934   | utf8Bytes raw > cap =
  935       ( truncateToBytes (cap - utf8Bytes contentTruncatedMarker) raw
  936           <> contentTruncatedMarker
  937       , True )
  938   | otherwise = (raw, False)
  939 
  940 -- | Truncate text to at most @cap@ UTF-8 bytes without splitting a code
  941 -- point (the cut walks back over continuation bytes).
  942 truncateToBytes :: Int -> Text -> Text
  943 truncateToBytes cap t =
  944   let bs = TE.encodeUtf8 t
  945   in if BS.length bs <= cap
  946        then t
  947        else TE.decodeUtf8With lenientDecode (BS.take (cutBoundary cap bs) bs)
  948 
  949 -- | The largest cut point @<= cap@ that does not split a UTF-8 code
  950 -- point: walks back from @cap@ while the byte at the cut is a
  951 -- continuation byte (@0b10xxxxxx@).
  952 cutBoundary :: Int -> BS.ByteString -> Int
  953 cutBoundary cap bs = go (min cap (BS.length bs))
  954   where
  955     go p
  956       | p <= 0 = 0
  957       | p >= BS.length bs = p
  958       | isContinuation (BS.index bs p) = go (p - 1)
  959       | otherwise = p
  960     isContinuation w = w >= 0x80 && w <= 0xBF
  961 
  962 -- | UTF-8 encoded byte length of a text.
  963 utf8Bytes :: Text -> Int
  964 utf8Bytes = BS.length . TE.encodeUtf8

  982 -- | Render a single message as readable text. The index is the 0-based
  983 -- position within the full transcript (offset + relative index), shown
  984 -- so the model can reference specific messages.
  985 renderMessage :: Int -> Int -> Message -> Text
  986 renderMessage offset relIdx msg =
  987   let globalIdx = offset + relIdx
  988       roleStr = case msgRole msg of
  989         User      -> "User"
  990         Assistant -> "Assistant"
  991       contentText = T.intercalate "\n" (map renderBlock (msgContent msg))
  992   in "## [" <> roleStr <> "] (#" <> T.pack (show globalIdx) <> ")\n" <> contentText

  994 -- | Render a content block as readable text.
  995 renderBlock :: ContentBlock -> Text
  996 renderBlock (CbText t) = t
  997 renderBlock (CbThinking t) = "[thinking] " <> t
  998 renderBlock (CbToolUse{cbName = OpName n}) = "[tool call: " <> n <> "]"
  999 renderBlock (CbToolResult{cbParts = parts, cbIsError = isErr}) =
 1000   let label = if isErr then "[tool error]" else "[tool result]"
 1001       content = T.intercalate "\n" [t | TrpText t <- parts]
 1002   in if T.null content then label else label <> " " <> content

 1004 -- | Render the pagination footer, telling the model how to page further.
 1005 -- The @displayOffset@ is the 0-based index in the transcript where the
 1006 -- window starts (used for "messages X-Y"). The @userOffset@ is the offset
 1007 -- value the caller passed (used for "read with offset=Z" guidance) — in
 1008 -- front-to-back mode these are the same; in back-to-front mode they differ.
 1009 renderPageFooter :: ReadOrder -> Int -> Int -> Int -> Int -> Bool -> Text
 1010 renderPageFooter order displayOffset userOffset count total hasMore
 1011   | total == 0 = "(0 messages in transcript)"
 1012   | count == 0 = pastEndMsg
 1013   | hasMore    = moreMsg
 1014   | otherwise  = endMsg
 1015   where
 1016     nextOffset = userOffset + count
 1017     rangeText  = T.pack (show (displayOffset + 1)) <> "-"
 1018                    <> T.pack (show (displayOffset + count))
 1019     totalText  = T.pack (show total)
 1020     pastEndMsg = case order of
 1021       OrderAsc ->
 1022         "[offset " <> T.pack (show userOffset) <> " is past end of transcript ("
 1023           <> totalText <> " messages); read with offset=0 to start over]"
 1024       OrderDesc ->
 1025         "[offset " <> T.pack (show userOffset) <> " is past the beginning of transcript ("
 1026           <> totalText <> " messages); read with offset=0 to start from the end]"
 1027     moreMsg = case order of
 1028       OrderAsc ->
 1029         "[messages " <> rangeText <> " of " <> totalText <> "; "
 1030           <> T.pack (show (total - displayOffset - count))
 1031           <> " more - read with offset=" <> T.pack (show nextOffset)
 1032           <> " for the next window]"
 1033       OrderDesc ->
 1034         "[messages " <> rangeText <> " of " <> totalText <> "; "
 1035           <> T.pack (show displayOffset)
 1036           <> " earlier - read with offset=" <> T.pack (show nextOffset)
 1037           <> " for the next page back]"
 1038     endMsg = case order of
 1039       OrderAsc ->
 1040         "[messages " <> rangeText <> " of " <> totalText <> " (end of transcript)]"
 1041       OrderDesc ->

 1044 -- | Render the around-mode page footer: the count-based global range
 1045 -- (the same @messages X-Y of Z@ the offset footer reports) plus guidance
 1046 -- pointing at @around_index@ anchors for the adjacent windows. The
 1047 -- anchors are the message indices just outside each window edge; an
 1048 -- anchored window always contains its anchor message, so stepping to
 1049 -- the edge anchors never skips messages.
 1050 renderAroundPageFooter :: Int -> Int -> Int -> Text
 1051 renderAroundPageFooter displayOffset count total
 1052   | total == 0 = "(0 messages in transcript)"
 1053   | before == 0 && after == 0 =
 1054       "[messages " <> rangeText <> " of " <> totalText <> " (whole transcript)]"
 1055   | otherwise =
 1056       "[messages " <> rangeText <> " of " <> totalText <> "; " <> partText
 1057         <> " - read with " <> T.intercalate " or " anchorTexts <> "]"
 1058   where
 1059     before = displayOffset
 1060     windowEnd = displayOffset + count
 1061     after = total - windowEnd
 1062     rangeText = T.pack (show (displayOffset + 1)) <> "-" <> T.pack (show windowEnd)
 1063     totalText = T.pack (show total)
 1064     partText = case (before, after) of
 1065       (0, a) -> T.pack (show a) <> " after"
 1066       (b, 0) -> T.pack (show b) <> " before"
 1067       (b, a) -> T.pack (show b) <> " before, " <> T.pack (show a) <> " after"
 1068     anchorTexts =
 1069       [ "around_index=" <> T.pack (show (displayOffset - 1))
 1070           <> " for an earlier window" | before > 0 ]
 1071         <> [ "around_index=" <> T.pack (show windowEnd)
 1072                <> " for a later window" | after > 0 ]

    2 -- | SETUP_REPO (Untrusted): clone a remote repository into the session's
    3 -- workspace so the model can work with its code and discover any skills it
    4 -- ships. The opcode is VCS-agnostic in name and result shape (\"cloned at
    5 -- \<path\>\") but git-only in implementation for now; a future @vcs@ arg or
    6 -- URL-scheme dispatch can add others without changing the contract.
    7 --
    8 -- Untrusted because upstream repo content is untrusted — a clone is a
    9 -- filesystem mutation of untrusted-origin data, so it goes through the
   10 -- sandboxed 'UntrustedIO' + the 'Autonomy' confirmation gate, not a
   11 -- trusted shell. This is strictly safer than the prior pattern of the
   12 -- model running @git clone@ via 'SHELL_EXEC' and hoping.
   13 --
   14 -- Semantics:
   15 --
   16 --   * Shallow clone (@git clone --depth 1@) to keep large repos cheap.
   17 --   * Multiple repos are supported: each clones into
   18 --     @\<workdir\>\/\<sanitized-repo-name\>\/@.
   19 --   * Optional @branch@ (@{url, branch}@): a fresh clone checks that branch
   20 --     (or tag — a tag yields a detached HEAD) out via a single
   21 --     @--branch=\<ref\>@ argv token. The value reaches a subprocess argv, so
   22 --     it is carried by the validated 'Seal.Tools.Args.BranchName' newtype,
   23 --     never raw 'Text'.
   24 --   * Idempotent: if the target dir already holds a clone of the same
   25 --     remote URL and the requested branch is already checked out, the call
   26 --     is a no-op (\"Repo already exists — no-op\"). A differing / unknown /
   27 --     detached current branch is SWITCHED ('switchBranch': fetch the branch
   28 --     ref at depth 1, then check it out) — never a re-clone, and the
   29 --     directory is never deleted.
   30 --     If the dir exists but its remote differs, the call errors (\"a
   31 --     different repo already occupies that path\") rather than silently
   32 --     clobbering.
   33 --   * The clone runs in the workspace root (cwd = workdir) via
   34 --     'uioShellExec'; the URL is validated to a safe charset so it cannot
   35 --     inject shell metacharacters into the @\/bin\/sh -c@ single-arg
   36 --     boundary.
   37 --
   38 -- Credential resolution (W3 — design §4.2/§4.3, rev 3): 'setupRepoOp' takes
   39 -- a 'CloneDeps' closed-over param (mirrors @secretManageOp (cdVault deps)@).
   40 -- 'cloneRepoIO' does @lookupRepoByUrl@ against the registry
   41 -- (@cdRepoReg@) → if the URL matches a registered repo, resolve the
   42 -- credential via the no-disk seam ('Seal.SourceControl.Clone.resolveCloneTarget'
   43 -- using @cdVault@/@cdSshAgent@/@cdPinnedKnownHosts@/@cdKeyfilesDir@) and
   44 -- clones via @gh repo clone@ with @GH_TOKEN@ (PAT/MachineUser: the raw
   45 -- token from 'ceRawToken' injected as @GH_TOKEN@ env) or @git clone@ via
   46 -- 'uioShellExecGitEnv' (deploy key: @SSH_AUTH_SOCK@ + @GIT_SSH_COMMAND@).
   47 -- If the URL is NOT registered, fall through to a bare-URL clone (public
   48 -- repos, backward-compat). Stays Untrusted; credential resolution via
   49 -- 'liftIO' in the trusted plane. NO @runLocal@/@BackendExec@.

   95 -- | SETUP_REPO opcode. Input: @{url: Text, branch?: Text}@. Authorize:
   96 -- autonomy must not be 'Deny' and the input must parse + validate
   97 -- ('parseSetupRepoInput'). Run: a shallow @git clone@ into
   98 -- @\<workdir\>\/\<sanitized-repo-name\>@ via 'uioShellExec', after a
   99 -- branch-aware idempotency check (read the existing remote + the branch it
  100 -- has checked out). 'orRecorded': the target path + status (secret-free).
  101 --
  102 -- No 'WorkspaceRoot' argument: the clone's root is the 'UIOEnv' the
  103 -- dispatcher supplies (the per-session workdir), so the opcode never needs
  104 -- one — and the delegation worker (which has no workspace root in scope but
  105 -- does have the child's 'UIOEnv') can build this op for a handoff session's
  106 -- pre-turn checkout.
  107 setupRepoOp :: CloneDeps -> AutonomyLevel -> Opcode
  108 setupRepoOp deps autonomy = UntrustedOpcode
  109   { uoName = OpName "SETUP_REPO"
  110   , uoDesc = "Clone a remote repository into the session workspace (shallow, idempotent; an optional branch or tag is checked out). Discover any skills it ships via the available-skills catalog on the next turn."
  111   , uoInSchema = setupRepoSchema
  112   , uoOutSchema = object []
  113   , uoAuthorize = \v ->
  114       case parseSetupRepoInput v of
  115         Left err -> Left err
  116         Right _ -> case autonomy of
  117           Deny -> Left "SETUP_REPO denied by autonomy policy"
  118           _    -> Right ()
  119   , uoRun = \v ->
  120       let recorded = object [ "url" .= (urlField v :: Maybe Text) ]
  121       in case parseSetupRepoInput v of
  122            Left err -> pure (OpResult [TrpText err] True recorded)
  123            Right (url, mBranch) -> runSetupRepo deps url mBranch recorded
  124   }

  143 urlField :: Value -> Maybe Text
  144 urlField = parseMaybe (withObject "in" (.: "url"))

  146 -- | The optional @branch@ field. An absent field, @null@, or a non-string
  147 -- value is treated as absent (the remote's default branch).
  148 branchField :: Value -> Maybe Text
  149 branchField v = case parseMaybe (withObject "in" (.:? "branch")) v of
  150   Just (Just b) -> Just b
  151   _             -> Nothing

  153 -- | Parse + validate the SETUP_REPO input into the cloned URL and the
  154 -- optional 'BranchName'. Shared by 'uoAuthorize' and 'uoRun' so the two
  155 -- cannot disagree about what a valid input is. An absent, empty, or
  156 -- whitespace-only @branch@ is 'Nothing'. The 'Left' arm is the
  157 -- user-facing message (the callers render it verbatim).
  158 parseSetupRepoInput :: Value -> Either Text (Text, Maybe BranchName)
  159 parseSetupRepoInput v = do
  160   rawUrl <- case urlField v of
  161     Just u  -> Right u
  162     Nothing -> Left "SETUP_REPO requires {url:string}"
  163   url <- case validateRepoUrl rawUrl of
  164     Right u  -> Right u
  165     Left err -> Left ("SETUP_REPO: invalid url: " <> err)
  166   mBranch <- case branchField v of
  167     Nothing -> Right Nothing
  168     Just raw
  169       | T.null (T.strip raw) -> Right Nothing
  170       | otherwise -> case mkBranchName raw of
  171           Right b  -> Right (Just b)
  172           Left err -> Left ("SETUP_REPO: invalid branch: " <> err)
  173   pure (url, mBranch)

  175 -- | Validate a repo URL. Accepts @https://@, @http://@, @git://@,
  176 -- @ssh://@, and the @git\@host:path@ SCP-style form. Rejects any URL
  177 -- containing shell metacharacters that could break out of the
  178 -- @\/bin\/sh -c@ single-arg boundary (the @uioShellExec@ contract).
  179 -- Returns the trimmed URL on success, or an error message.
  180 validateRepoUrl :: Text -> Either Text Text
  181 validateRepoUrl raw =
  182   let url = T.strip raw
  183   in if T.null url
  184        then Left "url is empty"
  185        else if not (hasAcceptableScheme url)
  186               then Left "url must use https://, http://, git://, ssh://, or git@host:path"
  187               else if T.any isShellMetachar url
  188                      then Left "url contains forbidden shell metacharacters"
  189                      else Right url
  190 
  191 -- | True if the URL begins with an acceptable scheme or is SCP-style
  192 -- (@git\@host:path@, more generally @user\@host:path@).
  193 hasAcceptableScheme :: Text -> Bool
  194 hasAcceptableScheme url =
  195   any (`T.isPrefixOf` url)
  196       [ "https://", "http://", "git://", "ssh://" ]
  197   || isScpStyle url
  198 
  199 -- | Heuristic for @user\@host:path@ (SCP-style). Has an @\@ before the
  200 -- first @/@ (if any), and a @:@ after the host, and no scheme prefix.
  201 isScpStyle :: Text -> Bool
  202 isScpStyle url =
  203   case T.breakOn "@" url of
  204     (user, rest) | not (T.null user), not (T.null rest) ->
  205       let afterAt = T.drop 1 rest
  206       in T.any (== ':') afterAt
  207              && not (":" `T.isPrefixOf` afterAt)   -- not ':host:path' (no user)
  208              && not ("/" `T.isPrefixOf` user)      -- not a path with an '@'
  209     _ -> False
  210 
  211 -- | Shell metacharacters forbidden in a repo URL (the @\/bin\/sh -c@
  212 -- single-arg boundary treats these specially). A legit git URL never
  213 -- contains them; their presence signals an injection attempt.
  214 isShellMetachar :: Char -> Bool
  215 isShellMetachar c = c `elem` (";`$\\\"'|\n\r&<>(){}*?[]#!~" :: String)

  217 -- | 'normalizeRepoUrl' is the shared normalizer in "Seal.SourceControl.Repo"
  218 -- (W1 moved it there so both 'ISA.Ops.Repo' and 'lookupRepoByUrl' use the SAME
  219 -- normalizer). Re-exported here for backward compatibility with existing
  220 -- callers (e.g. 'runSetupRepo's idempotency check below).

  222 sanitizeRepoName :: Text -> Text
  223 sanitizeRepoName url =
  224   -- Strip trailing slashes first so a trailing slash doesn't yield an
  225   -- empty last segment. Then take the last '/' segment; fall back to the
  226   -- ':' split for SCP-style (git@host:path); then the whole url.
  227   let trimmed = T.dropWhileEnd (== '/') url
  228       lastSeg = case T.breakOnEnd "/" trimmed of
  229         (_, seg) | not (T.null seg) -> seg
  230         _ -> case T.breakOnEnd ":" trimmed of
  231                (_, seg) | not (T.null seg) -> seg
  232                _ -> trimmed
  233       withoutGit = fromMaybe lastSeg (T.stripSuffix ".git" lastSeg)
  234       sanitized = T.map (\c -> if isAlphaNum c || c == '_' || c == '-' then c else '-') withoutGit
  235       trimmedName = T.dropWhile (== '-') (T.dropWhileEnd (== '-') sanitized)
  236   in if T.null trimmedName then "repo" else trimmedName

  238 -- | The outcome of a clone attempt. Shared by the opcode (model-invoked)
  239 -- and the @POST /api/sessions/:id/setup-repo@ endpoint (web combo box).
  240 data CloneResult
  241   = CloneCloned Text    -- ^ cloned into <repo-name> (shallow)
  242   | CloneNoop Text      -- ^ repo already exists at <repo-name> (branch matches)
  243   | CloneConflict Text Text  -- ^ a different repo occupies <repo-name> (existing url)
  244   | CloneSwitched Text Text  -- ^ switched an existing clone to <branch> (<repo-name>)
  245   | CloneFailed Text    -- ^ clone, switch, or idempotency check failed (<error>)
  246   deriving stock (Eq, Show)

  248 -- | Which executor arm a branch SWITCH uses, chosen from the resolved
  249 -- credential kind (mirrors 'cloneWithCredential''s dispatch).
  250 --
  251 -- A PAT / machine-user credential is delivered as
  252 -- @GIT_CONFIG_PARAMETERS='http.extraHeader=Authorization: Basic …'@ inside
  253 -- 'ceEnvExtras'. @GIT_CONFIG_PARAMETERS@ is a member of
  254 -- 'Seal.Tools.Exec.UntrustedIO.secretEnvKeys', so the remote arm of
  255 -- 'uioShellExecEnv' pipes it over SSH **stdin** instead of rendering it into
  256 -- the remote command string. @uioShellExecGitEnv@'s remote arm instead
  257 -- renders the extras into the remote command prefix — using it for a PAT
  258 -- switch would put the @Authorization@ header on the remote argv. Hence:
  259 -- secret-bearing credential → 'SwitchViaShellEnv'; deploy key (agent
  260 -- forwarding, no secret in the env) → 'SwitchViaGitEnv'.
  261 data SwitchTransport
  262   = SwitchViaShellEnv  -- ^ PAT / machine user (@ceRawToken@ is 'Just')
  263   | SwitchViaGitEnv    -- ^ deploy key (@ceRawToken@ is 'Nothing')
  264   deriving stock (Eq, Show)
  265 
  266 -- | Pick the switch executor arm from the resolved credential: a raw token
  267 -- means a PAT/machine-user credential (see 'SwitchTransport'); no token
  268 -- means a deploy key.
  269 switchTransport :: Maybe ByteString -> SwitchTransport
  270 switchTransport (Just _rawToken) = SwitchViaShellEnv
  271 switchTransport Nothing          = SwitchViaGitEnv

  273 -- | The @--branch=\<ref\>@ argv token for a requested branch — a SINGLE
  274 -- token (@--branch='x'@, never @--branch x@), so the value can never be
  275 -- read as a separate option. Empty when no branch was requested, so the
  276 -- emitted clone command is byte-identical to the pre-branch one.
  277 branchFlag :: Maybe BranchName -> Text
  278 branchFlag Nothing  = ""
  279 branchFlag (Just b) = " --branch=" <> shellQ (branchNameText b)

  281 -- | Parse the idempotency-probe output: the existing clone's origin URL,
  282 -- then (second line) the branch @git@ currently has checked out
  283 -- (@git rev-parse --abbrev-ref HEAD@ — the literal @HEAD@ for a detached
  284 -- checkout). A one-line response is tolerated: the current branch is then
  285 -- unknown ('Nothing'), which is what the remote test fixture's
  286 -- @\"__NONE__\\n\"@ reply produces. A @\"__NONE__\"@ URL means "no clone
  287 -- here".
  288 parseRepoProbe :: Text -> (Text, Maybe Text)
  289 parseRepoProbe out = case T.lines out of
  290   []             -> ("", Nothing)
  291   (urlLine : ls) -> (T.strip urlLine, firstNonBlank ls)
  292 
  293 -- | The first non-blank line of a (possibly empty) list of lines.
  294 firstNonBlank :: [Text] -> Maybe Text
  295 firstNonBlank [] = Nothing
  296 firstNonBlank (l : ls)
  297   | T.null (T.strip l) = firstNonBlank ls
  298   | otherwise          = Just (T.strip l)
  299 
  300 -- | The first line of a command's output, trimmed (reads
  301 -- @rev-parse --abbrev-ref HEAD@).
  302 firstLine :: Text -> Text
  303 firstLine t = case T.lines t of
  304   (l : _) -> T.strip l
  305   []      -> ""

  307 -- | Run the clone (or no-op) via an 'UntrustedIO' capability and return a
  308 -- structured 'CloneResult'. This is the shared seam between the
  309 -- model-invoked 'SETUP_REPO' opcode and the web combo box's
  310 -- @POST /api/sessions/:id/setup-repo@ endpoint — both build the same
  311 -- 'UntrustedIO' (sandboxed to the session workdir) and call this.
  312 --
  313 -- Credential resolution (W3): @lookupRepoByUrl@ against the registry
  314 -- (@cdRepoReg@) → if the URL matches a registered repo, resolve the
  315 -- credential via the no-disk seam ('resolveCloneTarget' using @cdVault@/
  316 -- @cdSshAgent@/@cdPinnedKnownHosts@/@cdKeyfilesDir@) and run @git clone@
  317 -- via 'uioShellExecEnv' (deploy key: @SSH_AUTH_SOCK@ + @GIT_SSH_COMMAND@
  318 -- env) or 'uioBinExecEnv' (PAT: @http.extraHeader@ argv) with the resolved
  319 -- env. If the URL is NOT registered, fall through to a bare-URL clone
  320 -- (public repos, backward-compat).
  321 --
  322 -- With @mBranch@: a fresh clone checks that branch out ('branchFlag'); an
  323 -- existing clone of the same repo is switched when the branch differs
  324 -- ('switchBranch'), and a no-op when it already matches.
  325 cloneRepoIO :: CloneDeps -> UntrustedIO -> Text -> Maybe BranchName -> IO CloneResult
  326 cloneRepoIO deps uio url mBranch = do
  327   let repoName = sanitizeRepoName url
  328       cleanUrl = T.strip url
  329       mCwdPath = case mkRemotePath "." of
  330         Right rp -> Just rp
  331         Left _   -> Nothing
  332       -- Idempotency probe: the existing origin URL, then the branch git
  333       -- currently has checked out (a second line — see 'parseRepoProbe').
  334       checkCmd = "if [ -d " <> shellQ repoName <> "/.git ]; then git -C "
  335                  <> shellQ repoName <> " config --get remote.origin.url; git -C "
  336                  <> shellQ repoName <> " rev-parse --abbrev-ref HEAD; else rm -rf "
  337                  <> shellQ repoName <> "; echo __NONE__; fi"
  338       -- No clone yet: look the URL up in the repo registry. If registered,
  339       -- resolve the credential via the no-disk seam; else fall through to a
  340       -- bare-URL clone (public repos).
  341       cloneFresh = do
  342         eRepos <- rrhList (cdRepoReg deps)
  343         let mRepo = case eRepos of
  344               Right repos -> lookupRepoByUrl cleanUrl
  345                                (RepoRegistry (Map.fromList [(srId r, r) | r <- repos]))
  346               Left _      -> Nothing
  347         case mRepo of
  348           Just repo -> cloneWithCredential deps uio repo repoName mBranch mCwdPath
  349           Nothing   -> cloneBareUrl uio cleanUrl repoName mBranch mCwdPath
  350   checkRes <- UIORec.uioShellExec uio (shellCmd checkCmd) mCwdPath
  351   case checkRes of
  352     Left err -> pure (CloneFailed ("idempotency check failed: " <> renderUntrustedErr err))
  353     Right out -> case parseRepoProbe out of
  354       (existingUrl, mCurrent)
  355         | existingUrl == "__NONE__" -> cloneFresh
  356         | normalizeRepoUrl existingUrl /= normalizeRepoUrl cleanUrl ->
  357             pure (CloneConflict repoName existingUrl)
  358         | otherwise -> case mBranch of
  359             -- No branch requested: same repo ⇒ no-op (unchanged behavior).
  360             Nothing -> pure (CloneNoop repoName)
  361             -- Requested branch already checked out ⇒ no-op. A differing,
  362             -- unknown, or detached ("HEAD") current branch ⇒ switch.
  363             Just b
  364               | mCurrent == Just (branchNameText b) -> pure (CloneNoop repoName)
  365               | otherwise -> switchBranch deps uio cleanUrl repoName b mCwdPath
  366 
  367 -- | Clone a registered repo via the no-disk seam. Resolves the credential
  368 -- via 'resolveCloneTarget', then branches on the credential kind:
  369 --
  370 --   * **PAT/MachineUser** ('ceRawToken' is 'Just'): clones via
  371 --     @gh repo clone <url> <dest> -- --depth 1@ with @GH_TOKEN@ injected
  372 --     into the process env. This mirrors the BIN_EXEC @gh@ path
  373 --     ('runGhWithCredentials' in "Seal.ISA.Ops.Bin") — the raw token
  374 --     bytes (from 'ceRawToken') go to @GH_TOKEN@, never to argv or disk.
  375 --     @gh@ authenticates via the token, not via @http.extraHeader@.
  376 --
  377 --   * **Deploy key** ('ceRawToken' is 'Nothing'): clones via
  378 --     @git clone@ with the agent-forwarding env (@SSH_AUTH_SOCK@ /
  379 --     @GIT_SSH_COMMAND@) — @gh@ cannot use an SSH agent.
  380 cloneWithCredential
  381   :: CloneDeps -> UntrustedIO -> SourceRepo -> Text -> Maybe BranchName
  382   -> Maybe RemotePath -> IO CloneResult
  383 cloneWithCredential deps uio repo repoName mBranch mCwdPath = do
  384   eTarget <- resolveCloneTarget deps repo
  385   case eTarget of
  386     Left err -> pure (CloneFailed ("credential resolution failed: " <> renderCloneError err))
  387     Right target -> withCloneTarget target $ \env -> do
  388       case ceRawToken env of
  389         Just tokenBytes -> do
  390           let token = map (toEnum . fromIntegral) (BS.unpack tokenBytes)
  391               envExtras = ("GH_TOKEN", token) : ceEnvExtras env
  392               ghCmd = "gh repo clone " <> shellQ (ceUrl env) <> " "
  393                       <> shellQ repoName <> " -- --depth 1" <> branchFlag mBranch
  394           cloneRes <- UIORec.uioShellExecEnv uio envExtras (shellCmd ghCmd) mCwdPath
  395           case cloneRes of
  396             Left err -> pure (CloneFailed ("clone failed (transport): " <> renderUntrustedErr err))
  397             Right _out -> verifyClone uio repoName _out mCwdPath
  398         Nothing -> do
  399           let gitConfigArgs = map T.unpack (ceGitConfigArgs env)
  400               cloneCmd = "git " <> T.unwords (map T.pack gitConfigArgs)
  401                          <> " clone --depth 1" <> branchFlag mBranch
  402                          <> " -- " <> shellQ (ceUrl env) <> " " <> shellQ repoName
  403           cloneRes <- UIORec.uioShellExecGitEnv uio (ceEnvExtras env) (ceKnownHostsContent env) (shellCmd cloneCmd) mCwdPath
  404           case cloneRes of
  405             Left err -> pure (CloneFailed ("clone failed (transport): " <> renderUntrustedErr err))
  406             Right _out -> verifyClone uio repoName _out mCwdPath
  407 
  408 -- | Clone a bare URL (no credential — public repo, backward-compat).
  409 cloneBareUrl
  410   :: UntrustedIO -> Text -> Text -> Maybe BranchName -> Maybe RemotePath
  411   -> IO CloneResult
  412 cloneBareUrl uio cleanUrl repoName mBranch mCwdPath = do
  413   let cloneCmd = "git clone --depth 1" <> branchFlag mBranch
  414                  <> " -- " <> shellQ cleanUrl <> " " <> shellQ repoName
  415   cloneRes <- UIORec.uioShellExec uio (shellCmd cloneCmd) mCwdPath
  416   case cloneRes of
  417     Left err -> pure (CloneFailed ("clone failed (transport): " <> renderUntrustedErr err))
  418     Right _out -> verifyClone uio repoName _out mCwdPath

  420 -- | Verify a clone actually landed by checking for @<repoName>/.git@.
  421 -- 'uioShellExec' returns 'Right' even on non-zero exit (it surfaces
  422 -- stderr to the model for SHELL_EXEC), so the exit code is unreliable
  423 -- for distinguishing success from failure. The filesystem is the source
  424 -- of truth. Returns 'CloneCloned' on success, 'CloneFailed' (with the
  425 -- clone's output text so the user sees /why/) on failure.
  426 verifyClone :: UntrustedIO -> Text -> Text -> Maybe RemotePath -> IO CloneResult
  427 verifyClone uio repoName cloneOut mCwdPath = do
  428   let verifyCmd = "test -d " <> shellQ repoName <> "/.git && echo __OK__ || echo __MISSING__"
  429   vRes <- UIORec.uioShellExec uio (shellCmd verifyCmd) mCwdPath
  430   case vRes of
  431     Left err -> pure (CloneFailed ("clone verify failed: " <> renderUntrustedErr err))
  432     Right vOut
  433       | "__OK__" `T.isInfixOf` vOut -> do
  434           _ <- tryCodegraphInit uio repoName mCwdPath
  435           pure (CloneCloned repoName)
  436       | otherwise -> pure (CloneFailed
  437                             ("clone did not land — git output:\n"
  438                              <> T.strip cloneOut))

  440 -- | Switch an EXISTING clone to @branch@ (the idempotent second call with a
  441 -- different branch) without re-cloning. The credential is resolved exactly
  442 -- like 'cloneRepoIO' does (registry lookup → 'resolveCloneTarget' →
  443 -- 'withCloneTarget'), then ONE shell command runs: check the branch out if
  444 -- it already exists locally, else fetch its ref at depth 1 and check it
  445 -- out. The fetch refspec is fully qualified (@refs\/heads\/\<b\>@), so a tag
  446 -- name fails loudly instead of DWIM-ing to the tag. The result is verified
  447 -- with @rev-parse --abbrev-ref HEAD@; on a mismatch 'CloneFailed' carries
  448 -- git's output and the directory (plus any work in it) is left intact —
  449 -- never deleted. On success 'tryCodegraphInit' re-runs (the graph would
  450 -- otherwise describe the old branch) and the result is 'CloneSwitched'.
  451 switchBranch
  452   :: CloneDeps -> UntrustedIO -> Text -> Text -> BranchName -> Maybe RemotePath
  453   -> IO CloneResult
  454 switchBranch deps uio cleanUrl repoName branch mCwdPath = do
  455   eRepos <- rrhList (cdRepoReg deps)
  456   let mRepo = case eRepos of
  457         Right repos -> lookupRepoByUrl cleanUrl
  458                          (RepoRegistry (Map.fromList [(srId r, r) | r <- repos]))
  459         Left _      -> Nothing
  460   case mRepo of
  461     -- Unregistered (public) repo: no credential to resolve.
  462     Nothing -> runSwitch (UIORec.uioShellExec uio) (shellCmd (switchCmd repoName b))
  463     Just repo -> do
  464       eTarget <- resolveCloneTarget deps repo
  465       case eTarget of
  466         Left err -> pure (CloneFailed ("credential resolution failed: " <> renderCloneError err))
  467         Right target -> withCloneTarget target $ \env ->
  468           runSwitch (switchExec env) (shellCmd (switchCmd repoName b))
  469   where
  470     b = branchNameText branch
  471     -- Dispatch on the credential kind (see 'SwitchTransport'): a PAT must
  472     -- ride 'uioShellExecEnv' (secret extras over stdin on the remote arm).
  473     switchExec env = case switchTransport (ceRawToken env) of
  474       SwitchViaShellEnv -> UIORec.uioShellExecEnv uio (ceEnvExtras env)
  475       SwitchViaGitEnv ->
  476         UIORec.uioShellExecGitEnv uio (ceEnvExtras env) (ceKnownHostsContent env)
  477     runSwitch exec cmd = do
  478       res <- exec cmd mCwdPath
  479       case res of
  480         Left err -> pure (CloneFailed ("switch failed (transport): " <> renderUntrustedErr err))
  481         Right out -> verifySwitch exec out
  482     verifySwitch exec switchOut = do
  483       let verifyCmd = "git -C " <> shellQ repoName <> " rev-parse --abbrev-ref HEAD"
  484       vRes <- exec (shellCmd verifyCmd) mCwdPath
  485       case vRes of
  486         Left err -> pure (CloneFailed ("switch verify failed: " <> renderUntrustedErr err))
  487         Right vOut
  488           | firstLine vOut == b -> do
  489               _ <- tryCodegraphInit uio repoName mCwdPath
  490               pure (CloneSwitched repoName b)
  491           | otherwise -> pure (CloneFailed
  492               ("switch to " <> b <> " did not take — git output:\n"
  493                <> T.strip switchOut <> "\n" <> T.strip vOut))

  495 -- | The one shell command that moves an existing clone onto @branch@:
  496 -- check it out directly when the local branch ref exists, otherwise fetch
  497 -- that branch's ref (at depth 1) into a local branch ref and check it out.
  498 -- Every interpolated value is shell-quoted.
  499 switchCmd :: Text -> Text -> Text
  500 switchCmd repoName b =
  501   "if git -C " <> qr <> " rev-parse --verify --quiet " <> shellQ ("refs/heads/" <> b)
  502     <> " >/dev/null\n"
  503   <> "then git -C " <> qr <> " checkout " <> shellQ b <> "\n"
  504   <> "else git -C " <> qr <> " fetch --depth 1 origin "
  505     <> shellQ ("refs/heads/" <> b <> ":refs/heads/" <> b)
  506     <> " && git -C " <> qr <> " checkout " <> shellQ b <> "\n"
  507   <> "fi"
  508   where
  509     qr = shellQ repoName

  511 -- | Render a 'CloneResult' as the user-facing text. @url@ is the requested
  512 -- URL and @mBranch@ the requested branch (if any) — 'CloneCloned' and
  513 -- 'CloneNoop' carry only the repo name, so the branch is passed in.
  514 renderCloneResult :: Text -> Maybe Text -> CloneResult -> Text
  515 renderCloneResult url mBranch res = case res of
  516   CloneCloned name ->
  517     "Cloned " <> T.strip url <> " into " <> name <> " (shallow"
  518       <> branchNote mBranch <> ")."
  519   CloneNoop name ->
  520     "Repo already exists — no-op (" <> name <> branchNote mBranch <> ")."
  521   CloneConflict name existing ->
  522     "A different repo already occupies " <> name <> " (existing: " <> existing <> ")."
  523   CloneSwitched name b ->
  524     "Repo already exists — switched to branch " <> b <> " (" <> name <> ")."
  525   CloneFailed err ->
  526     "SETUP_REPO failed — could not clone " <> T.strip url <> ":\n" <> err
  527   where
  528     branchNote = maybe "" (", branch " <>)
  529 
  530 -- | Run the clone (or no-op / switch) and build the 'OpResult' (opcode
  531 -- path; wraps 'cloneRepoIO' with the audit 'orRecorded' payload). A
  532 -- 'CloneSwitched' is a NON-error result (the clone is usable on the
  533 -- requested branch — the turn engine's post-repo hooks must still run).
  534 runSetupRepo :: CloneDeps -> Text -> Maybe BranchName -> Value -> UIO OpResult
  535 runSetupRepo deps url mBranch recorded = do
  536   uio <- uioUntrustedIO
  537   res <- uioLiftIO (cloneRepoIO deps uio url mBranch)
  538   let (target, status, isErr) = case res of
  539         CloneCloned name      -> (name, "cloned", False)
  540         CloneNoop name        -> (name, "noop", False)
  541         CloneConflict name _  -> (name, "conflict", True)
  542         CloneSwitched name _  -> (name, "switched", False)
  543         CloneFailed _         -> ("", "failed", True)
  544   pure (OpResult [TrpText (renderCloneResult url (branchNameText <$> mBranch) res)]
  545                  isErr (recordWith recorded target status))

  547 -- | Construct a 'ShellCommand', total on our internally-built command
  548 -- strings (they contain no NULs). 'mkShellCommand' only fails on NUL, so
  549 -- this never triggers the error branch in practice.
  550 shellCmd :: Text -> ShellCommand
  551 shellCmd t = case mkShellCommand t of
  552   Right c  -> c
  553   Left _e  -> error "SETUP_REPO: internal: shell command rejected (NUL?) — unreachable"
  554 
  555 -- | Re-build the recorded object with the target repo name + a status.
  556 recordWith :: Value -> Text -> Text -> Value
  557 recordWith _ target status =
  558   object [ "target" .= target, "status" .= status ]

  560 -- | Single-quote a shell token (the safe way to embed a literal in a
  561 -- @\/bin\/sh -c@ single-arg command). Any embedded @'@ becomes @'\''@.
  562 -- The URL has already been validated to contain no shell metacharacters,
  563 -- but quoting is defense-in-depth.
  564 shellQ :: Text -> Text
  565 shellQ t = "'" <> T.concatMap (\c -> if c == '\'' then "'\\''" else T.singleton c) t <> "'"

  567 -- | Best-effort: run @codegraph init@ in the freshly cloned repo directory
  568 -- so the agent has a pre-built knowledge graph on the next turn. This is
  569 -- a soft-failure operation — if @codegraph@ is not installed (exit 127) or
  570 -- fails for any reason (unsupported language, large repo, etc.), the
  571 -- clone result is unaffected. The @codegraph@ binary is looked up via
  572 -- @command -v@ first so we don't spawn a process we know will fail; if
  573 -- not found, we skip silently.
  574 --
  575 -- Security: this runs on the untrusted plane via 'uioShellExec', which is
  576 -- correct — @codegraph init@ parses untrusted repo content (tree-sitter on
  577 -- upstream code). No new attack surface is introduced.
  578 tryCodegraphInit :: UntrustedIO -> Text -> Maybe RemotePath -> IO ()
  579 tryCodegraphInit uio repoName mCwdPath = do
  580   -- Check if codegraph is installed (command -v returns 0 if found).
  581   let checkCmd = "command -v codegraph >/dev/null 2>&1"
  582   checkRes <- UIORec.uioShellExec uio (shellCmd checkCmd) mCwdPath
  583   case checkRes of
  584     Left _ -> pure ()   -- uioShellExec failed entirely — skip
  585     Right _ -> do
  586       -- codegraph exists; run init with the repo directory as cwd.
  587       -- @codegraph init@ (no path arg) initializes the project in the
  588       -- current directory. The @|| true@ ensures best-effort: any failure
  589       -- (unsupported language, etc.) is swallowed.
  590       let repoPath = case mkRemotePath repoName of
  591             Right rp -> Just rp
  592             Left _   -> Nothing
  593       let initCmd = "codegraph init 2>&1 || true"
  594       _ <- UIORec.uioShellExec uio (shellCmd initCmd) repoPath
  595       pure ()

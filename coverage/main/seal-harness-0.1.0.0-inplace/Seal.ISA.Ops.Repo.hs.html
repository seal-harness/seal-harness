    2 -- | SETUP_REPO (Untrusted): clone a remote repository into the session's
    3 -- workspace so the model can work with its code and discover any skills it
    4 -- ships. The opcode is VCS-agnostic in name and result shape (\"cloned at
    5 -- \<path\>\") but git-only in implementation for now; a future @vcs@ arg or
    6 -- URL-scheme dispatch can add others without changing the contract.
    7 --
    8 -- Untrusted because upstream repo content is untrusted — a clone is a
    9 -- filesystem mutation of untrusted-origin data, so it goes through the
   10 -- sandboxed 'UntrustedIO' + the 'Autonomy' confirmation gate, not a
   11 -- trusted shell. This is strictly safer than the prior pattern of the
   12 -- model running @git clone@ via 'SHELL_EXEC' and hoping.
   13 --
   14 -- Semantics:
   15 --
   16 --   * Shallow clone (@git clone --depth 1@) to keep large repos cheap.
   17 --   * Multiple repos are supported: each clones into
   18 --     @\<workdir\>\/\<sanitized-repo-name\>\/@.
   19 --   * Idempotent: if the target dir already holds a clone of the same
   20 --     remote URL, the call is a no-op (\"Repo already exists — no-op\").
   21 --     If the dir exists but its remote differs, the call errors (\"a
   22 --     different repo already occupies that path\") rather than silently
   23 --     clobbering.
   24 --   * The clone runs in the workspace root (cwd = workdir) via
   25 --     'uioShellExec'; the URL is validated to a safe charset so it cannot
   26 --     inject shell metacharacters into the @\/bin\/sh -c@ single-arg
   27 --     boundary.
   28 --
   29 -- Credential resolution (W3 — design §4.2/§4.3, rev 3): 'setupRepoOp' takes
   30 -- a 'CloneDeps' closed-over param (mirrors @secretGetOp (cdVault deps)@).
   31 -- 'cloneRepoIO' does @lookupRepoByUrl@ against the registry
   32 -- (@cdRepoReg@) → if the URL matches a registered repo, resolve the
   33 -- credential via the no-disk seam ('Seal.SourceControl.Clone.resolveCloneTarget'
   34 -- using @cdVault@/@cdSshAgent@/@cdPinnedKnownHosts@/@cdKeyfilesDir@) and
   35 -- clones via @gh repo clone@ with @GH_TOKEN@ (PAT/MachineUser: the raw
   36 -- token from 'ceRawToken' injected as @GH_TOKEN@ env) or @git clone@ via
   37 -- 'uioShellExecGitEnv' (deploy key: @SSH_AUTH_SOCK@ + @GIT_SSH_COMMAND@).
   38 -- If the URL is NOT registered, fall through to a bare-URL clone (public
   39 -- repos, backward-compat). Stays Untrusted; credential resolution via
   40 -- 'liftIO' in the trusted plane. NO @runLocal@/@BackendExec@.

   79 -- | SETUP_REPO opcode. Input: @{url: Text}@. Authorize: autonomy must not
   80 -- be 'Deny' and the URL must validate ('validateRepoUrl'). Run: a shallow
   81 -- @git clone@ into @\<workdir\>\/\<sanitized-repo-name\>@ via
   82 -- 'uioShellExec', after an idempotency check (read the existing remote).
   83 -- 'orRecorded': the url + target path + status (secret-free).
   84 setupRepoOp :: CloneDeps -> WorkspaceRoot -> AutonomyLevel -> Opcode
   85 setupRepoOp deps _wsRoot autonomy = UntrustedOpcode
   86   { uoName = OpName "SETUP_REPO"
   87   , uoDesc = "Clone a remote repository into the session workspace (shallow, idempotent). Discover any skills it ships via the available-skills catalog on the next turn."
   88   , uoInSchema = setupRepoSchema
   89   , uoOutSchema = object []
   90   , uoAuthorize = \v ->
   91       case urlField v of
   92         Nothing -> Left "SETUP_REPO requires {url:string}"
   93         Just url -> case validateRepoUrl url of
   94           Left err -> Left ("SETUP_REPO: invalid url: " <> err)
   95           Right _ -> case autonomy of
   96             Deny -> Left "SETUP_REPO denied by autonomy policy"
   97             _   -> Right ()
   98   , uoRun = \v -> do
   99       let mUrl = urlField v
  100           recorded = object [ "url" .= (mUrl :: Maybe Text) ]
  101       case mUrl of
  102         Nothing -> pure (OpResult [TrpText "SETUP_REPO requires {url:string}"] True recorded)
  103         Just url -> case validateRepoUrl url of
  104           Left err -> pure (OpResult [TrpText ("SETUP_REPO: invalid url: " <> err)] True recorded)
  105           Right url' -> runSetupRepo deps url' recorded
  106   }

  124 -- | Validate a repo URL. Accepts @https://@, @http://@, @git://@,
  125 -- @ssh://@, and the @git\@host:path@ SCP-style form. Rejects any URL
  126 -- containing shell metacharacters that could break out of the
  127 -- @\/bin\/sh -c@ single-arg boundary (the @uioShellExec@ contract).
  128 -- Returns the trimmed URL on success, or an error message.
  129 validateRepoUrl :: Text -> Either Text Text
  130 validateRepoUrl raw =
  131   let url = T.strip raw
  132   in if T.null url
  133        then Left "url is empty"
  134        else if not (hasAcceptableScheme url)
  135               then Left "url must use https://, http://, git://, ssh://, or git@host:path"
  136               else if T.any isShellMetachar url
  137                      then Left "url contains forbidden shell metacharacters"
  138                      else Right url
  139 
  140 -- | True if the URL begins with an acceptable scheme or is SCP-style
  141 -- (@git\@host:path@, more generally @user\@host:path@).
  142 hasAcceptableScheme :: Text -> Bool
  143 hasAcceptableScheme url =
  144   any (`T.isPrefixOf` url)
  145       [ "https://", "http://", "git://", "ssh://" ]
  146   || isScpStyle url
  147 
  148 -- | Heuristic for @user\@host:path@ (SCP-style). Has an @\@ before the
  149 -- first @/@ (if any), and a @:@ after the host, and no scheme prefix.
  150 isScpStyle :: Text -> Bool
  151 isScpStyle url =
  152   case T.breakOn "@" url of
  153     (user, rest) | not (T.null user), not (T.null rest) ->
  154       let afterAt = T.drop 1 rest
  155       in T.any (== ':') afterAt
  156              && not (":" `T.isPrefixOf` afterAt)   -- not ':host:path' (no user)
  157              && not ("/" `T.isPrefixOf` user)      -- not a path with an '@'
  158     _ -> False
  159 
  160 -- | Shell metacharacters forbidden in a repo URL (the @\/bin\/sh -c@
  161 -- single-arg boundary treats these specially). A legit git URL never
  162 -- contains them; their presence signals an injection attempt.
  163 isShellMetachar :: Char -> Bool
  164 isShellMetachar c = c `elem` (";`$\\\"'|\n\r&<>(){}*?[]#!~" :: String)

  166 -- | 'normalizeRepoUrl' is the shared normalizer in "Seal.SourceControl.Repo"
  167 -- (W1 moved it there so both 'ISA.Ops.Repo' and 'lookupRepoByUrl' use the SAME
  168 -- normalizer). Re-exported here for backward compatibility with existing
  169 -- callers (e.g. 'runSetupRepo's idempotency check below).
  170 
  171 sanitizeRepoName :: Text -> Text
  172 sanitizeRepoName url =
  173   -- Strip trailing slashes first so a trailing slash doesn't yield an
  174   -- empty last segment. Then take the last '/' segment; fall back to the
  175   -- ':' split for SCP-style (git@host:path); then the whole url.
  176   let trimmed = T.dropWhileEnd (== '/') url
  177       lastSeg = case T.breakOnEnd "/" trimmed of
  178         (_, seg) | not (T.null seg) -> seg
  179         _ -> case T.breakOnEnd ":" trimmed of
  180                (_, seg) | not (T.null seg) -> seg
  181                _ -> trimmed
  182       withoutGit = fromMaybe lastSeg (T.stripSuffix ".git" lastSeg)
  183       sanitized = T.map (\c -> if isAlphaNum c || c == '_' || c == '-' then c else '-') withoutGit
  184       trimmedName = T.dropWhile (== '-') (T.dropWhileEnd (== '-') sanitized)
  185   in if T.null trimmedName then "repo" else trimmedName

  187 -- | The outcome of a clone attempt. Shared by the opcode (model-invoked)
  188 -- and the @POST /api/sessions/:id/setup-repo@ endpoint (web combo box).
  189 data CloneResult
  190   = CloneCloned Text    -- ^ cloned into <repo-name> (shallow)
  191   | CloneNoop Text      -- ^ repo already exists at <repo-name>
  192   | CloneConflict Text Text  -- ^ a different repo occupies <repo-name> (existing url)
  193   | CloneFailed Text    -- ^ clone or idempotency check failed (<error>)
  194   deriving stock (Eq, Show)
  195 
  196 -- | Run the clone (or no-op) via an 'UntrustedIO' capability and return a
  197 -- structured 'CloneResult'. This is the shared seam between the
  198 -- model-invoked 'SETUP_REPO' opcode and the web combo box's
  199 -- @POST /api/sessions/:id/setup-repo@ endpoint — both build the same
  200 -- 'UntrustedIO' (sandboxed to the session workdir) and call this.
  201 --
  202 -- Credential resolution (W3): @lookupRepoByUrl@ against the registry
  203 -- (@cdRepoReg@) → if the URL matches a registered repo, resolve the
  204 -- credential via the no-disk seam ('resolveCloneTarget' using @cdVault@/
  205 -- @cdSshAgent@/@cdPinnedKnownHosts@/@cdKeyfilesDir@) and run @git clone@
  206 -- via 'uioShellExecEnv' (deploy key: @SSH_AUTH_SOCK@ + @GIT_SSH_COMMAND@
  207 -- env) or 'uioBinExecEnv' (PAT: @http.extraHeader@ argv) with the resolved
  208 -- env. If the URL is NOT registered, fall through to a bare-URL clone
  209 -- (public repos, backward-compat).
  210 cloneRepoIO :: CloneDeps -> UntrustedIO -> Text -> IO CloneResult
  211 cloneRepoIO deps uio url = do
  212   let repoName = sanitizeRepoName url
  213       mCwdPath = case mkRemotePath "." of
  214         Right rp -> Just rp
  215         Left _   -> Nothing
  216       checkCmd = "if [ -d " <> shellQ repoName <> "/.git ]; then git -C "
  217                  <> shellQ repoName <> " config --get remote.origin.url; else rm -rf "
  218                  <> shellQ repoName <> "; echo __NONE__; fi"
  219   checkRes <- UIORec.uioShellExec uio (shellCmd checkCmd) mCwdPath
  220   case checkRes of
  221     Left err -> pure (CloneFailed ("idempotency check failed: " <> renderUntrustedErr err))
  222     Right existing -> do
  223       let existingUrl = T.strip (T.filter (/= '\n') existing)
  224           cleanUrl = T.strip url
  225       if existingUrl /= "__NONE__"
  226         then if normalizeRepoUrl existingUrl == normalizeRepoUrl cleanUrl
  227                then pure (CloneNoop repoName)
  228                else pure (CloneConflict repoName existingUrl)
  229         else do
  230           -- Look up the URL in the repo registry. If registered, resolve
  231           -- the credential via the no-disk seam; else fall through to a
  232           -- bare-URL clone (public repos).
  233           eRepos <- rrhList (cdRepoReg deps)
  234           let mRepo = case eRepos of
  235                 Right repos -> lookupRepoByUrl cleanUrl (RepoRegistry (Map.fromList [(srId r, r) | r <- repos]))
  236                 Left _       -> Nothing
  237           case mRepo of
  238             Just repo -> cloneWithCredential deps uio repo repoName mCwdPath
  239             Nothing   -> cloneBareUrl uio cleanUrl repoName mCwdPath

  241 -- | Clone a registered repo via the no-disk seam. Resolves the credential
  242 -- via 'resolveCloneTarget', then branches on the credential kind:
  243 --
  244 --   * **PAT/MachineUser** ('ceRawToken' is 'Just'): clones via
  245 --     @gh repo clone <url> <dest> -- --depth 1@ with @GH_TOKEN@ injected
  246 --     into the process env. This mirrors the BIN_EXEC @gh@ path
  247 --     ('runGhWithCredentials' in "Seal.ISA.Ops.Bin") — the raw token
  248 --     bytes (from 'ceRawToken') go to @GH_TOKEN@, never to argv or disk.
  249 --     @gh@ authenticates via the token, not via @http.extraHeader@.
  250 --
  251 --   * **Deploy key** ('ceRawToken' is 'Nothing'): clones via
  252 --     @git clone@ with the agent-forwarding env (@SSH_AUTH_SOCK@ /
  253 --     @GIT_SSH_COMMAND@) — @gh@ cannot use an SSH agent.
  254 cloneWithCredential
  255   :: CloneDeps -> UntrustedIO -> SourceRepo -> Text -> Maybe RemotePath
  256   -> IO CloneResult
  257 cloneWithCredential deps uio repo repoName mCwdPath = do
  258   eTarget <- resolveCloneTarget deps repo
  259   case eTarget of
  260     Left err -> pure (CloneFailed ("credential resolution failed: " <> renderCloneError err))
  261     Right target -> withCloneTarget target $ \env -> do
  262       case ceRawToken env of
  263         Just tokenBytes -> do
  264           let token = map (toEnum . fromIntegral) (BS.unpack tokenBytes)
  265               envExtras = ("GH_TOKEN", token) : ceEnvExtras env
  266               ghCmd = "gh repo clone " <> shellQ (ceUrl env) <> " "
  267                       <> shellQ repoName <> " -- --depth 1"
  268           cloneRes <- UIORec.uioShellExecEnv uio envExtras (shellCmd ghCmd) mCwdPath
  269           case cloneRes of
  270             Left err -> pure (CloneFailed ("clone failed: " <> renderUntrustedErr err))
  271             Right _out -> verifyClone uio repoName _out mCwdPath
  272         Nothing -> do
  273           let gitConfigArgs = map T.unpack (ceGitConfigArgs env)
  274               cloneCmd = "git " <> T.unwords (map T.pack gitConfigArgs)
  275                          <> " clone --depth 1 -- " <> shellQ (ceUrl env) <> " " <> shellQ repoName
  276           cloneRes <- UIORec.uioShellExecGitEnv uio (ceEnvExtras env) (ceKnownHostsContent env) (shellCmd cloneCmd) mCwdPath
  277           case cloneRes of
  278             Left err -> pure (CloneFailed ("clone failed: " <> renderUntrustedErr err))
  279             Right _out -> verifyClone uio repoName _out mCwdPath

  281 -- | Clone a bare URL (no credential — public repo, backward-compat).
  282 cloneBareUrl :: UntrustedIO -> Text -> Text -> Maybe RemotePath -> IO CloneResult
  283 cloneBareUrl uio cleanUrl repoName mCwdPath = do
  284   let cloneCmd = "git clone --depth 1 -- " <> shellQ cleanUrl <> " " <> shellQ repoName
  285   cloneRes <- UIORec.uioShellExec uio (shellCmd cloneCmd) mCwdPath
  286   case cloneRes of
  287     Left err -> pure (CloneFailed ("clone failed: " <> renderUntrustedErr err))
  288     Right _out -> verifyClone uio repoName _out mCwdPath

  290 -- | Verify a clone actually landed by checking for @<repoName>/.git@.
  291 -- 'uioShellExec' returns 'Right' even on non-zero exit (it surfaces
  292 -- stderr to the model for SHELL_EXEC), so the exit code is unreliable
  293 -- for distinguishing success from failure. The filesystem is the source
  294 -- of truth. Returns 'CloneCloned' on success, 'CloneFailed' (with the
  295 -- clone's output text so the user sees /why/) on failure.
  296 verifyClone :: UntrustedIO -> Text -> Text -> Maybe RemotePath -> IO CloneResult
  297 verifyClone uio repoName cloneOut mCwdPath = do
  298   let verifyCmd = "test -d " <> shellQ repoName <> "/.git && echo __OK__ || echo __MISSING__"
  299   vRes <- UIORec.uioShellExec uio (shellCmd verifyCmd) mCwdPath
  300   case vRes of
  301     Left err -> pure (CloneFailed ("clone verify failed: " <> renderUntrustedErr err))
  302     Right vOut
  303       | "__OK__" `T.isInfixOf` vOut -> do
  304           _ <- tryCodegraphInit uio repoName mCwdPath
  305           pure (CloneCloned repoName)
  306       | otherwise -> pure (CloneFailed
  307                             ("clone did not land — git output: "
  308                              <> T.strip (T.filter (/= '\n') cloneOut)))

  310 -- | Run the clone (or no-op) and build the 'OpResult' (opcode path; wraps
  311 -- 'cloneRepoIO' with the audit 'orRecorded' payload).
  312 runSetupRepo :: CloneDeps -> Text -> Value -> UIO OpResult
  313 runSetupRepo deps url recorded = do
  314   uio <- uioUntrustedIO
  315   res <- uioLiftIO (cloneRepoIO deps uio url)
  316   pure $ case res of
  317     CloneCloned repoName ->
  318       OpResult [TrpText ("Cloned " <> T.strip url <> " into " <> repoName <> " (shallow).")]
  319                False (recordWith recorded repoName "cloned")
  320     CloneNoop repoName ->
  321       OpResult [TrpText ("Repo already exists — no-op (" <> repoName <> ").")]
  322                False (recordWith recorded repoName "noop")
  323     CloneConflict repoName existing ->
  324       OpResult [TrpText ("A different repo already occupies " <> repoName <> " (existing: " <> existing <> ").")]
  325                True (recordWith recorded repoName "conflict")
  326     CloneFailed err ->
  327       OpResult [TrpText ("SETUP_REPO: " <> err)] True (recordWith recorded "" "failed")

  329 -- | Construct a 'ShellCommand', total on our internally-built command
  330 -- strings (they contain no NULs). 'mkShellCommand' only fails on NUL, so
  331 -- this never triggers the error branch in practice.
  332 shellCmd :: Text -> ShellCommand
  333 shellCmd t = case mkShellCommand t of
  334   Right c  -> c
  335   Left _e  -> error "SETUP_REPO: internal: shell command rejected (NUL?) — unreachable"

  337 -- | Re-build the recorded object with the target repo name + a status.
  338 recordWith :: Value -> Text -> Text -> Value
  339 recordWith _ target status =
  340   object [ "target" .= target, "status" .= status ]

  342 -- | Single-quote a shell token (the safe way to embed a literal in a
  343 -- @\/bin\/sh -c@ single-arg command). Any embedded @'@ becomes @'\''@.
  344 -- The URL has already been validated to contain no shell metacharacters,
  345 -- but quoting is defense-in-depth.
  346 shellQ :: Text -> Text
  347 shellQ t = "'" <> T.concatMap (\c -> if c == '\'' then "'\\''" else T.singleton c) t <> "'"

  349 -- | Best-effort: run @codegraph init@ in the freshly cloned repo directory
  350 -- so the agent has a pre-built knowledge graph on the next turn. This is
  351 -- a soft-failure operation — if @codegraph@ is not installed (exit 127) or
  352 -- fails for any reason (unsupported language, large repo, etc.), the
  353 -- clone result is unaffected. The @codegraph@ binary is looked up via
  354 -- @command -v@ first so we don't spawn a process we know will fail; if
  355 -- not found, we skip silently.
  356 --
  357 -- Security: this runs on the untrusted plane via 'uioShellExec', which is
  358 -- correct — @codegraph init@ parses untrusted repo content (tree-sitter on
  359 -- upstream code). No new attack surface is introduced.
  360 tryCodegraphInit :: UntrustedIO -> Text -> Maybe RemotePath -> IO ()
  361 tryCodegraphInit uio repoName mCwdPath = do
  362   -- Check if codegraph is installed (command -v returns 0 if found).
  363   let checkCmd = "command -v codegraph >/dev/null 2>&1"
  364   checkRes <- UIORec.uioShellExec uio (shellCmd checkCmd) mCwdPath
  365   case checkRes of
  366     Left _ -> pure ()   -- uioShellExec failed entirely — skip
  367     Right _ -> do
  368       -- codegraph exists; run init with the repo directory as cwd.
  369       -- @codegraph init@ (no path arg) initializes the project in the
  370       -- current directory. The @|| true@ ensures best-effort: any failure
  371       -- (unsupported language, etc.) is swallowed.
  372       let repoPath = case mkRemotePath repoName of
  373             Right rp -> Just rp
  374             Left _   -> Nothing
  375       let initCmd = "codegraph init 2>&1 || true"
  376       _ <- UIORec.uioShellExec uio (shellCmd initCmd) repoPath
  377       pure ()

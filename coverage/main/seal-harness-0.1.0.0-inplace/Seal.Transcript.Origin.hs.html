    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Rendering a message's 'MessageOrigin' into the text the model reads.
    3 --
    4 -- @conversation.jsonl@ stores each message's origin as a structured field
    5 -- ('Seal.Providers.Class.msgOrigin'), never inside the text: the user's words
    6 -- stay exactly as typed on disk, which keeps search, session naming, exports,
    7 -- and the web transcript view clean. The model, however, only ever sees role +
    8 -- content blocks — providers have no place to put a structured sender. So the
    9 -- turn loop renders the origin as a short tag when it builds the provider
   10 -- request ('annotateOrigins'), applying it to the WHOLE message list so every
   11 -- user message in the history carries its own sender, not just the current
   12 -- one. That is what lets the model tell several senders apart in a group chat.
   13 --
   14 -- The tag is therefore a rendering, not data: never persist an annotated
   15 -- message back to the transcript, and never treat the tag as a parse target
   16 -- (a user is free to type @[telegram:123]@ themselves — the stored origin is
   17 -- the only trusted source).
   18 module Seal.Transcript.Origin
   19   ( originFromSource
   20   , renderOriginTag
   21   , annotateOrigins
   22   ) where
   23 
   24 import Data.Text (Text)
   25 import Data.Text qualified as T

   34 -- | The 'MessageOrigin' of an inbound chat-channel message.
   35 --
   36 -- Mirrors the @erMeta@ the loop stamps into @entries.jsonl@
   37 -- ('Seal.Agent.Loop.requestMeta') so the entry-level and message-level
   38 -- provenance agree. 'Nothing' (no 'MessageSource') means the message did not
   39 -- arrive over an authenticated chat channel — the web gateway and the CLI
   40 -- bypass 'MessageSource' — and such a message carries no origin at all.
   41 originFromSource :: MessageSource -> MessageOrigin
   42 originFromSource ms = MessageOrigin
   43   { moChannel = channelKindToText (msChannelKind ms)
   44   , moUserId = userIdText <$> msUserId ms
   45   , moConversationId = Just (conversationIdText (msConversationId ms))
   46   , moPasskeyId = if channelKindToText (msChannelKind ms) == "web"
   47       then case userIdText <$> msUserId ms of
   48              Just u  -> Just u
   49              Nothing -> Map.lookup "passkeyId" (msOpen ms)
   50       else Map.lookup "passkeyId" (msOpen ms)
   51   , moClientIp = case Map.lookup "clientIp" (msOpen ms) of
   52       Just i  -> Just i
   53       Nothing -> Map.lookup "ip" (msOpen ms)
   54   }

   56 -- | The tag a message's origin renders to: @[channel:userId]@ when the
   57 -- channel authenticated a sender, @[channel]@ otherwise. Neither the channel
   58 -- tag (@[a-z]+@) nor a 'UserId' (whose charset excludes brackets and spaces)
   59 -- can break the shape, so the tag is unambiguous and single-line.
   60 renderOriginTag :: MessageOrigin -> Text
   61 renderOriginTag o = case moUserId o of
   62   Just u  -> "[" <> moChannel o <> ":" <> u <> "]"
   63   Nothing -> "[" <> moChannel o <> "]"
   64 
   65 -- | Render every user message's origin into its text, leaving everything else
   66 -- alone. Applied to the message list of a provider request only — the result
   67 -- is what the model reads, never what the transcript stores.
   68 --
   69 -- Assistant messages and tool-result messages carry no origin, so they pass
   70 -- through untouched; a user message with no origin does too. Idempotent: a
   71 -- text block that already starts with its own tag is left as is, so an
   72 -- accidental second pass cannot double-prefix.
   73 annotateOrigins :: [Message] -> [Message]
   74 annotateOrigins = map annotateOne
   75 
   76 -- | Render one message's origin, prefixing the first text block (a message
   77 -- may lead with non-text blocks; the tag goes on the first text so it reads
   78 -- before the words). A user message with no text block at all — a bare tool
   79 -- result — has nowhere to put a tag and is returned unchanged.
   80 annotateOne :: Message -> Message
   81 annotateOne m = case (msgRole m, msgOrigin m) of
   82   (User, Just o) -> case tagFirstText (renderOriginTag o) (msgContent m) of
   83     Nothing     -> m
   84     Just blocks -> m { msgContent = blocks }
   85   _ -> m
   86 
   87 -- | Prefix @tag@ to the first 'CbText' block, or 'Nothing' when there is none
   88 -- (or when that block already starts with the tag).
   89 tagFirstText :: Text -> [ContentBlock] -> Maybe [ContentBlock]
   90 tagFirstText tag = go
   91   where
   92     go [] = Nothing
   93     go (CbText t : rest)
   94       | tag `T.isPrefixOf` t = Just (CbText t : rest)
   95       | otherwise            = Just (CbText (tag <> " " <> t) : rest)
   96     go (b : rest) = (b :) <$> go rest

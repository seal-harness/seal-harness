    1 -- | A UUID-backed durable harness identity (the registry key). Minted at
    2 -- spawn time; stamped as a tmux @seal_id marker on the harness window so a
    3 -- window can be re-identified after a rename or reconnect. The UUID is
    4 -- generated in-repo from 'System.Random' (the repo has @uuid-types@ but
    5 -- not the full @uuid@ package, so v4 generation is hand-rolled from two
    6 -- random 'Word64's with the v4 version/variant bits set).
    7 --
    8 -- This module re-exports the type + pure functions from
    9 -- 'Seal.Gateway.Types.HarnessId' (the canonical home in the
   10 -- 'seal-gateway-types' library stanza). The 'newHarnessId' IO action
   11 -- stays here because it requires 'System.Random' (a server-side concern).
   12 module Seal.Harness.Id
   13   ( HarnessId (..)
   14   , newHarnessId
   15   , parseHarnessId
   16   , harnessIdToText
   17   , isValidHarnessIdText
   18   ) where
   19 
   20 import Data.Bits ((.&.), (.|.), complement)
   21 import Data.Text qualified as T
   22 import Data.UUID.Types qualified as U
   23 import Data.Word (Word64)
   24 import System.Random (randomIO)
   25 
   26 import Seal.Gateway.Types.HarnessId
   27 
   28 -- | Mint a fresh random HarnessId (UUID v4). IO because it reads randomness.
   29 newHarnessId :: IO HarnessId
   30 newHarnessId = do
   31   w1 <- randomIO
   32   w2 <- randomIO
   33   -- Set version (4) in the high nibble of byte 6 (bits 48-51 of w1) and
   34   -- variant (10) in the high bits of byte 8 (bits 56-57 of w2).
   35   let v1 = (w1 .&. complement 0x0000F000) .|. (0x4000 :: Word64)   -- version 4
   36       v2 = (w2 .|. 0x8000000000000000) .&. complement 0x4000000000000000  -- variant 10
   37   pure (HarnessId (T.pack (U.toString (U.fromWords64 v1 v2))))

    2 -- | Deciding whether the gateway may start, and in what state — the design §10
    3 -- table as two functions.
    4 --
    5 -- The split matters for testability: 'preflight' does the I/O (read
    6 -- @security.toml@ as text, read @credentials.toml@) and 'bootPlan' is pure, so
    7 -- every refusal is reachable in a unit test without a filesystem, a socket, or
    8 -- a running server.
    9 --
   10 -- Two properties this module exists to guarantee:
   11 --
   12 -- * /Totality./ Every combination of (section present?, mode?, credentials
   13 --   present?, credential version, coordinates, bind address) lands on exactly
   14 --   one row. There is no "fall through to something reasonable" arm — a state
   15 --   the table does not describe is a refusal to boot, not a guess.
   16 -- * /One place./ The mode defaulting that used to live in
   17 --   'Seal.Security.AccessControl.resolveAccessControl' — "no mode line means
   18 --   passkey if credentials exist" — is here, so a caller cannot independently
   19 --   decide what an absent @mode@ means and drift from it. In particular the
   20 --   rule that guards the whole feature: absence never means @none@.
   21 module Seal.Config.BootPlan
   22   ( BindCoordinates (..)
   23   , BootInputs (..)
   24   , CredentialsPresence (..)
   25   , BootRefusal (..)
   26   , bootRefusalMessage
   27   , BootPlan (..)
   28   , preflight
   29   , bootPlan
   30   ) where

   52 -- | The address the gateway would bind, which decides whether credentials can

   55 data BindCoordinates = BindCoordinates
   56   { bcHost :: !Text
   57   , bcPort :: !Int
   58   } deriving stock (Eq, Show)

   60 -- | Everything 'bootPlan' needs, gathered by 'preflight'.
   61 data BootInputs = BootInputs
   62   { biSecurityRaw :: !Text
   63     -- ^ The @security.toml@ bytes as they are on disk. Kept because the
   64     -- legacy-layout check has to see the document, not a decoded summary of
   65     -- it: decoding is what loses a table the codec does not know.
   66   , biSection     :: !(Maybe AccessControlFileConfig)
   67     -- ^ 'Nothing' when the file is absent or has no @[access_control]@.
   68   , biCredentials :: !CredentialsPresence
   69   } deriving stock (Eq, Show)
   70 
   71 -- | What @credentials.toml@ turned out to be. Three states, because boot treats
   72 -- them differently: absent is normal (nothing enrolled yet), invalid is a
   73 -- refusal, and present-but-empty is a corruption signal — the store's writers
   74 -- cannot produce it (design §9.2), so seeing it means something else did.
   75 data CredentialsPresence
   76   = CredentialsAbsent
   77   | CredentialsPresent !Int !(Map Text CredentialFileConfig)
   78   | CredentialsInvalid Text
   79   deriving stock (Eq, Show)
   80 
   81 -- | Why the gateway will not start. Every constructor names the thing the
   82 -- operator has to change; none of them is a catch-all.
   83 data BootRefusal
   84   = BrSecurityUnparseable Text
   85   | BrLegacyLayout
   86   | BrCredentialsInvalid Text
   87   | BrCredentialsEmpty
   88   | BrCredentialsVersion !Int
   89   | BrModeInvalid Text
   90   | BrCoordinatesInvalid Text
   91   | BrUnsafeBind Text
   92   | BrPublicBindWithLoopbackOrigin
   93   deriving stock (Eq, Show)
   94 
   95 -- | The operator-facing text for a refusal. Each message names the command or
   96 -- the edit that fixes it — a refusal that only says "no" leaves the operator
   97 -- with a gateway that will not start and no way forward.
   98 bootRefusalMessage :: BootRefusal -> Text
   99 bootRefusalMessage refusal = case refusal of
  100   BrSecurityUnparseable err ->
  101     "security.toml could not be read: " <> err
  102       <> "\n  Fix ~/.seal/security.toml by hand; `seal serve` will not start until it parses."
  103   BrLegacyLayout ->
  104     "security.toml still holds the pre-split [access_control.credentials] tables.\n"
  105       <> "  Run `seal access-control migrate-credentials` to move them to credentials.toml."
  106   BrCredentialsInvalid err ->
  107     "credentials.toml could not be read: " <> err
  108       <> "\n  Run `seal access-control reset --yes` (with the daemon stopped) and enroll again."
  109   BrCredentialsEmpty ->
  110     "credentials.toml exists but holds no credentials.\n"
  111       <> "  The store is never written empty on purpose; treat this as corruption.\n"
  112       <> "  Run `seal access-control reset --yes` (with the daemon stopped) and enroll again."
  113   BrCredentialsVersion version ->
  114     "credentials.toml has format version " <> tshow version
  115       <> ", which this build does not understand.\n"
  116       <> "  Upgrade seal, or restore a backup of ~/.seal/credentials.toml."
  117   BrModeInvalid raw ->
  118     "access_control.mode is \"" <> raw <> "\", which is not a valid mode.\n"
  119       <> "  Use \"passkey\" (the default) or \"none\"."
  120   BrCoordinatesInvalid err ->
  121     "[access_control] origin/rp_id are unusable: " <> err
  122       <> "\n  Set origin to an absolute URL (or unset it for a loopback bind)."
  123   BrUnsafeBind reason ->
  124     "refusing to serve passkey-protected access with an unusable origin: " <> reason
  125       <> "\n  Bind a loopback address, or run behind a TLS reverse proxy and set "
  126       <> "[access_control] origin to its https:// URL."
  127   BrPublicBindWithLoopbackOrigin ->
  128     "the gateway binds a non-loopback address but [access_control] origin is a "
  129       <> "loopback URL.\n"
  130       <> "  Browsers reaching it from another machine cannot use a localhost "
  131       <> "origin, and the setup window would be reachable from the network.\n"
  132       <> "  Set origin to the address clients actually use."
  133 
  134 -- | Everything the runtime needs, already decided. The handle built from this
  135 -- never reads @credentials.toml@ again, which is what makes a mid-flight edit
  136 -- of the store unable to change the running gateway's behaviour.
  137 data BootPlan = BootPlan
  138   { bpMode      :: !AccessMode
  139   , bpOrigin    :: !(Maybe Text)
  140   , bpRpId      :: !(Maybe Text)
  141   , bpCreds     :: !(Map Text Credential)
  142   , bpMintGrant :: !(Maybe GrantSource)
  143     -- ^ 'Just' when boot should mint an enrollment grant: nothing enrolled,
  144     -- nothing usable, or access control is off. Never minted when a usable
  145     -- credential exists — there is already an authority.
  146   } deriving stock (Eq, Show)

  148 -- | Read the two files and hand 'bootPlan' what it needs.
  149 --
  150 -- Reads @security.toml@ as text /and/ decodes it: the raw document is what the
  151 -- legacy check needs, and the decoded section is what the table needs.
  152 preflight :: SealPaths -> IO (Either BootRefusal BootInputs)
  153 preflight paths = do
  154   securityExists <- doesFileExist (securityFilePath paths)
  155   raw <- if securityExists then TIO.readFile (securityFilePath paths) else pure ""
  156   creds <- credentialsPresence paths
  157   -- A document that does not decode leaves the section absent here; 'bootPlan'
  158   -- refuses it with the decoder's own message. Deciding it in one place keeps
  159   -- the refusals testable without files, and keeps the two copies from
  160   -- disagreeing about which condition wins.
  161   pure (Right BootInputs
  162     { biSecurityRaw = raw
  163     , biSection     = either (const Nothing) scAccessControl (decodeSecurityConfig raw)
  164     , biCredentials = creds })
  165 
  166 credentialsPresence :: SealPaths -> IO CredentialsPresence
  167 credentialsPresence paths = do
  168   loaded <- loadCredentials (credentialsFilePath paths)
  169   pure $ case loaded of
  170     Left err               -> CredentialsInvalid err
  171     Right Nothing          -> CredentialsAbsent
  172     Right (Just file)      -> CredentialsPresent (crdVersion file) (crdCredentials file)
  173 
  174 -- | The design §10 table.
  175 bootPlan :: BindCoordinates -> BootInputs -> Either BootRefusal BootPlan
  176 bootPlan coords inputs = do
  177   -- The two document-level refusals. Both live here rather than in 'preflight'
  178   -- because 'BootInputs' carries the raw text precisely so the decision stays
  179   -- pure. Order matters: a document that does not even parse is refused first,
  180   -- since 'hasLegacyLayout' answers False for it and would otherwise let a
  181   -- corrupt file through as "no legacy layout".
  182   case decodeSecurityConfig (biSecurityRaw inputs) of
  183     Left err -> Left (BrSecurityUnparseable err)
  184     Right _  -> Right ()
  185   -- Legacy before anything about the store: a pre-split install has no
  186   -- credentials.toml at all, so reporting the missing store would send the
  187   -- operator to `reset --yes` when the answer is `migrate-credentials`.
  188   if hasLegacyLayout (biSecurityRaw inputs) then Left BrLegacyLayout else Right ()
  189   creds <- plannedCredentials (biCredentials inputs)
  190   mode <- plannedMode inputs creds
  191   (origin, rpId) <- case resolveCoordinates (bcHost coords) (bcPort coords)
  192                                             (biSection inputs) mode of
  193     Left err -> Left (BrCoordinatesInvalid err)
  194     Right resolved -> Right resolved
  195   checkBind coords mode origin rpId
  196   pure BootPlan
  197     { bpMode      = mode
  198     , bpOrigin    = origin
  199     , bpRpId      = rpId
  200     , bpCreds     = creds
  201     , bpMintGrant = mintFor mode rpId creds
  202     }

  204 -- | The credential map, or a refusal. The version is checked before anything
  205 -- else: a future format must not be read with this build's assumptions.
  206 plannedCredentials :: CredentialsPresence -> Either BootRefusal (Map Text Credential)
  207 plannedCredentials presence = case presence of
  208   CredentialsAbsent        -> Right Map.empty
  209   CredentialsInvalid err   -> Left (BrCredentialsInvalid err)
  210   CredentialsPresent v raw
  211     | v /= credentialsVersion -> Left (BrCredentialsVersion v)
  212     | Map.null raw            -> Left BrCredentialsEmpty
  213     | otherwise               -> case traverse credentialFromFile (Map.toList raw) of
  214         Left err -> Left (BrCredentialsInvalid err)
  215         Right ok -> Right (Map.fromList ok)
  216 
  217 -- | Mode, including the one defaulting rule: no @mode@ line means passkey when
  218 -- credentials exist and unconfigured otherwise. @none@ is never inferred.
  219 plannedMode :: BootInputs -> Map Text Credential -> Either BootRefusal AccessMode
  220 plannedMode inputs creds = case biSection inputs >>= acfcMode of
  221   Nothing
  222     | Map.null creds -> Right AccessUnconfigured
  223     | otherwise      -> Right AccessPasskey
  224   Just raw -> case parseAccessMode raw of
  225     Left err  -> Left (BrModeInvalid err)
  226     Right mode -> Right mode
  227 
  228 -- | The two bind checks, kept as separate constructors so a failure tells the
  229 -- operator which of the two distinct problems they have.
  230 checkBind
  231   :: BindCoordinates -> AccessMode -> Maybe Text -> Maybe Text
  232   -> Either BootRefusal ()
  233 checkBind coords mode origin rpId
  234   -- Most specific first: a public bind pointed at a loopback origin is that
  235   -- mistake, not the broader "not https" one, and the two need different fixes.
  236   | not loopbackBind && maybe False isLoopbackHost rpId =
  237       Left BrPublicBindWithLoopbackOrigin
  238   | mode == AccessPasskey && not loopbackBind && not originIsHttps =
  239       Left (BrUnsafeBind "the bind address is not loopback and the origin is not https")
  240   | otherwise = Right ()
  241   where
  242     loopbackBind = isLoopbackHost (bcHost coords)
  243     originIsHttps = maybe False ("https://" `T.isPrefixOf`) origin

  245 -- | Which boot states mint an enrollment grant (design §11).
  246 --
  247 -- A usable credential means an assertion is available, so nothing is minted. In
  248 -- @open@ mode the grant is what stops a network client enrolling its own
  249 -- credential and flipping a public gateway into one locked with a key the
  250 -- operator does not hold.
  251 mintFor :: AccessMode -> Maybe Text -> Map Text Credential -> Maybe GrantSource
  252 mintFor mode rpId creds = case mode of
  253   AccessUnconfigured -> Just CliToken
  254   AccessOpen         -> Just CliToken
  255   AccessPasskey      -> case credentialAvailability rpId creds of
  256     Usable       -> Nothing
  257     NoCredentials -> Just CliToken
  258     UnusableOnly -> Just CliToken
  259 tshow :: Show a => a -> Text

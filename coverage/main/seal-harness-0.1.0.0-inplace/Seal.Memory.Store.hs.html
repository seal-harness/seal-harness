    3 -- | The file-based memory store. Memory files live as plain text (no
    4 -- frontmatter) under @\<root\>\/active\/\<path\>.md@. Archived memories live
    5 -- under @\<root\>\/archived\/\<dir\>\/\<timestamp\>-\<filename\>.md@.
    6 --
    7 -- The store is write-once: 'msWrite' fails if the file already exists in
    8 -- @active\/@. To update a fact, archive the old memory ('msArchive') then
    9 -- write the new one. Files are never deleted — archive is a move.
   10 --
   11 -- No git auto-commit. The store manages its own immutability via the
   12 -- write-once + archive model. The 'fileMemoryStore' is the production
   13 -- backend; 'noneMemoryStore' is an in-memory backend for tests.
   14 module Seal.Memory.Store
   15   ( MemoryStore (..)
   16   , fileMemoryStore
   17   , noneMemoryStore
   18   ) where

   49 -- | The memory store capability. Each operation is IO.
   50 data MemoryStore = MemoryStore
   51   { msWrite   :: MemoryPath -> Text -> IO (Either Text ())
   52     -- ^ Write a new memory file. Fails if the path already exists in active/.
   53   , msRead    :: MemoryPath -> IO (Either Text (Text, Bool))
   54     -- ^ Read a memory by path. Returns (content, isArchived).
   55     --   Falls back to archived/ if not in active/.
   56   , msList    :: Text -> Bool -> IO [MemoryPath]
   57     -- ^ List memory paths matching a prefix. includeArchived flag.
   58   , msArchive :: MemoryPath -> IO (Either Text MemoryPath)
   59     -- ^ Move active/<path> to archived/<path>/<timestamp>-<filename>.
   60   , msSearch :: Text -> Bool -> IO [(MemoryPath, Text)]
   61     -- ^ Substring search over memory content. Returns (path, content) pairs.
   62     --   includeArchived flag.
   63   }
   64 
   65 -- | The on-disk file extension for memory files.
   66 mdExt :: FilePath
   67 mdExt = ".md"
   68 
   69 -- | Convert a 'MemoryPath' to a relative 'FilePath' with .md extension.
   70 pathToFile :: MemoryPath -> FilePath
   71 pathToFile mp =
   72   T.unpack (T.intercalate "/" (memoryPathSegments mp)) <.> mdExt
   73 
   74 -- | Format a UTC timestamp as @YYYYMMDDThhmmssZ@.
   75 formatTimestamp :: UTCTime -> String
   76 formatTimestamp = formatTime defaultTimeLocale "%Y%m%dT%H%M%SZ"
   77 
   78 -- | The production disk-backed memory store. @root@ is the memory root
   79 -- (e.g. @~\/.seal\/memory@). Files live under @root\/active\/@ and
   80 -- @root\/archived\/@.
   81 fileMemoryStore :: FilePath -> IO MemoryStore
   82 fileMemoryStore root = do
   83   let activeDir   = root </> "active"
   84       archivedDir = root </> "archived"
   85   createDirectoryIfMissing True activeDir
   86   createDirectoryIfMissing True archivedDir
   87   pure MemoryStore
   88     { msWrite = \mp content -> do
   89         let relPath = pathToFile mp
   90             absPath = activeDir </> relPath
   91         exists <- doesFileExist absPath
   92         if exists
   93           then pure (Left ("memory already exists: " <> memoryPathText mp))
   94           else do
   95             let dir = takeDirectory absPath
   96             createDirectoryIfMissing True dir
   97             -- Atomic write: temp file in the same directory, then rename.
   98             let tmp = absPath <.> "tmp"
   99             TIO.writeFile tmp content
  100             renameFile tmp absPath
  101             pure (Right ())
  102     , msRead = \mp -> do
  103         let relPath = pathToFile mp
  104             activePath = activeDir </> relPath
  105         activeExists <- doesFileExist activePath
  106         if activeExists
  107           then do
  108             content <- TIO.readFile activePath
  109             pure (Right (content, False))
  110           else do
  111             -- Fall back to archived/: walk the archived tree for a file
  112             -- matching the path (by filename suffix, since archived files
  113             -- have a timestamp prefix).
  114             mArchived <- findArchived archivedDir mp
  115             case mArchived of
  116               Just archivedPath -> do
  117                 content <- TIO.readFile archivedPath
  118                 pure (Right (content, True))
  119               Nothing -> pure (Left ("memory not found: " <> memoryPathText mp))
  120     , msList = \prefix includeArchived -> do
  121         activeEntries <- listTree activeDir
  122         archivedEntries <-
  123           if includeArchived
  124             then listTree archivedDir
  125             else pure []
  126         -- For archived entries, strip the timestamp prefix from the filename
  127         -- so the path matches what the agent would use.
  128         let activePaths = mapMaybe (relPathToMemoryPath activeDir) activeEntries
  129             archivedPaths = mapMaybe (archivedRelPathToMemoryPath archivedDir) archivedEntries
  130             allPaths = activePaths <> archivedPaths
  131             filtered = filter (matchesPrefix prefix) allPaths
  132         pure (sortOn memoryPathText filtered)
  133     , msArchive = \mp -> do
  134         let relPath = pathToFile mp
  135             activePath = activeDir </> relPath
  136         exists <- doesFileExist activePath
  137         if not exists
  138           then pure (Left ("memory not found: " <> memoryPathText mp))
  139           else do
  140             now <- getCurrentTime
  141             let ts = formatTimestamp now
  142                 filename = takeFileName relPath
  143                 archivedFilename = ts <> "-" <> filename
  144                 -- Preserve the directory hierarchy under archived/.
  145                 dirPart = takeDirectory relPath
  146                 archivedRelPath =
  147                   if dirPart == "."
  148                     then archivedFilename
  149                     else dirPart </> archivedFilename
  150                 archivedAbsPath = archivedDir </> archivedRelPath
  151             createDirectoryIfMissing True (takeDirectory archivedAbsPath)
  152             renameFile activePath archivedAbsPath
  153             -- The archived path as a MemoryPath (with timestamp prefix).
  154             -- We construct it from the relative path under archived/.
  155             let archivedMP = case mkMemoryPath (T.pack archivedRelPath) of
  156                   Right p  -> p
  157                   Left _   -> mp  -- fallback (shouldn't happen)
  158             pure (Right archivedMP)
  159     , msSearch = \query includeArchived -> do
  160         activeEntries <- listTree activeDir
  161         archivedEntries <-
  162           if includeArchived
  163             then listTree archivedDir
  164             else pure []
  165         let activePaths = mapMaybe (relPathToMemoryPath activeDir) activeEntries
  166             archivedPaths = mapMaybe (archivedRelPathToMemoryPath archivedDir) archivedEntries
  167         activeResults <- mapM (readAndFilter activeDir query) activePaths
  168         archivedResults <-
  169           if includeArchived
  170             then mapM (readAndFilterArchived query) (zip (map (archivedDir </>) archivedEntries) archivedPaths)
  171             else pure []
  172         pure (catMaybes activeResults <> catMaybes archivedResults)
  173     }
  174   where
  175     -- Recursively list all files under a directory, returning relative paths.
  176     listTree :: FilePath -> IO [FilePath]
  177     listTree dir = do
  178       entries <- listDirectory dir
  179       concat <$> forM entries \entry -> do
  180         let absEntry = dir </> entry
  181         isFile <- doesFileExist absEntry
  182         if isFile
  183           then pure [entry]
  184           else do
  185             subEntries <- listTree absEntry
  186             pure (map (entry </>) subEntries)
  187 
  188     -- Convert a relative file path (from listTree) to a MemoryPath by
  189     -- stripping the .md extension.
  190     relPathToMemoryPath :: FilePath -> FilePath -> Maybe MemoryPath
  191     relPathToMemoryPath _baseDir relPath = do
  192       -- Strip .md extension
  193       let withoutExt = if mdExt `isInfixOf` relPath
  194                          then take (length relPath - length mdExt) relPath
  195                          else relPath
  196       either (const Nothing) Just (mkMemoryPath (T.pack withoutExt))
  197 
  198     -- Like relPathToMemoryPath but strips the timestamp prefix from the
  199     -- filename (e.g. "20260919T143000Z-tz.md" → "tz").
  200     archivedRelPathToMemoryPath :: FilePath -> FilePath -> Maybe MemoryPath
  201     archivedRelPathToMemoryPath _baseDir relPath = do
  202       let filename = takeFileName relPath
  203           dirPart = takeDirectory relPath
  204           -- Strip timestamp prefix: everything up to and including the first "-"
  205           strippedFilename = case dropWhile (/= '-') filename of
  206             ""       -> filename
  207             (_:rest)  -> rest  -- drop the '-' itself
  208           withoutExt = if mdExt `isInfixOf` strippedFilename
  209                          then take (length strippedFilename - length mdExt) strippedFilename
  210                          else strippedFilename
  211           -- Reconstruct the path without the timestamp
  212           newPath = if dirPart == "."
  213                       then withoutExt
  214                       else dirPart </> withoutExt
  215       either (const Nothing) Just (mkMemoryPath (T.pack newPath))
  216 
  217     -- Check if a MemoryPath matches a prefix.
  218     matchesPrefix :: Text -> MemoryPath -> Bool
  219     matchesPrefix prefix mp =
  220       T.null prefix || prefix `T.isPrefixOf` memoryPathText mp
  221 
  222     -- Read a memory file and return (path, content) if the query is a
  223     -- substring of the content (case-insensitive). 'Nothing' if no match.
  224     readAndFilter :: FilePath -> Text -> MemoryPath -> IO (Maybe (MemoryPath, Text))
  225     readAndFilter baseDir query mp = do
  226       let relPath = pathToFile mp
  227           absPath = baseDir </> relPath
  228       exists <- doesFileExist absPath
  229       if not exists
  230         then pure Nothing
  231         else do
  232           content <- TIO.readFile absPath
  233           if query `T.isInfixOf` T.toCaseFold content
  234             then pure (Just (mp, content))
  235             else pure Nothing
  236 
  237     -- Like 'readAndFilter' but reads from an explicit archived path.
  238     readAndFilterArchived :: Text -> (FilePath, MemoryPath) -> IO (Maybe (MemoryPath, Text))
  239     readAndFilterArchived query (absPath, mp) = do
  240       exists <- doesFileExist absPath
  241       if not exists
  242         then pure Nothing
  243         else do
  244           content <- TIO.readFile absPath
  245           if query `T.isInfixOf` T.toCaseFold content
  246             then pure (Just (mp, content))
  247             else pure Nothing
  248 
  249 -- | Find an archived file matching the given memory path. Archived files
  250 -- have a timestamp prefix, so we walk the directory tree and match by
  251 -- filename suffix.
  252 findArchived :: FilePath -> MemoryPath -> IO (Maybe FilePath)
  253 findArchived archivedDir mp = do
  254   let relPath = pathToFile mp
  255       dirPart = takeDirectory relPath
  256       filename = takeFileName relPath
  257       searchDir = if dirPart == "."
  258                     then archivedDir
  259                     else archivedDir </> dirPart
  260   dirExists <- doesDirectoryExist searchDir
  261   if not dirExists
  262     then pure Nothing
  263     else do
  264       entries <- listDirectory searchDir
  265       let candidates = [e | e <- entries, e /= takeFileName relPath, matchArchived filename e]
  266       case candidates of
  267         (match:_) -> pure (Just (searchDir </> match))
  268         []        -> pure Nothing
  269   where
  270     -- Match an archived filename (with timestamp prefix) against the
  271     -- original filename. E.g. "20260919T143000Z-tz.md" matches "tz.md".
  272     matchArchived original archived =
  273       let stripped = case dropWhile (/= '-') archived of
  274             ""       -> archived
  275             (_:rest)  -> rest
  276       in stripped == original

  278 -- | The in-memory store for tests. Uses IORefs to track active and
  279 -- archived memories.
  280 noneMemoryStore :: IO MemoryStore
  281 noneMemoryStore = do
  282   activeRef   <- newIORef (Map.empty :: Map Text Text)
  283   archivedRef <- newIORef (Map.empty :: Map Text Text)
  284   pure MemoryStore
  285     { msWrite = \mp content -> do
  286         let key = memoryPathText mp
  287         active <- readIORef activeRef
  288         if Map.member key active
  289           then pure (Left ("memory already exists: " <> key))
  290           else do
  291             modifyIORef' activeRef (Map.insert key content)
  292             pure (Right ())
  293     , msRead = \mp -> do
  294         let key = memoryPathText mp
  295         active <- readIORef activeRef
  296         case Map.lookup key active of
  297           Just content -> pure (Right (content, False))
  298           Nothing -> do
  299             archived <- readIORef archivedRef
  300             case Map.lookup key archived of
  301               Just content -> pure (Right (content, True))
  302               Nothing      -> pure (Left ("memory not found: " <> key))
  303     , msList = \prefix includeArchived -> do
  304         active <- readIORef activeRef
  305         archived <- readIORef archivedRef
  306         let activePaths = map fst (Map.toAscList active)
  307             archivedPaths = map fst (Map.toAscList archived)
  308             allPaths = activePaths <> (if includeArchived then archivedPaths else [])
  309             filtered = filter (prefix `T.isPrefixOf`) allPaths
  310         pure (rights (map mkMemoryPath filtered))
  311     , msArchive = \mp -> do
  312         let key = memoryPathText mp
  313         active <- readIORef activeRef
  314         case Map.lookup key active of
  315           Nothing -> pure (Left ("memory not found: " <> key))
  316           Just content -> do
  317             modifyIORef' activeRef (Map.delete key)
  318             modifyIORef' archivedRef (Map.insert key content)
  319             pure (Right mp)
  320     , msSearch = \query includeArchived -> do
  321         active <- readIORef activeRef
  322         archived <- readIORef archivedRef
  323         let activeResults = [ (mp, c)
  324                             | (k, c) <- Map.toAscList active
  325                             , query `T.isInfixOf` T.toCaseFold c
  326                             , Right mp <- [mkMemoryPath k] ]
  327             archivedResults = [ (mp, c)
  328                               | includeArchived
  329                               , (k, c) <- Map.toAscList archived
  330                               , query `T.isInfixOf` T.toCaseFold c
  331                               , Right mp <- [mkMemoryPath k] ]
  332         pure (activeResults <> archivedResults)
  333     }

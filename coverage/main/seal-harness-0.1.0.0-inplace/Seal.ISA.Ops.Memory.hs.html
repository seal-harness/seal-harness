    2 -- | The Memory opcode group. The consolidated entry point is
    3 -- 'memoryManageOp' (\"MEMORY_MANAGE\"), which dispatches on an @action@
    4 -- field to one of five handlers: @write@, @read@, @list@, @search@,
    5 -- @archive@.
    6 --
    7 -- The legacy opcodes ('memoryWriteOp', 'memoryReadOp', 'memoryListOp',
    8 -- 'memorySearchOp', 'memoryArchiveOp') remain as thin shims that delegate
    9 -- to the same handlers with a pre-set action. This preserves backward
   10 -- compatibility during the transition period — existing call sites, tests,
   11 -- and downstream consumers continue to work unchanged.
   12 --
   13 -- All memory opcodes are 'Trusted' — the dispatcher writes the session
   14 -- transcript; the opcodes operate on the file-based memory store
   15 -- ('Seal.Memory.Store') and the embedding backend ('Seal.Memory.Embedding').
   16 --
   17 -- The memory system is write-once: the @write@ action fails if the path
   18 -- already exists in @active\/@. To update a fact, archive the old memory
   19 -- (@archive@ action) then write the new one. Files are never deleted —
   20 -- archive is a move from @active\/@ to @archived\/@ with a timestamp
   21 -- prefix.
   22 --
   23 -- The @read@ action falls back to @archived\/@ if the file isn't in
   24 -- @active\/@, returning the content with an @archived: true@ flag.
   25 --
   26 -- The @search@ action delegates to the 'EmbeddingBackend'. The null
   27 -- backend returns empty results (not an error), with a substring fallback.
   28 module Seal.ISA.Ops.Memory
   29   ( -- * Consolidated opcode
   30     memoryManageOp
   31     -- * Legacy shims (delegate to the same handlers)
   32   , memoryWriteOp
   33   , memoryReadOp
   34   , memoryListOp
   35   , memorySearchOp
   36   , memoryArchiveOp
   37   ) where

   60 data MemoryAction
   61   = MemWrite
   62   | MemRead
   63   | MemList
   64   | MemSearch
   65   | MemArchive
   66 
   67 parseMemoryAction :: Value -> Either Text MemoryAction
   68 parseMemoryAction v =
   69   case parseMaybe (withObject "in" (.: "action")) v of
   70     Just t -> case t :: Text of
   71       "write"   -> Right MemWrite
   72       "read"    -> Right MemRead
   73       "list"    -> Right MemList
   74       "search"  -> Right MemSearch
   75       "archive" -> Right MemArchive
   76       other     -> Left ("unknown memory action: " <> other)
   77     Nothing -> Left "missing or invalid action field"

   79 -- ---------------------------------------------------------------------------
   80 -- Field extractors
   81 -- ---------------------------------------------------------------------------

   83 -- | Extract the @path@ string field from a JSON object.
   84 pathField :: Value -> Maybe Text
   85 pathField = parseMaybe (withObject "in" (.: "path"))
   86 
   87 -- | Extract the @content@ string field (defaults to empty when absent).
   88 contentField :: Value -> Text
   89 contentField v = fromMaybe "" (parseMaybe (withObject "in" (.: "content")) v)
   90 
   91 -- | Extract the optional @prefix@ string field for LIST.
   92 prefixField :: Value -> Text
   93 prefixField v =
   94   case parseMaybe (withObject "in" (.:? "prefix")) v :: Maybe (Maybe Text) of
   95     Just (Just t) -> t
   96     _             -> ""
   97 
   98 -- | Extract the optional @include_archived@ bool field (defaults to False).
   99 includeArchivedField :: Value -> Bool
  100 includeArchivedField v =
  101   case parseMaybe (withObject "in" (.:? "include_archived")) v :: Maybe (Maybe Bool) of
  102     Just (Just b) -> b
  103     _             -> False
  104 
  105 -- | Extract the optional @query@ string field for SEARCH.
  106 queryField :: Value -> Maybe Text
  107 queryField v =
  108   case parseMaybe (withObject "in" (.:? "query")) v :: Maybe (Maybe Text) of
  109     Just (Just q) -> Just q
  110     _             -> Nothing
  111 
  112 -- | Extract the optional @limit@ integer field for SEARCH (defaults to 10).
  113 limitField :: Value -> Int
  114 limitField v =
  115   case parseMaybe (withObject "in" (.:? "limit")) v :: Maybe (Maybe Int) of
  116     Just (Just n) | n > 0 -> n
  117     _                     -> 10
  118 
  119 -- | Validate and parse the path field into a 'MemoryPath'.
  120 parsePath :: Value -> Either Text MemoryPath
  121 parsePath v =
  122   case pathField v of
  123     Nothing -> Left "missing path field"
  124     Just t  -> mkMemoryPath t

  126 -- ---------------------------------------------------------------------------
  127 -- Handlers (shared between MEMORY_MANAGE and legacy shims)
  128 -- ---------------------------------------------------------------------------
  129 
  130 handleWrite :: MemoryStore -> EmbeddingBackend -> Value -> App OpResult
  131 handleWrite store embedding v =
  132   case parsePath v of
  133     Left e -> pure (OpResult [TrpText ("invalid path: " <> e)] True (object []))
  134     Right mp -> do
  135       let content = contentField v
  136       result <- liftIO (msWrite store mp content)
  137       case result of
  138         Left e -> pure (OpResult [TrpText e] True (object ["path" .= memoryPathText mp]))
  139         Right _ -> do
  140           liftIO (ebIndex embedding (T.unpack (memoryPathText mp)) content)
  141           let recorded = object
  142                 [ "path" .= memoryPathText mp
  143                 , "content" .= content
  144                 ]
  145           pure (OpResult [TrpText "stored"] False recorded)
  146 
  147 handleRead :: MemoryStore -> Value -> App OpResult
  148 handleRead store v =
  149   case parsePath v of
  150     Left e -> pure (OpResult [TrpText ("invalid path: " <> e)] True (object []))
  151     Right mp -> do
  152       result <- liftIO (msRead store mp)
  153       case result of
  154         Left _e -> do
  155           let recorded = object
  156                 [ "path" .= memoryPathText mp
  157                 , "exists" .= False
  158                 ]
  159           pure (OpResult [TrpText ("not found: " <> memoryPathText mp)] False recorded)
  160         Right (content, isArchived) -> do
  161           let recorded = object
  162                 [ "path" .= memoryPathText mp
  163                 , "content" .= content
  164                 , "exists" .= True
  165                 , "archived" .= isArchived
  166                 ]
  167               msg = if isArchived
  168                       then "archived: " <> content
  169                       else content
  170           pure (OpResult [TrpText msg] False recorded)
  171 
  172 handleList :: MemoryStore -> Value -> App OpResult
  173 handleList store v = do
  174   let prefix = prefixField v
  175       includeArchived = includeArchivedField v
  176   entries <- liftIO (msList store prefix includeArchived)
  177   let rendered = T.intercalate "\n" (map memoryPathText entries)
  178         <> "\n---\n" <> T.pack (show (length entries)) <> " entries"
  179       recorded = object
  180         [ "prefix" .= prefix
  181         , "include_archived" .= includeArchived
  182         , "count" .= length entries
  183         ]
  184   pure (OpResult [TrpText rendered] False recorded)
  185 
  186 handleSearch :: EmbeddingBackend -> MemoryStore -> Value -> App OpResult
  187 handleSearch embedding store v =
  188   case queryField v of
  189     Nothing -> pure (OpResult [TrpText "missing query"] True (object []))
  190     Just query -> do
  191       let limit = limitField v
  192           includeArchived = includeArchivedField v
  193       results <- liftIO (ebSearch embedding query limit)
  194       substringResults <-
  195         if null results
  196           then liftIO (msSearch store query includeArchived)
  197           else pure []
  198       let totalResults = length results + length substringResults
  199           allResults = map renderEmbeddingResult results
  200                     <> map renderSubstringResult substringResults
  201           rendered = if null allResults
  202                        then "No results."
  203                        else T.intercalate "\n" allResults
  204                    <> "\n---\n" <> T.pack (show totalResults) <> " results"
  205           recorded = object
  206             [ "query" .= query
  207             , "limit" .= limit
  208             , "total_matches" .= totalResults
  209             ]
  210       pure (OpResult [TrpText rendered] False recorded)
  211   where
  212     renderEmbeddingResult sr =
  213       srPath sr <> " (score: " <> T.pack (show (srScore sr)) <> "):\n"
  214         <> srContent sr
  215     renderSubstringResult (mp, content) =
  216       memoryPathText mp <> ":\n" <> content
  217 
  218 handleArchive :: MemoryStore -> EmbeddingBackend -> Value -> App OpResult
  219 handleArchive store embedding v =
  220   case parsePath v of
  221     Left e -> pure (OpResult [TrpText ("invalid path: " <> e)] True (object []))
  222     Right mp -> do
  223       result <- liftIO (msArchive store mp)
  224       case result of
  225         Left e -> pure (OpResult [TrpText e] True (object ["path" .= memoryPathText mp]))
  226         Right archivedPath -> do
  227           liftIO (ebUnindex embedding (T.unpack (memoryPathText mp)))
  228           let recorded = object
  229                 [ "path" .= memoryPathText mp
  230                 , "archived_path" .= memoryPathText archivedPath
  231                 ]
  232           pure (OpResult [TrpText ("archived: " <> memoryPathText archivedPath)] False recorded)

  234 -- ---------------------------------------------------------------------------

  238 authorizeWrite :: Value -> Either Text ()
  239 authorizeWrite v =
  240   case parsePath v of
  241     Left e -> Left e
  242     Right _ -> case pathField v of
  243       Just _ -> case contentField v of
  244         "" -> Left "write requires non-empty content"
  245         _  -> Right ()
  246       Nothing -> Left "write requires path"
  247 
  248 authorizeRead :: Value -> Either Text ()
  249 authorizeRead = either Left (const (Right ())) . parsePath
  250 
  251 authorizeSearch :: Value -> Either Text ()
  252 authorizeSearch v =
  253   case queryField v of
  254     Nothing -> Left "search requires query"
  255     Just _  -> Right ()
  256 
  257 authorizeArchive :: Value -> Either Text ()
  258 authorizeArchive = either Left (const (Right ())) . parsePath
  259 
  260 -- | Authorize gate for MEMORY_MANAGE — dispatches per-action validation.
  261 authorizeManage :: Value -> Either Text ()
  262 authorizeManage v =
  263   case parseMemoryAction v of
  264     Left e -> Left e
  265     Right action -> case action of
  266       MemWrite   -> authorizeWrite v
  267       MemRead    -> authorizeRead v
  268       MemList    -> Right ()
  269       MemSearch  -> authorizeSearch v
  270       MemArchive -> authorizeArchive v

  272 -- ---------------------------------------------------------------------------
  273 -- Consolidated opcode: MEMORY_MANAGE

  276 -- | MEMORY_MANAGE: action-based entry point for all memory operations.
  277 -- The @action@ field discriminates between @write@, @read@, @list@,
  278 -- @search@, and @archive@. Per-action required-field validation happens
  279 -- in the authorize gate.
  280 memoryManageOp :: MemoryStore -> EmbeddingBackend -> Opcode
  281 memoryManageOp store embedding = TrustedOpcode
  282   { toName = OpName "MEMORY_MANAGE"
  283   , toTrust = Trusted
  284   , toDesc = "Manage memory files. Use action to select: write (create, write-once), read (by path, falls back to archived/), list (by prefix), search (semantic), archive (move to archived/)."
  285   , toInSchema = object
  286       [ "type" .= ("object" :: Text)
  287       , "properties" .= object
  288           [ fromText "action" .= object
  289               [ "type" .= ("string" :: Text)
  290               , "enum" .= (["write", "read", "list", "search", "archive"] :: [Text])
  291               , "description" .= ("Operation to perform." :: Text)
  292               ]
  293           , fromText "path" .= object
  294               [ "type" .= ("string" :: Text)
  295               , "description" .= ("Memory path relative to active/ (write, read, archive)." :: Text)
  296               ]
  297           , fromText "content" .= object
  298               [ "type" .= ("string" :: Text)
  299               , "description" .= ("Memory content, plain text (write only)." :: Text)
  300               ]
  301           , fromText "prefix" .= object
  302               [ "type" .= ("string" :: Text)
  303               , "description" .= ("Directory prefix to filter by (list only)." :: Text)
  304               ]
  305           , fromText "query" .= object
  306               [ "type" .= ("string" :: Text)
  307               , "description" .= ("Search query (search only)." :: Text)
  308               ]
  309           , fromText "limit" .= object
  310               [ "type" .= ("integer" :: Text)
  311               , "description" .= ("Max search results (default: 10)." :: Text)
  312               ]
  313           , fromText "include_archived" .= object
  314               [ "type" .= ("boolean" :: Text)
  315               , "description" .= ("Include archived memories (list, search; default: false)." :: Text)
  316               ]
  317           ]
  318       , "required" .= (["action"] :: [Text])
  319       ]
  320   , toOutSchema = object []
  321   , toAuthorize = authorizeManage
  322   , toBlocking = False
  323   , toRun = \_ v ->
  324       case parseMemoryAction v of
  325         Left e -> pure (OpResult [TrpText e] True (object []))
  326         Right action -> case action of
  327           MemWrite   -> handleWrite store embedding v
  328           MemRead    -> handleRead store v
  329           MemList    -> handleList store v
  330           MemSearch  -> handleSearch embedding store v
  331           MemArchive -> handleArchive store embedding v
  332   }

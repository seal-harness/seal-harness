    2 -- | The Memory opcode group: MEMORY_WRITE, MEMORY_RECALL, MEMORY_DELETE.
    3 -- All Audited — the dispatcher writes both the session transcript and the
    4 -- Audited log; the opcodes mutate the in-memory/SQLite/Markdown backend
    5 -- (the materialized view). 'orRecorded' carries the secret-free 'MemoryId'
    6 -- + op name + @was_new@ flag (so the audit log distinguishes create vs
    7 -- update); the memory CONTENT is agent-visible data (not a vault secret)
    8 -- and is recorded in full in both logs.
    9 --
   10 -- 'MEMORY_WRITE' is an upsert: if the memory already exists, its content
   11 -- and/or tags are updated (the original 'meSession' provenance and
   12 -- 'meCreatedAt' are preserved; only 'meUpdatedAt' is bumped); if not, a
   13 -- fresh entry is created. This merges the former MEMORY_STORE +
   14 -- MEMORY_UPDATE into a single opcode, eliminating one failure path.
   15 --
   16 -- 'MEMORY_RECALL' uses the dynamic-retrieval pager ('Seal.Core.Paging'):
   17 -- @page_size = clamp floor ceiling (round (coeff * sqrt total))@, where total
   18 -- is the number of matching memories.
   19 module Seal.ISA.Ops.Memory
   20   ( memoryWriteOp
   21   , memoryRecallOp
   22   , memoryDeleteOp
   23   ) where

   42 -- | Build a JSON-Schema object with a single required string property.
   43 singleStringSchema :: Text -> Text -> Value
   44 singleStringSchema fieldName fieldDesc =
   45   object
   46     [ "type" .= ("object" :: Text)
   47     , "properties" .= object
   48         [fromText fieldName .= object
   49            [ "type" .= ("string" :: Text)
   50            , "description" .= fieldDesc
   51            ]]
   52     , "required" .= ([fieldName] :: [Text])
   53     ]
   54 
   55 -- | Extract the @id@ string field from a JSON object.
   56 idField :: Value -> Maybe Text
   57 idField = parseMaybe (withObject "in" (.: "id"))
   58 
   59 -- | Extract the @content@ string field (defaults to empty when absent).
   60 contentField :: Value -> Text
   61 contentField v = fromMaybe "" (parseMaybe (withObject "in" (.: "content")) v)
   62 
   63 -- | Extract the optional @tags@ array field (defaults to [] when absent).
   64 tagsField :: Value -> [Text]
   65 tagsField v =
   66   case parseMaybe (withObject "in" (.:? "tags")) v :: Maybe (Maybe [Text]) of
   67     Just (Just ts) -> ts
   68     _              -> []
   69 
   70 -- | Extract the optional @query@ string field for RECALL (substring filter).
   71 queryField :: Value -> Maybe Text
   72 queryField v =
   73   case parseMaybe (withObject "in" (.:? "query")) v :: Maybe (Maybe Text) of
   74     Just (Just q) -> Just q
   75     _             -> Nothing
   76 
   77 -- | Extract the optional @limit@ integer field for RECALL.
   78 limitField :: Value -> Maybe Int
   79 limitField v =
   80   case parseMaybe (withObject "in" (.:? "limit")) v :: Maybe (Maybe Int) of
   81     Just (Just n) | n >= 0 -> Just n
   82     _                      -> Nothing
   83 
   84 -- | Extract the optional @offset@ integer field for RECALL (defaults to 0).
   85 offsetField :: Value -> Int
   86 offsetField v =
   87   case parseMaybe (withObject "in" (.:? "offset")) v :: Maybe (Maybe Int) of
   88     Just (Just n) | n >= 0 -> n
   89     _                      -> 0

   91 -- | MEMORY_WRITE: upsert a memory by id. If the memory already exists, its
   92 -- content and/or tags are updated (the original 'meSession' provenance and
   93 -- 'meCreatedAt' are preserved; only 'meUpdatedAt' is bumped); if not, a
   94 -- fresh entry is created with the current session as provenance. The
   95 -- content and tags are recorded in full (agent-visible data); 'orRecorded'
   96 -- carries the id + op name + content + tags + @was_new@ (secret-free).
   97 memoryWriteOp :: MemoryBackend -> SessionId -> Opcode
   98 memoryWriteOp backend session = TrustedOpcode
   99   { toName = OpName "MEMORY_WRITE"
  100   , toTrust = Trusted
  101   , toDesc = "Create or update an agent memory by id (upsert; preserves provenance on update)."
  102   , toInSchema = object
  103       [ "type" .= ("object" :: Text)
  104       , "properties" .= object
  105           [ fromText "id" .= object
  106               [ "type" .= ("string" :: Text)
  107               , "description" .= ("Memory id ([A-Za-z0-9_-]+)." :: Text)
  108               ]
  109           , fromText "content" .= object
  110               [ "type" .= ("string" :: Text)
  111               , "description" .= ("The memory content (agent-visible)." :: Text)
  112               ]
  113           , fromText "tags" .= object
  114               [ "type" .= ("array" :: Text)
  115               , "description" .= ("Optional tags." :: Text)
  116               ]
  117           ]
  118       , "required" .= (["id", "content"] :: [Text])
  119       ]
  120   , toOutSchema = object []
  121   , toAuthorize = maybe (Left "MEMORY_WRITE requires {id:string}") checkId . idField
  122   , toBlocking = False
  123   , toRun = \_ v -> do
  124       let mId = idField v >>= either (const Nothing) Just . mkMemoryId
  125       case mId of
  126         Nothing -> pure (OpResult [TrpText "invalid memory id"] True (object []))
  127         Just mid -> do
  128           mExisting <- liftIO (mbRecall backend mid)
  129           now <- liftIO getCurrentTime
  130           let (entry, wasNew) = case mExisting of
  131                 Just existing ->
  132                   ( existing
  133                       { meContent = contentField v
  134                       , meTags = tagsField v
  135                       , meUpdatedAt = now
  136                       }
  137                   , False
  138                   )
  139                 Nothing ->
  140                   ( MemoryEntry
  141                       { meId = mid
  142                       , meContent = contentField v
  143                       , meTags = tagsField v
  144                       , meCreatedAt = now
  145                       , meUpdatedAt = now
  146                       , meSession = session
  147                       }
  148                   , True
  149                   )
  150           liftIO (mbStore backend entry)
  151           let recorded = object
  152                 [ "id" .= memoryIdText mid
  153                 , "content" .= meContent entry
  154                 , "tags" .= meTags entry
  155                 , "created_at" .= meCreatedAt entry
  156                 , "updated_at" .= meUpdatedAt entry
  157                 , "session" .= meSession entry
  158                 , "was_new" .= wasNew
  159                 ]
  160           pure (OpResult [TrpText (if wasNew then "stored" else "updated")] False recorded)
  161   }
  162   where
  163     checkId t = either (Left . ("invalid memory id: " <>)) (const (Right ())) (mkMemoryId t)

  165 -- | MEMORY_RECALL: return a paged window of memories, optionally filtered by a
  166 -- substring query. Uses the dynamic-retrieval pager.
  167 memoryRecallOp :: PageParams -> MemoryBackend -> Opcode
  168 memoryRecallOp params backend = TrustedOpcode
  169   { toName = OpName "MEMORY_RECALL"
  170   , toTrust = Trusted
  171   , toDesc = "Recall agent memories, optionally filtered by a substring query."
  172   , toInSchema = object
  173       [ "type" .= ("object" :: Text)
  174       , "properties" .= object
  175           [ fromText "query" .= object
  176               [ "type" .= ("string" :: Text)
  177               , "description" .= ("Optional substring filter over content+tags." :: Text)
  178               ]
  179           , fromText "limit" .= object
  180               [ "type" .= ("integer" :: Text)
  181               , "description" .= ("Max items to return (clamped to the pager ceiling)." :: Text)
  182               ]
  183           , fromText "offset" .= object
  184               [ "type" .= ("integer" :: Text)
  185               , "description" .= ("0-based offset (defaults to 0)." :: Text)
  186               ]
  187           ]
  188       ]
  189   , toOutSchema = object []
  190   , toAuthorize = const (Right ())
  191   , toBlocking = False
  192   , toRun = \_ v -> do
  193       let mQuery = queryField v
  194           offset = offsetField v
  195           mLimit = limitField v
  196       allEntries <- liftIO (mbList backend)
  197       let filtered = filter (matches mQuery) allEntries
  198           page = paginate params offset mLimit filtered
  199           rendered = T.intercalate "\n" (map renderEntry (pgItems page))
  200                     <> "\n---\npage " <> T.pack (show (pgOffset page))
  201                     <> " of " <> T.pack (show (pgTotal page))
  202                     <> if pgHasMore page then " (has more)" else " (end)"
  203       let recorded = object
  204             [ "query" .= mQuery
  205             , "offset" .= offset
  206             , "limit" .= mLimit
  207             , "total" .= pgTotal page
  208             ]
  209       pure (OpResult [TrpText rendered] False recorded)
  210   }
  211   where
  212     matches mQuery e =
  213       case mQuery of
  214         Nothing -> True
  215         Just q  -> q `T.isInfixOf` meContent e || any (q `T.isInfixOf`) (meTags e)
  216     renderEntry e =
  217       memoryIdText (meId e) <> ": " <> meContent e

  219 -- | MEMORY_DELETE: remove a memory by id. Idempotent (deleting a missing id is
  220 -- a success with a "not present" message, not an error).
  221 memoryDeleteOp :: MemoryBackend -> Opcode
  222 memoryDeleteOp backend = TrustedOpcode
  223   { toName = OpName "MEMORY_DELETE"
  224   , toTrust = Trusted
  225   , toDesc = "Delete an agent memory by id (idempotent)."
  226   , toInSchema = singleStringSchema "id" "The memory id to delete."
  227   , toOutSchema = object []
  228   , toAuthorize = maybe (Left "MEMORY_DELETE requires {id:string}") checkId . idField
  229   , toBlocking = False
  230   , toRun = \_ v -> do
  231       let mId = idField v >>= either (const Nothing) Just . mkMemoryId
  232       case mId of
  233         Nothing -> pure (OpResult [TrpText "invalid memory id"] True (object []))
  234         Just mid -> do
  235           mExisting <- liftIO (mbRecall backend mid)
  236           liftIO (mbDelete backend mid)
  237           let msg = case mExisting of
  238                 Nothing -> "deleted (was not present)"
  239                 Just _  -> "deleted"
  240               recorded = object ["id" .= memoryIdText mid]
  241           pure (OpResult [TrpText msg] False recorded)
  242   }
  243   where
  244     checkId t = either (Left . ("invalid memory id: " <>)) (const (Right ())) (mkMemoryId t)

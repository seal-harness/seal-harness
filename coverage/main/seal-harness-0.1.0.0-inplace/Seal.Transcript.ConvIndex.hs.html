    2 -- | The @conversation.idx@ binary index: a file of contiguous
    3 -- little-endian 'Word64' values storing the byte offset of the start
    4 -- of each line in @conversation.jsonl@. N lines → N+1 offsets
    5 -- (offset[0] = 0, offset[N] = fileSize). Enables random-access reads
    6 -- of individual conversation lines without loading the full file into
    7 -- memory — preventing OOM on multi-GB transcripts.
    8 --
    9 -- The index is built alongside @conversation.jsonl@ by the
   10 -- 'Seal.Handles.Transcript.withIndexedTranscript' writer (append-only,
   11 -- fsync'd). For pre-existing sessions without an index, 'ensureIndex'
   12 -- builds it in a one-time full-file scan. Crash recovery (index
   13 -- shorter than conversation) appends only the missing tail.

   49 -- | Error ADT for 'ensureIndex'. The program pattern-matches on this
   50 -- to drive control flow: 'IndexMissing' → build from scratch;
   51 -- 'IndexStale' → tail-recovery; 'IndexCorrupt' → return Left (rebuild
   52 -- deferred to next startup).
   53 data ConvIndexError
   54   = IndexMissing
   55   | IndexStale
   56   | IndexCorrupt
   57   deriving stock (Eq, Show)

   60 word64Size :: Int

   63 -- | Read lines [start, end) from conversation.jsonl using the index.
   64 -- Returns 'Left' on: missing files, corrupt index, out-of-bounds offsets.
   65 -- Clamps @end@ to 'convLineCount' if it exceeds the available lines.
   66 -- Returns @Right []@ when @start >= end@ or @start >= lineCount@.
   67 readConvLines :: FilePath -> FilePath -> Int -> Int -> IO (Either Text [Message])
   68 readConvLines convPath idxPath start end
   69   | start >= end = pure (Right [])
   70   | otherwise = do
   71       convExists <- doesFileExist convPath
   72       idxExists  <- doesFileExist idxPath
   73       if not convExists
   74         then pure (Left "conversation.jsonl not found")
   75         else if not idxExists
   76           then pure (Left "conversation.idx not found")
   77           else do
   78             lc <- convLineCount idxPath
   79             if lc <= 0 || start >= lc
   80               then pure (Right [])
   81               else do
   82                 let end' = min end lc
   83                 eOffsets <- readOffsets idxPath start end'
   84                 case eOffsets of
   85                   Left e -> pure (Left e)
   86                   Right (offStart, offEnd) -> do
   87                     convSize <- withBinaryFile convPath ReadMode hFileSize
   88                     if offStart > offEnd || offEnd > fromIntegral (convSize :: Integer)
   89                       then pure (Left "index offset out of bounds")
   90                       else do
   91                         raw <- readByteRange convPath (fromIntegral offStart) (fromIntegral offEnd)
   92                         let lns = filter (not . BS.null) (BS.split 0x0a raw)
   93                             msgs = mapMaybe (A.decode . BL.fromStrict) lns
   94                         pure (Right msgs)

  128 -- | A pair of open handles for reading conversation + index files.
  129 -- Used by 'withConvIndex' to avoid per-entry open/close overhead.
  130 data ConvIndexHandles = ConvIndexHandles
  131   { cihConv :: Handle
  132   , cihIdx  :: Handle
  133   }

  135 -- | Open both conversation.jsonl and conversation.idx for reading, run the
  136 -- action with the handles, then close both. Use 'readConvLinesWith' inside
  137 -- the bracket to read lines without per-call open/close.
  138 withConvIndex :: FilePath -> FilePath -> (ConvIndexHandles -> IO a) -> IO a
  139 withConvIndex convPath idxPath action =
  140   withBinaryFile convPath ReadMode $ \convH ->
  141     withBinaryFile idxPath ReadMode $ \idxH ->
  142       action (ConvIndexHandles convH idxH)

  144 -- | Read lines [start, end) using pre-opened handles (no per-call open/close).
  145 -- Returns 'Left' on out-of-bounds offsets. Clamps @end@ to available lines.
  146 readConvLinesWith :: ConvIndexHandles -> Int -> Int -> Int -> IO (Either Text [Message])
  147 readConvLinesWith hs lineCount start end
  148   | start >= end = pure (Right [])
  149   | lineCount <= 0 || start >= lineCount = pure (Right [])
  150   | otherwise = do
  151       let end' = min end lineCount
  152       offStart <- readWord64At (cihIdx hs) start
  153       offEnd   <- readWord64At (cihIdx hs) end'
  154       convSize <- hFileSize (cihConv hs)
  155       if offStart > offEnd || offEnd > fromIntegral (convSize :: Integer)
  156         then pure (Left "index offset out of bounds")
  157         else do
  158           let len = fromIntegral (offEnd - offStart) :: Int
  159           hSeek (cihConv hs) AbsoluteSeek (fromIntegral offStart)
  160           allocaBytes len $ \ptr -> do
  161             _ <- hGetBuf (cihConv hs) ptr len
  162             bs <- BS.packCStringLen (castPtr ptr, len)
  163             let lns = filter (not . BS.null) (BS.split 0x0a bs)
  164                 msgs = mapMaybe (A.decode . BL.fromStrict) lns
  165             pure (Right msgs)

  167 -- | Read the byte offsets for lines [start, end) from the index.
  168 -- Returns (offset[start], offset[end]) — the byte range to read from
  169 -- conversation.jsonl.
  170 readOffsets :: FilePath -> Int -> Int -> IO (Either Text (Word64, Word64))
  171 readOffsets idxPath start end =
  172   withBinaryFile idxPath ReadMode $ \h -> do
  173     offStart <- readWord64At h start
  174     offEnd   <- readWord64At h end
  175     pure (Right (offStart, offEnd))
  176 
  177 -- | Read a single Word64 LE value at the given line index from an open handle.
  178 readWord64At :: Handle -> Int -> IO Word64
  179 readWord64At h lineIdx = do
  180   hSeek h AbsoluteSeek (fromIntegral (lineIdx * word64Size))
  181   allocaBytes word64Size $ \ptr -> do
  182     _ <- hGetBuf h ptr word64Size
  183     bs <- BS.packCStringLen (castPtr ptr, word64Size)
  184     pure (decodeWord64LE bs)
  185 
  186 -- | Read a byte range [start, end) from a file.
  187 readByteRange :: FilePath -> Int -> Int -> IO BS.ByteString
  188 readByteRange path startOff endOff =
  189   withBinaryFile path ReadMode $ \h -> do
  190     hSeek h AbsoluteSeek (fromIntegral startOff)
  191     let len = endOff - startOff
  192     allocaBytes len $ \ptr -> do
  193       _ <- hGetBuf h ptr len
  194       BS.packCStringLen (castPtr ptr, len)

  196 -- | Decode a little-endian Word64 from an 8-byte ByteString.
  197 decodeWord64LE :: BS.ByteString -> Word64
  198 decodeWord64LE bs =
  199   let w0 = fromIntegral (BS.index bs 0)
  200       w1 = fromIntegral (BS.index bs 1)
  201       w2 = fromIntegral (BS.index bs 2)
  202       w3 = fromIntegral (BS.index bs 3)
  203       w4 = fromIntegral (BS.index bs 4)
  204       w5 = fromIntegral (BS.index bs 5)
  205       w6 = fromIntegral (BS.index bs 6)
  206       w7 = fromIntegral (BS.index bs 7)
  207   in w0 .|. (w1 `shiftL` 8) .|. (w2 `shiftL` 16) .|. (w3 `shiftL` 24)
  208    .|. (w4 `shiftL` 32) .|. (w5 `shiftL` 40) .|. (w6 `shiftL` 48) .|. (w7 `shiftL` 56)

  210 -- | Total number of conversation lines (index entry count - 1).
  211 -- Returns 0 if the index file is missing or empty.
  212 convLineCount :: FilePath -> IO Int
  213 convLineCount idxPath = do
  214   exists <- doesFileExist idxPath
  215   if not exists
  216     then pure 0
  217     else do
  218       size <- fromIntegral <$> withBinaryFile idxPath ReadMode hFileSize
  219       let entryCount = size `div` word64Size
  220       pure (max 0 (entryCount - 1))

  222 -- | Build conversation.idx from an existing conversation.jsonl.
  223 -- One pass: scan the file, record byte offsets of each newline.
  224 -- Uses temp-file + atomic rename: writes to conversation.idx.tmp,
  225 -- fsyncs, renames to conversation.idx. Crash-safe.
  226 buildIndex :: FilePath -> FilePath -> IO (Either Text ())
  227 buildIndex convPath idxPath = do
  228   convExists <- doesFileExist convPath
  229   if not convExists
  230     then pure (Left "conversation.jsonl not found")
  231     else do
  232       offsets <- scanLineOffsets convPath
  233       let tmpPath = idxPath <> ".tmp"
  234       writeIndexFile tmpPath offsets
  235       renameFile tmpPath idxPath
  236       pure (Right ())

  238 -- | Scan a file and record the byte offset of the start of each line.
  239 -- Returns a list of Word64 offsets: [0, offset[1], offset[2], ..., fileSize].
  240 -- For N lines, returns N+1 offsets.
  241 scanLineOffsets :: FilePath -> IO [Word64]
  242 scanLineOffsets path = do
  243   withBinaryFile path ReadMode $ \h -> do
  244     size <- hFileSize h
  245     let fileSize = fromIntegral size :: Word64
  246     scan h 0 [0] fileSize
  247   where
  248     chunkSize :: Int
  249     chunkSize = 65536
  250     scan h offset acc fileSize = do
  251       allocaBytes chunkSize $ \ptr -> do
  252         n <- hGetBuf h ptr chunkSize
  253         if n == 0
  254           then pure (reverse (finalizeIdx fileSize acc))
  255           else do
  256             bs <- BS.packCStringLen (castPtr ptr, n)
  257             let nlIdxs = BS.elemIndices 10 bs
  258                 lineStarts = map (\i -> offset + fromIntegral i + 1) nlIdxs
  259                 newOffset = offset + fromIntegral n
  260             scan h newOffset (reverse lineStarts ++ acc) fileSize
  261 
  262 -- | Add the file size as the final offset, avoiding a duplicate when the
  263 -- file ends with a newline (the last line-start already equals fileSize).
  264 finalizeIdx :: Word64 -> [Word64] -> [Word64]
  265 finalizeIdx fileSize acc = case acc of
  266   (lastOff : _) | lastOff == fileSize -> acc
  267   _ -> fileSize : acc

  269 -- | Write a list of Word64 offsets to an index file (little-endian).
  270 writeIndexFile :: FilePath -> [Word64] -> IO ()
  271 writeIndexFile path offsets = do
  272   let flags = defaultFileFlags { append = False, creat = Just (0o600 :: FileMode) }
  273   fd <- openFd path WriteOnly flags
  274   mapM_ (writeIdxEntry fd) offsets
  275   closeFd fd

  277 -- | Result of validating an existing index file.
  278 data IndexStatus
  279   = IndexCurrent  -- ^ Index is up-to-date (lastOffset == convSize)
  280   | IndexBehind   -- ^ Index is behind (lastOffset < convSize) — tail-recovery needed
  281   deriving stock (Eq, Show)
  282 
  283 -- | Ensure the index exists and is up to date. Builds from scratch
  284 -- if missing; recovers the tail if stale; returns Left IndexCorrupt
  285 -- if structurally invalid (rebuild deferred to next startup — runtime
  286 -- rebuild would invalidate the daemon's long-lived fd via atomic rename).
  287 -- Idempotent and safe to call concurrently (MVar serializes all index writers).
  288 ensureIndex :: FilePath -> FilePath -> IO (Either ConvIndexError ())
  289 ensureIndex convPath idxPath = do
  290   convExists <- doesFileExist convPath
  291   idxExists  <- doesFileExist idxPath
  292   if not convExists
  293     then -- No conversation file; remove stale orphan index if present
  294       if idxExists
  295         then do removeFile idxPath; pure (Right ())
  296         else pure (Right ())
  297     else if not idxExists
  298       then buildIndex convPath idxPath >>= \case
  299         Left _ -> pure (Left IndexCorrupt)
  300         Right _ -> pure (Right ())
  301       else do
  302         -- Index exists; validate it
  303         eValid <- validateIndex convPath idxPath
  304         case eValid of
  305           Left _ -> pure (Left IndexCorrupt)
  306           Right IndexBehind -> do
  307             eRecover <- recoverTail convPath idxPath
  308             case eRecover of
  309               Left _ -> pure (Left IndexCorrupt)
  310               Right _ -> pure (Right ())
  311           Right IndexCurrent -> pure (Right ())

  313 -- | Validate the index file structurally. Returns:
  314 -- 'Right IndexCurrent' if the index is up-to-date.
  315 -- 'Right IndexBehind' if the index is behind (lastOffset < convSize).
  316 -- 'Left IndexCorrupt' if the index is structurally invalid.
  317 --
  318 -- O(1): reads only the first, middle, and last offset — not all N offsets.
  319 -- The previous O(N) validation (reading every offset and checking
  320 -- monotonicity) took 3.7s on a 724K-line index. The spot-check catches
  321 -- the common corruption cases (truncated file, wrong file, zeroed first
  322 -- offset) without the per-offset seek cost.
  323 validateIndex :: FilePath -> FilePath -> IO (Either ConvIndexError IndexStatus)
  324 validateIndex convPath idxPath = do
  325   idxSize <- withBinaryFile idxPath ReadMode hFileSize
  326   let entryCount = fromIntegral idxSize `div` word64Size
  327   if entryCount <= 1
  328     then pure (Left IndexCorrupt)  -- need ≥2 entries: offset[0]=0, offset[N]=fileSize
  329     else do
  330       (firstOff, midOff, lastOff) <- withBinaryFile idxPath ReadMode $ \h -> do
  331         f <- readWord64At h 0
  332         m <- readWord64At h (entryCount `div` 2)
  333         l <- readWord64At h (entryCount - 1)
  334         pure (f, m, l)
  335       convSize <- withBinaryFile convPath ReadMode hFileSize
  336       let corrupt = firstOff /= 0
  337                  || not (firstOff < midOff && midOff <= lastOff)
  338                  || fromIntegral lastOff > convSize
  339       if corrupt
  340         then pure (Left IndexCorrupt)
  341         else if fromIntegral lastOff < convSize
  342           then pure (Right IndexBehind)
  343           else pure (Right IndexCurrent)

  345 -- | Read all Word64 offsets from the index file.
  346 readAllOffsets :: FilePath -> IO [Word64]
  347 readAllOffsets idxPath =
  348   withBinaryFile idxPath ReadMode $ \h -> do
  349     size <- hFileSize h
  350     let count = fromIntegral size `div` word64Size
  351     if count == 0
  352       then pure []
  353       else do
  354         hSeek h AbsoluteSeek 0
  355         readNWord64s h count

  357 -- | Read N Word64 values sequentially from an open handle.
  358 readNWord64s :: Handle -> Int -> IO [Word64]
  359 readNWord64s h count = go 0 []
  360   where
  361     go i acc
  362       | i >= count = pure (reverse acc)
  363       | otherwise = do
  364           w <- readWord64At h i
  365           go (i + 1) (w : acc)

  367 -- | Recover the index tail: scan conversation.jsonl from the last index
  368 -- offset to EOF and append the missing offsets.
  369 recoverTail :: FilePath -> FilePath -> IO (Either Text ())
  370 recoverTail convPath idxPath = do
  371   offsets <- readAllOffsets idxPath
  372   case reverse offsets of
  373     [] -> pure (Left "cannot recover: index is empty")
  374     (lastOffW : _) -> do
  375       let lastOff = fromIntegral lastOffW :: Int
  376       convSize <- fromIntegral <$> withBinaryFile convPath ReadMode hFileSize
  377       let tailBytes = convSize - lastOff
  378       if tailBytes <= 0
  379         then pure (Right ())
  380         else do
  381           raw <- readByteRange convPath lastOff convSize
  382           let lns = BS.split 0x0a raw
  383               chunks = if not (null lns) && BS.null (last lns) then init lns else lns
  384               relOffsets = scanl (\off chunk -> off + fromIntegral (BS.length chunk) + 1) (fromIntegral lastOff) chunks
  385               newOffsets = drop 1 relOffsets
  386           let flags = defaultFileFlags { append = True, creat = Just (0o600 :: FileMode) }
  387           fd <- openFd idxPath WriteOnly flags
  388           mapM_ (writeIdxEntry fd) newOffsets
  389           closeFd fd
  390           pure (Right ())

  392 -- | Append a Word64 LE offset to the index file via the given fd.
  393 -- Used by the writer daemon and 'appendConversationMessage'.
  394 writeIdxEntry :: Fd -> Word64 -> IO ()
  395 writeIdxEntry fd w =
  396   let bs = BL.toStrict (toLazyByteString (word64LE w))
  397   in BSU.unsafeUseAsCStringLen bs $ \(ptr, len) ->
  398        writeBuf fd (castPtr ptr) len
  399 
  400 -- | Write a buffer to an fd, looping on short writes.
  401 writeBuf :: Fd -> Ptr a -> Int -> IO ()
  402 writeBuf fd ptr len = go 0
  403   where
  404     go written
  405       | written >= len = pure ()
  406       | otherwise = do
  407           n <- fromIntegral <$> fdWriteBuf fd (plusPtr ptr written) (fromIntegral (len - written))
  408           go (written + n)

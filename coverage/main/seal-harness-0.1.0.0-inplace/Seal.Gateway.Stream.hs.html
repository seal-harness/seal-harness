    2 -- | The WebSocket stream endpoint. Runs on a separate port from the REST +
    3 -- static WARP server (the @wai-app-websockets@ bridge isn't available; a
    4 -- separate port is the clean 7a path). Every connection is gated by the
    5 -- Origin allowlist + the broker's global cap. The wire protocol: on
    6 -- connect, a one-shot @hello@ (carrying @protocolVersion@ +
    7 -- @serverStartedAt@ so the client can detect server restarts); then a
    8 -- reader/writer race forwards 'BrokerEvent's from the broker to the WS
    9 -- peer while accepting @focus@ ops from the client.
   10 --
   11 -- When a @focus@ op carries a @since@ field (an entry id), the server
   12 -- replays all transcript entries after that id from disk, sends them as
   13 -- @entry@ events, then sends a @replay-end@ event so the client transitions
   14 -- from @replaying@ back to @live@. This closes the gap: entries broadcast
   15 -- while the WS was disconnected are no longer permanently lost.

   18   , StreamGuard (..)

   61 -- | The per-connection guard: the Origin allowlist + the global cap.
   62 -- Also carries the TabsHandle + SealPaths so the stream can send an
   63 -- initial @lists@ snapshot on connect (the frontend's @wsListsReceived@
   64 -- flag requires at least one @lists@ frame to switch from polling to WS).
   65 data StreamGuard = StreamGuard
   66   { sgAllowedOrigins :: [Text]
   67   , sgGlobalCap :: Int
   68   , sgTabsHandle :: TabsHandle
   69   , sgPaths :: SealPaths
   70   }

   72 -- | Run the WebSocket stream server on the given port. Blocks (run in a
   73 -- forked thread from @seal serve@).
   74 runStreamServer :: Text -> Int -> StreamGuard -> StreamBroker -> IO ()
   75 runStreamServer host port guard broker =
   76   WS.runServer (T.unpack host) port (streamApp guard broker)

   78 -- | The per-connection WS app: check the Origin, accept, send hello, then
   79 -- race the broker-forwarder against the focus-reader.
   80 streamApp :: StreamGuard -> StreamBroker -> PendingConnection -> IO ()
   81 streamApp guard broker pending = do
   82   let reqHead = WS.pendingRequest pending
   83       origin = lookupHeader "origin" reqHead
   84       allowed = map T.unpack (sgAllowedOrigins guard)
   85   case origin of
   86     Nothing                   -> acceptConn  -- no Origin header (local dev client); accept
   87     Just _  | null allowed    -> acceptConn  -- wildcard mode (host=0.0.0.0); accept any
   88     Just o | o `elem` allowed -> acceptConn
   89     Just o                    -> globalLogIO InfoS ("ws: rejected Origin " <> ls (T.pack (show o)))
   90   where
   91     acceptConn = do
   92       conn <- acceptRequest pending
   93       startedAt <- getCurrentTime
   94       sendTextData conn (A.encode (object
   95         [ "type" .= ("hello" :: Text)
   96         , "protocolVersion" .= ("v1" :: Text)
   97         , "serverStartedAt" .= T.pack (showIso startedAt)
   98         ]))
   99       let sendEvent (BeEntryRecorded sid v) =
  100             sendTextData conn (A.encode (object
  101               [ "type" .= ("entry" :: Text)
  102               , "sessionId" .= sessionIdText sid
  103               , "entry" .= v
  104               ]))
  105           sendEvent (BeEntryUpdate sid v) =
  106             sendTextData conn (A.encode (object
  107               [ "type" .= ("entry-update" :: Text)
  108               , "sessionId" .= sessionIdText sid
  109               , "entry" .= v
  110               ]))
  111           sendEvent (BeHarnessStatus v)    = sendTextData conn (A.encode v)
  112           sendEvent (BeListsSnapshot v)    = sendTextData conn (A.encode v)
  113           sendEvent (BeActivity sid v)     =
  114             sendTextData conn (A.encode (object
  115               [ "type" .= ("activity" :: Text)
  116               , "sessionId" .= sessionIdText sid
  117               , "activity" .= v
  118               ]))
  119           sendEvent (BeAsk sid v)          =
  120             sendTextData conn (A.encode (object
  121               [ "type" .= ("ask" :: Text)
  122               , "sessionId" .= sessionIdText sid
  123               , "ask" .= v
  124               ]))
  125           sendEvent (BeAskResolved sid v)  =
  126             sendTextData conn (A.encode (object
  127               [ "type" .= ("ask_resolved" :: Text)
  128               , "sessionId" .= sessionIdText sid
  129               , "ask" .= v
  130               ]))
  131           sendEvent BeAgentDefsChanged =
  132             sendTextData conn (A.encode (object
  133               [ "type" .= ("agent-defs-changed" :: Text)
  134               ]))
  135           sendEvent BeSkillsChanged =
  136             sendTextData conn (A.encode (object
  137               [ "type" .= ("skills-changed" :: Text)
  138               ]))
  139           sendEvent BeReposChanged =
  140             sendTextData conn (A.encode (object
  141               [ "type" .= ("repos-changed" :: Text)
  142               ]))
  143       let defaultSid = case mkSessionId "default" of Right s -> s; Left _ -> error "sid"
  144           closeConn = sendClose conn ("subscriber evicted" :: Text)
  145       subSessionRef <- subscribe broker defaultSid sendEvent closeConn
  146       globalLogIO InfoS "[ws] connected, initial focus=default"
  147       -- Send an initial lists snapshot AFTER subscribing so this connection
  148       -- is in the broker's subscriber list and actually receives the frame.
  149       -- Without this, the snapshot is broadcast to zero subscribers, the
  150       -- frontend never flips wsLive, and REST /api/lists polling stays on.
  151       broadcastListsSnapshot broker (sgTabsHandle guard) (sgPaths guard)
  152       withPingThread conn 30 (pure ()) $ do
  153         let readerLoop = forever $ do
  154               msg <- receiveData conn
  155               case A.decode msg :: Maybe ClientMessage of
  156                 Just (CmFocus focusOp) ->
  157                   case mkSessionId (foSession focusOp) of
  158                     Right s  -> do
  159                       globalLogIO InfoS ("[ws] focus → " <> ls (sessionIdText s))
  160                       updateSubscriberSession subSessionRef s
  161                       -- When the client sends a `since` entry id, replay
  162                       -- all transcript entries after that id from disk so
  163                       -- entries broadcast during the WS gap are recovered.
  164                       forM_ (foSince focusOp) $ \sinceId ->
  165                         replayEntriesSince conn (sgPaths guard) s sinceId
  166                     Left _e  -> sendErrorFrame conn "invalid session id"
  167                 Just (CmRequestEntries reqOp) ->
  168                   handleRequestEntries conn (sgPaths guard) reqOp
  169                 Nothing -> sendErrorFrame conn "unknown op"
  170         readerLoop `catch` \(_e :: SomeException) -> pure ()
  171 
  172 -- | Replay transcript entries after the given entry id, then send a
  173 -- @replay-end@ event so the client transitions from @replaying@ to @live@.
  174 -- Reads the full transcript from disk and filters to entries whose id is
  175 -- strictly after @sinceId@ (by position — the entry id is compared as a
  176 -- string, and entries with ids lexicographically greater than @sinceId@
  177 -- are sent). The frontend dedupes by id so any overlap with entries the
  178 -- client already has is harmless. Errors during replay are logged and
  179 -- swallowed (the client will still receive @replay-end@ and transition to
  180 -- @live@, falling back to the HTTP seed for any missing entries).
  181 --
  182 -- Entry ids are either the on-disk @teId@ (a Text) or a synthetic index
  183 -- (@\"0\"@, @\"1\"@, ...) when the indexed transcript format has no per-entry id. We
  184 -- compare by the @id@ field in the frontend JSON shape, which is the same
  185 -- shape @readTranscriptEntries@ returns.
  186 replayEntriesSince
  187   :: Connection -> SealPaths -> SessionId -> Text -> IO ()
  188 replayEntriesSince conn paths sid sinceId = do
  189   let go = do
  190         mMeta <- loadSessionMeta paths sid
  191         let model = maybe "" smModel mMeta
  192             fallbackTs = maybe "" (showIso . smCreatedAt) mMeta
  193         -- Read only the last 200 entries (limit) to avoid flooding the
  194         -- Read only the last 50 entries (limit) to avoid flooding the
  195         -- frontend with thousands of WS entry events on large sessions.
  196         -- The frontend dedupes by id, and the HTTP seed already provided
  197         -- the initial page. The replay only needs to catch up entries
  198         -- that arrived since the last visit.
  199         putStrLn "[ws] replayEntriesSince: calling readTranscriptEntriesTimed (ReadLastN 50)"
  200         (entries, _tt) <- readTranscriptEntriesTimed paths model fallbackTs sid (ReadLastN 50)
  201         let after = filterAfterId sinceId entries
  202         forM_ after $ \entry -> do
  203           sendTextData conn (A.encode (object
  204             [ "type" .= ("entry" :: Text)
  205             , "sessionId" .= sessionIdText sid
  206             , "entry" .= entry
  207             ]))
  208         -- Send replay-end with the last replayed entry id (or null when
  209         -- no entries were replayed — the client keeps its current
  210         -- lastEntryId in that case).
  211         let mLastId = case reverse after of
  212               [] -> Nothing
  213               (last_ : _) -> Just (extractId last_)
  214         sendTextData conn (A.encode (object
  215           [ "type" .= ("replay-end" :: Text)
  216           , "sessionId" .= sessionIdText sid
  217           , "lastReplayedEntryId" .= mLastId
  218           ]))
  219   go `catch` \(e :: SomeException) -> do
  220     globalLogIO InfoS ("[ws] replay error: " <> ls (T.pack (show e)))
  221     -- Even on error, send replay-end so the client isn't stuck in
  222     -- replaying status. lastReplayedEntryId is null (no confirmed replay).
  223     sendTextData conn (A.encode (object
  224       [ "type" .= ("replay-end" :: Text)
  225       , "sessionId" .= sessionIdText sid
  226       , "lastReplayedEntryId" .= (Nothing :: Maybe Text)
  227       ]))

  229 -- | Filter the frontend-shaped transcript entries to those whose @id@
  230 -- field is strictly after @sinceId@. Entry ids from the indexed transcript format
  231 -- are synthetic line indices (@\"0\"@, @\"1\"@, ...) which sort
  232 -- lexicographically the same as numerically for single-digit counts but
  233 -- diverge for multi-digit (e.g. @\"10\"@ < @\"2\"@ lexically). To be
  234 -- robust, we find the position of @sinceId@ in the list and take everything
  235 -- after it; if not found, we fall back to sending all entries (the
  236 -- frontend dedupes by id, so overlap is harmless — and this handles the
  237 -- case where the client's @lastEntryId@ is from a different transcript
  238 -- shape or the session was rebuilt).
  239 filterAfterId :: Text -> [A.Value] -> [A.Value]
  240 filterAfterId sinceId entries =
  241   fromMaybe entries (breakOnId sinceId entries)
  242 
  243 -- | Find the position of the entry whose @id@ matches @sinceId@ and
  244 -- return everything after it. Returns 'Nothing' when the id is not found.
  245 breakOnId :: Text -> [A.Value] -> Maybe [A.Value]
  246 breakOnId sinceId = go
  247   where
  248     go [] = Nothing
  249     go (v : vs) =
  250       if extractId v == sinceId
  251         then Just vs
  252         else go vs

  254 -- | Extract the @id@ field from a frontend-shaped transcript entry JSON.
  255 extractId :: A.Value -> Text
  256 extractId v = case v of
  257   A.Object o -> case KeyMap.lookup (Key.fromText "id") o of
  258     Just (A.String t) -> t
  259     _                 -> ""
  260   _ -> ""

  262 -- | Handle a @request-entries@ op: read the transcript from disk, slice it
  263 -- to the requested chunk, and send an @entries-chunk@ event. Includes a
  264 -- catch handler so a corrupt/missing transcript sends an error frame
  265 -- instead of propagating to the readerLoop's top-level catch and
  266 -- disconnecting the entire WS connection.
  267 handleRequestEntries :: Connection -> SealPaths -> RequestEntriesOp -> IO ()
  268 handleRequestEntries conn paths (RequestEntriesOp sidTxt mBefore mLimit) =
  269   case mkSessionId sidTxt of
  270     Left _ -> sendErrorFrame conn "invalid session id"
  271     Right sid -> do
  272       let go = do
  273             mMeta <- loadSessionMeta paths sid
  274             let model = maybe "" smModel mMeta
  275                 fallbackTs = maybe "" (showIso . smCreatedAt) mMeta
  276                 limit = clampLimit mLimit
  277             putStrLn $ "[ws] handleRequestEntries: limit=" <> show limit <> " mBefore=" <> show mBefore
  278             -- Use the paginated read to avoid loading all entries into
  279             -- memory. For the initial chunk (mBefore == Nothing), read
  280             -- only the last `limit` entries. For before/id pagination,
  281             -- read a wider window. For __beginning__, we need the FIRST
  282             -- entries — use ReadFirstN to read only the first `limit`
  283             -- entries instead of loading the entire transcript.
  284             (allEntries, tt) <- readTranscriptEntriesTimed paths model fallbackTs sid
  285               (case mBefore of
  286                 Nothing              -> ReadLastN limit
  287                 Just "__beginning__" -> ReadFirstN limit
  288                 Just _               -> ReadLastN (limit * 3))
  289             -- totalCount comes from the timings (full entry count before
  290             -- limiting), NOT from length allEntries (which is the limited count).
  291             let totalCount = ttEntryCount tt
  292                 (chunk, hasMore) = case mBefore of
  293                   Nothing ->
  294                     -- allEntries already contains only the last `limit`
  295                     -- entries; the chunk is all of them, hasMore = totalCount > limit
  296                     (allEntries, totalCount > limit)
  297                   Just before ->
  298                    if before == "__beginning__"
  299                      -- allEntries already contains only the first `limit`
  300                      -- entries (ReadFirstN); no take needed.
  301                      then (allEntries, totalCount > limit)
  302                      else
  303                     case entriesBeforeId before allEntries of
  304                       Just beforeEntries ->
  305                         let taken = drop (max 0 (length beforeEntries - limit)) beforeEntries
  306                             hasMoreBefore = length beforeEntries > limit
  307                         in (taken, hasMoreBefore)
  308                       Nothing -> ([], False)
  309             sendTextData conn (A.encode (A.object
  310               [ "type" A..= ("entries-chunk" :: Text)
  311               , "sessionId" A..= sidTxt
  312               , "entries" A..= chunk
  313               , "hasMore" A..= hasMore
  314               , "totalCount" A..= totalCount
  315               , "requestBefore" A..= mBefore
  316               ]))
  317       go `catch` \(e :: SomeException) -> do
  318         globalLogIO InfoS ("[ws] request-entries error: " <> ls (T.pack (show e)))
  319         sendTextData conn (A.encode (A.object
  320           [ "type" A..= ("entries-chunk" :: Text)
  321           , "sessionId" A..= sidTxt
  322           , "entries" A..= ([] :: [A.Value])
  323           , "requestBefore" A..= mBefore
  324           ]))

  326 -- | Return entries before the entry with id @before@ (exclusive).
  327 -- Returns 'Nothing' when the id is not found (distinct from
  328 -- 'filterAfterId' which falls back to all entries on a miss).
  329 entriesBeforeId :: Text -> [A.Value] -> Maybe [A.Value]
  330 entriesBeforeId beforeId = go
  331   where
  332     go [] = Nothing
  333     go (v : vs) =
  334       if extractId v == beforeId
  335         then Just []
  336         else case go vs of
  337           Just rest -> Just (v : rest)
  338           Nothing   -> Nothing

  340 -- | Clamp the limit to [1, 200], defaulting to 50.
  341 clampLimit :: Maybe Int -> Int
  342 clampLimit = min 200 . max 1 . fromMaybe 50

  344 -- | Send an error frame to the WS peer. Extracted from the inline
  345 -- pattern in 'readerLoop' for reuse.
  346 sendErrorFrame :: Connection -> Text -> IO ()
  347 sendErrorFrame conn msg = sendTextData conn (A.encode (object
  348   [ "type" .= ("error" :: Text), "message" .= msg ]))

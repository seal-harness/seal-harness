    2 -- | The WebSocket stream endpoint. Runs on a separate port from the REST +
    3 -- static WARP server (the @wai-app-websockets@ bridge isn't available; a
    4 -- separate port is the clean 7a path). Every connection is gated by the
    5 -- Origin allowlist + the broker's global cap. The wire protocol: on
    6 -- connect, a one-shot @hello@ (carrying @protocolVersion@ +
    7 -- @serverStartedAt@ so the client can detect server restarts); then a
    8 -- reader/writer race forwards 'BrokerEvent's from the broker to the WS
    9 -- peer while accepting @focus@ ops from the client.
   10 --
   11 -- When a @focus@ op carries a @since@ field (an entry id), the server
   12 -- replays all transcript entries after that id from disk, sends them as
   13 -- @entry@ events, then sends a @replay-end@ event so the client transitions
   14 -- from @replaying@ back to @live@. This closes the gap: entries broadcast
   15 -- while the WS was disconnected are no longer permanently lost.
   16 module Seal.Gateway.Stream
   17   ( runStreamServer
   18   , StreamGuard (..)
   19   , FocusOp (..)
   20   , filterAfterId
   21   , extractId
   22   ) where

   54 -- | The per-connection guard: the Origin allowlist + the global cap.
   55 -- Also carries the TabsHandle + SealPaths so the stream can send an
   56 -- initial @lists@ snapshot on connect (the frontend's @wsListsReceived@
   57 -- flag requires at least one @lists@ frame to switch from polling to WS).
   58 data StreamGuard = StreamGuard
   59   { sgAllowedOrigins :: [Text]
   60   , sgGlobalCap :: Int
   61   , sgTabsHandle :: TabsHandle
   62   , sgPaths :: SealPaths
   63   }

   65 -- | Run the WebSocket stream server on the given port. Blocks (run in a
   66 -- forked thread from @seal serve@).
   67 runStreamServer :: Text -> Int -> StreamGuard -> StreamBroker -> IO ()
   68 runStreamServer host port guard broker =
   69   WS.runServer (T.unpack host) port (streamApp guard broker)

   71 -- | The per-connection WS app: check the Origin, accept, send hello, then
   72 -- race the broker-forwarder against the focus-reader.
   73 streamApp :: StreamGuard -> StreamBroker -> PendingConnection -> IO ()
   74 streamApp guard broker pending = do
   75   let reqHead = WS.pendingRequest pending
   76       origin = lookupHeader "origin" reqHead
   77       allowed = map T.unpack (sgAllowedOrigins guard)
   78   case origin of
   79     Nothing                   -> acceptConn  -- no Origin header (local dev client); accept
   80     Just _  | null allowed    -> acceptConn  -- wildcard mode (host=0.0.0.0); accept any
   81     Just o | o `elem` allowed -> acceptConn
   82     Just o                    -> globalLogIO InfoS ("ws: rejected Origin " <> ls (T.pack (show o)))
   83   where
   84     acceptConn = do
   85       conn <- acceptRequest pending
   86       startedAt <- getCurrentTime
   87       sendTextData conn (A.encode (object
   88         [ "type" .= ("hello" :: Text)
   89         , "protocolVersion" .= ("v1" :: Text)
   90         , "serverStartedAt" .= T.pack (showIso startedAt)
   91         ]))
   92       let sendEvent (BeEntryRecorded sid v) =
   93             sendTextData conn (A.encode (object
   94               [ "type" .= ("entry" :: Text)
   95               , "sessionId" .= sessionIdText sid
   96               , "entry" .= v
   97               ]))
   98           sendEvent (BeEntryUpdate sid v) =
   99             sendTextData conn (A.encode (object
  100               [ "type" .= ("entry-update" :: Text)
  101               , "sessionId" .= sessionIdText sid
  102               , "entry" .= v
  103               ]))
  104           sendEvent (BeHarnessStatus v)    = sendTextData conn (A.encode v)
  105           sendEvent (BeListsSnapshot v)    = sendTextData conn (A.encode v)
  106           sendEvent (BeActivity sid v)     =
  107             sendTextData conn (A.encode (object
  108               [ "type" .= ("activity" :: Text)
  109               , "sessionId" .= sessionIdText sid
  110               , "activity" .= v
  111               ]))
  112           sendEvent (BeAsk sid v)          =
  113             sendTextData conn (A.encode (object
  114               [ "type" .= ("ask" :: Text)
  115               , "sessionId" .= sessionIdText sid
  116               , "ask" .= v
  117               ]))
  118           sendEvent (BeAskResolved sid v)  =
  119             sendTextData conn (A.encode (object
  120               [ "type" .= ("ask_resolved" :: Text)
  121               , "sessionId" .= sessionIdText sid
  122               , "ask" .= v
  123               ]))
  124           sendEvent BeAgentDefsChanged =
  125             sendTextData conn (A.encode (object
  126               [ "type" .= ("agent-defs-changed" :: Text)
  127               ]))
  128           sendEvent BeSkillsChanged =
  129             sendTextData conn (A.encode (object
  130               [ "type" .= ("skills-changed" :: Text)
  131               ]))
  132           sendEvent BeReposChanged =
  133             sendTextData conn (A.encode (object
  134               [ "type" .= ("repos-changed" :: Text)
  135               ]))
  136       let defaultSid = case mkSessionId "default" of Right s -> s; Left _ -> error "sid"
  137           closeConn = sendClose conn ("subscriber evicted" :: Text)
  138       subSessionRef <- subscribe broker defaultSid sendEvent closeConn
  139       globalLogIO InfoS "[ws] connected, initial focus=default"
  140       -- Send an initial lists snapshot AFTER subscribing so this connection
  141       -- is in the broker's subscriber list and actually receives the frame.
  142       -- Without this, the snapshot is broadcast to zero subscribers, the
  143       -- frontend never flips wsLive, and REST /api/lists polling stays on.
  144       broadcastListsSnapshot broker (sgTabsHandle guard) (sgPaths guard)
  145       withPingThread conn 30 (pure ()) $ do
  146         let readerLoop = forever $ do
  147               msg <- receiveData conn
  148               case A.decode msg of
  149                 Just (focusOp :: FocusOp) ->
  150                   case mkSessionId (foSession focusOp) of
  151                     Right s  -> do
  152                       globalLogIO InfoS ("[ws] focus → " <> ls (sessionIdText s))
  153                       updateSubscriberSession subSessionRef s
  154                       -- When the client sends a `since` entry id, replay
  155                       -- all transcript entries after that id from disk so
  156                       -- entries broadcast during the WS gap are recovered.
  157                       forM_ (foSince focusOp) $ \sinceId ->
  158                         replayEntriesSince conn (sgPaths guard) s sinceId
  159                     Left _e  -> sendTextData conn (A.encode (object ["type" .= ("error" :: Text), "message" .= ("invalid session id" :: Text)]))
  160                 Nothing -> sendTextData conn (A.encode (object ["type" .= ("error" :: Text), "message" .= ("expected a focus op" :: Text)]))
  161         readerLoop `catch` \(_e :: SomeException) -> pure ()
  162 
  163 -- | Replay transcript entries after the given entry id, then send a
  164 -- @replay-end@ event so the client transitions from @replaying@ to @live@.
  165 -- Reads the full transcript from disk and filters to entries whose id is
  166 -- strictly after @sinceId@ (by position — the entry id is compared as a
  167 -- string, and entries with ids lexicographically greater than @sinceId@
  168 -- are sent). The frontend dedupes by id so any overlap with entries the
  169 -- client already has is harmless. Errors during replay are logged and
  170 -- swallowed (the client will still receive @replay-end@ and transition to
  171 -- @live@, falling back to the HTTP seed for any missing entries).
  172 --
  173 -- Entry ids are either the on-disk @teId@ (a Text) or a synthetic index
  174 -- (@\"0\"@, @\"1\"@, ...) when the two-file format has no per-entry id. We
  175 -- compare by the @id@ field in the frontend JSON shape, which is the same
  176 -- shape @readTranscriptEntries@ returns.
  177 replayEntriesSince
  178   :: Connection -> SealPaths -> SessionId -> Text -> IO ()
  179 replayEntriesSince conn paths sid sinceId = do
  180   let go = do
  181         mMeta <- loadSessionMeta paths sid
  182         let model = maybe "" smModel mMeta
  183             fallbackTs = maybe "" (showIso . smCreatedAt) mMeta
  184         entries <- readTranscriptEntries paths model fallbackTs sid
  185         let after = filterAfterId sinceId entries
  186         forM_ after $ \entry -> do
  187           sendTextData conn (A.encode (object
  188             [ "type" .= ("entry" :: Text)
  189             , "sessionId" .= sessionIdText sid
  190             , "entry" .= entry
  191             ]))
  192         -- Send replay-end with the last replayed entry id (or null when
  193         -- no entries were replayed — the client keeps its current
  194         -- lastEntryId in that case).
  195         let mLastId = case reverse after of
  196               [] -> Nothing
  197               (last_ : _) -> Just (extractId last_)
  198         sendTextData conn (A.encode (object
  199           [ "type" .= ("replay-end" :: Text)
  200           , "sessionId" .= sessionIdText sid
  201           , "lastReplayedEntryId" .= mLastId
  202           ]))
  203   go `catch` \(e :: SomeException) -> do
  204     globalLogIO InfoS ("[ws] replay error: " <> ls (T.pack (show e)))
  205     -- Even on error, send replay-end so the client isn't stuck in
  206     -- replaying status. lastReplayedEntryId is null (no confirmed replay).
  207     sendTextData conn (A.encode (object
  208       [ "type" .= ("replay-end" :: Text)
  209       , "sessionId" .= sessionIdText sid
  210       , "lastReplayedEntryId" .= (Nothing :: Maybe Text)
  211       ]))

  213 -- | Filter the frontend-shaped transcript entries to those whose @id@
  214 -- field is strictly after @sinceId@. Entry ids from the two-file format
  215 -- are synthetic line indices (@\"0\"@, @\"1\"@, ...) which sort
  216 -- lexicographically the same as numerically for single-digit counts but
  217 -- diverge for multi-digit (e.g. @\"10\"@ < @\"2\"@ lexically). To be
  218 -- robust, we find the position of @sinceId@ in the list and take everything
  219 -- after it; if not found, we fall back to sending all entries (the
  220 -- frontend dedupes by id, so overlap is harmless — and this handles the
  221 -- case where the client's @lastEntryId@ is from a different transcript
  222 -- shape or the session was rebuilt).
  223 filterAfterId :: Text -> [A.Value] -> [A.Value]
  224 filterAfterId sinceId entries =
  225   fromMaybe entries (breakOnId sinceId entries)
  226 
  227 -- | Find the position of the entry whose @id@ matches @sinceId@ and
  228 -- return everything after it. Returns 'Nothing' when the id is not found.
  229 breakOnId :: Text -> [A.Value] -> Maybe [A.Value]
  230 breakOnId sinceId = go
  231   where
  232     go [] = Nothing
  233     go (v : vs) =
  234       if extractId v == sinceId
  235         then Just vs
  236         else go vs
  237 
  238 -- | Extract the @id@ field from a frontend-shaped transcript entry JSON.
  239 extractId :: A.Value -> Text
  240 extractId v = case v of
  241   A.Object o -> case KeyMap.lookup (Key.fromText "id") o of
  242     Just (A.String t) -> t
  243     _                 -> ""
  244   _ -> ""

  246 -- | The focus op the client sends to change its focused session. Accepts
  247 -- both the frontend's shape (@{"op":"focus","sessionId":"..."}@) and the
  248 -- legacy shape (@{"session":"..."}@) for robustness. The optional @since@
  249 -- field requests replay of entries after the given entry id.
  250 data FocusOp = FocusOp
  251   { foSession :: Text
  252   , foSince   :: Maybe Text
  253   }
  254   deriving stock (Eq, Show)
  255 
  256 instance A.FromJSON FocusOp where
  257   parseJSON = A.withObject "focus" $ \o ->
  258     FocusOp
  259       <$> (o .: "sessionId" <|> o .: "session")
  260       <*> (o .:? "since")

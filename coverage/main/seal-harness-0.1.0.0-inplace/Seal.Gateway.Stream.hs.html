    2 -- | The WebSocket stream endpoint. Mounted INSIDE the REST + static WARP
    3 -- server, so the stream shares the gateway's port. Every connection is
    4 -- gated by the
    5 -- Origin allowlist + the broker's global cap. The wire protocol: on
    6 -- connect, a one-shot @hello@ (carrying @protocolVersion@ +
    7 -- @serverStartedAt@ so the client can detect server restarts); then a
    8 -- reader/writer race forwards 'BrokerEvent's from the broker to the WS
    9 -- peer while accepting @focus@ ops from the client.
   10 --
   11 -- When a @focus@ op carries a @since@ field (an entry id), the server
   12 -- replays all transcript entries after that id from disk, sends them as
   13 -- @entry@ events, then sends a @replay-end@ event so the client transitions
   14 -- from @replaying@ back to @live@. This closes the gap: entries broadcast
   15 -- while the WS was disconnected are no longer permanently lost.
   16 --
   17 -- Every connection also carries two keep-alives at the same interval: a
   18 -- protocol-level PING (invisible to browser JavaScript) and an
   19 -- application-level @heartbeat@ data frame (see 'heartbeatLoop').

   73 -- | The liveness of the WebSocket stream server, as reported by
   74 -- @\/api\/health@. The health endpoint used to report the CONFIGURED
   75 -- @ws_port@ unconditionally, so a server whose WS had failed to bind
   76 -- (port already in use) still advertised the dead port and the frontend
   77 -- connected to a non-WebSocket port and froze. The status is now derived
   78 -- from what actually happened at bind time.
   79 data StreamHealth = StreamHealth
   80   { shStatus :: Text
   81     -- ^ @\"starting\"@ (bind not yet attempted), @\"up\"@ (bound and
   82     -- accepting), @\"failed\"@ (the bind failed).
   83   , shPort  :: Int
   84     -- ^ The port the server was asked to bind.
   85   } deriving stock (Eq, Show)
   86 
   87 -- | The status string of a 'StreamHealth', as surfaced in @\/api\/health@.
   88 streamHealthStatus :: StreamHealth -> Text
   89 streamHealthStatus = shStatus

   91 -- | A human-readable, actionable description of a 'StreamHealth'. For a
   92 -- failed bind this names the port so the operator can see WHICH port
   93 -- collided — the whole point of reporting the failure.
   94 streamHealthMessage :: StreamHealth -> Text
   95 streamHealthMessage h =
   96   case shStatus h of
   97     "up"      -> "WebSocket stream server listening on port " <> tshow (shPort h)
   98     "failed"  -> "WebSocket stream server FAILED to bind port " <> tshow (shPort h)
   99                   <> " (port already in use?) — live transcript updates are unavailable"
  100     "down"    -> "WebSocket stream server is not running"
  101     _         -> "WebSocket stream server is starting on port " <> tshow (shPort h)
  102   where
  103     tshow :: Int -> Text
  104     tshow = T.pack . show

  106 -- | The per-connection guard: the Origin allowlist + the global cap.
  107 -- Also carries the TabsHandle + SealPaths so the stream can send an
  108 -- initial @lists@ snapshot on connect (the frontend's @wsListsReceived@
  109 -- flag requires at least one @lists@ frame to switch from polling to WS).
  110 data StreamGuard = StreamGuard
  111   { sgAllowedOrigins :: [Text]
  112   , sgGlobalCap :: Int
  113   , sgTabsHandle :: TabsHandle
  114   , sgPaths :: SealPaths
  115   -- | Interval in SECONDS for the connection keep-alives: the protocol-level
  116   -- PING ('withPingThread') and the application-level @{"type":"heartbeat"}@
  117   -- data frame. The protocol ping keeps intermediaries and dead-peer
  118   -- detection honest; the data frame is the only one a browser's JavaScript
  119   -- can observe (see 'heartbeatLoop'). Production uses 30s — comfortably
  120   -- inside the frontend's 90s watchdog. Tests override it to keep the suite
  121   -- fast, or raise it to keep heartbeat frames out of unrelated assertions.
  122   , sgHeartbeatSeconds :: Int
  123   }

  125 -- | The WebSocket stream endpoint, as a WAI middleware.
  126 --
  127 -- The WS server is mounted INSIDE the HTTP server, so the stream shares
  128 -- the gateway's port. This removes the second listener entirely: the old
  129 -- two-port split existed only because @wai-app-websockets@ (now
  130 -- @wai-websockets@) was missing from the nix snapshot when the split was
  131 -- chosen in 2026-07. With the bridge available, one port means one bind,
  132 -- one thing to go wrong -- and no chance of the WS silently dying while
  133 -- HTTP kept serving.
  134 --
  135 -- Non-WebSocket requests fall through to the supplied backup application,
  136 -- so this can wrap the whole gateway app.
  137 streamMiddleware
  138   :: StreamGuard
  139   -> StreamBroker
  140   -> Application
  141   -> Application
  142 streamMiddleware guard broker =
  143   websocketsOr defaultConnectionOptions (streamApp guard broker)

  145 -- | The per-connection WS app: check the Origin, accept, send hello, then
  146 -- race the broker-forwarder against the focus-reader.
  147 streamApp :: StreamGuard -> StreamBroker -> PendingConnection -> IO ()
  148 streamApp guard broker pending = do
  149   let reqHead = WS.pendingRequest pending
  150       origin = lookupHeader "origin" reqHead
  151       allowed = map T.unpack (sgAllowedOrigins guard)
  152   case origin of
  153     Nothing                   -> acceptConn  -- no Origin header (local dev client); accept
  154     Just _  | null allowed    -> acceptConn  -- wildcard mode (host=0.0.0.0); accept any
  155     Just o | o `elem` allowed -> acceptConn
  156     Just o                    -> globalLogIO InfoS ("ws: rejected Origin " <> ls (T.pack (show o)))
  157   where
  158     acceptConn = do
  159       conn <- acceptRequest pending
  160       startedAt <- getCurrentTime
  161       sendTextData conn (A.encode (object
  162         [ "type" .= ("hello" :: Text)
  163         , "protocolVersion" .= ("v1" :: Text)
  164         , "serverStartedAt" .= T.pack (showIso startedAt)
  165         ]))
  166       let sendEvent (BeEntryRecorded sid v) =
  167             sendTextData conn (A.encode (object
  168               [ "type" .= ("entry" :: Text)
  169               , "sessionId" .= sessionIdText sid
  170               , "entry" .= v
  171               ]))
  172           sendEvent (BeEntryUpdate sid v) =
  173             sendTextData conn (A.encode (object
  174               [ "type" .= ("entry-update" :: Text)
  175               , "sessionId" .= sessionIdText sid
  176               , "entry" .= v
  177               ]))
  178           sendEvent (BeHarnessStatus v)    = sendTextData conn (A.encode v)
  179           sendEvent (BeListsSnapshot v)    = sendTextData conn (A.encode v)
  180           sendEvent (BeActivity sid v)     =
  181             sendTextData conn (A.encode (object
  182               [ "type" .= ("activity" :: Text)
  183               , "sessionId" .= sessionIdText sid
  184               , "activity" .= v
  185               ]))
  186           sendEvent (BeAsk sid v)          =
  187             sendTextData conn (A.encode (object
  188               [ "type" .= ("ask" :: Text)
  189               , "sessionId" .= sessionIdText sid
  190               , "ask" .= v
  191               ]))
  192           sendEvent (BeAskResolved sid v)  =
  193             sendTextData conn (A.encode (object
  194               [ "type" .= ("ask_resolved" :: Text)
  195               , "sessionId" .= sessionIdText sid
  196               , "ask" .= v
  197               ]))
  198           sendEvent BeAgentDefsChanged =
  199             sendTextData conn (A.encode (object
  200               [ "type" .= ("agent-defs-changed" :: Text)
  201               ]))
  202           sendEvent BeSkillsChanged =
  203             sendTextData conn (A.encode (object
  204               [ "type" .= ("skills-changed" :: Text)
  205               ]))
  206           sendEvent BeReposChanged =
  207             sendTextData conn (A.encode (object
  208               [ "type" .= ("repos-changed" :: Text)
  209               ]))
  210       let defaultSid = case mkSessionId "default" of Right s -> s; Left _ -> error "sid"
  211           closeConn = sendClose conn ("subscriber evicted" :: Text)
  212       subSessionRef <- subscribe broker defaultSid sendEvent closeConn
  213       globalLogIO InfoS "[ws] connected, initial focus=default"
  214       -- Send an initial lists snapshot AFTER subscribing so this connection
  215       -- is in the broker's subscriber list and actually receives the frame.
  216       -- Without this, the snapshot is broadcast to zero subscribers, the
  217       -- frontend never flips wsLive, and REST /api/lists polling stays on.
  218       broadcastListsSnapshot broker (sgTabsHandle guard) (sgPaths guard)
  219       -- Keep-alives, at one interval: the protocol-level PING ('withPingThread')
  220       -- so intermediaries and dead-peer detection stay honest, and the
  221       -- application-level heartbeat frame ('heartbeatLoop') because that is
  222       -- the only one a browser's JavaScript can observe. The heartbeat is
  223       -- cancelled as soon as the reader loop ends (peer closed), so it never
  224       -- outlives the connection.
  225       let hbSeconds = sgHeartbeatSeconds guard
  226       withPingThread conn hbSeconds (pure ()) $ do
  227         let readerLoop = forever $ do
  228               msg <- receiveData conn
  229               case A.decode msg :: Maybe ClientMessage of
  230                 Just (CmFocus focusOp) ->
  231                   case mkSessionId (foSession focusOp) of
  232                     Right s  -> do
  233                       globalLogIO InfoS ("[ws] focus → " <> ls (sessionIdText s))
  234                       updateSubscriberSession subSessionRef s
  235                       -- When the client sends a `since` entry id, replay
  236                       -- all transcript entries after that id from disk so
  237                       -- entries broadcast during the WS gap are recovered.
  238                       forM_ (foSince focusOp) $ \sinceId ->
  239                         replayEntriesSince conn (sgPaths guard) s sinceId
  240                     Left _e  -> sendErrorFrame conn "invalid session id"
  241                 Just (CmRequestEntries reqOp) ->
  242                   handleRequestEntries conn (sgPaths guard) reqOp
  243                 Nothing -> sendErrorFrame conn "unknown op"
  244         concurrently_
  245           (heartbeatLoop conn hbSeconds)
  246           (readerLoop `catch` \(_e :: SomeException) -> pure ())

  248 -- | Push a content-free @{"type":"heartbeat"}@ frame every
  249 -- @intervalSeconds@ for as long as the connection lives.
  250 --
  251 -- WebSocket PING frames ('withPingThread') are answered below the browser's
  252 -- JavaScript API: `WebSocket.onmessage` never fires for them. A client that
  253 -- watches data frames — as ours does — therefore cannot distinguish a
  254 -- healthy idle connection from a dead one, and tears it down once its
  255 -- watchdog expires. On the web that showed up as a periodic redraw of the
  256 -- transcript view: the "reconnecting" banner appearing, the chat
  257 -- re-rendering, and the sidebar refreshing, roughly every 90 seconds of
  258 -- idleness. This frame is the keep-alive the browser can actually observe.
  259 -- It carries no state, so clients must not re-render on it.
  260 --
  261 -- A failed send (peer went away) is swallowed: the reader loop notices the
  262 -- closed connection promptly and cancels this thread.
  263 heartbeatLoop :: Connection -> Int -> IO ()
  264 heartbeatLoop conn intervalSeconds = forever $ do
  265   threadDelay (max 1 intervalSeconds * 1_000_000)
  266   sendTextData conn (A.encode (object [ "type" .= ("heartbeat" :: Text) ]))
  267     `catch` \(_e :: SomeException) -> pure ()

  269 -- | Replay transcript entries after the given entry id, then send a
  270 -- @replay-end@ event so the client transitions from @replaying@ to @live@.
  271 -- Reads the full transcript from disk and filters to entries whose id is
  272 -- strictly after @sinceId@ (by position — the entry id is compared as a
  273 -- string, and entries with ids lexicographically greater than @sinceId@
  274 -- are sent). The frontend dedupes by id so any overlap with entries the
  275 -- client already has is harmless. Errors during replay are logged and
  276 -- swallowed (the client will still receive @replay-end@ and transition to
  277 -- @live@, falling back to the HTTP seed for any missing entries).
  278 --
  279 -- Entry ids are either the on-disk @teId@ (a Text) or a synthetic index
  280 -- (@\"0\"@, @\"1\"@, ...) when the indexed transcript format has no per-entry id. We
  281 -- compare by the @id@ field in the frontend JSON shape, which is the same
  282 -- shape @readTranscriptEntries@ returns.
  283 replayEntriesSince
  284   :: Connection -> SealPaths -> SessionId -> Text -> IO ()
  285 replayEntriesSince conn paths sid sinceId = do
  286   let go = do
  287         mMeta <- loadSessionMeta paths sid
  288         let model = maybe "" smModel mMeta
  289             fallbackTs = maybe "" (showIso . smCreatedAt) mMeta
  290         -- Read only the last 200 entries (limit) to avoid flooding the
  291         -- Read only the last 50 entries (limit) to avoid flooding the
  292         -- frontend with thousands of WS entry events on large sessions.
  293         -- The frontend dedupes by id, and the HTTP seed already provided
  294         -- the initial page. The replay only needs to catch up entries
  295         -- that arrived since the last visit.
  296         putStrLn "[ws] replayEntriesSince: calling readTranscriptEntriesTimed (ReadLastN 50)"
  297         (entries, _tt) <- readTranscriptEntriesTimed paths model fallbackTs sid (ReadLastN 50)
  298         let after = filterAfterId sinceId entries
  299         forM_ after $ \entry -> do
  300           sendTextData conn (A.encode (object
  301             [ "type" .= ("entry" :: Text)
  302             , "sessionId" .= sessionIdText sid
  303             , "entry" .= entry
  304             ]))
  305         -- Send replay-end with the last replayed entry id (or null when
  306         -- no entries were replayed — the client keeps its current
  307         -- lastEntryId in that case).
  308         let mLastId = case reverse after of
  309               [] -> Nothing
  310               (last_ : _) -> Just (extractId last_)
  311         sendTextData conn (A.encode (object
  312           [ "type" .= ("replay-end" :: Text)
  313           , "sessionId" .= sessionIdText sid
  314           , "lastReplayedEntryId" .= mLastId
  315           ]))
  316   go `catch` \(e :: SomeException) -> do
  317     globalLogIO InfoS ("[ws] replay error: " <> ls (T.pack (show e)))
  318     -- Even on error, send replay-end so the client isn't stuck in
  319     -- replaying status. lastReplayedEntryId is null (no confirmed replay).
  320     sendTextData conn (A.encode (object
  321       [ "type" .= ("replay-end" :: Text)
  322       , "sessionId" .= sessionIdText sid
  323       , "lastReplayedEntryId" .= (Nothing :: Maybe Text)
  324       ]))

  326 -- | Filter the frontend-shaped transcript entries to those whose @id@
  327 -- field is strictly after @sinceId@. Entry ids from the indexed transcript format
  328 -- are synthetic line indices (@\"0\"@, @\"1\"@, ...) which sort
  329 -- lexicographically the same as numerically for single-digit counts but
  330 -- diverge for multi-digit (e.g. @\"10\"@ < @\"2\"@ lexically). To be
  331 -- robust, we find the position of @sinceId@ in the list and take everything
  332 -- after it; if not found, we fall back to sending all entries (the
  333 -- frontend dedupes by id, so overlap is harmless — and this handles the
  334 -- case where the client's @lastEntryId@ is from a different transcript
  335 -- shape or the session was rebuilt).
  336 filterAfterId :: Text -> [A.Value] -> [A.Value]
  337 filterAfterId sinceId entries =
  338   fromMaybe entries (breakOnId sinceId entries)
  339 
  340 -- | Find the position of the entry whose @id@ matches @sinceId@ and
  341 -- return everything after it. Returns 'Nothing' when the id is not found.
  342 breakOnId :: Text -> [A.Value] -> Maybe [A.Value]
  343 breakOnId sinceId = go
  344   where
  345     go [] = Nothing
  346     go (v : vs) =
  347       if extractId v == sinceId
  348         then Just vs
  349         else go vs

  351 -- | Extract the @id@ field from a frontend-shaped transcript entry JSON.
  352 extractId :: A.Value -> Text
  353 extractId v = case v of
  354   A.Object o -> case KeyMap.lookup (Key.fromText "id") o of
  355     Just (A.String t) -> t
  356     _                 -> ""
  357   _ -> ""

  359 -- | Handle a @request-entries@ op: read the transcript from disk, slice it
  360 -- to the requested chunk, and send an @entries-chunk@ event. Includes a
  361 -- catch handler so a corrupt/missing transcript sends an error frame
  362 -- instead of propagating to the readerLoop's top-level catch and
  363 -- disconnecting the entire WS connection.
  364 handleRequestEntries :: Connection -> SealPaths -> RequestEntriesOp -> IO ()
  365 handleRequestEntries conn paths (RequestEntriesOp sidTxt mBefore mLimit) =
  366   case mkSessionId sidTxt of
  367     Left _ -> sendErrorFrame conn "invalid session id"
  368     Right sid -> do
  369       let go = do
  370             mMeta <- loadSessionMeta paths sid
  371             let model = maybe "" smModel mMeta
  372                 fallbackTs = maybe "" (showIso . smCreatedAt) mMeta
  373                 limit = clampLimit mLimit
  374             putStrLn $ "[ws] handleRequestEntries: limit=" <> show limit <> " mBefore=" <> show mBefore
  375             -- Use the paginated read to avoid loading all entries into
  376             -- memory. For the initial chunk (mBefore == Nothing), read
  377             -- only the last `limit` entries. For before/id pagination,
  378             -- read a wider window. For __beginning__, we need the FIRST
  379             -- entries — use ReadFirstN to read only the first `limit`
  380             -- entries instead of loading the entire transcript.
  381             (allEntries, tt) <- readTranscriptEntriesTimed paths model fallbackTs sid
  382               (case mBefore of
  383                 Nothing              -> ReadLastN limit
  384                 Just "__beginning__" -> ReadFirstN limit
  385                 Just _               -> ReadLastN (limit * 3))
  386             -- totalCount comes from the timings (full entry count before
  387             -- limiting), NOT from length allEntries (which is the limited count).
  388             let totalCount = ttEntryCount tt
  389                 (chunk, hasMore) = case mBefore of
  390                   Nothing ->
  391                     -- allEntries already contains only the last `limit`
  392                     -- entries; the chunk is all of them, hasMore = totalCount > limit
  393                     (allEntries, totalCount > limit)
  394                   Just before ->
  395                    if before == "__beginning__"
  396                      -- allEntries already contains only the first `limit`
  397                      -- entries (ReadFirstN); no take needed.
  398                      then (allEntries, totalCount > limit)
  399                      else
  400                     case entriesBeforeId before allEntries of
  401                       Just beforeEntries ->
  402                         let taken = drop (max 0 (length beforeEntries - limit)) beforeEntries
  403                             hasMoreBefore = length beforeEntries > limit
  404                         in (taken, hasMoreBefore)
  405                       Nothing -> ([], False)
  406             sendTextData conn (A.encode (A.object
  407               [ "type" A..= ("entries-chunk" :: Text)
  408               , "sessionId" A..= sidTxt
  409               , "entries" A..= chunk
  410               , "hasMore" A..= hasMore
  411               , "totalCount" A..= totalCount
  412               , "requestBefore" A..= mBefore
  413               ]))
  414       go `catch` \(e :: SomeException) -> do
  415         globalLogIO InfoS ("[ws] request-entries error: " <> ls (T.pack (show e)))
  416         sendTextData conn (A.encode (A.object
  417           [ "type" A..= ("entries-chunk" :: Text)
  418           , "sessionId" A..= sidTxt
  419           , "entries" A..= ([] :: [A.Value])
  420           , "requestBefore" A..= mBefore
  421           ]))

  423 -- | Return entries before the entry with id @before@ (exclusive).
  424 -- Returns 'Nothing' when the id is not found (distinct from
  425 -- 'filterAfterId' which falls back to all entries on a miss).
  426 entriesBeforeId :: Text -> [A.Value] -> Maybe [A.Value]
  427 entriesBeforeId beforeId = go
  428   where
  429     go [] = Nothing
  430     go (v : vs) =
  431       if extractId v == beforeId
  432         then Just []
  433         else case go vs of
  434           Just rest -> Just (v : rest)
  435           Nothing   -> Nothing

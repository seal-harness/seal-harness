    2 -- | The shared logger handle for Seal Harness. Wraps a katip 'LogEnv' and
    3 -- provides 'IO'-level logging functions ('logIO') so that channel, transport,
    4 -- and gateway code (which runs in 'IO', not 'App') can emit structured katip
    5 -- logs without a monad transformer stack.
    6 --
    7 -- The logger is built once at startup via 'withSealLogger' (a bracket that
    8 -- closes the scribe on exit) and threaded through the shared dependency
    9 -- records ('ChannelDeps', 'SendDeps') and 'Env' ('envLogger'). Turn code
   10 -- refines the logger via 'withChannelContext' to attach 'ChannelContext'
   11 -- (channel kind, conversation id hash, session id) so every 'logIO' call
   12 -- within a turn automatically carries the metadata — context-free logging
   13 -- is never the easy path within a turn.
   14 --
   15 -- 'logIO' is best-effort: if the scribe write throws (e.g. closed handle,
   16 -- I/O error), the error is silently swallowed. The logger must never crash
   17 -- the caller, especially inside 'withExceptionLogging' where a logger crash
   18 -- would defeat the exception handler. Newlines in the 'LogStr' are escaped
   19 -- to @\\n@ / @\\r@ before emission (log-injection defense).

   22   , withSealLogger
   23   , newSealLogger
   24   , newSealLoggerWithScribe

   49 -- | The shared logger handle. Built once at startup, threaded through
   50 -- 'ChannelDeps', 'SendDeps', and 'Env'. Carries its own 'LogEnv' + context
   51 -- + namespace so 'logIO' can emit from 'IO' without a monad stack.
   52 data SealLogger = SealLogger
   53   { slLogEnv    :: LogEnv
   54   , slContext   :: LogContexts
   55   , slNamespace :: Namespace
   56   }
   57 
   58 -- | Bracket the logger's lifetime. Creates the stderr scribe at the given
   59 -- severity level, and optionally a file scribe that mirrors all console
   60 -- output to a log file (e.g. @seal.log@ in the project directory). Runs the
   61 -- action, closes both scribes on exit. Used at the startup site in
   62 -- 'Seal.AppMain.dispatch' (which serves all commands: serve, tui, etc.).
   63 --
   64 -- The file scribe uses 'K.ColorLog False' (no ANSI escape codes) so the log file
   65 -- is plain text and greppable. The file is opened in 'AppendMode' so
   66 -- restarts don't clobber previous logs. If the file cannot be opened
   67 -- (permissions, disk full), a warning is printed to stderr and the logger
   68 -- proceeds with stderr-only output (best-effort — file logging is a
   69 -- diagnostic aid, never a hard requirement).
   70 withSealLogger :: Text -> Maybe FilePath -> (SealLogger -> IO a) -> IO a
   71 withSealLogger logLevel mLogFile = bracket makeLogEnv' closeLogEnv'
   72   where
   73     makeLogEnv' = do
   74       let sev = fromMaybe InfoS (textToSeverity logLevel)
   75       scribe <- mkHandleScribeWithFormatter bracketFormat
   76         K.ColorIfTerminal stderr (permitItem sev) K.V2
   77       le0 <- initLogEnv "seal-harness" "production"
   78       le1 <- registerScribe "stderr" scribe defaultScribeSettings le0
   79       le2 <- case mLogFile of
   80         Nothing -> pure le1
   81         Just path -> do
   82           eHandle <- try (openFile path AppendMode) :: IO (Either SomeException Handle)
   83           case eHandle of
   84             Left e -> do
   85               hPutStrLn stderr ("Warning: could not open log file " <> path <> ": " <> show e)
   86               pure le1
   87             Right h -> do
   88               hSetBuffering h LineBuffering
   89               fileScribe <- mkHandleScribeWithFormatter bracketFormat
   90                 (K.ColorLog False) h (permitItem sev) K.V2
   91               registerScribe "file" fileScribe defaultScribeSettings le1
   92       pure SealLogger
   93         { slLogEnv = le2
   94         , slContext = mempty
   95         , slNamespace = "seal-harness"
   96         }
   97     closeLogEnv' logger = void (closeScribes (slLogEnv logger))

   99 -- | Build a logger without a bracket (for tests). The caller is responsible
  100 -- for closing the scribe via 'closeSealLogger' (or process exit).
  101 newSealLogger :: Text -> IO SealLogger
  102 newSealLogger logLevel = do
  103   let sev = fromMaybe InfoS (textToSeverity logLevel)
  104   scribe <- mkHandleScribeWithFormatter bracketFormat
  105     K.ColorIfTerminal stderr (permitItem sev) K.V2
  106   le <- registerScribe "stderr" scribe defaultScribeSettings
  107     =<< initLogEnv "seal-harness" "production"
  108   pure SealLogger
  109     { slLogEnv = le
  110     , slContext = mempty
  111     , slNamespace = "seal-harness"
  112     }

  114 -- | Build a logger with a custom scribe (for tests that capture log output).
  115 -- The scribe is registered with the given permit severity.
  116 newSealLoggerWithScribe :: Scribe -> Severity -> IO SealLogger
  117 newSealLoggerWithScribe scribe _sev = do
  118   le <- registerScribe "capture" scribe defaultScribeSettings
  119     =<< initLogEnv "seal-harness" "test"
  120   pure SealLogger
  121     { slLogEnv = le
  122     , slContext = mempty
  123     , slNamespace = "seal-harness"
  124     }

  126 -- | Close the logger's scribes (for non-bracket construction).
  127 closeSealLogger :: SealLogger -> IO ()
  128 closeSealLogger logger = void (closeScribes (slLogEnv logger))

  130 -- | A logger with a no-op scribe (for tests that don't assert log output).
  131 -- Uses a scribe that accepts all items but does nothing with them.
  132 testSealLogger :: IO SealLogger
  133 testSealLogger = do
  134   let noopScribe = Scribe
  135         { liPush = \_ -> pure ()
  136         , scribePermitItem = permitItem DebugS
  137         , scribeFinalizer = pure ()
  138         }
  139   newSealLoggerWithScribe noopScribe DebugS

  141 -- | Emit a log line from 'IO'. Uses the logger's 'LogEnv' + context +
  142 -- namespace. Newlines in the 'LogStr' are escaped to @\\n@ / @\\r@ before
  143 -- emission (log-injection defense). If the scribe write throws, the error
  144 -- is silently swallowed (best-effort logging — the logger must never crash
  145 -- the caller).
  146 logIO :: SealLogger -> Severity -> LogStr -> IO ()
  147 logIO logger sev msg =
  148   runKatipContextT (slLogEnv logger) (slContext logger)
  149     (slNamespace logger) (logFM sev (escapeNewlines msg))
  150   `catch` \(_ :: SomeException) -> pure ()
  151 
  152 -- | Refine the logger with additional structured context. Returns a new
  153 -- 'SealLogger' whose 'slContext' is the merge
  154 -- @slContext <> liftPayload item@. This is how 'ChannelContext' is
  155 -- attached: the turn code calls @withChannelContext logger ctx@ once, then
  156 -- threads the refined logger through all subsequent 'logIO' calls.
  157 withChannelContext :: K.LogItem a => SealLogger -> a -> SealLogger
  158 withChannelContext logger item =
  159   logger { slContext = slContext logger <> liftPayload item }
  160 
  161 -- | Escape newlines in a 'LogStr' to prevent log injection. Replaces @\n@
  162 -- with @\\n@ and @\r@ with @\\r@ in the textual representation.
  163 escapeNewlines :: LogStr -> LogStr
  164 escapeNewlines = K.ls . escapeText . T.pack . show . unLogStr
  165   where
  166     escapeText :: Text -> Text
  167     escapeText = T.replace "\n" "\\n" . T.replace "\r" "\\r"

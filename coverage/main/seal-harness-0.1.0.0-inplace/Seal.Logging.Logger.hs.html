    2 -- | The shared logger handle for Seal Harness. Wraps a katip 'LogEnv' and
    3 -- provides 'IO'-level logging functions ('logIO') so that channel, transport,
    4 -- and gateway code (which runs in 'IO', not 'App') can emit structured katip
    5 -- logs without a monad transformer stack.
    6 --
    7 -- The logger is built once at startup via 'withSealLogger' (a bracket that
    8 -- closes the scribe on exit) and threaded through the shared dependency
    9 -- records ('ChannelDeps', 'SendDeps') and 'Env' ('envLogger'). Turn code
   10 -- refines the logger via 'withChannelContext' to attach 'ChannelContext'
   11 -- (channel kind, conversation id hash, session id) so every 'logIO' call
   12 -- within a turn automatically carries the metadata — context-free logging
   13 -- is never the easy path within a turn.
   14 --
   15 -- 'logIO' is best-effort: if the scribe write throws (e.g. closed handle,
   16 -- I/O error), the error is silently swallowed. The logger must never crash
   17 -- the caller, especially inside 'withExceptionLogging' where a logger crash
   18 -- would defeat the exception handler. Newlines in the 'LogStr' are escaped
   19 -- to @\\n@ / @\\r@ before emission (log-injection defense).

   21   ( SealLogger (..)
   22   , withSealLogger
   23   , newSealLogger
   24   , newSealLoggerWithScribe
   25   , closeSealLogger
   26   , testSealLogger
   27   , logIO
   28   , withChannelContext
   29   , escapeNewlines
   30   ) where

   49 -- | The shared logger handle. Built once at startup, threaded through
   50 -- 'ChannelDeps', 'SendDeps', and 'Env'. Carries its own 'LogEnv' + context
   51 -- + namespace so 'logIO' can emit from 'IO' without a monad stack.
   52 data SealLogger = SealLogger
   53   { slLogEnv    :: LogEnv
   54   , slContext   :: LogContexts
   55   , slNamespace :: Namespace
   56   }
   57 
   58 -- | Bracket the logger's lifetime. Creates the stderr scribe at the given
   59 -- severity level, runs the action, closes the scribe on exit. Used at all
   60 -- 4 startup sites ('runServeMain', 'runSignalMain', 'runTelegramMain',
   61 -- 'runTui').
   62 withSealLogger :: Text -> (SealLogger -> IO a) -> IO a
   63 withSealLogger logLevel = bracket makeLogEnv' closeLogEnv'
   64   where
   65     makeLogEnv' = do
   66       let sev = fromMaybe InfoS (textToSeverity logLevel)
   67       scribe <- mkHandleScribeWithFormatter bracketFormat
   68         K.ColorIfTerminal stderr (permitItem sev) K.V2
   69       le <- registerScribe "stderr" scribe defaultScribeSettings
   70         =<< initLogEnv "seal-harness" "production"
   71       pure SealLogger
   72         { slLogEnv = le
   73         , slContext = mempty
   74         , slNamespace = "seal-harness"
   75         }
   76     closeLogEnv' logger = void (closeScribes (slLogEnv logger))
   77 
   78 -- | Build a logger without a bracket (for tests). The caller is responsible
   79 -- for closing the scribe via 'closeSealLogger' (or process exit).
   80 newSealLogger :: Text -> IO SealLogger
   81 newSealLogger logLevel = do
   82   let sev = fromMaybe InfoS (textToSeverity logLevel)
   83   scribe <- mkHandleScribeWithFormatter bracketFormat
   84     K.ColorIfTerminal stderr (permitItem sev) K.V2
   85   le <- registerScribe "stderr" scribe defaultScribeSettings
   86     =<< initLogEnv "seal-harness" "production"
   87   pure SealLogger
   88     { slLogEnv = le
   89     , slContext = mempty
   90     , slNamespace = "seal-harness"
   91     }

   93 -- | Build a logger with a custom scribe (for tests that capture log output).
   94 -- The scribe is registered with the given permit severity.
   95 newSealLoggerWithScribe :: Scribe -> Severity -> IO SealLogger
   96 newSealLoggerWithScribe scribe _sev = do
   97   le <- registerScribe "capture" scribe defaultScribeSettings
   98     =<< initLogEnv "seal-harness" "test"
   99   pure SealLogger
  100     { slLogEnv = le
  101     , slContext = mempty
  102     , slNamespace = "seal-harness"
  103     }

  105 -- | Close the logger's scribes (for non-bracket construction).
  106 closeSealLogger :: SealLogger -> IO ()
  107 closeSealLogger logger = void (closeScribes (slLogEnv logger))

  109 -- | A logger with a no-op scribe (for tests that don't assert log output).
  110 -- Uses a scribe that accepts all items but does nothing with them.
  111 testSealLogger :: IO SealLogger
  112 testSealLogger = do
  113   let noopScribe = Scribe
  114         { liPush = \_ -> pure ()
  115         , scribePermitItem = permitItem DebugS
  116         , scribeFinalizer = pure ()
  117         }
  118   newSealLoggerWithScribe noopScribe DebugS

  120 -- | Emit a log line from 'IO'. Uses the logger's 'LogEnv' + context +
  121 -- namespace. Newlines in the 'LogStr' are escaped to @\\n@ / @\\r@ before
  122 -- emission (log-injection defense). If the scribe write throws, the error
  123 -- is silently swallowed (best-effort logging — the logger must never crash
  124 -- the caller).
  125 logIO :: SealLogger -> Severity -> LogStr -> IO ()
  126 logIO logger sev msg =
  127   runKatipContextT (slLogEnv logger) (slContext logger)
  128     (slNamespace logger) (logFM sev (escapeNewlines msg))
  129   `catch` \(_ :: SomeException) -> pure ()
  130 
  131 -- | Refine the logger with additional structured context. Returns a new
  132 -- 'SealLogger' whose 'slContext' is the merge
  133 -- @slContext <> liftPayload item@. This is how 'ChannelContext' is
  134 -- attached: the turn code calls @withChannelContext logger ctx@ once, then
  135 -- threads the refined logger through all subsequent 'logIO' calls.
  136 withChannelContext :: K.LogItem a => SealLogger -> a -> SealLogger
  137 withChannelContext logger item =
  138   logger { slContext = slContext logger <> liftPayload item }
  139 
  140 -- | Escape newlines in a 'LogStr' to prevent log injection. Replaces @\n@
  141 -- with @\\n@ and @\r@ with @\\r@ in the textual representation.
  142 escapeNewlines :: LogStr -> LogStr
  143 escapeNewlines = K.ls . escapeText . T.pack . show . unLogStr
  144   where
  145     escapeText :: Text -> Text
  146     escapeText = T.replace "\n" "\\n" . T.replace "\r" "\\r"

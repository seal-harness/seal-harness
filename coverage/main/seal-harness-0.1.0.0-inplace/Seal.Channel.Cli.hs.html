   91 -- | Map a 'Disposition' to its channel effect.
   92 --
   93 -- Extracted for testability: callers supply a 'ChannelCaps' and a handler for
   94 -- plain (agent-bound) text; no Haskeline context is required. Routing plain
   95 -- text through an injected handler keeps this function testable without a live
   96 -- provider.
   97 interpretDisposition :: ChannelCaps -> (Text -> IO ()) -> Disposition -> IO ()
   98 interpretDisposition caps plainHandler d = do
   99   mFollowUp <- case d of
  100     DispatchAction a -> runCommandAction a caps
  101     ShowText t       -> ccSend caps t >> pure Nothing
  102     PlainMessage t   -> plainHandler t >> pure Nothing
  103     Rejected msg     -> ccSend caps msg >> pure Nothing
  104   for_ mFollowUp plainHandler

  106 -- | Drive one plain-text turn through the agent loop. The seam the wiring test
  107 -- asserts against: a 'PlainMessage' becomes @runApp env (runTurn agentEnv t)@.
  108 -- Catches exceptions (including 'TranscriptError' from a dead writer daemon)
  109 -- so the TUI reports the error instead of crashing.
  110 handlePlain :: AgentEnv -> Env -> Text -> IO ()
  111 handlePlain agentEnv env t = do
  112   eResult <- withExceptionLogging (envLogger env) (aeLogPath agentEnv) "plain" $
  113     runApp env (runTurn agentEnv t)
  114   case eResult of
  115     Left errMsg -> ccSend (aeCaps agentEnv) ("turn failed: " <> errMsg)
  116     Right _     -> pure ()

  134 -- | Run the Haskeline TUI loop.
  135 --
  136 -- History is persisted at @\<state\>\/history@; the agent transcript is written
  137 -- under the session directory (@\<state\>\/sessions\/\<id\>\/transcript.jsonl@).
  138 -- EOF (Ctrl-D) exits. The provider and model are resolved from the active
  139 -- session on every turn so mid-session @\/model use@ changes take effect
  140 -- immediately.
  141 runCliTui
  142   :: SealPaths -> VaultRuntime -> RepoRegistryHandle -> ProviderRuntime -> SessionRuntime
  143   -> Registry -> PreprocessChain -> Backends -> TabsHandle -> AutonomyLevel
  144   -> AskReplyStore -> SealLogger -> HarnessRegistry -> TmuxRunner
  145   -> SessionAbortRegistry -> IO ()
  146 runCliTui paths rt repoReg pr sr registry chain backends tabsH autonomy askReply logger harnessReg tmuxRunner abortReg = do
  147   approvals <- newApprovalCache
  148   replies <- newReplyRegistry
  149   locks <- newSessionLocks
  150   execCache <- newSessionExecCache
  151   active0 <- readIORef (srActive sr)
  152   eSecCfg <- loadSecurityConfig (securityFilePath paths)
  153   let isRemote = either (const False) (isJust . untrustedExecConfigFromSecurity) eSecCfg
  154       histFile       = spState paths </> "history"
  155       innerSettings  = (defaultSettings :: Settings IO) { complete = noCompletion }
  156       hlSettings     = innerSettings { historyFile = Just histFile }
  157       caps = def
  158         { ccSend         = putStrLn . T.unpack
  159        , ccShowHuman    = putStrLn . T.unpack
  160         , ccPrompt       = \(AskPrompt prompt opts) ->
  161             runInputT innerSettings $ do
  162               mLine <- getInputLine (T.unpack (formatQuestionWithOptions prompt opts))
  163               pure (maybe "" T.pack mLine)
  164         , ccPromptSecret = \prompt ->
  165             runInputT innerSettings $ do
  166               mPass <- getPassword (Just '*') (T.unpack prompt)
  167               pure (maybe "" T.pack mPass)
  168         }
  169   -- Startup diagnostic: show which provider+model the active session will use
  170   -- for plain-text turns (resolved from config at session creation), and the
  171   -- bound default agent (if any).
  172   let agentLine = case smAgent active0 of
  173         Nothing -> ""
  174         Just aid -> "  agent: " <> agentDefIdText aid
  175   ccSend caps ("session: " <> smProvider active0 <> " / " <> smModel active0 <> agentLine)
  176   let skillBackend     = bSkills backends
  177       agentDefBackend   = bAgentDefs backends
  178     -- The /bg runner: mint a fresh persisted session from the config
  179     -- defaults, build a ChannelCaps whose ccPrompt routes through askHuman
  180     -- (notify = print the question via ccSend, so the confirmation appears
  181     -- at the > prompt), and fork a turn against a bg-session-scoped ISA
  182     -- registry. The fork lets the CLI loop keep reading input; the loop's
  183     -- deliverNextAnswerAny routes the next line as the confirmation answer.
  184     -- The assistant reply is delivered via the bg caps' ccSend (println).
  185     -- No tab/cursor state is touched. The bg session's ISA registry is
  186     -- built per-invocation (its own sid, its own transcript bracket) —
  187     -- unchanged by the per-turn refactor.
  188       bgRunner = BgRunner $ \prompt -> do
  189         cfg <- fromRight defaultRuntimeConfig <$> loadRuntimeConfig (prConfigPath pr)
  190         (mAgent, mProv, mModel) <- resolveDefaultAgent agentDefBackend cfg
  191         let (cfgProv, cfgModel) = defaultSessionSelection cfg
  192             provider = fromMaybe cfgProv mProv
  193             model    = fromMaybe cfgModel mModel
  194         meta <- newSession paths provider model "bg" mAgent
  195         let bgSid = smId meta
  196             bgCaps = def
  197               { ccSend = ccSend caps
  198               , ccShowHuman = ccShowHuman caps
  199               , ccPrompt = \(AskPrompt q opts) -> do
  200                   outcome <- askHumanWithOptions askReply bgSid q opts
  201                                (\_qid -> ccSend caps (formatQuestionWithOptions q opts))
  202                   pure (fromRight "" outcome)
  203               , ccPromptSecret = ccPromptSecret caps
  204               }
  205             td = TurnDeps
  206               { tdPaths        = paths
  207               , tdVault        = rt
  208               , tdProvider     = pr
  209               , tdResolve      = resolveSessionProvider pr
  210               , tdRepoReg      = repoReg
  211               , tdAutonomy     = autonomy
  212               , tdBroker       = Nothing
  213               , tdHarnessReg   = harnessReg
  214               , tdTmuxRunner   = tmuxRunner
  215               , tdHttpManager  = Just (prManager pr)
  216               , tdApprovals    = approvals
  217               , tdReplies      = replies
  218               , tdLocks        = locks
  219               , tdAbortReg     = abortReg
  220               , tdTabsHandle   = tabsH
  221               , tdLogger       = logger
  222               , tdIsRemote     = isRemote
  223               , tdBaseBackends = backends
  224               , tdExecCache    = execCache
  225               , tdRemoteRunner = Nothing
  226               , tdMkWorker    = Nothing
  227               }
  228             bgAdapter = TurnAdapter
  229               { taCaps          = bgCaps
  230               , taPreTurn       = \_ _ _ -> pure ()
  231               , taChannelLabel  = smChannel
  232               , taOnStop        = const Nothing
  233               , taOnUserMessage = const Nothing
  234               , taPostTurn      = \_ _ -> pure ()
  235               , taStartWiring   = \sessionBackends sid appEnv' eCfg' opCeiling m ->
  236                   TurnEngine.buildStartWiring td sessionBackends sid appEnv' eCfg' opCeiling (smChannel m)
  237               }
  238         void (forkIO (do
  239           outcome <- runSessionTurn td bgAdapter meta Nothing prompt
  240           case toError outcome of
  241             Just err -> ccSend caps ("bg failed: " <> err)
  242             Nothing  -> pure ()))
  243       registryWithBg = mkRegistry (registrySpecs registry <> [backgroundCommandSpec bgRunner, callCommandSpec callDispatcher, skillCommandSpec skillBackend callDispatcher])
  244       -- The /call dispatcher: dispatch an opcode against the active
  245       -- session's ISA registry + transcript under Full autonomy (the
  246       -- operator is the approver by typing /call). Returns the
  247       -- structured DispatchError/OpResult for the command to render.
  248       -- Rebuilt per invocation via `withCliTurn` so a `/new` swap of
  249       -- `srActive` flows the new sid's transcript + isaReg into the
  250       -- dispatch.
  251       callDispatcher callOpName val = do
  252         meta <- readIORef (srActive sr)
  253         let sid = smId meta
  254             td = TurnDeps
  255               { tdPaths        = paths
  256               , tdVault        = rt
  257               , tdProvider     = pr
  258               , tdResolve      = resolveSessionProvider pr
  259               , tdRepoReg      = repoReg
  260               , tdAutonomy     = autonomy
  261               , tdBroker       = Nothing
  262               , tdHarnessReg   = harnessReg
  263               , tdTmuxRunner   = tmuxRunner
  264               , tdHttpManager  = Just (prManager pr)
  265               , tdApprovals    = approvals
  266               , tdReplies      = replies
  267               , tdLocks        = locks
  268               , tdAbortReg     = abortReg
  269               , tdTabsHandle   = tabsH
  270               , tdLogger       = logger
  271               , tdIsRemote     = isRemote
  272               , tdBaseBackends = backends
  273               , tdExecCache    = execCache
  274               , tdRemoteRunner = Nothing
  275               , tdMkWorker    = Nothing
  276               }
  277         TurnEngine.callDispatcher td caps sid "cli" callOpName val
  278       plainHandler t = do
  279         meta <- readIORef (srActive sr)
  280         let td = TurnDeps
  281               { tdPaths        = paths
  282               , tdVault        = rt
  283               , tdProvider     = pr
  284               , tdResolve      = resolveSessionProvider pr
  285               , tdRepoReg      = repoReg
  286               , tdAutonomy     = autonomy
  287               , tdBroker       = Nothing  -- CLI has no WS broker
  288               , tdHarnessReg   = harnessReg
  289               , tdTmuxRunner   = tmuxRunner
  290               , tdHttpManager  = Just (prManager pr)
  291               , tdApprovals    = approvals
  292               , tdReplies      = replies
  293               , tdLocks        = locks
  294               , tdAbortReg     = abortReg
  295               , tdTabsHandle   = tabsH
  296               , tdLogger       = logger
  297               , tdIsRemote     = isRemote
  298               , tdBaseBackends = backends
  299               , tdExecCache    = execCache
  300               , tdRemoteRunner = Nothing
  301               , tdMkWorker    = Nothing
  302               }
  303             adapter = TurnAdapter
  304               { taCaps          = caps
  305               , taPreTurn       = \_ _ _ -> pure ()
  306               , taChannelLabel  = const "cli"
  307               , taOnStop        = const Nothing
  308               , taOnUserMessage = const Nothing
  309               , taPostTurn      = \_ _ -> pure ()
  310               , taStartWiring   = \sessionBackends sid appEnv' eCfg' opCeiling _meta ->
  311                   TurnEngine.buildStartWiring td sessionBackends sid appEnv' eCfg' opCeiling "cli"
  312               }
  313         outcome <- runSessionTurn td adapter meta Nothing t
  314         for_ (toError outcome) (ccSend caps)
  315   runInputT hlSettings (loop caps plainHandler tabsH registryWithBg)
  316   where
  317     loop :: ChannelCaps -> (Text -> IO ()) -> TabsHandle -> Registry -> InputT IO ()
  318     loop caps plainHandler th reg = do
  319       -- Ctrl-C handler: on interrupt, abort the active session's in-flight
  320       -- tool call (design Task 7 — CLI abort wiring). The interrupt is
  321       -- caught here (not propagated), so the loop continues; the next
  322       -- getInputLine re-prompts. The active session's abort flag is set
  323       -- via the SessionAbortRegistry.
  324       mLine <- withInterrupt
  325         (handleInterrupt
  326           (liftIO $ do
  327             active <- readIORef (srActive sr)
  328             setSessionAbort abortReg (smId active)
  329             ccSend caps "interrupted (in-flight tool call aborted)"
  330             pure Nothing
  331           )
  332           (getInputLine "> "))
  333       case mLine of
  334         Nothing   -> pure ()   -- EOF / Ctrl-D
  335         Just line -> do
  336           -- A forked /bg turn may have registered a pending confirmation
  337           -- (askHuman) for its background session. Deliver the next input
  338           -- line as that answer before any normal routing; if no ask is
  339           -- pending, deliverNextAnswerAny returns False and the line is
  340           -- routed normally. This mirrors the per-session deliverNextAnswer
  341           -- the inbox-driven channels run at the top of their loop, but is
  342           -- session-agnostic because the CLI has one input stream serving
  343           -- the active session plus any /bg background sessions.
  344           (_resolved, delivered) <- liftIO $ deliverNextAnswerResolvedAny askReply (T.pack line)
  345           if delivered
  346             then loop caps plainHandler th reg
  347             else do
  348               case Seal.Routing.Route.route (T.pack line) of
  349                 Right (Seal.Routing.Route.Focus idx) -> liftIO (focusTabH th idx) >>= \r -> liftIO $ ccSend caps (case r of Left e -> "focus: " <> e; Right _ -> "focused tab " <> T.singleton (tabIndexToChar idx))
  350                 Right (Seal.Routing.Route.Inject idx payload) -> liftIO $ do
  351                   _ <- focusTabH th idx
  352                   plainHandler payload
  353                 Right (Seal.Routing.Route.TabCommand tsc) -> liftIO (handleTabCommand caps th tsc)
  354                 Right Seal.Routing.Route.CurrentTab -> liftIO $ do
  355                   active <- readIORef (srActive sr)
  356                   tl <- snapshotTabs th
  357                   case lookupByRef tl (BoundSession (smId active)) of
  358                     Just t  -> ccSend caps (renderTab t)
  359                     Nothing -> ccSend caps "no current tab"
  360                 Right (Seal.Routing.Route.NewSession _args) -> do
  361                   -- /new is registered as a CommandSpec in the registry
  362                   -- (the CLI tracks "current" via srActive, not a cursor),
  363                   -- so re-route to the registry path below. Falling
  364                   -- through by re-parsing as SlashCommand.
  365                   d <- liftIO $ ingest reg chain (RawInbound (T.pack line))
  366                   liftIO $ interpretDisposition caps plainHandler d
  367                 Right (Seal.Routing.Route.SlashCommand _) -> do
  368                   d <- liftIO $ ingest reg chain (RawInbound (T.pack line))
  369                   liftIO $ interpretDisposition caps plainHandler d
  370                 Right (Seal.Routing.Route.Plain t) -> liftIO $ plainHandler t
  371                 Left (Seal.Routing.Route.ParseError e) -> liftIO $ ccSend caps e
  372               loop caps plainHandler th reg

  374 -- | Handle a parsed 'TabSlashCommand' by mutating the 'TabsHandle' and
  375 -- replying via the channel caps. Pure-ish (the handle mutations are STM).
  376 handleTabCommand :: ChannelCaps -> TabsHandle -> TabSlashCommand -> IO ()
  377 handleTabCommand caps tabsH = \case
  378   TabListCmd -> do
  379     tl <- snapshotTabs tabsH
  380     if tabCount tl == 0
  381       then ccSend caps "no tabs"
  382       else mapM_ (ccSend caps . renderTab) (tlTabs tl)
  383   TabNewCmd _mKind -> do
  384     r <- insertTabH tabsH (BoundSession placeholderSid) KindAi Nothing
  385     case r of
  386       Left e  -> ccSend caps ("tab new failed: " <> e)
  387       Right i -> ccSend caps ("tab " <> T.singleton (tabIndexToChar i) <> " created")
  388   TabCloseCmd idx force -> do
  389     r <- removeTabH tabsH idx
  390     case r of
  391       Left e  -> ccSend caps (if force == Force then "force close: " <> e else "close failed: " <> e)
  392       Right _ -> ccSend caps ("tab " <> T.singleton (tabIndexToChar idx) <> " closed")
  393   TabFocusCmd idx -> do
  394     r <- focusTabH tabsH idx
  395     case r of
  396       Left e  -> ccSend caps ("focus failed: " <> e)
  397       Right _ -> ccSend caps ("focused tab " <> T.singleton (tabIndexToChar idx))
  398   TabResumeCmd sid -> do
  399     r <- insertTabH tabsH (BoundSession sid) KindAi Nothing
  400     case r of
  401       Left e  -> ccSend caps ("resume failed: " <> e)
  402       Right i -> ccSend caps ("tab " <> T.singleton (tabIndexToChar i) <> " resumed")
  403   TabRenameCmd idx name -> do
  404     r <- renameTabH tabsH idx name
  405     case r of
  406       Left e  -> ccSend caps ("rename failed: " <> e)
  407       Right _ -> ccSend caps ("tab " <> T.singleton (tabIndexToChar idx) <> " renamed to " <> name)
  408   where
  409     placeholderSid = case mkSessionId "tab-session" of
  410       Right s -> s
  411       Left _  -> error "placeholder session id"

  413 -- | Render one tab as a single line: @<index>  <kind>  [label]@.
  414 renderTab :: Tab -> Text
  415 renderTab t =
  416   T.singleton (tabIndexToChar (tIndex t)) <> "  " <> T.pack (show (tKind t))
  417     <> maybe "" ("  " <>) (tLabel t)

  419 -- | Resolve the untrusted-execution 'UntrustedIO' capability handle from the
  420 -- 'SecurityConfig'. Absent section / mode=local → the local arm (real local
  421 -- executor wired to the workspace root). mode=remote + remote fully
  422 -- configured → the remote arm (the SSH executor; the opcodes run remotely
  423 -- via 'mkRemoteUntrustedIO'). mode=remote + remote absent/incomplete → the
  424 -- fail-closed stub so untrusted opcodes fail at call time (the
  425 -- 'UeExec ExecNotImplemented' error surfaces).
  426 untrustedIOFromSecurity :: WorkspaceRoot -> SecurityConfig -> UntrustedIO
  427 untrustedIOFromSecurity wsRoot cfg =
  428   case untrustedExecConfigFromSecurity cfg of
  429     Nothing -> defaultHandle
  430     Just uec ->
  431       case uecRemote uec of
  432         Nothing -> mkRemoteUntrustedIOStub  -- mode=remote, no remote configured
  433         Just sshCfg ->
  434           -- The remote SSH arm. The RemoteRunner is the real
  435           -- 'mkRealRemoteRunner' (System.Process-backed); the SSH argv +
  436           -- host-key pinning are inherited from 'Seal.Tools.Exec.Remote'.
  437           mkRemoteUntrustedIO sshCfg mkRealRemoteRunner
  438   where
  439 #if !defined(REMOTE_ONLY_UNTRUSTED)
  440     defaultHandle = mkLocalUntrustedIO wsRoot
  441 #else
  442     defaultHandle = mkRemoteUntrustedIOStub  -- local executor absent; fail-closed
  443     _ = wsRoot  -- keep wsRoot referenced under the flag (unused when local absent)
  444 #endif

    2 -- | The first-run setup state: what the wizard must show, computed from files
    3 -- on disk plus the live vault handle.
    4 --
    5 -- == Two rules this module exists to enforce
    6 --
    7 -- 1. /Never/ treat @vsLocked@ as a readiness signal on its own. A
    8 --    @per_access@ vault never populates the in-memory cache (every read goes
    9 --    to disk), so a perfectly working vault reports @vsLocked = True@
   10 --    forever. Only @startup@ mode can be genuinely locked in-process, and
   11 --    @reopen@ (which calls @vhUnlock@) is the in-product fix.
   12 -- 2. /Never/ decrypt while merely rendering the wizard. For a YubiKey vault,
   13 --    probing credentials means a physical touch, so the provider step is
   14 --    config-only there (§5.2); the explicit Test button is where a touch is
   15 --    allowed to happen.
   16 --
   17 -- Every 'SetupStep' detail is path-free: tilde form at most, no absolute path,
   18 -- no secret, no credential value. This module imports no vault read function
   19 -- beyond 'VaultStatus' — it cannot leak vault contents because it cannot read
   20 -- them.
   21 module Seal.Setup.State
   22   ( StepStatus (..)
   23   , SetupStep (..)
   24   , ProviderCard (..)
   25   , SetupState (..)
   26   , SetupInputs (..)
   27   , emptySetupInputs
   28   , deriveSetupState
   29   ) where
   30

   50 -- | The three states a wizard row can be in. There is deliberately no
   51 -- @StepSkipped@: the gate is blocking (operator decision), so "done" and
   52 -- "not done" are the only outcomes a required step can have.
   53 data StepStatus = StepReady | StepPending | StepBlocked
   54   deriving stock (Eq, Show)
   55 
   56 data SetupStep = SetupStep
   57   { stId       :: Text
   58   , stRequired :: Bool
   59   , stStatus   :: StepStatus
   60   , stDetail   :: Maybe Text
   61     -- ^ Path-free: tilde form at most. Never a secret or a credential value.
   62   , stActions  :: [Text]
   63     -- ^ @create@ | @resume@ | @reopen@ | @recheck@ | @configure@.
   64     -- @configure@ renders no action-bar button (the provider step has its own
   65     -- inline form); the rest map to buttons.
   66   } deriving stock (Eq, Show)
   67 
   68 -- | One row of the provider step's list. Carried on 'SetupState' so the
   69 -- provider step renders without a second call to @GET \/api\/providers\/all@
   70 -- (which decrypts once per provider name).
   71 data ProviderCard = ProviderCard
   72   { pcId           :: Text
   73   , pcLabel        :: Text
   74   , pcConfigured   :: Bool
   75   , pcIsDefault    :: Bool
   76   , pcDefaultModel :: Maybe Text
   77   , pcModels      :: [Text]
   78     -- ^ The operator's frequently-used models for this provider. Offered as
   79     -- suggestions by the new-tab model combobox.
   80   , pcRequiresBaseUrl :: Bool
   81     -- ^ True for providers with no built-in endpoint (the generic
   82     -- OpenAI-compatible option): the wizard must collect the URL, and the
   83     -- backend refuses to use the provider without one. Server-driven so the
   84     -- wizard never has to know which provider is which.
   85   , pcRequiresName :: Bool
   86     -- ^ True for the generic OpenAI-compatible option: each instance needs its
   87     -- own unique name, which becomes the config key and the vault-key stem.
   88   } deriving stock (Eq, Show)

   90 data SetupState = SetupState
   91   { ssRequiredComplete :: Bool
   92   , ssSteps            :: [SetupStep]
   93   , ssProviders        :: [ProviderCard]
   94   } deriving stock (Eq, Show)

   96 -- | Everything 'deriveSetupState' needs, as values.
   97 --
   98 -- The two @stat@ results ('siVaultFileExists', 'siIdentityReadable') are
   99 -- inputs rather than seams, so classification is testable without a
  100 -- filesystem; the caller does the two @stat@ calls against 'siPaths'.
  101 data SetupInputs = SetupInputs
  102   { siPaths              :: SealPaths
  103   , siRuntime            :: RuntimeConfig
  104   , siSecurity           :: SecurityConfig
  105   , siSecurityProblem    :: Maybe Text
  106     -- ^ 'Just' a summary when @security.toml@ exists but does not parse.
  107     -- 'Nothing' when it parsed (or was absent, in which case 'siSecurity' is
  108     -- 'defaultSecurityConfig').
  109   , siHandle             :: Maybe VaultHandle
  110   , siVaultFileExists    :: Bool
  111   , siIdentityReadable   :: Maybe Bool
  112     -- ^ 'Nothing' = no identity recorded; @Just False@ = recorded but not a
  113     -- readable regular file.
  114   , siHome               :: FilePath
  115     -- ^ The user's home directory, the base for the tilde form a detail may
  116     -- show. 'emptySetupInputs' guesses it as the seal home's parent, which is
  117     -- exactly right for the default @~/.seal@ layout; the handler can override
  118     -- it with the real home directory.
  119   , siConfiguredProviders :: IO [KnownProvider]
  120     -- ^ The @adProviders@ seam. Never invoked for a YubiKey vault.
  121   }
  122 
  123 -- | An "empty install": no config files on disk, no handle, no vault, no
  124 -- provider probing. The base a handler overrides field by field.
  125 emptySetupInputs :: SealPaths -> SetupInputs
  126 emptySetupInputs paths = SetupInputs
  127   { siPaths = paths
  128   , siRuntime = defaultRuntimeConfig
  129   , siSecurity = defaultSecurityConfig
  130   , siSecurityProblem = Nothing
  131   , siHandle = Nothing
  132   , siVaultFileExists = False
  133   , siIdentityReadable = Nothing
  134   , siConfiguredProviders = pure []
  135   , siHome = takeDirectory (spHome paths)
  136   }

  138 -- | The whole state. 'ssSteps' order is the wizard's order: vault first.
  139 deriveSetupState :: SetupInputs -> IO SetupState
  140 deriveSetupState inputs = do
  141   vault <- classifyVault inputs
  142   configured <- configuredProviderIds inputs
  143   let provider = classifyProvider inputs configured
  144       steps = [vault, provider]
  145   pure SetupState
  146     { ssRequiredComplete = all isSatisfied steps
  147     , ssSteps = steps
  148     , ssProviders = providerCards inputs configured
  149     }
  150   where
  151     -- A future non-required step (chat channels) must not be able to block the
  152     -- gate by accident: only required steps constrain completion.
  153     isSatisfied step = not (stRequired step) || stStatus step == StepReady
  154 
  155 -- ---------------------------------------------------------------------------
  156 -- Vault step (design §5.1, ordered by terminality)
  157 -- ---------------------------------------------------------------------------
  158 
  159 classifyVault :: SetupInputs -> IO SetupStep
  160 classifyVault inputs
  161   -- V0: the file exists but does not decode. Never attempt a write; the
  162   -- runtime case is "boot refused, then a later edit broke it".
  163   | Just summary <- siSecurityProblem inputs =
  164       pure (blockedVault
  165         ("security.toml could not be parsed (" <> summary
  166           <> "). Fix or restore it, then Re-check.")
  167         ["recheck"])
  168   -- V1: an orphan vault file. Precedes the identity check on purpose: when
  169   -- both are wrong the operator must attach a key first, and the reverse order
  170   -- would give advice that cannot fix the blocker.
  171   | siVaultFileExists inputs, isNothing recipient =
  172       pure (blockedVault (orphanDetail inputs) ["recheck"])
  173   -- V2: the recorded identity cannot be read.
  174   | isJust recipient, Just False <- siIdentityReadable inputs =
  175       pure (blockedVault (identityDetail inputs) ["recheck"])
  176   | otherwise = do
  177       locked <- handleLocked (siHandle inputs)
  178       pure (classifyWith locked)
  179   where
  180     security = siSecurity inputs
  181     recipient = scVaultRecipient security
  182     startupMode = scVaultUnlock security == Just "startup"
  183     hasHandle = isJust (siHandle inputs)
  184     vaultExists = siVaultFileExists inputs
  185 
  186     classifyWith locked
  187       -- V3: startup mode, handle present, genuinely locked.
  188       | hasHandle, startupMode, locked =
  189           blockedVault
  190             "The vault is locked (mode: `startup`). Reopen it here, or run \
  191             \`/vault unlock` in `seal tui`."
  192             ["reopen", "recheck"]
  193       -- V4: configured on disk but no handle in this process.
  194       | isJust recipient, vaultExists, not hasHandle =
  195           blockedVault "The vault is configured but not open in this process." ["reopen"]
  196       -- V5: key recorded, no vault file — one click from done.
  197       | isJust recipient, not vaultExists =
  198           pendingVault "Finish creating the vault (a key is already recorded)." ["resume"]
  199       -- V6: open, on disk, identity readable, and not genuinely locked.
  200       | hasHandle, vaultExists, not (startupMode && locked) =
  201           readyVault ("Key type: `" <> fromMaybe "unknown" (scVaultKeyType security) <> "`.")
  202       -- Nothing recorded, no vault file: the clean first run.
  203       | otherwise =
  204           pendingVault "Create an encrypted vault for API keys and tokens." ["create"]
  205 
  206 handleLocked :: Maybe VaultHandle -> IO Bool
  207 handleLocked Nothing = pure False
  208 handleLocked (Just handle) = vsLocked <$> vhStatus handle
  209 
  210 orphanDetail :: SetupInputs -> Text
  211 orphanDetail inputs =
  212   "A vault file exists at " <> displayPath inputs (configuredVaultPath inputs)
  213     <> " but no key is recorded. Run `seal tui`, then `/vault setup` to attach \
  214        \a key, then Re-check."
  215 
  216 identityDetail :: SetupInputs -> Text
  217 identityDetail inputs =
  218   "security.toml names `" <> identityBasename inputs <> "`, which cannot be \
  219   \read. Fix ~/.seal/security.toml or re-run `/vault setup` in `seal tui`, \
  220   \then Re-check."
  221 
  222 identityBasename :: SetupInputs -> Text
  223 identityBasename inputs = case scVaultIdentity (siSecurity inputs) of
  224   Nothing -> "the identity file"
  225   Just raw -> T.pack (takeFileName (T.unpack raw))
  226 
  227 configuredVaultPath :: SetupInputs -> FilePath
  228 configuredVaultPath inputs =
  229   maybe (vaultFilePath (siPaths inputs)) T.unpack (scVaultPath (siSecurity inputs))
  230 
  231 -- | Render a path for a detail. Tilde form when it is under the user's home
  232 -- (@~/.seal/config/vault/vault.age@); otherwise the file name alone. A detail
  233 -- must never carry an absolute path (design §5.1), so an unusual @SEAL_HOME@
  234 -- loses the location rather than disclosing it.
  235 displayPath :: SetupInputs -> FilePath -> Text
  236 displayPath inputs path = case stripPrefix (siHome inputs <> "/") path of
  237   Just rest -> "~/" <> T.pack rest
  238   Nothing   -> T.pack (takeFileName path)

  244 -- | Which provider ids count as configured. For a YubiKey vault this is
  245 -- config-only (@[providers.*]@ sections), because asking the vault would
  246 -- decrypt — and a decrypt on a YubiKey identity means a physical touch.
  247 configuredProviderIds :: SetupInputs -> IO (Set Text)
  248 configuredProviderIds inputs
  249   | isYubiKeyVault inputs =
  250       pure (Set.fromList (Map.keys (rcProviders (siRuntime inputs))))
  251   | otherwise = do
  252       builtins <- map providerLabel <$> siConfiguredProviders inputs
  253       customs <- configuredCustomProviderNames (siHandle inputs) (siRuntime inputs)
  254       pure (Set.fromList (builtins <> customs))
  255 
  256 classifyProvider :: SetupInputs -> Set Text -> SetupStep
  257 classifyProvider inputs configured =
  258   case rcDefaultProvider (siRuntime inputs) of
  259     Just label
  260       -- P1
  261       | isYubiKeyVault inputs || Set.member label configured ->
  262           readyProvider (label <> " · " <> modelFor inputs label)
  263       -- P2
  264       | otherwise ->
  265           pendingProvider ("Add credentials for " <> label <> ".") ["configure"]
  266     Nothing
  267       -- P3
  268       | not (Set.null configured) ->
  269           pendingProvider "Choose which provider new sessions should use." ["configure"]
  270       -- P4
  271       | otherwise ->
  272           pendingProvider
  273             "Add an LLM provider: anthropic, openrouter, ollama, or an \
  274             \OpenAI-compatible endpoint."
  275             ["configure"]

  277 -- | The model a session would actually use: the global default, else the
  278 -- provider's configured default, else the provider's hardcoded default. A
  279 -- custom OpenAI-compatible instance falls back to that implementation's
  280 -- default.
  281 modelFor :: SetupInputs -> Text -> Text
  282 modelFor inputs label =
  283   case rcDefaultModel (siRuntime inputs) of
  284     Just m -> m
  285     Nothing ->
  286       let ModelId m = resolveDefaultModelForName (siRuntime inputs) label
  287       in m

  289 providerCards :: SetupInputs -> Set Text -> [ProviderCard]
  290 providerCards inputs configured =
  291   [ ProviderCard
  292       { pcId = providerLabel known
  293       , pcLabel = providerDisplayName known
  294       , pcConfigured = Set.member (providerLabel known) configured
  295       , pcIsDefault = rcDefaultProvider (siRuntime inputs) == Just (providerLabel known)
  296       , pcDefaultModel = providerDefaultModel (siRuntime inputs) (providerLabel known)
  297       , pcModels = providerModels (siRuntime inputs) (providerLabel known)
  298       , pcRequiresBaseUrl = case known of
  299           OpenAICompatibleProvider -> True
  300           _ -> False
  301       , pcRequiresName = case known of
  302           OpenAICompatibleProvider -> True
  303           _ -> False
  304       }
  305   | known <- knownProviders
  306   ]
  307 
  308 -- | The card's display name. Branding is explicit rather than derived: a
  309 -- mechanical capitalise turns @openrouter@ into "Openrouter".
  310 providerDisplayName :: KnownProvider -> Text
  311 providerDisplayName known = case known of
  312   AnthropicProvider  -> "Anthropic"
  313   OllamaProvider     -> "Ollama"
  314   OpenRouterProvider -> "OpenRouter"
  315   OpenAICompatibleProvider -> "OpenAI-compatible"
  316 
  317 isYubiKeyVault :: SetupInputs -> Bool
  318 isYubiKeyVault inputs = scVaultKeyType (siSecurity inputs) == Just "yubikey"

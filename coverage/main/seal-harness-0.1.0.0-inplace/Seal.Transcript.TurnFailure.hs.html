    3 -- | Classifying a failed turn into one user-facing descriptor.
    4 --
    5 -- The harness already catches turn exceptions and logs them
    6 -- ('Seal.Logging.Exceptions.withExceptionLogging'), but on the web surface the
    7 -- sanitized result was dropped on the floor, so a failed turn was
    8 -- indistinguishable from a completed one. This module owns the /vocabulary/ of
    9 -- that failure: what kind it is, how it is phrased for each surface, and what
   10 -- crosses the wire.
   11 --
   12 -- == Why the cause is a closed enum and not @show e@
   13 --
   14 -- The first design draft carried @T.pack (show e)@ as the cause. That was
   15 -- wrong, and the design review caught it: the harness's redaction machinery
   16 -- ('Seal.Handles.Transcript.redactMessages') is /provenance/-based — it blanks
   17 -- tool results whose opcode is in @secretOpcodes@ — and there is no
   18 -- secret-/value/ scrubber anywhere in the tree. An arbitrary exception message
   19 -- can embed a vault value, a credential in a URI, or a path under @~/.seal@.
   20 -- That text would have been written into @conversation.jsonl@ (which the model
   21 -- reads) and fanned out to Telegram/Signal (which cannot be un-sent).
   22 --
   23 -- So the cause is a closed 'FailureCause' vocabulary. Leaks are structurally
   24 -- impossible rather than scrubbed after the fact, matching the harness's actual
   25 -- invariant: 'Seal.ISA.Opcode.orRecorded' is documented as "what the transcript
   26 -- records (secret-free)" — exclusion by construction.
   27 --
   28 -- The full raw @show e@ still goes to @seal.log@, joined to the user-facing
   29 -- report by 'tfRef'.

   53 -- | What went wrong, as a closed vocabulary. Adding a constructor is a wire
   54 -- change: update 'causeWire' and 'parseCause' together.
   55 data FailureCause
   56   = CauseUnicode            -- ^ invalid UTF-8 in tool/process output
   57   | CauseNoSpace            -- ^ ENOSPC, rendered as "No space left on device"
   58   | CauseTooManyFiles       -- ^ EMFILE, rendered as "Too many open files"
   59   | CauseTranscriptDead     -- ^ the transcript writer died or was unreachable
   60   | CauseProviderUnresolved -- ^ provider/model/vault-key resolution failed
   61   | CauseOther Text         -- ^ exception TYPE name only — never a message
   62   deriving stock (Eq, Show)
   63 
   64 -- | One failed turn, normalized for every surface.
   65 --
   66 -- The JSON instance emits exactly @{headline, cause, ref}@ and nothing else
   67 -- (design §3.2 — deliberately no @kind@ field: 'FailureCause' is the only
   68 -- taxonomy, so a second discriminator could only drift from it).
   69 data TurnFailure = TurnFailure
   70   { tfHeadline :: !Text
   71   -- ^ Plain language, user-facing, e.g. "invalid UTF-8 in tool output".
   72   , tfCause    :: FailureCause
   73   , tfRef      :: !Text
   74   -- ^ 8-hex correlation id. The SAME id 'Seal.Logging.Exceptions' logged, so
   75   -- the operator can find the full detail in @seal.log@.
   76   }
   77   deriving stock (Eq, Show)
   78 
   79 -- | Classify an exception. /Pure/ on purpose: the caller supplies the
   80 -- transcript daemon's recorded inner cause (which is an @IO@ read), so this
   81 -- stays testable without IO. Precedence is most-specific first.
   82 --
   83 -- @mInner@ matters because the motivating incident threw a 'TranscriptError'
   84 -- whose own message ("the writer died") said nothing about the real cause (a
   85 -- strict UTF-8 decode). Reporting the wrapper would have been useless; reporting
   86 -- the inner exception is the whole point.
   87 classify :: Maybe SomeException -> SomeException -> Text -> TurnFailure
   88 classify mInner outer ref =
   89   let cause = case fromException outer of
   90         Just (TranscriptError _ inner) ->
   91           maybe CauseTranscriptDead classifyCause (inner `firstJust` mInner)
   92         Nothing -> classifyCause outer
   93   in TurnFailure (headlineFor cause) cause ref
   94 
   95 firstJust :: Maybe a -> Maybe a -> Maybe a
   96 firstJust (Just x) _ = Just x
   97 firstJust Nothing  y = y
   98 
   99 -- | The exception -> cause mapping, after 'TranscriptError' unwrapping.
  100 classifyCause :: SomeException -> FailureCause
  101 classifyCause e
  102   | isUnicode e      = CauseUnicode
  103   | isNoSpace e      = CauseNoSpace
  104   | isTooManyFiles e = CauseTooManyFiles
  105   | otherwise        = CauseOther (typeName e)
  106   where
  107     shown = displayException e
  108     -- 'UnicodeException' exposes no constructor we can match, so the only
  109     -- faithful probe is the rendered message. Both phrasings are GHC's.
  110     isUnicode ex =
  111       let m = displayException ex
  112       in "Cannot decode byte" `isInfixOf` m || "Invalid UTF-8" `isInfixOf` m
  113     -- GHC renders ENOSPC/EMFILE as words: "resource exhausted (No space left
  114     -- on device)" / "(Too many open files)". The literal tokens never appear in
  115     -- the rendering, so probing for them would be dead code.
  116     isNoSpace _ = "No space left on device" `isInfixOf` shown
  117     isTooManyFiles _ = "Too many open files" `isInfixOf` shown
  118 
  119 -- | The exception's TYPE NAME only — never its message, which may carry
  120 -- secrets or paths. 'SomeException' erases the type, so this probes the handful
  121 -- of types the harness actually raises and otherwise reports a fixed literal.
  122 typeName :: SomeException -> Text
  123 typeName e =
  124   case fromException e :: Maybe IOError of
  125     Just _  -> "IOError"
  126     Nothing -> case fromException e :: Maybe ErrorCall of
  127       Just _  -> "ErrorCall"
  128       Nothing -> "Exception"
  129 
  130 -- | The plain-language headline shown to the user.
  131 headlineFor :: FailureCause -> Text
  132 headlineFor = \case
  133   CauseUnicode            -> "invalid UTF-8 in tool output"
  134   CauseNoSpace            -> "out of disk space"
  135   CauseTooManyFiles       -> "out of open file handles"
  136   CauseTranscriptDead     -> "the session transcript became unavailable"
  137   CauseProviderUnresolved -> "the model provider could not be resolved"
  138   CauseOther _            -> "an internal error stopped the turn"
  139 
  140 -- | The cause line. Fixed strings only: an exception /type name/ is not a
  141 -- secret under constraint 1, whereas its message body can be.
  142 renderCause :: FailureCause -> Text
  143 renderCause = \case
  144   CauseUnicode            -> "UnicodeException"
  145   CauseNoSpace            -> "resource exhausted (No space left on device)"
  146   CauseTooManyFiles       -> "resource exhausted (Too many open files)"
  147   CauseTranscriptDead     -> "session transcript unavailable"
  148   CauseProviderUnresolved -> "model provider could not be resolved"
  149   CauseOther tn           -> tn
  150 
  151 -- | The full user-facing report: the web transcript entry and the CLI line.
  152 renderForUser :: TurnFailure -> Text
  153 renderForUser f =
  154   "Turn failed \8212 " <> tfHeadline f <> "\n\n"
  155   <> "Cause: " <> renderCause (tfCause f) <> "\n"
  156   <> "Reference: " <> tfRef f
  157 
  158 -- | The chat-channel message (Telegram/Signal): headline + reference only.
  159 --
  160 -- This is the one surface that leaves the machine and cannot be un-sent, so it
  161 -- carries the least text. Requirement 6 asks only that a phone user learns the
  162 -- turn failed; it does not ask for the cause.
  163 renderForChat :: TurnFailure -> Text
  164 renderForChat f =
  165   "Turn failed \8212 " <> tfHeadline f <> " (ref " <> tfRef f <> ")"
  166 
  167 -- | The operator-log line, joined to the full detail in @seal.log@ by 'tfRef'.
  168 renderForLog :: TurnFailure -> Text
  169 renderForLog f =
  170   "turn failure: " <> tfHeadline f
  171   <> "; cause=" <> renderCause (tfCause f)
  172   <> "; ref=" <> tfRef f
  173 
  174 -- | What the /model/ sees on the next turn: decision 1 accepts that it should
  175 -- know its own prior turn failed, so it does not repeat the same mistake.
  176 -- Deliberately terse and closed-vocabulary, so a context full of prior failures
  177 -- does not derail the next attempt.
  178 renderForModel :: TurnFailure -> Message
  179 renderForModel f = Message Assistant [CbText text]
  180   where
  181     text = "A previous turn failed: " <> tfHeadline f
  182       <> " (cause: " <> renderCause (tfCause f)
  183       <> ", ref " <> tfRef f <> ")"
  184 
  185 -- | The wire tag for a cause. Lowercase kebab-case, matching the repo's
  186 -- existing event vocabulary (@harness-status@, @reply-delivered@).
  187 causeWire :: FailureCause -> Text
  188 causeWire = \case
  189   CauseUnicode            -> "unicode"
  190   CauseNoSpace            -> "no-space"
  191   CauseTooManyFiles       -> "too-many-files"
  192   CauseTranscriptDead     -> "transcript-dead"
  193   CauseProviderUnresolved -> "provider-unresolved"
  194   CauseOther tn           -> "other:" <> tn
  195 
  196 -- | Parse a wire tag. An /unknown/ tag degrades to 'CauseOther' carrying the
  197 -- raw string rather than failing: this entry type is tag-filtered and will
  198 -- outlive several deploys, so it must be forward-compatible for free.
  199 parseCause :: Text -> FailureCause
  200 parseCause t = case t of
  201   "unicode"             -> CauseUnicode
  202   "no-space"            -> CauseNoSpace
  203   "too-many-files"      -> CauseTooManyFiles
  204   "transcript-dead"     -> CauseTranscriptDead
  205   "provider-unresolved" -> CauseProviderUnresolved
  206   other                 -> maybe (CauseOther other) CauseOther (T.stripPrefix "other:" other)
  207 
  208 instance ToJSON TurnFailure where
  209   toJSON f = object
  210     [ "headline" .= tfHeadline f
  211     , "cause"    .= causeWire (tfCause f)
  212     , "ref"      .= tfRef f
  213     ]
  214 
  215 instance FromJSON TurnFailure where
  216   parseJSON = withObject "TurnFailure" $ \o ->
  217     TurnFailure
  218       <$> o .: "headline"
  219       <*> (parseCause <$> o .: "cause")
  220       <*> o .: "ref"

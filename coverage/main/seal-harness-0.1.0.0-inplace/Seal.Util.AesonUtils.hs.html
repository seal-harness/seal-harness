   46 -- ---------------------------------------------------------------------------
   47 -- Minimal inline of the bits of @Text.Casing@ the @lensy*Snake@/@lensy*Kebab@
   48 -- helpers need. Vendored here so this module stays dependency-free (the
   49 -- @text-casing@ Hackage package isn't in the pinned haskell.nix index).
   50 -- @fromHumps@ splits @FooBarBaz@ / @fooBarBaz@ into @[Foo, Bar, Baz]@ /
   51 -- @[foo, Bar, Baz]@; @toQuietSnake@ joins with @_@ and lowercases;
   52 -- @toKebab@ joins with @-@ and lowercases.
   53 -- ---------------------------------------------------------------------------
   54 
   55 fromHumps :: String -> [String]
   56 fromHumps = foldr go []
   57   where
   58     go c []          = [[c]]
   59     go c (w:ws)
   60       | isUpper c    = [c] : w : ws
   61       | otherwise    = (c:w) : ws
   62 
   63 toQuietSnake :: [String] -> String
   64 toQuietSnake = concatWith '_' . map (map toLower)
   65   where concatWith _ []       = ""
   66         concatWith sep (x:xs) = x <> foldr (\a acc -> sep : a <> acc) "" xs
   67 
   68 toKebab :: [String] -> String
   69 toKebab = concatWith '-' . map (map toLower)
   70   where concatWith _ []       = ""
   71         concatWith sep (x:xs) = x <> foldr (\a acc -> sep : a <> acc) "" xs

   73 lensyLenToJSON
   74   :: (Generic a, GToJSON Zero (Rep a)) => Int -> a -> Value
   75 lensyLenToJSON n = genericToJSON (lensyLenOptions n)
   76 
   77 lensyLenParseJSON
   78   :: (Generic a, GFromJSON Zero (Rep a)) => Int -> Value -> Parser a
   79 lensyLenParseJSON n = genericParseJSON (lensyLenOptions n)

   81 lensyLenOptions :: Int -> Options
   82 lensyLenOptions n = defaultOptions { fieldLabelModifier = lensyLenConstructorToNiceJson n }
   83 
   84 lensyLenConstructorToNiceJson :: Int -> String -> String
   85 lensyLenConstructorToNiceJson n fieldName = firstToLower $ drop n fieldName
   86   where
   87     firstToLower (c:cs) = toLower c : cs
   88     firstToLower _ = error $ "lensyLenConstructorToNiceJson: bad arguments: " <> show (n,fieldName)

  106 lensyOptions :: Options

  108   { fieldLabelModifier = lensyFieldNameToNiceJson
  109   , constructorTagModifier = lensyConstructorToNiceJson

  112 lensySnakeOptions :: Options
  113 lensySnakeOptions = defaultOptions { fieldLabelModifier = lensyFieldNameToSnakeJson }

  115 lensyFieldNameToNiceJson :: String -> String
  116 lensyFieldNameToNiceJson fieldName = dropWhile (=='_') $ dropWhile (/='_') $ dropWhile (=='_') fieldName

  118 lensyFieldNameToSnakeJson :: String -> String
  119 lensyFieldNameToSnakeJson fieldName = toQuietSnake $ fromHumps $ dropWhile (=='_') $ dropWhile (/='_') $ dropWhile (=='_') fieldName

  121 lensyConstructorToNiceJson :: String -> String
  122 lensyConstructorToNiceJson constructorName = dropWhile (=='_') $ dropWhile (/='_') constructorName

  143 -- | Generic 'ToJSON' using 'stripPrefixOptions' — strips the leading run
  144 -- of lowercase letters (the lens prefix) from each record field and
  145 -- lowercases the first remaining char, keeping camelCase for the rest.
  146 --   @teId → id@, @crMaxTokens → maxTokens@, @cbInputSchema → inputSchema@,
  147 --   @msgRole → role@, @uInput → input@, @tdName → name@.
  148 stripPrefixToJSON
  149   :: (Generic a, GToJSON Zero (Rep a)) => a -> Value
  150 stripPrefixToJSON = genericToJSON stripPrefixOptions
  151 
  152 -- | Generic 'FromJSON' using 'stripPrefixOptions'. See 'stripPrefixToJSON'.
  153 stripPrefixParseJSON
  154   :: (Generic a, GFromJSON Zero (Rep a)) => Value -> Parser a
  155 stripPrefixParseJSON = genericParseJSON stripPrefixOptions
  156 
  157 -- | 'Options' that strip the leading lowercase lens prefix from record
  158 -- field names (camelCase preserved). Constructor tags are left as-is
  159 -- (Seal's enums encode as bare tags like @"User"@ / @"Request"@).
  160 stripPrefixOptions :: Options
  161 stripPrefixOptions = defaultOptions { fieldLabelModifier = stripLensPrefixCamel }
  162 
  163 -- | Drop the leading run of lowercase letters, then lowercase the first
  164 -- remaining character. Examples:
  165 --
  166 --   * @teId       → id@
  167 --   * @teTimestamp → timestamp@
  168 --   * @crMaxTokens → maxTokens@
  169 --   * @cbInputSchema → inputSchema@
  170 --   * @cbForId    → forId@
  171 --   * @msgRole    → role@
  172 --   * @uInput     → input@
  173 --   * @tdName     → name@
  174 --   * @rsContent  → content@
  175 --
  176 -- Field names with no leading lowercase run (e.g. already-@lowerCamel@
  177 -- keys produced by hand-written instances) are returned unchanged except
  178 -- for the (no-op) first-char lowercasing.
  179 stripLensPrefixCamel :: String -> String
  180 stripLensPrefixCamel fieldName =
  181   case dropWhile isLower fieldName of
  182     ""           -> fieldName
  183     firstUpper:rest -> toLower firstUpper : rest

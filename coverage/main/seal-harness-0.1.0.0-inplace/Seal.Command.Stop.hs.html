    2 -- | The @\/stop@ slash command: abort a session's in-flight tool call (and
    3 -- any future tool call until the next turn starts). Wires the
    4 -- 'SessionAbortRegistry' (design Blocker Resolution #2) to the channel
    5 -- command surface.
    6 --
    7 -- The target session is resolved by an @'IO' 'SessionId'@ action closed
    8 -- over at construction time, so the same spec serves both wiring shapes:
    9 --
   10 -- * Single-session channels (CLI, Signal, Telegram) use
   11 --   'stopCommandSpec' with the process-global @srActive@ ref — the
   12 --   channel owns one session at a time.
   13 -- * The multi-session web gateway builds a per-request spec via
   14 --   'stopCommandSpecForSession' closing over the request's explicit
   15 --   'SessionId' (mirroring the per-request @call@/@skill@ rebuild in
   16 --   'Seal.Gateway.Send.runSlash'). A @\/stop@ typed in tab 3 must abort
   17 --   tab 3's session, not whatever @srActive@ happens to point at.
   18 --
   19 -- The stop message is recorded to the session's transcript (as an
   20 -- assistant @\"(stopped)\"@ message) so it appears cross-channel — a
   21 -- @\/stop@ from Telegram is visible in the web frontend's transcript
   22 -- view, and vice versa. The transcript write is delegated to a
   23 -- 'StopTranscriptWriter' built at the wiring site (it needs the
   24 -- 'SealPaths' + broker + session metadata).
   25 module Seal.Command.Stop
   26   ( stopCommandSpec
   27   , stopCommandSpecForSession
   28   , StopTranscriptWriter
   29   , noStopTranscriptWriter
   30   , mkStopTranscriptWriter
   31   ) where

   53 -- | A function that records the stop message to the session's transcript
   54 -- and broadcasts the new entry to WS subscribers. Built at the wiring
   55 -- site from 'SealPaths' + the broker + session metadata. 'Nothing' (via
   56 -- 'noStopTranscriptWriter') skips the transcript write — used in tests
   57 -- or contexts without a transcript.
   58 newtype StopTranscriptWriter = StopTranscriptWriter (Maybe (SessionId -> Text -> IO ()))
   59 
   60 -- | The no-op writer (no transcript write). Used in tests or contexts
   61 -- without a transcript.
   62 noStopTranscriptWriter :: StopTranscriptWriter
   63 noStopTranscriptWriter = StopTranscriptWriter Nothing
   64 
   65 -- | Build a 'StopTranscriptWriter' from the shared dependencies. Opens
   66 -- the session's transcript, writes the stop message as an assistant
   67 -- 'EKResponse' entry, and broadcasts the new entry via the broker.
   68 mkStopTranscriptWriter :: SealPaths -> Maybe StreamBroker -> StopTranscriptWriter
   69 mkStopTranscriptWriter paths mBroker =
   70   StopTranscriptWriter $ Just $ \sid stopMsg -> do
   71     let dir = sessionDir paths sid
   72     mMeta <- loadSessionMeta paths sid
   73     now <- getCurrentTime
   74     let model = maybe "" smModel mMeta
   75         createdAt = maybe now smCreatedAt mMeta
   76     withTwoFileTranscript dir $ \h -> do
   77       let assistantMsg = Message Assistant [CbText stopMsg]
   78           entry = EntryRecord
   79             { erId = ""
   80             , erTimestamp = now
   81             , erKind = EKResponse
   82             , erConvLen = 1
   83             , erEnvelope = Nothing
   84             , erUsage = Nothing
   85             , erStop = Nothing
   86             , erDurationMs = Nothing
   87             , erHarness = Nothing
   88             , erCorrelation = Nothing
   89             , erMeta = mempty
   90             }
   91       tfwRecordAndAck h (TwoFileWrite [assistantMsg] entry)
   92       broadcastNewEntries mBroker paths sid model createdAt

   94 -- | The @\/stop@ command spec for single-session channels (CLI, Signal,
   95 -- Telegram). Closes over the 'SessionAbortRegistry' + 'SessionRuntime' +
   96 -- 'StopTranscriptWriter'; the action reads @srActive@ at dispatch time
   97 -- to resolve the target session. The command takes no arguments.
   98 stopCommandSpec :: SessionAbortRegistry -> SessionRuntime -> StopTranscriptWriter -> CommandSpec
   99 stopCommandSpec abortReg sr = stopCommandSpecWith abortReg (smId <$> readIORef (srActive sr))
  100 
  101 -- | The @\/stop@ command spec for the multi-session web gateway. Closes
  102 -- over the 'SessionAbortRegistry' + an explicit 'SessionId' (the
  103 -- request's sid from the URL, threaded in via the per-request registry
  104 -- rebuild in 'Seal.Gateway.Send.runSlash') + 'StopTranscriptWriter'.
  105 stopCommandSpecForSession :: SessionAbortRegistry -> SessionId -> StopTranscriptWriter -> CommandSpec
  106 stopCommandSpecForSession abortReg sid = stopCommandSpecWith abortReg (pure sid)
  107 
  108 -- | The core spec builder, parameterized by the session-id resolution
  109 -- action + the transcript writer. Both public constructors delegate here.
  110 stopCommandSpecWith :: SessionAbortRegistry -> IO SessionId -> StopTranscriptWriter -> CommandSpec
  111 stopCommandSpecWith abortReg resolveSid writer = CommandSpec
  112   { csName         = CommandName "stop"
  113   , csAliases      = []
  114   , csGroup        = GroupSession
  115   , csSynopsis     = "Abort the active session's in-flight tool call"
  116   , csParserInfo   = stopParserInfo abortReg resolveSid writer
  117   , csAvailability = InteractiveOnly
  118   }
  119 
  120 stopParserInfo :: SessionAbortRegistry -> IO SessionId -> StopTranscriptWriter -> ParserInfo CommandAction
  121 stopParserInfo abortReg resolveSid writer =
  122   info (pure (stopAction abortReg resolveSid writer) <**> helper)
  123     (  progDesc "Abort the active session's in-flight tool call"
  124     <> header   "stop — abort the active session's in-flight tool call"
  125     )
  126 
  127 stopAction :: SessionAbortRegistry -> IO SessionId -> StopTranscriptWriter -> CommandAction
  128 stopAction abortReg resolveSid (StopTranscriptWriter mWriter) = commandAction $ \caps -> do
  129   sid <- resolveSid
  130   setSessionAbort abortReg sid
  131   let stopMsg = "(stopped)"
  132   case mWriter of
  133     Just writer -> writer sid stopMsg
  134     Nothing     -> pure ()
  135   ccSend caps ("aborted session " <> sessionIdText sid)

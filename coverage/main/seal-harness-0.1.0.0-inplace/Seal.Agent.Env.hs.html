   26 data AgentEnv = AgentEnv
   27   { aeProvider :: SomeProvider
   28     -- | The provider's label (e.g. @\"ollama\"@), used only for display —
   29     -- 'aeProvider' is existential and carries no name of its own.
   30   , aeProviderLabel :: Text
   31   , aeModel :: ModelId
   32   , aeSystem :: Maybe Text
   33     -- ^ The system prompt injected at the start of every turn. For the
   34     -- main session this comes from the bound default agent's 'adSystem';
   35     -- for a forked sub-agent it comes from the def's 'adSystem'.
   36   , aeRegistry :: Registry
   37   , aeTranscript :: TwoFileHandle
   38   , aeBackend :: BackendExec
   39   , aeUIOEnv :: UIOEnv
   40     -- ^ The untrusted-execution environment (carrying the 'UntrustedIO'
   41     -- capability handle + the Git 'CloneDeps' surface), threaded to
   42     -- 'Seal.ISA.Dispatch.dispatch' for Untrusted opcodes. Backend (local
   43     -- FS vs SSH transport) is selected once at wiring time via
   44     -- 'mkSessionExec' (which builds the 'UIOEnv' from the 'SecurityConfig');
   45     -- the opcode never sees the backend, only the capability. Trusted/Audited
   46     -- opcodes ignore it (the GADT 'Opcode' has no 'UIOEnv' field for them —
   47     -- type-level capability scoping, spec §4/§8). The 'CloneDeps' lives
   48     -- inside the 'UIOEnv' (via 'uieCloneDeps'), so Git opcodes source it
   49     -- from here rather than a separate field.
   50   , aeAbortFlag :: AbortFlag
   51     -- ^ The session-scoped abort flag. Set by the channel layer (Signal
   52     -- @\/stop@, CLI Ctrl+C, Web @POST /api/sessions/:id/stop@) to cancel
   53     -- in-flight tool calls. Polled by the dispatch wrapper
   54     -- ('Seal.Tools.Exec.Timeout.runWithTimeoutAbortRetry') during the
   55     -- three-way race. Cleared once at 'runTurn' entry (before any tool
   56     -- call); a mid-turn abort keeps the flag set until the next turn.
   57     -- Looked up from a 'Seal.Tools.Exec.Abort.SessionAbortRegistry' by
   58     -- 'SessionId' (the web gateway has no 'AgentEnv' at the @/stop@ call
   59     -- site — the registry is the vehicle, design Blocker Resolution #2).
   60   , aeToolTimeout :: ToolTimeoutConfig
   61     -- ^ The per-call timeout/retry config (default 120s, max 600s, 3
   62     -- retries, 2s base delay, 2.0 backoff, 50KB output cap). Loaded from
   63     -- @config.toml@ @[tool_timeout]@ at startup via
   64     -- 'Seal.Config.File.toolTimeoutConfig'.
   65   , aeCaps :: ChannelCaps
   66   , aeSession :: SessionId
   67   , aeMaxTurns :: Int
   68   , aeChannel :: Text
   69     -- ^ The channel this turn's message arrived on (e.g. @\"telegram\"@,
   70     -- @\"web\"@, @\"cli\"@). This is the /arrival/ channel, NOT the
   71     -- session's channel-of-origin — a web message sent to a
   72     -- Telegram-created session carries @\"web\"@, not @\"telegram\"@. Each
   73     -- wiring site passes the channel the message actually came in on:
   74     -- the web gateway passes @"web"@, the CLI passes @"cli"@, and the
   75     -- inbox-channel loop passes @smChannel@ (correct because an inbox
   76     -- session is always created on the channel its messages arrive on).
   77     -- 'runTurn' stamps this into every request entry's @erMeta@ @channel@
   78     -- field so the frontend can attribute every user message to the
   79     -- channel it came from. This is the single source of truth for channel
   80     -- provenance — 'MessageSource' carries the finer-grained conversation
   81     -- id + user id, but the channel label itself always comes from here.
   82   , aeMessageSource :: Maybe MessageSource
   83     -- ^ The authenticated-transport identity of the inbound message this
   84     -- turn is answering. 'Nothing' for the CLI TUI and the web (which
   85     -- bypass 'MessageSource'); @'Just' ms@ for channels that carry one
   86     -- (Signal, Telegram). 'runTurn' folds the 'msConversationId' into the
   87     -- request 'EntryRecord's @erMeta@ @conversationId@ field. The
   88     -- @channel@ field comes from 'aeChannel' (above), NOT from here, so
   89     -- every channel — including those with no 'MessageSource' — is
   90     -- attributed.
   91   , aeAutonomy :: AutonomyLevel
   92     -- ^ The operator-selected autonomy level. 'Full' (@--yolo@) bypasses the
   93     -- human-confirmation gate for Untrusted opcodes (they run immediately after
   94     -- the ACK-before-execute audit). 'Supervised' (the default) prompts the
   95     -- human via 'ccPrompt' before executing any Untrusted opcode; a non-"yes"
   96     -- reply cancels the call (the model sees a denied result). 'Deny' is
   97     -- enforced by the opcode's own authorize gate (rejected before the
   98     -- dispatcher runs).
   99   , aeApprovals :: ApprovalCache
  100     -- ^ The approval cache for Untrusted opcodes under 'Supervised' autonomy.
  101     -- Records "for this session" and "always" approvals so subsequent calls
  102     -- to the same opcode skip the prompt. 'ScopeRejected' entries short-
  103     -- circuit to denied. Threaded from the channel wiring (web, CLI, Signal).
  104   , aeDebugRequestsPath :: Maybe FilePath
  105     -- ^ When 'Just', every 'CompletionRequest' sent to the LLM is appended
  106     -- (redundantly, in full) to this file as one JSONL line per request.
  107     -- The contract: each line is the complete 'CompletionRequest' exactly as
  108     -- passed to the provider — including the full 'crMessages' history — so
  109     -- we can debug whether the two-file storage format is correctly feeding
  110     -- the session history to the LLM. 'Nothing' (the default) means no
  111     -- debug file is written.
  112   , aeOnEntry :: IO ()
  113     -- ^ A hook called by the loop after each transcript entry is recorded
  114     -- (response entries, tool-result entries, approval-evidence entries).
  115     -- The web channel wires this to 'broadcastNewEntries' so the frontend
  116     -- sees new entries live — including tool calls that are pending
  117     -- confirmation — rather than only at the end of the turn. The CLI and
  118     -- Signal channels set this to @pure ()@ (no live broadcast needed).
  119   , aeOnUserMessage :: Maybe (IO ())
  120     -- ^ When 'Just action', the loop records the initial user message with
  121     -- 'tfwRecordAndAck' (synchronously fsync'd) and then runs @action@ —
  122     -- guaranteeing the user message is durable on disk before @action@
  123     -- runs. Used by the @/bg@ channel path to broadcast a @lists@ snapshot
  124     -- whose snippet (the first user message) is populated, so the web
  125     -- sidebar shows the session name immediately rather than after the
  126     -- first LLM response. 'Nothing' (the default) keeps the async
  127     -- 'tfwRecordAsync' write (no fsync latency at turn start).
  128   , aeOnStop     :: Maybe (Text -> IO ())
  129     -- ^ When 'Just fanout', the loop calls @fanout text@ with the final
  130     -- user-visible text at every stop branch (final answer, truncation
  131     -- give-up, max-turns stop, provider error). This is the chat-channel
  132     -- notification hook: wiring sites bind it to 'replyFanout' against the
  133     -- session's 'ReplyRegistry' so every subscribed chat channel (Telegram,
  134     -- Signal, …) attached to the tab receives the stop notice — not just
  135     -- the arrival channel (which 'ccSend' covers). 'Nothing' (the default
  136     -- for tests and the standalone CLI) means no fan-out; the arrival
  137     -- channel alone is notified via 'ccSend'.
  138   , aeStopFanoutDone :: IORef Bool
  139     -- ^ Flipped to 'True' by the loop's final-answer stop branch after
  140     -- 'aeOnStop' delivers the final text, so the turn engine's bracket
  141     -- cleanup skips 'fanoutLastReply' (which would re-send the identical
  142     -- last-assistant text — the double-delivery bug). The error/abort/
  143     -- max-turns stop branches do NOT set it: the engine's cleanup must
  144     -- still deliver the partial/notice text on those paths.
  145   , aeOnToolCall :: OpName -> Value -> IO ()
  146     -- ^ The loop calls @aeOnToolCall opName input@ before each tool
  147     -- dispatch. Chat channels wire this (via the turn adapter) to the
  148     -- 'StreamProgress' manager so tool-call progress messages are
  149     -- sent/edited on the chat platform. The default is a no-op (the turn
  150     -- engine passes @\\_ _ -> pure ()@ when no adapter hook is wired).
  151   , aeOnTextDelta :: Maybe (Text -> IO ())
  152     -- ^ When 'Just hook', the loop calls @hook delta@ for each text
  153     -- delta from the provider stream. Chat channels wire this to the
  154     -- 'StreamProgress' manager for progressive text edits. 'Nothing'
  155     -- (the default) means per-delta sends go through 'ccSend' (the
  156     -- existing CLI/web path) or are skipped (chat channels with
  157     -- ccStreaming = False).
  158   , aeOnDemandSchemas :: Bool
  159     -- ^ When 'True', the loop emits stub @input_schema@s in the @tools@
  160     -- field (via 'Seal.ISA.Registry.registryToolDefs'') to save tokens,
  161     -- and the registry is expected to include the @OPCODE_DESCRIBE@ /
  162     -- @OPCODE_LIST@ opcodes so the model can fetch full schemas on demand.
  163     -- 'False' (the default) sends full schemas inline, matching the
  164     -- pre-flag behavior.
  165   , aeLogPath :: Maybe FilePath
  166     -- ^ When 'Just', turn lifecycle events (start/end/duration) and failures
  167     -- (exceptions, provider errors) are appended to this per-session log file
  168     -- (@\<sessionDir\>\/seal.log@). Best-effort: a write error is swallowed.
  169     -- Records ONLY events not already in @entries.jsonl@ / @conversation.jsonl@.
  170     -- 'Nothing' disables per-session logging (the default for tests / fake
  171     -- transcripts).
  172   }

  174 -- | A parameter object bundling the per-turn inputs to 'mkSessionAgentEnv'.
  175 -- The fields are collected into one record so call sites construct it with
  176 -- named-field syntax (no positional-counting mistakes) and future additions
  177 -- are a one-field change. Lives here (next to 'AgentEnv') so the unified
  178 -- turn engine ('Seal.Core.TurnEngine') can import it without a cycle with
  179 -- 'Seal.Channel.Cli'.
  180 data TurnEnv = TurnEnv
  181   { teCaps          :: ChannelCaps
  182   , teProvider      :: SomeProvider
  183   , teProviderLabel :: Text
  184   , teModel         :: ModelId
  185   , teSession       :: SessionId
  186   , teSystem        :: Maybe Text
  187   , teRegistry      :: Registry
  188   , teTranscript    :: TwoFileHandle
  189   , teUioEnv        :: UIOEnv
  190   , teDebugReqPath  :: Maybe FilePath
  191   , teAutonomy      :: AutonomyLevel
  192   , teApprovals     :: ApprovalCache
  193   , teOnEntry       :: IO ()
  194   , teOnDemand      :: Bool
  195   , teLogPath       :: Maybe FilePath
  196   , teMaxTurns      :: Int
  197   , teOnUserMessage :: Maybe (IO ())
  198   , teChannel       :: Text
  199   , teOnStop        :: Maybe (Text -> IO ())
  200   , teStopFanoutDone :: IORef Bool
  201     -- ^ Set by the loop's final-answer stop branch AFTER 'teOnStop' fires,
  202     -- so the turn engine's bracket cleanup skips the redundant
  203     -- 'fanoutLastReply' re-send (double delivery to subscribed chat
  204     -- channels). 'Nothing' is not possible here — the engine always
  205     -- supplies the ref; direct 'TurnEnv' constructions in tests pass a
  206     -- fresh ref.
  207   , teOnToolCall    :: OpName -> Value -> IO ()
  208   , teOnTextDelta   :: Maybe (Text -> IO ())
  209   , teAbortFlag     :: AbortFlag
  210   , teToolTimeout   :: ToolTimeoutConfig
  211   }

  213 -- | Build the per-turn 'AgentEnv' from a 'TurnEnv'. The 'aeBackend' is
  214 -- always 'localBackend' (Trusted/Audited opcodes run in-process; Untrusted
  215 -- opcodes source their backend from 'aeUIOEnv', not from here). The
  216 -- 'aeMessageSource' is left 'Nothing' here — the unified turn engine sets
  217 -- it via a record update after construction (it's an inbound-property, not
  218 -- a per-turn-env field).
  219 mkSessionAgentEnv :: TurnEnv -> AgentEnv
  220 mkSessionAgentEnv te = AgentEnv
  221   { aeProvider   = teProvider te
  222   , aeProviderLabel = teProviderLabel te
  223   , aeModel      = teModel te
  224   , aeSystem     = teSystem te
  225   , aeRegistry   = teRegistry te
  226   , aeTranscript = teTranscript te
  227   , aeBackend    = localBackend
  228   , aeUIOEnv     = teUioEnv te
  229   , aeCaps       = teCaps te
  230   , aeSession    = teSession te
  231   , aeMaxTurns   = teMaxTurns te
  232   , aeChannel    = teChannel te
  233   , aeMessageSource = Nothing
  234   , aeAutonomy   = teAutonomy te
  235   , aeApprovals  = teApprovals te
  236   , aeDebugRequestsPath = teDebugReqPath te
  237   , aeOnEntry    = teOnEntry te
  238   , aeOnUserMessage = teOnUserMessage te
  239   , aeOnStop     = teOnStop te
  240   , aeStopFanoutDone = teStopFanoutDone te
  241   , aeOnToolCall = teOnToolCall te
  242   , aeOnTextDelta = teOnTextDelta te
  243   , aeOnDemandSchemas = teOnDemand te
  244   , aeLogPath    = teLogPath te
  245   , aeAbortFlag  = teAbortFlag te
  246   , aeToolTimeout = teToolTimeout te
  247   }

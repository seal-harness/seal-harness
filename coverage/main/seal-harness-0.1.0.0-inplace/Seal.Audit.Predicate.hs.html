    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Pure predicate: should this opcode invocation be written to the
    3 -- cross-session audit log?
    4 --
    5 -- The audit log captures only /mutating/ operations on the evolutionary
    6 -- stores (skills, agent definitions, vault secrets, memory). Read-only
    7 -- operations and non-evolutionary mutations (agent lifecycle, session
    8 -- creation, harness management) are excluded — their audit trail lives in
    9 -- the session transcript and the in-process runtime registries.
   10 --
   11 -- The predicate is an explicit allowlist of @(opcode, action)@ pairs. An
   12 -- opcode not listed here is never audited, regardless of its action. An
   13 -- action not listed for a listed opcode is not audited. A 'Nothing' action
   14 -- (opcode has no @action@ field) is never audited — all audited opcodes
   15 -- are action-based.
   16 module Seal.Audit.Predicate
   17   ( shouldAudit
   18   , extractAction
   19   , extractTargetId
   20   ) where
   21 
   22 import Data.Aeson (Value, withObject, (.:))
   23 import Data.Aeson.Types (parseMaybe)
   24 import Control.Applicative ((<|>))
   25 import Data.Maybe (fromMaybe)
   26 import Data.Text (Text)
   27 import Data.Map.Strict (Map)
   28 import Data.Map.Strict qualified as Map
   29 import Data.Set (Set)
   30 import Data.Set qualified as Set
   31 
   32 import Seal.Core.Types (OpName (..))
   33 
   34 -- | The audited actions for each opcode. Only the listed actions are
   35 -- audited; all other actions (read, list, search, load) are excluded.
   36 auditedActions :: Map OpName (Set Text)
   37 auditedActions = Map.fromList
   38   [ (OpName "SKILL_MANAGE", Set.fromList ["write", "delete"])
   39   , (OpName "AGENT_DEF_MANAGE", Set.fromList ["write", "delete"])
   40   , (OpName "SECRET_MANAGE", Set.fromList ["put", "delete"])
   41   , (OpName "MEMORY_MANAGE", Set.fromList ["write", "archive"])
   42   ]
   43 
   44 -- | Should this opcode invocation be written to the audit log?
   45 -- Returns 'True' only when both the opcode and its action are in the
   46 -- allowlist. A 'Nothing' action (no @action@ field in the input) is
   47 -- never audited.
   48 shouldAudit :: OpName -> Maybe Text -> Bool
   49 shouldAudit opName mAction =
   50   case mAction of
   51     Nothing -> False
   52     Just action -> case Map.lookup opName auditedActions of
   53       Nothing -> False
   54       Just allowed -> Set.member action allowed
   55 
   56 -- | Extract the @action@ field from an opcode's input JSON. Returns
   57 -- 'Nothing' when the input has no @action@ field (e.g. legacy single-action
   58 -- opcodes). All audited opcodes are action-based, so 'Nothing' means
   59 -- "not audited".
   60 extractAction :: Value -> Maybe Text
   61 extractAction = parseMaybe (withObject "input" (.: "action"))
   62 
   63 -- | Extract the target identifier from an opcode's input JSON. The field
   64 -- name varies by opcode: @id@ for skills and agent defs, @name@ for vault
   65 -- secrets, @path@ for memory. Falls back through these in order.
   66 extractTargetId :: Value -> Text
   67 extractTargetId v =
   68   let mId = parseMaybe (withObject "input" (.: "id")) v
   69       mName = parseMaybe (withObject "input" (.: "name")) v
   70       mPath = parseMaybe (withObject "input" (.: "path")) v
   71   in fromMaybe "-" (mId <|> mName <|> mPath)

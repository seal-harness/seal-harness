    2 -- | The Agent opcode group: three Audited definition opcodes
    3 -- (@AGENT_DEF_WRITE@, @AGENT_DEF_READ@, @AGENT_DEF_LIST@, @AGENT_DEF_DELETE@)
    4 -- and four Trusted lifecycle opcodes (@AGENT_INSTANCES@, @AGENT_START@,
    5 -- @AGENT_STATUS@, @AGENT_STOP@, @AGENT_INTERRUPT@). Only the *definition*
    6 -- mutations are Audited — running an instance is harness-internal, not an
    7 -- evolutionary mutation, so the lifecycle ops are Trusted.
    8 --
    9 -- @AGENT_DEF_WRITE@ is an upsert: if the def already exists, its name/system/
   10 -- tools are updated (the original 'adSession' provenance and 'adCreatedAt' are
   11 -- preserved; only 'adUpdatedAt' is bumped); if not, a fresh def is created.
   12 -- This merges the former AGENT_DEF_CREATE + AGENT_DEF_UPDATE into a single
   13 -- opcode. @orRecorded@ carries a @was_new@ flag so the audit log still
   14 -- distinguishes create vs update.
   15 --
   16 -- @AGENT_INSTANCES@ (renamed from @AGENT_LIST@) snapshots the in-process agent
   17 -- runtime (running instances), NOT the definitions. @AGENT_DEF_LIST@ lists the
   18 -- definitions. The rename stops the confusion between "list definitions" and
   19 -- "list running instances".
   20 --
   21 -- @AGENT_START@ is the Seal analog of Hermes' @delegate_task@: it spawns one
   22 -- or more child agents with isolated context, runs each against a goal to
   23 -- completion (synchronously), and returns a structured JSON result per child.
   24 -- The parent blocks until all children finish (or time out). See
   25 -- 'Seal.Agent.Runtime.Delegation' for the full feature list. The opcode is a
   26 -- thin shim over 'runDelegate' — it normalizes the model's input, resolves the
   27 -- def, delegates the worker construction to the wiring layer's
   28 -- 'AgentWorkerBuilder', and serializes the 'ChildResult' list to JSON.

   96 -- | Extract the @id@ string field from a JSON object.
   97 idField :: Value -> Maybe Text
   98 idField = parseMaybe (withObject "in" (.: "id"))

  100 -- | Extract a required text field (defaults to empty when absent).
  101 textField :: Text -> Value -> Text
  102 textField name v = fromMaybe "" (parseMaybe (withObject "in" (.: fromText name)) v)
  103 
  104 -- | Extract an optional text field (defaults to 'Nothing' when absent).
  105 textFieldMaybe :: Text -> Value -> Maybe Text
  106 textFieldMaybe name v =
  107   case parseMaybe (withObject "in" (.:? fromText name)) v :: Maybe (Maybe Text) of
  108     Just (Just t) -> Just t
  109     _             -> Nothing

  111 -- | Decode the @tools@ field: @\"all\"@ (or absent) -> 'AllowAll'; an array of
  112 -- opcode-name strings -> 'AllowOnly'. Malformed -> 'AllowAll' (permissive).
  113 toolsField :: Value -> AllowList OpName
  114 toolsField v =
  115   case parseMaybe (withObject "in" (.:? "tools")) v :: Maybe (Maybe Value) of
  116     Just (Just (String "all")) -> AllowAll
  117     Just (Just (Array xs))     -> AllowOnly (Set.fromList [ OpName t | String t <- V.toList xs ])
  118     _                          -> AllowAll

  120 -- | Resolve the @group@ field from the opcode input, preserving the existing
  121 -- def's group when the input omits it (mirrors the gateway @stampAgentDef@
  122 -- and the skill @stampSkill@ back-compat policy). An explicit empty string
  123 -- clears the group (sets 'Nothing'); a non-empty stripped value sets it.
  124 groupField :: Value -> Maybe AgentDef -> Maybe Text
  125 groupField v mExisting =
  126   case textFieldMaybe "group" v of
  127     Just g  -> let g' = T.strip g in if T.null g' then Nothing else Just g'
  128     Nothing -> adGroup =<< mExisting

  134 -- | AGENT_DEF_WRITE: upsert an agent definition by id. If the def already
  135 -- exists, its name/system/tools are updated (the original 'adSession'
  136 -- provenance and 'adCreatedAt' are preserved; only 'adUpdatedAt' is bumped);
  137 -- if not, a fresh def is created. The name, provider, model, system prompt,
  138 -- and tool list are recorded in full (agent-visible data); 'orRecorded'
  139 -- carries the id + op name + fields + @was_new@ (+ @unknown_tools@ when the
  140 -- tools list names opcodes the harness does not have — a def-author typo
  141 -- is discoverable in the audit trail). The optional @role@ field gates
  142 -- delegation (\"orchestrator\" | \"leaf\"); anything else fails the
  143 -- authorize gate (role controls spawning — it must be explicit, never
  144 -- permissive). Optional @description@ is a one-line catalog summary; both
  145 -- are sanitized via 'sanitizeAgentDefFields'.
  146 agentDefWriteOp :: AgentDefBackend -> SessionId -> Opcode
  147 agentDefWriteOp backend session = TrustedOpcode
  148   { toName = OpName "AGENT_DEF_WRITE"
  149   , toTrust = Trusted
  150   , toDesc = "Create or update an agent definition by id (upsert; preserves provenance on update)."
  151   , toInSchema = object
  152       [ "type" .= ("object" :: Text)
  153       , "properties" .= object
  154           [ fromText "id" .= object
  155               [ "type" .= ("string" :: Text)
  156               , "description" .= ("Agent def id ([A-Za-z0-9_-]+)." :: Text)
  157               ]
  158           , fromText "name" .= object
  159               [ "type" .= ("string" :: Text)
  160               , "description" .= ("Human-readable agent name." :: Text)
  161               ]
  162           , fromText "provider" .= object
  163               [ "type" .= ("string" :: Text)
  164               , "description" .= ("Provider label, e.g. \"ollama\"." :: Text)
  165               ]
  166           , fromText "model" .= object
  167               [ "type" .= ("string" :: Text)
  168               , "description" .= ("Model id, e.g. \"llama3\"." :: Text)
  169               ]
  170           , fromText "system" .= object
  171               [ "type" .= ("string" :: Text)
  172               , "description" .= ("Optional system prompt." :: Text)
  173               ]
  174           , fromText "tools" .= object
  175               [ "type" .= ("array" :: Text)
  176               , "description" .= ("Allowed opcode names, or \"all\"." :: Text)
  177               ]
  178           , fromText "group" .= object
  179               [ "type" .= ("string" :: Text)
  180               , "description" .= ("Optional display group (e.g. \"core\"). Omit for the default (ungrouped) section." :: Text)
  181               ]
  182           , fromText "role" .= object
  183               [ "type" .= ("string" :: Text)
  184               , "description" .= ("Optional delegation role: \"orchestrator\" (may spawn sub-agents, depth-capped) or \"leaf\" (default; cannot spawn). The def is authoritative — AGENT_START's per-task role can only narrow an orchestrator def to leaf, never widen a leaf." :: Text)
  185               ]
  186           , fromText "description" .= object
  187               [ "type" .= ("string" :: Text)
  188               , "description" .= ("Optional one-line description rendered into the <available_agents> catalog (single line; control characters and catalog-fence tokens are sanitized)." :: Text)
  189               ]
  190           ]
  191       , "required" .= (["id", "name", "provider", "model"] :: [Text])
  192       ]
  193   , toOutSchema = object []
  194   , toAuthorize = \v ->
  195       case idField v of
  196         Nothing -> Left "AGENT_DEF_WRITE requires {id:string}"
  197         Just idTxt -> case checkId idTxt of
  198           Left e -> Left e
  199           Right () -> case T.strip <$> textFieldMaybe "role" v of
  200             Nothing -> Right ()
  201             Just "" -> Right ()   -- empty role = unset (leaf)
  202             Just r
  203               | r == "orchestrator" || r == "leaf" -> Right ()
  204               | otherwise ->
  205                 Left ("AGENT_DEF_WRITE: role must be \"orchestrator\" or \"leaf\" (got: " <> r <> ")")
  206   , toBlocking = False
  207   , toRun = \_ v -> do
  208       let mId = idField v >>= either (const Nothing) Just . mkAgentDefId
  209           roleField vv = case T.strip <$> textFieldMaybe "role" vv of
  210             Just ""  -> Nothing
  211             r        -> r
  212       case mId of
  213         Nothing -> pure (OpResult [TrpText "invalid agent def id"] True (object []))
  214         Just aid -> do
  215           mExisting <- liftIO (adbRead backend aid)
  216           now <- liftIO getCurrentTime
  217           let (def0, wasNew) = case mExisting of
  218                 Just existing ->
  219                   ( existing
  220                       { adName = textField "name" v
  221                       , adSystem = textFieldMaybe "system" v
  222                       , adTools = toolsField v
  223                       , adGroup = groupField v (Just existing)
  224                       , adRole = roleField v
  225                       , adDescription = sanitizeAgentTextField agentFieldCapSmall <$> textFieldMaybe "description" v
  226                       , adUpdatedAt = now
  227                       }
  228                   , False
  229                   )
  230                 Nothing ->
  231                   ( AgentDef
  232                       { adId = aid
  233                       , adName = textField "name" v
  234                       , adProvider = textField "provider" v
  235                       , adModel = ModelId (textField "model" v)
  236                       , adSystem = textFieldMaybe "system" v
  237                       , adTools = toolsField v
  238                       , adGroup = groupField v Nothing
  239                       , adRole = roleField v
  240                       , adDescription = sanitizeAgentTextField agentFieldCapSmall <$> textFieldMaybe "description" v
  241                       , adCreatedAt = now
  242                       , adUpdatedAt = now
  243                       , adSession = session
  244                       }
  245                   , True
  246                   )
  247               def = sanitizeAgentDefFields def0
  248               unknownTools =
  249                 case adTools def of
  250                   AllowOnly xs ->
  251                     [ t | OpName t <- Set.toList xs, Set.notMember t knownOpNames ]
  252                   AllowAll -> []
  253           liftIO (adbUpdate backend def)
  254           let recorded = encodeDefRecorded def wasNew unknownTools
  255           pure (OpResult [TrpText (if wasNew then "defined" else "updated")] False recorded)
  256   }
  257   where
  258     checkId t = either (Left . ("invalid agent def id: " <>)) (const (Right ())) (mkAgentDefId t)
  259     -- The universe of opcode names the harness actually exposes. A tools
  260     -- entry outside this set is silently dropped from the child registry
  261     -- (intersection semantics) but recorded here so a def-author typo is
  262     -- discoverable in the audit trail.
  263     knownOpNames :: Set.Set Text
  264     knownOpNames = Set.fromList
  265       [ "SHOW_HUMAN", "ASK_HUMAN", "SECRET_GET"
  266       , "MEMORY_WRITE", "MEMORY_RECALL", "MEMORY_DELETE"
  267       , "SKILL_WRITE", "SKILL_LOAD", "SKILL_LIST", "SKILL_DELETE"
  268       , "AGENT_DEF_WRITE", "AGENT_DEF_READ", "AGENT_DEF_LIST", "AGENT_DEF_DELETE"
  269       , "AGENT_INSTANCES", "AGENT_START", "AGENT_STATUS", "AGENT_STOP", "AGENT_INTERRUPT"
  270       , "SEARCH_FILES", "FILE_READ", "FILE_WRITE", "FILE_PATCH"
  271       , "SHELL_EXEC", "SETUP_REPO", "BIN_EXEC", "PROCESS_MANAGE"
  272       , "WEB_FETCH", "WEB_SEARCH"
  273       , "HARNESS_LIST", "HARNESS_START", "HARNESS_STOP"
  274       , "OPCODE_DESCRIBE", "OPCODE_LIST"
  275       ]

  401 -- ---------------------------------------------------------------------------
  402 -- AGENT_START (synchronous goal-driven delegation)
  403 -- ---------------------------------------------------------------------------
  404 
  405 -- | The wiring-layer bundle the AGENT_START opcode closes over. The
  406 -- 'AgentWorkerBuilder' resolves the def's provider+model, opens a fresh
  407 -- two-file transcript under @\<parent-session\>\/agents\/\<child-id\>@, builds
  408 -- a fresh 'AgentEnv' bound to the new session + child transcript, runs the
  409 -- turn with the goal as the first user message, and reports the outcome via
  410 -- 'ChildWorkerOutcome'. The 'DelegationConfig' / 'SpawnPauseFlag' /
  411 -- 'ParentActivity' are process-global (or per-channel) and threaded in
  412 -- here so the opcode doesn't read config.
  413 data AgentStartWiring = AgentStartWiring
  414   { aswDefBackend   :: AgentDefBackend
  415   , aswRuntime      :: AgentRuntime
  416   , aswConfig       :: IO DelegationConfig
  417     -- ^ Re-read the [delegation] config per AGENT_START call (so config
  418     -- changes take effect without a restart). The IO action reads
  419     -- @config.toml@ and returns the resolved 'DelegationConfig'.
  420   , aswPauseFlag    :: SpawnPauseFlag
  421   , aswParentActivity :: Maybe ParentActivity
  422   , aswMintSession  :: IO SessionId
  423     -- ^ Mint a fresh 'SessionId' for a child.
  424   , aswParentDepth  :: Int
  425     -- ^ The parent's delegation depth (0 for a top-level turn).
  426   , aswWorker       :: AgentWorkerBuilder
  427     -- ^ The worker-builder (closes over per-turn 'AgentEnv' deps).
  428   , aswGate         :: AgentStartGate
  429     -- ^ The role/kill-switch gate (issue #154 §3.2): a nested (child-side)
  430     -- AGENT_START carries the spawning child's effective role + the
  431     -- resolved kill-switch state so the op can reject with the dedicated
  432     -- leaf/kill-switch messages. A top-level turn's wiring passes
  433     -- 'gateOpen'.
  434   }
  435 
  436 -- | The role/switch condition the nested AGENT_START enforces before it
  437 -- will spawn. Leaf children (and orchestrator children while the kill
  438 -- switch is off) get a present-but-rejecting op whose authorize returns
  439 -- the dedicated error.
  440 data AgentStartGate = AgentStartGate
  441   { gEffectiveRole :: Maybe Text
  442     -- ^ The spawning agent's effective role ('Nothing' = leaf).
  443   , gOrchEnabled   :: Bool
  444     -- ^ The resolved @delegation.orchestrator_enabled@.
  445   }
  446 
  447 -- | The open gate for top-level (operator-authorized) turns: spawning is
  448 -- governed only by the depth cap, spawn-pause, and per-spawn resolver
  449 -- checks.
  450 gateOpen :: AgentStartGate
  451 gateOpen = AgentStartGate { gEffectiveRole = Just "orchestrator", gOrchEnabled = True }
  452 
  453 -- | AGENT_START: spawn one or more child agents, run each against a goal to
  454 -- completion, return a JSON result per child. Input is either
  455 -- @{id, goal, context?, role?}@ (single mode) or @{tasks: [{id, goal, context?, role?}, ...]}@
  456 -- (batch mode). Returns JSON with a @results@ array, one entry per task.
  457 agentStartOp :: AgentStartWiring -> Opcode
  458 agentStartOp wiring = TrustedOpcode
  459   { toName = OpName "AGENT_START"
  460   , toTrust = Trusted
  461   , toDesc = "Spawn one or more child agents, run each against a goal to completion, return a JSON result per child. Single mode: {id, goal, context?, role?}. Batch mode: {tasks: [{id, goal, context?, role?}, ...]}."
  462   , toInSchema = object
  463       [ "type" .= ("object" :: Text)
  464       , "properties" .= object
  465           [ fromText "id" .= object
  466               [ "type" .= ("string" :: Text)
  467               , "description" .= ("Agent def id (single-task mode)." :: Text)
  468               ]
  469           , fromText "goal" .= object
  470               [ "type" .= ("string" :: Text)
  471               , "description" .= ("The task goal — becomes the child's first user message (single-task mode)." :: Text)
  472               ]
  473           , fromText "context" .= object
  474               [ "type" .= ("string" :: Text)
  475               , "description" .= ("Optional background context appended to the child's system prompt." :: Text)
  476               ]
  477           , fromText "role" .= object
  478               [ "type" .= ("string" :: Text)
  479               , "description" .= ("Optional narrow-only role hint: only \"leaf\" is meaningful per-task (downgrades an orchestrator def's child to leaf). Spawning capability comes from the def's role field — a leaf def can never be widened by task input." :: Text)
  480               ]
  481           , fromText "tasks" .= object
  482               [ "type" .= ("array" :: Text)
  483               , "description" .= ("Batch mode: array of {id, goal, context?, role?}. Cap on parallelism is delegation.max_concurrent_children." :: Text)
  484               ]
  485           ]
  486       , "required" .= (["goal"] :: [Text])
  487       ]
  488   , toOutSchema = object []
  489   , toAuthorize = \v ->
  490       let shapeGate =
  491             -- Require either a top-level goal (single) or a tasks array (batch).
  492             let hasGoal = case textFieldMaybe "goal" v of { Just _ -> True; Nothing -> False }
  493                 hasTasks = case parseMaybe (withObject "in" (.:? "tasks")) v :: Maybe (Maybe Value) of { Just (Just _) -> True; _ -> False }
  494             in if hasGoal || hasTasks
  495                  then Right ()
  496                  else Left "AGENT_START requires {goal:string} (single) or {tasks:array} (batch)."
  497           roleGate = case (gEffectiveRole (aswGate wiring), gOrchEnabled (aswGate wiring)) of
  498             (Just "orchestrator", True) -> Right ()
  499             (Just "orchestrator", False) -> Left killSwitchMsg
  500             (_, _) -> Left leafMsg
  501       in shapeGate *> roleGate
  502   , toBlocking = False
  503   , toRun = \_ v -> do
  504       input <- liftIO (parseInput v)
  505       case input of
  506         Left err -> pure (OpResult [TrpText err] True (object []))
  507         Right di -> do
  508           cfg <- liftIO (aswConfig wiring)
  509           let (_, _, _, orchEnabled) = resolveDelegationConfig cfg
  510           eResults <- liftIO (runDelegate
  511                                 cfg
  512                                 (aswPauseFlag wiring)
  513                                 (aswParentActivity wiring)
  514                                 (aswParentDepth wiring)
  515                                 di
  516                                 (resolveTask (aswDefBackend wiring)
  517                                              (aswRuntime wiring)
  518                                              (aswMintSession wiring)
  519                                              (aswParentDepth wiring)
  520                                              orchEnabled
  521                                              (aswWorker wiring)))
  522           case eResults of
  523             Left err -> pure (OpResult [TrpText err] True (object []))
  524             Right results -> do
  525               -- Register each finished child in the runtime registry so
  526               -- AGENT_INSTANCES / AGENT_STATUS / AGENT_STOP can observe
  527               -- it after the synchronous run completes. The child is
  528               -- recorded with status Stopped (the synchronous worker has
  529               -- already finished by the time we reach here). Zipped with
  530               -- the input tasks to recover the AgentDefId (the ChildResult
  531               -- carries the SubagentId but not the def id).
  532               let tasks = diTasks di
  533                   childDepth = aswParentDepth wiring + 1
  534               zipWithM_ (\t r -> liftIO (registerChild (aswRuntime wiring) childDepth t r)) tasks results
  535               let rendered = encodeResultsJson results
  536               pure (OpResult [TrpText rendered] False (object ["results" .= results]))
  537    }

  539 -- | Parse the model's input into a 'DelegateInput'. Single mode requires
  540 -- @id@ + @goal@; batch mode requires a @tasks@ array of @{id, goal, ...}@.
  541 parseInput :: Value -> IO (Either Text DelegateInput)
  542 parseInput v =
  543   case parseMaybe (withObject "in" (.:? "tasks")) v :: Maybe (Maybe Value) of
  544     Just (Just (Array arr)) | not (V.null arr) -> do
  545       tasks <- mapM parseTask (V.toList arr)
  546       case sequence tasks of
  547         Left err     -> pure (Left err)
  548         Right tsList -> pure (Right (DiBatch tsList))
  549     _ -> do
  550       let mDefId = idField v
  551           mGoal  = textFieldMaybe "goal" v
  552       case (mDefId, mGoal) of
  553         (Just defId, Just goal) | not (T.null goal) ->
  554           pure (Right (DiSingle (ChildTask defId goal (textFieldMaybe "context" v) (textFieldMaybe "role" v))))
  555         (Just _, Just _) -> pure (Left "AGENT_START requires a non-empty 'goal'.")
  556         (Just _, Nothing) -> pure (Left "AGENT_START single-task mode requires a 'goal'.")
  557         (Nothing, _) -> pure (Left "AGENT_START requires an 'id' (agent def id) in single-task mode, or a 'tasks' array in batch mode.")
  558 
  559 -- | Parse one element of the @tasks@ array into a 'ChildTask'.
  560 parseTask :: Value -> IO (Either Text ChildTask)
  561 parseTask v =
  562   case idField v of
  563     Nothing -> pure (Left "Each task requires an 'id' (agent def id).")
  564     Just defId ->
  565       case textFieldMaybe "goal" v of
  566         Nothing -> pure (Left "Each task requires a 'goal'.")
  567         Just goal | T.null goal -> pure (Left "Each task requires a non-empty 'goal'.")
  568                  | otherwise -> pure (Right (ChildTask defId goal (textFieldMaybe "context" v) (textFieldMaybe "role" v)))
  569 
  570 -- | Resolve a task to its def + worker + fresh session id. Returns Left if
  571 -- the def id is invalid, the def doesn't exist, or the effective-role /
  572 -- kill-switch gate rejects the spawn (issue #154 §3.2 item 5 — the
  573 -- dispatch-time TOCTOU gate: the def is in hand here, which is the only
  574 -- place the effective role is computable). @orchEnabled@ is the resolved
  575 -- @delegation.orchestrator_enabled@; @parentDepth@ is the spawning
  576 -- parent's depth (consumed for the error message context and forwarded
  577 -- via the wiring, not stored here).
  578 resolveTask
  579   :: AgentDefBackend
  580   -> AgentRuntime
  581   -> IO SessionId
  582   -> Int
  583   -> Bool
  584   -> AgentWorkerBuilder
  585   -> ChildTask
  586   -> IO (Either Text (AgentDef, AgentWorkerBuilder, SessionId))
  587 resolveTask defBackend _runtime mintSession _parentDepth orchEnabled worker task = do
  588   case mkAgentDefId (ctDefId task) of
  589     Left err -> pure (Left err)
  590     Right aid -> do
  591       mDef <- adbRead defBackend aid
  592       case mDef of
  593         Nothing  -> pure (Left ("agent def not found: " <> ctDefId task))
  594         Just def -> do
  595           let role = effectiveRole (adRole def) (ctRole task)
  596           if role == Just "orchestrator" && not orchEnabled
  597             then pure (Left killSwitchMsg)
  598             else do
  599               sid <- mintSession
  600               pure (Right (def, worker, sid))
  601 
  602 -- | The dedicated kill-switch error (§3.2 item 6). Distinct from the
  603 -- depth/leaf/pause messages so the parent transcript distinguishes all
  604 -- spawn-failure causes.
  605 killSwitchMsg :: Text
  606 killSwitchMsg = "Delegation spawning is disabled: delegation.orchestrator_enabled = false. Re-trying will not succeed until the operator re-enables it."
  607 
  608 -- | The dedicated leaf-role error (§3.2 item 6): a leaf agent cannot
  609 -- spawn — actionable for both the model and the operator.
  610 leafMsg :: Text
  611 leafMsg = "AGENT_START is not available to this agent: its definition is a leaf (role: leaf). Ask the operator to grant the orchestrator role if delegation is required."
  612 
  613 -- | Register a finished child in the runtime registry (post-hoc; the worker
  614 -- ran synchronously to completion). Records the instance with status
  615 -- 'Stopped' (the synchronous child has already finished by the time this is
  616 -- called), so AGENT_INSTANCES / AGENT_STATUS / AGENT_STOP can observe it.
  617 -- Recovers the 'AgentDefId' from the task's @ctDefId@ (the ChildResult
  618 -- carries the 'SubagentId' but not the def id). The recorded depth is the
  619 -- CHILD's own depth — the wiring's parent depth + 1 (issue #154: the
  620 -- former hardcoded 0 recorded every child at the root).
  621 registerChild :: AgentRuntime -> Int -> ChildTask -> ChildResult -> IO ()
  622 registerChild runtime childDepth task result =
  623   case mkAgentDefId (ctDefId task) of
  624     Left _ -> pure ()  -- malformed def id; skip registration
  625     Right aid ->
  626       case crChildSession result of
  627         Nothing -> pure ()  -- no child session; skip (error/timeout case)
  628         Just session -> do
  629           registerCompletedAgent runtime aid (crSubagentId result) session childDepth
  630 
  631 -- | Extract the task list from a 'DelegateInput' (in order).
  632 diTasks :: DelegateInput -> [ChildTask]
  633 diTasks (DiSingle t) = [t]
  634 diTasks (DiBatch ts) = ts

  640 agentStatusOp :: AgentRuntime -> Opcode
  641 agentStatusOp runtime = TrustedOpcode
  642   { toName = OpName "AGENT_STATUS"
  643   , toTrust = Trusted
  644   , toDesc = "Read one running agent's status by subagent_id."
  645   , toInSchema = singleStringSchema "subagent_id" "The subagent id (from AGENT_START's result)."
  646   , toOutSchema = object []
  647   , toAuthorize = maybe (Left "AGENT_STATUS requires {subagent_id:string}") checkSubagentId . subagentIdField
  648   , toBlocking = False
  649   , toRun = \_ v -> do
  650       let mSid = subagentIdField v
  651       case mSid of
  652         Nothing -> pure (OpResult [TrpText "invalid subagent id"] True (object []))
  653         Just sid -> do
  654           mStatus <- liftIO (agentStatus runtime sid)
  655           case mStatus of
  656             Nothing -> pure (OpResult [TrpText "not running"] False (object ["subagent_id" .= subagentIdText sid, "status" .= ("stopped" :: Text)]))
  657             Just s  -> pure (OpResult [TrpText (renderStatus s)] False (object ["subagent_id" .= subagentIdText sid, "status" .= renderStatus s]))
  658   }
  659   where
  660     checkSubagentId _ = Right ()

  691 -- | AGENT_INTERRUPT: request that a single running subagent stop at its next
  692 -- iteration boundary. Unlike AGENT_STOP (which hard-kills the thread),
  693 -- AGENT_INTERRUPT sets a flag the worker polls between turns, letting it
  694 -- exit cleanly. Trusted.
  695 agentInterruptOp :: AgentRuntime -> Opcode
  696 agentInterruptOp runtime = TrustedOpcode
  697   { toName = OpName "AGENT_INTERRUPT"
  698   , toTrust = Trusted
  699   , toDesc = "Request that a running subagent stop at its next iteration boundary (cooperative; the worker polls an interrupt flag between turns)."
  700   , toInSchema = singleStringSchema "subagent_id" "The subagent id to interrupt."
  701   , toOutSchema = object []
  702   , toAuthorize = maybe (Left "AGENT_INTERRUPT requires {subagent_id:string}") (const (Right ())) . subagentIdField
  703   , toBlocking = False
  704   , toRun = \_ v -> do
  705       let mSid = subagentIdField v
  706       case mSid of
  707         Nothing -> pure (OpResult [TrpText "invalid subagent id"] True (object []))
  708         Just sid -> do
  709           found <- liftIO (interruptAgent runtime sid)
  710           let msg = if found then "interrupt requested" else "subagent not running"
  711           pure (OpResult [TrpText msg] False (object ["subagent_id" .= subagentIdText sid, "found" .= found]))
  712   }

  714 -- ---------------------------------------------------------------------------
  715 -- JSON encoding of ChildResult
  716 -- ---------------------------------------------------------------------------
  717 
  718 -- | Render the results list as a JSON string for the model (the @orParts@
  719 -- text the model sees). One line per result:
  720 -- @subagent_id | status | summary-or-error@. The full structured JSON goes
  721 -- into 'orRecorded' via the 'ToJSON ChildResult' instance defined in
  722 -- 'Seal.Agent.Runtime.Delegation'.
  723 encodeResultsJson :: [ChildResult] -> Text
  724 encodeResultsJson [] = "(no results)"
  725 encodeResultsJson rs = T.intercalate "\n" (map renderOne rs)
  726   where
  727     renderOne r =
  728       subagentIdText (crSubagentId r) <> " | " <>
  729       T.pack (show (crStatus r)) <> " | " <>
  730       fromMaybe "(no summary)" (summaryOrError r)
  731     summaryOrError r = case crSummary r of
  732       Just s  -> Just s
  733       Nothing -> crError r
  734 
  735 renderStatus :: AgentStatus -> Text
  736 renderStatus = \case
  737   Starting     -> "starting"
  738   Running      -> "running"
  739   Idle         -> "idle"
  740   Stopped      -> "stopped"
  741   Interrupted  -> "interrupted"
  742   Crashed m    -> "crashed: " <> m
  743 
  744 -- | Extract the @subagent_id@ string field from a JSON object.
  745 subagentIdField :: Value -> Maybe SubagentId
  746 subagentIdField v = do
  747   t <- parseMaybe (withObject "in" (.: "subagent_id")) v
  748   if T.null t then Nothing else Just (SubagentId t)

  754 -- | Build a JSON-Schema object with a single required string property.
  755 singleStringSchema :: Text -> Text -> Value
  756 singleStringSchema fieldName fieldDesc =
  757   object
  758     [ "type" .= ("object" :: Text)
  759     , "properties" .= object
  760         [fromText fieldName .= object
  761            [ "type" .= ("string" :: Text)
  762            , "description" .= fieldDesc
  763            ]]
  764     , "required" .= ([fieldName] :: [Text])
  765     ]

  767 -- | Encode the secret-free 'AgentDef' fields into the 'orRecorded' payload.

  770 encodeDefRecorded d wasNew unknownTools = object $

  776   , "tools"      .= encodeTools (adTools d)

  781   , "updated_at" .= adUpdatedAt d

  792 -- | Encode an 'AllowList OpName' for the recorded payload: @\"all\"@ for

  796 encodeTools (AllowOnly xs) = Array (V.fromList [ String t | OpName t <- Set.toList xs ])

  798 -- | Render an 'AgentDef' as a Markdown-ish text block for the model.

  802   <> "provider: " <> adProvider d <> "\n"
  803   <> "model: " <> modelName <> "\n"
  804   <> "system: " <> fromMaybe "(none)" (adSystem d) <> "\n"
  805   <> "tools: " <> renderTools (adTools d)

  812 renderTools (AllowOnly xs) = T.intercalate ", " [ t | OpName t <- Set.toList xs ]

    2 -- | The Agent opcode group: three Audited definition opcodes
    3 -- (@AGENT_DEF_WRITE@, @AGENT_DEF_READ@, @AGENT_DEF_LIST@, @AGENT_DEF_DELETE@)
    4 -- and four Trusted lifecycle opcodes (@AGENT_INSTANCES@, @AGENT_START@,
    5 -- @AGENT_STATUS@, @AGENT_STOP@, @AGENT_INTERRUPT@). Only the *definition*
    6 -- mutations are Audited — running an instance is harness-internal, not an
    7 -- evolutionary mutation, so the lifecycle ops are Trusted.
    8 --
    9 -- @AGENT_DEF_WRITE@ is an upsert: if the def already exists, its name/system/
   10 -- tools are updated (the original 'adSession' provenance and 'adCreatedAt' are
   11 -- preserved; only 'adUpdatedAt' is bumped); if not, a fresh def is created.
   12 -- This merges the former AGENT_DEF_CREATE + AGENT_DEF_UPDATE into a single
   13 -- opcode. @orRecorded@ carries a @was_new@ flag so the audit log still
   14 -- distinguishes create vs update.
   15 --
   16 -- @AGENT_INSTANCES@ (renamed from @AGENT_LIST@) snapshots the in-process agent
   17 -- runtime (running instances), NOT the definitions. @AGENT_DEF_LIST@ lists the
   18 -- definitions. The rename stops the confusion between "list definitions" and
   19 -- "list running instances".
   20 --
   21 -- @AGENT_START@ is the Seal analog of Hermes' @delegate_task@: it spawns one
   22 -- or more child agents with isolated context, runs each against a goal to
   23 -- completion (synchronously), and returns a structured JSON result per child.
   24 -- The parent blocks until all children finish (or time out). See
   25 -- 'Seal.Agent.Runtime.Delegation' for the full feature list. The opcode is a
   26 -- thin shim over 'runDelegate' — it normalizes the model's input, resolves the
   27 -- def, delegates the worker construction to the wiring layer's
   28 -- 'AgentWorkerBuilder', and serializes the 'ChildResult' list to JSON.

  106 -- | Extract the @id@ string field from a JSON object.
  107 idField :: Value -> Maybe Text
  108 idField = parseMaybe (withObject "in" (.: "id"))
  109 
  110 -- | Extract a required text field (defaults to empty when absent).
  111 textField :: Text -> Value -> Text
  112 textField name v = fromMaybe "" (parseMaybe (withObject "in" (.: fromText name)) v)
  113 
  114 -- | Extract an optional text field (defaults to 'Nothing' when absent).
  115 textFieldMaybe :: Text -> Value -> Maybe Text
  116 textFieldMaybe name v =
  117   case parseMaybe (withObject "in" (.:? fromText name)) v :: Maybe (Maybe Text) of
  118     Just (Just t) -> Just t
  119     _             -> Nothing
  120 
  121 -- | Extract an optional boolean field (defaults to 'Nothing' when absent
  122 -- or non-boolean). Mirrors 'Seal.ISA.Ops.Session.boolField'.
  123 boolField :: Value -> Text -> Maybe Bool
  124 boolField v key = join (parseMaybe (withObject "in" (.:? fromText key)) v)
  125 
  126 -- | Decode the @tools@ field: @\"all\"@ (or absent) -> 'AllowAll'; an array of
  127 -- opcode-name strings -> 'AllowOnly'. Malformed -> 'AllowAll' (permissive).
  128 toolsField :: Value -> AllowList OpName
  129 toolsField v =
  130   case parseMaybe (withObject "in" (.:? "tools")) v :: Maybe (Maybe Value) of
  131     Just (Just (String "all")) -> AllowAll
  132     Just (Just (Array xs))     -> AllowOnly (Set.fromList [ OpName t | String t <- V.toList xs ])
  133     _                          -> AllowAll
  134 
  135 -- | Resolve the @group@ field from the opcode input, preserving the existing
  136 -- def's group when the input omits it (mirrors the gateway @stampAgentDef@
  137 -- and the skill @stampSkill@ back-compat policy). An explicit empty string
  138 -- clears the group (sets 'Nothing'); a non-empty stripped value sets it.
  139 groupField :: Value -> Maybe AgentDef -> Maybe Text
  140 groupField v mExisting =
  141   case textFieldMaybe "group" v of
  142     Just g  -> let g' = T.strip g in if T.null g' then Nothing else Just g'
  143     Nothing -> adGroup =<< mExisting

  149 data AgentDefAction = AdWrite | AdRead | AdList | AdDelete
  150 
  151 parseAgentDefAction :: Value -> Either Text AgentDefAction
  152 parseAgentDefAction v =
  153   case parseMaybe (withObject "in" (.: "action")) v of
  154     Just t -> case t :: Text of
  155       "write"  -> Right AdWrite
  156       "read"   -> Right AdRead
  157       "list"   -> Right AdList
  158       "delete" -> Right AdDelete
  159       other    -> Left ("unknown agent def action: " <> other)
  160     Nothing -> Left "missing or invalid action field"
  161 
  162 -- | Shared id validator (used by write/read/delete).
  163 checkAgentDefId :: Text -> Either Text ()
  164 checkAgentDefId t = either (Left . ("invalid agent def id: " <>)) (const (Right ())) (mkAgentDefId t)
  165 
  166 -- | Authorize gate for AGENT_DEF_MANAGE — dispatches per-action validation.
  167 authorizeAgentDefManage :: Value -> Either Text ()
  168 authorizeAgentDefManage v =
  169   case parseAgentDefAction v of
  170     Left e -> Left e
  171     Right action -> case action of
  172       AdWrite  -> authorizeDefWrite v
  173       AdRead   -> maybe (Left "read requires {id:string}") checkAgentDefId . idField $ v
  174       AdList   -> Right ()
  175       AdDelete -> maybe (Left "delete requires {id:string}") checkAgentDefId . idField $ v
  176 
  177 -- | Authorize the write action (shared with the legacy AGENT_DEF_WRITE shim).
  178 authorizeDefWrite :: Value -> Either Text ()
  179 authorizeDefWrite v =
  180   case idField v of
  181     Nothing -> Left "AGENT_DEF_WRITE requires {id:string}"
  182     Just idTxt -> case checkAgentDefId idTxt of
  183       Left e -> Left e
  184       Right () -> case T.strip <$> textFieldMaybe "role" v of
  185         Nothing -> Right ()
  186         Just "" -> Right ()
  187         Just r
  188           | r == "orchestrator" || r == "leaf" -> Right ()
  189           | otherwise ->
  190             Left ("AGENT_DEF_WRITE: role must be \"orchestrator\" or \"leaf\" (got: " <> r <> ")")

  192 -- | Handle the write action (shared with the legacy AGENT_DEF_WRITE shim).
  193 handleDefWrite :: AgentDefBackend -> SessionId -> Value -> App OpResult
  194 handleDefWrite backend session v = do
  195   let mId = idField v >>= either (const Nothing) Just . mkAgentDefId
  196       roleField vv = case T.strip <$> textFieldMaybe "role" vv of
  197         Just ""  -> Nothing
  198         r        -> r
  199   case mId of
  200     Nothing -> pure (OpResult [TrpText "invalid agent def id"] True (object []))
  201     Just aid -> do
  202       mExisting <- liftIO (adbRead backend aid)
  203       now <- liftIO getCurrentTime
  204       let (def0, wasNew) = case mExisting of
  205             Just existing ->
  206               ( existing
  207                   { adName = textField "name" v
  208                   , adSystem = textFieldMaybe "system" v
  209                   , adTools = toolsField v
  210                   , adGroup = groupField v (Just existing)
  211                   , adRole = roleField v
  212                   , adDescription = sanitizeAgentTextField agentFieldCapSmall <$> textFieldMaybe "description" v
  213                   , adUpdatedAt = now
  214                   }
  215               , False
  216               )
  217             Nothing ->
  218               ( AgentDef
  219                   { adId = aid
  220                   , adName = textField "name" v
  221                   , adProvider = textField "provider" v
  222                   , adModel = ModelId (textField "model" v)
  223                   , adSystem = textFieldMaybe "system" v
  224                   , adTools = toolsField v
  225                   , adGroup = groupField v Nothing
  226                   , adRole = roleField v
  227                   , adDescription = sanitizeAgentTextField agentFieldCapSmall <$> textFieldMaybe "description" v
  228                   , adCreatedAt = now
  229                   , adUpdatedAt = now
  230                   , adSession = session
  231                   }
  232               , True
  233               )
  234           def = sanitizeAgentDefFields def0
  235           unknownTools =
  236             case adTools def of
  237               AllowOnly xs ->
  238                 [ t | OpName t <- Set.toList xs, Set.notMember t knownOpNames ]
  239               AllowAll -> []
  240       liftIO (adbUpdate backend def)
  241       let recorded = encodeDefRecorded def wasNew unknownTools
  242       pure (OpResult [TrpText (if wasNew then "defined" else "updated")] False recorded)
  243 
  244 -- | Handle the read action (shared with the legacy AGENT_DEF_READ shim).
  245 handleDefRead :: AgentDefBackend -> Value -> App OpResult
  246 handleDefRead backend v = do
  247   let mId = idField v >>= either (const Nothing) Just . mkAgentDefId
  248   case mId of
  249     Nothing -> pure (OpResult [TrpText "invalid agent def id"] True (object []))
  250     Just aid -> do
  251       mDef <- liftIO (adbRead backend aid)
  252       case mDef of
  253         Nothing -> pure (OpResult [TrpText "agent def not found"] True (object ["id" .= agentDefIdText aid]))
  254         Just d  -> do
  255           let rendered = renderDef d
  256               recorded = encodeDefRecorded d False []
  257           pure (OpResult [TrpText rendered] False recorded)
  258 
  259 -- | Handle the list action (shared with the legacy AGENT_DEF_LIST shim).
  260 handleDefList :: AgentDefBackend -> App OpResult
  261 handleDefList backend = do
  262   defs <- liftIO (adbList backend)
  263   let rendered = case defs of
  264         [] -> "(no agent definitions)"
  265         _  -> T.intercalate "\n"
  266                 [ agentDefIdText (adId d) <> roleSuffix (adRole d) <> ": " <> adName d
  267                     <> " (" <> adProvider d <> "/" <> modelName <> ")"
  268                 | d <- defs, let ModelId modelName = adModel d ]
  269       recorded = object
  270         [ "count" .= length defs
  271         , "ids" .= fmap (agentDefIdText . adId) defs
  272         , "roles" .= object
  273             [ fromText (agentDefIdText (adId d)) .= adRole d | d <- defs ]
  274         ]
  275   pure (OpResult [TrpText rendered] False recorded)
  276 
  277 -- | Handle the delete action (shared with the legacy AGENT_DEF_DELETE shim).
  278 handleDefDelete :: AgentDefBackend -> Value -> App OpResult
  279 handleDefDelete backend v = do
  280   let mId = idField v >>= either (const Nothing) Just . mkAgentDefId
  281   case mId of
  282     Nothing -> pure (OpResult [TrpText "invalid agent def id"] True (object []))
  283     Just aid -> do
  284       mExisting <- liftIO (adbRead backend aid)
  285       liftIO (adbDelete backend aid)
  286       let msg = case mExisting of
  287             Nothing -> "deleted (was not present)"
  288             Just _  -> "deleted"
  289           recorded = object ["id" .= agentDefIdText aid]
  290       pure (OpResult [TrpText msg] False recorded)

  524 -- | The role/switch condition the nested AGENT_START enforces before it

  533 -- | The open gate for top-level (operator-authorized) turns: spawning is
  534 -- governed only by the depth cap, spawn-pause, and per-spawn resolver

  536 gateOpen :: AgentStartGate
  537 gateOpen = AgentStartGate { gEffectiveRole = Just "orchestrator", gOrchEnabled = True }

  539 -- | Parse the model's input into a 'DelegateInput'. Single mode requires
  540 -- @id@ + @goal@; batch mode requires a @tasks@ array of @{id, goal, ...}@.
  541 parseInput :: Value -> IO (Either Text DelegateInput)
  542 parseInput v =
  543   case parseMaybe (withObject "in" (.:? "tasks")) v :: Maybe (Maybe Value) of
  544     Just (Just (Array arr)) | not (V.null arr) -> do
  545       tasks <- mapM parseTask (V.toList arr)
  546       case sequence tasks of
  547         Left err     -> pure (Left err)
  548         Right tsList -> pure (Right (DiBatch tsList))
  549     _ -> do
  550       let mDefId = idField v
  551           mGoal  = textFieldMaybe "goal" v
  552       case (mDefId, mGoal) of
  553         (Just defId, Just goal) | not (T.null goal) ->
  554           pure (Right (DiSingle (ChildTask defId goal (textFieldMaybe "context" v) (textFieldMaybe "role" v) (fromMaybe False (boolField v "isolate_workdir")))))
  555         (Just _, Just _) -> pure (Left "AGENT_START requires a non-empty 'goal'.")
  556         (Just _, Nothing) -> pure (Left "AGENT_START single-task mode requires a 'goal'.")
  557         (Nothing, _) -> pure (Left "AGENT_START requires an 'id' (agent def id) in single-task mode, or a 'tasks' array in batch mode.")
  558 
  559 -- | Parse one element of the @tasks@ array into a 'ChildTask'.
  560 parseTask :: Value -> IO (Either Text ChildTask)
  561 parseTask v =
  562   case idField v of
  563     Nothing -> pure (Left "Each task requires an 'id' (agent def id).")
  564     Just defId ->
  565       case textFieldMaybe "goal" v of
  566         Nothing -> pure (Left "Each task requires a 'goal'.")
  567         Just goal | T.null goal -> pure (Left "Each task requires a non-empty 'goal'.")
  568                  | otherwise -> pure (Right (ChildTask defId goal (textFieldMaybe "context" v) (textFieldMaybe "role" v) (fromMaybe False (boolField v "isolate_workdir"))))
  569 
  570 -- | Resolve a task to its def + worker. Returns Left if the def id is
  571 -- invalid, the def doesn't exist, or the effective-role / kill-switch gate
  572 -- rejects the spawn. The session id is NOT minted here — 'spawnOne' in
  573 -- 'runDelegateAsync' mints the session before calling the resolver,
  574 -- eliminating the double-mint that doubled the collision surface.
  575 resolveTask
  576   :: AgentDefBackend
  577   -> AgentRuntime
  578   -> Int
  579   -> Bool
  580   -> AgentWorkerBuilder
  581   -> ChildTask
  582   -> IO (Either Text (AgentDef, AgentWorkerBuilder))
  583 resolveTask defBackend _runtime _parentDepth orchEnabled worker task = do
  584   case mkAgentDefId (ctDefId task) of
  585     Left err -> pure (Left err)
  586     Right aid -> do
  587       mDef <- adbRead defBackend aid
  588       case mDef of
  589         Nothing  -> pure (Left ("agent def not found: " <> ctDefId task))
  590         Just def -> do
  591           let role = effectiveRole (adRole def) (ctRole task)
  592           if role == Just "orchestrator" && not orchEnabled
  593             then pure (Left killSwitchMsg)
  594             else pure (Right (def, worker))
  595 
  596 -- | The dedicated kill-switch error. Distinct from the depth/leaf/pause
  597 -- messages so the parent transcript distinguishes all spawn-failure causes.
  598 killSwitchMsg :: Text
  599 killSwitchMsg = "Delegation spawning is disabled: delegation.orchestrator_enabled = false. Re-trying will not succeed until the operator re-enables it."
  600 
  601 -- | The dedicated leaf-role error: a leaf agent cannot spawn — actionable
  602 -- for both the model and the operator.
  603 leafMsg :: Text
  604 leafMsg = "AGENT_START is not available to this agent: its definition is a leaf (role: leaf). Ask the operator to grant the orchestrator role if delegation is required."

  606 -- | Register a finished child in the runtime registry (post-hoc; the worker
  607 -- ran synchronously to completion). Records the instance with status
  608 -- 'Stopped' (the synchronous child has already finished by the time this is
  609 -- ---------------------------------------------------------------------------
  610 -- Agent runtime action enum (AGENT_MANAGE)

  613 data AgentAction
  614   = AgInstances
  615   | AgStart
  616   | AgStatus
  617   | AgStop
  618   | AgInterrupt

  620 parseAgentAction :: Value -> Either Text AgentAction
  621 parseAgentAction v =
  622   case parseMaybe (withObject "in" (.: "action")) v of
  623     Just t -> case t :: Text of
  624       "instances" -> Right AgInstances
  625       "start"     -> Right AgStart
  626       "status"    -> Right AgStatus
  627       "stop"      -> Right AgStop
  628       "interrupt" -> Right AgInterrupt
  629       other       -> Left ("unknown agent action: " <> other)
  630     Nothing -> Left "missing or invalid action field"

  632 -- | Authorize gate for the `start` action (shared with the legacy
  633 -- AGENT_START shim). Checks both the input shape (goal or tasks present)
  634 -- and the role/kill-switch gate from the wiring.
  635 authorizeStart :: AgentStartWiring -> Value -> Either Text ()
  636 authorizeStart wiring v =
  637   let shapeGate =
  638         let hasGoal = case textFieldMaybe "goal" v of { Just _ -> True; Nothing -> False }
  639             hasTasks = case parseMaybe (withObject "in" (.:? "tasks")) v :: Maybe (Maybe Value) of { Just (Just _) -> True; _ -> False }
  640         in if hasGoal || hasTasks
  641              then Right ()
  642              else Left "AGENT_START requires {goal:string} (single) or {tasks:array} (batch)."
  643       roleGate = case (gEffectiveRole (aswGate wiring), gOrchEnabled (aswGate wiring)) of
  644         (Just "orchestrator", True) -> Right ()
  645         (Just "orchestrator", False) -> Left killSwitchMsg
  646         (_, _) -> Left leafMsg
  647   in shapeGate *> roleGate

  649 -- | Authorize gate for AGENT_MANAGE — dispatches per-action validation.
  650 authorizeAgentManage :: AgentStartWiring -> Value -> Either Text ()
  651 authorizeAgentManage wiring v =
  652   case parseAgentAction v of
  653     Left e -> Left e
  654     Right action -> case action of
  655       AgInstances -> Right ()
  656       AgStart     -> authorizeStart wiring v
  657       AgStatus    -> maybe (Left "status requires {subagent_id:string}") (const (Right ())) . subagentIdField $ v
  658       AgStop      -> maybe (Left "stop requires {subagent_id:string}") (const (Right ())) . subagentIdField $ v
  659       AgInterrupt -> maybe (Left "interrupt requires {subagent_id:string}") (const (Right ())) . subagentIdField $ v

  661 -- | Handle the instances action (shared with the legacy AGENT_INSTANCES shim).
  662 handleInstances :: AgentRuntime -> App OpResult
  663 handleInstances runtime = do
  664   insts <- liftIO (listAgents runtime)
  665   let rendered = case insts of
  666         [] -> "(no agents running)"
  667         _  -> T.intercalate "\n"
  668                 [ subagentIdText (aiSubagentId i) <> ": " <> agentDefIdText (aiId i) <> " — " <> renderStatus (aiStatus i)
  669                 | i <- insts ]
  670       recorded = object
  671         [ "count" .= length insts
  672         , "ids" .= fmap (subagentIdText . aiSubagentId) insts
  673         ]
  674   pure (OpResult [TrpText rendered] False recorded)

  676 -- | Handle the start action (shared with the legacy AGENT_START shim).
  677 -- Async: forks children, returns immediately with per-child SpawnInfo.
  678 handleStart :: AgentStartWiring -> Value -> App OpResult
  679 handleStart wiring v = do
  680   input <- liftIO (parseInput v)
  681   case input of
  682     Left err -> pure (OpResult [TrpText err] True (object []))
  683     Right di -> do
  684       cfg <- liftIO (aswConfig wiring)
  685       let (_, _, _, orchEnabled) = resolveDelegationConfig cfg
  686           runtime = aswRuntime wiring
  687           parentSid = aswParentSession wiring
  688           paths = aswPaths wiring
  689           callback :: AgentCompletionCallback
  690           callback result = do
  691             registerCompletedAgentResult runtime (crSubagentId result) result
  692             -- Append the completion message to a sidecar file. The turn
  693             -- engine reads this file at the start of the parent's next turn
  694             -- and injects the messages into the conversation. This avoids
  695             -- interfering with the single-writer daemon's in-memory diff
  696             -- state (tfsWritten) during the ongoing turn — writing directly
  697             -- to conversation.jsonl while the daemon is active causes the
  698             -- daemon's diff to desynchronize, corrupting the transcript.
  699             appendCompletionToSidecar paths parentSid (completionMessage result)
  700           spawnCb :: SpawnCallback
  701           spawnCb sid def childSid =
  702             registerRunningAgent runtime (adId def) sid childSid (aswParentDepth wiring + 1)
  703       eSpawnInfos <- liftIO (runDelegateAsync
  704                                cfg
  705                                (aswPauseFlag wiring)
  706                                (aswParentActivity wiring)
  707                                (aswParentDepth wiring)
  708                                di
  709                                (resolveTask (aswDefBackend wiring)
  710                                             runtime
  711                                             (aswParentDepth wiring)
  712                                             orchEnabled
  713                                             (aswWorker wiring))
  714                                callback
  715                                spawnCb
  716                                (aswMintSession wiring))
  717       case eSpawnInfos of
  718         Left err -> pure (OpResult [TrpText err] True (object []))
  719         Right spawnInfos ->
  720           pure (OpResult [TrpText (encodeSpawnInfos spawnInfos)] False
  721                  (object ["results" .= fmap toJSONSpawnInfo spawnInfos]))
  722 
  723 -- | Handle the status action (shared with the legacy AGENT_STATUS shim).
  724 -- Enriched: when the registry has a completed entry with aiResult, the
  725 -- response includes summary, child_session, exit_reason, and duration.
  726 handleStatus :: AgentRuntime -> Value -> App OpResult
  727 handleStatus runtime v = do
  728   let mSid = subagentIdField v
  729   case mSid of
  730     Nothing -> pure (OpResult [TrpText "invalid subagent id"] True (object []))
  731     Just sid -> do
  732       mInst <- liftIO (agentInstanceBySubagentId runtime sid)
  733       case mInst of
  734         Nothing -> pure (OpResult [TrpText "not running"] False
  735                          (object ["subagent_id" .= subagentIdText sid, "status" .= ("stopped" :: Text)]))
  736         Just inst -> do
  737           let s = aiStatus inst
  738               baseRecord = [ "subagent_id" .= subagentIdText sid
  739                            , "status" .= renderStatus s
  740                            ]
  741           case aiResult inst of
  742             Nothing ->
  743               pure (OpResult [TrpText (renderStatus s)] False (object baseRecord))
  744             Just result ->
  745               let enriched = baseRecord
  746                     ++ [ "summary" .= crSummary result
  747                        , "child_session" .= fmap sessionIdText (crChildSession result)
  748                        , "exit_reason" .= T.pack (show (crExitReason result))
  749                        , "duration_seconds" .= crDurationSeconds result
  750                        ]
  751                   statusText = renderStatus s <> maybe "" ("\n" <>) (crSummary result)
  752               in pure (OpResult [TrpText statusText] False (object enriched))

  754 -- | Handle the stop action (shared with the legacy AGENT_STOP shim).
  755 handleStop :: AgentRuntime -> Value -> App OpResult
  756 handleStop runtime v = do
  757   let mSid = subagentIdField v

  759     Nothing -> pure (OpResult [TrpText "invalid subagent id"] True (object []))

  761       _ <- liftIO (stopAgent runtime sid)

 1041 -- | Build the user-role harness message appended to the parent's
 1042 -- @conversation.jsonl@ when a child completes. The parent's next turn sees
 1043 -- this as a user message, so the model learns the child finished without
 1044 -- polling 'AGENT_STATUS'. The message is secret-free (only the summary,
 1045 -- status, child session, and exit reason — never @crParts@).
 1046 completionMessage :: ChildResult -> Text
 1047 completionMessage r =
 1048   let sid = subagentIdText (crSubagentId r)
 1049       status = T.pack (show (crStatus r))
 1050       mSummary = crSummary r
 1051       mChildSession = sessionIdText <$> crChildSession r
 1052       exitReason = T.pack (show (crExitReason r))
 1053       header = "[subagent " <> sid <> " completed] status=" <> status
 1054                  <> " exit=" <> exitReason
 1055       sessionLine = case mChildSession of
 1056         Just cs -> " child_session=" <> cs
 1057         Nothing -> ""
 1058       summaryLine = case mSummary of
 1059         Just s  -> "\n" <> s
 1060         Nothing -> case crError r of
 1061           Just e  -> "\nerror: " <> e
 1062           Nothing -> ""
 1063   in header <> sessionLine <> summaryLine

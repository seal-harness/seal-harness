    2 -- | The in-process agent runtime registry. Tracks live subagent instances
    3 -- spawned by AGENT_START so the parent (or operator) can list them, read
    4 -- status, and interrupt by id.
    5 --
    6 -- The registry is keyed by 'SubagentId' (def-id + random suffix), so multiple
    7 -- concurrent children of the same def don't collide. Each 'AgentInstance'
    8 -- binds a 'SubagentId' to its def id, fresh 'SessionId', status, the forked
    9 -- worker's 'ThreadId' (for interrupt), and the parent's delegation depth.
   10 --
   11 -- Scope: in-process instances only — no tmux/harness integration (that is
   12 -- the separate Phase 3 Harnesses group, a different concern). Lifecycle ops
   13 -- (@AGENT_INSTANCES@ / @AGENT_START@ / @AGENT_STATUS@ / @AGENT_STOP@ /
   14 -- @AGENT_INTERRUPT@) are Trusted (not Audited) because running an instance is
   15 -- harness-internal, not an evolutionary mutation.
   16 --
   17 -- 'startAgent' takes the worker action @IO ()@ as a parameter so the
   18 -- registry is testable without dragging in 'Seal.Agent.Loop.runTurn' /
   19 -- 'Seal.Types.Env'. The 'Seal.ISA.Ops.Agent' opcode builds the worker from a
   20 -- fresh 'AgentEnv' over a def's provider/model/system/tools and passes it in.

   50 -- | The lifecycle status of a running agent instance. 'Crashed' carries the
   51 -- exception message; 'Interrupted' means the parent/operator requested a
   52 -- stop via 'interruptAgent' (sets a flag the worker polls between turns).
   53 data AgentStatus = Starting | Running | Idle | Stopped | Interrupted | Crashed Text
   54   deriving stock (Eq, Show)
   55 
   56 -- | One running agent instance, bound to a definition, a fresh session, and
   57 -- a subagent id (the registry key).
   58 data AgentInstance = AgentInstance
   59   { aiId         :: AgentDefId
   60     -- ^ The def the child was spawned from (for display / lookup).
   61   , aiSubagentId :: SubagentId
   62     -- ^ The registry key (def-id + random suffix). Unique per spawn.
   63   , aiSession    :: SessionId
   64     -- ^ The fresh session the child runs under.
   65   , aiStatus     :: AgentStatus
   66   , aiThreadId   :: ThreadId
   67     -- ^ The forked worker thread (killed on stop / interrupt).
   68   , aiDepth      :: Int
   69     -- ^ The parent's delegation depth (0 = top-level parent). The child's
   70     -- depth is @aiDepth + 1@; checked against @max_spawn_depth@ before
   71     -- spawning.
   72   , aiResult     :: Maybe ChildResult
   73     -- ^ The 'ChildResult' when the child has completed ('Nothing' while
   74     -- running, 'Just' after completion). Populated by
   75     -- 'registerCompletedAgentResult'. Allows 'AGENT_STATUS' to return
   76     -- summary + child_session + exit_reason after the child finishes.
   77   } deriving stock (Eq, Show)
   78 
   79 -- | The STM-backed registry of running instances, keyed by 'SubagentId'. No
   80 -- two running instances share a subagent id: 'startAgent' generates a fresh
   81 -- one per spawn. 'stopAgent' removes the entry.
   82 newtype AgentRuntime = AgentRuntime (TVar (Map SubagentId AgentInstance))
   83 
   84 -- | Build an empty runtime.
   85 newAgentRuntime :: IO AgentRuntime
   86 newAgentRuntime = AgentRuntime <$> newTVarIO Map.empty

   88 -- | Start a new agent instance bound to the given def id + fresh session +
   89 -- subagent id. The worker action is forked; its 'ThreadId' is recorded.
   90 -- Returns @Left err@ if an instance is already running for this subagent id
   91 -- (shouldn't happen since subagent ids are freshly minted, but the check is
   92 -- race-safe). The worker's status transitions to 'Running' once the fork
   93 -- succeeds, or 'Crashed' on exception.
   94 startAgent :: AgentRuntime -> AgentDefId -> SubagentId -> SessionId -> Int -> IO () -> IO (Either Text AgentInstance)
   95 startAgent (AgentRuntime tv) aid subagentId session depth worker = do
   96   tid <- forkIO (runWorker tv subagentId worker)
   97   mInst <- atomically $ do
   98     insts <- readTVar tv
   99     if Map.member subagentId insts
  100       then pure Nothing  -- lost the race; shouldn't happen with random ids
  101       else do
  102         let inst = AgentInstance aid subagentId session Running tid depth Nothing
  103         writeTVar tv (Map.insert subagentId inst insts)
  104         pure (Just inst)
  105   case mInst of
  106     Just inst -> pure (Right inst)
  107     Nothing   -> do
  108       killThread tid
  109       pure (Left "agent already running for this subagent id")

  111 -- | Register a running agent instance WITHOUT forking a worker thread.
  112 -- Used by 'runDelegateAsync' which forks its own threads (via 'forkIO')
  113 -- and needs the instance in the registry at spawn time so 'AGENT_INSTANCES'
  114 -- lists it while running. The 'ThreadId' is the CHILD's forked thread id
  115 -- (passed in by 'spawnOne' after the fork) so 'stopAgent' can kill the
  116 -- correct thread. Idempotent: re-registering overwrites.
  117 registerRunningAgent
  118   :: AgentRuntime -> AgentDefId -> SubagentId -> SessionId -> Int -> ThreadId -> IO ()
  119 registerRunningAgent (AgentRuntime tv) aid subagentId session depth tid =
  120   atomically $ do
  121     let inst = AgentInstance aid subagentId session Running tid depth Nothing
  122     modifyTVar' tv (Map.insert subagentId inst)
  123 
  124 -- | Register a synchronously-completed child in the runtime registry. The
  125 -- synchronous delegation model runs the worker to completion BEFORE this is
  126 -- called (the 'ChildResult' payload IS the post-hoc record), so the instance
  127 -- is recorded with status 'Stopped' (the child has already finished) and the
  128 -- @ThreadId@ of the registering thread (the parent's). This makes the child
  129 -- visible to 'AGENT_INSTANCES' / 'AGENT_STATUS' / 'AGENT_STOP' after the
  130 -- synchronous run completes. 'AGENT_STOP' removes it; 'AGENT_INTERRUPT'
  131 -- flips its status to 'Interrupted' (a no-op for an already-stopped child,
  132 -- but harmless). Idempotent: re-registering the same subagent id overwrites
  133 -- the prior entry (the second registration wins).
  134 registerCompletedAgent
  135   :: AgentRuntime -> AgentDefId -> SubagentId -> SessionId -> Int -> IO ()
  136 registerCompletedAgent (AgentRuntime tv) aid subagentId session depth = do
  137   tid <- myThreadId
  138   atomically $ do
  139     insts <- readTVar tv
  140     let inst = AgentInstance aid subagentId session Stopped tid depth Nothing
  141     writeTVar tv (Map.insert subagentId inst insts)
  142 
  143 -- | Register a completed async child's result in the runtime registry.
  144 -- Updates the existing entry's status to 'Stopped' (or 'Crashed' if the
  145 -- result status is 'CsError') and stores the full 'ChildResult' in
  146 -- 'aiResult' so 'AGENT_STATUS' can return summary + child_session +
  147 -- exit_reason after the child finishes. If no entry exists for this
  148 -- subagent id (e.g. the child was a resolve error that never registered),
  149 -- this is a no-op.
  150 registerCompletedAgentResult
  151   :: AgentRuntime -> SubagentId -> ChildResult -> IO ()
  152 registerCompletedAgentResult (AgentRuntime tv) subagentId result = do
  153   tid <- myThreadId
  154   atomically $ do
  155     insts <- readTVar tv
  156     case Map.lookup subagentId insts of
  157       Nothing -> pure ()  -- resolve-error child never registered
  158       Just inst -> do
  159         let newStatus = case crStatus result of
  160               CsError -> Crashed (fromMaybe "unknown error" (crError result))
  161               _       -> Stopped
  162             inst' = inst { aiStatus = newStatus, aiResult = Just result, aiThreadId = tid }
  163         writeTVar tv (Map.insert subagentId inst' insts)
  164 
  165 -- | Run the worker action, transitioning the instance to 'Crashed' on
  166 -- exception. A normal completion leaves the status as 'Running' until the
  167 -- registry's 'stopAgent' is called (the worker is the child turn; reaching
  168 -- its end without exception is a clean completion that the orchestrator
  169 -- observes via 'runDelegate').
  170 runWorker :: TVar (Map SubagentId AgentInstance) -> SubagentId -> IO () -> IO ()
  171 runWorker tv subagentId worker =
  172   catch worker $ \(e :: SomeException) ->
  173     atomically (modifyTVar' tv (Map.adjust (\i -> i { aiStatus = Crashed (T.pack (show e)) }) subagentId))
  174 
  175 -- | Stop a running agent instance: kill the thread, remove the entry.
  176 -- Idempotent (stopping a non-running subagent id is a success with a
  177 -- \"not running\" message, not an error). Always leaves the registry without
  178 -- an entry for this subagent id, so a fresh 'startAgent' can proceed.
  179 stopAgent :: AgentRuntime -> SubagentId -> IO (Either Text ())
  180 stopAgent (AgentRuntime tv) subagentId = do
  181   mInst <- atomically $ do
  182     insts <- readTVar tv
  183     case Map.lookup subagentId insts of
  184       Nothing -> pure Nothing
  185       Just i  -> do
  186         writeTVar tv (Map.delete subagentId insts)
  187         pure (Just i)
  188   case mInst of
  189     Nothing -> pure (Right ())
  190     Just i  -> do
  191       killThread (aiThreadId i)
  192       pure (Right ())
  193 
  194 -- | Request that a single running subagent stop at its next iteration
  195 -- boundary. Does NOT hard-kill the worker thread (Haskell can't safely); sets
  196 -- the instance's status to 'Interrupted' so the worker (which polls its
  197 -- 'crhInterrupted' hook between turns) can exit cleanly. Returns 'True' if a
  198 -- matching subagent was found.
  199 --
  200 -- The worker-builder is responsible for polling the 'crhInterrupted' IORef
  201 -- between turn iterations and returning 'CerInterrupted' when it flips. The
  202 -- registry cannot force the worker to stop — only the worker can decide to
  203 -- exit. 'interruptAgent' /sets the flag and the status/; the worker /observes
  204 -- the flag/.
  205 interruptAgent :: AgentRuntime -> SubagentId -> IO Bool
  206 interruptAgent (AgentRuntime tv) subagentId = do
  207   mInst <- atomically $ do
  208     insts <- readTVar tv
  209     case Map.lookup subagentId insts of
  210       Nothing -> pure Nothing
  211       Just i  -> do
  212         let i' = i { aiStatus = Interrupted }
  213         writeTVar tv (Map.insert subagentId i' insts)
  214         pure (Just i')
  215   pure (isJust mInst)

  217 -- | Snapshot all running instances.
  218 listAgents :: AgentRuntime -> IO [AgentInstance]
  219 listAgents (AgentRuntime tv) = Map.elems <$> readTVarIO tv

  221 -- | Read one instance's status. @Nothing@ if not running.
  222 agentStatus :: AgentRuntime -> SubagentId -> IO (Maybe AgentStatus)
  223 agentStatus (AgentRuntime tv) subagentId =
  224   fmap aiStatus . Map.lookup subagentId <$> readTVarIO tv

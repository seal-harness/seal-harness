    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | A pure static lookup table for model context windows and max output
    3 -- tokens, keyed by model-id prefix. Used by the gateway's
    4 -- @/api/providers/:p/models@ + @/api/providers/:p/models/:m/context@
    5 -- endpoints (Phase 7b T11). The real @/v1/models@ upstream call needs vault
    6 -- credentials + a live HTTP request and is out of scope for T11; this table
    7 -- ships a static, best-known approximation. Unknown model ids yield @0@.
    8 module Seal.Providers.ContextWindow
    9   ( modelContextWindow
   10   , modelMaxOutputTokens
   11   ) where
   12 
   13 import Data.Text (Text)
   14 import Data.Text qualified as T
   15 
   16 -- | The context window size for a model id, by prefix match. 0 if unknown.
   17 modelContextWindow :: Text -> Int
   18 modelContextWindow m
   19   | "claude-sonnet-" `T.isPrefixOf` m = 200000
   20   | "claude-opus-"   `T.isPrefixOf` m = 200000
   21   | "claude-haiku-"  `T.isPrefixOf` m = 200000
   22   | "gpt-4o-"        `T.isPrefixOf` m = 128000
   23   | "gpt-4-turbo"    `T.isPrefixOf` m = 128000
   24   | "llama3.1"       `T.isPrefixOf` m = 128000
   25   | "llama3"         `T.isPrefixOf` m = 8192
   26   | otherwise                          = 0
   27 
   28 -- | The max output tokens for a model id, by prefix match. 0 if unknown.
   29 modelMaxOutputTokens :: Text -> Int
   30 modelMaxOutputTokens m
   31   | "claude-"   `T.isPrefixOf` m = 64000
   32   | "gpt-4o-"   `T.isPrefixOf` m = 16384
   33   | "llama3"    `T.isPrefixOf` m = 4096
   34   | otherwise                     = 0

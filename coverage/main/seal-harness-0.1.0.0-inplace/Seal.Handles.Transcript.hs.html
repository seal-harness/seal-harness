  103 -- | Handle to the legacy single-file transcript writer. All fields are
  104 -- functions so the type is uniform between the real (file-backed) and fake
  105 -- (in-memory) variants.
  106 data TranscriptHandle = TranscriptHandle
  107   { recordAndAck :: TranscriptEntry -> IO ()
  108   -- ^ Enqueue an entry and block until it is flushed and fsync'd to disk.
  109   , recordAsync :: TranscriptEntry -> IO ()
  110   -- ^ Fire-and-forget enqueue; no durability guarantee before return.
  111   , closeTranscript :: IO ()
  112   -- ^ No-op for the file-backed handle (bracket closes the fd); available
  113   -- for callers that want an explicit close point.
  114   }

  116 -- | A unit of work for the single-writer daemon. 'Write' carries an entry and
  117 -- an optional ack slot (filled after fsync). 'Shutdown' tells the daemon to
  118 -- drain any remaining queued writes, fill its ack slot, and exit.
  119 data Item
  120   = Write TranscriptEntry (Maybe (TMVar ()))
  121   | Shutdown (TMVar ())
  122 
  123 -- | Write the strict 'ByteString' to 'fd' using the Posix primitive, which
  124 -- bypasses any user-space buffer. Loops until the whole buffer is written,
  125 -- advancing past each (possibly short) write; 'fdWriteBuf' itself retries the
  126 -- underlying syscall on @EINTR@, so an interrupted write resumes rather than
  127 -- truncating the line.
  128 writeFd :: Fd -> BS.ByteString -> IO ()
  129 writeFd fd bs =
  130   BSU.unsafeUseAsCStringLen bs $ \(ptr, len) ->
  131     go (castPtr ptr) len
  132   where
  133     go :: Ptr Word8 -> Int -> IO ()
  134     go _ 0 = pure ()
  135     go ptr remaining = do
  136       written <- fromIntegral <$> fdWriteBuf fd ptr (fromIntegral remaining)
  137       go (ptr `plusPtr` written) (remaining - written)

  139 -- | Open the transcript file in O_APPEND mode, spawn the single-writer daemon,
  140 -- run @action@, then close. Every 'recordAndAck' blocks until 'fileSynchronise'
  141 -- returns, guaranteeing the entry is durable before the caller proceeds.
  142 withTranscript :: FilePath -> (TranscriptHandle -> IO a) -> IO a
  143 withTranscript path action = do
  144   q <- newTQueueIO
  145   let flags = defaultFileFlags
  146         { append = True
  147         , creat = Just (0o600 :: FileMode)
  148         }
  149       ack = maybe (pure ()) (\tv -> atomically (putTMVar tv ()))
  150       writeEntry fd e = do
  151         writeFd fd (encodeEntryRaw e <> "\n")
  152         fileSynchronise fd
  153       -- Drain every currently-queued write (used at shutdown), fsyncing and
  154       -- acking each, so nothing is lost before the fd is closed.
  155       drain fd = do
  156         next <- atomically (tryReadTQueue q)
  157         case next of
  158           Nothing -> pure ()
  159           Just (Write e mack) -> writeEntry fd e >> ack mack >> drain fd
  160           Just (Shutdown done) -> atomically (putTMVar done ()) >> drain fd
  161       -- Single-writer daemon: process writes; on Shutdown, drain the queue,
  162       -- signal completion, and exit (no 'forever', so it stops deterministically).
  163       daemon fd = do
  164         item <- atomically (readTQueue q)
  165         case item of
  166           Write e mack -> writeEntry fd e >> ack mack >> daemon fd
  167           Shutdown done -> drain fd >> atomically (putTMVar done ())
  168       -- Cleanup: tell the daemon to drain+exit and WAIT for it before closing
  169       -- the fd, so no queued write races (or writes to) a closed fd.
  170       shutdown fd = do
  171         done <- newEmptyTMVarIO
  172         atomically (writeTQueue q (Shutdown done))
  173         atomically (takeTMVar done)
  174         closeFd fd
  175   bracket (openFd path WriteOnly flags) shutdown $ \fd -> do
  176     void $ forkIO (daemon fd)
  177     let enqueue e = atomically (writeTQueue q (Write e Nothing))
  178         ackWrite e = do
  179           tv <- newEmptyTMVarIO
  180           atomically (writeTQueue q (Write e (Just tv)))
  181           atomically (takeTMVar tv)
  182     action TranscriptHandle
  183       { recordAndAck = ackWrite
  184       , recordAsync = enqueue
  185       , closeTranscript = pure ()
  186       }

  207 -- | A structured write request for the indexed transcript writer. The writer diffs
  208 -- @itwMessages@ against the conversation as it exists on disk and appends only
  209 -- the new lines (with @CbToolResult@ parts redacted), then appends the entry
  210 -- record. Crash semantics: messages first, then the entry line, so a torn tail
  211 -- leaves at most orphan message lines / a malformed last line — both already
  212 -- tolerated by the skip-malformed decode path.
  213 data IndexedTranscriptWrite = IndexedTranscriptWrite
  214   { itwMessages :: [Message]
  215   -- ^ The full message list in effect at this turn (the conversation prefix +
  216   -- any new messages). The writer diffs against the on-disk conversation and
  217   -- appends only the new lines.
  218   , itwEntry    :: EntryRecord
  219   -- ^ The event line to append to @entries.jsonl@.
  220   }

  222 -- | Handle to the indexed transcript writer. 'itwRecordAndAck' blocks until both files
  223 -- are fsync'd; 'itwRecordAsync' is fire-and-forget.
  224 data IndexedTranscriptHandle = IndexedTranscriptHandle
  225   { itwRecordAndAck :: IndexedTranscriptWrite -> IO ()
  226   , itwRecordAsync  :: IndexedTranscriptWrite -> IO ()
  227   , itwReadConversation :: IO [Message]
  228   -- ^ Read back the on-disk conversation (for tests / replay).
  229   , itwReadEntries     :: IO [EntryRecord]
  230   -- ^ Read back the on-disk entry log.
  231   , itwSetSecretOps    :: Set OpName -> IO ()
  232   -- ^ Set the set of opcode names whose tool results should be redacted from
  233   -- the on-disk conversation file. Call this after building the ISA registry
  234   -- so the writer knows which 'CbToolResult's may carry secrets. Results from
  235   -- opcodes NOT in this set (e.g. SHELL_EXEC, FILE_READ) pass through verbatim.
  236   , itwCloseTranscript :: IO ()
  237   , itwIsAlive :: IO Bool
  238   -- ^ Is the single-writer daemon still running? Returns 'False' after a
  239   -- write error killed it. Callers that depend on durability (the dispatch
  240   -- ACK-before-execute gate) should check this before trusting a prior write.
  241   , itwDiagnostics :: IO TranscriptDiagnostics
  242   -- ^ Counters for the writer's internal invariants. Both counters are
  243   -- expected to stay at zero: a non-zero value means the transcript and
  244   -- the conversation handed to the model have diverged (see
  245   -- @docs\/plans\/2026-10-22-subagent-completion-delivery-diagnosis.md@).
  246   , itwRecordedLength :: IO Int
  247   -- ^ How many conversation messages the writer has recorded so far. The
  248   -- writer's own view of the conversation — compare against
  249   -- @length <$> itwReadConversation@ to spot a reader that is behind the
  250   -- writer (a completion injection the model never saw).
  251   }

  253 -- | Invariant counters for the indexed transcript writer.
  254 --
  255 -- Both counters describe ways the transcript can drift away from the
  256 -- conversation the model is actually given:
  257 --
  258 -- * 'tdInjectedTailMisses' — a read that did not contain a completion
  259 --   injection the writer had already been asked to record. The model is
  260 --   then told that a sub-agent is still running while its verdict sits in
  261 --   the transcript.
  262 -- * 'tdDesyncEvents' — a write whose incoming conversation was not a
  263 --   prefix-extension of the conversation the writer had recorded. The
  264 --   writer's positional diff then slices the wrong range and silently
  265 --   drops messages from @conversation.jsonl@.
  266 data TranscriptDiagnostics = TranscriptDiagnostics
  267   { tdInjectedTailMisses :: Int
  268   -- ^ Reads that missed a pending completion injection.
  269   , tdDesyncEvents :: Int
  270   -- ^ Writes whose incoming conversation diverged from the writer's.
  271   , tdDesyncDroppedMessages :: Int
  272   -- ^ Total messages the writer had to drop because of a desync.
  273   }
  274   deriving stock (Eq, Show)

  276 -- | A diagnostics record with every counter at zero.

  278 emptyTranscriptDiagnostics = TranscriptDiagnostics
  279   { tdInjectedTailMisses = 0
  280   , tdDesyncEvents = 0

  292 -- | The daemon's accumulated state: the conversation lines written so far (so
  293 -- the diff can be computed without re-reading the file each turn) and the
  294 -- fsync'd fds.
  295 data IndexedTranscriptState = IndexedTranscriptState
  296   { itsConvFd :: Fd
  297   , itsEntriesFd :: Fd
  298   , itsConvIdxFd :: Fd
  299   -- ^ The fd for @conversation.idx@ (O_APPEND). Appended after each
  300   -- conversation line is written + fsync'd.
  301   , itsWritten :: [Message]
  302   -- ^ The conversation as it exists on disk, in order. Grown by each write
  303   -- so the next diff can be computed in-memory.
  304   , itsSecretOpsRef :: IORef (Set OpName)
  305   -- ^ The set of opcode names whose tool results may carry secrets and must
  306   -- be redacted from the on-disk conversation file. An IORef so the caller
  307   -- can set it after building the registry (the registry construction may
  308   -- depend on values only available inside the 'withIndexedTranscript'
  309   -- callback, e.g. session id or worker functions).
  310   , itsPriorEnv :: Maybe Envelope
  311   -- ^ The effective envelope at the most recent 'EKRequest' entry. Used by
  312   -- 'writeOne' to compute a minimal 'EnvelopeDelta' — only fields that
  313   -- differ from the prior envelope are emitted, so a stable system prompt
  314   -- + tools array isn't re-serialized on every turn (a 146-turn session
  315   -- with a 36KB system prompt would otherwise waste ~5MB). 'Nothing' until
  316   -- the first request entry lands.
  317   }

  324 -- | Open both files in O_APPEND mode, spawn the single-writer daemon, run
  325 -- @action@, then close. Every 'itwRecordAndAck' blocks until both files are
  326 -- fsync'd.
  327 --
  328 -- If the session directory already contains a legacy @transcript.jsonl@ and no
  329 -- @conversation.jsonl@, the legacy file is left untouched (the legacy read path
  330 -- handles it); the new writer simply creates the two new files alongside it.
  331 withIndexedTranscript :: FilePath -> (IndexedTranscriptHandle -> IO a) -> IO a
  332 withIndexedTranscript dir action = do
  333   q <- newTQueueIO
  334   aliveRef <- newTVarIO True
  335   diagTv <- newTVarIO emptyTranscriptDiagnostics
  336   writtenLenTv <- newTVarIO 0
  337   pendingInjectedTv <- newTVarIO []
  338   let sessionLog = dir </> "seal.log"
  339   let convPath     = dir </> "conversation.jsonl"
  340       entriesPath  = dir </> "entries.jsonl"
  341       idxPath      = dir </> "conversation.idx"
  342       flags = defaultFileFlags
  343         { append = True
  344         , creat = Just (0o600 :: FileMode)
  345         }
  346       ack = maybe (pure ()) (\tv -> atomically (putTMVar tv ()))
  347       writeOne st (IndexedTranscriptWrite msgs entry) = do
  348         secretOps <- readIORef (itsSecretOpsRef st)
  349         let redactedMsgs = redactMessages secretOps msgs msgs
  350             writtenLen = length (itsWritten st)
  351             -- Positional suffix extraction: the entry's erConvLen tells us
  352             -- the total conversation length after this write (for
  353             -- request/response) or the number of new lines (for harness).
  354             -- For request/response: new = take (erConvLen - writtenLen)
  355             --   (drop writtenLen redactedMsgs)
  356             -- For harness: erConvLen is a delta, so new = take erConvLen
  357             --   (drop writtenLen redactedMsgs)
  358             -- For compaction: erConvLen is the new cursor (total), same as
  359             --   request/response.
  360             newLen = case erKind entry of
  361               EKHarness -> erConvLen entry
  362               _ -> max 0 (erConvLen entry - writtenLen)
  363             new = take newLen (drop writtenLen redactedMsgs)
  364         checkAlignment st redactedMsgs entry
  365         -- 1. Append new conversation lines + index entries, fsync both.
  366         convOffset <- fromIntegral <$> fdSeek (itsConvFd st) SeekFromEnd 0
  367         newOffset <- foldM (\off m -> do
  368             let bs = encodeConvLine (ConvLine m) <> "\n"
  369             writeFd (itsConvFd st) bs
  370             let off' = off + fromIntegral (BS.length bs)
  371             writeIdxEntry (itsConvIdxFd st) off'
  372             pure off'
  373           ) convOffset new
  374         _ <- newOffset `seq` pure ()
  375         fileSynchronise (itsConvFd st)
  376         fileSynchronise (itsConvIdxFd st)
  377         -- 2. Compute a minimal envelope delta for EKRequest entries. The
  378         -- caller passes a FULL delta (every field set to 'Just'); the writer
  379         -- compares it against the prior effective envelope and emits only
  380         -- the fields that differ. This keeps a stable system prompt + tools
  381         -- array from being re-serialized on every turn (a 146-turn session
  382         -- with a 36KB system prompt would otherwise waste ~5MB). The first
  383         -- request (no prior envelope) emits the full delta so
  384         -- reconstruction has a baseline to fold from.
  385         let (entry', mNextEnv) = case erKind entry of
  386               EKRequest -> case erEnvelope entry of
  387                 Just delta ->
  388                   let baseline = fromMaybe defaultEnv (itsPriorEnv st)
  389                       nextEnv = applyDelta baseline delta
  390                       minimal = minimalDelta (itsPriorEnv st) nextEnv
  391                   in (entry { erEnvelope = minimal }, Just nextEnv)
  392                 Nothing -> (entry, itsPriorEnv st)
  393               _ -> (entry, itsPriorEnv st)
  394         -- 3. Append the entry line, fsync.
  395         writeFd (itsEntriesFd st) (encodeEntryRecordRaw entry' <> "\n")
  396         fileSynchronise (itsEntriesFd st)
  397         let st' = st { itsWritten = itsWritten st <> new, itsPriorEnv = mNextEnv }
  398         atomically (writeTVar writtenLenTv (length (itsWritten st')))
  399         -- The injection is on disk now: a later read that still misses it is
  400         -- a torn read, not a race with the writer.
  401         when (erHarness entry == Just completionHarnessTag) $
  402           atomically (writeTVar pendingInjectedTv [])
  403         pure st'
  404       -- | Assert that the caller's conversation is a prefix-extension of
  405       -- the conversation the writer has recorded. When it is not, the
  406       -- positional diff below slices the wrong range and silently drops
  407       -- messages from @conversation.jsonl@ — the transcript then no longer
  408       -- matches the conversation the model was given.
  409       --
  410       -- Two cheap checks, no full prefix scan:
  411       --
  412       -- * a non-harness entry whose @erConvLen@ is below the writer's
  413       --   length means the caller's list is SHORTER than what the writer
  414       --   has recorded, so the writer will drop @writtenLen - erConvLen@
  415       --   messages;
  416       -- * a non-harness entry whose last already-written message differs
  417       --   from the caller's message at the same position means the two
  418       --   conversations have drifted apart (same length, different
  419       --   content) — the file and the model's context disagree.
  420       checkAlignment st redactedMsgs entry = do
  421         let writtenLen = length (itsWritten st)
  422             isHarness = erKind entry == EKHarness
  423             -- A write that carries no conversation messages (the
  424             -- ACK-before-execute entries, the session preamble) appends
  425             -- nothing to @conversation.jsonl@ and cannot desync it.
  426             carriesMessages = not (null redactedMsgs)
  427             -- Two conversations are "the same conversation" when they
  428             -- start with the same message. A caller that writes a
  429             -- DIFFERENT conversation (the @\/model@ and @\/stop@
  430             -- notices, which append one message to a session whose
  431             -- conversation they did not read) is not desyncing anything
  432             -- — its write is a no-op against the writer's state, which is
  433             -- a separate (pre-existing) issue.
  434             sameConversation =
  435               carriesMessages
  436               && not (null (itsWritten st))
  437               && elementAt 0 redactedMsgs == elementAt 0 (itsWritten st)
  438             -- The caller's conversation length: its own claim, capped by
  439             -- the number of messages it actually handed over.
  440             callerLen = min (erConvLen entry) (length redactedMsgs)
  441             tooShort =
  442               not isHarness
  443               && carriesMessages
  444               && sameConversation
  445               && callerLen < writtenLen
  446             tailMismatch =
  447               not isHarness
  448               && carriesMessages
  449               && writtenLen > 0
  450               && case (lastOf (itsWritten st), elementAt (writtenLen - 1) redactedMsgs) of
  451                    (Just a, Just b) -> a /= b
  452                    _                -> False
  453         when (tooShort || tailMismatch) $ do
  454           let dropped = max 0 (writtenLen - callerLen)
  455           atomically (modifyTVar' diagTv $ \d -> d
  456             { tdDesyncEvents = tdDesyncEvents d + 1
  457             , tdDesyncDroppedMessages = tdDesyncDroppedMessages d + dropped
  458             })
  459           globalLogIO ErrorS
  460             (ls ("[transcript-diag] DESYNC: the caller's conversation is not a "
  461               <> "prefix-extension of the writer's. writerLen=" <> T.pack (show writtenLen)
  462               <> " callerLen=" <> T.pack (show callerLen)
  463               <> " dropped=" <> T.pack (show dropped)
  464               <> " tailMismatch=" <> (if tailMismatch then "true" else "false")))
  465           appendSessionLog (Just sessionLog) "ERROR"
  466             ("[transcript-diag] DESYNC: caller conversation is not a "
  467               <> "prefix-extension of the writer's conversation. writerLen="
  468               <> T.pack (show writtenLen) <> " callerLen=" <> T.pack (show callerLen)
  469               <> " droppedMessages=" <> T.pack (show dropped))
  470       drain st = do
  471         next <- atomically (tryReadTQueue q)
  472         case next of
  473           Nothing -> pure st
  474           Just (ITWrite w mack) -> do
  475             st' <- writeOne st w
  476             ack mack
  477             drain st'
  478           Just (ITShutdown done) -> do
  479             atomically (putTMVar done ())
  480             drain st
  481       daemon st = do
  482         item <- atomically (readTQueue q)
  483         case item of
  484           ITWrite w mack -> do
  485             st' <- writeOne st w
  486             ack mack
  487             daemon st'
  488           ITShutdown done -> do
  489             st' <- drain st
  490             atomically (putTMVar done ())
  491             pure st'
  492       shutdown convFd entriesFd idxFd done = do
  493         atomically (writeTQueue q (ITShutdown done))
  494         atomically (takeTMVar done)
  495         closeFd convFd
  496         closeFd entriesFd
  497         closeFd idxFd
  498   -- Read existing conversation (if any) so the diff starts from the right
  499   -- baseline. A torn legacy file or a fresh session yields [].
  500   existingConv <- do
  501     exists <- doesFileExist convPath
  502     if exists
  503       then readConversation <$> BS.readFile convPath
  504       else pure []
  505   -- Build the index if missing (one-time migration for pre-existing sessions).
  506   -- For fresh sessions, write the initial [0] offset so the index is valid
  507   -- (not empty) before the daemon's first write.
  508   convExists <- doesFileExist convPath
  509   idxExists  <- doesFileExist idxPath
  510   if convExists && not idxExists
  511     then void (buildIndex convPath idxPath)
  512     else when (not convExists && not idxExists) $ do
  513       -- Create the index with the initial [0] offset
  514       let idxFlags = defaultFileFlags { append = False, creat = Just (0o600 :: FileMode) }
  515       idxFd0 <- openFd idxPath WriteOnly idxFlags
  516       writeIdxEntry idxFd0 0
  517       fileSynchronise idxFd0
  518       closeFd idxFd0
  519   bracket
  520     (do convFd     <- openFd convPath    WriteOnly flags
  521         entriesFd  <- openFd entriesPath WriteOnly flags
  522         idxFd      <- openFd idxPath     WriteOnly flags
  523         pure (convFd, entriesFd, idxFd))
  524     (\(convFd, entriesFd, idxFd) -> do
  525         done <- newEmptyTMVarIO
  526         shutdown convFd entriesFd idxFd done)
  527     $ \(convFd, entriesFd, idxFd) -> do
  528         secretOpsRef <- newIORef Set.empty
  529         let st0 = IndexedTranscriptState convFd entriesFd idxFd existingConv secretOpsRef Nothing
  530             -- The daemon with an exception handler: if writeOne throws, mark
  531             -- the daemon dead, fail every queued ACK (so callers blocked on
  532             -- takeTMVar unblock immediately), and log to stderr. The daemon
  533             -- then exits — no further writes are attempted.
  534             safeDaemon st =
  535               daemon st `catch` \e -> do
  536                 atomically (writeTVar aliveRef False)
  537                 globalLogIO ErrorS ("[transcript] writer daemon died: " <> ls (T.pack (show (e :: SomeException))))
  538                 -- Drain remaining queue items WITHOUT filling their ACK slots
  539                 -- (the writes never completed). Callers blocked on takeTMVar
  540                 -- unblock via the aliveRef check in ackWrite's orElse. Shutdown
  541                 -- items still get their done signal so the bracket cleanup
  542                 -- completes.
  543                 let drainRemaining = do
  544                       item <- atomically (tryReadTQueue q)
  545                       case item of
  546                         Nothing -> pure ()
  547                         Just (ITWrite _ _mack) -> drainRemaining
  548                         Just (ITShutdown done) -> do
  549                           atomically (putTMVar done ())
  550                 drainRemaining
  551                 pure st
  552             -- | Assert that a reader about to build the model's context can
  553             -- see the completion injection the writer has been asked to
  554             -- record. The expectation is dropped as soon as the writer has
  555             -- actually written the injection, so a read that misses it is a
  556             -- genuine race — the model is about to be told that a finished
  557             -- sub-agent is still running.
  558             checkVisibility msgs = do
  559               pending <- readTVarIO pendingInjectedTv
  560               unless (null pending) $ do
  561                 let missing = [ m | m <- pending, m `notElem` msgs ]
  562                 unless (null missing) $ do
  563                   atomically (modifyTVar' diagTv $ \d -> d
  564                     { tdInjectedTailMisses = tdInjectedTailMisses d + 1 })
  565                   writerLen <- readTVarIO writtenLenTv
  566                   let detail =
  567                         "injected=" <> T.pack (show (length pending))
  568                         <> " missingFromReader=" <> T.pack (show (length missing))
  569                         <> " readerLen=" <> T.pack (show (length msgs))
  570                         <> " writerLen=" <> T.pack (show writerLen)
  571                   globalLogIO ErrorS
  572                     (ls ("[transcript-diag] COMPLETION NOT VISIBLE TO READER: the "
  573                       <> "writer has been asked to record a sub-agent completion but the "
  574                       <> "conversation the model is given does not contain it. " <> detail))
  575                   appendSessionLog (Just sessionLog) "ERROR"
  576                     ("[transcript-diag] COMPLETION NOT VISIBLE TO READER. " <> detail)
  577         void $ forkIO (void (safeDaemon st0))
  578         -- | Remember a completion injection the writer has been asked to
  579         -- record but has not written yet. A read that happens before the
  580         -- write lands hands the model a conversation without the
  581         -- completion — the model then reports the sub-agent as still
  582         -- running while its verdict is on its way to disk.
  583         let notePendingInjection w =
  584               when (erHarness (itwEntry w) == Just completionHarnessTag) $ do
  585                 let n = erConvLen (itwEntry w)
  586                     msgs = itwMessages w
  587                 when (n > 0 && n <= length msgs) $
  588                   atomically (writeTVar pendingInjectedTv (take n (reverse msgs)))
  589         -- Close the handle: mark the writer dead so a further durable
  590         -- write raises 'TranscriptError' instead of hanging. The daemon
  591         -- thread itself is stopped by the bracket's shutdown (enqueuing a
  592         -- second 'ITShutdown' here would deadlock that shutdown, which
  593         -- waits on the daemon's ack). The writer has no other way to be
  594         -- stopped — it is a bare `forkIO` with no exposed 'ThreadId' — so
  595         -- without this the "the writer died" path is unreachable from a
  596         -- test. Never called in production.
  597         let closeHandle = atomically (writeTVar aliveRef False)
  598         let enqueue w = do
  599               notePendingInjection w
  600               atomically (writeTQueue q (ITWrite w Nothing))
  601             ackWrite w = do
  602               notePendingInjection w
  603               alive <- readTVarIO aliveRef
  604               if not alive
  605                 then throwIO (TranscriptError "transcript writer daemon is dead")
  606                 else do
  607                   tv <- newEmptyTMVarIO
  608                   timeoutVar <- newEmptyTMVarIO
  609                   atomically (writeTQueue q (ITWrite w (Just tv)))
  610                   void $ forkIO $ do
  611                     threadDelay defaultAckTimeoutUs
  612                     void $ atomically (tryPutTMVar timeoutVar ())
  613                   mResult <- atomically $
  614                     (Right <$> takeTMVar tv)
  615                       `orElse` (Left <$> takeTMVar timeoutVar)
  616                       `orElse` (do
  617                          stillAlive <- readTVar aliveRef
  618                          if stillAlive then retry else pure (Left ()))
  619                   case mResult of
  620                     Right () -> pure ()
  621                     Left ()  -> do
  622                       stillAlive <- readTVarIO aliveRef
  623                       if stillAlive
  624                         then throwIO (TranscriptError "transcript write timed out (daemon stuck)")
  625                         else throwIO (TranscriptError "transcript writer daemon died during write")
  626         action IndexedTranscriptHandle
  627           { itwRecordAndAck = ackWrite
  628           , itwRecordAsync  = enqueue
  629           , itwReadConversation = do
  630               msgs <- readConversation <$> BS.readFile convPath
  631               checkVisibility msgs
  632               pure msgs
  633           , itwReadEntries     = readEntries entriesPath
  634           , itwSetSecretOps    = writeIORef secretOpsRef
  635           , itwCloseTranscript = closeHandle
  636           , itwIsAlive = readTVarIO aliveRef
  637           , itwDiagnostics = readTVarIO diagTv
  638           , itwRecordedLength = readTVarIO writtenLenTv
  639           }

  709 -- | Redact tool-result parts that may carry secret values. Only results from
  710 -- secret-producing opcodes (those whose name is in the provided set) are
  711 -- redacted — Untrusted opcodes (SHELL_EXEC, FILE_READ, etc.) produce non-secret
  712 -- output that passes through verbatim so the frontend can display it.
  713 --
  714 -- The @allMsgs@ list (the full conversation) is scanned for 'CbToolUse' blocks
  715 -- whose 'cbName' is in the secret-opcode-name set; their 'cbId's form the set
  716 -- of tool-call ids whose results should be redacted. Only the @newMsgs@ are
  717 -- returned (after redaction); @allMsgs@ is only used for the scan.
  718 redactMessages :: Set OpName -> [Message] -> [Message] -> [Message]
  719 redactMessages secretOps allMsgs newMsgs =
  720   let secretIds = buildSecretIds secretOps allMsgs
  721   in map (redactMessage secretIds) newMsgs
  722   where
  723     buildSecretIds :: Set OpName -> [Message] -> Set ToolCallId
  724     buildSecretIds ops = foldr go mempty
  725       where
  726         go (Message _ blocks) acc = acc <> foldr goBlock mempty blocks
  727           where
  728             goBlock (CbToolUse tcid name _) a
  729               | name `Set.member` ops = tcid `Set.insert` a
  730               | otherwise             = a
  731             goBlock _ a = a
  732 
  733     redactMessage :: Set ToolCallId -> Message -> Message
  734     redactMessage secretIds (Message role blocks) =
  735       Message role (map (redactBlock secretIds) blocks)
  736       where
  737         redactBlock ids (CbToolResult tcid parts isErr)
  738           | tcid `Set.member` ids = CbToolResult tcid (map redactPart parts) isErr
  739           | otherwise             = CbToolResult tcid parts isErr
  740         redactBlock _ other = other
  741 
  742         redactPart (TrpText t)
  743           | not (T.null t) = TrpText "<redacted:secret>"
  744           | otherwise      = TrpText t

  749 defaultEnv = Envelope (ModelId "") Nothing [] ToolAuto 0

  751 -- | Compute the minimal 'EnvelopeDelta' that, when applied to the prior
  752 -- envelope, yields the next envelope. 'Nothing' fields mean "inherit" — so
  753 -- a field that's unchanged from the prior is emitted as 'Nothing' and
  754 -- omitted from the on-disk JSON entirely. When there is no prior envelope
  755 -- (the first request), the full delta is returned so reconstruction has a
  756 -- baseline to fold from.
  757 minimalDelta :: Maybe Envelope -> Envelope -> Maybe EnvelopeDelta
  758 minimalDelta mPrior next = case mPrior of
  759   Nothing -> Just EnvelopeDelta
  760     { edModel = Just (envModel next)
  761     , edSystem = Just (envSystem next)
  762     , edTools = Just (envTools next)
  763     , edToolChoice = Just (envToolChoice next)
  764     , edMaxTokens = Just (envMaxTokens next)
  765     }
  766   Just prior ->
  767     let d = EnvelopeDelta
  768           { edModel = if envModel next == envModel prior then Nothing else Just (envModel next)
  769           , edSystem = if envSystem next == envSystem prior then Nothing else Just (envSystem next)
  770           , edTools = if envTools next == envTools prior then Nothing else Just (envTools next)
  771           , edToolChoice = if envToolChoice next == envToolChoice prior then Nothing else Just (envToolChoice next)
  772           , edMaxTokens = if envMaxTokens next == envMaxTokens prior then Nothing else Just (envMaxTokens next)
  773           }
  774     in if d == emptyEnvelopeDelta then Nothing else Just d

  776 -- | Append a single 'Message' to a session's @conversation.jsonl@ file
  777 -- using direct @O_APPEND@ + @fsync@, and append the new byte offset to
  778 -- @conversation.idx@. This is used by the async delegation completion
  779 -- callback — which runs in a forked child thread that may outlive the
  780 -- parent's 'withIndexedTranscript' bracket (and its single-writer daemon).
  781 -- POSIX @O_APPEND@ guarantees each @write()@ atomically seeks to
  782 -- end-of-file, so concurrent writes (from the daemon during the parent's
  783 -- turn, or from another completion callback) do not interleave.
  784 --
  785 -- IO errors are swallowed (best-effort): a completion write failure must
  786 -- not crash the forked child thread. The error is logged via the global
  787 -- katip scribe.
  788 appendConversationMessage :: SealPaths -> SessionId -> Message -> IO ()
  789 appendConversationMessage paths sid msg =
  790   catch @IOException
  791     ( do
  792         let convPath = sessionConversationPath paths sid
  793             idxPath  = sessionConversationIndexPath paths sid
  794             flags = defaultFileFlags { append = True, creat = Just (0o600 :: FileMode) }
  795         convFd <- openFd convPath ReadWrite flags
  796         let bs = encodeConvLine (ConvLine msg) <> "\n"
  797         BSU.unsafeUseAsCStringLen bs $ \(ptr, len) -> do
  798           _ <- fdWriteBuf convFd (castPtr ptr) (fromIntegral len)
  799           pure ()
  800         fileSynchronise convFd
  801         -- Append the new EOF offset to the index
  802         newOffset <- fromIntegral <$> fdSeek convFd SeekFromEnd 0
  803         idxFd <- openFd idxPath WriteOnly flags
  804         writeIdxEntry idxFd newOffset
  805         fileSynchronise idxFd
  806         closeFd idxFd
  807         closeFd convFd
  808     )
  809     ( \e -> globalLogIO ErrorS
  810               (ls ("appendConversationMessage failed: " <> T.pack (show e)))
  811     )

  867         globalLogIO ErrorS

    2 -- | The @/call@ command: directly invoke a named ISA opcode with
    3 -- user-supplied JSON arguments, bypassing the LLM. The opcode name is
    4 -- matched case-sensitively against the active session's ISA registry
    5 -- (the same registry the agent loop dispatches against). The argument
    6 -- payload is parsed as JSON; missing/invalid payloads are surfaced as
    7 -- plain-text errors.
    8 --
    9 -- Runs the opcode under 'Full' autonomy semantics: the operator is the
   10 -- human approver by typing the command, so the per-call confirmation
   11 -- gate ('Supervised' autonomy) is skipped — exactly mirroring how a
   12 -- turn under @--yolo@ dispatches. The opcode's own authorize gate still
   13 -- applies (e.g. BIN_EXEC's allow-list, SHELL_EXEC's policy).
   14 module Seal.Command.Call
   15   ( callCommandSpec
   16   , CallDispatcher
   17   , renderOpResult
   18   , renderDispatchError
   19   ) where

   38 -- | The channel-supplied dispatcher. Built by the wiring at each channel
   39 -- (CLI TUI, web) with the right 'Seal.ISA.Registry.Registry',
   40 -- 'Seal.Handles.Transcript.TwoFileHandle', 'Seal.ISA.Opcode.BackendExec',
   41 -- 'Seal.Tools.Exec.Types.ExecBackend', and 'Seal.Types.App.Env' for the
   42 -- active session. Returns the structured 'DispatchError' \/ 'OpResult'
   43 -- so the command can render either outcome.
   44 type CallDispatcher = OpName -> Value -> IO (Either DispatchError OpResult)
   45 
   46 -- | The @/call@ command spec. Closes over the channel-supplied
   47 -- 'CallDispatcher'. The dispatcher is per-session (it threads the active
   48 -- session's transcript + registry); the wiring rebuilds the spec on
   49 -- session change.
   50 callCommandSpec :: CallDispatcher -> CommandSpec
   51 callCommandSpec dispatcher = CommandSpec
   52   { csName         = CommandName "call"
   53   , csAliases      = []
   54   , csGroup        = GroupTools
   55   , csSynopsis     = "Invoke an ISA opcode directly with JSON args"
   56   , csParserInfo   = callParserInfo dispatcher
   57   , csAvailability = InteractiveOnly
   58   }
   59 
   60 callParserInfo :: CallDispatcher -> ParserInfo CommandAction
   61 callParserInfo dispatcher =
   62   info (callParser dispatcher <**> helper)
   63     (  progDesc "Invoke an ISA opcode directly (bypassing the LLM)"
   64     <> header   "call — directly dispatch an opcode with JSON args"
   65     )
   66 
   67 callParser :: CallDispatcher -> Parser CommandAction
   68 callParser dispatcher = callCmd dispatcher
   69   <$> opNameArg
   70   <*> jsonArg
   71 
   72 -- | @/call OP NAME JSON-ARGS@ — invoke the opcode with the parsed JSON
   73 -- value. If the JSON argument is omitted, an empty object is passed (for
   74 -- opcodes that take no input). The command line is echoed as the first
   75 -- line of the output so the "Command output" bubble is self-contained
   76 -- (the web channel clears the optimistic "You" bubble on slash responses).
   77 callCmd :: CallDispatcher -> Text -> Maybe Text -> CommandAction
   78 callCmd dispatcher opNameText mJson = commandAction $ \caps -> do
   79   ccSend caps (echoLine opNameText mJson)
   80   case mkOpNameText opNameText of
   81     Left err -> ccSend caps ("invalid opcode name: " <> err)
   82     Right opName ->
   83       case mJson of
   84         Nothing -> run dispatcher caps opName (Aeson.object [])
   85         Just raw ->
   86           case eitherDecode' (BL.fromStrict (TE.encodeUtf8 raw)) of
   87             Left err   -> ccSend caps
   88               ( "invalid JSON: " <> T.pack err
   89                 <> "\nThe JSON arg must be a complete JSON value (object, array, or string)."
   90                 <> "\nFor a bare token like a name, wrap it as a JSON object, e.g.:"
   91                 <> "\n  /call " <> opNameText <> " {\"name\":\"" <> raw <> "\"}" )
   92             Right val  -> run dispatcher caps opName val
   93 
   94 -- | Reconstruct the command line for the echo header.
   95 echoLine :: Text -> Maybe Text -> Text
   96 echoLine op mJson = "$ /call " <> op <> maybe "" (" " <>) mJson
   97 
   98 -- | Validate the opcode name (only the constructor check; the lookup
   99 -- happens in the dispatcher).
  100 mkOpNameText :: Text -> Either Text OpName
  101 mkOpNameText t
  102   | T.null t            = Left "opcode name is empty"
  103   | T.any (== '\0') t   = Left "opcode name contains NUL"
  104   | otherwise           = Right (OpName t)
  105 
  106 run :: CallDispatcher -> ChannelCaps -> OpName -> Value -> IO ()
  107 run dispatcher caps opName val = do
  108   res <- dispatcher opName val
  109   case res of
  110     Left e -> ccSend caps (renderDispatchError e)
  111     Right r -> mapM_ (ccSend caps) (renderOpResult r)
  112 
  113 -- | Render a 'DispatchError' as a single line.
  114 renderDispatchError :: DispatchError -> Text
  115 renderDispatchError = \case
  116   OpNotFound (OpName n) -> "opcode not found: " <> n
  117   Denied why            -> "denied: " <> why
  118   ExecFailed why         -> "exec failed: " <> why
  119 
  120 -- | Render an 'OpResult' as one line per text part. Non-text parts are
  121 -- shown via their 'Show' instance. The error flag is prefixed when set
  122 -- so the operator can tell a failed run from a successful one.
  123 renderOpResult :: OpResult -> [Text]
  124 renderOpResult r =
  125   let prefix = if orIsError r then "[error] " else ""
  126       body = case orParts r of
  127         [] -> [prefix <> "(no output)"]
  128         xs -> [ prefix <> t | TrpText t <- xs ]
  129   in if null body then [prefix <> "(no output)"] else body

  135 -- | Required opcode-name argument. The metavar @OP@ hints at the
  136 -- opcodes' convention (e.g. @SHELL_EXEC@, @FILE_READ@).
  137 opNameArg :: Parser Text
  138 opNameArg = T.pack <$> strArgument
  139   ( metavar "OP"
  140   <> help "Opcode name (e.g. SHELL_EXEC, FILE_READ, BIN_EXEC)" )
  141 
  142 -- | Optional JSON argument. Parsed by 'eitherDecode'' at dispatch time;
  143 -- parse errors are surfaced to the user. Quoting is up to the shell
  144 -- (the tokenizer handles double-quoted tokens).
  145 jsonArg :: Parser (Maybe Text)
  146 jsonArg = optional (T.pack <$> strArgument
  147   ( metavar "JSON"
  148   <> help "Opcode input as a JSON object (default: {})" ))

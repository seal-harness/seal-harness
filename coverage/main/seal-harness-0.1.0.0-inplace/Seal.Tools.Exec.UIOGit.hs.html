    1 -- | The Git-specific 'UIO' operations — the 'CloneDeps' surface lifted into
    2 -- the 'UIO' monad. This is a SEPARATE module from 'Seal.Tools.Exec.UIO'
    3 -- (which holds the general 'UntrustedIO' 'uio*' surface) because the
    4 -- 'CloneDeps' capability must be lexically scoped to Git opcodes ONLY
    5 -- (restoring the round-3 property that non-Git opcodes cannot name these
    6 -- functions). A non-Git opcode that wants 'uioCdRepoRegList' must add an
    7 -- import of this module — caught at review + enforced by the W-A3 CI grep
    8 -- guard (asserts 'Seal.Tools.Exec.UIOGit' is referenced only from
    9 -- @src\/Seal\/ISA\/Ops\/Git.hs@ + this definition site).
   10 --
   11 -- The 'uioCd*' functions take the 'CloneDeps' from 'UIOEnv' via the internal
   12 -- 'ReaderT' (no 'MonadIO' — the 'UIO' newtype hides it). Opcode bodies call
   13 -- these directly: @uioCdRepoRegList@, @uioResolveClone@, @uioWithClone@ —
   14 -- never @liftIO (rrhList (cdRepoReg deps))@.
   15 module Seal.Tools.Exec.UIOGit
   16   ( uioCdRepoRegList
   17   , uioResolveClone
   18   , uioWithClone
   19   , uioCdCloneDeps
   20   ) where
   21

   23 import Control.Monad.Reader (ask, asks, runReaderT)

   33 -- | List the repo registry entries (the @cdRepoReg@ surface). Returns the
   34 -- 'Either' 'Text' error from a corrupt @repos.toml@ (surfaced to the opcode
   35 -- as-is).
   36 uioCdRepoRegList :: UIO (Either Text [SourceRepo])
   37 uioCdRepoRegList = UIO $ do
   38   env <- ask
   39   liftIO (rrhList (cdRepoReg (uieCloneDeps env)))

   41 -- | Resolve a 'SourceRepo' to a 'CloneTarget' (the credential-resolution
   42 -- surface — fetches the secret from the vault, runs the per-op ssh-agent
   43 -- lifecycle for deploy keys). Returns 'Either' 'CloneError' so the opcode
   44 -- can surface a structured error (vault-locked vs. no-credential vs.
   45 -- agent-failed).
   46 uioResolveClone :: SourceRepo -> UIO (Either CloneError CloneTarget)
   47 uioResolveClone repo = UIO $ do
   48   env <- ask
   49   liftIO (resolveCloneTarget (uieCloneDeps env) repo)
   50 
   51 -- | Scope the authenticated 'CloneEnv' to a continuation (CPS — mirrors
   52 -- 'withCloneTarget' / 'Seal.Security.Secrets.withApiKey'). The cleanup
   53 -- ALWAYS runs, even if the continuation throws — bracket semantics. The
   54 -- 'UIOEnv' is threaded from the outer context, so 'CloneDeps' +
   55 -- 'UntrustedIO' are preserved across the CPS boundary.
   56 uioWithClone :: CloneTarget -> (CloneEnv -> UIO a) -> UIO a
   57 uioWithClone target k = UIO $ do
   58   env <- ask
   59   let runInner cloneEnv = runReaderT (unUIO (k cloneEnv)) env
   60   liftIO (withCloneTarget target runInner)
   61 
   62 -- | Read the 'CloneDeps' from the 'UIOEnv' (for Git opcodes that need
   63 -- direct access to a field not covered by the 'uioCd*' functions above).
   64 uioCdCloneDeps :: UIO CloneDeps
   65 uioCdCloneDeps = UIO $ asks uieCloneDeps

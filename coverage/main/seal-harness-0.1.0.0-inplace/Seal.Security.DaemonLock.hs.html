    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The daemon liveness lock: @\<state\>\/serve.lock@.
    3 --
    4 -- @seal serve@ takes an exclusive advisory lock on this file for its whole
    5 -- lifetime; the commands that must not race a live daemon (@enroll --cancel@,
    6 -- @disable@, @reset@, @migrate-credentials@) probe it with 'daemonHoldsLock'.
    7 --
    8 -- The lock is a POSIX @fcntl@ record lock, which the kernel holds on behalf of
    9 -- the process and releases when the process exits, however it exits. That is
   10 -- the property this module exists for: a daemon that crashed on Tuesday must
   11 -- not make @reset@ refuse on Wednesday. The file's /existence/ therefore means
   12 -- nothing; only the lock's state does.
   13 --
   14 -- Why not the @flock@-style @fileLock@: the @unix@ package shipped with GHC
   15 -- 9.12 (2.8.8.0) exposes only the record-lock API ('getLock', 'setLock',
   16 -- 'waitToSetLock'), and a record lock has the same released-on-death
   17 -- behaviour. The one difference worth knowing: closing /any/ descriptor for
   18 -- this file drops this process's lock on it, so nothing else in the daemon may
   19 -- open @serve.lock@.
   20 module Seal.Security.DaemonLock
   21   ( ServeLock
   22   , acquireServeLock
   23   , releaseServeLock
   24   , daemonHoldsLock
   25   ) where
   26 
   27 import Control.Exception (IOException, try)
   28 import Data.Maybe (isJust)
   29 import Data.Text (Text)
   30 import Data.Text qualified as T
   31 import GHC.IO.Device (SeekMode (AbsoluteSeek))
   32 import System.Posix.IO
   33   ( FileLock, LockRequest (..), OpenFileFlags (creat), OpenMode (ReadWrite)
   34   , closeFd, defaultFileFlags, getLock, openFd, setLock )
   35 import System.Posix.Types (Fd, ProcessID)
   36 
   37 -- | A held lock. Keep it alive for as long as the daemon should look alive;
   38 -- 'releaseServeLock' (or process exit) drops it.
   39 newtype ServeLock = ServeLock Fd
   40 
   41 -- | The whole-file, exclusive lock the daemon holds: lock byte 0 to EOF.
   42 wholeFile :: FileLock
   43 wholeFile = (WriteLock, AbsoluteSeek, 0, 0)
   44 
   45 -- | Take the lock, or report that someone else holds it.
   46 --
   47 -- Non-blocking by design: a second @seal serve@ against the same
   48 -- @SEAL_HOME@ is a mistake to report, not a process to queue behind.
   49 acquireServeLock :: FilePath -> IO (Either Text ServeLock)
   50 acquireServeLock path = do
   51   opened <- try (openFd path ReadWrite defaultFileFlags { creat = Just 0o600 })
   52   case opened of
   53     Left (err :: IOException) -> pure (Left (couldNotOpen path err))
   54     Right fd -> do
   55       locked <- try (setLock fd wholeFile)
   56       case locked of
   57         Right () -> pure (Right (ServeLock fd))
   58         Left (_ :: IOException) -> do
   59           closeFd fd
   60           pure (Left ("another `seal serve` is already running for this SEAL_HOME ("
   61                        <> T.pack path <> " is locked).\n"
   62                        <> "  Stop it first: a second daemon would fight it for the port."))
   63 
   64 -- | Drop the lock early. Process exit does this too.
   65 releaseServeLock :: ServeLock -> IO ()
   66 releaseServeLock (ServeLock fd) = closeFd fd
   67 
   68 -- | Is a live process holding the lock?
   69 --
   70 -- 'getLock' asks the kernel who holds the lock without taking it, which is
   71 -- exactly the question here. A missing lock file, an unreadable one, or a file
   72 -- left behind by a crashed daemon all answer @False@.
   73 daemonHoldsLock :: FilePath -> IO Bool
   74 daemonHoldsLock path = do
   75   opened <- try (openFd path ReadWrite defaultFileFlags { creat = Just 0o600 })
   76   case opened of
   77     -- If the file cannot be opened at all, no daemon can be holding a lock on
   78     -- it, and the command that probes is about to fail on its own write with a
   79     -- better message than "a daemon is running".
   80     Left (_ :: IOException) -> pure False
   81     Right fd -> do
   82       current <- try (getLock fd wholeFile)
   83               :: IO (Either IOException (Maybe (ProcessID, FileLock)))
   84       closeFd fd
   85       pure (either (const True) isJust current)
   86 
   87 couldNotOpen :: FilePath -> IOException -> Text
   88 couldNotOpen path err = T.pack path <> ": could not open the daemon lock: "
   89   <> T.pack (show err)

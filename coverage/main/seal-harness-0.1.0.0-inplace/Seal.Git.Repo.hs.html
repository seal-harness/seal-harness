    2 -- | Git-backed versioning for @~\/.seal\/config@. The config directory is a
    3 -- git repo (initialized on first run if absent). Skills, agent defs, and
    4 -- memory entries live as Markdown files under @config\/skills@,
    5 -- @config\/agents@, @config\/memory@; disk is canonical and git is the

    9 -- The vault file (@config\/vault\/vault.age@) is ciphertext — it lives in the

   39 -- | A handle to the config git repo. The repo root is the @config@ directory.
   40 newtype ConfigRepo = ConfigRepo
   41   { crRoot :: FilePath
   42   -- ^ Absolute path to the git repo (= the @config@ / directory).
   43   }

   45 -- | Open the config repo handle (no IO validation). The caller is responsible
   46 -- for ensuring the repo exists via 'ensureConfigRepo' at startup.
   47 openConfigRepo :: FilePath -> ConfigRepo
   48 openConfigRepo = ConfigRepo

   50 -- | Ensure @config\/@ is a git repo: @git init@ if @.git@ is absent, make
   51 -- an initial empty commit if the repo has no commits yet (so commit-on-write
   52 -- has a base to build on), and create the three store subdirectories
   53 -- (@skills@, @agents@, @memory@) so the Markdown backends can write files
   54 -- into them. Idempotent.
   55 ensureConfigRepo :: FilePath -> IO ()
   56 ensureConfigRepo root = do
   57   -- The root may not exist yet (first run); create it so `git init` has a
   58   -- working directory. createDirectoryIfMissing True creates all parents.
   59   createDirectoryIfMissing True root
   60   let gitDir = root </> ".git"
   61   exists <- doesDirectoryExist gitDir
   62   if exists
   63     then ensureInitialCommit root
   64     else do
   65       _ <- runGit root ["init"]
   66       -- Set a local identity so commits succeed in environments without a
   67       -- global git config (e.g. CI runners). Local config is scoped to this
   68       -- repo and never touches the user's global settings.
   69       _ <- runGit root ["config", "user.name", "seal"]
   70       _ <- runGit root ["config", "user.email", "seal@localhost"]
   71       ensureInitialCommit root
   72   -- The three evolutionary-store subdirectories. Created here (not in the
   73   -- backends) so a hand-edit user who drops a file finds the dirs ready, and
   74   -- so the backends' TIO.writeFile never hits a missing-parent-dir error.
   75   mapM_ (createDirectoryIfMissing True . (root </>)) ["skills", "agents", "memory"]
   76 
   77 -- | Make an initial empty commit if the repo has no commits yet. Without one,
   78 -- @git commit@ on the first real write fails (nothing to commit against, and
   79 -- @HEAD@ is unborn). Idempotent.
   80 ensureInitialCommit :: FilePath -> IO ()
   81 ensureInitialCommit root = do
   82   has <- gitHasCommits' root
   83   if has
   84     then pure ()
   85     else do
   86       -- Commit an empty tree so HEAD is born. -m with --allow-empty.
   87       _ <- runGit root ["commit", "--allow-empty", "-m", "seal: initialize config repo"]
   88       pure ()
   89 
   90 -- | True if the repo has at least one commit on HEAD.
   91 gitHasCommits :: ConfigRepo -> IO Bool
   92 gitHasCommits = gitHasCommits' . crRoot
   93 
   94 gitHasCommits' :: FilePath -> IO Bool
   95 gitHasCommits' root = do
   96   (ec, _out, _err) <- runGit root ["rev-parse", "--verify", "HEAD"]
   97   pure (ec == ExitSuccess)

   99 -- | Stage one path (relative to the repo root). @git add <path>@.
  100 gitAdd :: ConfigRepo -> FilePath -> IO ()
  101 gitAdd repo path = do
  102   (ec, _out, err) <- runGit (crRoot repo) ["add", "--", path]
  103   case ec of
  104     ExitSuccess -> pure ()
  105     ExitFailure n -> ioError (userError ("git add failed (" <> show n <> "): " <> T.unpack err))
  106 
  107 -- | Commit staged changes with a message. Fails if nothing is staged (the
  108 -- caller is responsible for staging via 'gitAdd' first). Returns @True@ if a
  109 -- commit was created, @False@ if there was nothing to commit.
  110 gitCommit :: ConfigRepo -> Text -> IO Bool
  111 gitCommit repo msg = do
  112   (ec, _out, err) <- runGit (crRoot repo) ["commit", "-m", T.unpack msg]
  113   case ec of
  114     ExitSuccess   -> pure True
  115     -- git exits 1 when there is nothing staged; treat as "nothing to commit".
  116     ExitFailure 1 -> pure False
  117     ExitFailure n -> ioError (userError ("git commit failed (" <> show n <> "): " <> T.unpack err))
  118 
  119 -- | Stage one path and commit it with a message. Returns @True@ if a commit
  120 -- was created. The convenience wrapper for the model-authored write path
  121 -- (each opcode mutation is one file → one commit).
  122 gitCommitAll :: ConfigRepo -> FilePath -> Text -> IO Bool
  123 gitCommitAll repo path msg = do
  124   gitAdd repo path
  125   gitCommit repo msg

  127 -- | Run @git@ in @root@ with the given args. Returns the exit code, stdout,
  128 -- and stderr (as decoded Text). Catches process-launch failures (git not on
  129 -- PATH) as an ExitFailure-like result rather than throwing.
  130 runGit :: FilePath -> [String] -> IO (ExitCode, Text, Text)
  131 runGit root args = do
  132   res <- try @SomeException (readProcessBinaryCwd (Just root) "git" args BS.empty)
  133   case res of
  134     Right (ec, out, err) ->
  135       pure (ec, TE.decodeUtf8Lenient out, TE.decodeUtf8Lenient err)
  136     Left e ->
  137       pure (ExitFailure 127, "", T.pack (show e))
  138 
  139 -- | Run a process with an optional working directory and a strict
  140 -- 'ByteString' stdin, capturing stdout and stderr as strict 'ByteString's.
  141 -- Used by the git seam (commit messages may carry non-ASCII bytes).
  142 readProcessBinaryCwd :: Maybe FilePath -> FilePath -> [String] -> ByteString
  143                      -> IO (ExitCode, ByteString, ByteString)
  144 readProcessBinaryCwd mCwd cmdPath args input =
  145   withCreateProcess
  146     ( (proc cmdPath args)
  147         { cwd = mCwd
  148         , std_in = CreatePipe, std_out = CreatePipe, std_err = CreatePipe
  149         }
  150     ) $ \mIn mOut mErr ph -> do
  151       (hIn, hOut, hErr) <- case (mIn, mOut, mErr) of
  152         (Just a, Just b, Just c) -> pure (a, b, c)
  153         _ -> error "readProcessBinaryCwd: pipe creation failed (unreachable)"
  154       BS.hPutStr hIn input
  155       hFlush hIn
  156       hClose hIn
  157       out <- BS.hGetContents hOut
  158       err <- BS.hGetContents hErr
  159       ec  <- waitForProcess ph
  160       let !_ = BS.length out
  161           !_ = BS.length err
  162       pure (ec, out, err)

  164 -- | Like 'readProcessBinaryCwd' but with an explicit environment override.
  165 -- The given @[(String,String)]@ is MERGED over the current process
  166 -- environment ('System.Environment.getEnvironment'), so the child still
  167 -- inherits @PATH@, @HOME@, etc. Used by the clone seam
  168 -- ('Seal.SourceControl.Clone') to inject @GIT_ASKPASS@ / @GIT_SSH_COMMAND@ /
  169 -- @GIT_TERMINAL_PROMPT@ without re-implementing 'withCreateProcess'.
  170 --
  171 -- Security invariant: the @extras@ MUST carry only non-secret values (paths
  172 -- + flags) — the clone seam ensures the secret bytes live only in the temp
  173 -- files referenced by those paths, never in the env values themselves.
  174 readProcessBinaryCwdEnv
  175   :: Maybe FilePath
  176   -> [(String, String)]
  177   -> FilePath
  178   -> [String]
  179   -> ByteString
  180   -> IO (ExitCode, ByteString, ByteString)
  181 readProcessBinaryCwdEnv mCwd extras cmdPath args input = do
  182   inherited <- getEnvironment
  183   let mergedEnv = Map.toList (Map.union (Map.fromList extras) (Map.fromList inherited))
  184   withCreateProcess
  185     ( (proc cmdPath args)
  186         { cwd = mCwd
  187         , env = Just mergedEnv
  188         , std_in = CreatePipe, std_out = CreatePipe, std_err = CreatePipe
  189         }
  190     ) $ \mIn mOut mErr ph -> do
  191       (hIn, hOut, hErr) <- case (mIn, mOut, mErr) of
  192         (Just a, Just b, Just c) -> pure (a, b, c)
  193         _ -> error "readProcessBinaryCwdEnv: pipe creation failed (unreachable)"
  194       BS.hPutStr hIn input
  195       hFlush hIn
  196       hClose hIn
  197       out <- BS.hGetContents hOut
  198       err <- BS.hGetContents hErr
  199       ec  <- waitForProcess ph
  200       let !_ = BS.length out
  201           !_ = BS.length err
  202       pure (ec, out, err)

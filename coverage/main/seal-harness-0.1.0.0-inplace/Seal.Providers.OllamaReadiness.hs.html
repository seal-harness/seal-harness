    2 -- | Pure classification of Ollama's observed readiness.
    3 --
    4 -- The module exists because @configured@ answers "did the operator express
    5 -- intent?" and nothing else. On a machine with no daemon, no binary and no
    6 -- models, @configured@ is still @True@ for a keyless local Ollama — so the
    7 -- provider dropdowns, the CRUD list, the setup wizard and @\/provider list@
    8 -- all asserted usability they had never observed. This module is the second,
    9 -- observed axis.
   10 --
   11 -- Everything here is pure and total. The I/O — the HTTP probe, the cache, the
   12 -- single-flight gate, @findExecutable@, the RAM reader — lives in
   13 -- "Seal.Providers.OllamaReadinessProbe".
   14 --
   15 -- == Why "refused" is decided on errno, never on the message
   16 --
   17 -- Measured on GHC 9.12 \/ http-client against a real closed port:
   18 --
   19 -- > refusal:      type=NoSuchThing  msg="does not exist"  errno=Just 61
   20 -- > DNS failure:  type=NoSuchThing  msg="does not exist"  errno=Nothing
   21 --
   22 -- The two are byte-identical in exception type and message. A substring match
   23 -- on @\"refused\"@ never fires, which would make 'RdNotRunning' unreachable
   24 -- and re-create the very bug this module exists to fix. 'classifyProbeFailure'
   25 -- therefore discriminates on 'ioe_errno' alone, and everything it cannot
   26 -- positively identify is 'OpeOther' — "do not claim to know".

   62 -- | The second, observed axis. 'RdPending' means /this process has not
   63 -- finished a probe yet/ — it is the only state that renders as nothing,
   64 -- because "we have not looked" must never be shown as a verdict.
   65 data OllamaReadiness
   66   = RdPending          -- ^ no probe has completed yet in this process
   67   | RdUnknown          -- ^ a probe ran and failed unexplainably
   68   | RdNotInstalled     -- ^ loopback URL, nothing listening, no binary on PATH
   69   | RdNotRunning       -- ^ loopback URL, connection REFUSED
   70   | RdNoModels         -- ^ daemon answered, zero models
   71   | RdNoFittingModel   -- ^ daemon answered, models exist, none usable (advisory)
   72   | RdReady !Int       -- ^ N usable models; @n >= 1@ by construction
   73   | RdCloud            -- ^ ollama.com: no probe performed
   74   deriving stock (Eq, Show)
   75 
   76 -- | Why a probe did not produce a tag list. 'OpeRefused' is the only
   77 -- constructor that permits the 'RdNotRunning' claim.
   78 data OllamaProbeError
   79   = OpeRefused       -- ^ only: ConnectionFailure with @ioe_errno == eCONNREFUSED@
   80   | OpeBlocked       -- ^ refused by the SSRF floor / URL-shape check, pre-request
   81   | OpeTimeout
   82   | OpeBadResponse   -- ^ something answered, but not with Ollama's JSON (or over the cap)
   83   | OpeHttp !Int     -- ^ non-2xx status; a 3xx lands here (redirects are never followed)
   84   | OpeOther         -- ^ any other transport failure (reset, TLS, DNS, non-IOError)
   85   deriving stock (Eq, Show)
   86 
   87 -- | Where the base URL points. Split from the binary check because only the
   88 -- default-loopback spelling may refine its wording from @findExecutable@
   89 -- (D3.3): a Docker-hosted daemon is "not running", not "not installed".
   90 data OllamaHostKind
   91   = OhCloud           -- ^ host is ollama.com: no probe at all
   92   | OhDefaultLoopback -- ^ literally 'defaultOllamaBaseUrl': binary wording allowed
   93   | OhLoopback        -- ^ 127\/8 \/ [::1] \/ localhost, any port
   94   | OhOtherHost       -- ^ LAN \/ remote
   95   deriving stock (Eq, Show)
   96 
   97 -- | One installed model, as @\/api\/tags@ reports it.
   98 --
   99 -- The two optional fields are what make the fit judgement hedgeable rather
  100 -- than assertive: an absent size or an absent RAM figure is /no evidence/, and
  101 -- no evidence must never become a claim (design D5.3).
  102 data OllamaTag = OllamaTag
  103   { otName       :: !Text
  104   , otSizeBytes  :: !(Maybe Word64)
  105     -- ^ 'Nothing' = the daemon reported no size.
  106   , otRemoteHost :: !(Maybe Text)
  107     -- ^ 'Just' = the model runs server-side (cloud), so it is never sized
  108     --   against local RAM.
  109   } deriving stock (Eq, Show)
  110 
  111 -- | The pure inputs to classification. Assembled by the I/O module.
  112 data OllamaProbe = OllamaProbe
  113   { opHostKind      :: !OllamaHostKind
  114   , opTags          :: !(Either OllamaProbeError [OllamaTag])
  115   , opBinaryPresent :: !Bool
  116     -- ^ Only consulted for 'OhDefaultLoopback' \/ 'OhLoopback' refusals.
  117   , opTotalRam      :: !(Maybe Word64)
  118     -- ^ Unused in WU-A; WU-B's fit predicate consumes it.
  119   } deriving stock (Eq, Show)
  120 
  121 -- | The state machine. Total, and deliberately narrow: every branch that is
  122 -- not positively identified degrades to 'RdUnknown'.
  123 classifyOllamaReadiness :: OllamaProbe -> OllamaReadiness
  124 classifyOllamaReadiness p = case opHostKind p of
  125   -- An ollama.com endpoint is not probed at all, so there is nothing to
  126   -- classify — and "not ready" would simply be false.
  127   OhCloud -> RdCloud
  128   _ -> case opTags p of
  129     Right [] -> RdNoModels
  130     Right tags -> classifyTags (opTotalRam p) tags
  131     Left OpeRefused
  132       -- A refusal is evidence about *this* machine only when the URL points
  133       -- at this machine. For a LAN or remote host, our own PATH says nothing.
  134       | opHostKind p == OhDefaultLoopback || opHostKind p == OhLoopback ->
  135           if opBinaryPresent p then RdNotRunning else RdNotInstalled
  136       | otherwise -> RdUnknown
  137     -- Everything else — blocked, timed out, malformed, an HTTP status, a
  138     -- reset, TLS, DNS — is "we do not know", never "not running".
  139     Left _ -> RdUnknown
  140 
  141 -- | Count the usable models, or report that none of the /sized/ ones fit.
  142 --
  143 -- The rules, all from design D5.3, are deliberately symmetric — RAM and size
  144 -- are treated the same way, because an asymmetry would emit
  145 -- @May Not Fit in RAM@ with zero evidence on one side:
  146 --
  147 -- * A /cloud/ model is always usable and is never sized locally
  148 --   ('isCloudModel').
  149 -- * An unknown size counts as usable: we cannot substantiate a problem.
  150 -- * Unknown RAM means /every/ local model is usable, and never
  151 --   'RdNoFittingModel'.
  152 -- * 'RdNoFittingModel' requires at least one /known/ size, and none of the
  153 --   known-size local models fitting.
  154 --
  155 -- The "at least one known size" requirement is satisfied /by construction/
  156 -- rather than by a separate guard, which is worth stating because it is the
  157 -- kind of thing that looks like a missing case: 'isUsable' returns 'True' for
  158 -- an unknown size, so @usable == 0@ can only happen when /every/ tag had a
  159 -- known size and none of them fit. If a size is ever unknown, the model counts
  160 -- as usable and we take the @RdReady@ branch — exactly the D5.3 rule
  161 -- ("every size unknown ⇒ RdReady with the full count"), reached without an
  162 -- extra test. An explicit @any hasKnownSize@ guard here would be dead code.
  163 classifyTags :: Maybe Word64 -> [OllamaTag] -> OllamaReadiness
  164 classifyTags mRam tags =
  165   case mRam of
  166     -- No RAM figure: nothing to judge against, so everything is usable.
  167     Nothing -> RdReady (length tags)
  168     Just ram ->
  169       let usable = length (filter (isUsable ram) tags)
  170       in if usable > 0
  171            then RdReady usable
  172            else RdNoFittingModel
  173 
  174 -- | Is a model usable at the given total RAM?
  175 --
  176 -- Cloud models always are. A local model is usable when its size is unknown
  177 -- (no evidence) or when 1.25x its size still fits in RAM. The comparison is
  178 -- exact integer arithmetic — @size * 5 < ram * 4@ — so the predicate stays
  179 -- pure and exact, and never rounds in the operator's favour or against it.
  180 --
  181 -- The arithmetic is done in 'Integer', not 'Word64'. @size@ comes from the
  182 -- daemon, so it is untrusted input: a size near @2^64@ would wrap a 'Word64'
  183 -- multiplication and make an absurd model compare as fitting. @Integer@ is
  184 -- arbitrary-precision, so that cannot happen. RAM is bounded in practice, but
  185 -- widening both sides keeps the comparison obviously exact rather than
  186 -- exact-by-luck.
  187 isUsable :: Word64 -> OllamaTag -> Bool
  188 isUsable ram t
  189   | isCloudModel t = True
  190   | otherwise = case otSizeBytes t of
  191       Nothing   -> True
  192       Just size -> (toInteger size * 5) < (toInteger ram * 4)
  193 
  194 -- | Is this model served remotely?
  195 --
  196 -- True when @remote_host@ is present /or/ the name carries the @:cloud@
  197 -- suffix. @remote_host@ is authoritative when the two disagree: the suffix
  198 -- alone is a naming convention, while the field is what the daemon actually
  199 -- does.
  200 isCloudModel :: OllamaTag -> Bool
  201 isCloudModel t = case otRemoteHost t of
  202   Just _  -> True
  203   Nothing -> ":cloud" `T.isSuffixOf` otName t
  204 
  205 -- | Parse @MemTotal:@ from one @\/proc\/meminfo@ line, as bytes.
  206 --
  207 -- @kB@ is the only accepted unit. A non-kB suffix yields 'Nothing' rather than
  208 -- a silent rescale: a wrong 1000x guess would be worse than "RAM unknown",
  209 -- which degrades safely to "assume usable". @MemTotal: 0 kB@ is 'Just 0' — a
  210 -- real (if pathological) kernel value, not an absence.
  211 parseMemTotal :: Text -> Maybe Word64
  212 parseMemTotal line = do
  213   rest <- T.stripPrefix "MemTotal:" line
  214   case T.words rest of
  215     [valueT, unitT]
  216       | unitT == "kB" -> (* 1024) <$> readDecimal valueT
  217     _ -> Nothing
  218   where
  219     readDecimal t = case reads (T.unpack t) of
  220       [(n, "")] -> Just n
  221       _         -> Nothing

  223 -- | How many models 'RdReady' counts. Exposed so the wire and the label
  224 -- cannot disagree.
  225 readinessUsableCount :: OllamaReadiness -> Maybe Int
  226 readinessUsableCount (RdReady n) = Just n
  227 readinessUsableCount _           = Nothing

  229 -- | The CLI's rendering (@\/provider list@). Total over all eight
  230 -- constructors; 'RdPending' is the empty string, and the caller omits the
  231 -- segment when the text is empty rather than printing a stray separator.
  232 readinessText :: OllamaReadiness -> Text
  233 readinessText = \case
  234   RdPending        -> ""
  235   RdUnknown        -> "Status Unknown"
  236   RdNotInstalled   -> "Ollama Not Found"
  237   RdNotRunning     -> "Not Running"
  238   RdNoModels       -> "No Models Pulled"
  239   RdNoFittingModel -> "May Not Fit in RAM"
  240   RdReady 1        -> "Ready · 1 Model"
  241   RdReady n        -> "Ready · " <> T.pack (show n) <> " Models"
  242   RdCloud          -> "Cloud"

  261 -- | Turn an 'HttpException' into the probe's own error type. The /only/ way
  262 -- to get 'OpeRefused' is a @ConnectionFailure@ whose payload unwraps to an
  263 -- 'IOException' with @ioe_errno == Just n@ and @Errno n == eCONNREFUSED@.
  264 --
  265 -- Note the unwrap: @ConnectionFailure@ carries a 'SomeException', not an
  266 -- 'IOException', so matching on an 'IOException' pattern directly does not
  267 -- typecheck. The trailing catch-all is an intentional fail-safe, not an
  268 -- oversight — it absorbs any future 'HttpException' shape as "do not claim to
  269 -- know" rather than inventing a state.
  270 classifyProbeFailure :: HttpException -> OllamaProbeError
  271 classifyProbeFailure (HttpExceptionRequest _ (ConnectionFailure se)) =
  272   case fromException se :: Maybe IOException of
  273     Just ioe -> case ioe_errno ioe of
  274       Just n | Errno n == eCONNREFUSED -> OpeRefused
  275       _                                -> OpeOther
  276     Nothing -> OpeOther
  277 classifyProbeFailure (HttpExceptionRequest _ ResponseTimeout) = OpeTimeout
  278 classifyProbeFailure _ = OpeOther

  284 -- | The probe's per-request timeout. Two seconds, deliberately /not/ the
  285 -- provider's five-minute 'Seal.Providers.Ollama.ollamaResponseTimeoutMicro':
  286 -- the cache caps in-flight probes at one globally, so a probe that never
  287 -- returns would hold that slot forever and starve every later refresh.
  288 probeTimeoutMicro :: Int
  289 probeTimeoutMicro = 2_000_000
  290 
  291 -- | Validate a base URL and build the probe request. Pure, so the URL-shape
  292 -- and SSRF rules are testable without a socket.
  293 --
  294 -- Rejected, in order of cheapness:
  295 --
  296 -- * a scheme other than http\/https ('OpeBlocked');
  297 -- * any authority carrying userinfo ('OpeBlocked') — @http:\/\/x\@169.254.169.254\/@
  298 --   is the bypass this rule closes;
  299 -- * a URL the SSRF floor blocks ('OpeBlocked') — the cloud-metadata literals;
  300 -- * a URL http-client cannot parse ('OpeBadResponse').
  301 --
  302 -- The resulting request is built from 'parseRequest' plus explicit fields. It
  303 -- never goes through @withHeaders@, so it cannot acquire the vault key: the
  304 -- probe is keyless by construction (D2.1).
  305 probeTargetFor :: Text -> Either OllamaProbeError Request
  306 probeTargetFor baseUrl
  307   | not (isHttpScheme baseUrl) = Left OpeBlocked
  308   | carriesUserinfo baseUrl = Left OpeBlocked
  309   | isAlwaysBlockedUrl baseUrl = Left OpeBlocked
  310   | otherwise = case parseRequest (T.unpack ("GET " <> baseUrl <> "/api/tags")) of
  311       Left _ -> Left OpeBadResponse
  312       Right req -> Right req
  313           { requestHeaders  = [ (hContentType, "application/json") ]
  314           -- Never follow a redirect: a 3xx is reported, not chased. This is
  315           -- what keeps a metadata redirect from being reached at all.
  316           , redirectCount   = 0
  317           , responseTimeout = responseTimeoutMicro probeTimeoutMicro
  318           }

  320 -- | Everything after @scheme://@. 'T.breakOn' leaves the delimiter in the
  321 -- suffix, so it must be dropped — keeping it would make the authority
  322 -- @"://localhost:11434"@ and every host comparison fail.
  323 afterSchemeOf :: Text -> Text
  324 afterSchemeOf url = T.drop 3 (snd (T.breakOn "://" url))
  325 
  326 -- | The scheme check. Deliberately not a URL parser: it only needs to know
  327 -- whether the prefix is one of the two allowed schemes.
  328 isHttpScheme :: Text -> Bool
  329 isHttpScheme url =
  330   "http://" `T.isPrefixOf` lowered || "https://" `T.isPrefixOf` lowered
  331   where lowered = T.toLower url
  332 
  333 -- | Does the authority carry userinfo (an @\@@ before the path)? This is the
  334 -- D2.8 bypass guard. The last @\@@ in the authority is the separator, because
  335 -- a password may itself contain one.
  336 carriesUserinfo :: Text -> Bool
  337 carriesUserinfo url =
  338   let afterScheme = afterSchemeOf url
  339       authority = T.takeWhile (\c -> c /= '/' && c /= '?' && c /= '#') afterScheme
  340   in T.any (== '@') authority
  341 
  342 -- ---------------------------------------------------------------------------
  343 -- Host kind
  344 -- ---------------------------------------------------------------------------
  345 
  346 -- | Which host kind a base URL is. The comparison is on the authority (after
  347 -- userinfo and port stripping), and @localhost@ must match /exactly/ — a
  348 -- @localhost.attacker.tld@ is 'OhOtherHost', not the default loopback.
  349 hostKindFor :: Text -> OllamaHostKind
  350 hostKindFor baseUrl
  351   | host == "ollama.com" = OhCloud
  352   | stripTrailingSlash baseUrl == defaultOllamaBaseUrl = OhDefaultLoopback
  353   | isLoopbackHost host = OhLoopback
  354   | otherwise = OhOtherHost
  355   where
  356     host = hostOf baseUrl
  357 
  358 -- | The authority's host: userinfo stripped, port stripped, IPv6 brackets
  359 -- stripped and lowercased. A small local twin of
  360 -- @Seal.Web.UrlSafety@'s internal helper (which is deliberately not exported);
  361 -- keeping it here means the readiness rules are testable without widening that
  362 -- module's public surface.
  363 hostOf :: Text -> Text
  364 hostOf url =
  365   let afterScheme = afterSchemeOf url
  366       authority = T.takeWhile (\c -> c /= '/' && c /= '?' && c /= '#') afterScheme
  367       afterUserinfo = snd (T.breakOnEnd "@" authority)
  368   in T.toLower $ case T.uncons afterUserinfo of
  369        Just ('[', rest) -> T.takeWhile (/= ']') rest
  370        _                -> T.takeWhile (/= ':') afterUserinfo
  371 
  372 -- | The loopback aliases that count as "this machine" for state purposes
  373 -- (D3.4): @localhost@, all of @127\/8@, and the IPv6 loopback @::1@. Only the
  374 -- exact spellings count — a hostname that merely starts with one does not.
  375 isLoopbackHost :: Text -> Bool
  376 isLoopbackHost host =
  377   host == "localhost"
  378   || host == "::1"
  379   || case T.stripPrefix "127." host of
  380        Just rest -> not (T.null rest) && T.all isDigitable rest
  381        Nothing   -> False
  382   where
  383     isDigitable c = isDigit c || c == '.'

  389 -- | The single-flight key: the base URL with one trailing slash removed. Two
  390 -- spellings that differ only by that slash must share one in-flight probe
  391 -- (D2.10).
  392 singleFlightKey :: Text -> Text
  393 singleFlightKey = stripTrailingSlash
  394 
  395 -- | Drop exactly one trailing slash. A local twin of the same rule in
  396 -- @Seal.Providers.Ollama@, which keeps it private; duplicating three lines is
  397 -- cheaper than widening that module's surface, and both spellings must agree
  398 -- so the cache key is stable.
  399 stripTrailingSlash :: Text -> Text
  400 stripTrailingSlash t = fromMaybe t (T.stripSuffix "/" t)

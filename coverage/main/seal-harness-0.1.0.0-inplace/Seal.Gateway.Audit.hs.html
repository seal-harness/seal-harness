    2 -- | The access-control audit log: @\<state\>\/access-control.log@.
    3 --
    4 -- Detective, not preventive (design §15). It exists so an operator can answer
    5 -- "when did this passkey appear, and what authorised it?" after the fact —
    6 -- under the accepted risk R2 (same-uid code can truncate it) it is a detective
    7 -- aid, not tamper-evidence. That sentence is the honest scope; nothing here
    8 -- should be read as stronger.
    9 --
   10 -- Two properties are load-bearing and both are enforced structurally:
   11 --
   12 -- * __The field set is closed.__ 'AuditFields' is a record, and 'auditLine'
   13 --   renders exactly its fields. A caller cannot add a request-derived value —
   14 --   no @rawId@, no client-supplied origin, no presented token, no digest, no
   15 --   @User-Agent@ — because there is nowhere to put one.
   16 -- * __Refusals cannot flood.__ 'admitRefusal' aggregates them per
   17 --   @(source, minute)@ on a fixed-size table with backoff, and they go to a
   18 --   separate file, so a pre-auth flood cannot rotate the state-changing
   19 --   evidence away.
   20 --
   21 -- The log is opened @O_APPEND|O_CREAT|O_NOFOLLOW@ at @0600@ and refused if the
   22 -- target is a symlink or not a regular file: a log that can be pointed at
   23 -- another file is a write primitive.

   62 -- | The enumerated events. Nothing else may be logged, so the log's vocabulary
   63 -- is a compile-time fact rather than a convention.
   64 data AuditEvent
   65   = CredentialAdded
   66   | CredentialRemoved
   67   | CredentialRelabeled
   68   | GrantMinted
   69   | GrantConsumed
   70   | GrantRefused
   71   | ModeEnabled
   72   | ModeDisabled
   73   deriving stock (Eq, Ord, Show)
   74 
   75 auditEventText :: AuditEvent -> Text
   76 auditEventText event = case event of
   77   CredentialAdded -> "credential_added"
   78   CredentialRemoved -> "credential_removed"
   79   CredentialRelabeled -> "credential_relabeled"
   80   GrantMinted -> "grant_minted"
   81   GrantConsumed -> "grant_consumed"
   82   GrantRefused -> "grant_refused"
   83   ModeEnabled -> "mode_enabled"
   84   ModeDisabled -> "mode_disabled"

   86 -- | The closed field set.
   87 data AuditFields = AuditFields

  109 -- | Render one line.
  110 --
  111 -- Values are sanitised: control characters become spaces and each value is
  112 -- capped, so a credential label an operator typed cannot forge a log line or
  113 -- blow the file up. The event names come from 'auditEventText', never from
  114 -- input.
  115 auditLine :: UTCTime -> AuditFields -> Text
  116 auditLine now fields = T.intercalate " " (timestampPart : map renderField present)
  117   where
  118     timestampPart = "ts=" <> isoTimeText now
  119     present = [ ("event", auditEventText (afEvent fields)) ]
  120       <> [ ("credential_id", value) | Just value' <- [afCredentialId fields]
  121                                     , let value = sanitize 128 value' ]
  122       <> [ ("label", sanitize 64 value) | Just value <- [afLabel fields] ]
  123       <> [ ("rp_id", sanitize 128 value) | Just value <- [afRpId fields] ]
  124       <> [ ("origin", sanitize 200 value) | Just value <- [afOrigin fields] ]
  125       <> [ ("via", grantSourceText source) | Just source <- [afVia fields] ]
  126       <> [ ("remaining_s", tshow value) | Just value <- [afRemainingSeconds fields] ]
  127     renderField (key, value) = key <> "=" <> value
  128 
  129 -- | Percent-encode everything that could break the line format.
  130 --
  131 -- The log is @key=value@ pairs separated by single spaces, so a value must never
  132 -- contain a space, an @=@ or a control character: otherwise a credential label
  133 -- an operator typed (or a hostile one) could append its own fields. Encoding
  134 -- rather than stripping keeps the value recognisable.
  135 sanitize :: Int -> Text -> Text
  136 sanitize limit = T.take limit . T.concatMap escape
  137   where
  138     escape c
  139       | c == ' '  = "%20"
  140       | c == '='  = "%3D"
  141       | c == '"'  = "%22"
  142       | c == '\n' = "%0A"
  143       | c == '\r' = "%0D"
  144       | c < ' ' || c == '\DEL' = "%" <> pad (showHexChar c)
  145       | otherwise = T.singleton c
  146     pad raw = if T.length raw < 2 then "0" <> raw else raw
  147     showHexChar c =
  148       let n = fromEnum c
  149       in T.map toUpperChar (T.pack [ hexDigit (n `div` 16), hexDigit (n `mod` 16) ])
  150     toUpperChar ch = if ch >= 'a' && ch <= 'f' then toEnum (fromEnum ch - 32) else ch
  151     hexDigit n
  152       | n < 10 = toEnum (fromEnum '0' + n)
  153       | otherwise = toEnum (fromEnum 'A' + n - 10)
  154 
  155 isoTimeText :: UTCTime -> Text
  156 isoTimeText stamp =
  157   let seconds = floor (utcTimeToPOSIXSeconds stamp) :: Integer
  158       daySeconds = seconds `mod` 86400
  159       hours = daySeconds `div` 3600
  160       minutes = (daySeconds `mod` 3600) `div` 60
  161       secs = daySeconds `mod` 60
  162       days = seconds `div` 86400
  163       (year, month, day) = civilFromDays days
  164       pad2 n = if n < 10 then "0" <> tshow n else tshow n
  165   in tshow year <> "-" <> pad2 month <> "-" <> pad2 day <> "T"
  166        <> pad2 hours <> ":" <> pad2 minutes <> ":" <> pad2 secs <> "Z"
  167 
  168 -- | Howard Hinnant's @civil_from_days@: days since the Unix epoch to a calendar
  169 -- date, without pulling a formatter into a hot path.
  170 civilFromDays :: Integer -> (Integer, Integer, Integer)
  171 civilFromDays days =
  172   let z = days + 719468
  173       era = (if z >= 0 then z else z - 146096) `div` 146097
  174       doe = z - era * 146097
  175       yoe = (doe - doe `div` 1460 + doe `div` 36524 - doe `div` 146096) `div` 365
  176       y = yoe + era * 400
  177       doy = doe - (365 * yoe + yoe `div` 4 - yoe `div` 100)
  178       mp = (5 * doy + 2) `div` 153
  179       d = doy - (153 * mp + 2) `div` 5 + 1
  180       m = if mp < 10 then mp + 3 else mp - 9
  181   in (if m <= 2 then y + 1 else y, m, d)

  186 -- | State-changing events, in one file.
  187 auditLogPath :: SealPaths -> FilePath
  188 auditLogPath paths = spState paths </> "access-control.log"

  190 -- | Refusals, in another: a flood must not rotate the evidence away.
  191 auditRefusalLogPath :: SealPaths -> FilePath
  192 auditRefusalLogPath paths = spState paths </> "access-control-refusals.log"

  194 -- | Rotate at 4 MiB.
  195 auditMaxBytes :: FileOffset
  196 auditMaxBytes = 4 * 1024 * 1024
  197 
  198 -- | Append one event.
  199 --
  200 -- Failures are returned rather than thrown: an audit write must never fail the
  201 -- request or the command it describes.
  202 appendAudit :: SealPaths -> UTCTime -> AuditFields -> IO (Either Text ())
  203 appendAudit paths now fields = do
  204   let path = case afEvent fields of
  205         GrantRefused -> auditRefusalLogPath paths
  206         _            -> auditLogPath paths
  207   rotateIfNeeded path
  208   appended <- try (appendLine path (auditLine now fields))
  209   pure $ case appended of
  210     Left (err :: IOException) -> Left (T.pack path <> ": " <> T.pack (show err))
  211     Right () -> Right ()
  212 
  213 appendLine :: FilePath -> Text -> IO ()
  214 appendLine path line = do
  215   fd <- openFd path WriteOnly defaultFileFlags
  216           { append = True, creat = Just 0o600, nofollow = True }
  217   status <- getFileStatus path
  218   -- A symlink is refused by O_NOFOLLOW; anything else that is not a regular
  219   -- file (a fifo, a device) is refused here, because appending to it is not
  220   -- writing a log.
  221   if not (isRegularFile status)
  222     then hClose =<< fdToHandle fd
  223     else do
  224       handle <- fdToHandle fd
  225       BS.hPut handle (TE.encodeUtf8 (line <> "\n"))
  226       hClose handle
  227 
  228 rotateIfNeeded :: FilePath -> IO ()
  229 rotateIfNeeded path = do
  230   exists <- doesFileExist path
  231   when exists $ do
  232     status <- getFileStatus path
  233     when (fileSize status > auditMaxBytes) $ do
  234       let rotated = path <> ".1"
  235       rotatedExists <- doesFileExist rotated
  236       when rotatedExists (removeFile rotated)
  237       renameFile path rotated

  239 -- | Aggregation state for refusal events.
  240 --
  241 -- Keyed on the /source/ and the minute, never on the presented value: an
  242 -- attacker-chosen key would be an unbounded allocation on a pre-auth path.
  243 newtype AuditGate = AuditGate (MVar (Map (Text, Integer) RefusalSlot))
  244 
  245 data RefusalSlot = RefusalSlot
  246   { rsCount    :: !Int
  247   , rsLastEmit :: !UTCTime
  248   }
  249 
  250 newAuditGate :: IO AuditGate
  251 newAuditGate = AuditGate <$> newMVar Map.empty
  252 
  253 -- | May this refusal be written?
  254 --
  255 -- One line per @(source, minute)@ at most, and exponential backoff on repeats
  256 -- (1 s doubling to a 60 s cap), so a pre-auth flood produces a handful of lines
  257 -- rather than a gigabyte. The table is capped: once it is full, further keys are
  258 -- dropped rather than allowed to grow it.
  259 admitRefusal :: AuditGate -> UTCTime -> Text -> IO Bool
  260 admitRefusal (AuditGate cells) now source =
  261   modifyMVar cells $ \table -> do
  262     let key = (source, minuteOf now)
  263         backoff count = fromIntegral (min (60 :: Int) (2 ^ max 0 (count - 1)))
  264     case Map.lookup key table of
  265       Just slot ->
  266         if realToFrac (diffUTCTime now (rsLastEmit slot)) < (backoff (rsCount slot) :: Double)
  267           then pure (table, False)
  268           else pure (Map.insert key slot { rsCount = rsCount slot + 1, rsLastEmit = now } table, True)
  269       Nothing ->
  270         if Map.size table >= maxRefusalKeys
  271           then pure (table, False)
  272           else pure (Map.insert key (RefusalSlot 1 now) table, True)
  273 
  274 maxRefusalKeys :: Int
  275 maxRefusalKeys = 256
  276 
  277 minuteOf :: UTCTime -> Integer
  278 minuteOf stamp = floor (utcTimeToPOSIXSeconds stamp) `div` 60

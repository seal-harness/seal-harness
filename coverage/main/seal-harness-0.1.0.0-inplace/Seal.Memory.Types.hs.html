    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The memory entry model. 'MemoryId' is a smart-constructed newtype with the
    3 -- same charset predicate as 'Seal.Core.Types.SessionId' (@[A-Za-z0-9_-]+@,
    4 -- non-empty, no leading dot). 'MemoryEntry' is the agent's own persistent
    5 -- memory — content the agent chooses to remember across sessions. It is NOT a
    6 -- vault secret (those live in 'Seal.Security.Secrets'); memory content is
    7 -- agent-visible data, recorded in full as a Markdown file under
    8 -- @config\/memory\/@ (disk is canonical; git is the versioning layer).
    9 module Seal.Memory.Types
   10   ( MemoryId (..)
   11   , mkMemoryId
   12   , isValidMemoryId
   13   , memoryIdText
   14   , MemoryEntry (..)
   15   ) where

   21 import GHC.Generics (Generic)

   25 -- | Opaque memory key. Smart-constructed via 'mkMemoryId'; the charset
   26 -- predicate guards every path / filename position.
   27 newtype MemoryId = MemoryId Text
   28   deriving stock (Eq, Ord, Show)
   29   deriving newtype (ToJSON, FromJSON)
   30 
   31 -- | @[A-Za-z0-9_-]+@, non-empty, no leading dot. Mirrors 'isValidSessionId'.
   32 isValidMemoryId :: Text -> Bool
   33 isValidMemoryId t =
   34   not (T.null t)
   35     && T.head t /= '.'
   36     && T.all (`elem` chars) t
   37   where
   38     chars = ['A' .. 'Z'] <> ['a' .. 'z'] <> ['0' .. '9'] <> "_-"
   39 
   40 mkMemoryId :: Text -> Either Text MemoryId
   41 mkMemoryId t
   42   | isValidMemoryId t = Right (MemoryId t)
   43   | otherwise         = Left ("invalid memory id: " <> T.pack (show t))
   44 
   45 memoryIdText :: MemoryId -> Text
   46 memoryIdText (MemoryId t) = t
   47 
   48 -- | One agent memory entry. The content is agent-visible data (not a vault
   49 -- secret); it is recorded in full as a Markdown file. 'meSession' is the
   50 -- originating session (provenance).
   51 data MemoryEntry = MemoryEntry
   52   { meId        :: MemoryId
   53   , meContent   :: Text
   54   , meTags      :: [Text]
   55   , meCreatedAt :: UTCTime
   56   , meUpdatedAt :: UTCTime
   57   , meSession   :: SessionId
   58   } deriving stock (Eq, Show, Generic)
   59 
   60 instance ToJSON MemoryEntry
   61 instance FromJSON MemoryEntry

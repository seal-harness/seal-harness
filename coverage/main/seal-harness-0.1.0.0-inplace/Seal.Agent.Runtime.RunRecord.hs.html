    3 -- | The durable sub-agent run record registry. Tracks every spawned child's
    4 -- full lifecycle from creation through completion (or cancellation), with
    5 -- generation tokens to prevent stale completions, lifecycle hook emission,
    6 -- session cleanup, and disk persistence for restart recovery.
    7 --
    8 -- This module is the foundation for push-based completion notification:
    9 -- instead of the parent polling @AGENT_MANAGE status@, the harness tracks
   10 -- each child run in a 'SubagentRunRecord' and delivers completion
   11 -- automatically via the sidecar mechanism (see 'Seal.Handles.Transcript').
   12 --
   13 -- The registry is STM-backed (like 'Seal.Agent.Runtime.Registry') and
   14 -- thread-safe. Records are keyed by run id (the 'SubagentId' text).

  171 -- | The core tracking record for each spawned child. Stored in the in-memory
  172 -- registry AND persisted to disk so completion can be recovered after
  173 -- restarts. See the design doc for the full field-by-field rationale.
  174 data SubagentRunRecord = SubagentRunRecord
  175   { rrrRunId                    :: !RunId
  176     -- ^ Unique run identifier (the 'SubagentId' text).
  177   , rrrSubagentId               :: !SubagentId
  178     -- ^ The 'SubagentId' for cross-referencing with 'AgentRuntime'.
  179   , rrrChildSessionKey          :: !SessionId
  180     -- ^ The child's session ID.
  181   , rrrParentSessionKey         :: !SessionId
  182     -- ^ The parent's session ID.
  183   , rrrControllerSessionKey     :: !SessionId
  184     -- ^ Who controls this run (usually the parent).
  185   , rrrStartedAt                :: !UTCTime
  186     -- ^ When the run was created.
  187   , rrrEndedAt                  :: !(Maybe UTCTime)
  188     -- ^ Set when the run reaches a terminal state.
  189   , rrrOutcome                  :: !RunOutcome
  190     -- ^ Terminal outcome (ok \/ error \/ timeout \/ killed \/ unknown).
  191   , rrrEndedReason              :: !(Maybe EndReason)
  192     -- ^ Why the run ended.
  193   , rrrExpectsCompletionMessage :: !Bool
  194     -- ^ Does the parent want push notification? 'False' for foreground mode.
  195   , rrrFrozenResultText         :: !(Maybe Text)
  196     -- ^ The child's final output, captured at completion.
  197   , rrrAnnounceRetryCount       :: !Int
  198     -- ^ Delivery retry counter.
  199   , rrrLastAnnounceRetryAt      :: !(Maybe UTCTime)
  200     -- ^ When the last delivery retry was attempted.
  201   , rrrSpawnMode                :: !SpawnMode
  202     -- ^ Foreground (blocking) or background (async with push).
  203   , rrrCleanup                  :: !CleanupMode
  204     -- ^ Whether to delete or keep the child session on completion.
  205   , rrrDepth                    :: !Int
  206     -- ^ Spawn depth (guard against runaway recursion).
  207   , rrrWakeOnDescendantSettle   :: !Bool
  208     -- ^ Re-invoke the child after its own descendants finish.
  209   , rrrCleanupHandled           :: !Bool
  210     -- ^ Has session cleanup been performed?
  211   , rrrCleanupCompletedAt       :: !(Maybe UTCTime)
  212     -- ^ When cleanup was completed.
  213   , rrrEndedHookEmittedAt       :: !(Maybe UTCTime)
  214     -- ^ When the @subagent_ended@ hook was emitted (idempotency guard).
  215   , rrrSuppressAnnounceReason   :: !(Maybe Text)
  216     -- ^ If set, completion announcement is suppressed (e.g. \"killed\").
  217   , rrrGenerationToken          :: !(Maybe GenerationToken)
  218     -- ^ Unique per-spawn token; stale tokens are rejected by 'completeRun'.
  219   } deriving stock (Eq, Show)
  220     deriving (Generic)

  301 -- | The STM-backed registry of run records, keyed by 'RunId'. Thread-safe.
  302 -- Lifecycle hooks are stored in an 'IORef' list (appended on registration).
  303 -- Completion signals (WU-7 Path A) are per-run @TMVar@s stored in a
  304 -- separate @TVar@ map; 'awaitCompletion' blocks on the @TMVar@ while
  305 -- 'completeRun' and 'cancelRunsForParent' fill it.
  306 data RunRecordRegistry = RunRecordRegistry
  307   { rrrRegistry  :: !(TVar (Map RunId SubagentRunRecord))
  308   , rrrHooks     :: !(IORef [SubagentRunRecord -> IO ()])
  309   , rrrSignals   :: !(TVar (Map RunId (TMVar SubagentRunRecord)))
  310   }
  311 
  312 -- | Build an empty registry.
  313 newRunRecordRegistry :: IO RunRecordRegistry
  314 newRunRecordRegistry = do
  315   tv <- newTVarIO Map.empty
  316   hooksRef <- newIORef []
  317   signalsTv <- newTVarIO Map.empty
  318   pure (RunRecordRegistry tv hooksRef signalsTv)

  324 -- | Create a new run record and store it in the registry. The generation
  325 -- token is freshly minted; 'completeRun' will reject calls with a
  326 -- different token.
  327 createRun
  328   :: RunRecordRegistry
  329   -> SubagentId
  330   -> SessionId
  331      -- ^ child session
  332   -> SessionId
  333      -- ^ parent session (also the controller by default)
  334   -> Int
  335      -- ^ spawn depth
  336   -> SpawnMode
  337   -> IO SubagentRunRecord
  338 createRun reg subagentId childSid parentSid depth mode = do
  339   now <- getCurrentTime
  340   token <- GenerationToken <$> randomRIO (1, maxBound :: Int)
  341   let runId = subagentIdText subagentId
  342       expectsMsg = case mode of
  343         SpawnForeground -> False
  344         SpawnBackground -> True
  345       rec = SubagentRunRecord
  346         { rrrRunId                    = runId
  347         , rrrSubagentId               = subagentId
  348         , rrrChildSessionKey          = childSid
  349         , rrrParentSessionKey         = parentSid
  350         , rrrControllerSessionKey     = parentSid
  351         , rrrStartedAt                = now
  352         , rrrEndedAt                  = Nothing
  353         , rrrOutcome                  = OutcomeUnknown
  354         , rrrEndedReason              = Nothing
  355         , rrrExpectsCompletionMessage = expectsMsg
  356         , rrrFrozenResultText         = Nothing
  357         , rrrAnnounceRetryCount       = 0
  358         , rrrLastAnnounceRetryAt      = Nothing
  359         , rrrSpawnMode                = mode
  360         , rrrCleanup                  = CleanupDelete
  361         , rrrDepth                    = depth
  362         , rrrWakeOnDescendantSettle   = False
  363         , rrrCleanupHandled           = False
  364         , rrrCleanupCompletedAt       = Nothing
  365         , rrrEndedHookEmittedAt       = Nothing
  366         , rrrSuppressAnnounceReason   = Nothing
  367         , rrrGenerationToken          = Just token
  368         }
  369   atomically $ modifyTVar' (rrrRegistry reg) (Map.insert runId rec)
  370   pure rec

  372 -- | Mark a run as completed. Checks the generation token — if the token
  373 -- doesn't match (or is 'Nothing' when the record has one), the call is
  374 -- rejected (returns 'Nothing'). This prevents stale completions from
  375 -- corrupting re-spawned jobs. Idempotent: completing an already-completed
  376 -- run returns 'Nothing'. Emits the @subagent_ended@ lifecycle hook exactly
  377 -- once (guarded by 'rrrEndedHookEmittedAt').
  378 completeRun
  379   :: RunRecordRegistry
  380   -> RunId
  381   -> Maybe GenerationToken
  382   -> ChildResult
  383   -> UTCTime
  384   -> IO (Maybe SubagentRunRecord)
  385 completeRun reg runId mToken result now = do
  386   mRec <- atomically $ do
  387     records <- readTVar (rrrRegistry reg)
  388     case Map.lookup runId records of
  389       Nothing -> pure Nothing
  390       Just rec
  391         | isJust (rrrEndedAt rec) -> pure Nothing
  392             -- Already completed — idempotent no-op.
  393         | tokenMismatch (rrrGenerationToken rec) mToken -> pure Nothing
  394             -- Stale token — reject.
  395         | otherwise -> do
  396             let (outcome, endReason) = outcomeForResult result
  397                 rec' = rec
  398                   { rrrEndedAt = Just now
  399                   , rrrOutcome = outcome
  400                   , rrrEndedReason = Just endReason
  401                   , rrrFrozenResultText = frozenText result
  402                   , rrrEndedHookEmittedAt = Just now
  403                   }
  404             writeTVar (rrrRegistry reg) (Map.insert runId rec' records)
  405             fillCompletionSignal reg rec'  -- WU-7 Path A
  406             pure (Just rec')
  407   -- Emit lifecycle hooks OUTSIDE the STM transaction (callbacks are IO).
  408   forM_ mRec $ \rec -> do
  409     hooks <- readIORef (rrrHooks reg)
  410     forM_ hooks $ \hook -> hook rec
  411   pure mRec

  413 -- | List all runs for a given parent session.
  414 listRunsForParent :: RunRecordRegistry -> SessionId -> IO [SubagentRunRecord]
  415 listRunsForParent reg parentSid = do
  416   records <- readTVarIO (rrrRegistry reg)
  417   pure (filter (\r -> rrrParentSessionKey r == parentSid)
  418                (Map.elems records))
  419 
  420 -- | Find the latest run for a given child session (most recently created).
  421 findLatestRunForChild
  422   :: RunRecordRegistry -> SessionId -> IO (Maybe SubagentRunRecord)
  423 findLatestRunForChild reg childSid = do
  424   records <- readTVarIO (rrrRegistry reg)
  425   let matching = filter (\r -> rrrChildSessionKey r == childSid)
  426                          (Map.elems records)
  427   pure (case matching of
  428           [] -> Nothing
  429           xs -> Just (latestByStartedAt xs))
  430   where
  431     latestByStartedAt = foldr1 (\a b ->
  432       if rrrStartedAt a >= rrrStartedAt b then a else b)
  433 
  434 -- | Count non-terminal (pending) runs for a given session as parent.
  435 countPendingDescendants :: RunRecordRegistry -> SessionId -> IO Int
  436 countPendingDescendants reg parentSid = do
  437   records <- readTVarIO (rrrRegistry reg)
  438   pure (length (filter (\r -> rrrParentSessionKey r == parentSid
  439                           && isNothing (rrrEndedAt r))
  440                        (Map.elems records)))
  441 
  442 -- | Check whether a session has any pending (non-terminal) descendants.
  443 -- Returns 'True' when there are pending runs whose parent is this session.
  444 descendantSettled :: RunRecordRegistry -> SessionId -> IO Bool
  445 descendantSettled reg sid = do
  446   n <- countPendingDescendants reg sid
  447   pure (n == 0)
  448 
  449 -- | Check whether a child's completion delivery should be deferred until
  450 -- its descendants settle. Returns 'True' when the child session still has
  451 -- pending descendants (the child's results would be partial). Once all
  452 -- descendants reach a terminal state, this returns 'False' and the child's
  453 -- completion can be delivered upward.
  454 shouldDeferDelivery :: RunRecordRegistry -> SessionId -> IO Bool
  455 shouldDeferDelivery reg childSid = do
  456   settled <- descendantSettled reg childSid
  457   pure (not settled)

  460 -- Announce retry / backoff (WU-6)
  461 ----------------------------------------------------------------------------
  462 
  463 -- | The maximum number of announce retry attempts before the completion
  464 -- is stored for delivery on parent reactivation.
  465 maxAnnounceRetries :: Int
  466 maxAnnounceRetries = 3
  467 
  468 -- | The base backoff delay (seconds) for the first retry. Subsequent
  469 -- retries double the delay (exponential backoff: 1s, 2s, 4s).
  470 announceBackoffBase :: Double
  471 announceBackoffBase = 1.0
  472 
  473 -- | Compute the exponential backoff delay (seconds) for a given attempt
  474 -- number. Attempt 0 = base, attempt 1 = base * 2, attempt 2 = base * 4,
  475 -- etc. Capped to avoid overflow on very large attempt counts.
  476 computeAnnounceBackoff :: Int -> Double
  477 computeAnnounceBackoff attempt =
  478   announceBackoffBase * (2 ^ safeAttempt)
  479   where
  480     safeAttempt = min attempt 30  -- cap to prevent overflow
  481 
  482 -- | Check whether the announce retries are exhausted (the retry count has
  483 -- reached or exceeded 'maxAnnounceRetries').
  484 announceRetriesExhausted :: Int -> Bool
  485 announceRetriesExhausted n = n >= maxAnnounceRetries
  486 
  487 -- | Record an announce attempt: increment the retry counter and record
  488 -- the timestamp. Returns the updated record, or 'Nothing' if the run id
  489 -- is not found. This is used by the delivery retry loop to track how many
  490 -- attempts have been made and when the last one occurred.
  491 recordAnnounceAttempt
  492   :: RunRecordRegistry
  493   -> RunId
  494   -> UTCTime
  495   -> IO (Maybe SubagentRunRecord)
  496 recordAnnounceAttempt reg runId now = do
  497   atomically $ do
  498     records <- readTVar (rrrRegistry reg)
  499     case Map.lookup runId records of
  500       Nothing -> pure Nothing
  501       Just rec -> do
  502         let rec' = rec
  503               { rrrAnnounceRetryCount = rrrAnnounceRetryCount rec + 1
  504               , rrrLastAnnounceRetryAt = Just now
  505               }
  506         writeTVar (rrrRegistry reg) (Map.insert runId rec' records)
  507         pure (Just rec')

  509 ----------------------------------------------------------------------------
  510 -- Completion signal (WU-7 dual-path detection)
  511 ----------------------------------------------------------------------------
  512 
  513 -- | A completion signal (WU-7 Path A): a @TMVar@ that is filled exactly
  514 -- once when a run reaches a terminal state. A watcher thread can block on
  515 -- 'awaitCompletion' to detect completion without polling.
  516 newtype CompletionSignal = CompletionSignal (TMVar SubagentRunRecord)
  517 
  518 instance Show CompletionSignal where
  519   show _ = "CompletionSignal"
  520 
  521 -- | Register a completion signal for a run. Returns 'Just' the signal if
  522 -- the run exists, or 'Nothing' if the run id is not found. The signal is
  523 -- filled by 'completeRun' or 'cancelRunsForParent' when the run reaches a
  524 -- terminal state. Registering twice for the same run returns the same
  525 -- 'TMVar' (idempotent).
  526 registerCompletionSignal
  527   :: RunRecordRegistry
  528   -> RunId
  529   -> IO (Maybe CompletionSignal)
  530 registerCompletionSignal reg runId = do
  531   atomically $ do
  532     records <- readTVar (rrrRegistry reg)
  533     if Map.member runId records
  534       then do
  535         signals <- readTVar (rrrSignals reg)
  536         case Map.lookup runId signals of
  537           Just tmv -> pure (Just (CompletionSignal tmv))
  538           Nothing -> do
  539             tmv <- newEmptyTMVar
  540             writeTVar (rrrSignals reg) (Map.insert runId tmv signals)
  541             pure (Just (CompletionSignal tmv))
  542       else pure Nothing
  543 
  544 -- | Block until the completion signal is filled, or the timeout (micro-
  545 -- seconds) elapses. Returns 'Just' the terminal record on completion, or
  546 -- 'Nothing' on timeout. Uses 'registerDelay' + 'readTMVar' racing for a
  547 -- pure STM timeout (no forked threads needed).
  548 awaitCompletion :: CompletionSignal -> Int -> IO (Maybe SubagentRunRecord)
  549 awaitCompletion (CompletionSignal tmv) timeoutMicros = do
  550   timer <- registerDelay timeoutMicros
  551   atomically $ do
  552     mRec <- tryReadTMVar tmv
  553     case mRec of
  554       Just rec -> pure (Just rec)
  555       Nothing -> do
  556         expired <- readTVar timer
  557         if expired
  558           then pure Nothing
  559           else retry
  560 
  561 -- | Fill the completion signal for a run (called by 'completeRun' and
  562 -- 'cancelRunsForParent'). If no signal is registered, this is a no-op.
  563 -- If the signal is already filled, this is a no-op (idempotent — the
  564 -- first completion wins).
  565 fillCompletionSignal :: RunRecordRegistry -> SubagentRunRecord -> STM ()
  566 fillCompletionSignal reg rec = do
  567   signals <- readTVar (rrrSignals reg)
  568   case Map.lookup (rrrRunId rec) signals of
  569     Nothing -> pure ()  -- no signal registered
  570     Just tmv -> do
  571       mExisting <- tryReadTMVar tmv
  572       case mExisting of
  573         Just _  -> pure ()  -- already filled — first wins
  574         Nothing -> putTMVar tmv rec
  575 
  576 -- | Cancel all pending runs for a parent session, recursively walking the
  577 -- parent→child linkage so grandchildren (and deeper descendants) are also
  578 -- cancelled. Marks each cancelled run as killed and sets the suppress
  579 -- reason. Returns all cancelled records (the full subtree). Already-
  580 -- completed runs are not affected and are not traversed through — only
  581 -- pending runs whose child session may itself be a parent of further
  582 -- pending runs are followed.
  583 cancelRunsForParent
  584   :: RunRecordRegistry
  585   -> SessionId
  586   -> UTCTime
  587   -> Text
  588      -- ^ suppress reason (e.g. \"killed\")
  589   -> IO [SubagentRunRecord]
  590 cancelRunsForParent reg parentSid now reason = do
  591   cancelled <- atomically $ do
  592     records <- readTVar (rrrRegistry reg)
  593     -- BFS/DFS over the parent→child linkage. Start with runs whose parent
  594     -- is the cancelled session; for each cancelled child, follow its child
  595     -- session key to find runs where THAT session is the parent.
  596     let allRecs = Map.elems records
  597         -- Index pending runs by their parent session key for O(1) lookup.
  598         pendingByParent :: Map.Map SessionId [SubagentRunRecord]
  599         pendingByParent = Map.fromListWith (++)
  600           [ (rrrParentSessionKey r, [r])
  601           | r <- allRecs, isNothing (rrrEndedAt r)
  602           ]
  603         -- Collect the full subtree of pending runs reachable from parentSid.
  604         collect :: [SessionId] -> [SubagentRunRecord] -> [SubagentRunRecord]
  605         collect [] acc = acc
  606         collect (sid : rest) acc =
  607           case Map.lookup sid pendingByParent of
  608             Nothing -> collect rest acc
  609             Just children ->
  610               let newAcc = acc ++ children
  611                   -- Each cancelled child's session becomes a new parent to
  612                   -- explore (its child_session_key is where grandchildren
  613                   -- attach as parent_session_key).
  614                   newFront = map rrrChildSessionKey children ++ rest
  615               in collect newFront newAcc
  616         pending = collect [parentSid] []
  617         updated = map (\r -> r { rrrEndedAt = Just now
  618                                , rrrOutcome = OutcomeKilled
  619                                , rrrEndedReason = Just EndKilled
  620                                , rrrSuppressAnnounceReason = Just reason
  621                                , rrrEndedHookEmittedAt = Just now
  622                                })
  623                       pending
  624         newMap = foldr (\r m -> Map.insert (rrrRunId r) r m) records updated
  625     writeTVar (rrrRegistry reg) newMap
  626     -- Fill completion signals for all cancelled runs (WU-7 Path A)
  627     forM_ updated (fillCompletionSignal reg)
  628     pure updated
  629   -- Emit hooks for cancelled runs
  630   hooks <- readIORef (rrrHooks reg)
  631   forM_ cancelled $ \rec -> forM_ hooks $ \hook -> hook rec
  632   pure cancelled
  633

  638 -- | Register a callback to be invoked when a run reaches a terminal state
  639 -- (completion or cancellation). The callback receives the final
  640 -- 'SubagentRunRecord'. Callbacks are called outside STM transactions.
  641 registerEndedHook
  642   :: RunRecordRegistry -> (SubagentRunRecord -> IO ()) -> IO ()
  643 registerEndedHook reg hook =
  644   atomicModifyIORef' (rrrHooks reg) (\hooks -> (hooks ++ [hook], ()))

  650 -- | The file path for a run record on disk.
  651 runRecordPath :: FilePath -> RunId -> FilePath
  652 runRecordPath sessionDirPath runId =
  653   sessionDirPath </> "agents" </> T.unpack runId </> "run-record.json"
  654 
  655 -- | Save a run record to disk as JSON (atomic write). Creates the parent
  656 -- directory if needed. Best-effort: IO errors are swallowed.
  657 saveRunRecordToDisk :: FilePath -> SubagentRunRecord -> IO ()
  658 saveRunRecordToDisk sessionDirPath rec = do
  659   let path = runRecordPath sessionDirPath (rrrRunId rec)
  660       tmp  = path <.> "tmp"
  661   eWrite <- try @IOException $ do
  662     createDirectoryIfMissing True (takeDirectory path)
  663     BS.writeFile tmp (BSL.toStrict (encode rec))
  664     renameFile tmp path
  665   case eWrite of
  666     Left _err -> pure ()  -- best-effort
  667     Right _   -> pure ()
  668 
  669 -- | Load a run record from a JSON file. Returns 'Nothing' if the file
  670 -- doesn't exist or can't be parsed.
  671 loadRunRecord :: FilePath -> RunId -> IO (Maybe SubagentRunRecord)
  672 loadRunRecord sessionDirPath runId = do
  673   let path = runRecordPath sessionDirPath runId
  674   exists <- doesFileExist path
  675   if not exists
  676     then pure Nothing
  677     else do
  678       eRaw <- try @IOException (BS.readFile path)
  679       case eRaw of
  680         Left _ -> pure Nothing
  681         Right raw -> case eitherDecodeStrict' raw of
  682           Left _   -> pure Nothing
  683           Right rec -> pure (Just rec)

  689 -- | Check whether a provided token matches the record's token.
  690 tokenMismatch :: Maybe GenerationToken -> Maybe GenerationToken -> Bool
  691 tokenMismatch (Just expected) (Just provided) = expected /= provided
  692 tokenMismatch (Just _)        Nothing         = True
  693 tokenMismatch Nothing         _               = False

  695 -- | Map a 'ChildResult' to its 'RunOutcome' and 'EndReason'.
  696 outcomeForResult :: ChildResult -> (RunOutcome, EndReason)
  697 outcomeForResult result =
  698   case crStatus result of
  699     CsCompleted   -> (OutcomeOk, EndComplete)
  700     CsFailed      -> (OutcomeOk, EndComplete)
  701     CsTimeout     -> (OutcomeTimeout, EndTimeout)
  702     CsInterrupted -> (OutcomeKilled, EndKilled)
  703     CsError       -> (OutcomeError, EndError)
  704 
  705 -- | Extract the frozen result text from a 'ChildResult'. Prefers the
  706 -- summary; falls back to the error message.
  707 frozenText :: ChildResult -> Maybe Text
  708 frozenText result =
  709   case crSummary result of
  710     Just s  -> Just s
  711     Nothing -> crError result

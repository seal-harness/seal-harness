    2 -- | Shared Markdown frontmatter codec for the file-backed evolutionary stores
    3 -- (skills, agent defs, memory). Format:
    4 --
    5 -- @
    6 -- ---
    7 -- id: greet
    8 -- description: greeting skill
    9 -- tags: ["a", "b"]
   10 -- updated_at: 2026-07-05T13:39:00Z
   11 -- ---
   12 --
   13 -- body line one
   14 -- @
   15 --
   16 -- The frontmatter is flat @key: value@ lines (one per line). Values are
   17 -- either bare scalars (rendered as-is) or JSON literals (for lists / nulls /
   18 -- bools), parsed permissively. This is a hand-rolled minimal codec — not full
   19 -- YAML — to avoid adding a YAML dependency. The body is everything after the
   20 -- closing @---@ fence.
   21 module Seal.Store.Markdown

   23   , parseFrontmatter
   24   , renderFrontmatter
   25   , fmLookup
   26   , fmLookupList

   32   ) where

   43 -- | A flat frontmatter map (key -> raw value text, as written in the file).
   44 type Frontmatter = Map Text Text
   45 
   46 -- | Split a document into (frontmatter, body). The frontmatter is the text
   47 -- between the opening and closing @---@ fences (exclusive); the body is
   48 -- everything after the closing fence, with one leading blank line stripped.
   49 -- A document with no frontmatter returns @('Map.empty', body)@.
   50 splitFrontmatter :: Text -> (Frontmatter, Text)
   51 splitFrontmatter content =
   52   case T.stripPrefix "---" (T.dropWhile (== '\r') content) of
   53     Nothing -> (Map.empty, content)
   54     Just afterOpen ->
   55       case T.uncons afterOpen of
   56         Just ('\n', rest) -> goBody rest
   57         Just ('\r', rest) -> goBody (T.drop 1 rest)
   58         _                 -> (Map.empty, content)
   59   where
   60     goBody rest =
   61       case T.breakOn "\n---" rest of
   62         (fmRaw, afterBreak) ->
   63           let fm = parseFrontmatter fmRaw
   64               body = case T.stripPrefix "\n---" afterBreak of
   65                 Nothing   -> afterBreak  -- malformed; keep as-is
   66                 Just after -> T.dropWhile (== '\n') after
   67           in (fm, body)
   68 
   69 -- | Split a frontmatter fence off the front of a document, returning the
   70 -- **raw** inner block (unparsed) and the body after the closing fence.
   71 -- Mirrors PureClaw's @extractFrontmatter@: recognizes a leading @"---\\n"@
   72 -- and a terminating @"\\n---\\n"@. Returns @('Nothing', originalInput)@ when
   73 -- the document does not start with a fence or the fence is not closed. Use
   74 -- this when the frontmatter is a non-YAML-ish dialect (e.g. TOML, as in
   75 -- PureClaw-style @AGENTS.md@) and the caller wants to decode the inner
   76 -- block with a dedicated codec rather than 'parseFrontmatter'.
   77 splitFrontmatterRaw :: Text -> (Maybe Text, Text)
   78 splitFrontmatterRaw input =
   79   case T.stripPrefix "---\n" input of
   80     Nothing -> (Nothing, input)
   81     Just rest ->
   82       case T.breakOn "\n---\n" rest of
   83         (_, "") -> (Nothing, input)  -- no closing fence
   84         (inner, afterBreak) ->
   85           let body = T.drop (T.length ("\n---\n" :: Text)) afterBreak
   86           in (Just inner, body)
   87 
   88 -- | Parse a block of @key: value@ lines into a 'Frontmatter' map. Blank lines
   89 -- and lines starting with @#@ are skipped.
   90 parseFrontmatter :: Text -> Frontmatter
   91 parseFrontmatter block =
   92   Map.fromList
   93     [ (T.strip k, T.strip v)
   94     | line <- T.lines block
   95     , not (T.null (T.strip line))
   96     , not ("#" `T.isPrefixOf` T.strip line)
   97     , let (k, v) = case T.breakOn ": " line of
   98             (kk, vv) | T.null vv -> (kk, "")
   99                      | otherwise -> (kk, T.drop 2 vv)
  100     ]
  101 
  102 -- | Render a 'Frontmatter' map as @---\\nkey: value\\n...\\n---@. Keys are
  103 -- emitted in 'Map.toAscList' order (deterministic for stable git diffs).
  104 renderFrontmatter :: Frontmatter -> Text
  105 renderFrontmatter fm =
  106   T.unlines (["---"] <> [k <> ": " <> v | (k, v) <- Map.toAscList fm] <> ["---"])
  107 
  108 -- | Look up a scalar value by key.
  109 fmLookup :: Text -> Frontmatter -> Maybe Text
  110 fmLookup = Map.lookup
  111 
  112 -- | Look up a JSON-array value by key, returning the list of string elements.
  113 -- A value of @[\"a\", \"b\"]@ yields @["a","b"]@. Non-array values yield
  114 -- 'Nothing'.
  115 fmLookupList :: Text -> Frontmatter -> Maybe [Text]
  116 fmLookupList key fm = case Map.lookup key fm of
  117   Nothing  -> Nothing
  118   Just raw -> case decode (BL.fromStrict (TE.encodeUtf8 raw)) of
  119     Just (Array xs) -> Just [ t | String t <- toList xs ]
  120     _               -> Nothing
  121 
  122 -- | Encode a document: render the frontmatter fence, a blank line, then the
  123 -- body.
  124 encodeDoc :: Frontmatter -> Text -> Text
  125 encodeDoc fm body = renderFrontmatter fm <> "\n" <> body
  126 
  127 -- | Decode a document: split into frontmatter + body.
  128 decodeDoc :: Text -> (Frontmatter, Text)
  129 decodeDoc = splitFrontmatter
  130

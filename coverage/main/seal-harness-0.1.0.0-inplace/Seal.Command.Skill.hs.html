    2 -- | The @/skill@ command group: list defined skills, show one skill's
    3 -- full body, and load a skill into the current session. Skills are stored
    4 -- as structured records materialized from the Audited log into the
    5 -- in-memory 'SkillBackend'; @/skill list@ and @/skill info@ read that
    6 -- backend directly (no filesystem discovery, no audit-trail entry).
    7 -- @/skill load@ dispatches the 'SKILL_LOAD' opcode via the channel-supplied
    8 -- 'CallDispatcher' (the same closure @/call@ uses), which records an
    9 -- 'EKHarness' entry to the session transcript with
   10 -- @erMeta.op.name = "SKILL_LOAD"@ and @erMeta.input.id = <id>@ — the
   11 -- audit-trail attribution that distinguishes a skill load from a user
   12 -- pasting the body.

   14   ( skillCommandSpec
   15   , renderSkillLine
   16   , renderSkillInfo
   17   , skillLoadFollowUpText
   18   ) where

   39 -- | The synthetic user message used as the follow-up turn text when
   40 -- @/skill load@ is invoked WITHOUT a trailing message (e.g.
   41 -- @/skill load greet@). The skill body has already been written to
   42 -- @conversation.jsonl@ as an 'Assistant' message by
   43 -- 'Seal.ISA.Dispatch.recordSkillLoadResult'; this text becomes the 'User'
   44 -- message that triggers the LLM turn, so the model sees the skill
   45 -- immediately followed by a prompt to act on it. When the user supplies a
   46 -- trailing message (e.g. @/skill load start #123@), THAT text is used
   47 -- instead — the user's intent takes precedence over the synthetic prompt.
   48 skillLoadFollowUpText :: Text
   49 skillLoadFollowUpText =
   50   "Skill loaded. Follow the instructions in the skill above."

   52 -- | The @/skill@ command spec. Closes over the 'SkillBackend' (for
   53 -- @/skill list@ \/ @/skill info@, which read the materialized view of the
   54 -- Audited log directly) and the channel-supplied 'CallDispatcher' (for
   55 -- @/skill load@, which dispatches the 'SKILL_LOAD' opcode against the
   56 -- active session's ISA registry + transcript — the same dispatcher
   57 -- @/call@ uses, so the audit trail records both paths uniformly).
   58 skillCommandSpec :: SkillBackend -> CallDispatcher -> CommandSpec
   59 skillCommandSpec backend dispatcher = CommandSpec
   60   { csName         = CommandName "skill"
   61   , csAliases      = []
   62   , csGroup        = GroupSkills
   63   , csSynopsis     = "List defined skills, show one skill's body, or load one into the session"
   64   , csParserInfo   = skillParserInfo backend dispatcher
   65   , csAvailability = InteractiveOnly
   66   }
   67 
   68 skillParserInfo :: SkillBackend -> CallDispatcher -> ParserInfo CommandAction
   69 skillParserInfo backend dispatcher =
   70   info (skillParser backend dispatcher <**> helper)
   71     (  progDesc "Inspect or load agent skills"
   72     <> header   "skill — list, show, or load a skill"
   73     )
   74 
   75 skillParser :: SkillBackend -> CallDispatcher -> Parser CommandAction
   76 skillParser backend dispatcher = hsubparser
   77   (  command "list"
   78        (info (pure (listCmd backend)) (progDesc "List all defined skills (id + description)"))
   79   <> command "info"
   80        (info (infoCmd backend <$> skillArg)
   81              (progDesc "Show one skill's full body"))
   82   <> command "load"
   83        (info (loadCmd dispatcher <$> skillArg <*> messageArg)
   84              (progDesc "Load one skill into the current session, with an optional trailing message"))
   85   <> metavar "COMMAND"
   86   )
   87 
   88 -- | Required skill-id argument.
   89 skillArg :: Parser Text
   90 skillArg = T.pack <$> strArgument (metavar "SKILL" <> help "Skill id (e.g. greet)")
   91 
   92 -- | Optional trailing message (everything after the skill id). Joined with
   93 -- spaces so @/skill load foo do something@ yields the message @do something@.
   94 -- Empty when no trailing text is supplied (just a skill load, no message).
   95 messageArg :: Parser Text
   96 messageArg = T.intercalate " " . map T.pack <$> many (strArgument (metavar "MESSAGE..." <> help "Optional message appended after the skill body"))

   98 listCmd :: SkillBackend -> CommandAction
   99 listCmd backend = CommandAction $ \caps -> do
  100   skills <- sbList backend
  101   if null skills
  102     then ccSend caps "no skills defined"
  103     else mapM_ (ccSend caps . renderSkillLine) (sortSkills skills)
  104   pure Nothing
  105 
  106 -- | Sort skills alphabetically by group name then by skill id, for
  107 -- deterministic @/skill list@ output.
  108 sortSkills :: [Skill] -> [Skill]
  109 sortSkills = sortBy (comparing (\s -> (fromMaybe "" (skGroup s), skillIdText (skId s))))
  110 
  111 infoCmd :: SkillBackend -> Text -> CommandAction
  112 infoCmd backend raw = CommandAction $ \caps ->
  113   case mkSkillId raw of
  114     Left err -> ccSend caps err >> pure Nothing
  115     Right sid -> do
  116       mSkill <- sbRead backend sid
  117       case mSkill of
  118         Nothing -> ccSend caps ("skill not found: " <> skillIdText sid) >> pure Nothing
  119         Just s  -> mapM_ (ccSend caps) (renderSkillInfo s) >> pure Nothing

  121 -- | @/skill load <id> [message...]@ — dispatch the 'SKILL_LOAD' opcode with
  122 -- @{"id": <id>, "message": <message>}@ via the channel-supplied
  123 -- 'CallDispatcher'. Mirrors @/call@'s pattern: echo a header line first (so
  124 -- the "Command output" bubble is self-contained).
  125 --
  126 -- The optional trailing @message@ is forwarded to the dispatcher as a
  127 -- @message@ field in the opcode input. 'recordSkillLoadResult' writes the
  128 -- skill body to @conversation.jsonl@ as an 'Assistant' message. This
  129 -- command returns a follow-up turn text ('Just' text) so the channel
  130 -- dispatch site forks an LLM turn immediately — the model sees the skill
  131 -- body (from the transcript) followed by the turn text (as the 'User'
  132 -- message 'runTurn' writes). When a trailing message was supplied, it
  133 -- becomes the turn text (the user's intent); when no message was supplied,
  134 -- a synthetic prompt ('skillLoadFollowUpText') is used instead.
  135 --
  136 -- On success, the skill body is NOT rendered via 'ccSend' — the dispatcher
  137 -- records a second 'EKHarness' entry to the transcript carrying the
  138 -- @orRecorded@ value (which includes the body), and the frontend renders
  139 -- that entry as a collapsible tool-call box. This keeps the slash bubble
  140 -- to just the echo line and avoids duplicating the body in the "command
  141 -- output — not saved" transient bubble.
  142 --
  143 -- On error (dispatch 'Left' or 'Right' with the error flag set), the error
  144 -- text IS rendered via 'ccSend' so the user sees it in the slash bubble —
  145 -- error paths produce no transcript body entry, so the slash bubble is
  146 -- the only surface for the error message. Error paths return 'Nothing'
  147 -- (no follow-up turn).
  148 loadCmd :: CallDispatcher -> Text -> Text -> CommandAction
  149 loadCmd dispatcher raw message = CommandAction $ \caps -> do
  150   ccSend caps ("$ /skill load " <> raw)
  151   case mkSkillId raw of
  152     Left err -> ccSend caps err >> pure Nothing
  153     Right sid -> do
  154       let input = object
  155             [ "id" .= skillIdText sid
  156             , "message" .= message
  157             ]
  158       res <- dispatcher (OpName "SKILL_LOAD") input
  159       case res of
  160         Left e  -> do
  161           ccSend caps (renderDispatchError e)
  162           pure Nothing
  163         Right r
  164           | orIsError r -> do
  165               mapM_ (ccSend caps) (renderOpResult r)
  166               pure Nothing
  167           | otherwise   ->
  168               -- Success: signal a follow-up turn. Use the user's trailing
  169               -- message when non-empty; otherwise the synthetic prompt.
  170               let turnText = if T.null (T.strip message)
  171                                then skillLoadFollowUpText
  172                                else message
  173               in pure (Just turnText)
  174 
  175 -- | One line per skill for @/skill list@.
  176 renderSkillLine :: Skill -> Text
  177 renderSkillLine s =
  178   "[" <> skillIdText (skId s) <> "] " <> skDescription s
  179 
  180 -- | Multi-line detail for @/skill info@.
  181 renderSkillInfo :: Skill -> [Text]
  182 renderSkillInfo s =
  183   [ "id:          " <> skillIdText (skId s)
  184   , "description: " <> skDescription s
  185   , "updated:     " <> T.pack (show (skUpdatedAt s))
  186   , "session:     " <> T.pack (show (skSession s))
  187   , ""
  188   , skBody s

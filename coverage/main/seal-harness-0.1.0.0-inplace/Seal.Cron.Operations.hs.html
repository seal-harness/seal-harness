   26 -- | The partial-update record for 'updateJob'. 'Nothing' means "leave
   27 -- this field unchanged". For 'cuTimezone' and 'cuAgentDefId', 'Nothing'
   28 -- means unchanged (NOT "clear the field").
   29 data CronUpdate = CronUpdate
   30   { cuName       :: Maybe Text
   31   , cuSchedule   :: Maybe Text
   32   , cuTimezone   :: Maybe Text
   33   , cuPrompt     :: Maybe Text
   34   , cuAgentDefId :: Maybe Text
   35   , cuProvider   :: Maybe Text
   36     -- ^ 'Nothing' = leave unchanged; 'Just ""' = clear (fall back to the
   37     -- configured default); 'Just name' = set. Validated with 'isProviderName'
   38     -- exactly as on create, so an update cannot smuggle in a value that
   39     -- 'createJob' would reject.
   40   , cuModel      :: Maybe Text
   41     -- ^ Same three-way convention as 'cuProvider'.
   42   , cuEnabled    :: Maybe Bool
   43   , cuOneShot    :: Maybe Bool
   44   }
   45 
   46 -- | Create a new cron job. Slugifies the name, dedups against existing
   47 -- ids, validates all fields, and defaults @enabled@ to 'False'.
   48 -- Returns 'Left' with a human-readable error on validation failure.
   49 createJob
   50   :: CronStore
   51   -> Text          -- ^ name (1–100 chars)
   52   -> Text          -- ^ schedule (5-field cron expression)
   53   -> Maybe Text    -- ^ timezone (IANA identifier, validated against tzdb)
   54   -> Text          -- ^ prompt (1–4000 chars)
   55   -> Maybe Text    -- ^ agent_def_id (validated with 'mkAgentDefId')
   56   -> Maybe Text    -- ^ provider (validated with 'isProviderName'; @""@ = absent)
   57   -> Maybe Text    -- ^ model (@""@ = absent)
   58   -> Bool          -- ^ enabled
   59   -> Bool          -- ^ one_shot
   60   -> IO (Either Text CronJob)
   61 createJob store name schedStr mTzStr prompt mAgentDefId mProvider mModel enabled oneShot
   62   | T.null name || T.length name > 100 =
   63       pure (Left "name must be 1-100 characters")
   64   | T.null prompt || T.length prompt > 4000 =
   65       pure (Left "prompt must be 1-4000 characters")
   66   -- A '/'-leading prompt is routed by the gateway to the slash-command path,
   67   -- which never starts a plain turn — so a scheduled run could never report
   68   -- completion and would look permanently in-flight. Reject it at the write
   69   -- boundary, where the operator gets an immediate, actionable error, instead
   70   -- of failing invisibly five times at fire time.
   71   | "/" `T.isPrefixOf` T.stripStart prompt =
   72       pure (Left "prompt must not start with '/'")
   73   | otherwise =
   74       case mkCronExpr schedStr of
   75         Left e -> pure (Left e)
   76         Right expr -> do
   77           mValidTz <- validateTimezone mTzStr
   78           case mValidTz of
   79             Left e -> pure (Left e)
   80             Right mTz ->
   81               case validateAgentDefId mAgentDefId of
   82                 Left e -> pure (Left e)
   83                 Right mValidAid ->
   84                   case validateProvider mProvider of
   85                     Left e -> pure (Left e)
   86                     Right mValidProv -> do
   87                       allJobs <- csList store False
   88                       if length allJobs >= 100
   89                         then pure (Left "maximum job count (100) reached")
   90                         else do
   91                           let baseSlug = slugify name
   92                           existingIds <- map (cronJobIdText . cjId) <$> csList store False
   93                           let slug = dedupSlug baseSlug existingIds
   94                           case mkCronJobId slug of
   95                             Left e -> pure (Left e)
   96                             Right jid -> do
   97                               now <- getCurrentTime
   98                               let job = CronJob
   99                                     { cjId = jid, cjName = name
  100                                     , cjSchedule = expr, cjTimezone = mTz
  101                                     , cjPrompt = prompt
  102                                     , cjAgentDefId = mValidAid
  103                                     , cjProvider = mValidProv
  104                                     , cjModel = validateModel mModel
  105                                     , cjEnabled = enabled
  106                                     , cjOneShot = oneShot
  107                                     , cjCreatedAt = now, cjUpdatedAt = now
  108                                     , cjLastFire = Nothing
  109                                     , cjRunCount = 0
  110                                     , cjConsecutiveFailures = 0 }
  111                               csCreate store job
  112                               pure (Right job)
  113 
  114 -- | Update an existing cron job, patching only the supplied fields.
  115 -- Applies the same validation as 'createJob' to each supplied field.
  116 -- Returns 'Left' if the job doesn't exist or a field is invalid.
  117 updateJob
  118   :: CronStore
  119   -> CronJobId
  120   -> CronUpdate
  121   -> IO (Either Text CronJob)
  122 updateJob store jid upd = do
  123   mJob <- csRead store jid
  124   case mJob of
  125     Nothing -> pure (Left ("cron job not found: " <> cronJobIdText jid))
  126     Just job -> updateJobValidated store job upd
  127 
  128 -- | Internal: validate all supplied fields and persist the update.
  129 updateJobValidated :: CronStore -> CronJob -> CronUpdate -> IO (Either Text CronJob)
  130 updateJobValidated store job upd
  131   -- Validate name
  132   | Just name <- cuName upd
  133   , T.null name || T.length name > 100 =
  134       pure (Left "name must be 1-100 characters")
  135   -- Validate prompt
  136   | Just prompt <- cuPrompt upd
  137   , T.null prompt || T.length prompt > 4000 =
  138       pure (Left "prompt must be 1-4000 characters")
  139   | Just prompt <- cuPrompt upd
  140   , "/" `T.isPrefixOf` T.stripStart prompt =
  141       pure (Left "prompt must not start with '/'")
  142   | otherwise = do
  143       -- Validate schedule
  144       case cuSchedule upd of
  145         Just schedStr ->
  146           case mkCronExpr schedStr of
  147             Left e -> pure (Left e)
  148             Right _ -> continueAfterSchedule
  149         Nothing -> continueAfterSchedule
  150   where
  151     continueAfterSchedule = do
  152       mValidTz <- validateTimezone (cuTimezone upd)
  153       case mValidTz of
  154         Left e -> pure (Left e)
  155         Right mTz ->
  156           case validateAgentDefId (cuAgentDefId upd) of
  157             Left e -> pure (Left e)
  158             Right mValidAid ->
  159               case validateProvider (cuProvider upd) of
  160                 Left e -> pure (Left e)
  161                 Right mValidProv -> do
  162                   now <- getCurrentTime
  163                   let updatedJob =
  164                         applyUpdate job upd mTz mValidAid mValidProv now
  165                   csSave store updatedJob
  166                   pure (Right updatedJob)

  168 -- | Merge a validated 'CronUpdate' into an existing job. Pure, so the
  169 -- three-way "leave alone / clear / set" rule is readable in one place.
  170 --
  171 -- Provider and model follow the same convention: 'Nothing' means the client
  172 -- did not supply the field (an absent key and an explicit JSON @null@ both
  173 -- decode to 'Nothing'), @Just ""@ means the operator emptied the form field
  174 -- and wants the job back on Seal's configured default, and @Just x@ sets it.
  175 applyUpdate
  176   :: CronJob
  177   -> CronUpdate
  178   -> Maybe CronTimeZone   -- ^ validated timezone ('Nothing' = unchanged)
  179   -> Maybe Text           -- ^ validated agent def id ('Nothing' = unchanged)
  180   -> Maybe Text           -- ^ validated provider ('Nothing' = absent/empty)
  181   -> UTCTime
  182   -> CronJob
  183 applyUpdate job upd mTz mValidAid mValidProv now = job
  184   { cjName     = fromMaybe (cjName job) (cuName upd)
  185   , cjSchedule = case cuSchedule upd of
  186       Nothing -> cjSchedule job
  187       Just sc ->
  188         case mkCronExpr sc of
  189           Right e -> e
  190           Left _  -> cjSchedule job
  191   , cjTimezone = mTz <|> cjTimezone job
  192   , cjPrompt   = fromMaybe (cjPrompt job) (cuPrompt upd)
  193   , cjAgentDefId = mValidAid <|> cjAgentDefId job
  194   , cjProvider = applyTriState (cjProvider job) (cuProvider upd) mValidProv
  195   , cjModel    = applyTriState (cjModel job) (cuModel upd)
  196                    (validateModel (cuModel upd))
  197   , cjEnabled  = fromMaybe (cjEnabled job) (cuEnabled upd)
  198   , cjOneShot  = fromMaybe (cjOneShot job) (cuOneShot upd)
  199   , cjUpdatedAt = now
  200   }

  202 -- | Resolve one tri-state field. @supplied@ is the raw client value, needed to
  203 -- tell "key absent" from "@Just \"\"@" (both would otherwise collapse to
  204 -- 'Nothing'); @validated@ is the smart-constructed value produced by the
  205 -- validator that already ran on this same field — it is 'Nothing' when nothing
  206 -- usable was supplied.
  207 applyTriState :: Maybe Text -> Maybe Text -> Maybe Text -> Maybe Text
  208 applyTriState current supplied validated =
  209   case supplied of
  210     Nothing -> current
  211     Just raw
  212       | T.null raw -> Nothing
  213       | otherwise  -> validated

  215 -- | Enable or disable a job. Idempotent — setting enabled to its current
  216 -- value is a no-op that still returns 'Right'. Returns 'Left' if the job
  217 -- doesn't exist.
  218 setJobEnabled :: CronStore -> CronJobId -> Bool -> IO (Either Text CronJob)
  219 setJobEnabled store jid newEnabled = do
  220   mJob <- csRead store jid
  221   case mJob of
  222     Nothing -> pure (Left ("cron job not found: " <> cronJobIdText jid))
  223     Just job -> do
  224       now <- getCurrentTime
  225       let updatedJob = job { cjEnabled = newEnabled, cjUpdatedAt = now }
  226       csSave store updatedJob
  227       pure (Right updatedJob)

  229 -- | Delete a job. Idempotent — deleting a non-existent job returns
  230 -- 'Right ()'. Never fails.
  231 deleteJob :: CronStore -> CronJobId -> IO (Either Text ())
  232 deleteJob store jid = do
  233   csDelete store jid
  234   pure (Right ())

  236 -- | Ensure a slug does not collide with existing ids by appending a
  237 -- numeric suffix. Owned here so both the opcode and the web path dedup
  238 -- identically.
  239 dedupSlug :: Text -> [Text] -> Text
  240 dedupSlug base existingIds
  241   | base `notElem` existingIds = base
  242   | otherwise = go (2 :: Int)
  243   where
  244     go :: Int -> Text
  245     go n = let c = base <> "-" <> T.pack (show n)
  246            in if c `notElem` existingIds then c else go (n + 1)

  248 -- | Validate a timezone string: format check via 'mkCronTimeZone',
  249 -- then existence check via 'validateCronTimeZone'. 'Nothing' input
  250 -- means "no timezone supplied" and yields 'Right Nothing'.
  251 validateTimezone :: Maybe Text -> IO (Either Text (Maybe CronTimeZone))
  252 validateTimezone Nothing = pure (Right Nothing)
  253 validateTimezone (Just tzStr) = case mkCronTimeZone tzStr of
  254   Left e -> pure (Left e)
  255   Right tz -> do
  256     eValid <- validateCronTimeZone tz
  257     pure $ case eValid of
  258       Left e -> Left e
  259       Right _ -> Right (Just tz)
  260 
  261 -- | Validate an agent_def_id string with 'mkAgentDefId'. 'Nothing' input
  262 -- means "no agent_def_id supplied" and yields 'Right Nothing'.
  263 validateAgentDefId :: Maybe Text -> Either Text (Maybe Text)
  264 validateAgentDefId Nothing = Right Nothing
  265 validateAgentDefId (Just aidStr) = case mkAgentDefId aidStr of
  266   Left e -> Left e
  267   Right _ -> Right (Just aidStr)

  269 -- | Validate an optional provider name. @Nothing@ and @Just ""@ both mean
  270 -- "no provider supplied" and yield 'Right Nothing' — an emptied form field is
  271 -- an absent choice, not a provider named the empty string.
  272 --
  273 -- The check is '<Seal.Providers.Registry.isProviderName>': a built-in provider
  274 -- label, or a syntactically valid custom instance name. A syntactically valid
  275 -- but *unconfigured* name cannot be rejected here — this module is shared by
  276 -- the opcode and the web gateway and holds no config handle. Such a name
  277 -- surfaces as @\"unknown provider in session: …\"@ inside the spawned session's
  278 -- transcript at fire time; see the design note in
  279 -- @docs/plans/2026-11-15-cron-provider-model-plan.md@ (D2).
  280 validateProvider :: Maybe Text -> Either Text (Maybe Text)
  281 validateProvider Nothing = Right Nothing
  282 validateProvider (Just provStr)
  283   | T.null provStr = Right Nothing
  284   | isProviderName provStr = Right (Just provStr)
  285   | otherwise = Left ("invalid provider name: " <> provStr)
  286 
  287 -- | Validate an optional model id. @Nothing@ and @Just ""@ both mean "no model
  288 -- supplied". Model ids are opaque to the harness (every provider names its own),
  289 -- so the only rule is that an emptied field is an absent choice.
  290 validateModel :: Maybe Text -> Maybe Text
  291 validateModel Nothing = Nothing
  292 validateModel (Just m) = if T.null m then Nothing else Just m

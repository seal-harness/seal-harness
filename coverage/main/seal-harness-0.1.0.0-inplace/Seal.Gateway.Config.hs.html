    3 -- origins. The server binds loopback by default; a non-loopback host emits

    7 -- The section is defined here and configured in @~\/.seal\/security.toml@

   44 -- | The gateway config section.
   45 data GatewayConfig = GatewayConfig
   46   { gcPort           :: Int        -- ^ default 8080 (REST + static + the WS stream)
   47   , gcHost           :: Text
   48     -- ^ default "localhost" — the one loopback name browsers treat as a
   49     -- passkey-capable secure context. Handed to Warp as the bind address.
   50   , gcStaticDir      :: Maybe Text  -- ^ the frontend dist dir (Nothing = no static serving)
   51   , gcAllowedOrigins :: [Text]     -- ^ the WS Origin allowlist (default localhost:8080)
   52   , gcFileExplorerEnabled :: Maybe Bool -- ^ enable the file explorer (default True; set False to disable)
   53   , gcTlsCert        :: Maybe Text
   54     -- ^ PEM certificate chain. Set together with 'gcTlsKey' to serve HTTPS
   55     -- directly, with no reverse proxy in front.
   56   , gcTlsKey         :: Maybe Text
   57     -- ^ PEM private key.
   58   , gcTls            :: Bool
   59     -- ^ Serve HTTPS. The PEM paths default to @keys\/tls\/@{cert,key}@.on.pem
   60     -- under @SEAL_HOME@, so this flag alone is the whole configuration.
   61   , gcTlsHosts       :: [Text]
   62     -- ^ Names to put in a generated certificate's subject alternative name.
   63     -- These come first, in order. Everything else is derived: the origin's
   64     -- host, the bind host (if it names a machine), the machine hostname, every
   65     -- local interface address, and loopback — see
   66     -- 'Seal.Security.TlsCert.sanHostsFor'.
   67   } deriving stock (Eq, Show)
   68 
   69 -- | The compiled-in default. Loopback + port 8080 + the local origin.
   70 defaultGatewayConfig :: GatewayConfig
   71 defaultGatewayConfig = GatewayConfig
   72   { gcPort = 8080
   73   , gcHost = "localhost"
   74   , gcStaticDir = Nothing
   75   , gcAllowedOrigins = ["http://localhost:8080"]
   76   , gcFileExplorerEnabled = Nothing
   77   , gcTlsCert = Nothing
   78   , gcTlsKey = Nothing
   79   , gcTls = False
   80   , gcTlsHosts = []
   81   }

   83 -- ---------------------------------------------------------------------------
   84 -- Host validation
   85 -- ---------------------------------------------------------------------------
   86 --
   87 -- @[gateway] host@ is handed straight to Warp (see 'Seal.Gateway.Server.warpSettingsFor'),
   88 -- so a value that cannot bind is now a boot refusal. Before the host was wired
   89 -- to the listener, any string was survivable: it only fed the printed URL, the
   90 -- derived origins, and the exposed-bind warning.
   91 
   92 -- | A @[gateway] host@ value that could never be bound. Purely syntactic.
   93 --
   94 -- 'gatewayHostResolutionError' asks the resolver about the values that pass
   95 -- this check, so the two refusals stay independently testable.
   96 -- Every clause inspects @raw@, never a trimmed copy: the caller hands this
   97 -- exact string to Warp ('Seal.Gateway.Server.warpSettingsFor'), so a value that
   98 -- passes here has to be one the resolver will accept. Trimming first would let
   99 -- @host = " localhost "@ through both gates and fail inside the bind, after the
  100 -- banner, with a message that names neither the key nor the mistake.
  101 gatewayHostError :: Text -> Maybe Text
  102 gatewayHostError raw
  103   | T.null (T.strip raw) = Just (hostRefusal "it is empty or blank" emptyFix)
  104   | T.any isSpace raw    = Just (hostRefusal "it contains whitespace" spacedFix)
  105   | T.any (== '/') raw   = Just (hostRefusal "it looks like a URL or a path" bareFix)
  106   | T.any (`elem` bracketed) raw =
  107       Just (hostRefusal "it is bracketed"
  108              "write an IPv6 literal without brackets, e.g. \"::1\"")
  109   | otherwise = Nothing
  110   where
  111     bracketed = "[]" :: String
  112     emptyFix = "set a hostname (\"localhost\") or a wildcard (\"0.0.0.0\")"
  113     spacedFix = "remove the surrounding or embedded whitespace: \"localhost\""
  114     bareFix = "write the bare host or address, e.g. \"localhost\" or \"0.0.0.0\""
  115 
  116 hostRefusal :: Text -> Text -> Text
  117 hostRefusal why fix =
  118   "[gateway] host is not a bindable address: " <> why <> ".\n  " <> fix

  120 -- | Does @host@ resolve to at least one address?
  121 --
  122 -- "Resolvable", not "local": Warp may legitimately bind any address this
  123 -- machine has, and a tailnet or LAN address is a normal @[gateway] host@ value.
  124 -- A name that resolves to an address the machine does not own still fails at
  125 -- bind time with Warp's own error; this check exists to catch typos and
  126 -- unusable spellings before anything starts listening.
  127 hostResolves :: Text -> IO Bool
  128 hostResolves raw = do
  129   let hints = defaultHints { addrFlags = [AI_PASSIVE], addrSocketType = Stream }
  130   resolved <-
  131     try (getAddrInfo (Just hints) (Just (T.unpack raw)) Nothing)
  132       :: IO (Either IOException [AddrInfo])
  133   pure (either (const False) (not . null) resolved)
  134 
  135 -- | The resolver-backed half of the host check.
  136 --
  137 -- The resolver is a parameter so the refusal branch is testable without DNS;
  138 -- @seal serve@ passes 'hostResolves'.
  139 gatewayHostResolutionError :: (Text -> IO Bool) -> Text -> IO (Maybe Text)
  140 gatewayHostResolutionError resolves raw = do
  141   ok <- resolves raw
  142   pure $ if ok
  143     then Nothing
  144     else Just
  145       ("[gateway] host = \"" <> raw <> "\" does not resolve to any address.\n"
  146         <> "  Fix or remove the line (the default is \"localhost\"), or set\n"
  147         <> "  host = \"0.0.0.0\" to bind every interface.")

  149 -- | Bidirectional tomland codec for the @[gateway]@ section. Every field is
  150 -- optional: a missing key decodes as 'Nothing', then 'withGatewayDefaults'
  151 -- (called by 'Seal.Command.Serve.runServeMain') fills in the
  152 -- 'defaultGatewayConfig' values. This lets a user set just @host@ or just
  153 -- @port@ without specifying the rest.
  154 gatewayConfigCodec :: Toml.TomlCodec PartialGatewayConfig
  155 gatewayConfigCodec = PartialGatewayConfig
  156   <$> Toml.dimap (fmap fromIntegral) (fmap fromIntegral) (Toml.dioptional (Toml.integer "port")) .= pgcPort
  157   <*> Toml.dioptional (Toml.text "host") .= pgcHost
  158   <*> Toml.dioptional (Toml.text "static_dir") .= pgcStaticDir
  159   <*> originsCodec .= pgcAllowedOrigins
  160   <*> Toml.dioptional (Toml.bool "file_explorer_enabled") .= pgcFileExplorerEnabled
  161   <*> Toml.dioptional (Toml.text "tls_cert") .= pgcTlsCert
  162   <*> Toml.dioptional (Toml.text "tls_key") .= pgcTlsKey
  163   <*> Toml.dioptional (Toml.bool "tls") .= pgcTls
  164   <*> hostsCodec .= pgcTlsHosts
  165 
  166 -- | Codec for the @[gateway]@ table itself — the one definition of where the
  167 -- section lives. Three consumers share it: @security.toml@'s document codec
  168 -- ('Seal.Config.Security.securityConfigCodec'), the pre-migration
  169 -- read-through in @seal access-control@ ('loadLegacyGatewayTable'), and the
  170 -- write guard that refuses to re-encode a @config.toml@ still carrying the
  171 -- table ('Seal.Config.File.updateRuntimeConfig').
  172 gatewayTableCodec :: Toml.TomlCodec (Maybe PartialGatewayConfig)
  173 gatewayTableCodec = Toml.dioptional (Toml.table gatewayConfigCodec "gateway")
  174 
  175 -- | Decode the @[gateway]@ table out of a TOML file, ignoring every other key.
  176 -- 'Nothing' means the file is absent, does not parse, or has no table.
  177 --
  178 -- Only for the upgrade window: while an install still has its gateway settings
  179 -- in @config.toml@, @seal access-control@ reads them from there instead of
  180 -- silently resolving the loopback defaults. The first @seal serve@/@seal tui@
  181 -- moves the table into @security.toml@ and this path goes quiet.
  182 loadLegacyGatewayTable :: FilePath -> IO (Maybe PartialGatewayConfig)
  183 loadLegacyGatewayTable path = do
  184   exists <- doesFileExist path
  185   if not exists then pure Nothing else do
  186     outcome <- try (TIO.readFile path) :: IO (Either IOException Text)
  187     pure $ case outcome of
  188       Left _     -> Nothing
  189       Right text -> case Toml.parse text of
  190         Left _     -> Nothing
  191         Right toml -> case Toml.runTomlCodec gatewayTableCodec toml of
  192           Success table -> table
  193           Failure _     -> Nothing

  195 -- | A partial gateway config where every field is optional. 'withGatewayDefaults'
  196 -- merges it with 'defaultGatewayConfig' to produce a complete 'GatewayConfig'.
  197 data PartialGatewayConfig = PartialGatewayConfig
  198   { pgcPort           :: Maybe Int
  199   , pgcHost           :: Maybe Text
  200   , pgcStaticDir      :: Maybe Text
  201   , pgcAllowedOrigins :: [Text]
  202   , pgcFileExplorerEnabled :: Maybe Bool
  203   , pgcTlsCert        :: Maybe Text
  204   , pgcTlsKey         :: Maybe Text
  205   , pgcTls            :: Maybe Bool
  206   , pgcTlsHosts       :: [Text]
  207   } deriving stock (Eq, Show)
  208 
  209 -- | Merge a partial gateway config with the compiled-in defaults, producing
  210 -- a complete 'GatewayConfig'. Each 'Nothing' field falls back to
  211 -- 'defaultGatewayConfig'; an empty 'pgcAllowedOrigins' also falls back (so
  212 -- omitting @allowed_origins@ keeps the safe loopback default rather than
  213 -- opening to all origins).
  214 withGatewayDefaults :: PartialGatewayConfig -> GatewayConfig
  215 withGatewayDefaults p = GatewayConfig
  216   { gcPort           = fromMaybe (gcPort defaultGatewayConfig) (pgcPort p)
  217   , gcHost           = fromMaybe (gcHost defaultGatewayConfig) (pgcHost p)
  218   , gcStaticDir      = pgcStaticDir p  -- Nothing = no static serving (the default)
  219   , gcAllowedOrigins = if null (pgcAllowedOrigins p)
  220                          then gcAllowedOrigins defaultGatewayConfig
  221                          else pgcAllowedOrigins p
  222   , gcFileExplorerEnabled = pgcFileExplorerEnabled p
  223   , gcTlsCert        = pgcTlsCert p
  224   , gcTlsKey         = pgcTlsKey p
  225   , gcTls            = fromMaybe False (pgcTls p)
  226   , gcTlsHosts       = pgcTlsHosts p
  227   }

  229 -- | Codec for the @allowed_origins@ array (a TOML array of strings).
  230 originsCodec :: Toml.TomlCodec [Text]
  231 originsCodec = Toml.dimap Just maybeToList (Toml.dioptional (Toml.arrayOf Toml._Text "allowed_origins"))
  232   where
  233     maybeToList Nothing   = []
  234     maybeToList (Just xs) = xs
  235 
  236 -- | Codec for the @tls_hosts@ array (a TOML array of strings).
  237 hostsCodec :: Toml.TomlCodec [Text]
  238 hostsCodec = Toml.dimap Just maybeToList (Toml.dioptional (Toml.arrayOf Toml._Text "tls_hosts"))
  239   where
  240     maybeToList Nothing   = []
  241     maybeToList (Just xs) = xs

  243 -- | Is the gateway configured to terminate TLS itself?
  244 --
  245 -- Serving HTTPS in-process means no reverse proxy is needed, which matters for
  246 -- passkeys: a WebAuthn ceremony requires a secure context, and a bare
  247 -- @http://@ LAN address is not one.
  248 tlsEnabled :: GatewayConfig -> Bool
  249 tlsEnabled cfg = gcTls cfg || (isJust (gcTlsCert cfg) && isJust (gcTlsKey cfg))
  250 
  251 -- | The PEM paths to load, given the defaults for this @SEAL_HOME@.
  252 --
  253 -- 'gcTls' opts in; an explicit path overrides the default; and setting both
  254 -- paths still turns TLS on by itself, which is the spelling the README used
  255 -- before the flag existed.
  256 effectiveTlsPaths :: FilePath -> FilePath -> GatewayConfig -> (FilePath, FilePath)
  257 effectiveTlsPaths defaultCert defaultKey cfg =
  258   ( maybe defaultCert T.unpack (gcTlsCert cfg)
  259   , maybe defaultKey T.unpack (gcTlsKey cfg) )
  260 
  261 -- | A TLS setting that cannot be served.
  262 --
  263 -- The two PEM paths are only meaningful together. A certificate without its
  264 -- key (or the reverse) is a typo, and saying so at boot beats failing inside
  265 -- the TLS handshake on the first connection.
  266 tlsConfigError :: GatewayConfig -> Maybe Text
  267 tlsConfigError cfg = case (gcTlsCert cfg, gcTlsKey cfg) of
  268   (Nothing, Nothing) -> Nothing
  269   (Just _, Just _)   -> Nothing
  270   (Just _, Nothing)  ->
  271     Just "[gateway] tls_cert is set but tls_key is not: both are required to serve HTTPS."
  272   (Nothing, Just _)  ->
  273     Just "[gateway] tls_key is set but tls_cert is not: both are required to serve HTTPS."
  274 
  275 -- | Is the configured origin consistent with serving HTTPS?
  276 --
  277 -- Once TLS is on, browsers reach the gateway at @https:\/\/@, so a verified
  278 -- origin that says @http:\/\/@ can never match: every registration and every
  279 -- assertion would be refused as an origin mismatch. The derived loopback
  280 -- origin (@http:\/\/localhost:\<port\>@) is exactly that case, so enabling TLS
  281 -- on a loopback bind also requires an explicit @origin@.
  282 tlsOriginError :: Bool -> Maybe Text -> Maybe Text
  283 tlsOriginError tls mOrigin
  284   | not tls = Nothing
  285   | otherwise = case mOrigin of
  286       Nothing -> Nothing
  287       Just o
  288         | "https://" `T.isPrefixOf` o -> Nothing
  289         | otherwise ->
  290             Just ("[gateway] tls_cert/tls_key are set but [access_control] origin is \""
  291                    <> o <> "\".\n  Clients will reach this gateway over https, so set origin to "
  292                    <> "the https URL they use (rp_id follows from its host).")

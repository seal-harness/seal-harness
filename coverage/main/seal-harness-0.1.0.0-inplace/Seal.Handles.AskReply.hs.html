    2 -- | The medium-agnostic ask/reply primitive.
    3 --
    4 -- Synchronous channels (the CLI TUI) can block inline on a direct prompt —
    5 -- the human is right there typing. Asynchronous channels (web, Signal,
    6 -- Telegram, …) cannot: the agent loop is running in one request/thread, and
    7 -- the human's answer will arrive in a *separate* future message. This module
    8 -- provides the shared, in-process store that bridges that gap without knowing
    9 -- anything about the transport.
   10 --
   11 -- The flow:
   12 --
   13 -- 1. The agent calls @ASK_HUMAN@, which runs @ccPrompt q@.
   14 -- 2. The channel's @ccPrompt@ closure calls 'askHuman', which mints a fresh
   15 --    'AskId', registers a 'PendingAsk' (with an empty answer slot) in the
   16 --    store, fires a medium-specific @notify@ callback (web: broadcast a WS
   17 --    event; Signal: @chSend@ the question to the peer; Telegram: send a
   18 --    message), and **blocks** on the answer slot (an STM 'TMVar').
   19 -- 3. The human's reply arrives through the channel's inbound path (web:
   20 --    POST @/api/sessions/:id/questions/:qid/answer@; Signal/Telegram: the
   21 --    next inbound message from that peer). That path calls 'deliverAnswer',
   22 --    which fills the answer slot and unblocks the agent-loop thread.
   23 -- 4. 'askHuman' returns the answer (or an 'AskOutcome' on timeout/cancel),
   24 --    and @ccPrompt@ maps it to 'Text' for the opcode.
   25 --
   26 -- The store is keyed by 'AskId'. 'pendingForSession' lets a channel list the
   27 -- open questions for a session (so the frontend can render them on
   28 -- reconnect). 'cancelAsk' abandons a question (timeout, session close).
   29 -- 'deliverAnswer' is idempotent on a given 'AskId' (a second answer is
   30 -- rejected).

  116 -- | Mint a fresh random 'AskId' (UUID v4). IO because it reads randomness.
  117 newAskId :: IO AskId
  118 newAskId = do
  119   w1 <- randomIO
  120   w2 <- randomIO
  121   let v1 = (w1 .&. complement 0x0000F000) .|. (0x4000 :: Word64)
  122       v2 = (w2 .|. 0x8000000000000000) .&. complement 0x4000000000000000
  123   pure (AskId (T.pack (U.toString (U.fromWords64 v1 v2))))

  129 -- | The scope of a human approval for an Untrusted opcode. Determines how
  130 -- long the approval is cached (so subsequent calls to the same opcode skip
  131 -- the prompt). 'ScopeOnce' never caches; 'ScopeForSession' caches for the
  132 -- session; 'ScopeAlways' caches globally; 'ScopeRejected' denies this call
  133 -- (and, for 'ScopeForSession'/'ScopeAlways', future calls to the same opcode).
  134 data ApprovalScope
  135   = ScopeOnce
  136   | ScopeForSession
  137   | ScopeAlways
  138   | ScopeRejected
  139   deriving stock (Eq, Show)

  171 -- | One discrete choice offered to the human alongside an @ASK_HUMAN@
  172 -- question. The 'qoLabel' is the value returned to the agent when the human
  173 -- picks this choice; the 'qoDescription' is a one-line explanation shown
  174 -- alongside the button (may be empty). Labels MUST be non-empty, ≤ 64
  175 -- bytes (UTF-8), and unique within one question's offer set — enforced by
  176 -- 'validateOptions'.
  177 data QuestionOption = QuestionOption
  178   { qoLabel       :: !Text
  179   , qoDescription :: !Text
  180   } deriving stock (Eq, Show, Generic)

  182 -- | ToJSON for 'QuestionOption' — encodes as @{label, description}@ (the
  183 -- 'qo' field prefix stripped so the wire shape matches the LLM schema + the
  184 -- frontend 'QuestionOption' type).
  185 instance ToJSON QuestionOption where
  186   toJSON (QuestionOption lbl desc) = object
  187     [ fromText "label" .= lbl
  188     , fromText "description" .= desc
  189     ]

  191 -- | FromJSON for 'QuestionOption' — decodes from @{label, description}@.

  198 -- | Validate a list of 'QuestionOption's for an @ASK_HUMAN@ invocation.
  199 -- Pure (no IO); returns 'Right' the validated list on success, 'Left' an
  200 -- error message on failure. Rules:
  201 --
  202 -- 1. Non-empty list (1 ≤ length).
  203 -- 2. At most 8 options.
  204 -- 3. Each 'qoLabel' non-empty and ≤ 64 bytes (UTF-8; 'Data.Text.length'
  205 --    counts code points, so we measure bytes explicitly to bound non-ASCII).
  206 -- 4. Each 'qoDescription' ≤ 200 chars.
  207 -- 5. Labels unique within the list (case-sensitive).
  208 validateOptions :: [QuestionOption] -> Either Text [QuestionOption]
  209 validateOptions [] = Left "options must be non-empty"
  210 validateOptions opts
  211   | length opts > 8 = Left "options must be at most 8"
  212   | otherwise = case foldr checkOpt (Right []) opts of
  213       Left e -> Left e
  214       Right _ ->
  215         let labels = map qoLabel opts
  216         in if length labels /= length (nub labels)
  217              then Left "option labels must be unique"
  218              else Right opts
  219   where
  220     checkOpt _ (Left e) = Left e
  221     checkOpt opt@(QuestionOption lbl desc) (Right acc)
  222       | T.null lbl          = Left "option label must be non-empty"
  223       | labelBytes > 64     = Left ("option label exceeds 64 bytes: " <> lbl)
  224       | T.length desc > 200 = Left ("option description exceeds 200 chars: " <> qoLabel opt)
  225       | otherwise           = Right (opt : acc)
  226       where
  227         labelBytes = BS.length (TE.encodeUtf8 lbl)

  229 -- | Format a question + its options as a numbered list for chat channels
  230 -- (Signal, CLI TUI). When the options are non-empty, the format is:
  231 --
  232 -- @
  233 -- \<question\>
  234 --
  235 -- 1) main — the default branch
  236 -- 2) develop — ...
  237 --
  238 -- Reply with a number or type your own answer.
  239 -- @
  240 --
  241 -- When the options are empty, just the question text (today's behavior).
  242 -- Pure (no IO).
  243 formatQuestionWithOptions :: Text -> [QuestionOption] -> Text
  244 formatQuestionWithOptions question [] = question
  245 formatQuestionWithOptions question opts =
  246   question
  247   <> "\n\n"
  248   <> T.intercalate "\n" (zipWith formatLine [1 :: Int ..] opts)
  249   <> "\n\nReply with a number or type your own answer."
  250   where
  251     formatLine n (QuestionOption lbl desc)
  252       | T.null desc = T.pack (show n <> ") ") <> lbl
  253       | otherwise   = T.pack (show n <> ") ") <> lbl <> " — " <> desc

  296 -- | The approval cache: maps @(SessionId, OpName)@ → 'ApprovalScope' so that
  297 -- "for this session" and "always" approvals short-circuit the prompt for
  298 -- subsequent calls to the same opcode. 'ScopeForSession' entries are keyed
  299 -- by session; 'ScopeAlways' entries use a synthetic global session key.
  300 data ApprovalCache = ApprovalCache
  301   { acSession  :: TVar (Map (SessionId, OpName) ApprovalScope)
  302   , acGlobal   :: TVar (Set OpName)
  303   }
  304 
  305 -- | Build a new, empty approval cache.
  306 newApprovalCache :: IO ApprovalCache
  307 newApprovalCache = ApprovalCache <$> newTVarIO Map.empty <*> newTVarIO Set.empty

  309 -- | Check the approval cache for a @(SessionId, OpName)@ pair. Returns
  310 -- 'Just ScopeRejected' if the opcode was previously rejected (for this
  311 -- session or globally), 'Just scope' if it was approved at that scope,
  312 -- 'Nothing' if no prior approval exists (the caller should prompt).
  313 checkApproval :: ApprovalCache -> SessionId -> OpName -> IO (Maybe ApprovalScope)
  314 checkApproval cache sid opName = do
  315   global <- readTVarIO (acGlobal cache)
  316   if opName `Set.member` global
  317     then pure (Just ScopeAlways)
  318     else do
  319       m <- readTVarIO (acSession cache)
  320       pure (Map.lookup (sid, opName) m)
  321 
  322 -- | Record an approval in the cache. 'ScopeOnce' is a no-op (no caching);
  323 -- 'ScopeForSession' inserts a session-scoped entry; 'ScopeAlways' inserts a
  324 -- global entry; 'ScopeRejected' inserts a session-scoped rejection (so
  325 -- subsequent calls in the same session short-circuit to denied). A
  326 -- 'ScopeRejected' does NOT elevate to global — the human may want to allow
  327 -- the opcode in another session.
  328 recordApproval :: ApprovalCache -> SessionId -> OpName -> ApprovalScope -> IO ()
  329 recordApproval cache sid opName scope =
  330   case scope of
  331     ScopeOnce       -> pure ()
  332     ScopeForSession -> atomically (modifyTVar' (acSession cache) (Map.insert (sid, opName) ScopeForSession))
  333     ScopeAlways     -> atomically (modifyTVar' (acGlobal cache) (Set.insert opName))
  334     ScopeRejected   -> atomically (modifyTVar' (acSession cache) (Map.insert (sid, opName) ScopeRejected))
  335 
  336 -- | Register a pending question, fire the @notify@ callback with its 'AskId',
  337 -- and block until the answer is delivered, the question is cancelled, or the
  338 -- timeout elapses. Returns 'Right' the reply, or 'Left' the outcome. The
  339 -- @notify@ callback is the medium-specific seam: the web broadcasts a WS
  340 -- event, Signal/Telegram @chSend@ the question to the peer. It is invoked
  341 -- /before/ blocking so the medium surfaces the question immediately. If
  342 -- @notify@ throws, the pending question is left registered (the caller may
  343 -- still deliver an answer); this is intentional — a flaky notify should not
  344 -- strand the agent loop, and 'cancelSessionAsks' cleans up on session close.
  345 askHuman
  346   :: AskReplyStore
  347   -> SessionId
  348   -> Text          -- ^ the question text
  349   -> (AskId -> IO ())  -- ^ the medium-specific notify callback
  350   -> IO (Either AskOutcome Text)
  351 askHuman store sid question notify = do
  352   r <- askHumanCore store sid question Nothing [] notify
  353   pure (case r of
  354     Left o   -> Left o
  355     Right ar -> Right (arText ar))

  357 -- | Like 'askHuman' but carries optional metadata (opcode name + input for
  358 -- the confirmation gate) in the 'PendingAsk'. The metadata is surfaced in the
  359 -- WS @ask@ event so the frontend can display the opcode name + input alongside
  360 -- the question. Returns the full 'AskReply' (scope + text).
  361 askHumanWithMeta
  362   :: AskReplyStore
  363   -> SessionId
  364   -> Text          -- ^ the question text
  365   -> Maybe Value   -- ^ optional metadata (opcode name + input)
  366   -> (AskId -> IO ())  -- ^ the medium-specific notify callback
  367   -> IO (Either AskOutcome AskReply)
  368 askHumanWithMeta store sid question meta =
  369   askHumanCore store sid question meta []

  371 -- | Like 'askHuman' but carries the offered 'QuestionOption's in the
  372 -- 'PendingAsk'. The options are surfaced in the WS @ask@ event + the
  373 -- @GET .../questions@ REST route so the frontend can render one button per
  374 -- option. The @notify@ callback fires before the blocking wait so the medium
  375 -- can surface the question + options immediately. Returns the human's typed

  377 askHumanWithOptions
  378   :: AskReplyStore
  379   -> SessionId

  382   -> (AskId -> IO ())  -- ^ the medium-specific notify callback
  383   -> IO (Either AskOutcome Text)
  384 askHumanWithOptions store sid question opts notify = do

  387     Left o   -> Left o

  390 -- | The shared core for 'askHuman'/'askHumanWithMeta'/'askHumanWithOptions'.
  391 askHumanCore
  392   :: AskReplyStore
  393   -> SessionId
  394   -> Text              -- ^ the question text
  395   -> Maybe Value       -- ^ optional metadata (opcode name + input)
  396   -> [QuestionOption]  -- ^ offered options (empty for open-ended + the confirmation gate)
  397   -> (AskId -> IO ())  -- ^ the medium-specific notify callback
  398   -> IO (Either AskOutcome AskReply)
  399 askHumanCore store sid question meta opts notify = do
  400   qid <- newAskId
  401   slot <- newEmptyTMVarIO
  402   now <- getCurrentTime
  403   let pa = PendingAsk { paId = qid, paSession = sid, paQuestion = question
  404                       , paCreatedAt = now, paMeta = meta, paOptions = opts, paSlot = slot }
  405   atomically (modifyTVar' (arsPending store) (Map.insert qid pa))
  406   notify qid
  407   timeoutUs <- readIORef (arsTimeoutUs store)
  408   if timeoutUs <= 0
  409     then atomically (takeTMVar slot)
  410     else do
  411       _ <- forkIO $ do
  412         threadDelay timeoutUs
  413         atomically $ do
  414           m <- readTVar (arsPending store)
  415           case Map.lookup qid m of
  416             Nothing -> pure ()
  417             Just pendingPa -> void (tryPutTMVar (paSlot pendingPa) (Left AoTimedOut))
  418       atomically (takeTMVar slot)

  420 -- | Deliver an answer to a pending question. Returns 'True' if the answer
  421 -- | was accepted (the question was pending and not yet answered), 'False'
  422 -- | otherwise (unknown id, already answered, or already cancelled). The
  423 -- | accepted answer unblocks the waiting 'askHuman' thread. Idempotent on a
  424 -- | given 'AskId': a second call is rejected.
  425 deliverAnswer :: AskReplyStore -> AskId -> AskReply -> IO Bool
  426 deliverAnswer store qid reply = do
  427   mPa <- atomically $ do
  428     m <- readTVar (arsPending store)
  429     case Map.lookup qid m of
  430       Nothing -> pure Nothing
  431       Just pa -> do
  432         let slot = paSlot pa
  433         won <- tryPutTMVar slot (Right reply)
  434         if won
  435           then writeTVar (arsPending store) (Map.delete qid m) >> pure (Just pa)
  436           else pure Nothing
  437   pure (case mPa of Just _ -> True; Nothing -> False)
  438 
  439 -- | Deliver an answer to the /oldest/ pending question for a session. Used by
  440 -- | inbox-driven async channels (Signal, Telegram) where the human's reply
  441 -- | arrives as the next inbound message from the peer — there is no ask-id in
  442 -- | the transport, so the FIFO queue of pending questions is matched in order.
  443 -- | Returns 'True' if a pending question was found and answered, 'False' if
  444 -- | the session has no pending questions (the answer is a spurious / unsolicited
  445 -- | inbound message the caller may route as a plain turn instead).
  446 deliverNextAnswer :: AskReplyStore -> SessionId -> Text -> IO Bool
  447 deliverNextAnswer store sid ans = do
  448   let reply = AskReply ScopeOnce ans
  449   mQid <- atomically $ do
  450     m <- readTVar (arsPending store)
  451     let matching = filter (\pa -> paSession pa == sid)
  452                   $ sortByCreatedAt (Map.elems m)
  453     case matching of
  454       [] -> pure Nothing
  455       pa : _ ->
  456         let slot = paSlot pa in do
  457           won <- tryPutTMVar slot (Right reply)
  458           if won
  459             then writeTVar (arsPending store) (Map.delete (paId pa) m)
  460                  >> pure (Just (paId pa))
  461             else
  462               writeTVar (arsPending store) (Map.delete (paId pa) m)
  463               >> pure Nothing
  464   case mQid of
  465     Just _  -> pure True
  466     Nothing ->
  467       do
  468         mQid2 <- atomically $ do
  469           m <- readTVar (arsPending store)
  470           let matching = filter (\pa -> paSession pa == sid)
  471                         $ sortByCreatedAt (Map.elems m)
  472           case matching of
  473             [] -> pure Nothing
  474             pa : _ ->
  475               let slot = paSlot pa in do
  476                 won <- tryPutTMVar slot (Right reply)
  477                 if won
  478                   then writeTVar (arsPending store) (Map.delete (paId pa) m)
  479                        >> pure (Just (paId pa))
  480                   else writeTVar (arsPending store) (Map.delete (paId pa) m)
  481                        >> pure Nothing
  482         pure (case mQid2 of Just _ -> True; Nothing -> False)
  483 
  484 -- | Sort pending questions oldest-first by 'paCreatedAt'. 'Map.elems' is
  485 -- unsorted (keyed by 'AskId'), so an explicit sort is needed for FIFO
  486 -- delivery.
  487 sortByCreatedAt :: [PendingAsk] -> [PendingAsk]
  488 sortByCreatedAt = sortBy (\a b -> compare (paCreatedAt a) (paCreatedAt b))
  489 
  490 -- | Deliver an inbound line as the answer to the oldest pending question
  491 -- across /all/ sessions (FIFO by 'paCreatedAt'). Returns 'True' if a
  492 -- pending question was found and answered, 'False' if none exists (the
  493 -- caller may route the line as a normal turn).
  494 --
  495 -- This is the session-agnostic counterpart of 'deliverNextAnswer': it serves
  496 -- channels with a single shared input stream that may answer questions for
  497 -- more than one session (e.g. the CLI TUI, where a @/bg@ turn forks a
  498 -- background session whose confirmation prompts must be answerable from the
  499 -- same @>@ prompt that serves the active session). Inbox-driven channels
  500 -- (Signal/Telegram) use the per-session 'deliverNextAnswer' because each
  501 -- conversation has its own inbound stream and cursor.
  502 deliverNextAnswerAny :: AskReplyStore -> Text -> IO Bool
  503 deliverNextAnswerAny store ans = do
  504   let reply = AskReply ScopeOnce ans
  505   mQid <- atomically $ do
  506     m <- readTVar (arsPending store)
  507     case sortByCreatedAt (Map.elems m) of
  508       [] -> pure Nothing
  509       pa : _ ->
  510         let slot = paSlot pa in do
  511           won <- tryPutTMVar slot (Right reply)
  512           if won
  513             then writeTVar (arsPending store) (Map.delete (paId pa) m)
  514                  >> pure (Just (paId pa))
  515             else
  516               writeTVar (arsPending store) (Map.delete (paId pa) m)
  517               >> pure Nothing
  518   pure (case mQid of Just _ -> True; Nothing -> False)

  520 -- | Resolve a 1-based numeric index into the pending ask's 'paOptions'.
  521 -- If the body (after 'T.strip') is a decimal 'Int' in @[1..length opts]@,
  522 -- return the indexed option's 'qoLabel'; otherwise return the body as-is.
  523 -- Pure (safe inside STM).
  524 resolveOptionIndex :: Text -> [QuestionOption] -> Text
  525 resolveOptionIndex body opts =
  526   let stripped = T.strip body
  527   in case TR.decimal stripped of
  528        Right (n, rest) | T.null rest && n >= 1 && n <= length opts ->
  529          qoLabel (opts !! (n - 1))
  530        _ -> body

  532 -- | Like 'deliverNextAnswer' but resolves a 1-based numeric index into the
  533 -- oldest pending ask's 'paOptions' to the indexed option's label. The
  534 -- resolution + the delivery happen in the /same/ STM transaction (gate:
  535 -- Architect #4, Security #2 — eliminates the TOCTOU race + the double-read
  536 -- of the original @resolveStockIndex@ design). 'T.strip' is applied before
  537 -- parsing; @02@ → 2 is accepted. Out-of-range numbers, non-numeric text,
  538 -- and asks with no options are delivered as-is ("Other"). Returns
  539 -- @(deliveredText, accepted)@.
  540 deliverNextAnswerResolved :: AskReplyStore -> SessionId -> Text -> IO (Text, Bool)
  541 deliverNextAnswerResolved store sid body = do
  542   mResult <- atomically $ do
  543     m <- readTVar (arsPending store)
  544     let matching = filter (\pa -> paSession pa == sid)
  545                   $ sortByCreatedAt (Map.elems m)
  546     case matching of
  547       [] -> pure Nothing
  548       pa : _ ->
  549         let resolved = resolveOptionIndex body (paOptions pa)
  550             slot = paSlot pa
  551             reply' = AskReply ScopeOnce resolved
  552         in do
  553           won <- tryPutTMVar slot (Right reply')
  554           if won
  555             then writeTVar (arsPending store) (Map.delete (paId pa) m)
  556                  >> pure (Just (resolved, True))
  557             else writeTVar (arsPending store) (Map.delete (paId pa) m)
  558                  >> pure (Just (resolved, False))
  559   pure (case mResult of
  560     Just (resolved, accepted) -> (resolved, accepted)
  561     Nothing                   -> (body, False))

  591 -- | 'True' if the question was pending and is now cancelled.
  592 cancelAsk :: AskReplyStore -> AskId -> IO Bool
  593 cancelAsk store qid = do
  594   mPa <- atomically $ do
  595     m <- readTVar (arsPending store)
  596     case Map.lookup qid m of
  597       Nothing -> pure Nothing
  598       Just pa -> do
  599         won <- tryPutTMVar (paSlot pa) (Left AoCancelled)
  600         if won
  601           then writeTVar (arsPending store) (Map.delete qid m) >> pure (Just pa)
  602           else pure Nothing
  603   pure (case mPa of Just _ -> True; Nothing -> False)

  605 -- | Cancel every pending question for a session (e.g. on session close or
  606 -- | agent termination). Each still-waiting 'askHuman' thread unblocks with
  607 -- | 'AoCancelled'. Already-answered questions are unaffected.
  608 cancelSessionAsks :: AskReplyStore -> SessionId -> IO ()
  609 cancelSessionAsks store sid = do
  610   toCancel <- atomically $ do
  611     m <- readTVar (arsPending store)
  612     let matching = Map.filter (\pa -> paSession pa == sid) m
  613         remaining = Map.difference m matching
  614     writeTVar (arsPending store) remaining
  615     pure (Map.elems matching)
  616   mapM_ (\pa -> atomically (void (tryPutTMVar (paSlot pa) (Left AoCancelled)))) toCancel

  618 -- | List the pending questions for a session, oldest-first. Used by a medium
  619 -- | to render open questions on reconnect (the web frontend polls this via
  620 -- | GET /api/sessions/:id/questions). Does /not/ expose the answer slot.
  621 -- | Includes the metadata (opcode name + input) when present so the frontend
  622 -- | can display it, and the offered options (for @ASK_HUMAN@) so the frontend
  623 -- | can render one button per option.
  624 pendingForSession
  625   :: AskReplyStore -> SessionId
  626   -> IO [PendingQuestionInfo]
  627 pendingForSession store sid =
  628   map (\pa -> PendingQuestionInfo
  629               { pqiId = paId pa
  630               , pqiSession = paSession pa
  631               , pqiQuestion = paQuestion pa
  632               , pqiCreatedAt = paCreatedAt pa
  633               , pqiMeta = paMeta pa
  634               , pqiOptions = paOptions pa
  635               })
  636   . filter (\pa -> paSession pa == sid)
  637   . Map.elems <$> readTVarIO (arsPending store)

  639 -- | Find a pending ask by an 8-hex prefix of its 'AskId'. Used by the
  640 -- Telegram inbound path: the @callback_data@ is @\"<8hex>:<label>\"@, and
  641 -- this scans the session's pending asks for one whose 'askIdText' starts
  642 -- with the prefix. Returns 'Nothing' if no match (the body was not a
  643 -- callback — fall through to 'deliverNextAnswerResolved').
  644 findByAskIdPrefix :: AskReplyStore -> SessionId -> Text -> IO (Maybe AskId)
  645 findByAskIdPrefix store sid prefix = do
  646   ps <- pendingForSession store sid
  647   pure (case filter (T.isPrefixOf prefix . askIdText . pqiId) ps of
  648     [] -> Nothing
  649     info : _ -> Just (pqiId info))

  651 -- | Look up a single pending question by id (for the delivery path to
  652 -- | validate). Returns 'Nothing' if the id is unknown or already answered.
  653 lookupAsk :: AskReplyStore -> AskId -> IO (Maybe (SessionId, Text))
  654 lookupAsk store qid = do
  655   m <- readTVarIO (arsPending store)
  656   case Map.lookup qid m of
  657     Nothing -> pure Nothing
  658     Just pa -> do
  659       mRes <- atomically (tryReadTMVar (paSlot pa))
  660       case mRes of
  661         Just _  -> pure Nothing
  662         Nothing -> pure (Just (paSession pa, paQuestion pa))

   23 -- | The image-provider interface.
   24 data ImageProvider = ImageProvider
   25   { ipGenerate :: Text -> IO (Either Text Text)   -- ^ prompt → image data (base64 or URL)
   26   , ipDescribe :: Text -> IO (Either Text Text)   -- ^ image ref → description
   27   }
   28 
   29 -- | The fail-closed default provider.
   30 noImageProvider :: ImageProvider
   31 noImageProvider = ImageProvider
   32   { ipGenerate = \_ -> pure (Left "IMAGE_GENERATE: no image provider configured")
   33   , ipDescribe = \_ -> pure (Left "IMAGE_DESCRIBE: no image provider configured")
   34   }
   35 
   36 -- | IMAGE_GENERATE opcode. Input: @{ prompt: Text }@.
   37 imageGenerateOp :: ImageProvider -> Opcode
   38 imageGenerateOp _prov = UntrustedOpcode
   39   { uoName = OpName "IMAGE_GENERATE"
   40   , uoDesc = "Generate an image from a prompt (provider-pluggable, fail-closed default)."
   41   , uoInSchema = imageSchema "prompt" "The image generation prompt."
   42   , uoOutSchema = object []
   43   , uoAuthorize = \v ->
   44       case promptField v of
   45         Nothing -> Left "IMAGE_GENERATE requires {prompt:string}"
   46         Just p | T.null p -> Left "IMAGE_GENERATE: prompt is empty"
   47                | otherwise -> Right ()
   48   , uoRun = \v -> do
   49       let p = fromMaybe "" (promptField v)
   50           recorded = object [ "prompt" .= p ]
   51       pure (OpResult [TrpText "IMAGE_GENERATE: no image provider configured"] True recorded)
   52   }
   53 
   54 -- | IMAGE_DESCRIBE opcode. Input: @{ image: Text }@.
   55 imageDescribeOp :: ImageProvider -> Opcode
   56 imageDescribeOp _prov = UntrustedOpcode
   57   { uoName = OpName "IMAGE_DESCRIBE"
   58   , uoDesc = "Describe an image (provider-pluggable, fail-closed default)."
   59   , uoInSchema = imageSchema "image" "The image reference (URL or base64)."
   60   , uoOutSchema = object []
   61   , uoAuthorize = \v ->
   62       case imageField v of
   63         Nothing -> Left "IMAGE_DESCRIBE requires {image:string}"
   64         Just i | T.null i -> Left "IMAGE_DESCRIBE: image is empty"
   65                | otherwise -> Right ()
   66   , uoRun = \v -> do
   67       let i = fromMaybe "" (imageField v)
   68           recorded = object [ "image" .= i ]
   69       pure (OpResult [TrpText "IMAGE_DESCRIBE: no image provider configured"] True recorded)
   70   }
   71 
   72 imageSchema :: Text -> Text -> Value
   73 imageSchema fieldName desc =
   74   object
   75     [ "type" .= ("object" :: Text)
   76     , "properties" .= object
   77         [ fromText fieldName .= object
   78             [ "type" .= ("string" :: Text)
   79             , "description" .= desc
   80             ]
   81         ]
   82     , "required" .= [fieldName]
   83     ]
   84 
   85 promptField :: Value -> Maybe Text
   86 promptField = parseMaybe (withObject "in" (.: "prompt"))
   87 
   88 imageField :: Value -> Maybe Text
   89 imageField = parseMaybe (withObject "in" (.: "image"))

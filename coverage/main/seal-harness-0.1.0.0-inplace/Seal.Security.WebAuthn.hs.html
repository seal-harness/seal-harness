    2 -- | WebAuthn (passkey) ceremony verification — the security core of the
    3 -- gateway's access control. Pure: every function here takes bytes and returns
    4 -- 'Either', so the whole surface is testable without a browser, a socket or a
    5 -- clock.
    6 --
    7 -- Scope (design §9):
    8 --
    9 -- * registration accepts @fmt = \"none\"@ only and checks the client data,
   10 --   the rpId hash, the presence and user-verification flags, and the attested
   11 --   credential data (id length, @rawId@ agreement, COSE key shape);
   12 -- * assertion checks the same client-data / rpId / flag invariants and
   13 --   verifies the ECDSA P-256 signature over
   14 --   @authData || SHA-256(clientDataJSON)@;
   15 -- * the sign-counter rule lives in 'checkSignCount' so the caller can run it
   16 --   under the same lock that persists the new value.
   17 --
   18 -- Attestation statements are deliberately not verified: we request
   19 -- @attestation: \"none\"@, and the WebAuthn spec does not require an RP to
   20 -- validate an attestation it did not ask for.

   70 -- | Everything that can make a ceremony fail. Callers map these to HTTP error
   71 -- codes via 'webAuthnErrorCode'; the constructors exist so the program can
   72 -- pattern-match on the failure class rather than parse a string.
   73 data WebAuthnError
   74   = WnMalformedClientData Text
   75   | WnWrongCeremony Text
   76   | WnChallengeMismatch
   77   | WnOriginMismatch Text
   78   | WnCrossOrigin
   79   | WnMalformedAuthenticatorData Text
   80   | WnRpIdMismatch
   81   | WnUserPresenceMissing
   82   | WnUserVerificationMissing
   83   | WnAttestedCredentialMissing
   84   | WnCredentialIdMismatch
   85   | WnCredentialIdLength Int
   86   | WnUnsupportedAttestation Text
   87   | WnMalformedCoseKey Text
   88   | WnSignatureInvalid
   89   | WnSignCountRegression Int Int
   90   deriving stock (Eq, Show)

   92 -- | A human-readable sentence (goes in the API's @error@ field).
   93 webAuthnErrorText :: WebAuthnError -> Text
   94 webAuthnErrorText err = case err of
   95   WnMalformedClientData d        -> "malformed clientDataJSON: " <> d
   96   WnWrongCeremony t              -> "unexpected ceremony type \"" <> t <> "\""
   97   WnChallengeMismatch            -> "the challenge does not match the one this server issued"
   98   WnOriginMismatch o             -> "the request origin (" <> o <> ") does not match this gateway's configured origin"
   99   WnCrossOrigin                  -> "cross-origin ceremonies are not accepted"
  100   WnMalformedAuthenticatorData d -> "malformed authenticator data: " <> d
  101   WnRpIdMismatch                 -> "the credential was registered for a different relying-party id"
  102   WnUserPresenceMissing          -> "the authenticator did not assert user presence"
  103   WnUserVerificationMissing      -> "the authenticator did not verify the user (PIN or biometric required)"
  104   WnAttestedCredentialMissing    -> "the registration response carried no credential"
  105   WnCredentialIdMismatch         -> "the credential id does not match the attested credential"
  106   WnCredentialIdLength n         -> "the credential id length (" <> tshow n <> ") is out of range"
  107   WnUnsupportedAttestation f     -> "unsupported attestation format \"" <> f <> "\""
  108   WnMalformedCoseKey d           -> "malformed public key: " <> d
  109   WnSignatureInvalid             -> "the signature does not verify"
  110   WnSignCountRegression old new  ->
  111     "the signature counter went backwards (" <> tshow new <> " after "
  112       <> tshow old <> "), which indicates a replayed assertion"

  151 data ClientData = ClientData
  152   { cdType        :: Text
  153   , cdChallenge   :: Text
  154   , cdOrigin      :: Text
  155   , cdCrossOrigin :: Bool
  156   } deriving stock (Eq, Show)

  158 -- | Parse @clientDataJSON@ with aeson (never by string matching).
  159 parseClientData :: ByteString -> Either WebAuthnError ClientData
  160 parseClientData raw =
  161   case A.eitherDecodeStrict' raw :: Either String Value of
  162     Left e -> Left (WnMalformedClientData (T.pack e))
  163     Right (Object o) -> ClientData
  164       <$> field o "type"
  165       <*> field o "challenge"
  166       <*> field o "origin"
  167       <*> pure (case KeyMap.lookup (Key.fromText "crossOrigin") o of
  168                   Just (Bool b) -> b
  169                   _             -> False)
  170     Right _ -> Left (WnMalformedClientData "not a JSON object")
  171   where
  172     field o k = case KeyMap.lookup (Key.fromText k) o of
  173       Just (String s) -> Right s
  174       Just _          -> Left (WnMalformedClientData ("field \"" <> k <> "\" is not a string"))
  175       Nothing         -> Left (WnMalformedClientData ("missing field \"" <> k <> "\""))

  177 -- | Check the ceremony type, the challenge and the origin.
  178 --
  179 -- Both sides of the challenge comparison are base64url-decoded, so a
  180 -- re-encoded (but equivalent) challenge cannot be used to smuggle a mismatch
  181 -- past an exact text comparison — and a different challenge never compares
  182 -- equal by accident.
  183 checkClientData :: Ceremony -> Text -> Text -> ClientData -> Either WebAuthnError ()
  184 checkClientData ceremony expectedChallenge expectedOrigin cd = do
  185   if cdType cd /= ceremonyText ceremony
  186     then Left (WnWrongCeremony (cdType cd))
  187     else Right ()
  188   if cdCrossOrigin cd
  189     then Left WnCrossOrigin
  190     else Right ()
  191   case (decodeB64Url (cdChallenge cd), decodeB64Url expectedChallenge) of
  192     (Right a, Right b) | a == b -> Right ()
  193     (Right _, Right _)          -> Left WnChallengeMismatch
  194     -- One side is not valid base64url: fall back to a padding-insensitive
  195     -- text comparison so a browser that only re-pads still verifies.
  196     _ | stripPadding (cdChallenge cd) == stripPadding expectedChallenge -> Right ()
  197       | otherwise -> Left WnChallengeMismatch
  198   case normalizeOrigin (cdOrigin cd) of
  199     Just o | o == normalizeOrigin' expectedOrigin -> Right ()
  200     _ -> Left (WnOriginMismatch (cdOrigin cd))
  201   where
  202     normalizeOrigin' = fromMaybe expectedOrigin . normalizeOrigin
  203 
  204 -- | Normalise an origin to @scheme://host[:port]@ with scheme and host
  205 -- lower-cased, a trailing slash removed, and the scheme's default port
  206 -- elided. Returns 'Nothing' for anything that is not an absolute
  207 -- @scheme://host@ URL.
  208 normalizeOrigin :: Text -> Maybe Text
  209 normalizeOrigin t0 =
  210   case T.breakOn "://" (T.dropWhileEnd (== '/') (T.strip t0)) of
  211     (scheme, rest)
  212       | T.null rest || T.null scheme -> Nothing
  213       | not (T.all validSchemeChar scheme) -> Nothing
  214       | otherwise ->
  215           let (authority, path) = T.break (== '/') (T.drop 3 rest)
  216           in if not (T.null path) || T.null authority
  217                then Nothing
  218                else
  219                  let schemeL = T.toLower scheme
  220                      (host, port) = splitHostPort authority
  221                      hostL = T.toLower host
  222                      defaultPort = case schemeL of
  223                        "https" -> Just "443"
  224                        "http"  -> Just "80"
  225                        _       -> Nothing
  226                  in if T.null hostL
  227                       then Nothing
  228                       else Just $ case port of
  229                         Just p | Just p /= defaultPort -> schemeL <> "://" <> hostL <> ":" <> p
  230                         _                              -> schemeL <> "://" <> hostL
  231   where
  232     validSchemeChar c =
  233       isAsciiLower c || isAsciiUpper c || isDigit c || c == '+' || c == '-' || c == '.'
  234 
  235 -- | Split @host[:port]@. A bare IPv6 literal is returned unchanged with no
  236 -- port (the gateway never derives an IPv6 origin; a wildcard bind has no
  237 -- derivable origin at all).
  238 splitHostPort :: Text -> (Text, Maybe Text)
  239 splitHostPort authority =
  240   case T.breakOnEnd ":" authority of
  241     (withColon, port)
  242       | T.null withColon -> (authority, Nothing)
  243       | T.any (== ':') (T.dropEnd 1 withColon) -> (authority, Nothing)
  244       | T.null port -> (authority, Nothing)
  245       | T.all isDigit port -> (T.dropEnd 1 withColon, Just port)
  246       | otherwise -> (authority, Nothing)

  252 -- | The parsed fixed header of @authenticatorData@ plus the attested
  253 -- credential data when the @AT@ flag is set.
  254 data AuthenticatorData = AuthenticatorData
  255   { adRpIdHash   :: ByteString
  256   , adFlags      :: Word8
  257   , adSignCount  :: Int
  258   , adCredential :: Maybe (ByteString, CosePublicKey)
  259   , adRaw        :: ByteString
  260     -- ^ The exact bytes, needed verbatim for signature verification.
  261   } deriving stock (Eq, Show)
  262 
  263 flagUserPresent, flagUserVerified, flagAttested :: Word8
  264 flagUserPresent  = 0x01
  265 flagUserVerified = 0x04
  266 flagAttested     = 0x40
  267 
  268 adUserPresent :: AuthenticatorData -> Bool
  269 adUserPresent ad = adFlags ad .&. flagUserPresent /= 0
  270 
  271 adUserVerified :: AuthenticatorData -> Bool
  272 adUserVerified ad = adFlags ad .&. flagUserVerified /= 0
  273 
  274 adHasAttestedCredential :: AuthenticatorData -> Bool
  275 adHasAttestedCredential ad = adFlags ad .&. flagAttested /= 0
  276 
  277 -- | Parse @authenticatorData@. Requires at least the 37-byte fixed header;
  278 -- when @AT@ is set, parses @aaguid(16) || credIdLen(2) || credId || COSE key@.
  279 parseAuthenticatorData :: ByteString -> Either WebAuthnError AuthenticatorData
  280 parseAuthenticatorData raw
  281   | BS.length raw < 37 =
  282       Left (WnMalformedAuthenticatorData "shorter than the 37-byte header")
  283   | flags .&. flagAttested == 0 =
  284       Right AuthenticatorData
  285         { adRpIdHash = rpIdHash, adFlags = flags, adSignCount = signCount
  286         , adCredential = Nothing, adRaw = raw }
  287   | otherwise = do
  288       (credId, credIdLen, keyBytes) <- parseAttested rest
  289       if credIdLen < 1 || credIdLen > 1023
  290         then Left (WnCredentialIdLength credIdLen)
  291         else do
  292           cose <- case decodeCborPrefix keyBytes of
  293             Left e           -> Left (WnMalformedCoseKey e)
  294             Right (value, _) -> Right value
  295           key <- parseCoseKey cose
  296           Right AuthenticatorData
  297             { adRpIdHash = rpIdHash, adFlags = flags, adSignCount = signCount
  298             , adCredential = Just (credId, key), adRaw = raw }
  299   where
  300     rpIdHash  = BS.take 32 raw
  301     flags     = BS.index raw 32
  302     signCount = be32 (BS.take 4 (BS.drop 33 raw))
  303     rest      = BS.drop 37 raw
  304 
  305 -- | Split @aaguid(16) || credIdLen(2) || credId@ off the front, returning the
  306 -- id, its declared length and the remaining (COSE key + optional extensions)
  307 -- bytes.
  308 parseAttested :: ByteString -> Either WebAuthnError (ByteString, Int, ByteString)
  309 parseAttested r
  310   | BS.length r < 18 =
  311       Left (WnMalformedAuthenticatorData "truncated attested credential data")
  312   | otherwise =
  313       let afterAaguid = BS.drop 16 r
  314           len = be16 (BS.take 2 afterAaguid)
  315           credId = BS.take len (BS.drop 2 afterAaguid)
  316       in if BS.length credId /= len
  317            then Left (WnMalformedAuthenticatorData "truncated credential id")
  318            else Right (credId, len, BS.drop (2 + len) afterAaguid)
  319 
  320 -- | Parse a COSE @EC2@ public key: @{1: 2, 3: -7, -1: 1, -2: x, -3: y}@.
  321 parseCoseKey :: Cbor -> Either WebAuthnError CosePublicKey
  322 parseCoseKey cose = do
  323   kty <- intField 1
  324   alg <- intField 3
  325   crv <- intField (-1)
  326   x   <- bytesField (-2)
  327   y   <- bytesField (-3)
  328   if kty /= 2
  329     then Left (WnMalformedCoseKey "kty is not EC2 (2)")
  330     else if alg /= -7
  331       then Left (WnMalformedCoseKey "alg is not ES256 (-7)")
  332       else if crv /= 1
  333         then Left (WnMalformedCoseKey "crv is not P-256 (1)")
  334         else if BS.length x /= 32 || BS.length y /= 32
  335           then Left (WnMalformedCoseKey "x/y are not 32 bytes each")
  336           else do
  337             let key = CosePublicKey (x <> y)
  338             _ <- cosePublicKeyToPoint key
  339             Right key
  340   where
  341     intField k = case cborLookupInt k cose of
  342       Just (CborInt n) -> Right n
  343       Just _           -> Left (WnMalformedCoseKey ("field " <> tshow k <> " is not an integer"))
  344       Nothing          -> Left (WnMalformedCoseKey ("missing field " <> tshow k))
  345     bytesField k = case cborLookupInt k cose of
  346       Just (CborBytes b) -> Right b
  347       Just _             -> Left (WnMalformedCoseKey ("field " <> tshow k <> " is not a byte string"))
  348       Nothing            -> Left (WnMalformedCoseKey ("missing field " <> tshow k))
  349 
  350 -- | Convert a 64-byte @x || y@ key into a crypton 'PublicKey', rejecting a
  351 -- point that is not on the curve.
  352 cosePublicKeyToPoint :: CosePublicKey -> Either WebAuthnError PublicKey
  353 cosePublicKeyToPoint (CosePublicKey bytes)
  354   | BS.length bytes /= 64 = Left (WnMalformedCoseKey "key is not 64 bytes")
  355   | not (isPointValid curve point) = Left (WnMalformedCoseKey "point is not on P-256")
  356   | otherwise = Right (PublicKey curve point)
  357   where
  358     (xb, yb) = BS.splitAt 32 bytes
  359     point = Point (beInteger xb) (beInteger yb)
  360     curve = getCurveByName SEC_p256r1

  381 -- | Verify a @navigator.credentials.create()@ response.
  382 verifyRegistration :: RegistrationInput -> Either WebAuthnError RegistrationResult
  383 verifyRegistration ri = do
  384   cd <- parseClientData (riClientDataJSON ri)
  385   checkClientData CerCreate (riExpectedChallenge ri) (riExpectedOrigin ri) cd
  386   authData <- attestationAuthData (riAttestationObject ri)
  387   ad <- parseAuthenticatorData authData
  388   checkHeader (riExpectedRpId ri) ad
  389   (credId, key) <- case adCredential ad of
  390     Just pair -> Right pair
  391     Nothing   -> Left WnAttestedCredentialMissing
  392   if credId /= riRawId ri
  393     then Left WnCredentialIdMismatch
  394     else if BS.length credId < 1 || BS.length credId > 1023
  395       then Left (WnCredentialIdLength (BS.length credId))
  396       else Right RegistrationResult
  397              { rrCredentialId = credId
  398              , rrPublicKey = key
  399              , rrSignCount = adSignCount ad
  400              }

  402 -- | Pull @authData@ out of the attestation object, requiring
  403 -- @fmt = \"none\"@.
  404 attestationAuthData :: ByteString -> Either WebAuthnError ByteString
  405 attestationAuthData raw = do
  406   cose <- either (Left . WnUnsupportedAttestation) Right (decodeCbor raw)
  407   pairs <- case cose of
  408     CborMap ps -> Right ps
  409     _          -> Left (WnUnsupportedAttestation "attestation object is not a map")
  410   fmt <- case lookup (CborText "fmt") pairs of
  411     Just (CborText f) -> Right f
  412     _                 -> Left (WnUnsupportedAttestation "missing fmt")
  413   if fmt /= "none"
  414     then Left (WnUnsupportedAttestation fmt)
  415     else case lookup (CborText "authData") pairs of
  416       Just (CborBytes b) -> Right b
  417       _                  -> Left (WnUnsupportedAttestation "missing authData")
  418 
  419 -- | The invariants shared by both ceremonies.
  420 checkHeader :: Text -> AuthenticatorData -> Either WebAuthnError ()
  421 checkHeader expectedRpId ad = do
  422   if adRpIdHash ad /= sha256 (TE.encodeUtf8 expectedRpId)
  423     then Left WnRpIdMismatch
  424     else Right ()
  425   if not (adUserPresent ad)
  426     then Left WnUserPresenceMissing
  427     else Right ()
  428   if not (adUserVerified ad)
  429     then Left WnUserVerificationMissing
  430     else Right ()

  436 data AssertionInput = AssertionInput

  439   , aiSignature         :: ByteString

  446 -- | Verify a @navigator.credentials.get()@ response. Returns the new sign
  447 -- counter; the caller persists it under the same lock, so the
  448 -- read-check-write cannot interleave with a replayed assertion.
  449 verifyAssertion :: CosePublicKey -> AssertionInput -> Either WebAuthnError Int
  450 verifyAssertion key ai = do
  451   cd <- parseClientData (aiClientDataJSON ai)
  452   checkClientData CerGet (aiExpectedChallenge ai) (aiExpectedOrigin ai) cd
  453   ad <- parseAuthenticatorData (aiAuthenticatorData ai)
  454   checkHeader (aiExpectedRpId ai) ad
  455   newCount <- checkSignCount (aiStoredSignCount ai) (adSignCount ad)
  456   (r, s) <- either
  457     (Left . WnMalformedAuthenticatorData . ("signature: " <>))
  458     Right
  459     (decodeEcdsaSignature (aiSignature ai))
  460   pub <- cosePublicKeyToPoint key
  461   let message = aiAuthenticatorData ai <> sha256 (aiClientDataJSON ai)
  462   if verify SHA256 pub (Signature r s) message
  463     then Right newCount
  464     else Left WnSignatureInvalid

  466 -- | The WebAuthn sign-counter rule, stated exactly:
  467 --
  468 -- * a stored counter of 0 and a reported counter of 0 means the authenticator
  469 --   does not implement one — accept (the spec allows this for platform
  470 --   authenticators);
  471 -- * a reported counter of 0 when we stored a non-zero one is a regression;
  472 -- * otherwise the reported counter must be strictly greater.
  473 checkSignCount :: Int -> Int -> Either WebAuthnError Int
  474 checkSignCount stored new
  475   | stored == 0 && new == 0 = Right 0
  476   | new <= stored           = Left (WnSignCountRegression stored new)
  477   | otherwise               = Right new

  483 sha256 :: ByteString -> ByteString
  484 sha256 bs = convert (hash bs :: Digest SHA256)
  485 
  486 be16 :: ByteString -> Int
  487 be16 = fromIntegral . beInteger
  488 
  489 be32 :: ByteString -> Int
  490 be32 = fromIntegral . beInteger
  491 
  492 beInteger :: ByteString -> Integer
  493 beInteger = BS.foldl' (\acc w -> acc `shiftL` 8 + fromIntegral w) 0

  495 decodeB64Url :: Text -> Either Text ByteString
  496 decodeB64Url = either (Left . T.pack) Right . B64Url.decodeUnpadded . TE.encodeUtf8

  501 tshow :: Show a => a -> Text

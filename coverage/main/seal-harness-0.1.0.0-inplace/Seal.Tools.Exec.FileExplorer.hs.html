   60 -- | The error ADT for 'FileExplorerHandle' methods. Pinned for exhaustive
   61 -- pattern-match under @-Wall -Werror@.
   62 data FileExplorerErr
   63   = FeNotFound          -- ^ path does not exist
   64   | FeIsDirectory       -- ^ path is a directory (for content/download)
   65   | FeNotADirectory     -- ^ path is a file (for listing)
   66   | FePermissionDenied  -- ^ permission denied
   67   | FeIo !Text          -- ^ I/O error (message)
   68   | FeExec !Text        -- ^ remote SSH exec failure (message)
   69   | FeTruncated         -- ^ output exceeded the byte ceiling
   70   | FeDisabled          -- ^ file explorer disabled (non-loopback)
   71   deriving stock (Eq, Show)

   77 -- | One entry in a directory listing.
   78 data FileEntry = FileEntry
   79   { feName :: !Text              -- ^ basename
   80   , feIsDir :: !Bool             -- ^ True = directory, False = file
   81   , feIsSymlink :: !Bool         -- ^ True = symlink (not followed in listing)
   82   , feSize :: !(Maybe Integer)   -- ^ Nothing for directories/symlinks or when unavailable
   83   , feMtime :: !(Maybe UTCTime)  -- ^ Nothing when unavailable
   84   } deriving stock (Eq, Show)

   86 -- | The result of a directory listing.
   87 data FileListing = FileListing

   89   , flEntries :: ![FileEntry]
   90   , flTruncated :: !Bool         -- ^ True if the listing was cut short
   91   , flWarning :: !(Maybe Text)   -- ^ sensitive-directory warning
   92   } deriving stock (Eq, Show)

  107 -- | The file explorer capability handle. Abstracts local vs remote
  108 -- execution machine access. The @Int@ parameter on construction is the
  109 -- operator-configured byte ceiling for reads (same @maxScanBytes@ as
  110 -- @FILE_READ@).
  111 data FileExplorerHandle = FileExplorerHandle
  112   { feListDirectory :: FilePath -> IO (Either FileExplorerErr FileListing)
  113   , feListDirectories :: [FilePath] -> IO (Either FileExplorerErr [FileListing])
  114     -- ^ List several directories in ONE round trip. Returns listings in
  115     -- request order. On a truncated transport, only the directories that
  116     -- arrived complete are returned (never a partial listing): the caller is
  117     -- expected to re-request the remainder individually. See 'remoteListDirsCmd'.
  118   , feReadFile :: FilePath -> IO (Either FileExplorerErr FileContent)
  119   , feReadFileBytes :: FilePath -> IO (Either FileExplorerErr (ByteString, Bool))
  120     -- ^ (bytes up to ceiling, truncated?) — for the download endpoint.
  121     -- Returns raw bytes so binary files are not corrupted by UTF-8
  122     -- decode/encode round-trip. The 'Bool' is whether the file exceeded
  123     -- the byte ceiling.
  124   , feResolveHome :: IO (Either FileExplorerErr FilePath)
  125   }

  164 -- | Lexically collapse @.@ and @..@ segments in a path. Wraps
  165 -- 'lexicalCollapse' from "Seal.Security.Path".
  166 collapsePath :: FilePath -> FilePath
  167 collapsePath = joinPath . lexicalCollapse . splitDirectories

  169 -- | Check if any component of the path is a blocked (sensitive) name.
  170 -- Returns a warning message if so, 'Nothing' otherwise.
  171 checkBlocked :: FilePath -> Maybe Text
  172 checkBlocked path =
  173   if any (`elem` blockedNames) (splitDirectories path)
  174     then Just sensitiveWarning
  175     else Nothing

  177 -- | Map an 'ExecError' from the remote SSH runner to a 'FileExplorerErr'.
  178 -- The file explorer has no TOFU host-key adoption flow (unlike the CLI
  179 -- channel), so 'ExecHostKeyUnknown' gets a descriptive message with the
  180 -- manual fix instructions — same approach as the web channel's
  181 -- session-exec path in "Seal.Session.Workdir".
  182 mapRemoteExecError :: SshConfig -> ExecError -> FileExplorerErr
  183 mapRemoteExecError sshCfg = \case
  184   ExecHostKeyUnknown -> FeExec
  185     ("remote host key not in known_hosts (TOFU adoption not available from the file explorer). "
  186      <> "Add the key manually with: ssh-keyscan -H "
  187      <> getSshHost (scHost sshCfg)
  188      <> " >> " <> T.pack (scKnownHosts sshCfg))
  189   e -> FeExec (T.pack (show e))

  211 -- | The local file explorer. Reads via 'System.Directory' on the harness
  212 -- machine. The @Int@ is the operator-configured byte ceiling for reads,
  213 -- captured at construction time.
  214 mkLocalFileExplorer :: Int -> FileExplorerHandle
  215 mkLocalFileExplorer ceilingBytes = FileExplorerHandle
  216   { feListDirectory = localListDirectory ceilingBytes
  217   , feListDirectories = localListDirectories ceilingBytes
  218   , feReadFile = localReadFile ceilingBytes
  219   , feReadFileBytes = localReadFileBytes ceilingBytes
  220   , feResolveHome = localResolveHome
  221   }

  223 -- | Read at most @maxBytes@ from a local file. Returns the bytes read and
  224 -- a 'Bool' indicating whether the file was truncated (the file is larger
  225 -- than @maxBytes@).
  226 readBoundedFile :: FilePath -> Int -> IO (BS.ByteString, Bool)
  227 readBoundedFile path maxBytes = do
  228   bs <- BS.readFile path
  229   let len = BS.length bs
  230   if len <= maxBytes
  231     then pure (bs, False)
  232     else pure (BS.take maxBytes bs, True)

  234 -- | List a local directory. Applies 'lexicalCollapse' to the path first,
  235 -- checks existence/type, then enumerates entries with best-effort
  236 -- metadata. Entries are sorted: directories first, then files,
  237 -- alphabetically. Truncates at 'listingEntryLimit' entries.
  238 localListDirectory :: Int -> FilePath -> IO (Either FileExplorerErr FileListing)
  239 localListDirectory _ origPath = do
  240   let path = collapsePath origPath
  241       warning = checkBlocked path
  242   exists <- try (doesDirectoryExist path) :: IO (Either IOException Bool)
  243   case exists of
  244     Left ioErr
  245       | isDoesNotExistError ioErr -> pure (Left FeNotFound)
  246       | isPermissionError ioErr -> pure (Left FePermissionDenied)
  247       | otherwise -> pure (Left (FeIo (T.pack (show ioErr))))
  248     Right False -> do
  249       isFile <- try (doesFileExist path) :: IO (Either IOException Bool)
  250       case isFile of
  251         Right True -> pure (Left FeNotADirectory)
  252         Right False -> pure (Left FeNotFound)
  253         Left ioErr
  254           | isPermissionError ioErr -> pure (Left FePermissionDenied)
  255           | otherwise -> pure (Left FeNotFound)
  256     Right True -> do
  257       eNames <- try (listDirectory path) :: IO (Either IOException [FilePath])
  258       case eNames of
  259         Left ioErr
  260           | isPermissionError ioErr -> pure (Left FePermissionDenied)
  261           | otherwise -> pure (Left (FeIo (T.pack (show ioErr))))
  262         Right names -> do
  263           entries <- mapM (localEntry path) names
  264           let sorted = sortBy (comparing (\e -> (not (feIsDir e), feName e))) entries
  265               (kept, truncated) =
  266                 if length sorted > listingEntryLimit
  267                   then (take listingEntryLimit sorted, True)
  268                   else (sorted, False)
  269           pure (Right FileListing
  270             { flPath = T.pack path
  271             , flEntries = kept
  272             , flTruncated = truncated
  273             , flWarning = warning
  274             })

  276 -- | List several local directories. Sequential and cheap (no round trip per
  277 -- directory), sharing exactly the semantics of 'localListDirectory', so the
  278 -- batched and single-path arms cannot drift.
  279 localListDirectories :: Int -> [FilePath] -> IO (Either FileExplorerErr [FileListing])
  280 localListDirectories ceilingBytes = go []
  281   where
  282     go acc [] = pure (Right (reverse acc))
  283     go acc (p : ps) = do
  284       r <- localListDirectory ceilingBytes p
  285       case r of
  286         Left e -> pure (Left e)
  287         Right l -> go (l : acc) ps

  289 -- | Build a 'FileEntry' for one child of a local directory. Checks
  290 -- symlink, directory, file type, and best-effort size/mtime.
  291 localEntry :: FilePath -> FilePath -> IO FileEntry
  292 localEntry parent name = do
  293   let full = parent </> name
  294   eLink <- try (pathIsSymbolicLink full) :: IO (Either IOException Bool)
  295   case eLink of
  296     Right True -> pure FileEntry
  297       { feName = T.pack name
  298       , feIsDir = False
  299       , feIsSymlink = True
  300       , feSize = Nothing
  301       , feMtime = Nothing
  302       }
  303     _ -> do
  304       isDir <- try (doesDirectoryExist full) :: IO (Either IOException Bool)
  305       case isDir of
  306         Right True -> do
  307           eMtime <- try (getModificationTime full) :: IO (Either IOException UTCTime)
  308           pure FileEntry
  309             { feName = T.pack name
  310             , feIsDir = True
  311             , feIsSymlink = False
  312             , feSize = Nothing
  313             , feMtime = either (const Nothing) Just eMtime
  314             }
  315         _ -> do
  316           isFile <- try (doesFileExist full) :: IO (Either IOException Bool)
  317           case isFile of
  318             Right True -> do
  319               eSize <- try (getFileSize full) :: IO (Either IOException Integer)
  320               eMtime <- try (getModificationTime full) :: IO (Either IOException UTCTime)
  321               pure FileEntry
  322                 { feName = T.pack name
  323                 , feIsDir = False
  324                 , feIsSymlink = False
  325                 , feSize = either (const Nothing) Just eSize
  326                 , feMtime = either (const Nothing) Just eMtime
  327                 }
  328             _ -> pure FileEntry
  329               { feName = T.pack name
  330               , feIsDir = False
  331               , feIsSymlink = False
  332               , feSize = Nothing
  333               , feMtime = Nothing
  334               }

  336 -- | Read a local file as 'FileContent'. Applies 'lexicalCollapse', checks
  337 -- type, bounded read, UTF-8 decode, NUL-byte binary detection, line count
  338 -- via 'T.lines'.
  339 localReadFile :: Int -> FilePath -> IO (Either FileExplorerErr FileContent)
  340 localReadFile ceilingBytes origPath = do
  341   let path = collapsePath origPath
  342   eRes <- checkReadable path
  343   case eRes of
  344     Left err -> pure (Left err)
  345     Right () -> do
  346       eBs <- try (readBoundedFile path ceilingBytes)
  347                 :: IO (Either IOException (BS.ByteString, Bool))
  348       case eBs of
  349         Left ioErr
  350           | isPermissionError ioErr -> pure (Left FePermissionDenied)
  351           | isDoesNotExistError ioErr -> pure (Left FeNotFound)
  352           | otherwise -> pure (Left (FeIo (T.pack (show ioErr))))
  353         Right (bs, truncated) -> do
  354           let isBinary = BS.elem 0 bs
  355               content = if isBinary then "" else TE.decodeUtf8Lenient bs
  356               linesRead = if isBinary then 0 else length (T.lines content)
  357           pure (Right FileContent
  358             { fcPath = T.pack path
  359             , fcContent = content
  360             , fcTruncated = truncated
  361             , fcLinesRead = linesRead
  362             , fcIsBinary = isBinary
  363             })
  364 
  365 -- | Read a local file as raw 'ByteString'. Applies 'lexicalCollapse',
  366 -- checks type, bounded read — returns raw bytes (no UTF-8 decode).
  367 localReadFileBytes :: Int -> FilePath -> IO (Either FileExplorerErr (ByteString, Bool))
  368 localReadFileBytes ceilingBytes origPath = do
  369   let path = collapsePath origPath
  370   eRes <- checkReadable path
  371   case eRes of
  372     Left err -> pure (Left err)
  373     Right () -> do
  374       eBs <- try (readBoundedFile path ceilingBytes)
  375                 :: IO (Either IOException (BS.ByteString, Bool))
  376       case eBs of
  377         Left ioErr
  378           | isPermissionError ioErr -> pure (Left FePermissionDenied)
  379           | isDoesNotExistError ioErr -> pure (Left FeNotFound)
  380           | otherwise -> pure (Left (FeIo (T.pack (show ioErr))))
  381         Right (bs, truncated) -> pure (Right (bs, truncated))
  382 
  383 -- | Check that a path exists and is not a directory (for file-read
  384 -- methods). Returns 'Left FeNotFound' if missing, 'Left FeIsDirectory'
  385 -- if it's a directory, 'Right ()' if readable.
  386 checkReadable :: FilePath -> IO (Either FileExplorerErr ())
  387 checkReadable path = do
  388   eDir <- try (doesDirectoryExist path) :: IO (Either IOException Bool)
  389   case eDir of
  390     Left ioErr
  391       | isDoesNotExistError ioErr -> pure (Left FeNotFound)
  392       | isPermissionError ioErr -> pure (Left FePermissionDenied)
  393       | otherwise -> pure (Left (FeIo (T.pack (show ioErr))))
  394     Right True -> pure (Left FeIsDirectory)
  395     Right False -> do
  396       eFile <- try (doesFileExist path) :: IO (Either IOException Bool)
  397       case eFile of
  398         Right True -> pure (Right ())
  399         Right False -> pure (Left FeNotFound)
  400         Left ioErr
  401           | isPermissionError ioErr -> pure (Left FePermissionDenied)
  402           | otherwise -> pure (Left FeNotFound)
  403 
  404 -- | Resolve the home directory on the local machine.
  405 localResolveHome :: IO (Either FileExplorerErr FilePath)
  406 localResolveHome = do
  407   eHome <- try getHomeDirectory :: IO (Either IOException FilePath)
  408   pure $ case eHome of
  409     Left ioErr -> Left (FeIo (T.pack (show ioErr)))
  410     Right home -> Right home

  416 -- | The remote file explorer. Reads via SSH through a 'RemoteRunner' to
  417 -- the execution machine. The @Int@ is the operator-configured byte
  418 -- ceiling for reads, captured at construction time.
  419 --
  420 -- The home-directory resolution is cached in an 'IORef' per handle
  421 -- instance (per-process, never invalidated; if the SSH user's home
  422 -- changes, a server restart refreshes it).
  423 mkRemoteFileExplorer :: SshConfig -> RemoteRunner -> Int -> IO FileExplorerHandle
  424 mkRemoteFileExplorer sshCfg runner ceilingBytes = do
  425   homeCache <- newIORef Nothing
  426   pure $ FileExplorerHandle
  427     { feListDirectory = remoteListDirectory sshCfg runner ceilingBytes
  428     , feListDirectories = remoteListDirectories sshCfg runner ceilingBytes
  429     , feReadFile = remoteReadFile sshCfg runner ceilingBytes
  430     , feReadFileBytes = remoteReadFileBytes sshCfg runner ceilingBytes
  431     , feResolveHome = remoteResolveHome sshCfg runner homeCache
  432     }

  434 -- | List a remote directory via ONE SSH round trip. The command assigns
  435 -- the path to a shell variable first (via 'shellQuote'), then lists
  436 -- entries with type/size/mtime. Uses @--@ separator before the path
  437 -- argument in @ls@. Output is bounded by the 'RemoteRunner'\'s 50KB cap.
  438 remoteListDirectory
  439   :: SshConfig -> RemoteRunner -> Int -> FilePath
  440   -> IO (Either FileExplorerErr FileListing)
  441 remoteListDirectory sshCfg runner _ origPath = do
  442   let path = collapsePath origPath
  443       warning = checkBlocked path
  444       cmdText = remoteListCmd path
  445   case mkShellCommand cmdText of
  446     Left _ -> pure (Left (FeExec "invalid command (NUL in path?)"))
  447     Right cmd -> do
  448       res <- runRemoteShell runner sshCfg cmd
  449       case res of
  450         Left e -> pure (Left (mapRemoteExecError sshCfg e))
  451         Right out -> do
  452           let (cleanOut, runnerTruncated) = stripTruncationMarker out
  453           if notFoundMarker `T.isPrefixOf` cleanOut
  454             then pure (Left FeNotFound)
  455             else if notDirMarker `T.isPrefixOf` cleanOut
  456               then pure (Left FeNotADirectory)
  457               else do
  458                 let entries = sortBy
  459                       (comparing (\e -> (not (feIsDir e), feName e)))
  460                       (parseRemoteListing cleanOut)
  461                     (kept, listTruncated) =
  462                       if length entries > listingEntryLimit
  463                         then (take listingEntryLimit entries, True)
  464                         else (entries, False)
  465                 pure (Right FileListing
  466                   { flPath = T.pack path
  467                   , flEntries = kept
  468                   , flTruncated = runnerTruncated || listTruncated
  469                   , flWarning = warning
  470                   })

  472 -- | Read a remote file as 'FileContent' via SSH @head -c <ceil> -- "$p"@.
  473 -- UTF-8 decode, NUL-byte binary detection, line count via 'T.lines'.
  474 remoteReadFile
  475   :: SshConfig -> RemoteRunner -> Int -> FilePath
  476   -> IO (Either FileExplorerErr FileContent)
  477 remoteReadFile sshCfg runner ceilingBytes origPath = do
  478   let path = collapsePath origPath
  479       cmdText = remoteReadCmd path ceilingBytes
  480   case mkShellCommand cmdText of
  481     Left _ -> pure (Left (FeExec "invalid command (NUL in path?)"))
  482     Right cmd -> do
  483       res <- runRemoteShell runner sshCfg cmd
  484       case res of
  485         Left e -> pure (Left (mapRemoteExecError sshCfg e))
  486         Right out -> do
  487           let (cleanOut, runnerTruncated) = stripTruncationMarker out
  488           if notFoundMarker `T.isPrefixOf` cleanOut
  489             then pure (Left FeNotFound)
  490             else if isDirMarker `T.isPrefixOf` cleanOut
  491               then pure (Left FeIsDirectory)
  492               else do
  493                 let bs = TE.encodeUtf8 cleanOut
  494                     truncated = runnerTruncated || BS.length bs >= ceilingBytes
  495                     isBinary = BS.elem 0 bs
  496                     content = if isBinary then "" else cleanOut
  497                     linesRead = if isBinary then 0 else length (T.lines content)
  498                 pure (Right FileContent
  499                   { fcPath = T.pack path
  500                   , fcContent = content
  501                   , fcTruncated = truncated
  502                   , fcLinesRead = linesRead
  503                   , fcIsBinary = isBinary
  504                   })

  506 -- | Read a remote file as raw ByteString via SSH @head -c <ceil>
  507 -- -- "$p"@. Uses runRemoteBytes (raw ByteString transport -- no
  508 -- UTF-8 decode) so binary files are served without corruption.
  509 remoteReadFileBytes
  510   :: SshConfig -> RemoteRunner -> Int -> FilePath
  511   -> IO (Either FileExplorerErr (ByteString, Bool))
  512 remoteReadFileBytes sshCfg runner ceilingBytes origPath = do
  513   let path = collapsePath origPath
  514       cmdText = remoteReadCmd path ceilingBytes
  515       notFoundBs = TE.encodeUtf8 (notFoundMarker <> "\n")
  516       isDirBs = TE.encodeUtf8 (isDirMarker <> "\n")
  517   case mkShellCommand cmdText of
  518     Left _ -> pure (Left (FeExec "invalid command (NUL in path?)"))
  519     Right _ -> do
  520       res <- runRemoteBytes runner (sshExecArgv sshCfg cmdText)
  521       case res of
  522         Left e -> pure (Left (mapRemoteExecError sshCfg e))
  523         Right (bs, transportTruncated) -> do
  524           let truncated = transportTruncated || BS.length bs >= ceilingBytes
  525           if notFoundBs `BS.isPrefixOf` bs
  526             then pure (Left FeNotFound)
  527             else if isDirBs `BS.isPrefixOf` bs
  528               then pure (Left FeIsDirectory)
  529               else pure (Right (bs, truncated))
  530 -- | Resolve the home directory on the remote machine via SSH @echo
  531 -- $HOME@. The result is cached in an 'IORef' per handle instance.
  532 remoteResolveHome
  533   :: SshConfig -> RemoteRunner -> IORef (Maybe FilePath)
  534   -> IO (Either FileExplorerErr FilePath)
  535 remoteResolveHome sshCfg runner ref = do
  536   cached <- readIORef ref
  537   case cached of
  538     Just h -> pure (Right h)
  539     Nothing -> case mkShellCommand "echo $HOME" of
  540       Left _ -> pure (Left (FeExec "invalid command"))
  541       Right cmd -> do
  542         res <- runRemoteShell runner sshCfg cmd
  543         case res of
  544           Left e -> pure (Left (mapRemoteExecError sshCfg e))
  545           Right out -> do
  546             let home = T.unpack (T.strip out)
  547             if T.null (T.strip out)
  548               then pure (Left (FeIo "empty home directory"))
  549               else do
  550                 writeIORef ref (Just home)
  551                 pure (Right home)

  557 -- | Build the remote listing command. The path is 'shellQuote'-d and
  558 -- assigned to a shell variable (@p=...@) before use in compound
  559 -- commands. The @ls@ uses @--@ before the path argument. Entry type is
  560 -- emitted as a prefix tag: @S|@ (symlink), @D|@ (directory), @F|@ (file,
  561 -- with size and mtime), @O|@ (other). Error sentinels
  562 -- (@___SEAL_NOTFOUND___@, @___SEAL_NOTDIR___@) distinguish missing paths
  563 -- from non-directory paths.
  564 remoteListCmd :: FilePath -> Text
  565 remoteListCmd path =
  566   T.pack $ unwords
  567     [ "p=" <> shellQuote path <> ";"
  568     , "if [ ! -e \"$p\" ]; then echo \"" <> T.unpack notFoundMarker <> "\";"
  569     , "elif [ ! -d \"$p\" ]; then echo \"" <> T.unpack notDirMarker <> "\";"
  570     , "else ls -1A -- \"$p\" 2>/dev/null | while IFS= read -r name; do"
  571     , "full=\"$p/$name\";"
  572     , "if [ -L \"$full\" ]; then echo \"S|$name\";"
  573     , "elif [ -d \"$full\" ]; then echo \"D|$name\";"
  574     , "elif [ -f \"$full\" ]; then"
  575     , "sz=$(stat -c %s -- \"$full\" 2>/dev/null || stat -f%z \"$full\" 2>/dev/null || echo 0);"
  576     , "mt=$(stat -c %Y -- \"$full\" 2>/dev/null || stat -f%m \"$full\" 2>/dev/null || echo 0);"
  577     , "echo \"F|$name|$sz|$mt\";"
  578     , "else echo \"O|$name\"; fi;"
  579     , "done;"
  580     , "fi"
  581     ]

  583 -- | Build the remote read command. The path is 'shellQuote'-d and
  584 -- assigned to a shell variable, then @head -c <ceil> -- "$p"@ reads the
  585 -- first @ceilBytes@ bytes. Error sentinels distinguish missing paths
  586 -- (@___SEAL_NOTFOUND___@) from directories (@___SEAL_ISDIR___@).
  587 remoteReadCmd :: FilePath -> Int -> Text
  588 remoteReadCmd path ceilingBytes =
  589   T.pack $ unwords
  590     [ "p=" <> shellQuote path <> ";"
  591     , "if [ ! -e \"$p\" ]; then echo \"" <> T.unpack notFoundMarker <> "\";"
  592     , "elif [ -d \"$p\" ]; then echo \"" <> T.unpack isDirMarker <> "\";"
  593     , "else head -c " <> show ceilingBytes <> " -- \"$p\";"
  594     , "fi"
  595     ]

  601 -- | The per-directory framing tag emitted by 'remoteListDirsCmd'. Chosen so it
  602 -- cannot collide with an entry line (which starts with D\/F\/S\/O\/|).
  603 dirHeaderTag :: Text
  604 dirHeaderTag = "%%D|"
  605 
  606 -- | Build ONE shell command that lists N directories, framing each with
  607 -- 'dirHeaderTag' so the response can be split. Every path is shell-quoted
  608 -- (never interpolated raw), and the per-directory body is identical to
  609 -- 'remoteListCmd' so the two arms agree.
  610 --
  611 -- Output shape:
  612 --
  613 -- > %%D|/a
  614 -- > D|sub
  615 -- > F|f.txt|6|1700000000
  616 -- > %%D|/b
  617 -- > F|g.txt|7|1700000000
  618 remoteListDirsCmd :: [FilePath] -> Text
  619 remoteListDirsCmd paths = T.intercalate "; " (map one paths)
  620   where
  621     one :: FilePath -> Text
  622     one path = T.pack $ unwords
  623       [ "p=" <> shellQuote path <> ";"
  624       , "echo " <> shellQuote (T.unpack dirHeaderTag <> path) <> ";"
  625       , "if [ ! -e \"$p\" ]; then echo \"" <> T.unpack notFoundMarker <> "\";"
  626       , "elif [ ! -d \"$p\" ]; then echo \"" <> T.unpack notDirMarker <> "\";"
  627       , "else ls -1A -- \"$p\" 2>/dev/null | while IFS= read -r name; do"
  628       , "full=\"$p/$name\";"
  629       , "if [ -L \"$full\" ]; then echo \"S|$name\";"
  630       , "elif [ -d \"$full\" ]; then echo \"D|$name\";"
  631       , "elif [ -f \"$full\" ]; then"
  632       , "sz=$(stat -c %s -- \"$full\" 2>/dev/null || stat -f%z \"$full\" 2>/dev/null || echo 0);"
  633       , "mt=$(stat -c %Y -- \"$full\" 2>/dev/null || stat -f%m \"$full\" 2>/dev/null || echo 0);"
  634       , "echo \"F|$name|$sz|$mt\";"
  635       , "else echo \"O|$name\"; fi;"
  636       , "done;"
  637       , "fi"
  638       ]

  640 -- | Split a batched listing response into (path, body) pairs, in arrival
  641 -- order. A directory whose body was cut off by transport truncation is
  642 -- dropped entirely, so a caller never sees a partially-listed directory.
  643 parseBatchedListing :: Text -> [(Text, Text)]
  644 parseBatchedListing raw =
  645   let (clean, _truncated) = stripTruncationMarker raw
  646   in go clean
  647   where
  648     -- Accumulate (path, body-lines) blocks, then reverse each body.
  649     go txt =
  650       let ls = T.lines txt
  651           (blocks, curPath, curLines) = foldl step ([], Nothing, []) ls
  652           finish mpath mlines acc =
  653             case mpath of
  654               Nothing -> acc
  655               Just p  -> acc ++ [(p, T.unlines (reverse mlines))]
  656       in finish curPath curLines blocks
  657 
  658     step (acc, mpath, mlines) line =
  659       case T.stripPrefix dirHeaderTag line of
  660         Just p ->
  661           let acc' = case mpath of
  662                 Nothing -> acc
  663                 Just cur -> acc ++ [(cur, T.unlines (reverse mlines))]
  664           in (acc', Just p, [])
  665         Nothing -> (acc, mpath, line : mlines)
  666 
  667 -- | List several remote directories in ONE SSH round trip. Directories that
  668 -- did not arrive complete (transport truncation) are omitted from the result:
  669 -- returning a partial listing would be wrong, and the caller re-requests the
  670 -- remainder individually.
  671 remoteListDirectories
  672   :: SshConfig -> RemoteRunner -> Int -> [FilePath]
  673   -> IO (Either FileExplorerErr [FileListing])
  674 remoteListDirectories _sshCfg _runner _ceiling [] = pure (Right [])
  675 remoteListDirectories sshCfg runner _ceiling paths = do
  676   let collapsed = map collapsePath paths
  677       cmdText = remoteListDirsCmd collapsed
  678   case mkShellCommand cmdText of
  679     Left _ -> pure (Left (FeExec "invalid command (NUL in path?)"))
  680     Right cmd -> do
  681       res <- runRemoteShell runner sshCfg cmd
  682       case res of
  683         Left e -> pure (Left (mapRemoteExecError sshCfg e))
  684         Right out -> do
  685           let pairs = parseBatchedListing out
  686               toListing (p, body) =
  687                 let entries = sortBy
  688                       (comparing (\e -> (not (feIsDir e), feName e)))
  689                       (parseRemoteListing body)
  690                 in FileListing
  691                      { flPath = p
  692                      , flEntries = entries
  693                      , flTruncated = False
  694                      , flWarning = checkBlocked (T.unpack p)
  695                      }
  696           pure (Right (map toListing pairs))

  698 -- | Parse the remote listing output into 'FileEntry' list. Each line is
  699 -- @S|name@, @D|name@, @F|name|size|mtime@, or @O|name@. Malformed lines
  700 -- are silently skipped. Entry names containing @|@ are misparsed (a
  701 -- documented limitation — the QuickCheck property guards against this).
  702 parseRemoteListing :: Text -> [FileEntry]
  703 parseRemoteListing output =
  704   mapMaybe parseEntry (T.lines output)
  705   where
  706     parseEntry :: Text -> Maybe FileEntry
  707     parseEntry line
  708       | T.null line = Nothing
  709       | otherwise = case T.splitOn "|" line of
  710           ["S", name] -> Just FileEntry
  711             { feName = name
  712             , feIsDir = False
  713             , feIsSymlink = True
  714             , feSize = Nothing
  715             , feMtime = Nothing
  716             }
  717           ["D", name] -> Just FileEntry
  718             { feName = name
  719             , feIsDir = True
  720             , feIsSymlink = False
  721             , feSize = Nothing
  722             , feMtime = Nothing
  723             }
  724           ["F", name, sz, mt] -> Just FileEntry
  725             { feName = name
  726             , feIsDir = False
  727             , feIsSymlink = False
  728             , feSize = parseInteger sz
  729             , feMtime = parseMtime mt
  730             }
  731           ["O", name] -> Just FileEntry
  732             { feName = name
  733             , feIsDir = False
  734             , feIsSymlink = False
  735             , feSize = Nothing
  736             , feMtime = Nothing
  737             }
  738           _ -> Nothing
  739 
  740 -- | Strip the 'readBounded' truncation marker from the output. Returns
  741 -- the clean content and a 'Bool' indicating whether truncation was
  742 -- detected.
  743 stripTruncationMarker :: Text -> (Text, Bool)
  744 stripTruncationMarker txt =
  745   case T.breakOn truncationMarker txt of
  746     (content, rest)
  747       | T.null rest -> (txt, False)
  748       | otherwise -> (T.stripEnd content, True)
  749 
  750 -- | Parse a non-negative integer from 'Text'. Total — returns
  751 -- 'Nothing' on any non-numeric input (including empty).
  752 parseInteger :: Text -> Maybe Integer
  753 parseInteger txt =
  754   let t = T.dropWhile isSpace (T.strip txt)
  755   in if not (T.null t) && T.all isDigit t
  756        then readMaybe (T.unpack t)
  757        else Nothing
  758 
  759 -- | Parse a Unix epoch timestamp (seconds) into 'UTCTime'. Returns
  760 -- 'Nothing' for non-numeric input or @0@ (which means @stat@ failed and
  761 -- fell back to @echo 0@).
  762 parseMtime :: Text -> Maybe UTCTime
  763 parseMtime txt =
  764   case parseInteger txt of
  765     Just n | n > 0 -> Just (posixSecondsToUTCTime (fromIntegral n))
  766     _ -> Nothing

   69 -- | The real 'LocalExecHandle': wires 'System.Process' behind the two IO
   70 -- actions. The 'WorkspaceRoot' anchors cwd confinement for the shell exec.
   71 mkLocalExecHandle :: WorkspaceRoot -> LocalExecHandle
   72 mkLocalExecHandle wsRoot = LocalExecHandle
   73   { lehExecShell = \cmd mCwd -> do
   74       let argv = ["/bin/sh", "-c", T.unpack (textShellCommand cmd)]
   75       case mCwd of
   76         Nothing -> runShell argv Nothing
   77         Just rp -> do
   78           e <- mkSafePath wsRoot (T.unpack (getRemotePath rp))
   79           case e of
   80             Left _err -> pure (Left ExecNotImplemented)
   81             Right sp  -> runShell argv (Just (getSafePath sp))
   82   , lehExecBin = \bin bargs mCwd ->
   83       let binName  = T.unpack (textBinName bin)
   84           argTexts = map (T.unpack . textBinArg) bargs
   85           argv     = binName : argTexts
   86       in case mCwd of
   87            Nothing  -> runProgram argv Nothing
   88            Just rp  -> do
   89              e <- mkSafePath wsRoot (T.unpack (getRemotePath rp))
   90              case e of
   91                Left _err -> pure (Left ExecNotImplemented)
   92                Right sp  -> runProgram argv (Just (getSafePath sp))
   93   }

   95 -- | A 'LocalExecHandle' built from explicit IO action functions — the form
   96 -- the opcode tests use (mirrors the 'TmuxRunner' fake pattern). Real
   97 -- callers use 'mkLocalExecHandle'.
   98 mkLocalExecHandleFromFns
   99   :: (ShellCommand -> Maybe RemotePath -> IO (Either ExecError Text))
  100   -> (BinName -> [BinArg] -> Maybe RemotePath -> IO (Either ExecError Text))
  101   -> LocalExecHandle
  102 mkLocalExecHandleFromFns shellFn binFn = LocalExecHandle
  103   { lehExecShell = shellFn
  104   , lehExecBin = binFn

  107 -- | Run a shell command via @/bin/sh -c@. The shell itself always launches
  108 -- (it's at a fixed path), so a non-zero exit — including 127 ("command not
  109 -- found" inside the shell) — is a normal command failure, returned via
  110 -- 'Right' with the output + exit code annotation. Only an IOError (the shell
  111 -- binary itself couldn't launch) becomes 'Left ExecNotImplemented'.
  112 runShell :: [String] -> Maybe String -> IO (Either ExecError Text)
  113 runShell = runFixedArgv False
  114 
  115 -- | Run a named binary (resolved on PATH or by path). A 127 exit
  116 -- here means the binary itself is not on PATH, so it's mapped to
  117 -- 'Left ExecNotImplemented' (the executor is not available). Other non-zero
  118 -- exits are normal failures, returned via 'Right' with the output + exit code.
  119 runProgram :: [String] -> Maybe String -> IO (Either ExecError Text)
  120 runProgram = runFixedArgv True
  121 
  122 -- | Run a fixed-argv program, capturing stdout and stderr as Text. An optional
  123 -- cwd (already 'SafePath'-validated). When @treat127AsMissing@ is True, a 127
  124 -- exit is mapped to 'Left ExecNotImplemented' (the binary is not on PATH).
  125 -- Otherwise (shell mode), 127 is a normal command-not-found failure, returned
  126 -- via 'Right' with the exit code annotation. Any IOError becomes 'Left
  127 -- ExecNotImplemented'.
  128 --
  129 -- Uses 'withManagedProcess' (process-group spawning + kill-on-cleanup) and
  130 -- 'readBounded' (bounded output capture, default 50KB).
  131 runFixedArgv :: Bool -> [String] -> Maybe String -> IO (Either ExecError Text)
  132 runFixedArgv treat127AsMissing argv mCwd = do
  133   let (program, args) = case argv of
  134         (p : as) -> (p, as)
  135         []       -> error "runFixedArgv: empty argv (unreachable)"
  136       cp = (proc program args)
  137              { std_in = NoStream, std_out = CreatePipe, std_err = CreatePipe
  138              , cwd = mCwd
  139              , create_group = True
  140              }
  141       -- ^ @create_group = True@ puts the child in its own POSIX process
  142       -- group so 'withManagedProcess' can kill the whole group on cleanup
  143       -- (SIGTERM → grace → SIGKILL). This prevents orphans when the
  144       -- dispatch wrapper cancels the Haskell worker thread.
  145   res <- try @IOError
  146          (withManagedProcess cp $ \ph _mIn hOut hErr -> do
  147             out <- readBounded hOut defaultMaxOutputBytes
  148             err <- readBounded hErr defaultMaxOutputBytes
  149             ec  <- waitForProcess ph
  150             pure (ec, out, err))
  151   case res of
  152     Left _ioErr                     -> pure (Left ExecNotImplemented)  -- binary missing/launch fail
  153     Right (ExitSuccess, out, _)     -> pure (Right out)
  154     Right (ExitFailure 127, _, _)
  155       | treat127AsMissing            -> pure (Left ExecNotImplemented)  -- binary not on PATH
  156     Right (ExitFailure n, out, err)  -> pure (Right (formatExitResult n out err))
  157 
  158 -- | Format a non-zero exit result for the tool-call consumer. Combines stdout
  159 -- and stderr (if non-empty) and annotates the exit code so the frontend can
  160 -- surface it. The result is returned via 'Right' (not an 'ExecError') so the
  161 -- dispatcher records @is_error = False@ — the command ran successfully, it just
  162 -- returned a non-zero exit code. The frontend treats the exit code annotation
  163 -- as the success/failure signal.
  164 formatExitResult :: Int -> Text -> Text -> Text
  165 formatExitResult n out err =
  166   let parts = [ t | t <- [out, err], not (T.null (T.strip t)) ]
  167       body  = if null parts then "" else T.intercalate "\n" parts
  168   in body <> "\n[exit code: " <> T.pack (show n) <> "]"

  170 -- | Spawn a process in its own process group (@create_group = True@) and run
  171 -- an action with the process handle + stdin/stdout/stderr handles. On cleanup
  172 -- (normal exit OR async-exception cancellation), kill the whole process
  173 -- group: SIGTERM → grace period → SIGKILL. Then close handles and reap the
  174 -- zombie with a BOUNDED wait (non-blocking 'getProcessExitCode' poll with a
  175 -- 1s deadline) so cleanup never hangs.
  176 --
  177 -- When the dispatch wrapper
  178 -- ('Seal.Tools.Exec.Timeout.runWithTimeoutAbortRetry') cancels the Haskell
  179 -- worker thread via 'async'\'s 'cancel', the bracket cleanup runs (bracket
  180 -- is mask-on-cleanup) and kills the OS process group — no orphans. This is
  181 -- the key mechanism: killing the Haskell thread cascades to killing the OS
  182 -- process group.
  183 --
  184 -- The stdin handle is 'Maybe Handle' — 'Nothing' if @std_in = NoStream@,
  185 -- 'Just' if @std_in = CreatePipe@. The caller is responsible for writing
  186 -- the payload and closing the stdin handle (so the child sees EOF).
  187 withManagedProcess
  188   :: CreateProcess
  189   -> (ProcessHandle -> Maybe Handle -> Handle -> Handle -> IO a)
  190   -> IO a
  191 withManagedProcess cp action = bracket create cleanup runAction
  192   where
  193     -- The cp MUST have create_group=True (asserted at the call site in
  194     -- runFixedArgv; we re-set it here too for safety).
  195     cp' = cp { create_group = True }
  196     create = do
  197       (mIn, mOut, mErr, ph) <- createProcess cp'
  198       pidRef <- newIORef Nothing
  199       -- Capture the PID while the process is open.
  200       withProcessHandle ph $ \case
  201         OpenHandle pid -> writeIORef pidRef (Just (fromIntegral pid :: Int))
  202         _              -> pure ()
  203       mPid <- readIORef pidRef
  204       case (mOut, mErr) of
  205         (Just hOut, Just hErr) -> pure (ph, mIn, hOut, hErr, mPid)
  206         _                      -> error "withManagedProcess: pipe creation failed (unreachable)"
  207     cleanup (ph, mIn, hOut, hErr, mPid) = do
  208       -- Fast path: if the child ALREADY exited (the common success path),
  209       -- there is nothing to kill. Killing here would (a) tax every
  210       -- successful exec with the full SIGTERM→SIGKILL grace wait and
  211       -- (b) signal the backgrounded @ssh@ ControlMaster that shares the
  212       -- child's process group on the remote arm — destroying connection
  213       -- multiplexing (every subsequent remote op would pay a fresh
  214       -- TCP+key-exchange+auth handshake).
  215       mec <- getProcessExitCode ph
  216       case mec of
  217         Just _  -> pure ()
  218         Nothing -> case mPid of
  219           Nothing   -> pure ()
  220           Just pid  -> killProcessGroup ph pid defaultKillGraceMicros
  221       -- Close the stdin (if open), stdout, stderr handles.
  222       case mIn of
  223         Nothing -> pure ()
  224         Just hIn -> hClose hIn `catch` \(_ :: IOException) -> pure ()
  225       hClose hOut `catch` \(_ :: IOException) -> pure ()
  226       hClose hErr `catch` \(_ :: IOException) -> pure ()
  227       -- Reap the zombie with a BOUNDED wait (non-blocking poll, 1s deadline).
  228       -- If the child is stuck in an uninterruptible state, don't hang
  229       -- cleanup forever — the OS init process reaps it eventually.
  230       reapBounded ph
  231     -- Run the action with (ph, mIn, hOut, hErr).
  232     runAction (ph, mIn, hOut, hErr, _) = action ph mIn hOut hErr
  233 
  234 -- | Reap a zombie with a bounded wait. Polls 'getProcessExitCode' every
  235 -- 10ms for up to 1s; if the child hasn't exited, gives up (the OS init
  236 -- process reaps it eventually). Never hangs.
  237 reapBounded :: ProcessHandle -> IO ()
  238 reapBounded ph = go (100 :: Int)  -- 100 polls × 10ms = 1s deadline
  239   where
  240     go :: Int -> IO ()
  241     go 0 = pure ()  -- give up; zombie reaped by init eventually
  242     go n = do
  243       mec <- getProcessExitCode ph
  244       case mec of
  245         Just _  -> pure ()  -- reaped
  246         Nothing -> do
  247           threadDelay 10_000  -- 10ms
  248           go (n - 1)

  250 -- | Kill a process group: SIGTERM → bounded grace → SIGKILL.
  251 -- POSIX: 'signalProcessGroup' sends a signal to every process in the group.
  252 -- The @graceMicros@ is the SIGTERM→SIGKILL grace period. Because the child
  253 -- was spawned with @create_group = True@, its PGID equals its PID.
  254 --
  255 -- The grace period POLLS for the child's exit (every 10ms) instead of
  256 -- sleeping the full duration: a child that dies on SIGTERM — the normal
  257 -- case — lets cleanup return within ~10ms, and only a stubborn child pays
  258 -- the full grace before SIGKILL. The trailing SIGKILL is unconditional
  259 -- (idempotent on already-dead processes) so no group member can outlive
  260 -- the kill on the cancel path.
  261 killProcessGroup :: ProcessHandle -> Int -> Int -> IO ()
  262 killProcessGroup ph pid graceMicros = do
  263   -- The child is in its own group (create_group=True), so its PGID == its PID.
  264   -- 'signalProcessGroup' takes a ProcessGroupID; a CPid and ProcessGroupID
  265   -- are both newtype'd Int, so we coerce via fromIntegral.
  266   let pgid = fromIntegral pid :: Posix.ProcessGroupID
  267   -- SIGTERM the whole group.
  268   _ <- try @IOException (Posix.signalProcessGroup Posix.sigTERM pgid)
  269   -- Poll for exit within the grace budget (no blind sleep).
  270   _ <- pollExitBounded ph graceMicros
  271   -- SIGKILL the whole group (idempotent on already-dead processes).
  272   _ <- try @IOException (Posix.signalProcessGroup Posix.sigKILL pgid)
  273   pure ()
  274 
  275 -- | Poll 'getProcessExitCode' every 10ms up to @budgetMicros@. Returns
  276 -- 'True' if the child exited within budget.
  277 pollExitBounded :: ProcessHandle -> Int -> IO Bool
  278 pollExitBounded ph budgetMicros = go (budgetMicros `div` 10_000)
  279   where
  280     go :: Int -> IO Bool
  281     go 0 = pure False  -- out of budget; caller escalates to SIGKILL
  282     go n = do
  283       mec <- getProcessExitCode ph
  284       case mec of
  285         Just _  -> pure True
  286         Nothing -> do
  287           threadDelay 10_000
  288           go (n - 1)

  290 -- | Read at most N bytes from a handle, returning the content. If the
  291 -- stream has more than N bytes remaining, a truncation marker is appended:
  292 -- @[output truncated at N bytes — redirect to a file and FILE_READ with
  293 -- pagination for full output]@. The marker hints at the recovery workflow
  294 -- so the model learns the pattern.
  295 --
  296 -- After reading the bounded amount (and detecting non-EOF), the handle is
  297 -- CLOSED — this prevents a deadlock where the child blocks on a full pipe
  298 -- buffer while 'waitForProcess' waits for the child. Closing the handle
  299 -- causes the child's writes to fail (SIGPIPE/EPIPE), the child exits, and
  300 -- 'waitForProcess' returns. The truncation marker is the signal that output
  301 -- was lost.
  302 readBounded :: Handle -> Int -> IO Text
  303 readBounded h maxBytes = do
  304   bs <- BS.hGet h maxBytes
  305   eof <- hIsEOF h
  306   let content = TE.decodeUtf8 bs
  307   if eof
  308     then pure content
  309     else do
  310       -- There's more output. Close the handle so the child's writes fail
  311       -- (SIGPIPE), the child exits, and waitForProcess doesn't deadlock.
  312       hClose h
  313       pure (content <> "\n[output truncated at " <> T.pack (show maxBytes)
  314                      <> " bytes — redirect to a file and FILE_READ with pagination for full output]")

   28 -- | The evolutionary-store backends + the in-process agent runtime, created
   29 -- once at startup and shared between the command specs (which read them via
   30 -- @\/skill@ \/ @\/agent@) and the ISA opcodes (which mutate them). The
   31 -- skills and agent-def stores are disk-backed (Markdown files under
   32 -- @config\/@); disk is canonical and git is the versioning + audit layer.
   33 -- The memory store is file-based under @\<sealHome\>\/memory\/@ with
   34 -- @active\/@ and @archived\/@ directories (no git auto-commit — immutability
   35 -- is enforced by the write-once + archive model). The embedding backend is
   36 -- 'nullEmbeddingBackend' (engram wiring is deferred). The agent runtime is
   37 -- an in-process STM registry (lifecycle only — not persisted). The
   38 -- delegation knobs (config, pause flag, parent-activity cell) are
   39 -- process-global so AGENT_START calls across all channels share one pause /
   40 -- heartbeat state.
   41 data Backends = Backends
   42   { bMemory    :: Mem.MemoryStore
   43   , bEmbedding :: Emb.EmbeddingBackend
   44   , bSkills    :: Skill.SkillBackend
   45   , bAgentDefs :: Def.AgentDefBackend
   46   , bRuntime   :: AgentRuntime
   47   , bDelegationConfig :: IO DelegationConfig
   48     -- ^ Reload the [delegation] config per AGENT_START call (so config
   49     -- changes take effect without a restart). The IO action reads
   50     -- @config.toml@ and returns the resolved 'DelegationConfig'.
   51   , bSpawnPauseFlag :: SpawnPauseFlag
   52     -- ^ Process-global spawn-pause flag (operator can freeze new fan-out).
   53   , bParentActivity :: ParentActivity
   54     -- ^ Process-global parent-activity cell (heartbeat target).
   55   }
   56 
   57 -- | Construct the disk-backed backends for the given config repo and seal
   58 -- paths. Skills and agent defs read their directories on demand (no startup
   59 -- materialization needed — disk is canonical, so @\/skill list@ etc. just
   60 -- enumerate the dir). The memory store is constructed from
   61 -- @\<sealHome\>\/memory\/@ (NOT @\<configRoot\>\/memory\/@) — memory is a
   62 -- separate store with its own immutability guarantees. The embedding backend
   63 -- The embedding backend is passed in (resolved from config at the call
   64 -- site) so this function doesn't depend on 'RuntimeConfig' or
   65 -- 'Seal.Memory.EngramBackend'. The delegation knobs are process-global;
   66 -- the config is re-read per AGENT_START call so config changes take effect
   67 -- without a restart.
   68 newBackends :: SealPaths -> ConfigRepo -> Emb.EmbeddingBackend -> IO Backends
   69 newBackends paths repo embedding = do
   70   let skillsDir    = spConfig paths </> "skills"
   71       agentsDir    = spConfig paths </> "agents"
   72       memoryDir    = spHome paths </> "memory"
   73   rt          <- newAgentRuntime
   74   pauseFlag   <- newSpawnPauseFlag
   75   parentAct   <- newParentActivity
   76   memStore    <- Mem.fileMemoryStore memoryDir
   77   skills      <- Skill.unionSkillBackend <$> Skill.markdownSkillBackend skillsDir repo
   78   agentDefs   <- Def.markdownAgentDefBackend agentsDir repo
   79   pure (Backends
   80     { bMemory = memStore
   81     , bEmbedding = embedding
   82     , bSkills = skills
   83     , bAgentDefs = agentDefs
   84     , bRuntime = rt
   85     , bDelegationConfig = pure defaultDelegationConfig
   86     , bSpawnPauseFlag = pauseFlag
   87     , bParentActivity = parentAct
   88     })

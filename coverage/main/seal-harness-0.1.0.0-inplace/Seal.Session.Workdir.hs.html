   47 -- | The error type for workdir lifecycle operations.
   48 data WorkdirError
   49   = WdMkdirFailed FilePath Text      -- ^ local mkdir failed (path, reason)
   50   | WdRemoteMkdirFailed Text          -- ^ remote SSH mkdir failed (reason)
   51   | WdInvalidSessionId Text           -- ^ SessionId failed validation
   52   | WdNotUnderWorkdirsRoot FilePath    -- ^ cleanup path escaped workdirsRoot
   53   deriving stock (Eq, Show)
   54 
   55 -- | Create the per-session workdir at @<cache>/workdirs/<sid>@.
   56 -- Idempotent: if the workdir already exists, it is reused (NOT cleared —
   57 -- the operator may want to resume or inspect). Returns the workdir path
   58 -- on success. Fails closed on permission errors (does NOT fall back to
   59 -- the cwd — that would reintroduce the cross-session clobber bug).
   60 ensureSessionWorkdir :: SealPaths -> SessionId -> IO (Either WorkdirError FilePath)
   61 ensureSessionWorkdir paths sid = do
   62   let sidText = sessionIdText sid
   63   if not (isValidSessionId sidText)
   64     then pure (Left (WdInvalidSessionId sidText))
   65     else do
   66       let wdPath = sessionWorkdir paths sid
   67       eResult <- try (createDirectoryIfMissing True wdPath) :: IO (Either IOException ())
   68       pure $ case eResult of
   69         Left ioErr -> Left (WdMkdirFailed wdPath (T.pack (show ioErr)))
   70         Right _   -> Right wdPath
   71 
   72 -- | Remove the per-session workdir. Asserts the path is under
   73 -- 'workdirsRoot' (canonicalize + prefix check — defeats symlink swap).
   74 -- Idempotent: no error if the workdir is already gone. Returns
   75 -- 'Either WorkdirError ()' so cleanup failures are not silently
   76 -- swallowed.
   77 cleanupSessionWorkdir :: SealPaths -> SessionId -> IO (Either WorkdirError ())
   78 cleanupSessionWorkdir paths sid = do
   79   let sidText = sessionIdText sid
   80   if not (isValidSessionId sidText)
   81     then pure (Left (WdInvalidSessionId sidText))
   82     else do
   83       let wdPath = sessionWorkdir paths sid
   84           wdRoot = workdirsRoot paths
   85       exists <- doesDirectoryExist wdPath
   86       if not exists
   87         then pure (Right ())  -- already gone — idempotent
   88         else do
   89           -- Defense-in-depth: canonicalize and verify the path is under
   90           -- workdirsRoot before rm -rf (defeats symlink swap).
   91           canonWd <- canonicalizePath wdPath
   92           canonRoot <- canonicalizePath wdRoot
   93           if not (splitDirectories canonRoot `isPrefixOf` splitDirectories canonWd)
   94             then pure (Left (WdNotUnderWorkdirsRoot canonWd))
   95             else do
   96               eResult <- try (removeDirectoryRecursive canonWd) :: IO (Either IOException ())
   97               pure $ case eResult of
   98                 Left ioErr -> Left (WdMkdirFailed canonWd (T.pack (show ioErr)))
   99                 Right _   -> Right ()

  105 -- | Compute the remote per-session workdir path:
  106 -- @<scWorkspace>/workdirs/<sid>@. Pure — no IO, no SSH.
  107 remoteSessionWorkdirPath :: SshConfig -> SessionId -> Text
  108 remoteSessionWorkdirPath sshCfg sid =
  109   getRemotePath (scWorkspace sshCfg) <> "/workdirs/" <> sessionIdText sid
  110 
  111 -- | Create the per-session workdir on the REMOTE machine via SSH
  112 -- @mkdir -p@. Validates the 'SessionId' before the SSH call. Idempotent
  113 -- (mkdir -p is a no-op if the dir exists). Returns the remote workdir
  114 -- path (as 'Text') on success. Fails closed on SSH errors (does NOT
  115 -- fall back to the shared scWorkspace — that would reintroduce the
  116 -- cross-session clobber bug).
  117 ensureRemoteSessionWorkdir
  118   :: SshConfig -> RemoteRunner -> SessionId
  119   -> IO (Either WorkdirError Text)
  120 ensureRemoteSessionWorkdir sshCfg runner sid = do
  121   let sidText = sessionIdText sid
  122   if not (isValidSessionId sidText)
  123     then pure (Left (WdInvalidSessionId sidText))
  124     else do
  125       let remoteWd = remoteSessionWorkdirPath sshCfg sid
  126           cmdText = "mkdir -p '" <> T.unpack remoteWd <> "'"
  127       case mkShellCommand (T.pack cmdText) of
  128         Left _err -> pure (Left (WdRemoteMkdirFailed "invalid mkdir command"))
  129         Right cmd -> do
  130           res <- runRemoteShell runner sshCfg cmd
  131           pure $ case res of
  132             Left e   -> Left (WdRemoteMkdirFailed (T.pack (show e)))
  133             Right _  -> Right remoteWd

  139 -- | The fail-closed 'WorkspaceRoot' returned when workdir creation fails.
  140 -- Matches the convention used at the wiring sites (e.g. @Cli.hs@).
  141 failClosedRoot :: WorkspaceRoot
  142 failClosedRoot = WorkspaceRoot "/nonexistent-workdir-fail-closed"

  144 -- | The per-session execution bundle: the 'UIOEnv' (carrying 'UntrustedIO'
  145 -- + 'CloneDeps'), the 'WorkdirFs' for discovery, and the shared
  146 -- 'WorkspaceRoot' used by UIO + WorkdirFs + the ISA registry. Constructed
  147 -- by 'mkSessionExec' from the 'SecurityConfig'; the single 'RemoteRunner'
  148 -- is shared between 'seUIOEnv' (UIO's 'UntrustedIO') and 'seWorkdirFs'
  149 -- (via 'runUIOWithEnv') — one SSH connection, not two. On ANY workdir-
  150 -- creation failure, ALL three fields are stubs (fail-closed) — never a
  151 -- mix of real + stub.
  152 data SessionExec = SessionExec
  153   { seUIOEnv        :: UIOEnv
  154   , seWorkdirFs     :: WorkdirFs
  155   , seWorkspaceRoot :: WorkspaceRoot
  156   }
  157 
  158 -- | Construct the per-session 'SessionExec' from the 'SecurityConfig'.
  159 -- Mirrors 'mkSessionUntrustedIO' with the same three cases (local,
  160 -- remote + configured, remote + absent/incomplete) but additionally
  161 -- builds the 'WorkdirFs' and the shared 'WorkspaceRoot'. The
  162 -- 'RemoteRunner' is passed explicitly so tests can inject
  163 -- 'mkFakeRemoteRunnerRecording' (no real SSH); production wiring passes
  164 -- 'mkRealRemoteRunner'. The single runner is shared between 'seUIOEnv'
  165 -- and 'seWorkdirFs'.
  166 --
  167 -- When @mAdoption = Just hka@ and the remote workdir bootstrap fails with
  168 -- 'ExecHostKeyUnknown' (the host key is not in @known_hosts@), the harness
  169 -- attempts TOFU adoption: probes the host via @ssh-keyscan@, asks the human
  170 -- for confirmation, and appends the key on approval. On approval, the
  171 -- workdir bootstrap is retried. On rejection or when @mAdoption = Nothing@,
  172 -- the session fail-closes with a descriptive error (not the content-free
  173 -- 'ExecNotImplemented').
  174 --
  175 -- On ANY workdir-creation failure, returns a 'SessionExec' with both
  176 -- handles as stubs and 'seWorkspaceRoot' as 'failClosedRoot' — never mixed.
  177 mkSessionExec
  178   :: SealPaths -> SecurityConfig -> SessionId
  179   -> CloneDeps -> RemoteRunner -> Maybe HostKeyAdoption
  180   -> IO SessionExec
  181 mkSessionExec paths secCfg sid cloneDeps runner mAdoption =
  182   case untrustedExecConfigFromSecurity secCfg of
  183     Nothing -> do
  184       eWd <- ensureSessionWorkdir paths sid
  185       case eWd of
  186         Right wd ->
  187           let wsRoot  = WorkspaceRoot wd
  188               uio     = mkLocalUntrustedIO wsRoot
  189               uioEnv  = mkTestUIOEnv uio cloneDeps
  190               wfs     = mkLocalWorkdirFs wsRoot maxScanBytes
  191           in pure SessionExec
  192                { seUIOEnv        = uioEnv
  193                , seWorkdirFs     = wfs
  194                , seWorkspaceRoot = wsRoot
  195                }
  196         Left err -> pure (failClosedSessionExecWithReason cloneDeps
  197                             ("local workdir creation failed: " <> T.pack (show err)))
  198 
  199     Just uec ->
  200       case uecRemote uec of
  201         Nothing -> pure (failClosedSessionExecWithReason cloneDeps
  202                            "mode=remote but no remote SSH block configured (host/user/known_hosts/workspace required)")
  203 
  204         Just sshCfg -> do
  205           eRemoteWd <- ensureRemoteSessionWorkdir sshCfg runner sid
  206           case eRemoteWd of
  207             Left (WdRemoteMkdirFailed errText)
  208               | "ExecHostKeyUnknown" `T.isInfixOf` errText
  209                 -> case mAdoption of
  210                   Nothing ->
  211                     pure (failClosedSessionExecWithReason cloneDeps
  212                            ("remote host key not in known_hosts (TOFU adoption not available). "
  213                             <> "Host: " <> T.pack (show (scHost sshCfg))
  214                             <> ". Add the key manually with: ssh-keyscan -H "
  215                             <> hostText sshCfg
  216                             <> " >> " <> T.pack (scKnownHosts sshCfg)))
  217                   Just hka -> do
  218                     eAdopt <- hkaAdopt hka sshCfg
  219                     case eAdopt of
  220                       Left adoptErr -> pure (failClosedSessionExecWithReason cloneDeps
  221                                               ("host-key adoption failed: " <> adoptErr))
  222                       Right _ -> do
  223                         -- Retry the workdir bootstrap after adoption.
  224                         eRetry <- ensureRemoteSessionWorkdir sshCfg runner sid
  225                         case eRetry of
  226                           Left retryErr -> pure (failClosedSessionExecWithReason cloneDeps
  227                                                   ("remote workdir creation failed after host-key adoption: "
  228                                                    <> T.pack (show retryErr)))
  229                           Right remoteWdText -> buildRemoteExec sshCfg remoteWdText cloneDeps runner
  230             Left err -> pure (failClosedSessionExecWithReason cloneDeps
  231                                ("remote workdir creation failed: " <> T.pack (show err)))
  232             Right remoteWdText -> buildRemoteExec sshCfg remoteWdText cloneDeps runner
  233 
  234   where
  235     hostText cfg = case scHost cfg of
  236       h -> T.pack (show h)
  237 
  238 -- | Build the 'SessionExec' for a successful remote workdir bootstrap.
  239 buildRemoteExec :: SshConfig -> Text -> CloneDeps -> RemoteRunner
  240                 -> IO SessionExec
  241 buildRemoteExec sshCfg remoteWdText cloneDeps runner =
  242   case mkRemotePath remoteWdText of
  243     Left err -> pure (failClosedSessionExecWithReason cloneDeps
  244                        ("invalid remote workdir path: " <> err))
  245     Right remotePath ->
  246       let sshCfg' = sshCfg { scWorkspace = remotePath }
  247           wsRoot  = WorkspaceRoot (T.unpack (getRemotePath remotePath))
  248           uio     = mkRemoteUntrustedIO sshCfg' runner
  249           uioEnv  = mkTestUIOEnv uio cloneDeps
  250           wfs     = mkRemoteWorkdirFs uioEnv sshCfg' wsRoot maxScanBytes
  251       in pure SessionExec
  252            { seUIOEnv        = uioEnv
  253            , seWorkdirFs     = wfs
  254            , seWorkspaceRoot = wsRoot
  255            }

  257 -- | The fail-closed 'SessionExec': both handles are stubs, the root is
  258 -- 'failClosedRoot'. Used on ANY workdir-creation failure. Returns
  259 -- 'ExecNotImplemented' (no reason) — use 'failClosedSessionExecWithReason'
  260 -- when a descriptive error is available.
  261 failClosedSessionExec :: CloneDeps -> SessionExec
  262 failClosedSessionExec cloneDeps = SessionExec
  263   { seUIOEnv        = mkTestUIOEnv mkRemoteUntrustedIOStub cloneDeps
  264   , seWorkdirFs     = mkWorkdirFsStub
  265   , seWorkspaceRoot = failClosedRoot
  266   }
  267 
  268 -- | Like 'failClosedSessionExec' but the 'UntrustedIO' stub carries a
  269 -- descriptive reason (via 'mkRemoteUntrustedIOStubWithReason'). The model
  270 -- sees *why* every untrusted opcode is failing, not just
  271 -- 'ExecNotImplemented'.
  272 failClosedSessionExecWithReason :: CloneDeps -> Text -> SessionExec
  273 failClosedSessionExecWithReason cloneDeps reason = SessionExec
  274   { seUIOEnv        = mkTestUIOEnv (mkRemoteUntrustedIOStubWithReason reason) cloneDeps
  275   , seWorkdirFs     = mkWorkdirFsStub
  276   , seWorkspaceRoot = failClosedRoot
  277   }

  279 -- | Did this exec fail closed? (The workdir bootstrap failed and every
  280 -- handle is a stub.) The exec cache uses this to avoid memoizing failures.
  281 isFailClosedSessionExec :: SessionExec -> Bool
  282 isFailClosedSessionExec e = seWorkspaceRoot e == failClosedRoot

  284 -- | Construct the per-session 'UntrustedIO' from the 'SecurityConfig':
  285 --
  286 --   * @mode=local@ (or absent): create the local workdir via
  287 --     'ensureSessionWorkdir' and construct 'mkLocalUntrustedIO' with it
  288 --     as the 'WorkspaceRoot'.
  289 --
  290 --   * @mode=remote@ + remote configured: create the remote workdir via
  291 --     'ensureRemoteSessionWorkdir' (SSH @mkdir -p@), clone the
  292 --     'SshConfig' with @scWorkspace@ = the per-session remote workdir,
  293 --     and construct 'mkRemoteUntrustedIO' with the cloned config.
  294 --
  295 --   * @mode=remote@ + remote absent/incomplete: return the fail-closed
  296 --     stub ('mkRemoteUntrustedIOStub').
  297 --
  298 -- This is the single entry point the wiring sites call — it handles
  299 -- both local and remote, creates the workdir, and returns the handle.
  300 -- On workdir creation failure, returns the fail-closed stub (the session
  301 -- should surface the error and NOT proceed — the wiring site checks the
  302 -- 'WorkdirError' via 'ensureSessionWorkdir' separately if it needs to
  303 -- surface the error to the user).
  304 --
  305 -- Back-compat thin wrapper over 'mkSessionExec' (threads
  306 -- 'mkRealRemoteRunner' and no host-key adoption, preserving EXACT current
  307 -- semantics for callers that haven't been updated yet).
  308 mkSessionUntrustedIO
  309   :: SealPaths -> SecurityConfig -> SessionId -> IO UntrustedIO
  310 mkSessionUntrustedIO paths secCfg sid =
  311   uieUntrustedIO . seUIOEnv
  312     <$> mkSessionExec paths secCfg sid stubCloneDeps mkRealRemoteRunner Nothing

  174 -- | Resolve the system prompt for a session turn. This is the **single**
  175 -- implementation — used by all four surfaces (Web, TUI, Telegram, Signal).
  176 --
  177 -- The resolution pipeline:
  178 -- 1. If @smSystemOverride@ is set (non-empty), use it as the base.
  179 -- 2. Otherwise, read the bound agent's @adSystem@ from the workdir-aware
  180 --    agent-def backend.
  181 -- 3. Inject static guidance (parallel / tool-use / task-completion).
  182 -- 4. Inject the autoload skill body.
  183 -- 5. Inject the available-skills catalog (if enabled).
  184 --
  185 -- The flags ('autoloadId', 'injectCatalog', 'parallel', 'toolUse',
  186 -- 'taskCompletion') are passed as explicit args — the caller computes
  187 -- them from the loaded 'RuntimeConfig'. This matches the web path's
  188 -- existing convention and avoids re-reading config inside the resolver.
  189 resolveSystemPrompt
  190   :: Def.AgentDefBackend
  191   -> Skill.SkillBackend
  192   -> Maybe Text
  193   -- ^ The resolved auto-load skill id ('Nothing' disables injection).
  194   -> Bool
  195   -- ^ Whether to inject the @\<available_skills\>@ catalog.
  196   -> Bool
  197   -- ^ Whether to inject the @\<available_agents\>@ catalog (issue #154
  198   -- §3.4; the caller passes 'resolvedAvailableAgents').
  199   -> [AgentDef]
  200   -- ^ The agent-def catalog source (the per-turn union backend's
  201   -- 'adbList' — the caller fetches once; injection stays pure).
  202   -> Bool
  203   -- ^ Whether to inject the parallel tool-call guidance block.
  204   -> Bool
  205   -- ^ Whether to inject the tool-use enforcement guidance block.
  206   -> Bool
  207   -- ^ Whether to inject the task-completion guidance block.
  208   -> Maybe Text
  209   -- ^ The codegraph skill body to inject ('Nothing' when no repo with
  210   -- @.codegraph/@ is in the workdir — skips codegraph injection).
  211   -> SessionMeta
  212   -> IO (Maybe Text)
  213 resolveSystemPrompt agentDefBackend skillBackend autoloadId injectCatalog
  214                     injectAgents agentDefs
  215                     parallel toolUse taskCompletion mCodegraphBody meta = do
  216   base <- case smSystemOverride meta of
  217     Just t | not (T.null (T.strip t)) -> pure (Just t)
  218     _ -> case smAgent meta of
  219            Nothing  -> pure Nothing
  220            Just aid -> maybe Nothing adSystem <$> Def.adbRead agentDefBackend aid
  221   let withGuidance = injectStaticGuidance parallel toolUse taskCompletion base
  222   withAutoload <- injectAutoloadSkill skillBackend autoloadId withGuidance
  223   let withCodegraph = injectCodegraphSkill mCodegraphBody withAutoload
  224   withSkills <- if injectCatalog
  225                   then injectAvailableSkills skillBackend withCodegraph
  226                   else pure withCodegraph
  227   -- The agents catalog is appended AFTER the skills catalog (W3: skills
  228   -- keep their established tail position; agents follow).
  229   pure (if injectAgents
  230           then injectAvailableAgents agentDefs withSkills
  231           else withSkills)

  434 -- | The single turn body (design §5.2). Replaces @plainTurn@ (Send.hs),
  435 -- @runTurnOnSession@ (Loop.hs), and @withCliTurn@ (Cli.hs).
  436 --
  437 -- The engine owns (per the §5.2 step table): load meta, resolve provider,
  438 -- broadcast "thinking", the per-session lock (OUTSIDE the transcript
  439 -- bracket — serializes the entire turn), the transcript bracket,
  440 -- mkSessionExec, workdir-aware backends, the system prompt, the ISA
  441 -- registry, AgentEnv construction, runTurn, broadcast "idle" + new entries,
  442 -- reply fan-out, and auto-tab. The adapter owns: 'replySubscribe' +
  443 -- 'replyFanoutMessage' (@taPreTurn@, runs before the lock), the channel
  444 -- label, the on-stop/on-user-message hooks, the start-wiring builder, and
  445 -- the post-turn @lists@ refresh (@taPostTurn@).
  446 --
  447 -- Broadcast timing + exception safety: the "thinking" signal fires before
  448 -- the lock; the "idle" signal + 'fanoutLastReply' run in a 'bracket' cleanup
  449 -- so a turn killed mid-way (success, sync exception, async 'ThreadKilled')
  450 -- cannot leave the tab stuck in "thinking". This matches the existing web +
  451 -- channel paths; the CLI converges to the same guarantee.
  452 runSessionTurn
  453   :: TurnDeps -> TurnAdapter -> SessionMeta -> Maybe MessageSource -> Text -> IO TurnOutcome
  454 runSessionTurn td adapter meta mSrc t = do
  455   let paths = tdPaths td
  456       sid   = smId meta
  457       sessionDirPath = sessionDir paths sid
  458   eprov <- tdResolve td meta
  459   case eprov of
  460     Left err -> do
  461       -- Provider resolution failed: still run the adapter's pre-turn hook so
  462       -- the adapter isn't left in an inconsistent state (e.g. a channel
  463       -- handle that subscribed but never sees the turn complete). The
  464       -- engine's bracket is NOT entered (no "thinking" signal was sent).
  465       taPreTurn adapter sid meta t
  466       pure (TurnOutcome (Just err))
  467     Right (prov, model) -> do
  468       createDirectoryIfMissing True sessionDirPath
  469       saveSessionMeta paths meta
  470       -- [adapter] Pre-turn: subscribe + cross-channel mirror (inbox-driven
  471       -- channels); no-op for web/CLI.
  472       taPreTurn adapter sid meta t
  473       -- [engine] Broadcast "thinking" (before the lock, matches the existing
  474       -- web + channel paths).
  475       broadcastHarnessStatus (tdBroker td) sid "thinking"
  476       mErr <- bracket
  477         (pure ())
  478         (\_ -> do
  479           -- [engine] Guaranteed cleanup: signal idle + fan out the last
  480           -- assistant reply to subscribed chat channels (so a turn that
  481           -- dies mid-way still releases the tab + delivers any partial
  482           -- reply).
  483           broadcastHarnessStatus (tdBroker td) sid "idle"
  484           fanoutLastReply (tdReplies td) (tdBroker td) paths sid
  485           broadcastReplyDelivered (tdBroker td) sid)
  486         (\_ -> withSessionLock (tdLocks td) sid $
  487           withTwoFileTranscript sessionDirPath $ \tHandle ->

  539 runTurnBody
  540   :: TurnDeps -> TurnAdapter -> SessionMeta -> Maybe MessageSource -> Text
  541   -> SessionId -> SealPaths
  542   -> SomeProvider -> ModelId -> TwoFileHandle
  543   -> IO (Maybe Text)
  544 runTurnBody td adapter meta mSrc t sid paths prov model tHandle = do
  545   appEnv <- mkEnv (tdLogger td) defaultConfig
  546   eCfg <- loadRuntimeConfig (prConfigPath (tdProvider td))
  547   eSecCfg <- loadSecurityConfig (securityFilePath paths)
  548   let operatorCeiling = either (const defaultRetrievalMaxScanBytes) retrievalMaxScanBytes eCfg
  549   cloneDeps <- mkCloneDepsTurn td
  550   exec <- either (\_ _ _ _ -> pure (failClosedSessionExec cloneDeps))
  551                  (\sc _sid _cd runner -> cachedSessionExec (tdExecCache td) paths sc sid cloneDeps runner)
  552                  eSecCfg sid cloneDeps (fromMaybe mkRealRemoteRunner (tdRemoteRunner td))
  553   let wfs    = seWorkdirFs exec
  554       wsRoot = seWorkspaceRoot exec
  555       uioEnv = seUIOEnv exec
  556   -- [engine] One discovery scan per session (cache-backed): the result
  557   -- feeds autoBindRepoAgent AND both per-turn workdir backends, so a turn
  558   -- costs zero extra scans after the first (or after SETUP_REPO).
  559   metaEnv <- metaCacheEnvFor td paths eSecCfg
  560   (workdirDefs, workdirSkillsList) <- cachedWorkdirScan (tdExecCache td) sid wfs wsRoot metaEnv
  561   workdirSkills <- Skill.staticSkillBackend workdirSkillsList
  562   workdirAgentDefs <- Def.staticAgentDefBackend workdirDefs
  563   -- [engine] autoBindRepoAgent (design §5.2 step 3b — currently web-only;
  564   -- unifying means channels/CLI gain it, the desired behavioural
  565   -- convergence). Runs over the SAME cached defs — no extra scan.
  566   autoBindRepoAgentWith workdirDefs paths sid
  567   mMetaAfterBind <- loadSessionMeta paths sid
  568   let meta' = fromMaybe meta mMetaAfterBind

  570   -- base, never mutated). The workdir halves are static backends over the
  571   -- cache's scan result (see above).
  572   let sessionSkills = Skill.tripleUnionSkillBackend workdirSkills (bSkills (tdBaseBackends td))
  573       sessionBackends = (tdBaseBackends td)
  574         { bAgentDefs = Def.unionAgentDefBackend workdirAgentDefs (bAgentDefs (tdBaseBackends td)) }
  575       agentDefBackend = bAgentDefs sessionBackends
  576       autoloadId = either (const Nothing) resolvedAutoloadSkill eCfg
  577       injectCatalog = either (const True) resolvedAvailableSkills eCfg
  578       parallel = either (const True) resolvedParallelToolGuidance eCfg
  579       toolUse = either (const True) resolvedToolUseEnforcement eCfg
  580       taskCompletion = either (const True) resolvedTaskCompletionGuidance eCfg
  581   -- [engine] Check if any cloned repo has a .codegraph/ directory; if so,
  582   -- inject the codegraph skill body into the system prompt.
  583   mCodegraphBody <- codegraphSkillBodyFor wfs sessionSkills
  584   -- [engine] Agent catalog source (W3): ONE adbList on the per-turn union
  585   -- backend feeds both the prompt catalog and the nested spawn resolution.
  586   catalogAgentDefs <- Def.adbList agentDefBackend
  587   let injectAgents = either (const True) resolvedAvailableAgents eCfg
  588   -- [engine] System prompt (single resolver, honors smSystemOverride).
  589   mSystem <- resolveSystemPrompt agentDefBackend sessionSkills
  590               autoloadId injectCatalog injectAgents catalogAgentDefs
  591               parallel toolUse taskCompletion mCodegraphBody meta'
  592   turnAbortFlag <- lookupOrCreateAbortFlag (tdAbortReg td) sid
  593   let onDemand = either (const False) onDemandSchemas eCfg
  594       startWiring = taStartWiring adapter sessionBackends sid appEnv eCfg operatorCeiling meta'
  595       isaReg = buildSessionRegistry (tdVault td) cloneDeps sessionBackends wsRoot sid operatorCeiling
  596                  (tdAutonomy td) (either (const Nothing) rcWeb eCfg) startWiring
  597                  (tdHarnessReg td) (tdTmuxRunner td) (tdHttpManager td) (taCaps adapter) onDemand
  598   tfwSetSecretOps tHandle (ISA.secretOpNames isaReg)
  599   let onEntry = broadcastNewEntries (tdBroker td) paths sid (modelText model) (smCreatedAt meta')
  600       env = (mkSessionAgentEnv TurnEnv
  601               { teCaps          = taCaps adapter
  602               , teProvider      = prov
  603               , teProviderLabel = smProvider meta'
  604               , teModel         = model
  605               , teSession       = sid
  606               , teSystem        = mSystem
  607               , teRegistry      = isaReg
  608               , teTranscript    = tHandle
  609               , teUioEnv        = uioEnv
  610               , teDebugReqPath  = debugRequestsPath paths sid eCfg
  611               , teAutonomy      = tdAutonomy td
  612               , teApprovals     = tdApprovals td
  613               , teOnEntry       = onEntry
  614               , teOnDemand      = onDemand
  615               , teLogPath       = Just (sessionLogPath paths sid)
  616               , teMaxTurns      = either (const defaultMaxTurns) maxTurnsConfig eCfg
  617               , teOnUserMessage = taOnUserMessage adapter meta'
  618               , teChannel       = taChannelLabel adapter meta'
  619               , teOnStop        = taOnStop adapter sid
  620               , teAbortFlag     = turnAbortFlag
  621               , teToolTimeout   = either (const defaultToolTimeoutConfig) toolTimeoutConfig eCfg
  622               })
  623             { aeMessageSource = mSrc }
  624   eResult <- withExceptionLogging (tdLogger td) (Just (sessionLogPath paths sid)) "turn" $
  625     runApp appEnv (runTurn env t)
  626   case eResult of

  767 -- @channelCallDispatcher@, and the CLI's inline @callDispatcher@). Takes
  768 -- explicit 'TurnDeps' + 'SessionId' + 'ChannelCaps' + the channel label (for

  776 -- 'recordSetupRepoResult' + 'autoBindRepoAgent' + 'broadcastAgentDefsChanged'

  778 -- convergence — channels/CLI gain the auto-bind behavior.

  780   :: TurnDeps -> ChannelCaps -> SessionId -> Text

  783   let paths = tdPaths td
  784       sessionDirPath = sessionDir paths sid
  785   createDirectoryIfMissing True sessionDirPath
  786   mMeta <- loadSessionMeta paths sid
  787   withTwoFileTranscript sessionDirPath $ \tHandle -> do
  788     appEnv <- mkEnv (tdLogger td) defaultConfig
  789     eCfg <- loadRuntimeConfig (prConfigPath (tdProvider td))
  790     eSecCfg <- loadSecurityConfig (securityFilePath paths)
  791     let operatorCeiling = either (const defaultRetrievalMaxScanBytes) retrievalMaxScanBytes eCfg
  792     cloneDeps <- mkCloneDepsTurn td
  793     exec <- either (\_ _ _ _ -> pure (failClosedSessionExec cloneDeps))
  794                    (\sc _sid _cd runner -> cachedSessionExec (tdExecCache td) paths sc sid cloneDeps runner)
  795                    eSecCfg sid cloneDeps (fromMaybe mkRealRemoteRunner (tdRemoteRunner td))
  796     let wfs = seWorkdirFs exec
  797         wsRoot = seWorkspaceRoot exec
  798         uioEnv = seUIOEnv exec
  799     metaEnv <- metaCacheEnvFor td paths eSecCfg
  800     (workdirDefs, _) <- cachedWorkdirScan (tdExecCache td) sid wfs wsRoot metaEnv
  801     workdirAgentDefs <- Def.staticAgentDefBackend workdirDefs
  802     let onDemand = either (const False) onDemandSchemas eCfg
  803         sessionBackends = (tdBaseBackends td)
  804           { bAgentDefs = Def.unionAgentDefBackend workdirAgentDefs (bAgentDefs (tdBaseBackends td)) }
  805         startWiring = buildStartWiring td sessionBackends sid appEnv eCfg operatorCeiling channelLabel
  806         isaReg = buildSessionRegistry (tdVault td) cloneDeps sessionBackends wsRoot sid operatorCeiling
  807                    (tdAutonomy td) (either (const Nothing) rcWeb eCfg) startWiring
  808                    (tdHarnessReg td) (tdTmuxRunner td) (tdHttpManager td) caps onDemand
  809     tfwSetSecretOps tHandle (ISA.secretOpNames isaReg)
  810     callAbortFlag <- lookupOrCreateAbortFlag (tdAbortReg td) sid
  811     res <- runApp appEnv (dispatch isaReg tHandle localBackend uioEnv (either (const defaultToolTimeoutConfig) toolTimeoutConfig eCfg) callAbortFlag callOpName val)

  813       Right r -> do

  817             recordSetupRepoResult tHandle callOpName val r (Just channelLabel)

  820               -- discovery scan, re-scan fresh, and bind from the fresh defs.

  827               -- the workdir is set up and the agent is bound. This makes
  828               -- the System Prompt + Tools rows appear in the frontend
  829               -- BEFORE the first user message, not after it. Only recorded
  830               -- when no prior EKRequest entry exists (first time only).

  832               unless (any (\e -> erKind e == EKRequest) priorEntries) $ do
  833                 mMetaAfterBind <- loadSessionMeta paths sid
  834                 let meta' = fromMaybe metaFallback mMetaAfterBind
  835                     metaFallback = fromMaybe (error "callDispatcher: SETUP_REPO without session meta") mMeta
  836                     autoloadId = either (const Nothing) resolvedAutoloadSkill eCfg
  837                     injectCatalog = either (const True) resolvedAvailableSkills eCfg
  838                     parallel = either (const True) resolvedParallelToolGuidance eCfg
  839                     toolUse = either (const True) resolvedToolUseEnforcement eCfg
  840                     taskCompletion = either (const True) resolvedTaskCompletionGuidance eCfg
  841                 freshSkills <- Skill.staticSkillBackend freshSkillsList
  842                 -- Rebuild sessionBackends with the FRESH agent defs (after
  843                 -- autoBindRepoAgent re-bound the agent to the repo's
  844                 -- .agents/agents.md). The original sessionBackends was built
  845                 -- BEFORE the fresh scan, so it doesn't include the repo's
  846                 -- agent definition — using it would produce a preamble with
  847                 -- a stale (or missing) system prompt.
  848                 freshAgentDefs <- Def.staticAgentDefBackend freshDefs
  849                 let freshBackends = sessionBackends
  850                       { bAgentDefs = Def.unionAgentDefBackend freshAgentDefs (bAgentDefs (tdBaseBackends td)) }
  851                     sessionSkills = Skill.tripleUnionSkillBackend freshSkills (bSkills (tdBaseBackends td))
  852                 mCodegraphBody <- codegraphSkillBodyFor wfs sessionSkills
  853                 catalogAgentDefs' <- Def.adbList (bAgentDefs freshBackends)
  854                 let injectAgents = either (const True) resolvedAvailableAgents eCfg
  855                 mSystem <- resolveSystemPrompt
  856                   (bAgentDefs freshBackends) sessionSkills
  857                   autoloadId injectCatalog injectAgents catalogAgentDefs'
  858                   parallel toolUse taskCompletion mCodegraphBody meta'
  859                 let model = maybe (ModelId "") (ModelId . smModel) mMetaAfterBind
  860                 recordPreamble tHandle model mSystem isaReg
  861             broadcastAgentDefsChanged (tdBroker td)
  862           else recordSkillLoadResult tHandle callOpName val r (Just channelLabel)
  863         case mMeta of
  864           Just meta -> broadcastNewEntries (tdBroker td) paths sid (smModel meta) (smCreatedAt meta)
  865           Nothing   -> pure ()
  866       Left _  -> pure ()
  867     pure res

  928     { dwdPaths = tdPaths td
  929     , dwdParentSid = parentSid
  930     , dwdAppEnv = appEnv
  931     , dwdMkUIOEnv = \childSid -> do
  932         childCloneDeps <- mkCloneDepsTurn td
  933         eSecCfg <- loadSecurityConfig (securityFilePath (tdPaths td))
  934         seUIOEnv <$> either (\_ _ _ _ -> pure (failClosedSessionExec childCloneDeps))
  935                             (\sc _sid _cd runner -> cachedSessionExec (tdExecCache td) (tdPaths td) sc childSid childCloneDeps runner)
  936                             eSecCfg childSid childCloneDeps (fromMaybe mkRealRemoteRunner (tdRemoteRunner td))
  937     , dwdAutonomy = tdAutonomy td

 1083 -- | Resolve the child agent's system prompt. Mirrors the three original
 1084 -- @*ChildSystemPrompt@ helpers: the def's 'adSystem' (with the task context
 1085 -- appended) + the static guidance + autoload skill + available-skills
 1086 -- catalog. Uses 'tdBaseBackends' (the child registry doesn't get
 1087 -- workdir-aware skills — matching the original implementations).
 1088 childSystemPrompt
 1089   :: TurnDeps -> Either a RuntimeConfig -> Def.AgentDefBackend -> Bool
 1090   -> AgentDef -> ChildTask -> IO (Maybe Text)
 1091 childSystemPrompt td eCfg unionDefBackend orchEnabled agentDef task = do
 1092   let base = adSystem agentDef
 1093       ctx  = ctContext task
 1094       basePrompt = case (base, ctx) of
 1095         (Just b, Just c) | not (T.null c) -> Just (b <> "\n\nCONTEXT:\n" <> c)
 1096         (Just b, _)                       -> Just b
 1097         (Nothing, Just c)                 -> Just ("CONTEXT:\n" <> c)
 1098         (Nothing, Nothing)                -> Nothing
 1099       autoloadId = either (const Nothing) resolvedAutoloadSkill eCfg
 1100       injectCatalog = either (const True) resolvedAvailableSkills eCfg
 1101       parallel = either (const True) resolvedParallelToolGuidance eCfg
 1102       toolUse = either (const True) resolvedToolUseEnforcement eCfg
 1103       taskCompletion = either (const True) resolvedTaskCompletionGuidance eCfg
 1104       injectAgents = either (const True) resolvedAvailableAgents eCfg
 1105       -- W3 (§3.4): the effective role (def-authoritative, ctRole
 1106       -- narrowed) + the kill switch decide the CHILD's catalog plane —
 1107       -- the same predicate the registry's gate used (both planes gated
 1108       -- together, computed once in the adapter and threaded here). An
 1109       -- orchestrator child (+ switch on) gets the catalog; a leaf child
 1110       -- (or switch-off) gets the one-line leaf note.
 1111       effRole = Worker.effectiveRole (adRole agentDef) (ctRole task)
 1112       canSpawn = effRole == Just "orchestrator" && orchEnabled
 1113       withGuidance = injectStaticGuidance parallel toolUse taskCompletion basePrompt
 1114   withAutoload <- injectAutoloadSkill (bSkills (tdBaseBackends td)) autoloadId withGuidance
 1115   withSkills <- if injectCatalog
 1116     then injectAvailableSkills (bSkills (tdBaseBackends td)) withAutoload
 1117     else pure withAutoload
 1118   if not injectAgents
 1119     then pure withSkills
 1120     else if canSpawn
 1121       then do
 1122         agentDefs <- Def.adbList unionDefBackend
 1123         pure (injectAvailableAgents agentDefs withSkills)
 1124       else pure (Just (maybe leafAgentNote (\p -> p <> "\n\n" <> leafAgentNote) withSkills))

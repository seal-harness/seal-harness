   60 -- | Where a tool call runs. 'TbLocal' is the untrusted local executor
   61 -- (absent under the @-f remote-only-untrusted@ Cabal flag); 'TbTmux' is

   68   | TbContainer ContainerSpec

  102 -- | A container target name (e.g. @ubuntu-22.04@). Smart-constructed: rejects
  103 -- leading-dash (option injection), path separators (@/@ @\\@), colon
  104 -- (option-injection / port-syntax defense), control chars, empty.
  105 newtype ContainerTarget = ContainerTarget Text
  106   deriving stock (Eq, Show)
  107 
  108 mkContainerTarget :: Text -> Either Text ContainerTarget
  109 mkContainerTarget t
  110   | T.null t              = Left "container target is empty"
  111   | T.head t == '-'       = Left "container target must not start with '-'"
  112   | T.any isBad t         = Left "container target has invalid characters"
  113   | otherwise             = Right (ContainerTarget t)
  114   where
  115     isBad c = c == '/' || c == '\\' || c == ':' || isControl c
  116 
  117 getContainerTarget :: ContainerTarget -> Text
  118 getContainerTarget (ContainerTarget t) = t
  119 
  120 -- | An SSH host. Smart-constructed: rejects control chars, space, colon
  121 -- (the colon guards against @host:port@ being parsed as two argv tokens
  122 -- by some tooling).
  123 newtype SshHost = SshHost Text
  124   deriving stock (Eq, Show)
  125 
  126 mkSshHost :: Text -> Either Text SshHost
  127 mkSshHost t
  128   | T.null t        = Left "ssh host is empty"
  129   | T.any isBad t   = Left "ssh host has invalid characters"
  130   | otherwise       = Right (SshHost t)
  131   where
  132     isBad c = isControl c || c == ' ' || c == ':'
  133 
  134 getSshHost :: SshHost -> Text
  135 getSshHost (SshHost t) = t

  152 -- | A workspace-relative path on the remote machine. Smart-constructed:
  153 -- rejects leading-dash (option injection), and rejects a @..@ escape. 4a
  154 -- ships a permissive validator (the lexical re-anchoring against the
  155 -- remote root is the job of 'Seal.Security.Path.mkSafePathRemote' in 4g);
  156 -- 4a only needs a validated newtype the 'SshConfig' can carry.
  157 newtype RemotePath = RemotePath Text
  158   deriving stock (Eq, Ord, Show)
  159 
  160 mkRemotePath :: Text -> Either Text RemotePath
  161 mkRemotePath t
  162   | T.null t              = Left "remote path is empty"
  163   | T.head t == '-'       = Left "remote path must not start with '-'"
  164   | T.any isControl t     = Left "remote path has control characters"
  165   | otherwise             = Right (RemotePath t)
  166 
  167 getRemotePath :: RemotePath -> Text
  168 getRemotePath (RemotePath t) = t

  174 -- | The executor-layer error ADT. Lives here (not in
  175 -- 'Seal.Tools.Exec.Untrusted') to break the import cycle with
  176 -- 'Seal.Tools.Exec.Local'.
  177 data ExecError
  178   = ExecNotAllowed                -- ^ the operator policy denies the call
  179   | ExecLocalNotPermittedForUntrusted -- ^ 'selectUntrustedBackend' got a non-Ssh backend
  180   | ExecRemoteRequired            -- ^ mode=remote but no remote configured/reachable
  181   | ExecRemoteUnreachable         -- ^ SSH connect failed (not a host-key mismatch)
  182   | ExecHostKeyMismatch           -- ^ hard security failure; never bypassed
  183   | ExecNotImplemented            -- ^ the backend is stubbed (e.g. TbContainer, TbTmux-as-tool-call)
  184   deriving stock (Eq, Show)

  190 -- | The local-executor handle. The TYPE and its CONSTRUCTOR both live here
  191 -- (Haskell requires a type and its constructors in the same module — the
  192 -- 4a placeholder was an opaque `data LocalExecHandle` with no constructors;
  193 -- 4b-T1 widens it to the real handle). The smart constructor that wires
  194 -- the real `System.Process`-backed IO actions lives in
  195 -- 'Seal.Tools.Exec.Local' ('mkLocalExecHandle'); the tests also use the
  196 -- placeholder constructor directly for fakes.
  197 --
  198 -- The fields are the IO actions the Untrusted opcodes call. Each returns
  199 -- an 'Either ExecError' so the opcode can surface a structured error
  200 -- (fail-closed) instead of throwing.
  201 data LocalExecHandle = LocalExecHandle
  202   { lehExecShell :: ShellCommand -> Maybe RemotePath -> IO (Either ExecError Text)
  203     -- ^ run a validated 'ShellCommand' via @/bin/sh -c@ (single arg, fixed
  204     -- argv), with an optional cwd ('SafePath'-confined to the workspace root)
  205   , lehExecBin :: BinName -> [BinArg] -> Maybe RemotePath
  206                  -> IO (Either ExecError Text)
  207     -- ^ run a named binary (resolved on PATH or by absolute/relative path,
  208     -- via 'System.Process.proc' RawCommand — no shell) with validated argv
  209     -- args and an optional cwd. Used by the BIN_EXEC opcode.
  210   }
  211 
  212 -- | A placeholder constructor for tests that don't exercise the IO
  213 -- (mirrors the 4a opaque form). Real callers use
  214 -- 'Seal.Tools.Exec.Local.mkLocalExecHandle'.
  215 mkLocalExecHandlePlaceholder :: LocalExecHandle
  216 mkLocalExecHandlePlaceholder = LocalExecHandle
  217   { lehExecShell = \_ _ -> pure (Right "")
  218   , lehExecBin = \_ _ _ -> pure (Right "")
  219   }

  222 -- ExecBackend (added in 4a-T2)
  223 -- ---------------------------------------------------------------------------

  225 -- | The execution backend the dispatcher consumes. The 'EbLocal' arm is
  226 -- the untrusted local executor (absent under @-f remote-only-untrusted@);
  227 -- the 'EbRemote' arm is the SSH executor. 'selectExecBackend' (4a-T2)
  228 -- returns this; 'selectUntrustedBackend' (4a-T2) returns ONLY the remote
  229 -- arm (the spec's 'UntrustedExecBackend' is Ssh-only by construction).
  230 --
  231 -- No 'Eq' is derived: 'LocalExecHandle' carries IO actions which have no
  232 -- 'Eq' instances. A custom 'Show' is provided (the 'LocalExecHandle' shows
  233 -- as @<local>@, the 'SshConfig' via its own 'Show').
  234 data ExecBackend
  235   = EbLocal LocalExecHandle
  236   | EbRemote SshConfig
  237 
  238 instance Show ExecBackend where
  239   show (EbLocal _) = "EbLocal <local executor>"
  240   show (EbRemote c) = "EbRemote " <> show c
  241 
  242 -- | Structural 'Eq'. Two 'EbLocal' handles are always 'equal' (they carry
  243 -- opaque IO actions; no meaningful equality — used only by test `shouldBe`
  244 -- on the 'Left' arms, where both sides are 'Left'). Two 'EbRemote' compare
  245 -- by 'SshConfig'.
  246 instance Eq ExecBackend where
  247   EbLocal _ == EbLocal _ = True
  248   EbRemote a == EbRemote b = a == b
  249   _ == _ = False

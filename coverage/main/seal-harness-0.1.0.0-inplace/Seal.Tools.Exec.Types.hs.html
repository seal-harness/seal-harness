   61 -- | Where a tool call runs. 'TbLocal' is the untrusted local executor
   62 -- (absent under the @-f remote-only-untrusted@ Cabal flag); 'TbTmux' is

   64 -- SSH executor (Phase 4 4g); 'TbContainer' is forward-compat (stubbed).

   69   | TbContainer ContainerSpec

   80 -- | SSH coordinates for a 'TbSsh' tool call (the remote-only untrusted

   89   , scWorkspace  :: RemotePath     -- ^ remote workspace root (validated)

  103 -- | A container target name (e.g. @ubuntu-22.04@). Smart-constructed: rejects
  104 -- leading-dash (option injection), path separators (@/@ @\\@), colon
  105 -- (option-injection / port-syntax defense), control chars, empty.
  106 newtype ContainerTarget = ContainerTarget Text
  107   deriving stock (Eq, Show)
  108 
  109 mkContainerTarget :: Text -> Either Text ContainerTarget
  110 mkContainerTarget t
  111   | T.null t              = Left "container target is empty"
  112   | T.head t == '-'       = Left "container target must not start with '-'"
  113   | T.any isBad t         = Left "container target has invalid characters"
  114   | otherwise             = Right (ContainerTarget t)
  115   where
  116     isBad c = c == '/' || c == '\\' || c == ':' || isControl c
  117 
  118 getContainerTarget :: ContainerTarget -> Text
  119 getContainerTarget (ContainerTarget t) = t
  120 
  121 -- | An SSH host. Smart-constructed: rejects control chars, space, colon
  122 -- (the colon guards against @host:port@ being parsed as two argv tokens
  123 -- by some tooling).
  124 newtype SshHost = SshHost Text
  125   deriving stock (Eq, Show)
  126 
  127 mkSshHost :: Text -> Either Text SshHost
  128 mkSshHost t
  129   | T.null t        = Left "ssh host is empty"
  130   | T.any isBad t   = Left "ssh host has invalid characters"
  131   | otherwise       = Right (SshHost t)
  132   where
  133     isBad c = isControl c || c == ' ' || c == ':'
  134 
  135 getSshHost :: SshHost -> Text
  136 getSshHost (SshHost t) = t

  153 -- | A workspace-relative path on the remote machine. Smart-constructed:
  154 -- rejects leading-dash (option injection), and rejects a @..@ escape. 4a
  155 -- ships a permissive validator (the lexical re-anchoring against the
  156 -- remote root is the job of 'Seal.Security.Path.mkSafePathRemote' in 4g);
  157 -- 4a only needs a validated newtype the 'SshConfig' can carry.
  158 newtype RemotePath = RemotePath Text
  159   deriving stock (Eq, Ord, Show)
  160 
  161 mkRemotePath :: Text -> Either Text RemotePath
  162 mkRemotePath t
  163   | T.null t              = Left "remote path is empty"
  164   | T.head t == '-'       = Left "remote path must not start with '-'"
  165   | T.any isControl t     = Left "remote path has control characters"
  166   | otherwise             = Right (RemotePath t)
  167 
  168 getRemotePath :: RemotePath -> Text
  169 getRemotePath (RemotePath t) = t

  175 -- | The executor-layer error ADT. Lives here (not in
  176 -- 'Seal.Tools.Exec.Untrusted') to break the import cycle with
  177 -- 'Seal.Tools.Exec.Local'.
  178 data ExecError
  179   = ExecNotAllowed                -- ^ the operator policy denies the call
  180   | ExecLocalNotPermittedForUntrusted -- ^ 'selectUntrustedBackend' got a non-Ssh backend
  181   | ExecRemoteRequired            -- ^ mode=remote but no remote configured/reachable
  182   | ExecRemoteUnreachable         -- ^ SSH connect failed (not a host-key mismatch)
  183   | ExecHostKeyMismatch           -- ^ hard security failure; never bypassed
  184   | ExecHostKeyUnknown            -- ^ no host key in known_hosts (TOFU adoption candidate)
  185   | ExecNotImplemented            -- ^ the backend is stubbed (e.g. TbContainer, TbTmux-as-tool-call)
  186   | ExecNotImplementedReason Text -- ^ the backend is stubbed with a specific reason
  187   deriving stock (Eq, Show)

  193 -- | The local-executor handle. The TYPE and its CONSTRUCTOR both live here
  194 -- (Haskell requires a type and its constructors in the same module — the
  195 -- 4a placeholder was an opaque `data LocalExecHandle` with no constructors;
  196 -- 4b-T1 widens it to the real handle). The smart constructor that wires
  197 -- the real `System.Process`-backed IO actions lives in
  198 -- 'Seal.Tools.Exec.Local' ('mkLocalExecHandle'); the tests also use the
  199 -- placeholder constructor directly for fakes.
  200 --
  201 -- The fields are the IO actions the Untrusted opcodes call. Each returns
  202 -- an 'Either ExecError' so the opcode can surface a structured error
  203 -- (fail-closed) instead of throwing.
  204 data LocalExecHandle = LocalExecHandle
  205   { lehExecShell :: ShellCommand -> Maybe RemotePath -> IO (Either ExecError Text)
  206     -- ^ run a validated 'ShellCommand' via @/bin/sh -c@ (single arg, fixed
  207     -- argv), with an optional cwd ('SafePath'-confined to the workspace root)
  208   , lehExecBin :: BinName -> [BinArg] -> Maybe RemotePath
  209                  -> IO (Either ExecError Text)
  210     -- ^ run a named binary (resolved on PATH or by absolute/relative path,
  211     -- via 'System.Process.proc' RawCommand — no shell) with validated argv
  212     -- args and an optional cwd. Used by the BIN_EXEC opcode.
  213   }
  214 
  215 -- | A placeholder constructor for tests that don't exercise the IO
  216 -- (mirrors the 4a opaque form). Real callers use
  217 -- 'Seal.Tools.Exec.Local.mkLocalExecHandle'.
  218 mkLocalExecHandlePlaceholder :: LocalExecHandle
  219 mkLocalExecHandlePlaceholder = LocalExecHandle
  220   { lehExecShell = \_ _ -> pure (Right "")
  221   , lehExecBin = \_ _ _ -> pure (Right "")
  222   }

  228 -- | The execution backend the dispatcher consumes. The 'EbLocal' arm is
  229 -- the untrusted local executor (absent under @-f remote-only-untrusted@);
  230 -- the 'EbRemote' arm is the SSH executor. 'selectExecBackend' (4a-T2)
  231 -- returns this; 'selectUntrustedBackend' (4a-T2) returns ONLY the remote
  232 -- arm (the spec's 'UntrustedExecBackend' is Ssh-only by construction).
  233 --
  234 -- No 'Eq' is derived: 'LocalExecHandle' carries IO actions which have no
  235 -- 'Eq' instances. A custom 'Show' is provided (the 'LocalExecHandle' shows
  236 -- as @<local>@, the 'SshConfig' via its own 'Show').
  237 data ExecBackend
  238   = EbLocal LocalExecHandle
  239   | EbRemote SshConfig
  240 
  241 instance Show ExecBackend where
  242   show (EbLocal _) = "EbLocal <local executor>"
  243   show (EbRemote c) = "EbRemote " <> show c
  244 
  245 -- | Structural 'Eq'. Two 'EbLocal' handles are always 'equal' (they carry
  246 -- opaque IO actions; no meaningful equality — used only by test `shouldBe`
  247 -- on the 'Left' arms, where both sides are 'Left'). Two 'EbRemote' compare
  248 -- by 'SshConfig'.
  249 instance Eq ExecBackend where
  250   EbLocal _ == EbLocal _ = True
  251   EbRemote a == EbRemote b = a == b
  252   _ == _ = False

  254 -- | Format a non-zero exit result for the tool-call consumer. Combines
  255 -- stdout and stderr (if non-empty) and annotates the exit code so the
  256 -- frontend can surface it. The result is returned via 'Right' (not an
  257 -- 'ExecError') so the dispatcher records @is_error = False@ — the
  258 -- command ran successfully, it just returned a non-zero exit code. The
  259 -- frontend treats the exit code annotation as the success/failure
  260 -- signal.
  261 --
  262 -- Shared by the local executor ('Seal.Tools.Exec.Local'), the remote SSH
  263 -- executor ('Seal.Tools.Exec.Remote'), and the unified UntrustedIO
  264 -- handle ('Seal.Tools.Exec.UntrustedIO') so both arms format non-zero
  265 -- exits identically — the model sees the same shape whether the command
  266 -- ran locally or over SSH.
  267 formatExitResult :: Int -> Text -> Text -> Text
  268 formatExitResult n out err =
  269   let parts = [ t | t <- [out, err], not (T.null (T.strip t)) ]
  270       body  = if null parts then "" else T.intercalate "\n" parts
  271   in body <> "\n[exit code: " <> T.pack (show n) <> "]"

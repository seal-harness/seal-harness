   48 -- | Which search backend to use. 'ProviderParallel' is the zero-config
   49 -- default (free MCP endpoint, no API key needed).
   50 data SearchProvider
   51   = ProviderParallel    -- ^ Parallel Search (free MCP or REST with key)
   52   | ProviderSearXNG     -- ^ Self-hosted SearXNG instance
   53   | ProviderExa         -- ^ Exa search API (requires key)
   54   | ProviderFirecrawl   -- ^ Firecrawl search API (requires key)
   55   | ProviderCustom      -- ^ User-configured custom endpoint (legacy behavior)
   56   deriving stock (Eq, Show)
   57 
   58 -- | Parse a provider name from config text. Unknown values default to
   59 -- 'ProviderParallel' (the zero-config default).
   60 parseProvider :: Text -> SearchProvider
   61 parseProvider t = case T.toLower t of
   62   "searxng"   -> ProviderSearXNG
   63   "exa"       -> ProviderExa
   64   "firecrawl" -> ProviderFirecrawl
   65   "custom"    -> ProviderCustom
   66   _           -> ProviderParallel
   67 
   68 -- | Render a provider to its canonical string name (for 'orRecorded').
   69 providerName :: SearchProvider -> Text
   70 providerName ProviderParallel  = "parallel"
   71 providerName ProviderSearXNG   = "searxng"
   72 providerName ProviderExa       = "exa"
   73 providerName ProviderFirecrawl = "firecrawl"
   74 providerName ProviderCustom    = "custom"

   76 -- | A single normalized search result. All providers decode their
   77 -- provider-specific JSON into this shape before re-encoding for the model.
   78 data SearchResult = SearchResult
   79   { srTitle       :: Text
   80   , srUrl         :: Text
   81   , srDescription :: Text
   82   , srPosition    :: Int
   83   } deriving stock (Eq, Show)

   85 -- | Encode search results as JSON for the model ('orParts'). The shape is
   86 -- @{"results": [{"title","url","description","position"}, ...]}@.
   87 encodeResults :: [SearchResult] -> Text
   88 encodeResults results = TE.decodeUtf8 (BL.toStrict (A.encode body))
   89   where
   90     body = A.object ["results" .= map toValue results]
   91     toValue (SearchResult title url desc pos) = A.object
   92       [ "title"       .= title
   93       , "url"         .= url
   94       , "description" .= desc
   95       , "position"    .= pos
   96       ]

   98 -- | The configuration for WEB_SEARCH.
   99 data WebSearchConfig = WebSearchConfig
  100   { wscManager     :: Maybe Manager     -- ^ HTTP manager (Nothing = fail-closed)
  101   , wscProvider    :: SearchProvider    -- ^ which backend to use
  102   , wscEndpoint    :: Text             -- ^ endpoint URL (empty = use provider default)
  103   , wscAllowList   :: [Text]           -- ^ allowed domains (empty = all allowed)
  104   , wscAuthKey     :: Maybe Text       -- ^ vault key reference for API key
  105   , wscMaxResults  :: Int              -- ^ max results to return (0 = provider default)
  106   , wscVault       :: Maybe VaultRuntime  -- ^ vault for resolving API keys
  107   , wscSearXngUrl  :: Maybe Text       -- ^ SearXNG instance URL (Nothing = localhost:8888)
  108   }

  110 -- | WEB_SEARCH opcode. Input: @{ query: Text, limit?: Int }@. Dispatches to
  111 -- the configured provider ('wscProvider'). 'orRecorded' carries only the
  112 -- query, result count, and provider name — never API keys or response bodies.
  113 webSearchOp :: WebSearchConfig -> Opcode
  114 webSearchOp cfg = UntrustedOpcode
  115   { uoName = OpName "WEB_SEARCH"
  116   , uoDesc = "Search the web using the configured provider (parallel, searxng, exa, firecrawl, or custom). Returns ranked results with title, URL, and description."
  117   , uoInSchema = webSearchSchema
  118   , uoOutSchema = object []
  119   , uoAuthorize = \v ->
  120       case queryField v of
  121         Nothing -> Left "WEB_SEARCH requires {query:string}"
  122         Just q
  123           | T.null q -> Left "WEB_SEARCH: query is empty"
  124           | otherwise -> case limitField v of
  125               Just n | n < 1 -> Left "WEB_SEARCH: limit must be >= 1"
  126               _ -> Right ()
  127   , uoRun = \v -> do
  128       let q = fromMaybe "" (queryField v)
  129           userLimit = limitField v
  130           cfg' = cfg { wscMaxResults = fromMaybe (wscMaxResults cfg) userLimit }
  131       case wscManager cfg' of
  132         Nothing -> pure (OpResult
  133           [TrpText "WEB_SEARCH: no HTTP manager configured"]
  134           True (recorded cfg' q 0))
  135         Just mgr -> uioLiftIO (dispatchSearch mgr cfg' q)
  136   }

  138 -- | Build the secret-free 'orRecorded' metadata value for a search call.
  139 recorded :: WebSearchConfig -> Text -> Int -> Value
  140 recorded cfg q n = object
  141   [ "query"        .= q
  142   , "result_count" .= n
  143   , "provider"     .= providerName (wscProvider cfg)
  144   ]

  146 -- | A standalone 'recorded' that takes the provider name directly (used by
  147 -- stub/error paths that don't have a full config in scope).
  148 recorded' :: Text -> Text -> Int -> Value
  149 recorded' q provider n = object
  150   [ "query"        .= q
  151   , "result_count" .= n
  152   , "provider"     .= provider
  153   ]

  155 -- | Construct an error 'OpResult' with recorded metadata.
  156 mkError :: Text -> Text -> Text -> OpResult
  157 mkError q provider msg = OpResult
  158   [TrpText msg]
  159   True (recorded' q provider 0)

  161 -- | Decode a response body to 'Text' (lenient — never throws on bad UTF-8).
  162 bodyText :: BL.ByteString -> Text
  163 bodyText = TE.decodeUtf8With TEE.lenientDecode . BL.toStrict

  165 -- | Dispatch to the appropriate provider search function.
  166 dispatchSearch :: Manager -> WebSearchConfig -> Text -> IO OpResult
  167 dispatchSearch mgr cfg q = case wscProvider cfg of
  168   ProviderParallel  -> searchParallel mgr cfg q
  169   ProviderSearXNG   -> searchSearXNG mgr cfg q
  170   ProviderExa       -> searchExa mgr cfg q
  171   ProviderFirecrawl -> searchFirecrawl mgr cfg q
  172   ProviderCustom    -> searchCustom mgr cfg q

  174 -- | Resolve an API key from the vault if configured. Returns 'Nothing' if
  175 -- no vault key is configured or the vault is unavailable. The resolved key
  176 -- never appears in 'orRecorded' — it is injected only into the HTTP request.
  177 resolveApiKey :: WebSearchConfig -> IO (Maybe Text)
  178 resolveApiKey cfg = case (wscVault cfg, wscAuthKey cfg) of
  179   (Just rt, Just keyName) -> do
  180     eVal <- vaultGetByName rt keyName
  181     pure (either (const Nothing) Just eVal)
  182   _ -> pure Nothing

  188 searchParallel :: Manager -> WebSearchConfig -> Text -> IO OpResult
  189 searchParallel mgr cfg q = do
  190   mKey <- resolveApiKey cfg
  191   case mKey of
  192     Just key -> searchParallelRest mgr cfg q key
  193     Nothing  -> searchParallelMcp mgr cfg q

  195 -- | Mode A — MCP endpoint (zero-config, no API key). Uses JSON-RPC over HTTP
  196 -- (Streamable HTTP transport). The response may be @application/json@
  197 -- (plain JSON-RPC) or @text/event-stream@ (SSE).
  198 searchParallelMcp :: Manager -> WebSearchConfig -> Text -> IO OpResult
  199 searchParallelMcp mgr _cfg q = do
  200   eReq <- try (parseRequest "https://search.parallel.ai/mcp")
  201   case eReq of
  202     Left (_ :: HttpException) ->
  203       pure (mkError q "parallel" "WEB_SEARCH: failed to parse Parallel MCP URL")
  204     Right initReq -> do
  205       let body = A.encode $ A.object
  206             [ "jsonrpc" .= ("2.0" :: Text)
  207             , "method"  .= ("tools/call" :: Text)
  208             , "params"  .= A.object
  209                 [ "name"      .= ("web_search" :: Text)
  210                 , "arguments" .= A.object
  211                     [ "objective"      .= q
  212                     , "search_queries" .= [q]
  213                     ]
  214                 ]
  215             , "id"      .= (1 :: Int)
  216             ]
  217           req = initReq
  218             { method = methodPost
  219             , requestBody = RequestBodyLBS body
  220             , requestHeaders =
  221                 [ ("content-type", "application/json")
  222                 , ("accept", "application/json, text/event-stream")
  223                 ]
  224             }
  225       eResp <- try (httpLbs req mgr)
  226       case eResp of
  227         Left (_ :: HttpException) ->
  228           pure (mkError q "parallel" "WEB_SEARCH: Parallel MCP request failed")
  229         Right resp -> do
  230           let code = statusCode (responseStatus resp)
  231               respBody = responseBody resp
  232           if code >= 200 && code <= 299
  233             then parseParallelMcpResponse q respBody
  234             else pure (mkError q "parallel" $
  235               "WEB_SEARCH: Parallel MCP HTTP " <> T.pack (show code) <> ": "
  236               <> bodyText respBody)

  238 -- | Extract text content from the MCP @result.content@ array. Each item has
  239 -- @{type: "text", text: "..."}@; we concatenate all text items.
  240 extractMcpContentText :: Value -> Text
  241 extractMcpContentText val = fromMaybe "" $
  242   parseMaybe (withObject "result" (.: "content")) val >>= \case
  243     A.Array arr -> Just $ T.intercalate "\n"
  244       [ fromMaybe "" (parseMaybe (withObject "item" (.: "text")) item)
  245       | item <- V.toList arr
  246       , hasTextType item
  247       ]
  248     _ -> Nothing
  249   where
  250     hasTextType v =
  251       parseMaybe (withObject "item" (.: "type")) v == Just ("text" :: Text)
  252 
  253 -- | Parse the text content from Parallel MCP into a 'SearchResult' list.
  254 -- The text may be JSON with a @results@ array, or a bare JSON array, or
  255 -- plain text (in which case we return no structured results).
  256 parseParallelResults :: Text -> [SearchResult]
  257 parseParallelResults txt =
  258   case A.decode (BL.fromStrict (TE.encodeUtf8 txt)) :: Maybe Value of
  259     Just (A.Object o) ->
  260       case parseMaybe (withObject "r" (.: "results")) (A.Object o) of
  261         Just (A.Array arr) -> zipWith fromParallelJson [1..] (V.toList arr)
  262         _ -> []
  263     Just (A.Array arr) -> zipWith fromParallelJson [1..] (V.toList arr)
  264     _ -> []
  265   where
  266     fromParallelJson :: Int -> Value -> SearchResult
  267     fromParallelJson pos v = fromMaybe (SearchResult "" "" "" pos) $
  268       parseMaybe (withObject "result" $ \obj -> do
  269         url      <- obj .: "url"
  270         title    <- obj .: "title"
  271         excerpts <- obj .:? "excerpts" .!= []
  272         let desc = T.intercalate " " (map asText excerpts)
  273         pure (SearchResult title url desc pos)) v
  274 
  275     asText :: Value -> Text
  276     asText (A.String s) = s
  277     asText _ = ""
  278 
  279 parseParallelMcpResponse :: Text -> BL.ByteString -> IO OpResult
  280 parseParallelMcpResponse q respBody =
  281   case A.decode respBody :: Maybe Value of
  282     Nothing -> pure (mkError q "parallel" "WEB_SEARCH: Parallel MCP returned non-JSON")
  283     Just val -> case parseMaybe (withObject "rpc" (.: "result")) val of
  284       Nothing -> pure (mkError q "parallel" "WEB_SEARCH: Parallel MCP missing result field")
  285       Just resultVal -> do
  286         let contentText = extractMcpContentText resultVal
  287             results = parseParallelResults contentText
  288         pure (OpResult
  289           [TrpText (encodeResults results)]
  290           False (recorded' q "parallel" (length results)))

  292 -- | Mode B — REST API (with API key for higher limits).
  293 searchParallelRest :: Manager -> WebSearchConfig -> Text -> Text -> IO OpResult
  294 searchParallelRest mgr cfg q apiKey = do
  295   let endpoint = if T.null (wscEndpoint cfg)
  296                    then "https://api.parallel.ai/v1/search"
  297                    else wscEndpoint cfg
  298       maxR = min (wscMaxResults cfg) 20
  299       body = A.encode $ A.object
  300         [ "objective"      .= q
  301         , "search_queries" .= [q]
  302         , "mode"           .= ("fast" :: Text)
  303         , "max_results"    .= maxR

  307     Left (_ :: HttpException) ->

  310       let req = initReq
  311             { method = methodPost
  312             , requestBody = RequestBodyLBS body
  313             , requestHeaders =
  314                 [ ("content-type", "application/json")
  315                 , ("x-api-key", TE.encodeUtf8 apiKey)
  316                 ]
  317             }
  318       eResp <- try (httpLbs req mgr)
  319       case eResp of
  320         Left (_ :: HttpException) ->
  321           pure (mkError q "parallel" "WEB_SEARCH: Parallel REST request failed")
  322         Right resp -> do
  323           let code = statusCode (responseStatus resp)

  356 -- | URL-encode a 'Text' for use in a query string.
  357 urlEncodeText :: Text -> Text
  358 urlEncodeText = TE.decodeUtf8 . urlEncode False . TE.encodeUtf8
  359 
  360 searchSearXNG :: Manager -> WebSearchConfig -> Text -> IO OpResult
  361 searchSearXNG mgr cfg q = do
  362   let baseUrl = fromMaybe "http://localhost:8888" (wscSearXngUrl cfg)
  363   result <- searXngSearchOnce mgr cfg q baseUrl
  364   case result of
  365     Right opRes -> pure opRes
  366     Left errMsg -> do
  367       installed <- ensureSearXng mgr baseUrl
  368       if installed
  369         then do
  370           result2 <- searXngSearchOnce mgr cfg q baseUrl
  371           case result2 of
  372             Right opRes    -> pure opRes
  373             Left errMsg2   -> pure (mkError q "searxng" errMsg2)
  374         else pure (mkError q "searxng" $
  375           errMsg <> " Auto-install attempted but failed. Ensure Docker is running or start SearXNG manually.")
  376 
  377 searXngSearchOnce :: Manager -> WebSearchConfig -> Text -> Text -> IO (Either Text OpResult)
  378 searXngSearchOnce mgr cfg q baseUrl = do
  379   let endpoint = baseUrl <> "/search"
  380       maxR = wscMaxResults cfg
  381       fullUrl = endpoint <> "?" <> "q=" <> urlEncodeText q <> "&format=json&pageno=1"
  382   eReq <- try (parseRequest (T.unpack fullUrl))
  383   case eReq of
  384     Left (_ :: HttpException) ->
  385       pure (Left "WEB_SEARCH: invalid SearXNG URL")
  386     Right initReq -> do
  387       let req = initReq { requestHeaders = [("accept", "application/json")] }
  388       eResp <- try (httpLbs req mgr)
  389       case eResp of
  390         Left (_ :: HttpException) ->
  391           pure (Left "WEB_SEARCH: SearXNG request failed (is the instance running?)")
  392         Right resp -> do
  393           let code = statusCode (responseStatus resp)
  394               respBody = responseBody resp
  395           if code >= 200 && code <= 299
  396             then Right <$> parseSearXngResponse q maxR respBody
  397             else pure (Left $ "WEB_SEARCH: SearXNG HTTP " <> T.pack (show code) <> ": " <> bodyText respBody)
  398 
  399 parseSearXngResponse :: Text -> Int -> BL.ByteString -> IO OpResult
  400 parseSearXngResponse q maxR respBody =
  401   case A.decode respBody :: Maybe Value of
  402     Nothing -> pure (mkError q "searxng" "WEB_SEARCH: SearXNG returned non-JSON (is JSON format enabled in settings.yml?)")
  403     Just val -> case parseMaybe (withObject "r" (.: "results")) val of
  404       Just (A.Array arr) -> do
  405         let rawResults = V.toList arr
  406             scored = [(scoreOf r, r) | r <- rawResults]
  407             sorted = map snd (sortBy (flip compare `on` fst) scored)
  408             taken = take maxR sorted
  409             results = zipWith fromSearXngJson [1..] taken
  410         pure (OpResult [TrpText (encodeResults results)] False (recorded' q "searxng" (length results)))
  411       _ -> pure (mkError q "searxng" "WEB_SEARCH: SearXNG missing results array")
  412   where
  413     scoreOf :: Value -> Double
  414     scoreOf v = fromMaybe 0 $
  415       parseMaybe (withObject "r" (.: "score")) v >>= \case
  416         A.Number n -> Just (realToFrac n)
  417         _ -> Nothing
  418 
  419     fromSearXngJson :: Int -> Value -> SearchResult
  420     fromSearXngJson pos v = fromMaybe (SearchResult "" "" "" pos) $
  421       parseMaybe (withObject "r" $ \obj -> do
  422         url     <- obj .:  "url"
  423         title   <- obj .:  "title"
  424         content <- obj .:? "content" .!= ""
  425         pure (SearchResult title url content pos)) v

  539 parseFirecrawlResponse :: Text -> BL.ByteString -> IO OpResult
  540 parseFirecrawlResponse q respBody =
  541   case A.decode respBody :: Maybe Value of
  542     Nothing -> pure (mkError q "firecrawl" "WEB_SEARCH: Firecrawl returned non-JSON")
  543     Just val -> do
  544       let results = extractFirecrawlResults val
  545       if null results
  546         then pure (mkError q "firecrawl" "WEB_SEARCH: Firecrawl returned no results")
  547         else pure (OpResult [TrpText (encodeResults results)] False (recorded' q "firecrawl" (length results)))
  548 
  549 -- | Extract results from a Firecrawl response, trying multiple shapes:
  550 -- v2: @data.web[]@ → @data.results[]@ → @web[]@ → @results[]@ → @data@ (flat list, v1).
  551 extractFirecrawlResults :: Value -> [SearchResult]
  552 extractFirecrawlResults val =
  553   case tryPath [fromText "data", fromText "web"] val of
  554     Just rs -> rs
  555     Nothing -> case tryPath [fromText "data", fromText "results"] val of
  556       Just rs -> rs
  557       Nothing -> case tryPath [fromText "web"] val of
  558         Just rs -> rs
  559         Nothing -> case tryPath [fromText "results"] val of
  560           Just rs -> rs
  561           Nothing -> case parseMaybe (withObject "r" (.: "data")) val of
  562             Just (A.Array arr) -> zipWith fromFirecrawlJson [1..] (V.toList arr)
  563             _ -> []
  564   where
  565     tryPath :: [A.Key] -> Value -> Maybe [SearchResult]
  566     tryPath keys v = do
  567       let go []     val' = Just val'
  568           go (k:ks) val' = parseMaybe (withObject "o" (.: k)) val' >>= go ks
  569       resultVal <- go keys v
  570       case resultVal of
  571         A.Array arr -> Just (zipWith fromFirecrawlJson [1..] (V.toList arr))
  572         _ -> Nothing
  573 
  574     fromFirecrawlJson :: Int -> Value -> SearchResult
  575     fromFirecrawlJson pos v = fromMaybe (SearchResult "" "" "" pos) $
  576       parseMaybe (withObject "r" $ \obj -> do
  577         url   <- obj .:  "url"
  578         title <- obj .:  "title"
  579         desc  <- obj .:? "description" .!= ""
  580         pos'  <- obj .:? "position" .!= pos
  581         pure (SearchResult title url desc pos')) v

  587 searchCustom :: Manager -> WebSearchConfig -> Text -> IO OpResult
  588 searchCustom mgr cfg q
  589   | T.null (wscEndpoint cfg) = pure (mkError q "custom"
  590       "WEB_SEARCH: no search endpoint configured and no provider selected. Set [web] search_provider or search_endpoint.")
  591   | otherwise = do
  592       eReq <- try (parseRequest (T.unpack (wscEndpoint cfg)))
  593       case eReq of
  594         Left (_ :: HttpException) ->
  595           pure (mkError q "custom" ("WEB_SEARCH: invalid endpoint URL: " <> wscEndpoint cfg))
  596         Right initReq -> do
  597           let body = A.encode (A.object ["query" .= q])
  598               req = initReq
  599                 { method = methodPost
  600                 , requestBody = RequestBodyLBS body
  601                 , requestHeaders = [("content-type", "application/json")]
  602                 }
  603           eResp <- try (httpLbs req mgr)
  604           case eResp of
  605             Left (_ :: HttpException) ->
  606               pure (mkError q "custom" "WEB_SEARCH: HTTP request failed (connection or transport error)")
  607             Right resp -> do
  608               let code = statusCode (responseStatus resp)
  609                   respBody = responseBody resp
  610                   resultCount = countResults respBody
  611               if code >= 200 && code <= 299
  612                 then pure (OpResult [TrpText (bodyText respBody)] False (recorded' q "custom" resultCount))
  613                 else pure (mkError q "custom" $
  614                   "WEB_SEARCH: HTTP " <> T.pack (show code) <> ": " <> bodyText respBody)
  615 
  616 -- | Best-effort count of search results from the response JSON. Tries to
  617 -- parse the body as a JSON array and returns its length; falls back to 0
  618 -- when the body is not a JSON array (the count is metadata only and not
  619 -- security-critical).
  620 countResults :: BL.ByteString -> Int
  621 countResults bs =
  622   case A.decode bs :: Maybe Value of
  623     Just (A.Array arr) -> length arr
  624     Just (A.Object _)  -> 1  -- single result object
  625     _                  -> 0

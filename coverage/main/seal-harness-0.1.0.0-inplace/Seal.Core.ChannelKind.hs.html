    1 {-# LANGUAGE DeriveAnyClass #-}
    2 {-# LANGUAGE DeriveGeneric #-}
    3 {-# LANGUAGE OverloadedStrings #-}
    4 -- | The channel enumeration: the closed set of channels the runtime knows
    5 -- about, plus an 'Other' escape hatch for future channels. The lowercase
    6 -- tag from 'channelKindToText' is the value the transcript's @_te_metadata@
    7 -- @channel@ field carries.
    8 module Seal.Core.ChannelKind
    9   ( ChannelKind (..)
   10   , channelKindToText
   11   , channelKindFromText
   12   ) where
   13 
   14 import Data.Aeson (FromJSON, ToJSON)
   15 import Data.Text (Text)
   16 import Data.Text qualified as T
   17 import GHC.Generics (Generic)
   18 
   19 data ChannelKind
   20   = Cli | Web | Signal | Telegram | Background | Other
   21   deriving stock (Eq, Ord, Show, Enum, Bounded, Generic)
   22   deriving anyclass (ToJSON, FromJSON)
   23 
   24 -- | The lowercase tag for the transcript's metadata field.
   25 channelKindToText :: ChannelKind -> Text
   26 channelKindToText = \case
   27   Cli        -> "cli"
   28   Web        -> "web"
   29   Signal     -> "signal"
   30   Telegram   -> "telegram"
   31   Background -> "background"
   32   Other      -> "other"
   33 
   34 -- | Inverse of 'channelKindToText', case-insensitive. 'Nothing' for unknown
   35 -- tags (the caller decides whether to fall back to 'Other' or reject).
   36 channelKindFromText :: Text -> Maybe ChannelKind
   37 channelKindFromText t = case T.toCaseFold t of
   38   "cli"        -> Just Cli
   39   "web"        -> Just Web
   40   "signal"     -> Just Signal
   41   "telegram"   -> Just Telegram
   42   "background" -> Just Background
   43   "other"      -> Just Other
   44   _            -> Nothing

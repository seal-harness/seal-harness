    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | PROCESS_MANAGE (Untrusted): list or kill processes on the untrusted
    3 -- plane. @list@ runs @ps -o pid=,cmd=@ (fixed argv, bounded output); @kill@
    4 -- runs @kill <pid>@ with a validated positive-integer PID (rejects
    5 -- negative, self-pid). All IO through the 'UntrustedIO' seam; this module
    6 -- never imports 'System.Process'.
    7 module Seal.ISA.Ops.Process
    8   ( processManageOp
    9   ) where
   10 
   11 import Data.Aeson (Value, object, withObject, (.:), (.:?), (.=))
   12 import Data.Aeson.Types (parseMaybe)
   13 import Data.Text (Text)
   14 import Data.Text qualified as T

   18 import Seal.Providers.Class (ToolResultPart (..))

   23 -- | PROCESS_MANAGE opcode. Input: @{ action: "list" | "kill", pid?: Int }@.
   24 processManageOp :: WorkspaceRoot -> SecurityPolicy -> Opcode
   25 processManageOp _wsRoot policy = UntrustedOpcode
   26   { uoName = OpName "PROCESS_MANAGE"
   27   , uoDesc = "List or kill processes on the untrusted plane (bounded output, validated PID)."
   28   , uoInSchema = processManageSchema
   29   , uoOutSchema = object []
   30   , uoAuthorize = \v ->
   31       case actionField v of
   32         Nothing -> Left "PROCESS_MANAGE requires {action:string}"
   33         Just act
   34           | act == "list" -> checkAutonomy
   35           | act == "kill"  -> case pidField v of
   36               Nothing  -> Left "PROCESS_MANAGE: kill requires {pid:positive integer}"
   37               Just pid
   38                 | pid <= 0   -> Left "PROCESS_MANAGE: pid must be a positive integer"
   39                 | otherwise  -> checkAutonomy
   40           | otherwise -> Left ("PROCESS_MANAGE: unknown action \"" <> act <> "\"")
   41   , uoRun = \v -> do
   42       let act = actionField v
   43           mPid = pidField v
   44           recorded = object [ "action" .= act, "pid" .= mPid ]
   45       case act of
   46         Nothing -> pure (OpResult [TrpText "PROCESS_MANAGE: missing action"] True recorded)
   47         Just a
   48           | a == "list" -> do
   49               res <- uioProcessList
   50               pure $ case res of
   51                 Left err   -> OpResult [TrpText (renderUntrustedErr err)] True recorded
   52                 Right out -> OpResult [TrpText out] False recorded
   53           | a == "kill" -> case mPid of
   54               Nothing -> pure (OpResult [TrpText "PROCESS_MANAGE: missing pid"] True recorded)
   55               Just pid -> do
   56                 res <- uioProcessKill pid
   57                 pure $ case res of
   58                   Left err -> OpResult [TrpText (renderUntrustedErr err)] True recorded
   59                   Right _  -> OpResult [TrpText ("killed pid " <> T.pack (show pid))] False recorded
   60           | otherwise -> pure (OpResult [TrpText ("PROCESS_MANAGE: unknown action " <> a)] True recorded)
   61   }
   62   where
   63     checkAutonomy = case spAutonomy policy of
   64       Deny -> Left "PROCESS_MANAGE denied by autonomy policy"
   65       _   -> Right ()
   66 
   67 processManageSchema :: Value
   68 processManageSchema =
   69   object
   70     [ "type" .= ("object" :: Text)
   71     , "properties" .= object
   72         [ "action" .= object
   73             [ "type" .= ("string" :: Text)
   74             , "description" .= ("\"list\" (list processes) or \"kill\" (kill by PID)" :: Text)
   75             ]
   76         , "pid" .= object
   77             [ "type" .= ("integer" :: Text)
   78             , "description" .= ("Required for kill; positive integer." :: Text)
   79             ]
   80         ]

   82     ]

   84 actionField :: Value -> Maybe Text
   85 actionField = parseMaybe (withObject "in" (.: "action"))
   86 
   87 pidField :: Value -> Maybe Int
   88 pidField v = case parseMaybe (withObject "in" (.:? "pid")) v :: Maybe (Maybe Int) of
   89   Just (Just n) -> Just n
   90   _             -> Nothing

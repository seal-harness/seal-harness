    2 -- | The skill entry model. A *skill* is a named Markdown bundle (name +
    3 -- description + body) the agent can read into a prompt or update. 'SkillId' is
    4 -- a smart-constructed newtype with the charset predicate
    5 -- (@[A-Za-z0-9_\/-]+@, non-empty, no leading dot, no double\/trailing slash). The
    6 -- skill body is agent-visible data (not a vault secret); it is recorded in full
    7 -- in both the session transcript and the Audited log.

   10   , mkSkillId
   11   , isValidSkillId
   12   , skillIdText
   13   , skillIdSeparator
   14   , qualifiedSkillId

   16   , bareSkillIdText
   17   , Skill (..)
   18   ) where

   28 -- | Opaque skill key. Smart-constructed via 'mkSkillId'; the charset predicate
   29 -- guards every Audited-log / path / SQL-parameter position.
   30 newtype SkillId = SkillId Text
   31   deriving stock (Eq, Ord, Show)
   32   deriving newtype (ToJSON, FromJSON)
   33 
   34 -- | @[A-Za-z0-9_\/-]+@, non-empty, no leading dot, no double slash, no
   35 -- trailing slash. The forward slash is allowed so a skill id can carry its
   36 -- group as a prefix (e.g. @core\/dummy-test@), making the fully-qualified
   37 -- id unique across groups. A bare id (no slash) is also valid for ungrouped
   38 -- skills or when the group is unambiguous. Mirrors 'isValidMemoryId' but
   39 -- adds @\/@.
   40 isValidSkillId :: Text -> Bool
   41 isValidSkillId t =
   42   not (T.null t)
   43     && T.head t /= '.'
   44     && not (T.isInfixOf "//" t)
   45     && T.last t /= '/'
   46     && T.all (`elem` chars) t
   47   where
   48     chars = ['A' .. 'Z'] <> ['a' .. 'z'] <> ['0' .. '9'] <> "_-/"
   49 
   50 -- | The forward slash separator used to join a group and a bare id into a
   51 -- fully-qualified skill id.
   52 skillIdSeparator :: Char
   53 skillIdSeparator = '/'
   54 
   55 mkSkillId :: Text -> Either Text SkillId
   56 mkSkillId t
   57   | isValidSkillId t = Right (SkillId t)
   58   | otherwise        = Left ("invalid skill id: " <> T.pack (show t))
   59 
   60 -- | Construct a fully-qualified 'SkillId' from a group and a bare id.
   61 -- Returns the bare id as-is when the group is 'Nothing' or empty.
   62 qualifiedSkillId :: Maybe Text -> SkillId -> SkillId
   63 qualifiedSkillId Nothing sid = sid
   64 qualifiedSkillId (Just g) sid
   65   | T.null (T.strip g) = sid
   66   | otherwise = case mkSkillId (g <> T.singleton skillIdSeparator <> skillIdText sid) of
   67       Right fq -> fq
   68       Left _   -> sid  -- fail-soft: return bare id if the qualified form is invalid
   69 
   70 -- | Extract the group component from a 'SkillId' (the text before the
   71 -- first @\/@). Returns 'Nothing' for a bare id with no slash.
   72 skillIdGroup :: SkillId -> Maybe Text
   73 skillIdGroup (SkillId t) =
   74   case T.breakOn (T.singleton skillIdSeparator) t of
   75     (g, rest) | not (T.null rest) -> Just g
   76     _ -> Nothing
   77 
   78 -- | The bare (unqualified) id component — the text after the last @\/@.
   79 -- For a bare id (no slash), returns the id unchanged.
   80 bareSkillIdText :: SkillId -> Text
   81 bareSkillIdText (SkillId t) = snd (T.breakOnEnd (T.singleton skillIdSeparator) t)
   82 
   83 skillIdText :: SkillId -> Text
   84 skillIdText (SkillId t) = t
   85 
   86 -- | One agent skill. The body is agent-visible data (not a vault secret); it is
   87 -- recorded in full in both the session transcript and the Audited log.
   88 -- 'skSession' is the originating session (provenance).
   89 --
   90 -- 'skGroup' is an optional category for display grouping in the
   91 -- @\<available_skills\>@ catalog. It is derived from the on-disk parent
   92 -- directory name (@config\/skills\/\<group\>\/\<id\>.md@) and may be
   93 -- overridden via a @group:@ frontmatter key. 'Nothing' means the skill
   94 -- belongs to the default (ungrouped) section. The skill id may be either
   95 -- bare (e.g. @greet@) or fully-qualified with a group prefix
   96 -- (e.g. @core\/greet@) so two skills with the same bare name in different
   97 -- groups are distinct. The 'skGroup' field mirrors the group component of
   98 -- the id for display grouping when present.
   99 data Skill = Skill
  100   { skId          :: SkillId
  101   , skDescription :: Text
  102   , skBody        :: Text
  103   , skGroup       :: Maybe Text
  104   , skCreatedAt   :: UTCTime
  105   , skUpdatedAt   :: UTCTime
  106   , skSession     :: SessionId
  107   } deriving stock (Eq, Show, Generic)
  108 
  109 instance ToJSON Skill
  110 instance FromJSON Skill

    2 -- | TLS material for the gateway, generated with @openssl@.
    3 --
    4 -- Shelling out is deliberate and matches 'Seal.Security.Vault.Age' shelling out
    5 -- to @age@: X.509 generation is a large encoder that is easy to get subtly
    6 -- wrong, and a fixed-argv invocation of the system @openssl@ is both smaller
    7 -- and more likely to be correct than a hand-rolled one. The argv is never built
    8 -- from request data — the only interpolated values are the operator's own host
    9 -- names, passed as argv elements and, for the subject alternative names,
   10 -- through a generated OpenSSL config file.
   11 --
   12 -- __Two limits drive everything here.__ A browser rejects a certificate whose
   13 -- validity exceeds 398 days (@ERR_CERT_VALIDITY_TOO_LONG@) and rejects one
   14 -- whose names do not match the URL (@ERR_CERT_COMMON_NAME_INVALID@); neither is
   15 -- an interstitial the user can click through. So the generated pair is valid
   16 -- for 'certificateLifetimeDays' and carries every name and address the gateway
   17 -- can be reached at.
   18 --
   19 -- __A generated certificate does not make passkeys work by itself.__ Since
   20 -- Chrome 110 a browser disables WebAuthn outright on a certificate it does not
   21 -- trust, and clicking through the interstitial does not help. The certificate
   22 -- has to be installed on the client device, and it has to name a hostname: a
   23 -- WebAuthn RP ID can never be an IP address. This module exists so an operator
   24 -- can stand up encrypted transport without a proxy, not so that the trust step
   25 -- disappears.

  104 -- | The names to put in the certificate.

  129 -- | Build the SAN inputs from a gateway config plus what only the caller can
  130 -- discover (the hostname, the origin host, the interface addresses).
  131 sanInputsFor :: GatewayConfig -> Text -> Maybe Text -> [Text] -> SanInputs
  132 sanInputsFor cfg machineName originName addresses = SanInputs
  133   { siConfigured  = gcTlsHosts cfg
  134   , siBindHost    = gcHost cfg
  135   , siMachineName = machineName
  136   , siOriginName  = originName
  137   , siAddresses   = addresses
  138   }

  140 -- | The SAN list, in the order it is written (the first entry becomes the CN).
  141 --
  142 -- Operator-configured names come first because that is the name they said they
  143 -- use; everything else is derived, and @localhost@ plus both loopback literals
  144 -- are always present so a pair generated on this host is usable on this host.
  145 sanHostsFor :: SanInputs -> [Text]
  146 sanHostsFor si = dedupe (filter (not . T.null) ordered)
  147   where
  148     ordered =
  149       map normaliseName (siConfigured si)
  150         <> foldMap (pure . normaliseName) (siOriginName si)
  151         <> [normaliseName (siBindHost si) | namedHost]
  152         <> [normaliseName (siMachineName si) | machineNamed]
  153         <> map normaliseName (siAddresses si)
  154         <> ["localhost", "127.0.0.1", "::1"]
  155     namedHost = normaliseName (siBindHost si) `notElem` ["0.0.0.0", "::", "*", ""]
  156     machineNamed = not (T.null (normaliseName (siMachineName si)))
  157 
  158 -- | DNS names are case-insensitive, so they are lower-cased; an IPv6 zone id
  159 -- (@%en0@) is not part of the address and is dropped. Brackets are stripped
  160 -- because 'isIpLiteral' tolerates them but OpenSSL will not accept @[::1]@ as
  161 -- the value of an @IP.n@ entry.
  162 normaliseName :: Text -> Text
  163 normaliseName raw
  164   | T.isInfixOf ":" debracketed = fst (T.breakOn "%" debracketed)
  165   | otherwise                   = T.toLower debracketed
  166   where
  167     debracketed = T.dropAround (\c -> c == '[' || c == ']') (T.strip raw)
  168 
  169 -- | Pick the SAN key: @IP.n@ for a literal address, @DNS.n@ for a name.
  170 classifySanName :: Text -> SanName
  171 classifySanName raw
  172   | isIpLiteral normalised = SanIp normalised
  173   | otherwise              = SanDns normalised
  174   where
  175     normalised = normaliseName raw

  180 dedupe :: [Text] -> [Text]
  181 dedupe = go Set.empty
  182   where
  183     go _ [] = []
  184     go seen (x : xs)
  185       | Set.member (T.toLower x) seen = go seen xs
  186       | otherwise                     = x : go (Set.insert (T.toLower x) seen) xs

  188 -- | Filter and normalise the address texts an enumeration produced.
  189 --
  190 -- Pure and text-based on purpose: @Network.Info@ stores addresses in
  191 -- socket byte order, so its numeric fields are host-endianness dependent while
  192 -- its rendered form is not. Keeping the filtering on text means the rules
  193 -- (drop the all-zero "family not present" address, drop duplicates, agree with
  194 -- the compressed literals 'sanHostsFor' always adds) are testable without
  195 -- reproducing a byte order.
  196 addressNamesFor :: [Text] -> [Text]
  197 addressNamesFor = dedupe . filter isAssigned . map tidy
  198   where
  199     tidy = compressIpv6Text . T.toLower . T.strip
  200     isAssigned a = not (T.null a) && a /= "0.0.0.0" && a /= "::"
  201 
  202 -- | Compress the zero runs of a fully expanded IPv6 literal; any other text is
  203 -- returned unchanged. @Network.Info@ renders addresses in expanded form
  204 -- (@0:0:0:0:0:0:0:1@), and the loopback literals 'sanHostsFor' always adds are
  205 -- compressed (@::1@), so without this the same address would be listed twice.
  206 compressIpv6Text :: Text -> Text
  207 compressIpv6Text t =
  208   case traverse hexGroup (T.splitOn ":" t) of
  209     Just groups | length groups == 8 -> compressGroups groups
  210     _                                -> t
  211   where
  212     hexGroup g
  213       | T.null g  = Nothing
  214       | otherwise = case readHex (T.unpack g) of
  215           [(n, "")] -> Just n
  216           _         -> Nothing

  218 -- | Enumerate this machine's interface addresses.
  219 --
  220 -- A failure to enumerate is not a reason to refuse to start: the certificate
  221 -- still covers loopback, the hostname and anything in @tls_hosts@.
  222 localAddressNames :: IO [Text]
  223 localAddressNames = localAddressNamesWith getNetworkInterfaces

  225 -- | 'localAddressNames' with the enumeration action supplied, so a failing
  226 -- enumeration is a case a test can actually create.
  227 localAddressNamesWith :: IO [NetworkInterface] -> IO [Text]
  228 localAddressNamesWith enumerate = do
  229   result <- try enumerate :: IO (Either IOException [NetworkInterface])
  230   pure (either (const []) (addressNamesFor . concatMap rendered) result)
  231   where
  232     rendered iface = [T.pack (show (ipv4 iface)), T.pack (show (ipv6 iface))]

  234 -- | RFC 5952 text: lower-case hex groups, with the longest run of at least two
  235 -- zero groups compressed to @::@.
  236 compressGroups :: [Int] -> Text
  237 compressGroups groups =
  238   case longestZeroRun groups of
  239     Nothing -> T.intercalate ":" (map hex16 groups)
  240     Just (start, len) ->
  241       let (before, rest) = splitAt start groups
  242           after = drop len rest
  243       in T.intercalate ":" (map hex16 before) <> "::"
  244            <> T.intercalate ":" (map hex16 after)
  245 
  246 -- | Render the four 32-bit groups of an IPv6 address.
  247 renderIpv6 :: Word32 -> Word32 -> Word32 -> Word32 -> Text
  248 renderIpv6 a b c d = compressGroups (concatMap split32 [a, b, c, d])
  249   where
  250     split32 w = [fromIntegral (w `shiftR` 16), fromIntegral (w .&. 0xffff)] :: [Int]
  251 
  252 hex16 :: Int -> Text
  253 hex16 n = T.pack (showHex n "")
  254 
  255 longestZeroRun :: [Int] -> Maybe (Int, Int)
  256 longestZeroRun = pick . runs 0
  257   where
  258     runs _ [] = []
  259     runs i (x : xs)
  260       | x == 0 =
  261           let (zeros, rest) = span (== 0) (x : xs)
  262               n = length zeros
  263           in [(i, n) | n >= 2] <> runs (i + n) rest
  264       | otherwise = runs (i + 1) xs
  265     pick [] = Nothing
  266     -- The first longest run wins, which is what RFC 5952 requires.
  267     pick (r : rs) = Just (foldl' (\best@(_, bestLen) cur@(_, curLen)
  268                                    -> if curLen > bestLen then cur else best) r rs)

  270 -- | The OpenSSL config used for generation.
  271 --
  272 -- A config file rather than @-addext@: that option is OpenSSL 1.1.1+ only, and
  273 -- macOS still ships LibreSSL as @\/usr\/bin\/openssl@. @rsa:2048@ plus a config
  274 -- file works on both.
  275 --
  276 -- @CA:TRUE@ on a self-signed leaf is what lets the operator install this one
  277 -- file as a trusted root on a phone, which is the only way the certificate
  278 -- becomes acceptable to WebAuthn.
  279 renderOpensslConfig :: Text -> [Text] -> Text
  280 renderOpensslConfig primaryHost hosts = T.unlines $
  281   [ "[req]"
  282   , "prompt = no"
  283   , "distinguished_name = dn"
  284   , "x509_extensions = v3"
  285   , ""
  286   , "[dn]"
  287   , "CN = " <> primaryHost
  288   , ""
  289   , "[v3]"
  290   , "basicConstraints = critical, CA:TRUE"
  291   , "keyUsage = critical, digitalSignature, keyEncipherment, keyCertSign"
  292   , "extendedKeyUsage = serverAuth"
  293   , "subjectAltName = @alt"
  294   , ""
  295   , "[alt]"
  296   ] <> zipWith renderAlt [1 :: Int ..] hosts
  297   where
  298     renderAlt i host = case classifySanName host of
  299       SanIp ip  -> "IP." <> tshow i <> " = " <> ip
  300       SanDns nm -> "DNS." <> tshow i <> " = " <> nm

  302 -- | Operator guidance for the generated self-signed certificate.
  303 --
  304 -- The certificate is self-signed (with @CA:TRUE@, so installing it as a
  305 -- trusted root works). It has to be trusted by every browser that will run a
  306 -- WebAuthn ceremony; clicking through the interstitial does not count. This
  307 -- text is shared by @seal access-control configure@ and the @seal serve@
  308 -- startup warning so the two cannot drift. The exact trust command is
  309 -- OS-specific; the SSH-tunnel and Tailscale alternatives are the zero-trust
  310 -- paths for a harness on another machine.
  311 generatedCertificateTrustHint :: FilePath -> Text
  312 generatedCertificateTrustHint certPath = T.intercalate "\n" (header <> trust <> footer)
  313   where
  314     p = T.pack certPath
  315     header =
  316       [ "the generated certificate at " <> p <> " is self-signed."
  317       , "  Browsers disable WebAuthn until it is trusted; clicking through the warning is not enough."
  318       ]
  319     trust = case os of
  320       "darwin" ->
  321         [ "  Trust it in the System keychain, then fully quit the browser (Command-Q):"
  322         , "    sudo security add-trusted-cert -d -r trustRoot -k /Library/Keychains/System.keychain " <> p
  323         ]
  324       "mingw32" ->
  325         [ "  Trust it in the Windows Root store (admin PowerShell):"
  326         , "    certutil -addstore -f Root " <> p
  327         ]
  328       "linux" ->
  329         [ "  Trust it on this machine (Debian/Ubuntu):"
  330         , "    sudo cp " <> p <> " /usr/local/share/ca-certificates/seal.crt && sudo update-ca-certificates"
  331         , "  Fedora/RHEL: sudo trust anchor " <> p
  332         ]
  333       _ ->
  334         [ "  Install it in the OS trust store on every device that will use the gateway." ]
  335     footer =
  336       [ "  Other devices: install the same PEM and mark it trusted."
  337       , "  To avoid per-device trust entirely: bind loopback and use an SSH tunnel"
  338       , "  (`ssh -L 9090:localhost:8080 you@host`, then open http://localhost:9090),"
  339       , "  or use `tailscale serve` / a certificate from `tailscale cert <name>`."

  342 -- | The reason an existing pair cannot serve a legal certificate, if it cannot.
  343 --
  344 -- Pure, and deliberately narrow: it answers "would a browser reject this file
  345 -- outright", not "would a new one be better". Refreshing a merely stale name
  346 -- set is an operator action (delete the pair), not a silent re-key.
  347 certificateDefect :: (Day, Day) -> Maybe Text
  348 certificateDefect (notBefore, notAfter)
  349   | notAfter <= notBefore =
  350       Just "its expiry date is not after its start date"
  351   | window > toInteger certificateLifetimeDays =
  352       Just (tshow window <> " days of validity: browsers refuse more than "
  353               <> tshow certificateLifetimeDays <> " days")
  354   | otherwise = Nothing
  355   where
  356     window = diffDays notAfter notBefore

  358 -- | The validity window of the certificate in a PEM file, in whole days.
  359 readCertificateValidity :: FilePath -> IO (Either Text (Day, Day))
  360 readCertificateValidity path = do
  361   parsed <- try (readSignedObject path) :: IO (Either IOException [SignedCertificate])
  362   pure $ case parsed of
  363     Left e -> Left ("could not read " <> T.pack path <> ": " <> T.pack (show e))
  364     Right [] -> Left (T.pack path <> " holds no certificate")
  365     Right (c : _) ->
  366       case certValidity (getCertificate c) of
  367         (start, end) -> case (asDay start, asDay end) of
  368           (Just s, Just e) -> Right (s, e)
  369           _ -> Left ("the certificate in " <> T.pack path <> " has an unreadable validity window")
  370   where
  371     asDay dt =
  372       let date = dtDate dt
  373       in fromGregorianValid (toInteger (dateYear date))
  374                             (monthNumber (dateMonth date))
  375                             (dateDay date)
  376 
  377 -- | 1-based month number, spelled out rather than taken from 'Enum':
  378 -- @Data.Hourglass.Month@'s 'Enum' instance is zero-based, which silently
  379 -- shifts every parsed date by one month.
  380 monthNumber :: Month -> Int
  381 monthNumber = \case
  382   January   -> 1
  383   February  -> 2
  384   March     -> 3
  385   April     -> 4
  386   May       -> 5
  387   June      -> 6
  388   July      -> 7
  389   August    -> 8
  390   September -> 9
  391   October   -> 10
  392   November  -> 11
  393   December  -> 12

  395 -- | The SAN entries of the certificate in a PEM file.
  396 certificateSans :: FilePath -> IO [SanName]
  397 certificateSans path = do
  398   parsed <- try (readSignedObject path) :: IO (Either IOException [SignedCertificate])
  399   pure $ case parsed of
  400     Right (c : _) ->
  401       case extensionGet (certExtensions (getCertificate c)) of
  402         Just (ExtSubjectAltName alts) -> mapMaybe altName alts
  403         Nothing                       -> []
  404     _ -> []
  405   where
  406     altName (AltNameDNS d)  = Just (SanDns (T.pack d))
  407     altName (AltNameIP raw) = SanIp <$> renderAddressBytes raw
  408     altName _               = Nothing
  409 
  410 -- | Render the raw SAN address bytes the way 'renderIpv6' renders ours, so the
  411 -- two can be compared directly.
  412 renderAddressBytes :: ByteString -> Maybe Text
  413 renderAddressBytes raw
  414   | BS.length raw == 4 =
  415       Just (T.intercalate "." (map (tshow . toInteger) (BS.unpack raw)))
  416   | BS.length raw == 16 =
  417       Just (renderIpv6 (word32 0) (word32 1) (word32 2) (word32 3))
  418   | otherwise = Nothing
  419   where
  420     word32 i =
  421       foldl' (\acc k -> acc + (fromIntegral (BS.index raw (i * 4 + k)) `shiftL` (8 * (3 - k))))
  422              0 [0 .. 3 :: Int]

  424 -- | Does the pair load the way warp-tls will load it?
  425 --
  426 -- 'credentialLoadX509' is the same loader the TLS stack uses, so a truncated
  427 -- write or a certificate next to somebody else's key is caught at boot instead
  428 -- of on the first handshake.
  429 pairLoads :: FilePath -> FilePath -> IO (Either Text ())
  430 pairLoads certPath keyPath = do
  431   loaded <- try (credentialLoadX509 certPath keyPath)
  432   case loaded of
  433     Left (e :: IOException) -> pure (Left (T.pack (show e)))
  434     Right (Left err)        -> pure (Left (T.pack err))
  435     Right (Right _) -> do
  436       matches <- keyBelongsToCertificate certPath keyPath
  437       pure (if matches
  438               then Right ()
  439               else Left "the private key does not belong to the certificate")
  440 
  441 -- | Does the key in @keyPath@ pair with the certificate in @certPath@?
  442 --
  443 -- 'credentialLoadX509' parses both files but does not check that they belong
  444 -- together, so a cert next to somebody else's key loads happily and then fails
  445 -- every handshake. Comparing public keys closes that.
  446 --
  447 -- A key type with no comparable public half (anything but RSA, which is all
  448 -- this module generates) is reported as belonging: refusing to start over a key
  449 -- shape we do not model would be worse than the handshake error it prevents.
  450 keyBelongsToCertificate :: FilePath -> FilePath -> IO Bool
  451 keyBelongsToCertificate certPath keyPath = do
  452   certs <- try (readSignedObject certPath) :: IO (Either IOException [SignedCertificate])
  453   keys <- try (readKeyFile keyPath) :: IO (Either IOException [PrivKey])
  454   pure $ case (certs, keys) of
  455     (Right (c : _), Right (k : _)) ->
  456       samePublic (publicHalf k) (certPubKey (getCertificate c))
  457     _ -> True
  458   where
  459     samePublic (Just (PubKeyRSA a)) (PubKeyRSA b) = a == b
  460     samePublic _ _ = True
  461 
  462 publicHalf :: PrivKey -> Maybe PubKey
  463 publicHalf (PrivKeyRSA k) = Just (PubKeyRSA (private_pub k))
  464 publicHalf _              = Nothing

  474 -- | 'expiryWarning' when the certificate is within a month of expiry.
  475 certificateExpiryWarning :: FilePath -> IO (Maybe Text)
  476 certificateExpiryWarning certPath = do
  477   validity <- readCertificateValidity certPath
  478   today <- utctDay <$> getCurrentTime
  479   pure $ case validity of
  480     Right (_, notAfter)
  481       | diffDays notAfter today <= certificateExpiryWarningDays ->
  482           Just (expiryWarning notAfter)
  483     _ -> Nothing

  485 -- | How close to expiry is close enough to warn about. Part 1 has no rotation
  486 -- (that is the CA/leaf split), so this warning is the only thing standing
  487 -- between an operator and a listener that one day stops being accepted.
  488 certificateExpiryWarningDays :: Integer
  489 certificateExpiryWarningDays = 30

  511 ensurePair
  512   :: Bool -> FilePath -> FilePath -> FilePath -> [Text] -> IO (Either Text TlsCertOutcome)
  513 ensurePair manage binary certPath keyPath hosts = do
  514   certOk <- doesFileExist certPath
  515   keyOk  <- doesFileExist keyPath
  516   if not (certOk && keyOk)
  517     then generate False Nothing
  518     else if not manage
  519       then pure (Right TlsCertUnchanged)
  520       else do
  521         defect <- existingDefect
  522         case defect of
  523           Nothing   -> pure (Right TlsCertUnchanged)
  524           Just why  -> generate True (Just why)
  525   where
  526     existingDefect = do
  527       loaded <- pairLoads certPath keyPath
  528       case loaded of
  529         Left err -> pure (Just ("it does not load: " <> err))
  530         Right () -> do
  531           validity <- readCertificateValidity certPath
  532           pure (either Just certificateDefect validity)
  533 
  534     generate isReplace why = case hosts of
  535       [] -> pure (Left "no host names were given for the certificate")
  536       (primaryHost : _) -> do
  537           let dir = takeDirectory certPath
  538               confPath = certPath <> ".openssl.cnf"
  539               tmpCert = certPath <> ".tmp"
  540               tmpKey = keyPath <> ".tmp"
  541           createDirectoryIfMissing True dir
  542           setFileMode dir 0o700
  543           TIO.writeFile confPath (renderOpensslConfig primaryHost hosts)
  544           result <- try (readProcessWithExitCode binary (reqArgs confPath tmpCert tmpKey) "")
  545           _ <- try (removeFile confPath) :: IO (Either IOException ())
  546           case result of
  547             Left (e :: IOException) -> do
  548               discardTemps
  549               pure (Left ("could not run openssl: " <> T.pack (show e)
  550                             <> "\n  Install openssl, or point tls_cert/tls_key at an existing pair."))
  551             Right (ExitFailure code, _out, err) -> do
  552               discardTemps
  553               pure (Left ("openssl exited " <> tshow code <> ": " <> T.strip (T.pack err)))
  554             Right (ExitSuccess, _out, _err) -> do
  555               checked <- pairLoads tmpCert tmpKey
  556               case checked of
  557                 Left err -> do
  558                   discardTemps
  559                   pure (Left ("the generated pair did not load: " <> err))
  560                 Right () -> do
  561                   when isReplace keepPreviousPair
  562                   renameFile tmpCert certPath
  563                   renameFile tmpKey keyPath
  564                   -- openssl honours the umask; the key must not be world-readable
  565                   -- whatever it was.
  566                   setFileMode keyPath 0o600
  567                   pure (Right (if isReplace
  568                                  then TlsCertReplaced (fromMaybe "it could not serve a legal certificate" why)
  569                                  else TlsCertCreated))
  570       where
  571         reqArgs confPath tmpCert tmpKey =
  572           [ "req", "-x509", "-newkey", "rsa:2048", "-nodes"
  573           , "-days", show certificateLifetimeDays
  574           , "-keyout", tmpKey
  575           , "-out", tmpCert
  576           , "-config", confPath
  577           ]
  578         discardTemps =
  579           forM_ [certPath <> ".tmp", keyPath <> ".tmp"] $ \p ->
  580             void (try (removeFile p) :: IO (Either IOException ()))
  581         keepPreviousPair = do
  582           exists <- doesFileExist certPath
  583           when exists $ do
  584             copyFile certPath (certPath <> ".bak")
  585             keyExists <- doesFileExist keyPath
  586             when keyExists $ do
  587               copyFile keyPath (keyPath <> ".bak")
  588               setFileMode (keyPath <> ".bak") 0o600

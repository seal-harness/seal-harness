   45 -- | The per-subscriber state: the focused session (via an 'TVar' so the
   46 -- connection thread can update it on focus without re-subscribing), a
   47 -- send action, and an optional close action. The close action is called
   48 -- when 'subSend' throws (dead connection) so the WS socket is actively
   49 -- closed — this triggers the client's @onclose@ handler and reconnect
   50 -- rather than leaving a zombie connection that never receives another
   51 -- event but also never detects the failure.
   52 data Subscriber = Subscriber
   53   { subSessionRef :: TVar SessionId
   54   , subSend       :: BrokerEvent -> IO ()
   55   , subClose      :: IO ()
   56   }
   57 
   58 -- | The in-process broker. STM-backed: a 'TVar' of subscribers + a global
   59 -- cap. Also tracks the set of sessions currently in a @thinking@ turn so
   60 -- a freshly-connected web client can hydrate its sidebar without waiting
   61 -- for the next harness-status event (which would only arrive at the next
   62 -- turn boundary — leaving a mid-turn refresh stuck on Idle). The thinking
   63 -- map records /when/ thinking started so 'reconcileStaleThinking' can
   64 -- detect sessions stuck longer than a threshold (e.g. an unanswered
   65 -- ASK_HUMAN blocking the turn indefinitely).
   66 data StreamBroker = StreamBroker
   67   { sbSubs :: TVar [Subscriber]
   68   , sbCap :: Int
   69   , sbThinking :: TVar (Map.Map SessionId UTCTime)
   70   , sbEntryCursors :: TVar (Map.Map SessionId Int)
   71   }
   72 
   73 -- | Build a new broker with the given subscriber cap.
   74 newStreamBroker :: Int -> IO StreamBroker
   75 newStreamBroker cap =
   76   StreamBroker <$> newTVarIO [] <*> pure cap <*> newTVarIO Map.empty <*> newTVarIO Map.empty

   78 -- | Read the per-session entry broadcast cursor: the number of entries
   79 -- already fanned out for this session. Zero for a never-broadcast session.
   80 readEntryCursor :: StreamBroker -> SessionId -> IO Int
   81 readEntryCursor broker sid =
   82   fromMaybe 0 . Map.lookup sid <$> readTVarIO (sbEntryCursors broker)
   83 
   84 -- | Incremental entry broadcast (issue #198): given the session's FULL
   85 -- frontend-shaped transcript (as 'readTranscriptEntries' returns it),
   86 -- return only the entries that have not been broadcast for this session
   87 -- yet, and advance the per-session cursor. The cursor is positional
   88 -- (the number of entries already sent), so successive calls during a
   89 -- turn fan out each entry exactly once — linear broadcast volume instead
   90 -- of the historical O(N²) full-transcript re-send per recorded entry.
   91 --
   92 -- A cursor AHEAD of the transcript (the session was rebuilt / the
   93 -- transcript shrank) clamps to zero and resends everything: subscribers
   94 -- see an idempotent replay rather than silently missing entries. The web
   95 -- frontend dedupes by entry id, and idempotent replays are the failure
   96 -- mode every consumer already tolerates.
   97 --
   98 -- Concurrency: 'Seal.Core.TurnEngine.runTurnBody' runs turns under
   99 -- 'withSessionLock' so per-session calls are serialized; the atomic
  100 -- stateTVar below keeps the cursor consistent even when the
  101 -- slash-command writers ('mkModelTranscriptWriter',
  102 -- 'mkStopTranscriptWriter') race a turn.
  103 takeNewEntries
  104   :: StreamBroker -> SessionId -> [(Int, a)] -> IO [(Int, a)]
  105 takeNewEntries broker sid entries = do
  106   let total = length entries
  107   start <- atomically $ stateTVar (sbEntryCursors broker) $ \cursors ->
  108     let prior = fromMaybe 0 (Map.lookup sid cursors)
  109     in if prior > total
  110          then (0, Map.insert sid total cursors)
  111          else (prior, Map.insert sid total cursors)
  112   pure (drop start entries)

  114 -- | Subscribe a new connection. If the global cap is exceeded, the subscribe
  115 -- is a no-op (the over-cap subscriber is never added — it should close).
  116 -- Returns the session 'TVar' so the caller can update the focused session
  117 -- via 'updateSubscriberSession' when the client sends a @focus@ op.
  118 --
  119 -- The @close@ action is called when the broker detects this subscriber's
  120 -- send has thrown (dead connection), so the WS socket is actively closed
  121 -- and the client reconnects. Pass @pure ()@ when no cleanup is needed
  122 -- (tests).
  123 subscribe :: StreamBroker -> SessionId -> (BrokerEvent -> IO ()) -> IO () -> IO (TVar SessionId)
  124 subscribe broker session sendfn closefn = do
  125   ref <- newTVarIO session
  126   atomically $ do
  127     subs <- readTVar (sbSubs broker)
  128     when (length subs < sbCap broker) $
  129       writeTVar (sbSubs broker) (subs <> [Subscriber ref sendfn closefn])
  130   pure ref
  131 
  132 -- | Update a subscriber's focused session. Called when the client sends a
  133 -- @focus@ op so subsequent 'BeEntryRecorded' events for the new session are
  134 -- delivered.
  135 updateSubscriberSession :: TVar SessionId -> SessionId -> IO ()
  136 updateSubscriberSession ref sid = atomically (writeTVar ref sid)
  137 
  138 -- | Fan one event to every subscriber whose focused session matches. For
  139 -- 'BeListsSnapshot' (a broadcast to all), every subscriber receives it
  140 -- regardless of focus.
  141 --
  142 -- A subscriber whose 'subSend' throws (e.g. a closed WebSocket connection
  143 -- raising 'Network.WebSockets.ConnectionClosed') is pruned from the
  144 -- subscriber list and its 'subClose' action is invoked. Closing the
  145 -- connection actively (rather than just silently removing it from the
  146 -- list) ensures the client's @onclose@ fires and it reconnects — without
  147 -- this, a dead connection would linger as a zombie: the reader loop is
  148 -- still alive, the client thinks it's connected, but no events arrive.
  149 -- The close action is best-effort (swallowed if it throws) so a failure
  150 -- during cleanup never propagates to the caller (e.g. a @seal serve@
  151 -- request thread running 'triggerBroadcast' after a slash command).
  152 broadcast :: StreamBroker -> BrokerEvent -> IO ()
  153 broadcast broker event = do
  154   subs <- readTVarIO (sbSubs broker)
  155   live <- filterM (deliverTo event) subs
  156   -- Drop any subscribers whose send threw OR timed out (dead / slow
  157   -- connections). Call their close actions so the WS socket is actively
  158   -- closed and the client reconnects. The length check avoids a needless
  159   -- STM write when everyone survived.
  160   when (length live < length subs) $ do
  161     let dead = filter (\s -> subSessionRef s `notElem` map subSessionRef live) subs
  162     forM_ dead $ \s -> subClose s `catch` \(_ :: SomeException) -> pure ()
  163     atomically $ writeTVar (sbSubs broker) live
  164   where
  165     deliverTo ev s =
  166       (do
  167          ok <- shouldSend ev s
  168          if ok
  169            then do
  170              -- Bound the send: a peer that stops draining its socket (a
  171              -- zombie/half-open WS, or a backgrounded tab under TCP
  172              -- backpressure) makes 'sendTextData' block indefinitely.
  173              -- Without this timeout the *caller* blocks — and callers include
  174              -- HTTP request handlers that broadcast synchronously (e.g.
  175              -- handleTabNew -> triggerBroadcast), so a stuck WS would hang
  176              -- POST /api/tabs/new forever. A timed-out send is treated as a
  177              -- dead subscriber (pruned + closed, so the client reconnects).
  178              sent <- timeout sendTimeoutMicros (subSend s ev)
  179              pure (case sent of
  180                      Just () -> True
  181                      Nothing -> False)
  182            else pure True)
  183         `catch` \(_e :: SomeException) -> pure False
  184     -- 2 seconds: generous for a healthy client, short enough that a single
  185     -- stuck subscriber cannot stall a request thread for long.
  186     sendTimeoutMicros :: Int
  187     sendTimeoutMicros = 2_000_000
  188     -- All-subscriber events vs session-filtered events.
  189     -- BeActivity is ALL-subscriber: the sidebar renders tab status for
  190     -- EVERY open tab, so a turn on a channel-originated session (e.g.
  191     -- Telegram) must surface to a web client focused on a different
  192     -- session. The activity envelope carries its own sessionId, so the
  193     -- frontend's useSessionActivityStream keys it per-session without
  194     -- relying on the broker's focus filter.
  195     shouldSend ev s = case ev of
  196       BeListsSnapshot _  -> pure True
  197       BeHarnessStatus _  -> pure True
  198       BeActivity _ _      -> pure True
  199       BeAgentDefsChanged  -> pure True
  200       BeSkillsChanged     -> pure True
  201       BeReposChanged      -> pure True
  202       BeEntryRecorded sid _ -> matchSession s sid
  203       BeEntryUpdate sid _   -> matchSession s sid
  204       BeAsk sid _          -> matchSession s sid
  205       BeAskResolved sid _  -> matchSession s sid
  206     matchSession s sid = do
  207       subSid <- readTVarIO (subSessionRef s)
  208       pure (subSid == sid)

  232 -- | Read the set of sessions currently in a @thinking@ turn. Used by the
  233 -- lists-snapshot builders to hydrate a freshly-connected web client's
  234 -- sidebar (so a mid-turn refresh does not blank the thinking indicator).
  235 thinkingSessions :: StreamBroker -> IO (Set SessionId)
  236 thinkingSessions broker =
  237   Map.keysSet <$> readTVarIO (sbThinking broker)

  239 -- | Add ('True') or remove ('False') a session from the thinking set.
  240 -- Idempotent. Called by 'broadcastHarnessStatus' so the broker's
  241 -- in-memory state mirrors the events it fans out. When adding, records
  242 -- the current time so 'reconcileStaleThinking' can detect sessions that
  243 -- have been stuck thinking longer than a threshold (e.g. an unanswered
  244 -- ASK_HUMAN blocking the turn indefinitely — session
  245 -- 20260912-183908-767 issue #3).
  246 setThinking :: StreamBroker -> SessionId -> Bool -> IO ()
  247 setThinking broker sid thinking = do
  248   now <- getCurrentTime
  249   atomically $ modifyTVar' (sbThinking broker)
  250     (\m -> if thinking then Map.insert sid now m else Map.delete sid m)
  251 
  252 -- | Remove sessions that have been in the thinking set longer than the
  253 -- given threshold (in seconds). Returns the set of sessions that were
  254 -- cleared. Broadcasts an @idle@ harness-status for each cleared session
  255 -- so the web frontend updates. This is a safety net: a session blocked
  256 -- on an unanswered ASK_HUMAN (or a dead provider connection that slipped
  257 -- past the stream timeout) leaves the session permanently thinking in
  258 -- the broker's in-memory state. Without reconciliation, a page refresh
  259 -- shows the session as thinking forever.
  260 reconcileStaleThinking :: StreamBroker -> Int -> IO (Set SessionId)
  261 reconcileStaleThinking broker maxAgeSec = do
  262   now <- getCurrentTime
  263   let threshold = fromIntegral maxAgeSec :: Double
  264   stale <- atomically $ do
  265     m <- readTVar (sbThinking broker)
  266     let isStale t = realToFrac (now `diffUTCTime` t) >= threshold
  267         staleMap = Map.filter isStale m
  268         staleSids = Map.keysSet staleMap
  269     unless (Set.null staleSids) $
  270       writeTVar (sbThinking broker) (Map.difference m staleMap)
  271     pure staleSids
  272   -- Broadcast idle for each cleared session so the frontend updates.
  273   forM_ (Set.toList stale) $ \sid ->
  274     broadcast broker (BeActivity sid (object
  275       [ "kind" .= ("harness-status" :: Text)
  276       , "status" .= ("idle" :: Text)
  277       ]))
  278   pure stale

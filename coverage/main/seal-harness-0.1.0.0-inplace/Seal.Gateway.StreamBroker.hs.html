   30 -- | One event the broker fans out to subscribers.
   31 data BrokerEvent
   32   = BeEntryRecorded SessionId Value   -- ^ a transcript entry (the JSON the WS peer receives)
   33   | BeEntryUpdate SessionId Value      -- ^ a streaming entry whose text is still growing (the WS peer renders as an @entry-update@ event, replacing the in-place entry by id)
   34   | BeHarnessStatus Value             -- ^ a harness liveness change
   35   | BeListsSnapshot Value             -- ^ a refreshed tab/session snapshot
   36   | BeAsk SessionId Value             -- ^ a pending human-question from ASK_HUMAN (the JSON the WS peer renders)
   37   | BeAskResolved SessionId Value      -- ^ a pending question was answered/cancelled (the JSON carries the ask id)
   38   | BeActivity SessionId Value          -- ^ a per-session activity signal (harness-status / reply-delivered) the WS peer renders as an @activity@ envelope
   39   | BeAgentDefsChanged                 -- ^ agent defs were created/updated/deleted; clients should re-fetch
   40   | BeSkillsChanged                    -- ^ skills were created/updated/deleted; clients should re-fetch
   41   | BeReposChanged                     -- ^ the source-control repo registry was mutated; clients should re-fetch /api/repos
   42   deriving stock (Eq, Show)
   43 
   44 -- | The per-subscriber state: the focused session (via an 'TVar' so the
   45 -- connection thread can update it on focus without re-subscribing), a
   46 -- send action, and an optional close action. The close action is called
   47 -- when 'subSend' throws (dead connection) so the WS socket is actively
   48 -- closed — this triggers the client's @onclose@ handler and reconnect
   49 -- rather than leaving a zombie connection that never receives another
   50 -- event but also never detects the failure.
   51 data Subscriber = Subscriber
   52   { subSessionRef :: TVar SessionId
   53   , subSend       :: BrokerEvent -> IO ()
   54   , subClose      :: IO ()
   55   }
   56 
   57 -- | The in-process broker. STM-backed: a 'TVar' of subscribers + a global
   58 -- cap. Also tracks the set of sessions currently in a @thinking@ turn so
   59 -- a freshly-connected web client can hydrate its sidebar without waiting
   60 -- for the next harness-status event (which would only arrive at the next
   61 -- turn boundary — leaving a mid-turn refresh stuck on Idle).
   62 data StreamBroker = StreamBroker
   63   { sbSubs :: TVar [Subscriber]
   64   , sbCap :: Int
   65   , sbThinking :: TVar (Set SessionId)
   66   }
   67 
   68 -- | Build a new broker with the given subscriber cap.
   69 newStreamBroker :: Int -> IO StreamBroker
   70 newStreamBroker cap =
   71   StreamBroker <$> newTVarIO [] <*> pure cap <*> newTVarIO Set.empty
   72 
   73 -- | Subscribe a new connection. If the global cap is exceeded, the subscribe
   74 -- is a no-op (the over-cap subscriber is never added — it should close).
   75 -- Returns the session 'TVar' so the caller can update the focused session
   76 -- via 'updateSubscriberSession' when the client sends a @focus@ op.
   77 --
   78 -- The @close@ action is called when the broker detects this subscriber's
   79 -- send has thrown (dead connection), so the WS socket is actively closed
   80 -- and the client reconnects. Pass @pure ()@ when no cleanup is needed
   81 -- (tests).
   82 subscribe :: StreamBroker -> SessionId -> (BrokerEvent -> IO ()) -> IO () -> IO (TVar SessionId)
   83 subscribe broker session sendfn closefn = do
   84   ref <- newTVarIO session
   85   atomically $ do
   86     subs <- readTVar (sbSubs broker)
   87     when (length subs < sbCap broker) $
   88       writeTVar (sbSubs broker) (subs <> [Subscriber ref sendfn closefn])
   89   pure ref
   90 
   91 -- | Update a subscriber's focused session. Called when the client sends a
   92 -- @focus@ op so subsequent 'BeEntryRecorded' events for the new session are
   93 -- delivered.
   94 updateSubscriberSession :: TVar SessionId -> SessionId -> IO ()
   95 updateSubscriberSession ref sid = atomically (writeTVar ref sid)
   96 
   97 -- | Fan one event to every subscriber whose focused session matches. For
   98 -- 'BeListsSnapshot' (a broadcast to all), every subscriber receives it
   99 -- regardless of focus.
  100 --
  101 -- A subscriber whose 'subSend' throws (e.g. a closed WebSocket connection
  102 -- raising 'Network.WebSockets.ConnectionClosed') is pruned from the
  103 -- subscriber list and its 'subClose' action is invoked. Closing the
  104 -- connection actively (rather than just silently removing it from the
  105 -- list) ensures the client's @onclose@ fires and it reconnects — without
  106 -- this, a dead connection would linger as a zombie: the reader loop is
  107 -- still alive, the client thinks it's connected, but no events arrive.
  108 -- The close action is best-effort (swallowed if it throws) so a failure
  109 -- during cleanup never propagates to the caller (e.g. a @seal serve@
  110 -- request thread running 'triggerBroadcast' after a slash command).
  111 broadcast :: StreamBroker -> BrokerEvent -> IO ()
  112 broadcast broker event = do
  113   subs <- readTVarIO (sbSubs broker)
  114   live <- filterM (deliverTo event) subs
  115   -- Drop any subscribers whose send threw (dead connections). Call their
  116   -- close actions so the WS socket is actively closed and the client
  117   -- reconnects. The length check avoids a needless STM write when everyone
  118   -- survived.
  119   when (length live < length subs) $ do
  120     let dead = filter (\s -> subSessionRef s `notElem` map subSessionRef live) subs
  121     forM_ dead $ \s -> subClose s `catch` \(_ :: SomeException) -> pure ()
  122     atomically $ writeTVar (sbSubs broker) live
  123   where
  124     deliverTo ev s =
  125       (do
  126          ok <- shouldSend ev s
  127          if ok then subSend s ev >> pure True else pure True)
  128         `catch` \(_e :: SomeException) -> pure False
  129     -- All-subscriber events vs session-filtered events.
  130     -- BeActivity is ALL-subscriber: the sidebar renders tab status for
  131     -- EVERY open tab, so a turn on a channel-originated session (e.g.
  132     -- Telegram) must surface to a web client focused on a different
  133     -- session. The activity envelope carries its own sessionId, so the
  134     -- frontend's useSessionActivityStream keys it per-session without
  135     -- relying on the broker's focus filter.
  136     shouldSend ev s = case ev of
  137       BeListsSnapshot _  -> pure True
  138       BeHarnessStatus _  -> pure True
  139       BeActivity _ _      -> pure True
  140       BeAgentDefsChanged  -> pure True
  141       BeSkillsChanged     -> pure True
  142       BeReposChanged      -> pure True
  143       BeEntryRecorded sid _ -> matchSession s sid
  144       BeEntryUpdate sid _   -> matchSession s sid
  145       BeAsk sid _          -> matchSession s sid
  146       BeAskResolved sid _  -> matchSession s sid
  147     matchSession s sid = do
  148       subSid <- readTVarIO (subSessionRef s)
  149       pure (subSid == sid)
  150 
  151 -- | Push a refreshed tab/session snapshot to every connection.
  152 broadcastLists :: StreamBroker -> Value -> IO ()
  153 broadcastLists broker snap = broadcast broker (BeListsSnapshot snap)
  154 -- | Push an @agent-defs-changed@ invalidation signal to every connection.
  155 -- All subscribers receive it (agent defs are not session-scoped).
  156 broadcastAgentDefsChanged :: StreamBroker -> IO ()
  157 broadcastAgentDefsChanged broker = broadcast broker BeAgentDefsChanged
  158 
  159 -- | Push a @skills-changed@ invalidation signal to every connection.
  160 -- All subscribers receive it (skills are not session-scoped).
  161 broadcastSkillsChanged :: StreamBroker -> IO ()
  162 broadcastSkillsChanged broker = broadcast broker BeSkillsChanged
  163 
  164 -- | Push a @repos-changed@ invalidation signal to every connection.
  165 -- All subscribers receive it (the repo registry is not session-scoped).
  166 broadcastReposChanged :: StreamBroker -> IO ()
  167 broadcastReposChanged broker = broadcast broker BeReposChanged

  169 -- | The current subscriber count (for diagnostics / the global cap check).
  170 subscriberCount :: StreamBroker -> IO Int
  171 subscriberCount broker = length <$> readTVarIO (sbSubs broker)

  173 -- | Read the set of sessions currently in a @thinking@ turn. Used by the
  174 -- lists-snapshot builders to hydrate a freshly-connected web client's
  175 -- sidebar (so a mid-turn refresh does not blank the thinking indicator).
  176 thinkingSessions :: StreamBroker -> IO (Set SessionId)
  177 thinkingSessions broker = readTVarIO (sbThinking broker)
  178 
  179 -- | Add ('True') or remove ('False') a session from the thinking set.
  180 -- Idempotent. Called by 'broadcastHarnessStatus' so the broker's
  181 -- in-memory state mirrors the events it fans out.
  182 setThinking :: StreamBroker -> SessionId -> Bool -> IO ()
  183 setThinking broker sid thinking =
  184   atomically $ modifyTVar' (sbThinking broker)
  185     (\s -> if thinking then Set.insert sid s else Set.delete sid s)

   51 -- | The per-subscriber state: the focused session (via an 'TVar' so the
   52 -- connection thread can update it on focus without re-subscribing), a
   53 -- send action, and an optional close action. The close action is called
   54 -- when 'subSend' throws (dead connection) so the WS socket is actively
   55 -- closed — this triggers the client's @onclose@ handler and reconnect
   56 -- rather than leaving a zombie connection that never receives another
   57 -- event but also never detects the failure.
   58 data Subscriber = Subscriber
   59   { subSessionRef :: TVar SessionId
   60   , subSend       :: BrokerEvent -> IO ()
   61   , subClose      :: IO ()
   62   }
   63 
   64 -- | The in-process broker. STM-backed: a 'TVar' of subscribers + a global
   65 -- cap. Also tracks the set of sessions currently in a @thinking@ turn so
   66 -- a freshly-connected web client can hydrate its sidebar without waiting
   67 -- for the next harness-status event (which would only arrive at the next
   68 -- turn boundary — leaving a mid-turn refresh stuck on Idle). The thinking
   69 -- map records /when the session was last seen doing something/ (turn
   70 -- start via 'setThinking', then every session-scoped broadcast via
   71 -- 'touchThinking') so 'reconcileStaleThinking' can tell a quiet session
   72 -- from a stalled one — a long turn that keeps emitting tool calls is
   73 -- working, not stuck.
   74 data StreamBroker = StreamBroker
   75   { sbSubs :: TVar [Subscriber]
   76   , sbCap :: Int
   77   , sbThinking :: TVar (Map.Map SessionId UTCTime)
   78   , sbEntryCursors :: TVar (Map.Map SessionId Int)
   79   }
   80 
   81 -- | Build a new broker with the given subscriber cap.
   82 newStreamBroker :: Int -> IO StreamBroker
   83 newStreamBroker cap =
   84   StreamBroker <$> newTVarIO [] <*> pure cap <*> newTVarIO Map.empty <*> newTVarIO Map.empty

   86 -- | Read the per-session entry broadcast cursor: the number of entries
   87 -- already fanned out for this session. Zero for a never-broadcast session.
   88 readEntryCursor :: StreamBroker -> SessionId -> IO Int
   89 readEntryCursor broker sid =
   90   fromMaybe 0 . Map.lookup sid <$> readTVarIO (sbEntryCursors broker)

   92 -- | Incremental entry broadcast (issue #198): given the session's FULL
   93 -- frontend-shaped transcript (as 'readTranscriptEntries' returns it),
   94 -- return only the entries that have not been broadcast for this session
   95 -- yet, and advance the per-session cursor. The cursor is positional
   96 -- (the number of entries already sent), so successive calls during a
   97 -- turn fan out each entry exactly once — linear broadcast volume instead
   98 -- of the historical O(N²) full-transcript re-send per recorded entry.
   99 --
  100 -- A cursor AHEAD of the transcript (the session was rebuilt / the
  101 -- transcript shrank) clamps to zero and resends everything: subscribers
  102 -- see an idempotent replay rather than silently missing entries. The web
  103 -- frontend dedupes by entry id, and idempotent replays are the failure
  104 -- mode every consumer already tolerates.
  105 --
  106 -- Concurrency: 'Seal.Core.TurnEngine.runTurnBody' runs turns under
  107 -- 'withSessionLock' so per-session calls are serialized; the atomic
  108 -- stateTVar below keeps the cursor consistent even when the
  109 -- slash-command writers ('mkModelTranscriptWriter',
  110 -- 'mkStopTranscriptWriter') race a turn.
  111 takeNewEntries
  112   :: StreamBroker -> SessionId -> [(Int, a)] -> IO [(Int, a)]
  113 takeNewEntries broker sid entries = do
  114   let total = length entries
  115   start <- atomically $ stateTVar (sbEntryCursors broker) $ \cursors ->
  116     let prior = fromMaybe 0 (Map.lookup sid cursors)
  117     in if prior > total
  118          then (0, Map.insert sid total cursors)
  119          else (prior, Map.insert sid total cursors)
  120   pure (drop start entries)

  122 -- | Subscribe a new connection. If the global cap is exceeded, the subscribe
  123 -- is a no-op (the over-cap subscriber is never added — it should close).
  124 -- Returns the session 'TVar' so the caller can update the focused session
  125 -- via 'updateSubscriberSession' when the client sends a @focus@ op.
  126 --
  127 -- The @close@ action is called when the broker detects this subscriber's
  128 -- send has thrown (dead connection), so the WS socket is actively closed
  129 -- and the client reconnects. Pass @pure ()@ when no cleanup is needed
  130 -- (tests).
  131 subscribe :: StreamBroker -> SessionId -> (BrokerEvent -> IO ()) -> IO () -> IO (TVar SessionId)
  132 subscribe broker session sendfn closefn = do
  133   ref <- newTVarIO session
  134   atomically $ do
  135     subs <- readTVar (sbSubs broker)
  136     when (length subs < sbCap broker) $
  137       writeTVar (sbSubs broker) (subs <> [Subscriber ref sendfn closefn])
  138   pure ref

  140 -- | Update a subscriber's focused session. Called when the client sends a
  141 -- @focus@ op so subsequent 'BeEntryRecorded' events for the new session are
  142 -- delivered.
  143 updateSubscriberSession :: TVar SessionId -> SessionId -> IO ()
  144 updateSubscriberSession ref sid = atomically (writeTVar ref sid)

  146 -- | Fan one event to every subscriber whose focused session matches. For
  147 -- 'BeListsSnapshot' (a broadcast to all), every subscriber receives it
  148 -- regardless of focus.
  149 --
  150 -- A subscriber whose 'subSend' throws (e.g. a closed WebSocket connection
  151 -- raising 'Network.WebSockets.ConnectionClosed') is pruned from the
  152 -- subscriber list and its 'subClose' action is invoked. Closing the
  153 -- connection actively (rather than just silently removing it from the
  154 -- list) ensures the client's @onclose@ fires and it reconnects — without
  155 -- this, a dead connection would linger as a zombie: the reader loop is
  156 -- still alive, the client thinks it's connected, but no events arrive.
  157 -- The close action is best-effort (swallowed if it throws) so a failure
  158 -- during cleanup never propagates to the caller (e.g. a @seal serve@
  159 -- request thread running 'triggerBroadcast' after a slash command).
  160 --
  161 -- Every session-scoped event is also proof of life for that session's
  162 -- turn: it refreshes the session's last-activity stamp (see
  163 -- 'touchThinking') so the stale-thinking sweep measures /silence/, not
  164 -- turn duration.
  165 broadcast :: StreamBroker -> BrokerEvent -> IO ()
  166 broadcast broker event = do
  167   forM_ (eventSession event) (touchThinking broker)
  168   subs <- readTVarIO (sbSubs broker)
  169   live <- filterM (deliverTo event) subs
  170   -- Drop any subscribers whose send threw OR timed out (dead / slow
  171   -- connections). Call their close actions so the WS socket is actively
  172   -- closed and the client reconnects. The length check avoids a needless
  173   -- STM write when everyone survived.
  174   when (length live < length subs) $ do
  175     let dead = filter (\s -> subSessionRef s `notElem` map subSessionRef live) subs
  176     forM_ dead $ \s -> subClose s `catch` \(_ :: SomeException) -> pure ()
  177     atomically $ writeTVar (sbSubs broker) live
  178   where
  179     deliverTo ev s =
  180       (do
  181          ok <- shouldSend ev s
  182          if ok
  183            then do
  184              -- Bound the send: a peer that stops draining its socket (a
  185              -- zombie/half-open WS, or a backgrounded tab under TCP
  186              -- backpressure) makes 'sendTextData' block indefinitely.
  187              -- Without this timeout the *caller* blocks — and callers include
  188              -- HTTP request handlers that broadcast synchronously (e.g.
  189              -- handleTabNew -> triggerBroadcast), so a stuck WS would hang
  190              -- POST /api/tabs/new forever. A timed-out send is treated as a
  191              -- dead subscriber (pruned + closed, so the client reconnects).
  192              sent <- timeout sendTimeoutMicros (subSend s ev)
  193              pure (case sent of
  194                      Just () -> True
  195                      Nothing -> False)
  196            else pure True)
  197         `catch` \(_e :: SomeException) -> pure False
  198     -- 2 seconds: generous for a healthy client, short enough that a single
  199     -- stuck subscriber cannot stall a request thread for long.
  200     sendTimeoutMicros :: Int
  201     sendTimeoutMicros = 2_000_000
  202     -- All-subscriber events vs session-filtered events.
  203     -- BeActivity is ALL-subscriber: the sidebar renders tab status for
  204     -- EVERY open tab, so a turn on a channel-originated session (e.g.
  205     -- Telegram) must surface to a web client focused on a different
  206     -- session. The activity envelope carries its own sessionId, so the
  207     -- frontend's useSessionActivityStream keys it per-session without
  208     -- relying on the broker's focus filter.
  209     shouldSend ev s = case ev of
  210       BeListsSnapshot _  -> pure True
  211       BeHarnessStatus _  -> pure True
  212       BeActivity _ _      -> pure True
  213       BeAgentDefsChanged  -> pure True
  214       BeSkillsChanged     -> pure True
  215       BeReposChanged      -> pure True
  216       BeEntryRecorded sid _ -> matchSession s sid
  217       BeEntryUpdate sid _   -> matchSession s sid
  218       BeAsk sid _          -> matchSession s sid
  219       BeAskResolved sid _  -> matchSession s sid
  220     matchSession s sid = do
  221       subSid <- readTVarIO (subSessionRef s)
  222       pure (subSid == sid)

  224 -- | The session a broker event pertains to, if any. Used by 'broadcast'
  225 -- to refresh the session's last-activity stamp. 'BeListsSnapshot' and the
  226 -- three global invalidation signals are not session-scoped; the legacy
  227 -- 'BeHarnessStatus' (never emitted by the current dispatch path) carries
  228 -- no session id either.
  229 eventSession :: BrokerEvent -> Maybe SessionId
  230 eventSession = \case
  231   BeEntryRecorded sid _ -> Just sid
  232   BeEntryUpdate sid _   -> Just sid
  233   BeAsk sid _           -> Just sid
  234   BeAskResolved sid _   -> Just sid
  235   BeActivity sid _      -> Just sid
  236   BeHarnessStatus _     -> Nothing
  237   BeListsSnapshot _     -> Nothing
  238   BeAgentDefsChanged    -> Nothing
  239   BeSkillsChanged       -> Nothing
  240   BeReposChanged        -> Nothing

  264 -- | Read the set of sessions currently in a @thinking@ turn. Used by the
  265 -- lists-snapshot builders to hydrate a freshly-connected web client's
  266 -- sidebar (so a mid-turn refresh does not blank the thinking indicator).
  267 thinkingSessions :: StreamBroker -> IO (Set SessionId)
  268 thinkingSessions broker =
  269   Map.keysSet <$> readTVarIO (sbThinking broker)

  271 -- | Add ('True') or remove ('False') a session from the thinking set.
  272 -- Idempotent. Called by 'broadcastHarnessStatus' so the broker's
  273 -- in-memory state mirrors the events it fans out. When adding, stamps the
  274 -- session with the current time (its first "sign of life"); every later
  275 -- session-scoped broadcast refreshes it via 'touchThinking'.
  276 setThinking :: StreamBroker -> SessionId -> Bool -> IO ()
  277 setThinking broker sid thinking = do
  278   now <- getCurrentTime
  279   atomically $ modifyTVar' (sbThinking broker)
  280     (\m -> if thinking then Map.insert sid now m else Map.delete sid m)
  281 
  282 -- | Refresh a thinking session's last-activity stamp. Called by
  283 -- 'broadcast' for every session-scoped event (a recorded entry, an entry
  284 -- update, a tool-call activity, an ask), so the stale-thinking sweep
  285 -- measures /silence/ rather than turn length. A session that is not in
  286 -- the thinking set is left untouched: there is nothing to keep alive.
  287 touchThinking :: StreamBroker -> SessionId -> IO ()
  288 touchThinking broker sid = do
  289   now <- getCurrentTime
  290   atomically $ modifyTVar' (sbThinking broker) (Map.adjust (const now) sid)
  291 
  292 -- | Clear thinking sessions that have gone silent for longer than the
  293 -- given threshold (in seconds) /and/ that the caller reports as not busy,
  294 -- broadcasting an @idle@ harness-status for each. Returns the cleared
  295 -- set. This is a safety net for a turn that died without its bracket
  296 -- cleanup broadcasting @idle@ (a process killed mid-turn, an exception
  297 -- that escaped the bracket), so a page refresh does not show a dead
  298 -- session as thinking forever.
  299 --
  300 -- It must NEVER fire for a session that is still working. Every consumer
  301 -- of @harness-status idle@ treats it as "the turn is over": the web
  302 -- sidebar flips the tab to idle-unread (which sorts to the TOP of Active
  303 -- Tabs), the chat channels' @\/watch@ set drops it, and a @\/bg@
  304 -- conversation closes its connection. A mid-turn idle is therefore a
  305 -- user-visible lie, and the turn's remaining output streams into a tab
  306 -- that claims to be finished. (Observed: a 34-minute turn was cleared at
  307 -- the 30-minute mark while it was still emitting tool calls.)
  308 --
  309 -- Two independent guards, both of which must say "quiet" for a session to
  310 -- be cleared:
  311 --
  312 -- 1. /Silence./ The stamp is refreshed by 'touchThinking' on every
  313 --    session-scoped broadcast, so this measures the gap since the last
  314 --    entry / tool call / status change, not the age of the turn.
  315 -- 2. /The caller's busy predicate./ The gateway passes "a turn is in
  316 --    flight for this session, or it still has running background
  317 --    children" — a parent parked on a long, silent sub-agent run is alive,
  318 --    not stale. A busy session's stamp is restarted instead, so the next
  319 --    sweep leaves it alone.
  320 reconcileStaleThinking
  321   :: StreamBroker
  322   -> (SessionId -> IO Bool)
  323      -- ^ is this session busy? (turn in flight, or running children)
  324   -> Int
  325      -- ^ max silence (seconds) before a thinking session is cleared
  326   -> IO (Set SessionId)
  327 reconcileStaleThinking broker isBusy maxIdleSec = do
  328   now <- getCurrentTime
  329   let threshold = fromIntegral maxIdleSec :: Double
  330       silentEnough t = realToFrac (now `diffUTCTime` t) >= threshold
  331   candidates <- Map.filter silentEnough <$> readTVarIO (sbThinking broker)
  332   -- The busy check is IO (session locks + agent registry), so it is asked
  333   -- outside STM, per silent candidate.
  334   busy <- filterM (isBusy . fst) (Map.toList candidates)
  335   let busySids = Set.fromList (map fst busy)
  336       silentSids = Map.keysSet candidates `Set.difference` busySids
  337   -- A busy session is alive, just quiet: restart its clock so it stops
  338   -- being a candidate for the next sweeps.
  339   forM_ (Set.toList busySids) (touchThinking broker)
  340   -- Re-check under STM before deleting: a wake-up turn may have
  341   -- re-broadcast @thinking@ (a fresh stamp) since the snapshot above.
  342   cleared <- atomically $ do
  343     m <- readTVar (sbThinking broker)
  344     let doomed = Map.filterWithKey
  345           (\sid t -> Set.member sid silentSids && silentEnough t) m
  346     unless (Map.null doomed) $
  347       writeTVar (sbThinking broker) (m `Map.difference` doomed)
  348     pure (Map.keysSet doomed)
  349   unless (Set.null cleared) $
  350     globalLogIO WarningS (ls ("[idle-diag] reconcileStaleThinking: clearing "
  351       <> T.pack (show (Set.size cleared))
  352       <> " session(s) silent for >= " <> T.pack (show maxIdleSec)
  353       <> "s and not busy — broadcasting idle: "
  354       <> T.intercalate "," (map sessionIdText (Set.toList cleared))))
  355   -- Broadcast idle for each cleared session so the frontend updates.
  356   forM_ (Set.toList cleared) $ \sid ->
  357     broadcast broker (BeActivity sid (object
  358       [ "kind" .= ("harness-status" :: Text)
  359       , "status" .= ("idle" :: Text)
  360       ]))
  361   pure cleared

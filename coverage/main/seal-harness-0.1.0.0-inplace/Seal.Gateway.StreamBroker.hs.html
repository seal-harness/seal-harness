   44 -- | The per-subscriber state: the focused session (via an 'TVar' so the
   45 -- connection thread can update it on focus without re-subscribing), a
   46 -- send action, and an optional close action. The close action is called
   47 -- when 'subSend' throws (dead connection) so the WS socket is actively
   48 -- closed — this triggers the client's @onclose@ handler and reconnect
   49 -- rather than leaving a zombie connection that never receives another
   50 -- event but also never detects the failure.
   51 data Subscriber = Subscriber
   52   { subSessionRef :: TVar SessionId
   53   , subSend       :: BrokerEvent -> IO ()
   54   , subClose      :: IO ()
   55   }
   56 
   57 -- | The in-process broker. STM-backed: a 'TVar' of subscribers + a global
   58 -- cap. Also tracks the set of sessions currently in a @thinking@ turn so
   59 -- a freshly-connected web client can hydrate its sidebar without waiting
   60 -- for the next harness-status event (which would only arrive at the next
   61 -- turn boundary — leaving a mid-turn refresh stuck on Idle). The thinking
   62 -- map records /when/ thinking started so 'reconcileStaleThinking' can
   63 -- detect sessions stuck longer than a threshold (e.g. an unanswered
   64 -- ASK_HUMAN blocking the turn indefinitely).
   65 data StreamBroker = StreamBroker
   66   { sbSubs :: TVar [Subscriber]
   67   , sbCap :: Int
   68   , sbThinking :: TVar (Map.Map SessionId UTCTime)
   69   , sbEntryCursors :: TVar (Map.Map SessionId Int)
   70   }
   71 
   72 -- | Build a new broker with the given subscriber cap.
   73 newStreamBroker :: Int -> IO StreamBroker
   74 newStreamBroker cap =
   75   StreamBroker <$> newTVarIO [] <*> pure cap <*> newTVarIO Map.empty <*> newTVarIO Map.empty

   77 -- | Read the per-session entry broadcast cursor: the number of entries
   78 -- already fanned out for this session. Zero for a never-broadcast session.
   79 readEntryCursor :: StreamBroker -> SessionId -> IO Int
   80 readEntryCursor broker sid =
   81   fromMaybe 0 . Map.lookup sid <$> readTVarIO (sbEntryCursors broker)
   82 
   83 -- | Incremental entry broadcast (issue #198): given the session's FULL
   84 -- frontend-shaped transcript (as 'readTranscriptEntries' returns it),
   85 -- return only the entries that have not been broadcast for this session
   86 -- yet, and advance the per-session cursor. The cursor is positional
   87 -- (the number of entries already sent), so successive calls during a
   88 -- turn fan out each entry exactly once — linear broadcast volume instead
   89 -- of the historical O(N²) full-transcript re-send per recorded entry.
   90 --
   91 -- A cursor AHEAD of the transcript (the session was rebuilt / the
   92 -- transcript shrank) clamps to zero and resends everything: subscribers
   93 -- see an idempotent replay rather than silently missing entries. The web
   94 -- frontend dedupes by entry id, and idempotent replays are the failure
   95 -- mode every consumer already tolerates.
   96 --
   97 -- Concurrency: 'Seal.Core.TurnEngine.runTurnBody' runs turns under
   98 -- 'withSessionLock' so per-session calls are serialized; the atomic
   99 -- stateTVar below keeps the cursor consistent even when the
  100 -- slash-command writers ('mkModelTranscriptWriter',
  101 -- 'mkStopTranscriptWriter') race a turn.
  102 takeNewEntries
  103   :: StreamBroker -> SessionId -> [(Int, a)] -> IO [(Int, a)]
  104 takeNewEntries broker sid entries = do
  105   let total = length entries
  106   start <- atomically $ stateTVar (sbEntryCursors broker) $ \cursors ->
  107     let prior = fromMaybe 0 (Map.lookup sid cursors)
  108     in if prior > total
  109          then (0, Map.insert sid total cursors)
  110          else (prior, Map.insert sid total cursors)
  111   pure (drop start entries)

  113 -- | Subscribe a new connection. If the global cap is exceeded, the subscribe
  114 -- is a no-op (the over-cap subscriber is never added — it should close).
  115 -- Returns the session 'TVar' so the caller can update the focused session
  116 -- via 'updateSubscriberSession' when the client sends a @focus@ op.
  117 --
  118 -- The @close@ action is called when the broker detects this subscriber's
  119 -- send has thrown (dead connection), so the WS socket is actively closed
  120 -- and the client reconnects. Pass @pure ()@ when no cleanup is needed
  121 -- (tests).
  122 subscribe :: StreamBroker -> SessionId -> (BrokerEvent -> IO ()) -> IO () -> IO (TVar SessionId)
  123 subscribe broker session sendfn closefn = do
  124   ref <- newTVarIO session
  125   atomically $ do
  126     subs <- readTVar (sbSubs broker)
  127     when (length subs < sbCap broker) $
  128       writeTVar (sbSubs broker) (subs <> [Subscriber ref sendfn closefn])
  129   pure ref
  130 
  131 -- | Update a subscriber's focused session. Called when the client sends a
  132 -- @focus@ op so subsequent 'BeEntryRecorded' events for the new session are
  133 -- delivered.
  134 updateSubscriberSession :: TVar SessionId -> SessionId -> IO ()
  135 updateSubscriberSession ref sid = atomically (writeTVar ref sid)
  136 
  137 -- | Fan one event to every subscriber whose focused session matches. For
  138 -- 'BeListsSnapshot' (a broadcast to all), every subscriber receives it
  139 -- regardless of focus.
  140 --
  141 -- A subscriber whose 'subSend' throws (e.g. a closed WebSocket connection
  142 -- raising 'Network.WebSockets.ConnectionClosed') is pruned from the
  143 -- subscriber list and its 'subClose' action is invoked. Closing the
  144 -- connection actively (rather than just silently removing it from the
  145 -- list) ensures the client's @onclose@ fires and it reconnects — without
  146 -- this, a dead connection would linger as a zombie: the reader loop is
  147 -- still alive, the client thinks it's connected, but no events arrive.
  148 -- The close action is best-effort (swallowed if it throws) so a failure
  149 -- during cleanup never propagates to the caller (e.g. a @seal serve@
  150 -- request thread running 'triggerBroadcast' after a slash command).
  151 broadcast :: StreamBroker -> BrokerEvent -> IO ()
  152 broadcast broker event = do
  153   subs <- readTVarIO (sbSubs broker)
  154   live <- filterM (deliverTo event) subs
  155   -- Drop any subscribers whose send threw (dead connections). Call their
  156   -- close actions so the WS socket is actively closed and the client
  157   -- reconnects. The length check avoids a needless STM write when everyone
  158   -- survived.
  159   when (length live < length subs) $ do
  160     let dead = filter (\s -> subSessionRef s `notElem` map subSessionRef live) subs
  161     forM_ dead $ \s -> subClose s `catch` \(_ :: SomeException) -> pure ()
  162     atomically $ writeTVar (sbSubs broker) live
  163   where
  164     deliverTo ev s =
  165       (do
  166          ok <- shouldSend ev s
  167          if ok then subSend s ev >> pure True else pure True)
  168         `catch` \(_e :: SomeException) -> pure False
  169     -- All-subscriber events vs session-filtered events.
  170     -- BeActivity is ALL-subscriber: the sidebar renders tab status for
  171     -- EVERY open tab, so a turn on a channel-originated session (e.g.
  172     -- Telegram) must surface to a web client focused on a different
  173     -- session. The activity envelope carries its own sessionId, so the
  174     -- frontend's useSessionActivityStream keys it per-session without
  175     -- relying on the broker's focus filter.
  176     shouldSend ev s = case ev of
  177       BeListsSnapshot _  -> pure True
  178       BeHarnessStatus _  -> pure True
  179       BeActivity _ _      -> pure True
  180       BeAgentDefsChanged  -> pure True
  181       BeSkillsChanged     -> pure True
  182       BeReposChanged      -> pure True
  183       BeEntryRecorded sid _ -> matchSession s sid
  184       BeEntryUpdate sid _   -> matchSession s sid
  185       BeAsk sid _          -> matchSession s sid
  186       BeAskResolved sid _  -> matchSession s sid
  187     matchSession s sid = do
  188       subSid <- readTVarIO (subSessionRef s)
  189       pure (subSid == sid)

  213 -- | Read the set of sessions currently in a @thinking@ turn. Used by the
  214 -- lists-snapshot builders to hydrate a freshly-connected web client's
  215 -- sidebar (so a mid-turn refresh does not blank the thinking indicator).
  216 thinkingSessions :: StreamBroker -> IO (Set SessionId)
  217 thinkingSessions broker =
  218   Map.keysSet <$> readTVarIO (sbThinking broker)
  219 
  220 -- | Add ('True') or remove ('False') a session from the thinking set.
  221 -- Idempotent. Called by 'broadcastHarnessStatus' so the broker's
  222 -- in-memory state mirrors the events it fans out. When adding, records
  223 -- the current time so 'reconcileStaleThinking' can detect sessions that
  224 -- have been stuck thinking longer than a threshold (e.g. an unanswered
  225 -- ASK_HUMAN blocking the turn indefinitely — session
  226 -- 20260912-183908-767 issue #3).
  227 setThinking :: StreamBroker -> SessionId -> Bool -> IO ()
  228 setThinking broker sid thinking = do
  229   now <- getCurrentTime
  230   atomically $ modifyTVar' (sbThinking broker)
  231     (\m -> if thinking then Map.insert sid now m else Map.delete sid m)
  232 
  233 -- | Remove sessions that have been in the thinking set longer than the
  234 -- given threshold (in seconds). Returns the set of sessions that were
  235 -- cleared. Broadcasts an @idle@ harness-status for each cleared session
  236 -- so the web frontend updates. This is a safety net: a session blocked
  237 -- on an unanswered ASK_HUMAN (or a dead provider connection that slipped
  238 -- past the stream timeout) leaves the session permanently thinking in
  239 -- the broker's in-memory state. Without reconciliation, a page refresh
  240 -- shows the session as thinking forever.
  241 reconcileStaleThinking :: StreamBroker -> Int -> IO (Set SessionId)
  242 reconcileStaleThinking broker maxAgeSec = do
  243   now <- getCurrentTime
  244   let threshold = fromIntegral maxAgeSec :: Double
  245   stale <- atomically $ do
  246     m <- readTVar (sbThinking broker)
  247     let isStale t = realToFrac (now `diffUTCTime` t) >= threshold
  248         staleMap = Map.filter isStale m
  249         staleSids = Map.keysSet staleMap
  250     unless (Set.null staleSids) $
  251       writeTVar (sbThinking broker) (Map.difference m staleMap)
  252     pure staleSids
  253   -- Broadcast idle for each cleared session so the frontend updates.
  254   forM_ (Set.toList stale) $ \sid ->
  255     broadcast broker (BeActivity sid (object
  256       [ "kind" .= ("harness-status" :: Text)
  257       , "status" .= ("idle" :: Text)
  258       ]))
  259   pure stale

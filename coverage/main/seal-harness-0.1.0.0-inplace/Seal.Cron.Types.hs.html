   46 newtype CronJobId = CronJobId { unCronJobId :: Text }
   47   deriving stock (Eq, Ord, Show)
   48   deriving newtype (ToJSON)
   49 
   50 mkCronJobId :: Text -> Either Text CronJobId
   51 mkCronJobId t
   52   | T.null t = Left "cron job id must not be empty"
   53   | T.length t > 64 = Left "cron job id must be at most 64 characters"
   54   | T.head t == '-' = Left "cron job id must not start with a dash"
   55   | not (T.all isValidIdChar t) = Left "cron job id must contain only [a-z0-9-]"
   56   | otherwise = Right (CronJobId t)
   57 
   58 isValidIdChar :: Char -> Bool
   59 isValidIdChar c = isAsciiLower c || isDigit c || c == '-'
   60 
   61 cronJobIdText :: CronJobId -> Text
   62 cronJobIdText = unCronJobId
   63 
   64 instance FromJSON CronJobId where
   65   parseJSON = withObject "CronJobId" $ \o -> do
   66     t <- o .: "id"
   67     case mkCronJobId t of
   68       Left e -> fail (T.unpack e)
   69       Right cid -> pure cid

   75 -- | A validated 5-field cron expression (e.g. @"0 9 * * 1-5"@). The
   76 -- underlying 'CronSchedule' is parsed at construction time.
   77 data CronExpr = CronExpr
   78   { ceSchedule :: !CronSchedule
   79   , ceText     :: !Text
   80   }
   81 
   82 mkCronExpr :: Text -> Either Text CronExpr
   83 mkCronExpr t
   84   | T.null t = Left "cron expression must not be empty"
   85   | otherwise = case parseCronSchedule t of
   86       Left err -> Left ("invalid cron expression: " <> t <> " (" <> T.pack err <> ")")
   87       Right sched -> Right (CronExpr { ceSchedule = sched, ceText = t })
   88 
   89 cronExprText :: CronExpr -> Text
   90 cronExprText = ceText
   91 
   92 instance ToJSON CronExpr where
   93   toJSON = toJSON . ceText
   94 
   95 instance FromJSON CronExpr where
   96   parseJSON v = do
   97     t <- parseJSON v
   98     case mkCronExpr t of
   99       Left e -> fail (T.unpack e)
  100       Right e -> pure e
  101 
  102 instance Show CronExpr where
  103   show = T.unpack . ceText
  104 
  105 instance Eq CronExpr where
  106   a == b = ceText a == ceText b
  107 
  108 instance Ord CronExpr where
  109   compare a b = compare (ceText a) (ceText b)

  115 newtype CronTimeZone = CronTimeZone { unCronTimeZone :: Text }
  116   deriving stock (Eq, Ord, Show)
  117   deriving newtype (ToJSON)
  118 
  119 mkCronTimeZone :: Text -> Either Text CronTimeZone
  120 mkCronTimeZone t
  121   | T.null t = Left "timezone must not be empty"
  122   | not (isValidIanaTz t) = Left "timezone must be a valid IANA identifier (e.g. America/New_York)"
  123   | otherwise = Right (CronTimeZone t)
  124 
  125 -- | Validate a 'CronTimeZone' against the tz database. The pure
  126 -- 'mkCronTimeZone' constructor checks format only; this IO function
  127 -- additionally verifies the zone exists in the database (e.g.
  128 -- @"Not/AZone"@ passes the format check but is not a real zone). Used
  129 -- at the write boundary on both the opcode and web paths so a bad zone
  130 -- can never enter the store and kill the scheduler thread.
  131 validateCronTimeZone :: CronTimeZone -> IO (Either Text ())
  132 validateCronTimeZone tz = do
  133   eEx <- loadCronTZ tz
  134   pure $ case eEx of
  135     Left (_ :: SomeException) -> Left ("timezone not found in tz database: " <> cronTimeZoneText tz)
  136     Right _ -> Right ()
  137 
  138 isValidIanaTz :: Text -> Bool
  139 isValidIanaTz t =
  140   let parts = T.splitOn "/" t
  141   in length parts >= 2 && not (any T.null parts) && all isValidTzComponent parts
  142 
  143 isValidTzComponent :: Text -> Bool
  144 isValidTzComponent = T.all (\c -> isAscii c && (isAlphaNum c || c == '_' || c == '-'))
  145 
  146 cronTimeZoneText :: CronTimeZone -> Text
  147 cronTimeZoneText = unCronTimeZone
  148 
  149 -- | Load a zone from the host tz database, falling back to the database
  150 -- embedded in @tzdata@. The fallback keeps static release binaries portable:
  151 -- 'loadTZFromDB' resolves to a build-time Nix store data path that does not
  152 -- exist on a machine without Nix.
  153 loadCronTZ :: CronTimeZone -> IO (Either SomeException TZ)
  154 loadCronTZ zone = do
  155   let name = T.unpack (cronTimeZoneText zone)
  156   eSystem <- try (loadSystemTZ name)
  157   pure $ case eSystem of
  158     Right loaded -> Right loaded
  159     Left err -> case ZonesAll.tzByName (BS.pack name) of
  160       Just loaded -> Right loaded
  161       Nothing -> Left err
  162 
  163 instance FromJSON CronTimeZone where
  164   parseJSON v = do
  165     t <- parseJSON v
  166     case mkCronTimeZone t of
  167       Left e -> fail (T.unpack e)
  168       Right tz -> pure tz

  318 slugify :: Text -> Text
  319 slugify name =
  320   let lowered = T.map toLower name
  321       slugged = T.map (\c -> if isSlugChar c then c else '-') lowered
  322       collapsed = collapseDashes slugged
  323       trimmed = T.dropWhile (== '-') (T.dropWhileEnd (== '-') collapsed)
  324       truncated = T.take 32 trimmed
  325   in if T.null truncated then "job" else truncated
  326   where
  327     isSlugChar :: Char -> Bool
  328     isSlugChar c = isAsciiLower c || isDigit c || c == '-'
  329 
  330 collapseDashes :: Text -> Text
  331 collapseDashes = T.intercalate "-" . filter (not . T.null) . T.splitOn "-"

  333 -- ---------------------------------------------------------------------------
  334 -- Next fire computation
  335 -- ---------------------------------------------------------------------------
  336 
  337 -- | Compute the next fire time for a cron job, given the reference time
  338 -- (usually @now@). Uses the @cron@ package's 'expand' for schedule matching
  339 -- and the @tz@ package for timezone-aware computation.
  340 --
  341 -- When the job has a timezone, the reference time is converted to local
  342 -- time, the next fire is computed in local time, and the result is
  343 -- converted back to UTC.
  344 computeNextFire :: CronExpr -> Maybe CronTimeZone -> UTCTime -> IO (Maybe UTCTime)
  345 computeNextFire expr mTz refTime =
  346   case mTz of
  347     Nothing -> pure (nextFireUTC (ceSchedule expr) refTime)
  348     Just tz -> nextFireTZ (ceSchedule expr) tz refTime
  349 
  350 -- | Compute next fire in UTC (no timezone). Searches minute by minute
  351 -- from the reference time forward, up to a reasonable limit (366 days).
  352 nextFireUTC :: CronSchedule -> UTCTime -> Maybe UTCTime
  353 nextFireUTC sched refTime =
  354   case expand sched of
  355     Nothing -> Nothing
  356     Just ex -> findNextMatch ex refTime
  357 
  358 -- | Compute next fire with timezone awareness.
  359 nextFireTZ :: CronSchedule -> CronTimeZone -> UTCTime -> IO (Maybe UTCTime)
  360 nextFireTZ sched tz refTime = do
  361   eLoaded <- loadCronTZ tz
  362   case eLoaded of
  363     Left (_ :: SomeException) -> pure Nothing
  364     Right loadedTz -> do
  365       let localRef = utcToLocalTimeTZ loadedTz refTime
  366           pseudoUtc = localTimeToUTC utc localRef
  367       case nextFireUTC sched pseudoUtc of
  368         Nothing -> pure Nothing
  369         Just nextPseudoUtc -> do
  370           let nextLocal = utcToLocalTime utc nextPseudoUtc
  371           pure (Just (localTimeToUTCTZ loadedTz nextLocal))
  372 
  373 -- | Find the next time after @refTime@ that matches all cron fields.
  374 -- Searches up to 366 days forward (527040 minutes).
  375 findNextMatch :: Expanded -> UTCTime -> Maybe UTCTime
  376 findNextMatch ex refTime =
  377   let truncated = truncateToMinute refTime
  378       startMinute = addOneMinute truncated
  379   in go 0 startMinute
  380   where
  381     go :: Int -> UTCTime -> Maybe UTCTime
  382     go n t
  383       | n >= 366 * 24 * 60 = Nothing
  384       | matchesExpanded ex t = Just t
  385       | otherwise = go (n + 1) (addOneMinute t)
  386 
  387 -- | Check if a 'UTCTime' matches all fields of an 'Expanded' cron schedule.
  388 matchesExpanded :: Expanded -> UTCTime -> Bool
  389 matchesExpanded ex t =
  390   let (_year, month, day) = toGregorian (utctDay t)
  391       secs = utctDayTime t
  392       totalSecs = floor secs :: Int
  393       hour = totalSecs `div` 3600
  394       minute = (totalSecs `div` 60) `mod` 60
  395       dowNum = dowToInt (dayOfWeek (utctDay t))
  396   in  inSet minute (minF ex)
  397       && inSet hour (hourF ex)
  398       && inSet day (domF ex)
  399       && inSet month (monthF ex)
  400       && inSet dowNum (dowF ex)
  401 
  402 -- | Check if an 'Int' is in the expanded field's 'NonEmpty Int' list.
  403 inSet :: Int -> NonEmpty Int -> Bool
  404 inSet x = elem x . toList
  405 
  406 -- | Convert 'DayOfWeek' to cron integer (0=Sunday, 7=Sunday, 1=Monday..6=Saturday).
  407 -- The cron package uses 0-7 where both 0 and 7 mean Sunday.
  408 dowToInt :: DayOfWeek -> Int
  409 dowToInt Sunday = 0
  410 dowToInt Monday = 1
  411 dowToInt Tuesday = 2
  412 dowToInt Wednesday = 3
  413 dowToInt Thursday = 4
  414 dowToInt Friday = 5
  415 dowToInt Saturday = 6
  416 
  417 -- | Add one minute to a 'UTCTime'.
  418 addOneMinute :: UTCTime -> UTCTime
  419 addOneMinute = addUTCTime 60
  420 
  421 -- | Truncate a 'UTCTime' to the minute boundary (zero out seconds).
  422 -- Ensures cron fire times are always at second 0.
  423 truncateToMinute :: UTCTime -> UTCTime
  424 truncateToMinute t =
  425   let secs = floor (utctDayTime t) :: Int
  426   in t { utctDayTime = fromIntegral ((secs `div` 60) * 60) }

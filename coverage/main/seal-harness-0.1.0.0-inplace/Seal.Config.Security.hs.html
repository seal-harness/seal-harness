    3 -- | The security-critical, boot-only configuration that lives in
    4 -- @~\/.seal\/security.toml@ — a separate file from @config.toml@ (which holds
    5 -- the agent/operator-tunable 'RuntimeConfig'). No opcode can reach this module.
    6 --
    7 -- It is re-read through exactly two narrow writers, both reachable from the
    8 -- gateway and from the CLI:
    9 --
   10 -- * 'updateAccessControlSection' rewrites the @[access_control]@ table when the
   11 --   operator enrolls a passkey, waives the requirement, or manages credentials
   12 --   from the authenticated web UI. Its callback sees only
   13 --   'AccessControlFileConfig'.
   14 -- * 'updateVaultKeySection' records the three vault-key fields
   15 --   (@vault_recipient@, @vault_identity@, @vault_key_type@) when a vault is
   16 --   created from the web setup wizard. Its callback sees the current
   17 --   'SecurityConfig' but may only return 'VaultKeyFields'.
   18 --
   19 -- Neither callback can express a change to the other's section, nor to
   20 -- @untrusted_execution@: the type split (design §4 Approach E) keeps its
   21 -- guarantee, and it is a compile error, not a convention. Both run under
   22 -- 'securityWriteLock' and verify the re-encoded document before committing.
   23 --
   24 -- The file is therefore reachable from the HTTP gateway — but only through
   25 -- those two writers, only on a loopback-origin-guarded route, and only behind
   26 -- the passkey middleware (default-deny). Nothing here is an opcode surface.
   27 --
   28 -- Moving @untrusted_execution@ and the vault settings here means a future
   29 -- @CONFIG_UPDATE@ opcode (which operates on 'RuntimeConfig') and the Gateway's
   30 -- @updateRuntimeConfig@ caller physically cannot express a change to these
   31 -- fields — the type split makes it a compile error (design §4 Approach E).
   32 --
   33 -- @[gateway]@ joined them for the same reason: its bind host and port decide
   34 -- who can reach the agent, and its Origin allowlist and TLS paths decide how.
   35 -- The two narrow writers below assert it unchanged across their round-trip, so
   36 -- an enrollment or a vault-key write cannot move the listener either.
   37 --
   38 -- Every field is optional; a missing key decodes as 'Nothing' and a
   39 -- 'Nothing' value is omitted from the encoded output.

  109 -- | The boot-only, agent-immutable security configuration. Carries the
  110 -- @untrusted_execution@ section (remote-only enforcement) and the vault
  111 -- settings. Loaded once at boot from @security.toml@; never passed to any
  112 -- opcode or the Gateway.
  113 data SecurityConfig = SecurityConfig
  114   { scVaultPath :: Maybe Text
  115     -- ^ Absolute path to the vault file (default: @~\/.seal\/config\/vault\/vault.age@).
  116   , scVaultRecipient :: Maybe Text
  117     -- ^ age public key: @age1…@ or @age1yubikey1…@.
  118   , scVaultIdentity :: Maybe Text
  119     -- ^ Path to the identity file under @keys\/@, or a user-supplied path.
  120   , scVaultUnlock :: Maybe Text
  121     -- ^ @\"startup\"@ | @\"on_demand\"@ | @\"per_access\"@.
  122   , scVaultKeyType :: Maybe Text
  123     -- ^ Display label: @\"x25519\"@ | @\"yubikey\"@ | @\"user\"@.
  124   , scUntrustedExec :: Maybe UntrustedExecFileConfig
  125     -- ^ Optional @[untrusted_execution]@ section (remote-only untrusted
  126     -- execution). Absent means @mode=local@ (default). @mode=remote@
  127     -- fail-closes at call time when the remote block is absent or
  128     -- unreachable (boot still succeeds).
  129   , scAccessControl :: Maybe AccessControlFileConfig
  130     -- ^ Optional @[access_control]@ section. Absent means *unconfigured*: the
  131     -- gateway serves only the setup screen until the operator enrolls a
  132     -- passkey or explicitly waives (design §4). This is the ONE section the
  133     -- gateway may write, via 'updateAccessControlSection'.
  134   , scGateway :: Maybe PartialGatewayConfig
  135     -- ^ Optional @[gateway]@ section: bind host/port, the WS Origin allowlist,
  136     -- the static-dir override and the TLS settings. Boot-only and
  137     -- agent-immutable like the rest of this file — a bind address is not a
  138     -- setting to guess from, and no opcode or gateway config writer can
  139     -- express a change to it (the type split makes that a compile error).
  140     -- 'Seal.Gateway.Config.gatewayConfigFromSecurity' resolves it.
  141   } deriving stock (Eq, Show)

  220 -- | Every /value-bearing/ key path in a parsed document, as a list of pieces.
  221 --
  222 -- Table nodes are deliberately not emitted on their own. A table is a prefix
  223 -- of the paths underneath it, and whether it appears as a node or as a prefix
  224 -- depends only on spelling — a header or a dotted key. Emitting both would
  225 -- make "the document's data" mean two different things, and a rewrite that
  226 -- normalises a dotted key into a header would look like damage.
  227 --
  228 -- The three shapes tomland produces all have to be walked, or a path
  229 -- comparison silently misses one of them:
  230 --
  231 -- * an explicit table header — a prefix-tree 'Toml.Branch' whose prefix /is/
  232 --   the path;
  233 -- * a table's own keys — a 'Toml.TOML' payload reached through a
  234 --   'Toml.Branch' or 'Toml.Leaf';
  235 -- * a /dotted key/ — which tomland does NOT expand into nested tables. It
  236 --   keeps the whole path in one 'Toml.tomlPairs' key, so
  237 --   @credentials.\<id\>.public_key@ under @[access_control]@ arrives as the
  238 --   single pair key @(\"credentials\" :| [\"\<id\>\",\"public_key\"])@ and no
  239 --   prefix tree ever holds a @credentials@ node. This is why the check below
  240 --   cannot be a codec.
  241 tomlKeyPaths :: Toml.TOML -> [[Toml.Piece]]
  242 tomlKeyPaths = go []
  243   where
  244     go prefix toml =
  245          [ prefix <> keyPieces key
  246          | key <- HashMap.keys (Toml.tomlPairs toml) ]
  247       <> concatMap nodePaths (HashMap.elems (Toml.tomlTables toml))
  248 
  249     -- A node's own prefix is the authoritative absolute path: an implicit
  250     -- table header arrives as one multi-piece prefix under a single-piece
  251     -- map key, so the map key alone loses path segments.
  252     nodePaths tree = case tree of
  253       Toml.Branch pref mValue children ->
  254            maybe [] (go (keyPieces pref)) mValue
  255         <> concatMap nodePaths (HashMap.elems children)
  256       Toml.Leaf pref value -> go (keyPieces pref) value
  257 
  258 keyPieces :: Toml.Key -> [Toml.Piece]
  259 keyPieces = NonEmpty.toList . Toml.unKey
  260 
  261 -- | Expand dotted keys into nested tables.
  262 --
  263 -- tomland keeps a dotted key as one flat pair whose key carries the whole
  264 -- path: @untrusted_execution.mode = \"local\"@ becomes
  265 -- @(\"untrusted_execution\" :| [\"mode\"], …)@ in 'Toml.tomlPairs', and no
  266 -- @untrusted_execution@ node exists anywhere in the prefix tree. A codec looks
  267 -- up tables, so it finds nothing, decodes the section as absent — and the next
  268 -- write drops it. Silently: the document still parses, so nothing reports it.
  269 --
  270 -- Expanding first makes the document look the way the codec expects, so both
  271 -- spellings decode and re-encode identically. (tomland's own
  272 -- 'Toml.Type.TOML.insertKeyAnyVal' does not help: it stores the value at the
  273 -- full multi-piece key, which is exactly where it already was. The nesting has
  274 -- to be built.)
  275 expandDottedKeys :: Toml.TOML -> Toml.TOML
  276 expandDottedKeys toml = foldl' insert base dottedPaths
  277   where
  278     pairs       = HashMap.toList (Toml.tomlPairs toml)
  279     dottedPaths = [(keyPieces key, value) | (key, value) <- pairs, isDotted key]
  280     base = toml
  281       { Toml.tomlPairs  = HashMap.fromList
  282           [(key, value) | (key, value) <- pairs, not (isDotted key)]
  283       , Toml.tomlTables = fmap expandNode (Toml.tomlTables toml)
  284       }
  285     insert acc (pieces, value) = insertDotted pieces value acc
  286     isDotted key = case keyPieces key of
  287       _ : _ : _ -> True
  288       _         -> False
  289 
  290     expandNode node = case node of
  291       Toml.Branch pref mValue children ->
  292         Toml.Branch pref (fmap expandDottedKeys mValue) (fmap expandNode children)
  293       Toml.Leaf pref value -> Toml.Leaf pref (expandDottedKeys value)
  294 
  295     -- Insert the value at the dotted path, creating the intermediate tables as
  296     -- explicit 'Toml.Leaf' nodes — the form 'Toml.table' and 'Toml.tableMap'
  297     -- look up. The payload type is tomland's GADT-indexed 'Toml.Value', so it
  298     -- is inferred rather than named.
  299     insertDotted [] _ target = target
  300     insertDotted (piece : rest) value target = case rest of
  301       [] -> target { Toml.tomlPairs = HashMap.insert (piece Toml.:|| []) value
  302                                             (Toml.tomlPairs target) }
  303       _  -> target
  304         { Toml.tomlTables = HashMap.insert piece
  305             (Toml.Leaf (piece Toml.:|| []) (insertDotted rest value (nodeValue existing)))
  306             (Toml.tomlTables target) }
  307       where
  308         existing = HashMap.lookupDefault (Toml.Leaf (piece Toml.:|| []) mempty) piece
  309                      (Toml.tomlTables target)
  310 
  311     -- A node's own payload with any child nodes folded into it: the canonical
  312     -- explicit form, and the one 'fixNode' produces.
  313     nodeValue node = case node of
  314       Toml.Leaf _ value -> value
  315       Toml.Branch _ mValue children ->
  316         maybe mempty (\value -> value { Toml.tomlTables = children }) mValue
  317

  338 -- | Read the credential tables that still live in a pre-split @security.toml@.
  339 --
  340 -- An absent section is an empty map, not an error: that is the idempotent case.
  341 -- A document that does not parse is an error, because the caller is about to
  342 -- rewrite it and must not do so on a guess.
  343 readLegacyCredentials :: Text -> Either Text (Map Text CredentialFileConfig)
  344 readLegacyCredentials raw = case Toml.parse raw of
  345   Left err -> Left (Toml.unTomlParseError err)
  346   Right toml -> case Toml.runTomlCodec legacySectionCodec
  347                      (normalizeSecurityToml toml) of
  348     Success Nothing          -> Right Map.empty
  349     Success (Just section)   -> Right (lacfcCredentials section)
  350     Failure errs             -> Left (Toml.prettyTomlDecodeErrors errs)

  355 -- | Does this raw @security.toml@ still carry the pre-split credential tables?
  356 --
  357 -- Once the store moved to @credentials.toml@, a write that rewrites
  358 -- @security.toml@ without noticing the old tables would drop every credential —
  359 -- silently, on a document whose next boot then reads as unconfigured.
  360 --
  361 -- Detection walks the parsed key paths, it does not grep the text. A substring
  362 -- scan for @[access_control.credentials.@ misses the dotted-key spelling of the
  363 -- same document (@[access_control]@ plus @credentials.\<id\>.public_key = …@),
  364 -- and — as 'tomlKeyPaths' explains — so does anything that goes through the
  365 -- prefix tree, which is why this is not a codec either.
  366 --
  367 -- An unparseable document answers 'False': it is refused earlier, as
  368 -- @BrSecurityUnparseable@, so it can never reach the writer this guard protects.
  369 --
  370 -- Detection is not reading. Decoding the credentials back out for
  371 -- @migrate-credentials@ is a separate concern, and it must handle the
  372 -- dotted-key spelling too: 'normalizeAccessControlTable' only makes
  373 -- /prefix-tree/ nodes explicit, so a dotted-key file currently decodes to an
  374 -- empty credential map. A migration that reads through the codec alone would
  375 -- therefore move nothing and then report success.
  376 hasLegacyLayout :: Text -> Bool
  377 hasLegacyLayout raw = case Toml.parse raw of
  378   Left _     -> False
  379   Right toml -> any isCredentialPath (tomlKeyPaths toml)
  380   where
  381     isCredentialPath (p1 : p2 : _) = p1 == "access_control" && p2 == "credentials"
  382     isCredentialPath _             = False

  384 -- | Make the implicit @access_control@ node explicit before decoding.
  385 --
  386 -- tomland's 'Toml.tableMap' looks the section up under its @[access_control]@
  387 -- header, so an idiomatic hand-written file containing only
  388 -- @[access_control.credentials.\<id\>]@ sub-tables decodes as an EMPTY map
  389 -- (learnings 2026-08-02 §4). Mirrors
  390 -- 'Seal.SourceControl.Repo.normalizeReposTable'.
  391 normalizeAccessControlTable :: Toml.TOML -> Toml.TOML
  392 normalizeAccessControlTable t =
  393   t { Toml.tomlTables = HashMap.adjust fixAccessControl "access_control" (Toml.tomlTables t) }
  394   where
  395     -- Two levels are implicit in an idiomatic file: `access_control` and the
  396     -- `credentials` table beneath it. Both must be made explicit, or the
  397     -- nested `Toml.tableMap` lookup silently yields an empty map and the
  398     -- operator's passkeys disappear.
  399     fixAccessControl tree = fixChildIn "credentials" (fixNode "access_control" tree)
  400 
  401     fixChildIn child tree = case tree of
  402       Toml.Branch pref mv children ->
  403         Toml.Branch pref mv (HashMap.adjust (fixNode child) child children)
  404       Toml.Leaf pref v ->
  405         Toml.Leaf pref v { Toml.tomlTables =
  406                              HashMap.adjust (fixNode child) child (Toml.tomlTables v) }
  407 
  408 -- | Collapse whatever implicit path the parser produced under @root@ into a
  409 -- single explicit table node named @root@.
  410 fixNode :: Toml.Piece -> Toml.PrefixTree Toml.TOML -> Toml.PrefixTree Toml.TOML
  411 fixNode root tree = case tree of
  412   Toml.Branch pref Nothing children ->
  413     Toml.Leaf pref (mempty { Toml.tomlTables = children })
  414   Toml.Branch _ (Just _) _ -> tree
  415   Toml.Leaf (_ Toml.:|| []) _ -> tree
  416   Toml.Leaf (_ Toml.:|| (p : ps)) v ->
  417     Toml.Leaf (root Toml.:|| [])
  418       (mempty { Toml.tomlTables = Toml.single (p Toml.:|| ps) v })

  482 -- | Decode @security.toml@ from its text.
  483 --
  484 -- The I/O-free half of 'loadSecurityConfig', exported because the boot
  485 -- preflight needs the document as text (for the legacy-layout check) /and/ the
  486 -- decoded section at the same time, and reading the file twice could see two
  487 -- different documents.
  488 decodeSecurityConfig :: Text -> Either Text SecurityConfig
  489 decodeSecurityConfig contents = case Toml.parse contents of
  490   Left err   -> Left (Toml.unTomlParseError err)
  491   Right toml -> case Toml.runTomlCodec securityConfigCodec
  492                      (normalizeSecurityToml toml) of
  493     Success cfg  -> Right cfg
  494     Failure errs -> Left (Toml.prettyTomlDecodeErrors errs)
  495 
  496 -- | Whether a write is allowed to discard the pre-split credential tables.
  497 --
  498 -- A credential table in @security.toml@ is only ever a leftover from the
  499 -- layout before credentials moved to @credentials.toml@, and rewriting the
  500 -- document without migrating it first destroys it: the next boot then reads as
  501 -- unconfigured and every enrolled passkey is gone. So the exemption is a
  502 -- value, not a convention — the two callers that may pass 'DropCredentials'
  503 -- are @migrate-credentials@ and @reset --yes@, both CLI-only and both
  504 -- @--yes@-gated.
  505 data WriteMode = PreserveCredentials | DropCredentials
  506   deriving stock (Eq, Show)
  507 
  508 -- | Save @cfg@ to @path@ atomically.
  509 --
  510 -- The temp file is created with mode @0600@ at creation time (not chmod'ed
  511 -- afterwards), so there is never a window in which the vault recipient or a
  512 -- passkey public key is world-readable. It is then renamed over @path@, which
  513 -- is atomic on every filesystem this project supports.
  514 --
  515 -- Every writer funnels through here, so this is where the credential-preserving
  516 -- refusal lives. It reads the document /as it is on disk/, not the value being
  517 -- written: the caller has already decoded it, and decoding is exactly what
  518 -- loses a legacy table it does not understand.
  519 saveSecurityConfig :: FilePath -> SecurityConfig -> WriteMode -> IO (Either Text ())
  520 saveSecurityConfig path cfg mode = do
  521   refusal <- legacyCredentialRefusal path mode
  522   case refusal of
  523     Just err -> pure (Left err)
  524     Nothing  -> do
  525       let encoded = Toml.encode securityConfigCodec cfg
  526       outcome <- try (writeAtomicText path encoded) :: IO (Either IOException ())
  527       pure $ case outcome of
  528         Left err -> Left ("could not write " <> T.pack path <> ": "
  529                             <> T.pack (show err))
  530         Right () -> Right ()
  531 
  532 -- | Open a unique temp file in @path@'s directory, at mode 0600.
  533 --
  534 -- 'openBinaryTempFile' is @mkstemp@-style: it creates with @O_CREAT|O_EXCL@ and
  535 -- mode 0600, so a pre-created symlink or a colliding name makes the open fail
  536 -- rather than being followed. That is why no separate @O_NOFOLLOW@ is needed.
  537 -- Two writers targeting the same destination therefore never share a temp path.
  538 openUniqueTemp :: FilePath -> IO (FilePath, Handle)
  539 openUniqueTemp path =
  540   openBinaryTempFile (takeDirectory path) (takeFileName path <> ".tmp")
  541 
  542 -- | Write UTF-8 bytes through a unique temp file and rename into place.
  543 --
  544 -- Binary mode is deliberate: 'openBinaryTempFile' returns a binary handle, and
  545 -- encoding explicitly keeps non-ASCII paths/hosts from being written as latin1.
  546 writeAtomicText :: FilePath -> Text -> IO ()
  547 writeAtomicText path contents =
  548   bracketOnError (openUniqueTemp path) cleanup $ \(tmp, h) -> do
  549     BS.hPutStr h (TE.encodeUtf8 contents)
  550     hClose h
  551     setFileMode tmp 0o600
  552     renameFile tmp path
  553   where
  554     cleanup (tmp, h) = do
  555       ignoreIO (hClose h)
  556       ignoreIO (removeFile tmp)
  557     ignoreIO action = void (try action :: IO (Either IOException ()))
  558 
  559 legacyCredentialRefusal :: FilePath -> WriteMode -> IO (Maybe Text)
  560 legacyCredentialRefusal _    DropCredentials   = pure Nothing
  561 legacyCredentialRefusal path PreserveCredentials = do
  562   exists <- doesFileExist path
  563   if not exists
  564     then pure Nothing
  565     else do
  566       raw <- TIO.readFile path
  567       pure $ if hasLegacyLayout raw
  568         then Just ("refusing to rewrite " <> T.pack path <> ": it still holds the "
  569                     <> "pre-split [access_control.credentials] tables. Run "
  570                     <> "`seal access-control migrate-credentials` first.")
  571         else Nothing
  572 
  573 -- | Write a text file whose permissions are restricted from the moment it
  574 -- exists: @0600@ is applied at creation (via @openFd@'s @creat@ mode) and again
  575 -- afterwards to defeat a permissive umask, so there is never a window in which
  576 -- the contents are world-readable. Used for @security.toml@ and for the
  577 -- per-boot setup code.
  578 writeSecretFile :: FilePath -> Text -> IO ()
  579 writeSecretFile path contents =
  580   bracketOnError
  581     (openFd path WriteOnly defaultFileFlags
  582        { trunc = True, creat = Just 0o600, nofollow = True })
  583     closeFd
  584                  (\fd -> do
  585                      h <- fdToHandle fd
  586                      TIO.hPutStr h contents
  587                      hClose h
  588                      setFileMode path 0o600)
  589 
  590 -- | Process-wide lock serializing security config writes (design V7).
  591 --
  592 -- Exported because the credential store shares it: a credential write and a
  593 -- @security.toml@ write must not interleave (an enrollment raises the mode and
  594 -- writes a credential as one operation — design §6.1). It is
  595 -- /non-reentrant/: a caller that already holds it must use the unlocked
  596 -- primitives, never a writer that takes it. 'Seal.Config.Credentials'\\'s
  597 -- writers take it; nothing else may call them from inside it.
  598 {-# NOINLINE securityWriteLock #-}
  599 securityWriteLock :: MVar ()
  600 securityWriteLock = unsafePerformIO (newMVar ())
  601 
  602 -- | Load the security config at @path@, apply @f@, save. Propagates any load
  603 -- error as @Left Text@ without writing. This is the admin/boot path (e.g.
  604 -- @\/vault setup@); it is NEVER called by an opcode or the HTTP Gateway
  605 -- (those operate on 'RuntimeConfig' and physically cannot reach
  606 -- 'SecurityConfig' — design §4 E). Serialized behind 'securityWriteLock'
  607 -- to prevent lost-update races (design V7).
  608 updateSecurityConfig :: FilePath -> (SecurityConfig -> SecurityConfig) -> IO (Either Text ())
  609 updateSecurityConfig path f = withMVar securityWriteLock $ \_ -> do
  610   result <- loadSecurityConfig path
  611   case result of
  612     Left err  -> pure (Left err)
  613     Right cfg -> saveSecurityConfig path (f cfg) PreserveCredentials
  614 
  615 -- | One of the TWO narrow writers of @security.toml@ reachable from the
  616 -- gateway (the other is 'updateVaultKeySection'; they deliberately cannot touch
  617 -- each other's fields).
  618 --
  619 -- It loads the file, applies @f@ to the @[access_control]@ section alone, and
  620 -- re-encodes. The callback's type is @AccessControlFileConfig ->
  621 -- AccessControlFileConfig@, so no value the gateway can build will ever name a
  622 -- vault or @untrusted_execution@ field: rewriting those is a compile error,
  623 -- exactly as the 'RuntimeConfig' split intends.
  624 --
  625 -- The write is verified before it is committed: the freshly encoded document
  626 -- is decoded again and compared with the intended section, so a codec change
  627 -- that silently drops a field surfaces as @Left@ instead of as data loss.
  628 -- Serialized behind 'securityWriteLock' (the same lock the vault commands use
  629 -- for lost-update protection).
  630 updateAccessControlSection
  631   :: FilePath
  632   -> (AccessControlFileConfig -> AccessControlFileConfig)
  633   -> IO (Either Text AccessControlFileConfig)
  634 updateAccessControlSection path f = withMVar securityWriteLock $ \_ -> do
  635   loaded <- loadSecurityConfig path
  636   case loaded of
  637     Left err -> pure (Left err)
  638     Right cfg -> do
  639       let updated = f (fromMaybe emptyAccessControl (scAccessControl cfg))
  640           newCfg = cfg { scAccessControl = Just updated }
  641           encoded = Toml.encode securityConfigCodec newCfg
  642       case reDecode encoded of
  643         Just roundTripped
  644           | scAccessControl roundTripped == Just updated
  645           , scGateway       roundTripped == scGateway cfg ->
  646               -- The section writer never discards credentials: enrollment
  647               -- moves them out, it does not drop them.
  648               fmap (>> Right updated)
  649                 (saveSecurityConfig path newCfg PreserveCredentials)
  650         _ -> pure (Left "refusing to write security.toml: the encoded document does not round-trip the new [access_control] section")
  651   where
  652     reDecode encoded = case Toml.parse encoded of
  653       Left _     -> Nothing
  654       Right toml -> case Toml.runTomlCodec securityConfigCodec toml of
  655         Success c -> Just c
  656         Failure _ -> Nothing
  657 
  658 -- | The vault-key triple, and nothing else. The gateway's setup wizard writes
  659 -- exactly these three fields when it creates a vault.
  660 data VaultKeyFields = VaultKeyFields
  661   { vkfRecipient :: !Text
  662     -- ^ age public key (@age1…@ \/ @age1yubikey1…@).
  663   , vkfIdentity  :: !FilePath
  664     -- ^ Absolute path to the identity file.
  665   , vkfKeyType   :: !Text
  666     -- ^ Display label: @"x25519"@ \/ @"yubikey"@ \/ @"user"@.
  667   } deriving stock (Eq, Show)
  668 
  669 -- | The other narrow writer: the vault-key triple, and nothing else.
  670 --
  671 -- The callback sees the *current* config and returns either a rejection (which
  672 -- is how the setup wizard refuses to overwrite a key that appeared between its
  673 -- pre-check and this write — a compare-and-swap across processes, since the
  674 -- read and the write happen inside 'securityWriteLock') or the triple to
  675 -- record.
  676 --
  677 -- Like 'updateAccessControlSection' the write is verified before it is
  678 -- committed: the encoded document is decoded again and compared, so a codec
  679 -- change that silently drops a field surfaces as @Left@ instead of data loss.
  680 -- The @[access_control]@ and @[untrusted_execution]@ sections are asserted
  681 -- unchanged, so this writer cannot become a back door for them.
  682 updateVaultKeySection
  683   :: FilePath
  684   -> (SecurityConfig -> Either e VaultKeyFields)
  685   -> IO (Either Text (Either e VaultKeyFields))
  686 updateVaultKeySection path f = withMVar securityWriteLock $ \_ -> do
  687   loaded <- loadSecurityConfig path
  688   case loaded of
  689     Left err -> pure (Left err)
  690     Right cfg -> case f cfg of
  691       Left rejection -> pure (Right (Left rejection))
  692       Right fields -> do
  693         let newCfg = cfg
  694               { scVaultRecipient = Just (vkfRecipient fields)
  695               , scVaultIdentity  = Just (T.pack (vkfIdentity fields))
  696               , scVaultKeyType   = Just (vkfKeyType fields)
  697               }
  698             encoded = Toml.encode securityConfigCodec newCfg
  699         case reDecode encoded of
  700           Just roundTripped
  701             | scVaultRecipient roundTripped == scVaultRecipient newCfg
  702             , scVaultIdentity  roundTripped == scVaultIdentity newCfg
  703             , scVaultKeyType   roundTripped == scVaultKeyType newCfg
  704             , scAccessControl  roundTripped == scAccessControl cfg
  705             , scUntrustedExec  roundTripped == scUntrustedExec cfg
  706             , scGateway        roundTripped == scGateway cfg
  707             , scVaultPath      roundTripped == scVaultPath cfg
  708             , scVaultUnlock    roundTripped == scVaultUnlock cfg ->
  709                 fmap (>> Right (Right fields))
  710                   (saveSecurityConfig path newCfg PreserveCredentials)
  711           _ -> pure (Left "refusing to write security.toml: the encoded document does not round-trip the new vault key fields")
  712   where
  713     reDecode encoded = case Toml.parse encoded of
  714       Left _     -> Nothing
  715       Right toml -> case Toml.runTomlCodec securityConfigCodec toml of
  716         Success c -> Just c
  717         Failure _ -> Nothing

  719 -- | Remove the @[access_control]@ table entirely, returning the gateway to the
  720 -- unconfigured state. Reachable only from the @seal access-control reset@ CLI
  721 -- command (the human-driven path); no opcode and no HTTP route can call it.
  722 clearAccessControlSection :: FilePath -> WriteMode -> IO (Either Text ())
  723 clearAccessControlSection path mode = withMVar securityWriteLock $ \_ -> do
  724   loaded <- loadSecurityConfig path
  725   case loaded of
  726     Left err  -> pure (Left err)
  727     Right cfg -> saveSecurityConfig path (cfg { scAccessControl = Nothing }) mode

  729 -- | Resolve the @[gateway]@ section to the 'GatewayConfig' the boot path and
  730 -- the CLI consume. An absent section yields
  731 -- 'Seal.Gateway.Config.defaultGatewayConfig' (loopback, port 8080); present
  732 -- keys override it field by field.
  733 gatewayConfigFromSecurity :: SecurityConfig -> GatewayConfig
  734 gatewayConfigFromSecurity cfg =
  735   maybe defaultGatewayConfig withGatewayDefaults (scGateway cfg)

  737 -- | The section a first write starts from when the file has none.
  738 --
  739 -- 'acfcMode' is deliberately 'Nothing', not @Just \"none\"@. Any write to a
  740 -- section-less file would otherwise record an /implicit/ waiver, and invariant
  741 -- 1 says absence never means off: an open gateway has to be written out by a
  742 -- human. With 'Nothing', the write leaves the mode unstated, and boot resolves
  743 -- it from whether credentials exist — @passkey@ if they do, @unconfigured@ if
  744 -- they do not.
  745 emptyAccessControl :: AccessControlFileConfig
  746 emptyAccessControl = AccessControlFileConfig
  747   { acfcMode = Nothing
  748   , acfcOrigin = Nothing
  749   , acfcRpId = Nothing
  750   }

  752 -- | Resolve the 'SecurityConfig'\'s @[untrusted_execution]@ section to the
  753 -- typed 'UntrustedExecConfig' the dispatcher consumes. Returns
  754 -- 'Nothing' when the section is absent OR @mode=local@ (the default —
  755 -- no remote needed, the local executor is wired by the call site).
  756 -- Returns @Just (UemRemote, ...remote...)@ when @mode=remote@,
  757 -- with the remote 'SshConfig' if the remote block is fully present
  758 -- (host/user/known_hosts/workspace all set). A @mode=remote@ with a
  759 -- missing/incomplete remote block returns @Just (UemRemote, Nothing)@
  760 -- — fail-closed is at call time (spec §7 row 1), not at config resolution.
  761 untrustedExecConfigFromSecurity :: SecurityConfig -> Maybe UntrustedExecConfig
  762 untrustedExecConfigFromSecurity cfg =
  763   resolveByMode =<< scUntrustedExec cfg
  764 
  765 resolveByMode :: UntrustedExecFileConfig -> Maybe UntrustedExecConfig
  766 resolveByMode uec =
  767   let mode = if uefcMode uec == "remote" then UemRemote else UemLocal
  768   in case mode of
  769        UemRemote -> Just (UntrustedExecConfig UemRemote (resolveRemoteSsh uec))
  770        UemLocal  -> localModeResult uec
  771 
  772 -- | Resolve the [untrusted_execution.remote] sub-table to an 'SshConfig'.
  773 -- Shared by both the UemRemote branch and the hardened-build localModeResult.
  774 resolveRemoteSsh :: UntrustedExecFileConfig -> Maybe SshConfig
  775 resolveRemoteSsh uec = uefcRemote uec >>= \r ->
  776   uerfcHost r >>= \host ->
  777   uerfcUser r >>= \user ->
  778   uerfcKnownHosts r >>= \knownHosts ->
  779   uerfcWorkspace r >>= \workspace ->
  780   let port = fromMaybe 22 (uerfcPort r)
  781       identity = uerfcIdentity r
  782   in case ( mkSshHost host
  783           , mkSshUser user
  784           , mkRemotePath workspace
  785           ) of
  786        (Right h, Right u, Right w) -> Just SshConfig
  787          { scHost       = h
  788          , scUser       = u
  789          , scPort       = port
  790          , scIdentity   = identity
  791          , scKnownHosts = knownHosts
  792          , scWorkspace  = w
  793          }
  794        _ -> Nothing
  795 
  796 #if defined(REMOTE_ONLY_UNTRUSTED)
  797 localModeResult :: UntrustedExecFileConfig -> Maybe UntrustedExecConfig
  798 localModeResult uec =
  799   Just (UntrustedExecConfig UemRemote (resolveRemoteSsh uec))
  800 #else
  801 localModeResult :: UntrustedExecFileConfig -> Maybe UntrustedExecConfig
  802 localModeResult _ = Nothing
  803 #endif

  809 -- | What a mode raise needs in order to be undone.
  810 --
  811 -- The constructor is deliberately not exported. @RaisedFrom@ is the only value
  812 -- 'restoreMode' accepts as proof that /this/ code path raised the mode, so a
  813 -- caller cannot conjure one and lower the mode to @none@: the web path can turn
  814 -- access control on, and cannot turn it off, as a type fact rather than a
  815 -- convention. @AlreadyRaised@ restores nothing.
  816 data ModeRaise
  817   = AlreadyRaised
  818   | RaisedFrom !(Maybe Text)
  819   deriving stock (Eq, Show)
  820 
  821 -- | Raise @access_control.mode@ to @passkey@, remembering what it was.
  822 --
  823 -- __The caller must hold 'securityWriteLock'.__ It is not taken here because the
  824 -- only caller that needs it ('Seal.Config.Credentials.enrollCredential') takes
  825 -- it once around the raise /and/ the credential write, and the lock is
  826 -- non-reentrant.
  827 --
  828 -- @mode@ is the only field this touches; origin, rp_id and @untrusted_execution@
  829 -- are unreachable from here by construction.
  830 raiseMode :: FilePath -> IO (Either Text ModeRaise)
  831 raiseMode path = do
  832   loaded <- loadSecurityConfig path
  833   case loaded of
  834     Left err  -> pure (Left err)
  835     Right cfg -> do
  836       let section = fromMaybe emptyAccessControl (scAccessControl cfg)
  837       case acfcMode section of
  838         Just "passkey" -> pure (Right AlreadyRaised)
  839         previous -> do
  840           written <- saveSecurityConfig path
  841                        (cfg { scAccessControl = Just (section { acfcMode = Just "passkey" }) })
  842                        PreserveCredentials
  843           pure (RaisedFrom previous <$ written)
  844 
  845 -- | Undo a raise — but only if the on-disk mode is still the @passkey@ the
  846 -- raise wrote.
  847 --
  848 -- Compare-and-swap, not a blind write: if anything else moved the mode in the
  849 -- meantime (an operator running @disable --yes@, another ceremony), restoring
  850 -- would silently undo their decision. Leaving it alone is the safe direction —
  851 -- the gateway stays closed and the operator can still enroll.
  852 --
  853 -- __The caller must hold 'securityWriteLock'.__
  854 restoreMode :: FilePath -> ModeRaise -> IO (Either Text ())
  855 restoreMode _ AlreadyRaised = pure (Right ())
  856 restoreMode path (RaisedFrom previous) = do
  857   loaded <- loadSecurityConfig path
  858   case loaded of
  859     Left err  -> pure (Left err)
  860     Right cfg -> case scAccessControl cfg >>= acfcMode of
  861       Just "passkey" ->
  862         saveSecurityConfig path
  863           (cfg { scAccessControl =
  864                    fmap (\section -> section { acfcMode = previous }) (scAccessControl cfg) })
  865           PreserveCredentials
  866       _ -> pure (Right ())

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | SEARCH_FILES (Untrusted): search workspace files for a pattern. Runs
    3 -- @rg -n -- <pattern> <path>@ via the 'UntrustedIO' capability (fixed
    4 -- argv, no shell interpreter — the validated 'SearchPattern' newtype
    5 -- guards against option injection). 'SafePath'-confined. Bounded result
    6 -- count (operator-configured ceiling; the model can narrow via
    7 -- @max_results@). This module never imports 'System.Process'.
    8 module Seal.ISA.Ops.Search
    9   ( searchFilesOp
   10   ) where

   27 -- | SEARCH_FILES opcode. Input: @{ pattern: Text, path?: Text, max_results?:
   28 -- Int }@. The pattern must not start with @-@ (option injection). The
   29 -- @path@ defaults to @.@ (the workspace root). The result count is bounded
   30 -- by the operator-configured @maxResults@ ceiling; the model's
   31 -- @max_results@ request can only narrow it.
   32 searchFilesOp :: WorkspaceRoot -> SecurityPolicy -> Int -> Opcode
   33 searchFilesOp _wsRoot policy maxResults = UntrustedOpcode
   34   { uoName = OpName "SEARCH_FILES"
   35   , uoDesc = "Search workspace files for a pattern (rg, SafePath-confined, bounded)."
   36   , uoInSchema = searchFilesSchema
   37   , uoOutSchema = object []
   38   , uoAuthorize = \v ->
   39       case patternField v of
   40         Nothing -> Left "SEARCH_FILES requires {pattern:string}"
   41         Just p
   42           | T.null p -> Left "SEARCH_FILES: pattern is empty"
   43           | T.head p == '-' -> Left "SEARCH_FILES: pattern must not start with '-' (option injection)"
   44           | otherwise -> case spAutonomy policy of
   45               Deny -> Left "SEARCH_FILES denied by autonomy policy"
   46               _   -> Right ()
   47   , uoRun = \v -> do
   48       let pat    = fromMaybe "" (patternField v)
   49           pth    = fromMaybe "." (pathField v)
   50           limit  = clampResults maxResults (resultsField v)
   51       case mkSearchPattern pat of
   52         Left err -> pure (OpResult [TrpText ("SEARCH_FILES: " <> err)] True
   53                            (object ["pattern" .= pat, "path" .= pth]))
   54         Right pat' -> do
   55           let mPath = case mkRemotePath pth of
   56                 Right rp -> Just rp
   57                 Left _   -> Nothing
   58           res <- uioSearchFiles pat' mPath limit
   59           case res of
   60             Left err -> pure (OpResult [TrpText (renderUntrustedErr err)] True
   61                               (object ["pattern" .= pat, "path" .= pth]))
   62             Right out -> do
   63               let ls = take limit (T.lines out)
   64                   count = length ls
   65               pure (OpResult [TrpText (T.intercalate "\n" ls)] False
   66                      (object ["pattern" .= pat, "path" .= pth, "result_count" .= count]))
   67   }
   68 
   69 searchFilesSchema :: Value
   70 searchFilesSchema =
   71   object
   72     [ "type" .= ("object" :: Text)
   73     , "properties" .= object
   74         [ "pattern" .= object
   75             [ "type" .= ("string" :: Text)
   76             , "description" .= ("The search pattern (must not start with '-')." :: Text)
   77             ]
   78         , "path" .= object
   79             [ "type" .= ("string" :: Text)
   80             , "description" .= ("Optional workspace-relative path (defaults to '.')." :: Text)
   81             ]
   82         , "max_results" .= object
   83             [ "type" .= ("integer" :: Text)
   84             , "description" .= ("Max results to return (clamped to the operator ceiling)." :: Text)
   85             ]
   86         ]
   87     , "required" .= (["pattern"] :: [Text])
   88     ]
   89 
   90 patternField :: Value -> Maybe Text
   91 patternField = parseMaybe (withObject "in" (.: "pattern"))
   92 
   93 pathField :: Value -> Maybe Text
   94 pathField v = case parseMaybe (withObject "in" (.:? "path")) v :: Maybe (Maybe Text) of
   95   Just (Just t) -> Just t
   96   _             -> Nothing
   97 
   98 resultsField :: Value -> Maybe Int
   99 resultsField v = case parseMaybe (withObject "in" (.:? "max_results")) v :: Maybe (Maybe Int) of
  100   Just (Just n) | n >= 0 -> Just n
  101   _                      -> Nothing
  102 
  103 clampResults :: Int -> Maybe Int -> Int
  104 clampResults ceiling' mReq =
  105   case mReq of
  106     Nothing  -> ceiling'
  107     Just req -> max 1 (min ceiling' req)

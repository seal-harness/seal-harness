   25 sessionCommandSpec :: SessionRuntime -> CommandSpec
   26 sessionCommandSpec sr = CommandSpec
   27   { csName         = CommandName "session"
   28   , csAliases      = []
   29   , csGroup        = GroupSession
   30   , csSynopsis     = "List sessions and show the active one"
   31   , csParserInfo   = sessionParserInfo sr
   32   , csAvailability = InteractiveOnly
   33   }
   34 
   35 sessionParserInfo :: SessionRuntime -> ParserInfo CommandAction
   36 sessionParserInfo sr =
   37   info (sessionParser sr <**> helper)
   38     (  progDesc "Inspect chat sessions"
   39     <> header   "session — list sessions and show the active one"
   40     )
   41 
   42 sessionParser :: SessionRuntime -> Parser CommandAction
   43 sessionParser sr = hsubparser
   44   (  command "list"
   45        (info (pure (listCmd sr)) (progDesc "List all sessions (newest first)"))
   46   <> command "info"
   47        (info (pure (infoCmd sr)) (progDesc "Show the active session's details"))
   48   <> metavar "COMMAND"
   49   )
   50 
   51 listCmd :: SessionRuntime -> CommandAction
   52 listCmd sr = commandAction $ \caps -> do
   53   active <- readIORef (srActive sr)
   54   metas  <- listSessions (srPaths sr)
   55   if null metas
   56     then ccSend caps "no sessions yet"
   57     else do
   58       snippets <- mapM (firstUserMessageSnippet (srPaths sr) . smId) metas
   59       mapM_ (ccSend caps . uncurry (renderSessionLine (smId active)))
   60             (zip snippets metas)
   61 
   62 infoCmd :: SessionRuntime -> CommandAction
   63 infoCmd sr = commandAction $ \caps -> do
   64   active <- readIORef (srActive sr)
   65   mapM_ (ccSend caps) (renderSessionInfo active)
   66 
   67 -- | One line per session for @/session list@, marking the active one.
   68 -- Shows the session id, provider/model, the bound agent (if any), and the
   69 -- first user message snippet (if any) — the same title cascade the web
   70 -- frontend uses, minus the timestamp (which is redundant with the id).
   71 renderSessionLine :: SessionId -> Maybe Text -> SessionMeta -> Text
   72 renderSessionLine active mSnippet m =
   73   let mark = if smId m == active then "  (active)" else ""
   74       agentLabel = case (smAgentName m, smAgent m) of
   75         (Just name, _)   -> name
   76         (Nothing, Just a) -> agentDefIdText a
   77         (Nothing, Nothing) -> ""
   78       agentPart = if T.null agentLabel then "" else "  " <> agentLabel
   79       snippetPart = case mSnippet of
   80         Just s  -> "  " <> s
   81         Nothing -> ""
   82   in sessionIdText (smId m)
   83        <> "  " <> smProvider m <> "/" <> smModel m
   84        <> agentPart
   85        <> snippetPart
   86        <> mark
   87 
   88 -- | Multi-line detail for @/session info@.
   89 renderSessionInfo :: SessionMeta -> [Text]
   90 renderSessionInfo m =
   91   [ "id:          " <> sessionIdText (smId m)
   92   , "provider:    " <> smProvider m
   93   , "model:       " <> smModel m
   94   , "channel:     " <> smChannel m
   95   , "agent:       " <> maybe "(none)" agentDefIdText (smAgent m)
   96   , "created:     " <> T.pack (show (smCreatedAt m))
   97   , "last active: " <> T.pack (show (smLastActive m))
   98   ]

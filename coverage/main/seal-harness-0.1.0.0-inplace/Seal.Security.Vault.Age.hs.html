   31 -- | Run a process with a strict 'ByteString' stdin, capturing stdout and
   32 -- stderr as strict 'ByteString's. Used by the age encrypt/decrypt seam
   33 -- where the payload is binary (ciphertext) and must not be round-tripped
   34 -- through 'String'. Closes all handles and waits for the child before
   35 -- returning.
   36 readProcessBinary :: FilePath -> [String] -> ByteString
   37                  -> IO (ExitCode, ByteString, ByteString)
   38 readProcessBinary cmdPath args input =
   39   withCreateProcess
   40     ( (proc cmdPath args)
   41         { std_in = CreatePipe, std_out = CreatePipe, std_err = CreatePipe }
   42     ) $ \mIn mOut mErr ph -> do
   43       (hIn, hOut, hErr) <- case (mIn, mOut, mErr) of
   44         (Just a, Just b, Just c) -> pure (a, b, c)
   45         _ -> error "readProcessBinary: pipe creation failed (unreachable)"
   46       -- Feed stdin on a child thread so we can read stdout/stderr
   47       -- concurrently without deadlocking on a large payload.
   48       BS.hPutStr hIn input
   49       hFlush hIn
   50       hClose hIn
   51       -- Read stdout/stderr fully after stdin is closed (age reads the
   52       -- whole plaintext before emitting ciphertext, so this is safe).
   53       out <- BS.hGetContents hOut
   54       err <- BS.hGetContents hErr
   55       ec  <- waitForProcess ph
   56       -- Force the lazy hGetContents results so the handles are fully read
   57       -- before withCreateProcess closes them.
   58       let !_ = BS.length out
   59           !_ = BS.length err
   60       pure (ec, out, err)

   62 -- | The vault's error type. The first four constructors are matched on to
   63 -- drive control flow (so they earn being a sum type per the haskell-coder
   64 -- skill); 'VaultBackendError' is the single catch-all for failures that are
   65 -- only reported — @age@ stderr, "age not installed", corrupt vault data.
   66 data VaultError
   67   = VaultLocked
   68   | VaultNotFound
   69   | VaultAlreadyExists
   70   | VaultKeyNotFound Text
   71   | VaultBackendError Text
   72   deriving stock (Eq, Show)

   77 -- | Validate an age recipient (public key).
   78 --
   79 -- The shape is checked, the bech32 checksum cannot be: a wrong-but-well-formed
   80 -- recipient only fails when @age@ actually encrypts with it, which is why
   81 -- 'Seal.Vault.Provision' round-trips a nonce before recording anything.
   82 --
   83 -- Rejections: empty, longer than any real recipient, a control character or
   84 -- whitespace, a leading @-@ (option injection), and any character outside the
   85 -- bech32 alphabet used by age (@a-z0-9@ after the @age1@ prefix; yubikey
   86 -- recipients use @age1yubikey1@).
   87 mkAgeRecipient :: Text -> Either Text AgeRecipient
   88 mkAgeRecipient raw
   89   | T.null t = Left "recipient is empty"
   90   | T.length t > 512 = Left "recipient is implausibly long"
   91   | T.head t == '-' = Left "recipient must not start with '-'"
   92   | "age1" `T.isPrefixOf` t = checkRest (T.drop 4 t)
   93   | otherwise = Left "recipient must start with \"age1\""
   94   where
   95     t = T.strip raw
   96     checkRest rest
   97       | T.null rest = Left "recipient is truncated"
   98       | T.all isBech32Char rest = Right (AgeRecipient t)
   99       | otherwise = Left "recipient has characters outside the age alphabet"
  100     isBech32Char c = isAsciiLower c || isDigit c

  102 -- | The argv @age@ is invoked with to encrypt under a recipient.
  103 --
  104 -- Single-token option forms (@--recipient=\<value\>@) so no value can ever be
  105 -- reinterpreted as a flag, even if a future parser change lets one through.
  106 ageEncryptArgs :: AgeRecipient -> [String]
  107 ageEncryptArgs (AgeRecipient recipient) = ["--encrypt", "--recipient=" <> T.unpack recipient]
  108 
  109 -- | The argv @age@ is invoked with to decrypt with an identity file.
  110 ageDecryptArgs :: AgeIdentity -> [String]
  111 ageDecryptArgs (AgeIdentity identity) = ["--decrypt", "--identity=" <> T.unpack identity]

  113 -- | Encrypt/decrypt with credentials already captured in the closure.
  114 data VaultEncryptor = VaultEncryptor
  115   { veEncrypt :: ByteString -> IO (Either VaultError ByteString)
  116   , veDecrypt :: ByteString -> IO (Either VaultError ByteString)
  117   }

  119 -- | Build a real encryptor backed by @age@. Preflights @age --version@ and
  120 -- reports a 'VaultBackendError' install hint if the binary is absent.
  121 --
  122 -- Both credentials are validated here before anything is spawned: this is the
  123 -- single choke point every caller goes through (the CLI, the gateway, a
  124 -- hand-edited @security.toml@), so a malformed value can never reach @age@'s
  125 -- argv. The recipient's checksum cannot be checked without @age@ — that is what
  126 -- the provisioning round-trip is for.
  127 mkAgeEncryptor :: AgeRecipient -> AgeIdentity -> IO (Either VaultError VaultEncryptor)
  128 mkAgeEncryptor recipient identity =
  129   case (mkAgeRecipient (recipientText recipient), validateAgeIdentity identity) of
  130     (Left err, _) -> pure (Left (VaultBackendError ("invalid recipient: " <> err)))
  131     (_, Left err) -> pure (Left (VaultBackendError ("invalid identity path: " <> err)))
  132     (Right _, Right ()) -> do
  133       versionResult <- try @IOException (callProcessNoOutput "age" ["--version"])
  134       case versionResult of
  135         Right ExitSuccess ->
  136           pure (Right VaultEncryptor
  137             { veEncrypt = run (ageEncryptArgs recipient)
  138             , veDecrypt = run (ageDecryptArgs identity)
  139             })
  140         _ -> pure notInstalled  -- IOException (binary absent) OR ExitFailure
  141   where
  142     recipientText (AgeRecipient t) = t
  143     notInstalled =
  144       Left (VaultBackendError "age not installed; see https://age-encryption.org")
  145     run :: [String] -> ByteString -> IO (Either VaultError ByteString)
  146     run args input = do
  147       (code, out, err) <- readProcessBinary "age" args input
  148       pure $ case code of
  149         ExitSuccess   -> Right out
  150         ExitFailure _ -> Left (VaultBackendError (TE.decodeUtf8Lenient err))
  151 
  152 -- | Structural validation of an identity path: non-empty, absolute, no
  153 -- control characters, no leading @-@. Existence is checked by the caller
  154 -- ("Seal.Vault.IdentityPath" does it for user-supplied paths; the boot path
  155 -- needs the encryptor to exist before it can stat anything).
  156 validateAgeIdentity :: AgeIdentity -> Either Text ()
  157 validateAgeIdentity (AgeIdentity path) = do
  158   let t = T.strip path
  159   if T.null t then Left "identity path is empty" else Right ()
  160   if T.head t == '-' then Left "identity path must not start with '-'" else Right ()
  161   if not (T.isPrefixOf "/" t) then Left "identity path must be absolute" else Right ()
  162   if T.any (\c -> c < ' ' || c == '\DEL') t
  163     then Left "identity path contains a control character"
  164     else Right ()

  166 -- | Run a process, ignoring its stdout/stderr, returning only the exit code.
  167 -- Used for the @age --version@ preflight. Catches a launch failure as
  168 -- 'ExitFailure 127' (consistent with the git seam's convention).
  169 callProcessNoOutput :: FilePath -> [String] -> IO ExitCode
  170 callProcessNoOutput cmdPath args =
  171   withCreateProcess ((proc cmdPath args) { std_out = NoStream, std_err = NoStream }) $
  172     \_ _ _ ph -> waitForProcess ph

  174 -- | XOR-with-0xAB mock; reversible, no binary required.
  175 mkMockEncryptor :: VaultEncryptor
  176 mkMockEncryptor = VaultEncryptor
  177   { veEncrypt = pure . Right . BS.map (`xor` 0xAB)

   34 -- | The live Signal channel state.
   35 data SignalChannel = SignalChannel
   36   { scAllowList    :: AllowList UserId  -- ^ resolved sender allow-list
   37   , scChunkLimit   :: Int               -- ^ chunk limit for sends
   38   , scAccount      :: SignalAccount
   39   , scInbox        :: TQueue (MessageSource, Text)
   40   , scTransport    :: SignalTransport
   41   , scLastSender   :: IORef (Maybe UserId)
   42   , scReaderAlive  :: IORef Bool
   43     -- ^ 'True' while the reader thread is running. 'chReceive' stops
   44     -- returning once the inbox drains and this is 'False'.
   45   , scLogger       :: SealLogger
   46     -- ^ The shared logger for structured katip logging.
   47   }
   48 
   49 instance Channel SignalChannel where
   50   toHandle ch = ChannelHandle
   51     { chLabel       = "signal"
   52     , chSend         = sendChunked ch
   53     , chSendError    = \t -> sendChunked ch ("error: " <> t)
   54     , chSendChunk    = sendRaw ch
   55     , chPrompt       = \_ -> pure (Left Deferred)   -- Signal can't answer inline
   56     , chPromptSecret = \_ -> pure (Left Deferred)
   57     , chStreaming    = False  -- per-token messages flood the chat; send accumulated text once
   58     , chReadSecret   = pure Nothing                  -- vault is reached via the vault handle
   59     , chReceive      = receiveFromInbox ch
   60     , chLastChatId   = pure Nothing                  -- Signal addresses by user id, not chat id
   61     }

   63 -- | Run the reader thread with cleanup. Spawns a background thread that
   64 -- loops 'stReceive' + 'parseSignalEnvelope', allow-lists the sender, and
   65 -- pushes @(MessageSource, body)@ to 'scInbox'. On transport close or
   66 -- exception, the thread exits. The action runs with the channel; on exit
   67 -- the transport is closed and the reader is killed.
   68 withSignalChannel
   69   :: (AllowList UserId, Int)
   70   -> SignalAccount
   71   -> SignalTransport
   72   -> SealLogger
   73   -> (SignalChannel -> IO a)
   74   -> IO a
   75 withSignalChannel (allow, chunkLimit) account transport logger action =
   76   bracket before after (action . snd)
   77   where
   78     before = do
   79       inbox      <- newTQueueIO
   80       lastSender <- newIORef Nothing
   81       alive      <- newIORef True
   82       let ch = SignalChannel
   83             { scAllowList  = allow
   84             , scChunkLimit = chunkLimit
   85             , scAccount    = account
   86             , scInbox      = inbox
   87             , scTransport  = transport
   88             , scLastSender = lastSender
   89             , scReaderAlive = alive
   90             , scLogger     = logger
   91             }
   92       tid <- forkIO (readerLoop ch)
   93       pure (tid, ch)
   94     after (tid, _ch) = do
   95       killThread tid
   96       stClose transport
   97 
   98 -- | The background reader: loop 'stReceive', parse each inbound value,
   99 -- allow-list the sender, and push to the inbox. A malformed line or a
  100 -- non-allow-listed sender is logged via 'logIO' and dropped (not fatal).
  101 -- Exits when 'stReceive' returns 'Left' (the channel is closing).
  102 -- 'ThreadKilled' is logged at 'InfoS' (normal shutdown); other
  103 -- 'AsyncException's at 'WarningS'; synchronous exceptions at 'ErrorS'.
  104 readerLoop :: SignalChannel -> IO ()
  105 readerLoop ch = go
  106   where
  107     logger = scLogger ch
  108     go = do
  109       eVal <- try @SomeException (stReceive (scTransport ch))
  110       case eVal of
  111         Left e -> do
  112           case (fromException e :: Maybe AsyncException) of
  113             Just ThreadKilled ->
  114               logIO logger InfoS "reader thread stopped (channel shutting down)"
  115             Just _ ->
  116               logIO logger WarningS ("reader thread terminated by async exception: " <> ls (T.pack (show e)))
  117             Nothing ->
  118               logIO logger ErrorS ("reader thread exception: " <> ls (T.pack (show e)))
  119           writeIORef (scReaderAlive ch) False
  120         Right (Left err) -> do
  121           logIO logger ErrorS ("reader exiting: " <> ls err)
  122           writeIORef (scReaderAlive ch) False
  123         Right (Right val) -> do
  124           case parseSignalEnvelope val of
  125             Left err -> logIO logger WarningS ("envelope parse error: " <> ls err)
  126             Right env
  127               | T.null (seBody env) -> go
  128                 -- Skip non-data envelopes (receipts, typing, sync messages).
  129                 -- signal-cli emits these constantly; only dataMessage carries
  130                 -- real text. Avoids tight-cycling the loop with empty envelopes.
  131               | otherwise -> do
  132                 let sender = seSender env
  133                 logIO logger DebugS ("signal: inbound message from " <> ls (userIdText sender))
  134                 if isAllowed sender (scAllowList ch)
  135                   then do
  136                     case mkMessageSource (seConversationId env) Signal (Just sender) mempty of
  137                       Left err -> logIO logger WarningS ("MessageSource construction failed: " <> ls err)
  138                       Right ms -> do
  139                         writeIORef (scLastSender ch) (Just sender)
  140                         atomically (writeTQueue (scInbox ch) (ms, seBody env))
  141                   else logIO logger InfoS ("dropped non-allow-listed sender: " <> ls (userIdText sender))
  142                 go
  143 
  144 -- | Send a message, chunked to 'scChunkLimit', addressed to the last sender.
  145 -- If no peer has been seen yet, the send is dropped with a warning log.
  146 sendChunked :: SignalChannel -> Text -> IO ()
  147 sendChunked ch t = do
  148   mSender <- readIORef (scLastSender ch)
  149   case mSender of
  150     Nothing -> logIO (scLogger ch) WarningS "signal: dropping send — no last sender yet"
  151     Just _  -> mapM_ (sendRaw ch) (chunkMessage (scChunkLimit ch) t)
  152 
  153 -- | Send one chunk verbatim to the last sender (no further splitting).
  154 -- Used by 'chSendChunk' (the caller pre-split) and by 'sendChunked' for
  155 -- each split chunk. Addressed to the last sender.
  156 sendRaw :: SignalChannel -> Text -> IO ()
  157 sendRaw ch t = do
  158   mSender <- readIORef (scLastSender ch)
  159   case mSender of
  160     Nothing -> logIO (scLogger ch) WarningS "signal: dropping chunk — no last sender yet"
  161     Just uid -> stSend (scTransport ch) (userIdText uid) t
  162 
  163 -- | Pull the next @(MessageSource, body)@ from the inbox. Non-blocking:
  164 -- returns @(Nothing, "")@ when the inbox is empty AND the reader thread has
  165 -- exited (transport closed / EOF). Returns @(Nothing, "")@ immediately in
  166 -- that state so 'runSignalLoop' can terminate rather than block forever.
  167 receiveFromInbox :: SignalChannel -> IO (Maybe MessageSource, Text)
  168 receiveFromInbox ch = do
  169   mNext <- atomically (tryReadTQueue (scInbox ch))
  170   case mNext of
  171     Just (ms, body) -> pure (Just ms, body)
  172     Nothing -> do
  173       alive <- readIORef (scReaderAlive ch)
  174       if alive
  175         then do
  176           -- Inbox momentarily empty but reader still running: retry once
  177           -- after a brief yield to let the reader push the next envelope.
  178           -- (A real transport blocks in stReceive; a mock drains fast.)
  179           threadDelay 1000  -- 1ms
  180           receiveFromInbox ch
  181         else pure (Nothing, "")

   78 -- | The tab list. I1 (contiguous 0..n-1, removal compacts) + I2 (no two tabs
   79 -- share a 'TabRef') are enforced by the smart constructors. Hard 36-slot cap.
   80 newtype TabList = TabList { tlTabs :: [Tab] }
   81   deriving stock (Eq, Show, Generic)
   82   deriving anyclass (ToJSON, FromJSON)
   83 
   84 -- | Construct an empty 'TabList'.
   85 emptyTabList :: TabList
   86 emptyTabList = TabList []

   88 -- | The number of tabs (also the next free slot under I1).
   89 tabCount :: TabList -> Int
   90 tabCount = length . tlTabs

   92 -- | The maximum number of tabs (the hard 36-slot cap).
   93 maxTabs :: Int
   94 maxTabs = 36

   96 -- | Insert a tab at the lowest free slot (I1). 'Left' if the 'TabRef' is
   97 -- already present (I2) or the list is full (36).
   98 insertTab :: TabRef -> TabKind -> Maybe Text -> TabList -> Either Text TabList
   99 insertTab ref kind label (TabList ts)
  100   | ref `elem` map tRef ts = Left "tab ref already bound"
  101   | length ts >= maxTabs   = Left "tab list full (36 slots)"
  102   | otherwise =
  103       -- The index is the lowest free slot. Since the list is contiguous
  104       -- (I1), that's just `length ts` (the next slot after the last).
  105       let idx = mkTabIndex (length ts)
  106       in case idx of
  107            Right i -> Right (TabList (ts <> [Tab i ref kind label Live]))
  108            Left e  -> Left e  -- unreachable: length < 36 implies idx < 36
  109 
  110 -- | Look up a tab by index. 'Nothing' if the index is out of range.
  111 lookupTab :: TabList -> TabIndex -> Maybe Tab
  112 lookupTab (TabList ts) i = go ts
  113   where
  114     go [] = Nothing
  115     go (t:rest)
  116       | tIndex t == i = Just t
  117       | otherwise     = go rest
  118 
  119 -- | Look up a tab by its 'TabRef'. 'Nothing' if absent.
  120 lookupByRef :: TabList -> TabRef -> Maybe Tab
  121 lookupByRef (TabList ts) ref = go ts
  122   where
  123     go [] = Nothing
  124     go (t:rest)
  125       | tRef t == ref = Just t
  126       | otherwise    = go rest
  127 
  128 -- | Remove a tab by index. Compacts the list (I1: slots renumber to 0..n-1).
  129 -- 'Left' if the index is out of range.
  130 removeTab :: TabList -> TabIndex -> Either Text TabList
  131 removeTab (TabList ts) i =
  132   case break (\t -> tIndex t == i) ts of
  133     (_, []) -> Left "tab index out of range"
  134     (before, _t : after) ->
  135       -- Renumber the surviving tabs 0..n-2 (I1: contiguous after compaction).
  136       let survivors = before <> after
  137           renumbered = zipWith (\n t -> t { tIndex = mkIdx n }) [0..] survivors
  138       in Right (TabList renumbered)
  139   where
  140     mkIdx n = case mkTabIndex n of
  141       Right x -> x
  142       Left _  -> error ("removeTab: renumber out of range (unreachable, n=" <> show n <> ")")
  143 
  144 -- | Rename a tab (set its label). 'Left' if the index is out of range.
  145 renameTab :: TabList -> TabIndex -> Text -> Either Text TabList
  146 renameTab (TabList ts) i name =
  147   case go ts of
  148     Nothing -> Left "tab index out of range"
  149     Just ts' -> Right (TabList ts')
  150   where
  151     go [] = Nothing
  152     go (t:rest)
  153       | tIndex t == i = Just (t { tLabel = Just name } : rest)
  154       | otherwise     = case go rest of
  155           Just rest' -> Just (t : rest')
  156           Nothing   -> Nothing
  157 
  158 -- | Rebind a tab's 'TabRef' in place — preserve its index, kind, label, and
  159 -- status, only swapping @tRef@. Used by @\/new@ to point an existing tab at a
  160 -- fresh session.
  161 --
  162 -- I2 (no two tabs share a 'TabRef') is preserved: 'Left' if the new ref is
  163 -- already bound to a /different/ tab. Rebinding a tab to its own current
  164 -- ref is a no-op 'Right' (idempotent — the I2 check excludes the target
  165 -- slot itself). 'Left' if the index is out of range.
  166 rebindTab :: TabIndex -> TabRef -> TabList -> Either Text TabList
  167 rebindTab idx newRef (TabList ts) =
  168   case break (\t -> tIndex t == idx) ts of
  169     (_, []) -> Left "tab index out of range"
  170     (before, target : after) ->
  171       if newRef `elem` map tRef (before <> after)
  172         then Left "tab ref already bound to another tab"
  173         else Right (TabList (before <> [target { tRef = newRef }] <> after))
  174 
  175 -- | The current slot of a 'TabRef' (I3: resolved at read time). 'Nothing' if
  176 -- the ref is no longer in the list (the cursor is stale).
  177 slotOf :: TabList -> TabRef -> Maybe TabIndex
  178 slotOf tl ref = tIndex <$> lookupByRef tl ref

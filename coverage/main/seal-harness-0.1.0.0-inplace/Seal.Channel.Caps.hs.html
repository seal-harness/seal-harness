    1 {-# LANGUAGE OverloadedStrings #-}
    2 module Seal.Channel.Caps
    3   ( ChannelCaps(..)
    4   , AskPrompt(..)
    5   ) where
    6 
    7 import Data.Default (Default (..))
    8 import Data.Text (Text)
    9 
   10 import Seal.Handles.AskReply (QuestionOption (..))
   11 
   12 -- | A structured prompt for 'ccPrompt': the question text plus the offered
   13 -- 'QuestionOption's (empty for open-ended questions and the confirmation
   14 -- gate). Channels render the options in the richest way their transport
   15 -- supports (web: buttons + "Other" textarea; Signal/CLI: numbered list;
   16 -- Telegram: inline keyboard) and return the human's chosen text.
   17 data AskPrompt = AskPrompt
   18   { apQuestion :: !Text
   19   , apOptions  :: ![QuestionOption]
   20   }
   21 
   22 -- | A channel's interaction capabilities as a record of IO functions
   23 -- (house style: no type class; callers receive the handle and call fields
   24 -- directly). Web deferral of prompts is a later phase; the CLI TUI
   25 -- is always interactive.
   26 data ChannelCaps = ChannelCaps
   27   { ccSend         :: Text -> IO ()   -- ^ Emit one line to the user
   28   , ccPrompt       :: AskPrompt -> IO Text -- ^ Visible prompt; returns typed line
   29   , ccPromptSecret :: Text -> IO Text -- ^ Hidden (no-echo) prompt
   30   , ccStreaming    :: Bool
   31   -- ^ Whether the channel wants per-delta sends during streaming (CLI,
   32   -- web). When 'False' (Telegram, Signal), the loop skips per-delta
   33   -- 'ccSend' and sends the accumulated text once at the end via the
   34   -- normal stop path — avoiding a flood of per-token messages.
   35   , ccShowHuman    :: Text -> IO ()
   36   -- ^ Display a message to the human operator, guaranteed to be shown
   37   -- regardless of channel-specific verbosity settings or streaming
   38   -- suppression. Unlike 'ccSend' (which may be subject to future
   39   -- verbosity / intermediate-output suppression in chat channels), this
   40   -- is the SHOW_HUMAN contract: the message WILL reach the user. Web
   41   -- channels can leave this as a no-op (the message surfaces via the
   42   -- transcript + ToolCallBlock rendering); chat channels wire it to
   43   -- their 'chSend'.
   44   }
   45 
   46 -- | The default 'ChannelCaps': all IO actions are no-ops and streaming is
   47 -- enabled. Production callers override at least @ccSend@, @ccPrompt@, and
   48 -- @ccPromptSecret@; test callers often override only @ccSend@. The
   49 -- @ccStreaming@ default is 'True' (matching the CLI and web channels, where
   50 -- per-delta sends are desired or harmless).
   51 instance Default ChannelCaps where
   52   def = ChannelCaps
   53     { ccSend         = \_ -> pure ()
   54     , ccPrompt       = \_ -> pure ""
   55     , ccPromptSecret = \_ -> pure ""
   56     , ccStreaming    = True
   57     , ccShowHuman    = \_ -> pure ()
   58     }

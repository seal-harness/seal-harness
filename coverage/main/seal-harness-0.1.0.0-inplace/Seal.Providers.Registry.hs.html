   68 -- | Every provider Seal can build. M1 ships Anthropic only; later milestones
   69 -- extend this sum (the totality of the functions below forces each addition to
   70 -- be handled everywhere).
   71 data KnownProvider
   72   = AnthropicProvider
   73   | OllamaProvider
   74   | OllamaCloudProvider
   75     -- ^ Ollama's hosted inference over the OpenAI-compatible API
   76     -- (@https://ollama.com/v1@). Unlike 'OllamaProvider' this is a remote HTTPS
   77     -- endpoint with no keyless mode and no readiness probe. Same client as
   78     -- OpenRouter; the difference is a built-in endpoint.
   79   | OpenRouterProvider
   80   | OpenAIProvider
   81     -- ^ Real OpenAI (@https://api.openai.com/v1@), with a built-in endpoint.
   82     -- The generic 'OpenAICompatibleProvider' used to carry this id; the rename
   83     -- gives it to the thing the name always described.
   84   | OpenAICompatibleProvider
   85     -- ^ Any OpenAI-compatible @/v1/chat/completions@ endpoint the operator
   86     -- names (a self-hosted vLLM, LM Studio, …). Same client as OpenRouter
   87     -- ('mkOpenAICompatible'); the difference is that it has **no** built-in base
   88     -- URL — the endpoint is required input, not a default.
   89   deriving stock (Eq, Show, Enum, Bounded)

  106 -- | Parse a label (case-insensitive) back to a 'KnownProvider'.
  107 parseProvider :: Text -> Maybe KnownProvider
  108 parseProvider t =
  109   let needle = T.toCaseFold (T.strip t)
  110   in lookup needle [(T.toCaseFold (providerLabel p), p) | p <- knownProviders]

  112 -- | Validate a provider instance name. An instance name is the config table
  113 -- key (@[providers.\<name\>]@), the value sessions and agent defs store, and
  114 -- the stem of the instance's vault key. The grammar is an allow-list applied
  115 -- to the raw text (no normalisation): 1-32 characters of @a-z@, @0-9@ and
  116 -- @-@, no leading/trailing @-@, and no @--@. @all@ and @default@ are
  117 -- reserved because they are literal @\/api\/providers\/...@ routes.
  118 validateProviderName :: Text -> Either Text ()
  119 validateProviderName name
  120   | T.null name = Left "provider name must not be empty"
  121   | T.length name > 32 =
  122       Left "provider name must be at most 32 characters"
  123   | name `elem` ["all", "default"] =
  124       Left "\"all\" and \"default\" are reserved provider names"
  125   | T.any badChar name =
  126       Left "provider names may contain only a-z, 0-9 and \"-\""
  127   | "-" `T.isPrefixOf` name || "-" `T.isSuffixOf` name =
  128       Left "provider name must not start or end with \"-\""
  129   | "--" `T.isInfixOf` name =
  130       Left "provider name must not contain \"--\""
  131   | name `elem` map providerLabel knownProviders =
  132       Left (name <> " is a built-in provider type; choose another name"
  133               <> " (for example \"my-openai-endpoint\")")
  134   | otherwise = Right ()
  135   where
  136     badChar c =
  137       not (isAsciiLower c || isDigit c || c == '-')

  139 -- | Is @name@ either a built-in provider id or a syntactically valid custom
  140 -- instance name?
  141 isProviderName :: Text -> Bool
  142 isProviderName name =
  143   isJust (parseProvider name)
  144     || either (const False) (const True) (validateProviderName name)

  203 -- | Build a live provider. For Anthropic, stored OAuth tokens take precedence
  204 -- over an API key; a present-but-corrupt OAuth blob fails loudly (the user must
  205 -- re-run @\/provider login@) rather than silently falling back. The @callCounter@
  206 -- is shared across turns so Ollama-synthesized tool-call ids stay unique across
  207 -- a whole session (Ollama responses carry no id, so Seal synthesizes
  208 -- @call_<i>@; the counter must outlive any single 'Ollama' value, which is
  209 -- rebuilt each turn — see 'ProviderRuntime.prCallCounter').
  210 resolveProvider
  211   :: Maybe VaultHandle -> Manager -> RuntimeConfig -> KnownProvider -> ModelId
  212   -> IORef Int -> IO (Either Text SomeProvider)
  213 resolveProvider Nothing _ _ AnthropicProvider _ _ =
  214   pure (Left "vault not configured \x2014 run /vault setup")
  215 resolveProvider (Just vh) mgr _cfg AnthropicProvider model _ = do
  216   eOAuth <- vhGet vh "ANTHROPIC_OAUTH_TOKENS"
  217   case eOAuth of
  218     Right blob -> case deserializeTokens blob of
  219       Left e     -> pure (Left ("stored Anthropic OAuth tokens are unreadable ("
  220                                   <> e <> ") — run /provider login anthropic"))
  221       Right toks -> do
  222         ref <- newIORef toks
  223         let sess = OAuthSession
  224               { osTokens  = ref
  225               , osRefresh = refreshTokens mgr
  226               , osPersist = void . vhPut vh "ANTHROPIC_OAUTH_TOKENS" . serializeTokens
  227               }
  228         _ <- ensureFresh sess
  229         pure (Right (SomeProvider (mkAnthropicOAuth mgr sess model)))
  230     Left _ -> do
  231       eKey <- vhGet vh (vaultKeyName AnthropicProvider)
  232       pure $ case eKey of
  233         Left e         -> Left (credErr AnthropicProvider e)
  234         Right keyBytes -> Right (SomeProvider (mkAnthropic mgr (mkApiKey keyBytes) model))
  235 resolveProvider mvh mgr cfg OllamaProvider model callCounter
  236   | not (ollamaNeedsKey baseUrl) =
  237       -- local/custom host: keyless, never touch the vault
  238       Right . SomeProvider <$> mkOllama mgr baseUrl Nothing model callCounter
  239   | otherwise = case mvh of
  240       Nothing ->
  241         pure (Left "Ollama Cloud needs an API key \x2014 run /vault setup then /provider add ollama")
  242       Just vh -> do
  243         eKey <- vhGet vh (vaultKeyName OllamaProvider)
  244         case eKey of
  245           Right kb -> Right . SomeProvider <$> mkOllama mgr baseUrl (Just (mkApiKey kb)) model callCounter
  246           Left e   -> pure (Left (vaultErrText e))
  247   where
  248     baseUrl = providerBaseUrlFor cfg OllamaProvider
  249 resolveProvider mvh mgr cfg OpenRouterProvider model _ =
  250   -- OpenRouter requires a vault-stored API key; no keyless mode.
  251   case mvh of
  252     Nothing ->
  253       pure (Left "OpenRouter needs an API key \x2014 run /vault setup then /provider add openrouter")
  254     Just vh -> do
  255       eKey <- vhGet vh (vaultKeyName OpenRouterProvider)
  256       case eKey of
  257         Left e -> pure (Left (credErr OpenRouterProvider e))
  258         Right kb ->
  259           pure (Right (SomeProvider (mkOpenRouter mgr (Just (mkApiKey kb)) baseUrl model)))
  260   where
  261     baseUrl = providerBaseUrlFor cfg OpenRouterProvider
  262 
  263 resolveProvider mvh mgr cfg OllamaCloudProvider model _ =
  264   -- Remote HTTPS endpoint, key required: no keyless mode (D5, D6). Reuses the
  265   -- shared OpenAI-compatible client. No readiness probe.
  266   resolveOpenAICompatible mvh mgr
  267     (providerLabel OllamaCloudProvider)
  268     (providerBaseUrlFor cfg OllamaCloudProvider)
  269     (vaultKeyName OllamaCloudProvider) model
  270 
  271 resolveProvider mvh mgr cfg OpenAIProvider model _ =
  272   -- Built-in endpoint, key required. A [providers.openai] base_url still wins
  273   -- (D4); an explicitly empty one fails closed in the shared helper (item 21b).
  274   resolveOpenAICompatible mvh mgr
  275     (providerLabel OpenAIProvider)
  276     (providerBaseUrlFor cfg OpenAIProvider)
  277     (vaultKeyName OpenAIProvider) model

  287 -- | Build a provider for an operator-named instance. Built-ins resolve
  288 -- exactly as before; an unknown name is an OpenAI-compatible endpoint whose
  289 -- base URL and credential live under that instance's config/vault names.
  290 resolveProviderByName
  291   :: Maybe VaultHandle -> Manager -> RuntimeConfig -> Text -> ModelId
  292   -> IORef Int -> IO (Either Text SomeProvider)
  293 resolveProviderByName mvh mgr cfg name model counter =
  294   case parseProvider name of
  295     Just kp -> resolveProvider mvh mgr cfg kp model counter
  296     Nothing -> resolveOpenAICompatible mvh mgr name
  297       (fromMaybe "" (providerBaseUrl cfg name))
  298       (vaultKeyNameForName name) model

  300 -- | The shared OpenAI-compatible resolution. No built-in endpoint: the
  301 -- operator names it. Fail closed rather than silently reaching out to
  302 -- api.openai.com on their behalf, which would send a key meant for a
  303 -- self-hosted server to a third party.
  304 resolveOpenAICompatible
  305   :: Maybe VaultHandle -> Manager -> Text -> Text -> Text -> ModelId
  306   -> IO (Either Text SomeProvider)
  307 resolveOpenAICompatible mvh mgr name baseUrl keyName model
  308   | T.null baseUrl =
  309       pure (Left ("the OpenAI-compatible provider \"" <> name
  310         <> "\" needs a base URL — set [providers." <> name
  311         <> "] base_url (the setup wizard asks for it)"))
  312   | otherwise = case mvh of
  313       Nothing ->
  314         pure (Left ("the OpenAI-compatible provider \"" <> name
  315           <> "\" needs an API key — run /vault setup then /provider add "
  316           <> name))
  317       Just vh -> do
  318         eKey <- vhGet vh keyName
  319         case eKey of
  320           Left e -> pure (Left (credErrForName name e))
  321           Right kb ->
  322             -- The same client OpenRouter uses: that protocol is the whole
  323             -- point of the generic option.
  324             pure (Right (SomeProvider
  325               (mkOpenAICompatible mgr (Just (mkApiKey kb)) baseUrl model)))

  327 -- | The effective base URL for a provider: the configured
  328 -- @[providers.&lt;label&gt;] base_url@ when set, else that provider's own
  329 -- built-in default. Each provider reads its OWN config section — Ollama's URL
  330 -- is never used for another provider. Anthropic's endpoint is fixed inside the
  331 -- provider, so its value here is only a placeholder.
  332 providerBaseUrlFor :: RuntimeConfig -> KnownProvider -> Text
  333 providerBaseUrlFor cfg kp =
  334   fromMaybe (builtInBaseUrl kp) (providerBaseUrl cfg (providerLabel kp))
  335   where
  336     builtInBaseUrl OllamaProvider = defaultOllamaBaseUrl
  337     builtInBaseUrl OllamaCloudProvider = "https://ollama.com/v1"
  338     builtInBaseUrl OpenRouterProvider = defaultOpenRouterBaseUrl
  339     builtInBaseUrl AnthropicProvider = "https://api.anthropic.com"
  340     builtInBaseUrl OpenAIProvider = "https://api.openai.com/v1"
  341     -- Deliberately empty: a generic endpoint has no default (see
  342     -- 'resolveProvider'), and callers treat "" as "not configured".
  343     builtInBaseUrl OpenAICompatibleProvider = ""
  344 
  345 -- | The effective base URL for a provider name. Custom names have no
  346 -- built-in default: the empty string means "not configured".
  347 -- | Must the operator SUPPLY a base URL for this type? This is a *validation*
  348 -- question, not a visibility one: Ollama's base URL is optional yet must still
  349 -- be shown, because it is the only GUI path to a remote daemon (D7). Keeping
  350 -- this in the registry means the setup wizard, the Providers page and the
  351 -- resolver all read one definition instead of three hardcoded lists.
  352 providerRequiresBaseUrl :: KnownProvider -> Bool
  353 providerRequiresBaseUrl OpenAICompatibleProvider = True
  354 providerRequiresBaseUrl _ = False

  400 -- | The set of providers that are *configured* (have a usable credential or
  401 -- are usable keyless). A provider appears in @/provider list@ and in the
  402 -- frontend's provider select only when this returns 'True'. The rules mirror
  403 -- the auth labels in 'Seal.Command.Provider.listCmd':
  404 --
  405 --   * __Ollama__ is configured when its base URL needs no key (local daemon)
  406 --     OR an API key / OAuth blob is stored in the vault. A cloud URL
  407 --     (@ollama.com@) without a key is NOT configured.
  408 --   * __Anthropic__ is configured when either an OAuth blob or an API key is
  409 --     stored in the vault.
  410 --
  411 -- A missing vault ('Nothing') means no provider has credentials, so only a
  412 -- keyless local Ollama would survive the filter (and only when its base URL
  413 -- is the default local URL).
  414 configuredProviders :: Maybe VaultHandle -> RuntimeConfig -> IO [KnownProvider]
  415 configuredProviders mvh cfg = filterM isConfigured knownProviders
  416   where
  417     isConfigured kp = case kp of
  418       OllamaProvider -> do
  419         let baseUrl = fromMaybe defaultOllamaBaseUrl (providerBaseUrl cfg "ollama")
  420         if not (ollamaNeedsKey baseUrl)
  421           then pure True
  422           else hasVaultKey mvh (vaultKeyName OllamaProvider)
  423       AnthropicProvider ->
  424         hasVaultKey mvh (vaultKeyName AnthropicProvider)
  425           ||^ hasVaultKey mvh anthropicOAuthKey
  426       OpenRouterProvider ->
  427         hasVaultKey mvh (vaultKeyName OpenRouterProvider)
  428       -- Each new type reads its OWN vault key (D3 injectivity): the local
  429       -- OLLAMA_API_KEY must never make Ollama Cloud look configured.
  430       OllamaCloudProvider ->
  431         hasVaultKey mvh (vaultKeyName OllamaCloudProvider)
  432       OpenAIProvider ->
  433         hasVaultKey mvh (vaultKeyName OpenAIProvider)
  434       OpenAICompatibleProvider -> do
  435         -- BOTH halves are required: a base URL with no credential fails on the
  436         -- first call, and a credential with no base URL has nowhere to go.
  437         keyed <- hasVaultKey mvh (vaultKeyName OpenAICompatibleProvider)
  438         pure (keyed && not (T.null
  439           (fromMaybe "" (providerBaseUrl cfg (providerLabel OpenAICompatibleProvider)))))
  440 
  441 -- | Every custom (operator-named) provider instance in the config, sorted.
  442 -- A custom instance is any @[providers.\<name\>]@ section whose key is not a
  443 -- built-in provider id and passes the instance-name grammar.
  444 customProviderNames :: RuntimeConfig -> [Text]
  445 customProviderNames cfg =
  446   sort
  447     [ name
  448     | name <- Map.keys (rcProviders cfg)
  449     , isNothing (parseProvider name)
  450     , either (const False) (const True) (validateProviderName name)
  451     ]

  453 -- | Custom OpenAI-compatible instances that are configured: they have both a
  454 -- base URL and a stored credential.
  455 configuredCustomProviderNames :: Maybe VaultHandle -> RuntimeConfig -> IO [Text]
  456 configuredCustomProviderNames mvh cfg =
  457   filterM isConfigured (customProviderNames cfg)
  458   where
  459     isConfigured name = do
  460       keyed <- hasVaultKey mvh (vaultKeyNameForName name)
  461       pure (keyed && not (T.null
  462         (fromMaybe "" (providerBaseUrl cfg name))))

  464 -- | Every configured provider, built-in and custom, as instance names.
  465 configuredProviderNames :: Maybe VaultHandle -> RuntimeConfig -> IO [Text]
  466 configuredProviderNames mvh cfg = do
  467   builtins <- map providerLabel <$> configuredProviders mvh cfg
  468   customs <- configuredCustomProviderNames mvh cfg
  469   pure (builtins <> customs)

  471 -- | Does the vault hold a value for @k@? 'Nothing' (no vault) or a vault error
  472 -- (locked / key missing) → 'False'; a present key → 'True'.
  473 hasVaultKey :: Maybe VaultHandle -> Text -> IO Bool
  474 hasVaultKey Nothing _ = pure False
  475 hasVaultKey (Just vh) k = do
  476   r <- vhGet vh k
  477   pure $ case r of
  478     Right _ -> True
  479     Left _  -> False

  481 -- | Short-circuit OR for two 'IO' booleans (so the second is only run when the
  482 -- first is 'False'). Avoids touching OAuth when an API key is already present.
  483 infixr 2 ||^
  484 (||^) :: IO Bool -> IO Bool -> IO Bool
  485 a ||^ b = do
  486   x <- a
  487   if x then pure True else b

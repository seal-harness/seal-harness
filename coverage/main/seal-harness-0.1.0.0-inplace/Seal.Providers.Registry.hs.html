   93 -- | Parse a label (case-insensitive) back to a 'KnownProvider'.
   94 parseProvider :: Text -> Maybe KnownProvider
   95 parseProvider t =
   96   let needle = T.toCaseFold (T.strip t)
   97   in lookup needle [(T.toCaseFold (providerLabel p), p) | p <- knownProviders]

   99 -- | Validate a provider instance name. An instance name is the config table
  100 -- key (@[providers.\<name\>]@), the value sessions and agent defs store, and
  101 -- the stem of the instance's vault key. The grammar is an allow-list applied
  102 -- to the raw text (no normalisation): 1-32 characters of @a-z@, @0-9@ and
  103 -- @-@, no leading/trailing @-@, and no @--@. @all@ and @default@ are
  104 -- reserved because they are literal @\/api\/providers\/...@ routes.
  105 validateProviderName :: Text -> Either Text ()
  106 validateProviderName name
  107   | T.null name = Left "provider name must not be empty"
  108   | T.length name > 32 =
  109       Left "provider name must be at most 32 characters"
  110   | name `elem` ["all", "default"] =
  111       Left "\"all\" and \"default\" are reserved provider names"
  112   | T.any badChar name =
  113       Left "provider names may contain only a-z, 0-9 and \"-\""
  114   | "-" `T.isPrefixOf` name || "-" `T.isSuffixOf` name =
  115       Left "provider name must not start or end with \"-\""
  116   | "--" `T.isInfixOf` name =
  117       Left "provider name must not contain \"--\""
  118   | name `elem` ["anthropic", "ollama", "openrouter"] =
  119       Left (name <> " is a built-in provider type; choose another name")
  120   | otherwise = Right ()
  121   where
  122     badChar c =
  123       not (isAsciiLower c || isDigit c || c == '-')
  124 
  125 -- | Is @name@ either a built-in provider id or a syntactically valid custom
  126 -- instance name?
  127 isProviderName :: Text -> Bool
  128 isProviderName name =
  129   isJust (parseProvider name)
  130     || either (const False) (const True) (validateProviderName name)

  145 -- | The vault key for a provider name. Built-ins keep their legacy key; a
  146 -- custom OpenAI-compatible instance uses a name-derived key so two instances
  147 -- never share one secret.
  148 vaultKeyNameForName :: Text -> Text
  149 vaultKeyNameForName name = case parseProvider name of
  150   Just kp -> vaultKeyName kp
  151   Nothing -> "PROVIDER_APIKEY_" <> shoutySnake name
  152   where
  153     shoutySnake = T.map (\c -> if c == '-' then '_' else toUpper c)

  165 -- | A provider's default model: the configured value if given, else the

  181 -- | Build a live provider. For Anthropic, stored OAuth tokens take precedence
  182 -- over an API key; a present-but-corrupt OAuth blob fails loudly (the user must
  183 -- re-run @\/provider login@) rather than silently falling back. The @callCounter@
  184 -- is shared across turns so Ollama-synthesized tool-call ids stay unique across
  185 -- a whole session (Ollama responses carry no id, so Seal synthesizes
  186 -- @call_<i>@; the counter must outlive any single 'Ollama' value, which is
  187 -- rebuilt each turn — see 'ProviderRuntime.prCallCounter').
  188 resolveProvider
  189   :: Maybe VaultHandle -> Manager -> RuntimeConfig -> KnownProvider -> ModelId
  190   -> IORef Int -> IO (Either Text SomeProvider)
  191 resolveProvider Nothing _ _ AnthropicProvider _ _ =
  192   pure (Left "vault not configured \x2014 run /vault setup")
  193 resolveProvider (Just vh) mgr _cfg AnthropicProvider model _ = do
  194   eOAuth <- vhGet vh "ANTHROPIC_OAUTH_TOKENS"
  195   case eOAuth of
  196     Right blob -> case deserializeTokens blob of
  197       Left e     -> pure (Left ("stored Anthropic OAuth tokens are unreadable ("
  198                                   <> e <> ") — run /provider login anthropic"))
  199       Right toks -> do
  200         ref <- newIORef toks
  201         let sess = OAuthSession
  202               { osTokens  = ref
  203               , osRefresh = refreshTokens mgr
  204               , osPersist = void . vhPut vh "ANTHROPIC_OAUTH_TOKENS" . serializeTokens
  205               }
  206         _ <- ensureFresh sess
  207         pure (Right (SomeProvider (mkAnthropicOAuth mgr sess model)))
  208     Left _ -> do
  209       eKey <- vhGet vh (vaultKeyName AnthropicProvider)
  210       pure $ case eKey of
  211         Left e         -> Left (credErr AnthropicProvider e)
  212         Right keyBytes -> Right (SomeProvider (mkAnthropic mgr (mkApiKey keyBytes) model))
  213 resolveProvider mvh mgr cfg OllamaProvider model callCounter
  214   | not (ollamaNeedsKey baseUrl) =
  215       -- local/custom host: keyless, never touch the vault
  216       Right . SomeProvider <$> mkOllama mgr baseUrl Nothing model callCounter
  217   | otherwise = case mvh of
  218       Nothing ->
  219         pure (Left "Ollama Cloud needs an API key \x2014 run /vault setup then /provider add ollama")
  220       Just vh -> do
  221         eKey <- vhGet vh (vaultKeyName OllamaProvider)
  222         case eKey of
  223           Right kb -> Right . SomeProvider <$> mkOllama mgr baseUrl (Just (mkApiKey kb)) model callCounter
  224           Left e   -> pure (Left (vaultErrText e))
  225   where
  226     baseUrl = providerBaseUrlFor cfg OllamaProvider
  227 resolveProvider mvh mgr cfg OpenRouterProvider model _ =
  228   -- OpenRouter requires a vault-stored API key; no keyless mode.
  229   case mvh of
  230     Nothing ->
  231       pure (Left "OpenRouter needs an API key \x2014 run /vault setup then /provider add openrouter")
  232     Just vh -> do
  233       eKey <- vhGet vh (vaultKeyName OpenRouterProvider)
  234       case eKey of
  235         Left e -> pure (Left (credErr OpenRouterProvider e))
  236         Right kb ->
  237           pure (Right (SomeProvider (mkOpenRouter mgr (Just (mkApiKey kb)) baseUrl model)))
  238   where
  239     baseUrl = providerBaseUrlFor cfg OpenRouterProvider
  240 
  241 resolveProvider mvh mgr cfg OpenAICompatibleProvider model _ =
  242   resolveOpenAICompatible mvh mgr "openai"
  243     (providerBaseUrlFor cfg OpenAICompatibleProvider)
  244     (vaultKeyName OpenAICompatibleProvider) model

  246 -- | Build a provider for an operator-named instance. Built-ins resolve
  247 -- exactly as before; an unknown name is an OpenAI-compatible endpoint whose
  248 -- base URL and credential live under that instance's config/vault names.
  249 resolveProviderByName
  250   :: Maybe VaultHandle -> Manager -> RuntimeConfig -> Text -> ModelId
  251   -> IORef Int -> IO (Either Text SomeProvider)
  252 resolveProviderByName mvh mgr cfg name model counter =
  253   case parseProvider name of
  254     Just kp -> resolveProvider mvh mgr cfg kp model counter
  255     Nothing -> resolveOpenAICompatible mvh mgr name
  256       (fromMaybe "" (providerBaseUrl cfg name))
  257       (vaultKeyNameForName name) model

  259 -- | The shared OpenAI-compatible resolution. No built-in endpoint: the
  260 -- operator names it. Fail closed rather than silently reaching out to
  261 -- api.openai.com on their behalf, which would send a key meant for a
  262 -- self-hosted server to a third party.
  263 resolveOpenAICompatible
  264   :: Maybe VaultHandle -> Manager -> Text -> Text -> Text -> ModelId
  265   -> IO (Either Text SomeProvider)
  266 resolveOpenAICompatible mvh mgr name baseUrl keyName model
  267   | T.null baseUrl =
  268       pure (Left ("the OpenAI-compatible provider \"" <> name
  269         <> "\" needs a base URL — set [providers." <> name
  270         <> "] base_url (the setup wizard asks for it)"))
  271   | otherwise = case mvh of
  272       Nothing ->
  273         pure (Left ("the OpenAI-compatible provider \"" <> name
  274           <> "\" needs an API key — run /vault setup then /provider add "
  275           <> name))
  276       Just vh -> do
  277         eKey <- vhGet vh keyName
  278         case eKey of
  279           Left e -> pure (Left (credErrForName name e))
  280           Right kb ->
  281             -- The same client OpenRouter uses: that protocol is the whole
  282             -- point of the generic option.
  283             pure (Right (SomeProvider
  284               (mkOpenAICompatible mgr (Just (mkApiKey kb)) baseUrl model)))

  286 -- | The effective base URL for a provider: the configured
  287 -- @[providers.&lt;label&gt;] base_url@ when set, else that provider's own
  288 -- built-in default. Each provider reads its OWN config section — Ollama's URL
  289 -- is never used for another provider. Anthropic's endpoint is fixed inside the
  290 -- provider, so its value here is only a placeholder.
  291 providerBaseUrlFor :: RuntimeConfig -> KnownProvider -> Text
  292 providerBaseUrlFor cfg kp =
  293   fromMaybe (builtInBaseUrl kp) (providerBaseUrl cfg (providerLabel kp))
  294   where
  295     builtInBaseUrl OllamaProvider     = defaultOllamaBaseUrl
  296     builtInBaseUrl OpenRouterProvider = defaultOpenRouterBaseUrl
  297     builtInBaseUrl AnthropicProvider  = "https://api.anthropic.com"
  298     -- Deliberately empty: a generic endpoint has no default (see
  299     -- 'resolveProvider'), and callers treat "" as "not configured".
  300     builtInBaseUrl OpenAICompatibleProvider = ""
  301 
  302 -- | The effective base URL for a provider name. Custom names have no
  303 -- built-in default: the empty string means "not configured".
  304 providerBaseUrlForName :: RuntimeConfig -> Text -> Text
  305 providerBaseUrlForName cfg name
  306   | Just kp <- parseProvider name = providerBaseUrlFor cfg kp
  307   | otherwise = fromMaybe "" (providerBaseUrl cfg name)

  334 vaultErrText :: VaultError -> Text

  336   VaultLocked         -> "vault is locked — run /vault unlock"
  337   VaultNotFound       -> "vault not found — run /vault setup"

  342 -- | The set of providers that are *configured* (have a usable credential or
  343 -- are usable keyless). A provider appears in @/provider list@ and in the
  344 -- frontend's provider select only when this returns 'True'. The rules mirror
  345 -- the auth labels in 'Seal.Command.Provider.listCmd':
  346 --
  347 --   * __Ollama__ is configured when its base URL needs no key (local daemon)
  348 --     OR an API key / OAuth blob is stored in the vault. A cloud URL
  349 --     (@ollama.com@) without a key is NOT configured.
  350 --   * __Anthropic__ is configured when either an OAuth blob or an API key is
  351 --     stored in the vault.
  352 --
  353 -- A missing vault ('Nothing') means no provider has credentials, so only a
  354 -- keyless local Ollama would survive the filter (and only when its base URL
  355 -- is the default local URL).
  356 configuredProviders :: Maybe VaultHandle -> RuntimeConfig -> IO [KnownProvider]
  357 configuredProviders mvh cfg = filterM isConfigured knownProviders
  358   where
  359     isConfigured kp = case kp of
  360       OllamaProvider -> do
  361         let baseUrl = fromMaybe defaultOllamaBaseUrl (providerBaseUrl cfg "ollama")
  362         if not (ollamaNeedsKey baseUrl)
  363           then pure True
  364           else hasVaultKey mvh (vaultKeyName OllamaProvider)
  365       AnthropicProvider ->
  366         hasVaultKey mvh (vaultKeyName AnthropicProvider)
  367           ||^ hasVaultKey mvh anthropicOAuthKey
  368       OpenRouterProvider ->
  369         hasVaultKey mvh (vaultKeyName OpenRouterProvider)
  370       OpenAICompatibleProvider -> do
  371         -- BOTH halves are required: a base URL with no credential fails on the
  372         -- first call, and a credential with no base URL has nowhere to go.
  373         keyed <- hasVaultKey mvh (vaultKeyName OpenAICompatibleProvider)
  374         pure (keyed && not (T.null
  375           (fromMaybe "" (providerBaseUrl cfg (providerLabel OpenAICompatibleProvider)))))
  376 
  377 -- | Every custom (operator-named) provider instance in the config, sorted.
  378 -- A custom instance is any @[providers.\<name\>]@ section whose key is not a
  379 -- built-in provider id and passes the instance-name grammar.
  380 customProviderNames :: RuntimeConfig -> [Text]
  381 customProviderNames cfg =
  382   sort
  383     [ name
  384     | name <- Map.keys (rcProviders cfg)
  385     , isNothing (parseProvider name)
  386     , either (const False) (const True) (validateProviderName name)
  387     ]
  388 
  389 -- | Custom OpenAI-compatible instances that are configured: they have both a
  390 -- base URL and a stored credential.
  391 configuredCustomProviderNames :: Maybe VaultHandle -> RuntimeConfig -> IO [Text]
  392 configuredCustomProviderNames mvh cfg =
  393   filterM isConfigured (customProviderNames cfg)
  394   where
  395     isConfigured name = do
  396       keyed <- hasVaultKey mvh (vaultKeyNameForName name)
  397       pure (keyed && not (T.null
  398         (fromMaybe "" (providerBaseUrl cfg name))))

  400 -- | Every configured provider, built-in and custom, as instance names.
  401 configuredProviderNames :: Maybe VaultHandle -> RuntimeConfig -> IO [Text]
  402 configuredProviderNames mvh cfg = do
  403   builtins <- map providerLabel <$> configuredProviders mvh cfg
  404   customs <- configuredCustomProviderNames mvh cfg
  405   pure (builtins <> customs)

  407 -- | Does the vault hold a value for @k@? 'Nothing' (no vault) or a vault error
  408 -- (locked / key missing) → 'False'; a present key → 'True'.
  409 hasVaultKey :: Maybe VaultHandle -> Text -> IO Bool
  410 hasVaultKey Nothing _ = pure False
  411 hasVaultKey (Just vh) k = do
  412   r <- vhGet vh k
  413   pure $ case r of
  414     Right _ -> True
  415     Left _  -> False

  417 -- | Short-circuit OR for two 'IO' booleans (so the second is only run when the
  418 -- first is 'False'). Avoids touching OAuth when an API key is already present.
  419 infixr 2 ||^
  420 (||^) :: IO Bool -> IO Bool -> IO Bool
  421 a ||^ b = do
  422   x <- a
  423   if x then pure True else b

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The @/tab new harness@ (or @/tab resume@) attach-wizard state machine:
    3 -- snapshot the running harnesses + recent sessions, number them @[1-9a-z]@,
    4 -- @0@ cancels, a @\/@-prefixed reply cancels and runs that command instead.
    5 module Seal.Tabs.Wizard
    6   ( AttachTarget (..)
    7   , WizardState (..)
    8   , WizardReply (..)
    9   , buildWizard
   10   , handleReply
   11   ) where
   12 
   13 import Data.Char (isDigit)
   14 import Data.Text (Text)
   15 import Data.Text qualified as T
   16 
   17 import Seal.Handles.Tab (TabKind, TabIndex, mkTabIndex, tabIndexToInt)
   18 import Seal.Tabs.Types (TabRef)
   19 
   20 -- | One attachable target (a running harness or a recent session).
   21 data AttachTarget = AttachTarget
   22   { atLabel :: Text
   23   , atRef   :: TabRef
   24   } deriving stock (Eq, Show)
   25 
   26 -- | The wizard state: a numbered list of targets + the pending tab kind.
   27 -- Targets are numbered 1..n (slot 0 is reserved for "cancel").
   28 data WizardState = WizardState
   29   { wsTargets :: [(TabIndex, AttachTarget)]
   30   , wsKind    :: TabKind
   31   } deriving stock (Eq, Show)
   32 
   33 -- | The wizard reply: attach to a ref, cancel, or run a /-prefixed command
   34 -- instead.
   35 data WizardReply
   36   = WizardAttach TabRef
   37   | WizardCancel
   38   | WizardSlash Text   -- ^ a /-prefixed reply: cancel + run this command
   39   deriving stock (Eq, Show)
   40 
   41 -- | Build the wizard state from the running harnesses + recent sessions.
   42 -- Numbers the targets 1..n (slot 0 is reserved for "cancel").
   43 buildWizard :: TabKind -> [AttachTarget] -> WizardState
   44 buildWizard kind targets =
   45   WizardState
   46     { wsTargets = zip (map mkIdx [1..]) targets
   47     , wsKind    = kind
   48     }
   49   where
   50     mkIdx n = case mkTabIndex n of
   51       Right i -> i
   52       Left _  -> error ("buildWizard: target index out of range (n=" <> show n <> ")")
   53 
   54 -- | Handle one reply. 'Right WizardReply' on success; 'Left Text' on a
   55 -- malformed reply (out-of-range number, non-numeric, empty).
   56 handleReply :: WizardState -> Text -> Either Text WizardReply
   57 handleReply ws reply
   58   | T.null reply             = Left "empty reply"
   59   | T.head reply == '/'      = Right (WizardSlash (T.drop 1 reply))
   60   | T.all isDigit reply &&
   61     T.length reply <= 2      =
   62       case mkTabIndex (read (T.unpack reply)) of
   63         Right idx
   64           | tabIndexToInt idx == 0 -> Right WizardCancel
   65           | otherwise -> case lookup idx (wsTargets ws) of
   66               Just t  -> Right (WizardAttach (atRef t))
   67               Nothing -> Left "no such target"
   68         Left _ -> Left "no such target"
   69   | otherwise                = Left "not a number (or '/command')"

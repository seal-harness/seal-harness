    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The on-disk session metadata record ('session.json'): the session's
    3 -- selected provider label + model id (never a key), its channel of origin, the
    4 -- bound agent definition (if any — set at init from @default_agent@), its
    5 -- cloned repo, and timestamps. The 'FromJSON' is tolerant (missing 'channel'
    6 -- defaults to @\"cli\"@, missing 'agent' defaults to 'Nothing') so
    7 -- older/partial files still load.
    8 --
    9 -- Also home to the single reader for that file ('readSessionMeta') — a leaf
   10 -- module, so every layer (the turn engine, the gateway, the delegation
   11 -- worker, the opcodes) and the tests can load a session without import
   12 -- cycles.
   13 module Seal.Session.Meta
   14   ( SessionMeta (..)
   15   , readSessionMeta
   16   ) where
   17 
   18 import Data.Aeson
   19   ( FromJSON (..), ToJSON (..), object, withObject, (.:), (.:?), (.!=), (.=) )
   20 import Data.Text (Text)
   21 import Data.Time (UTCTime)
   22 import System.Directory (doesFileExist)
   23 
   24 import Seal.Agent.Def.Types (AgentDefId (..))
   25 import Seal.Config.Paths (SealPaths, sessionMetaPath)
   26 import Seal.Core.Types (SessionId)
   27 import Seal.Util.StrictIO (decodeFileStrict)
   28 
   29 -- | Load a session's 'SessionMeta' from
   30 -- @\<state\>\/sessions\/\<sid\>\/session.json@. Returns 'Nothing' when the
   31 -- file is missing or undecodable. The existence check is required:
   32 -- 'decodeFileStrict' reads with @sourceFile@, which throws on a missing path.
   33 readSessionMeta :: SealPaths -> SessionId -> IO (Maybe SessionMeta)
   34 readSessionMeta paths sid = do
   35   let mp = sessionMetaPath paths sid
   36   exists <- doesFileExist mp
   37   if exists then decodeFileStrict mp else pure Nothing
   38 
   39 data SessionMeta = SessionMeta
   40   { smId         :: SessionId
   41   , smProvider   :: Text      -- ^ Provider label, e.g. @\"anthropic\"@.
   42   , smModel      :: Text      -- ^ Model id, e.g. @\"claude-opus-4-8\"@.
   43   , smChannel    :: Text      -- ^ Channel that created the session, e.g. @\"cli\"@.
   44   , smParent     :: !(Maybe SessionId)
   45     -- ^ The session that spawned this session ('Nothing' for top-level
   46     -- sessions). Written by the delegation worker when a child transcript
   47     -- is created. Audit metadata only — the lineage guard uses the
   48     -- baked chain threaded through the spawn path, not this field.
   49   , smAgent      :: Maybe AgentDefId
   50     -- ^ The agent definition bound to this session at init (from
   51     -- @default_agent@ in @config.toml@). 'Nothing' means no agent is bound.
   52   , smSystemOverride :: Maybe Text
   53     -- ^ An ad-hoc system prompt supplied via the Session setup screen's
   54     -- \"Use a one-off agent file\" upload. When 'Just', 'plainTurn' prefers
   55     -- this over the bound agent's 'adSystem'. 'Nothing' means no override
   56     -- (the agent's prompt, if any, is used).
   57   , smAgentName :: Maybe Text
   58     -- ^ Display label for the session's active agent (shown in the
   59     -- sidebar / chat header as the @agent@ field of 'SessionInfo').
   60     -- Populated whenever the session has an effective agent — either the
   61     -- bound 'smAgent' (set to the agent def's id) or a one-off uploaded
   62     -- file (set to the file's frontmatter @id@, or the filename when the
   63     -- file has no frontmatter). 'Nothing' when no agent is active.
   64   , smDescription :: Maybe Text
   65     -- ^ A user-set display title for the session (the chat-header pencil).
   66     -- 'Nothing' means no explicit title; the UI then falls back to the
   67     -- auto-summary, first-message snippet, or agent name. Persisted to
   68     -- @session.json@ so it survives restarts and propagates to the
   69     -- sidebar (which reads the live session list, not a client-side
   70     -- override).
   71   , smSealFrontMatter :: Bool
   72     -- ^ Whether the harness injects its own front matter into the session's
   73     -- system prompt (the static behavioral guidance blocks, the auto-loaded
   74     -- @seal-usage@ skill, the codegraph skill, and the available-skills /
   75     -- available-agents catalogs). 'True' (the default) injects it;
   76     -- 'False' means the system prompt is the selected agent's prompt and
   77     -- nothing else. Set by the session-setup form's "Add Seal Harness
   78     -- front matter" toggle.
   79   , smRepoUrl :: Maybe Text
   80     -- ^ The URL of the repository cloned into the session's workdir via
   81     -- @SETUP_REPO@ (either agent-invoked or the web setup-repo endpoint).
   82     -- 'Nothing' when no repo has been cloned. Persisted to @session.json@
   83     -- so the sidebar can show the repo name without scanning the workdir.
   84   , smRepoBranch :: Maybe Text
   85     -- ^ The branch or tag requested for 'smRepoUrl' through @SETUP_REPO@'s
   86     -- optional @branch@ argument. 'Nothing' means the remote's default
   87     -- branch (the *request*, not a resolved ref -- no git query is made).
   88     -- Recorded so a later clone can repeat the same checkout: an
   89     -- @AGENT_MANAGE action=handoff@ session inherits both the repo and
   90     -- this branch from the session that handed off.
   91   , smCreatedAt  :: UTCTime
   92   , smLastActive :: UTCTime
   93   } deriving stock (Eq, Show)
   94 
   95 instance ToJSON SessionMeta where
   96   toJSON m = object
   97     [ "id"          .= smId m
   98     , "provider"    .= smProvider m
   99     , "model"       .= smModel m
  100     , "channel"     .= smChannel m
  101     , "parent"      .= smParent m
  102     , "agent"       .= smAgent m
  103     , "system_override" .= smSystemOverride m
  104     , "agent_name"  .= smAgentName m
  105     , "description" .= smDescription m
  106     , "seal_front_matter" .= smSealFrontMatter m
  107     , "repo_url"    .= smRepoUrl m
  108     , "repo_branch" .= smRepoBranch m
  109     , "created_at"  .= smCreatedAt m
  110     , "last_active" .= smLastActive m
  111     ]
  112 
  113 instance FromJSON SessionMeta where
  114   parseJSON = withObject "SessionMeta" $ \o -> SessionMeta
  115     <$> o .:  "id"
  116     <*> o .:  "provider"
  117     <*> o .:  "model"
  118     <*> o .:? "channel" .!= "cli"
  119     <*> o .:? "parent"
  120     <*> o .:? "agent"
  121     <*> o .:? "system_override"
  122     <*> o .:? "agent_name"
  123     <*> o .:? "description"
  124     <*> o .:? "seal_front_matter" .!= True
  125     <*> o .:? "repo_url"
  126     <*> o .:? "repo_branch"
  127     <*> o .:  "created_at"
  128     <*> o .:  "last_active"

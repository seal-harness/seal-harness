   16 -- | Proof that a (program, args) pair passed policy. Constructor unexported:
   17 -- the only way to obtain one is 'authorize' or 'authorizeShell', so an
   18 -- executor that demands an 'AuthorizedCommand' cannot be handed an unchecked
   19 -- command.
   20 newtype AuthorizedCommand = AuthorizedCommand (FilePath, [Text])
   21   deriving stock (Eq, Show)
   22 
   23 authorizedProgram :: AuthorizedCommand -> (FilePath, [Text])
   24 authorizedProgram (AuthorizedCommand p) = p
   25 
   26 data CommandError
   27   = CommandNotAllowed Text
   28   | CommandInAutonomyDeny
   29   deriving stock (Eq, Show)
   30 
   31 authorize :: SecurityPolicy -> FilePath -> [Text] -> Either CommandError AuthorizedCommand
   32 authorize policy program args
   33   | spAutonomy policy == Deny             = Left CommandInAutonomyDeny
   34   | isCommandAllowed policy (CommandName base) = Right (AuthorizedCommand (program, args))
   35   | otherwise                             = Left (CommandNotAllowed base)
   36   where
   37     base = T.pack (takeFileName program)
   38 
   39 authorizeShell :: SecurityPolicy -> Text -> Either CommandError AuthorizedCommand
   40 authorizeShell policy command
   41   | spAutonomy policy == Deny                       = Left CommandInAutonomyDeny
   42   | isCommandAllowed policy (CommandName "shell")   =
   43       Right (AuthorizedCommand ("/bin/sh", ["-c", command]))
   44   | otherwise                                       = Left (CommandNotAllowed "shell")

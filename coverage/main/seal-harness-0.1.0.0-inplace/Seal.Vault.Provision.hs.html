    1 {-# LANGUAGE OverloadedStrings #-}
    2 {-# OPTIONS_GHC -Wno-partial-fields #-}
    3 -- | The vault provisioning core: one place that turns a backend choice into a
    4 -- recorded key, a fresh vault file, and a live 'VaultHandle'.
    5 --
    6 -- == What this module is
    7 --
    8 -- The CLI's @\/vault setup@ and the web setup wizard's @POST \/api\/setup\/vault@
    9 -- both end up here. It performs the eight ordered steps of design §7.4:
   10 --
   11 -- 1. pre-check (config parses, no key recorded, no vault file) — no lock held;
   12 -- 2. take the @O_CREAT|O_EXCL@ setup lock next to @security.toml@ (0600, stale
   13 --    after 30 s);
   14 -- 3. obtain the key from the injected 'SetupKeygen' seam;
   15 -- 4. build the encryptor and round-trip a nonce (under 'timeout': a YubiKey
   16 --    can sit waiting for a touch or a PIN);
   17 -- 5. record the three vault fields through the narrow CAS writer;
   18 -- 6. @vhInit@ the vault at the configured path;
   19 -- 7. @vhList@ it, so success is never reported for a file this key cannot read;
   20 -- 8. publish the handle in the runtime's 'IORef'.
   21 --
   22 -- Every partial state the ordering can produce is benign and resumable
   23 -- (@resume@ is the one-click recovery for "key recorded, no vault file"),
   24 -- which is why @VaultSpecResume@ is carved out of both the CAS write and the
   25 -- "key already recorded" refusal.
   26 --
   27 -- == What this module is not
   28 --
   29 -- It does not authenticate the caller and it does not read key material back
   30 -- out. @~\/.seal\/@ is writable by the operator's own processes, so a same-UID
   31 -- local process that could call this could already edit @security.toml@
   32 -- directly; this endpoint adds no capability for that class. Cross-UID and
   33 -- LAN peers are covered by the gateway's passkey middleware, not here. See
   34 -- design §7.3 and the PR body's residual-threat paragraph.
   35 --
   36 -- == Leakage rule
   37 --
   38 -- 'renderProvisionError' is the only rendering of a 'ProvisionError' that may
   39 -- reach a response body, and it never echoes the constructor's detail text:
   40 -- the detail can contain an identity path, a vault path, or @age@'s stderr.
   41 -- The one permitted path-bearing field is the 'ProvisionNeedsTty' hint, which
   42 -- is a runnable command built by 'defaultProvisionDeps' from the tilde form of
   43 -- the keys directory (design §7.2 permits an absolute form for mutation errors
   44 -- when @SEAL_HOME@ is non-default).

   98 -- | What the caller asked for. Mirrors the wire shapes in design §9.2.
   99 data VaultSpec
  100   = VaultSpecLocal
  101   | VaultSpecYubiKey { vsTouch :: Bool, vsPin :: Bool }
  102   | VaultSpecUser { vsRecipient :: AgeRecipient, vsIdentity :: VaultIdentityPath }
  103   | VaultSpecResume
  104   deriving stock (Eq, Show)
  105 
  106 -- | Everything that can go wrong, as a value. The handler maps these to HTTP
  107 -- statuses (design §9.2); nothing here carries a response-ready message — that
  108 -- is 'renderProvisionError'\'s job.
  109 data ProvisionError
  110   = ProvisionAlreadyExists
  111   | ProvisionInProgress
  112   | ProvisionKeygenMissing Text
  113     -- ^ @age-keygen@ \/ @age-plugin-yubikey@ is not on PATH.
  114   | ProvisionNeedsTty Text
  115     -- ^ The plugin needs a terminal; carries the runnable command.
  116   | ProvisionInvalidInput Text
  117   | ProvisionRoundTripFailed Text
  118     -- ^ Detail is for the log. The rendered reason is fixed.
  119   | ProvisionKeygenFailed Text
  120   | ProvisionVaultInitFailed
  121   | ProvisionVaultUnreadable
  122   | ProvisionConfigWriteFailed Text
  123   deriving stock (Eq, Show)
  124 
  125 -- | What the caller gets to see: the public recipient and the display label.
  126 -- Never the identity path, never the vault path, never any key material.
  127 data ProvisionOutcome = ProvisionOutcome
  128   { poRecipient :: Text
  129   , poKeyType   :: Text
  130   } deriving stock (Eq, Show)
  131 
  132 -- | How a key is obtained. Injected so the core is testable without
  133 -- @age-keygen@ or a YubiKey on the machine.
  134 data SetupKeygen = SetupKeygen
  135   { skLocal :: IO (Either ProvisionError ResolvedKey)
  136   , skYubi  :: Text -> Bool -> Bool -> IO (Either ProvisionError ResolvedKey)
  137     -- ^ name, touch, pin. The name is always the server-side literal
  138     -- @\"default\"@ (never request-derived).
  139   , skUser  :: AgeRecipient -> VaultIdentityPath -> Either Text ResolvedKey
  140   }
  141 
  142 -- | How the encryptor is built and verified.
  143 data SetupEncryptor = SetupEncryptor
  144   { seMk     :: AgeRecipient -> AgeIdentity -> IO (Either VaultError VaultEncryptor)
  145   , seVerify :: VaultEncryptor -> IO (Either Text ())
  146     -- ^ Encrypt/decrypt a nonce. This is the only real check of a recipient's
  147     -- bech32 checksum (a wrong-but-well-formed key fails here).
  148   }
  149 
  150 data ProvisionDeps = ProvisionDeps
  151   { pdKeygen      :: SetupKeygen
  152   , pdEncryptor   :: SetupEncryptor
  153   , pdWriteConfig :: (SecurityConfig -> Either ProvisionError VaultKeyFields)
  154                   -> IO (Either Text (Either ProvisionError VaultKeyFields))
  155     -- ^ The narrow CAS writer. Its callback sees the current config inside the
  156     -- write lock, so "no key appeared between the pre-check and now" is a real
  157     -- compare-and-swap, not a read-then-write race.
  158   , pdLockPath    :: FilePath
  159   }

  165 -- | Parse a setup request body. Pure shape checks happen here; the two
  166 -- user-supplied credentials go through their smart constructors, so nothing
  167 -- unvalidated can reach a 'VaultSpec'.
  168 parseVaultSpec :: Value -> IO (Either ProvisionError VaultSpec)
  169 parseVaultSpec v =
  170   case parseMaybe parseBackend v of
  171     Nothing -> pure (Left (ProvisionInvalidInput "request body must name a backend"))
  172     Just PBBackendLocal  -> pure (Right VaultSpecLocal)
  173     Just PBBackendResume -> pure (Right VaultSpecResume)
  174     Just (PBBackendYubi touch pin) -> pure (Right (VaultSpecYubiKey touch pin))
  175     Just (PBBackendUser rawRecipient rawIdentity) ->
  176       case mkAgeRecipient rawRecipient of
  177         Left err -> pure (Left (ProvisionInvalidInput ("recipient: " <> err)))
  178         Right recipient -> do
  179           identity <- mkVaultIdentityPath (T.unpack rawIdentity)
  180           pure $ case identity of
  181             Left err -> Left (ProvisionInvalidInput ("identity_path: " <> err))
  182             Right path -> Right (VaultSpecUser recipient path)
  183 
  184 -- | The pure half of 'parseVaultSpec', kept separate so the 'IO' smart
  185 -- constructors do not have to live inside a 'Parser'.
  186 data ParsedBackend
  187   = PBBackendLocal
  188   | PBBackendResume
  189   | PBBackendYubi Bool Bool
  190   | PBBackendUser Text Text
  191 
  192 parseBackend :: Value -> Parser ParsedBackend
  193 parseBackend = withObject "vault setup request" $ \o -> do
  194   backend <- o .: "backend"
  195   case T.toLower backend of
  196     "local"   -> pure PBBackendLocal
  197     "resume"  -> pure PBBackendResume
  198     -- The CLI's default for both policies is "on", so an omitted field means
  199     -- "true" — the safe reading (a missing touch requirement would otherwise
  200     -- silently downgrade the token).
  201     "yubikey" -> PBBackendYubi
  202       <$> o .:? "touch" .!= True
  203       <*> o .:? "pin" .!= True
  204     "user"    -> PBBackendUser <$> o .: "recipient" <*> o .: "identity_path"
  205     other     -> fail ("unknown backend: " <> T.unpack other)

  207 -- ---------------------------------------------------------------------------
  208 -- Provisioning
  209 -- ---------------------------------------------------------------------------

  211 -- | Run the eight ordered steps of design §7.4.
  212 provisionVault
  213   :: SealPaths
  214   -> IORef (Maybe VaultHandle)
  215   -> ProvisionDeps
  216   -> VaultSpec
  217   -> IO (Either ProvisionError ProvisionOutcome)
  218 provisionVault paths handleRef deps spec = do
  219   loaded <- loadSecurityConfig (securityFilePath paths)
  220   case loaded of
  221     Left err -> pure (Left (ProvisionConfigWriteFailed err))
  222     Right cfg -> do
  223       checked <- precheck paths spec cfg
  224       case checked of
  225         Left err -> pure (Left err)
  226         Right () -> withSetupLock (pdLockPath deps) $ do
  227           keyResult <- obtainKey deps cfg spec
  228           case keyResult of
  229             Left err -> pure (Left err)
  230             Right key -> buildVerifyWriteInit key
  231   where
  232     buildVerifyWriteInit key = do
  233       built <- seMk (pdEncryptor deps)
  234         (AgeRecipient (rkRecipient key))
  235         (AgeIdentity (rkIdentity key))
  236       case built of
  237         Left verr -> pure (Left (ProvisionKeygenFailed (vaultErrorText verr)))
  238         Right encryptor -> do
  239           verified <- roundTrip (pdEncryptor deps) encryptor
  240           case verified of
  241             Left err -> pure (Left err)
  242             Right () -> do
  243               written <- recordKey deps spec key
  244               case written of
  245                 Left err -> pure (Left err)
  246                 Right () -> initAndPublish paths handleRef key encryptor
  247 
  248 -- | Steps 5–8, plus the @resume@ carve-out: @resume@ writes no config (the key
  249 -- is already there) but still creates and verifies the vault file.
  250 recordKey
  251   :: ProvisionDeps -> VaultSpec -> ResolvedKey
  252   -> IO (Either ProvisionError ())
  253 recordKey deps spec key = case spec of
  254   VaultSpecResume -> pure (Right ())
  255   _ -> do
  256     result <- pdWriteConfig deps $ \current ->
  257       -- The CAS: a recipient that appeared since the pre-check means another
  258       -- writer won; refuse rather than overwrite a live vault's key.
  259       case scVaultRecipient current of
  260         Just _  -> Left ProvisionAlreadyExists
  261         Nothing -> Right VaultKeyFields
  262           { vkfRecipient = rkRecipient key
  263           , vkfIdentity  = T.unpack (rkIdentity key)
  264           , vkfKeyType   = rkKeyType key
  265           }
  266     pure $ case result of
  267       Left err -> Left (ProvisionConfigWriteFailed err)
  268       Right (Left rejection) -> Left rejection
  269       Right (Right _) -> Right ()
  270 
  271 initAndPublish
  272   :: SealPaths
  273   -> IORef (Maybe VaultHandle)
  274   -> ResolvedKey
  275   -> VaultEncryptor
  276   -> IO (Either ProvisionError ProvisionOutcome)
  277 initAndPublish paths handleRef key encryptor = do
  278   loaded <- loadSecurityConfig (securityFilePath paths)
  279   let configuredPath = either (const Nothing) scVaultPath loaded
  280       vaultPath = maybe (vaultFilePath paths) T.unpack configuredPath
  281       vaultCfg = VaultConfig
  282         { vcPath    = vaultPath
  283         , vcKeyType = rkKeyType key
  284         -- 'UnlockOnDemand' (the CLI's choice): a freshly created vault has an
  285         -- empty cache, and 'vhList' below must reach the disk rather than
  286         -- report 'VaultLocked' for a vault that is perfectly readable.
  287         , vcUnlock  = UnlockOnDemand
  288         }
  289   h <- openVault vaultCfg encryptor
  290   initialized <- vhInit h
  291   case initialized of
  292     Left _ -> pure (Left ProvisionVaultInitFailed)
  293     Right () -> do
  294       -- Step 7: never report success for a file this key cannot read. The file
  295       -- is left in place (it may hold data a different key can open).
  296       readable <- vhList h
  297       case readable of
  298         Left _ -> pure (Left ProvisionVaultUnreadable)
  299         Right _ -> do
  300           writeIORef handleRef (Just h)
  301           pure (Right (ProvisionOutcome (rkRecipient key) (rkKeyType key)))
  302 
  303 -- | Step 1: refuse before taking any lock or writing anything.
  304 precheck :: SealPaths -> VaultSpec -> SecurityConfig -> IO (Either ProvisionError ())
  305 precheck paths spec cfg = case spec of
  306   VaultSpecResume -> case (scVaultRecipient cfg, scVaultIdentity cfg) of
  307     (Just _, Just _) -> noVaultFile
  308     _ -> pure (Left (ProvisionInvalidInput
  309            "resume requires a recorded vault key; there is nothing to resume"))
  310   _ -> case scVaultRecipient cfg of
  311     Just _  -> pure (Left ProvisionAlreadyExists)
  312     Nothing -> noVaultFile
  313   where
  314     noVaultFile = do
  315       exists <- doesFileExist (vaultPathFor paths cfg)
  316       pure (if exists then Left ProvisionAlreadyExists else Right ())
  317 
  318 vaultPathFor :: SealPaths -> SecurityConfig -> FilePath
  319 vaultPathFor paths cfg = maybe (vaultFilePath paths) T.unpack (scVaultPath cfg)
  320 
  321 -- | Step 3. @resume@ takes the recorded key: that is the whole point of the
  322 -- state (V5: key recorded, vault file missing).
  323 obtainKey
  324   :: ProvisionDeps -> SecurityConfig -> VaultSpec
  325   -> IO (Either ProvisionError ResolvedKey)
  326 obtainKey deps cfg spec = case spec of
  327   VaultSpecLocal -> skLocal (pdKeygen deps)
  328   VaultSpecYubiKey touch pin -> skYubi (pdKeygen deps) "default" touch pin
  329   VaultSpecUser recipient identity ->
  330     pure (either (Left . ProvisionInvalidInput) Right
  331             (skUser (pdKeygen deps) recipient identity))
  332   VaultSpecResume -> case (scVaultRecipient cfg, scVaultIdentity cfg) of
  333     (Just recipient, Just identity) ->
  334       pure (Right (ResolvedKey recipient identity
  335                      (fromMaybe "unknown" (scVaultKeyType cfg))))
  336     _ -> pure (Left (ProvisionInvalidInput "no recorded vault key to resume"))
  337 
  338 -- | Step 4, under a timeout: a YubiKey identity can block on a touch or a PIN.
  339 roundTrip :: SetupEncryptor -> VaultEncryptor -> IO (Either ProvisionError ())
  340 roundTrip encryptorDeps encryptor = do
  341   outcome <- timeout roundTripTimeoutMicros (seVerify encryptorDeps encryptor)
  342   pure $ case outcome of
  343     Nothing -> Left (ProvisionRoundTripFailed "round-trip verify timed out")
  344     Just (Left err) -> Left (ProvisionRoundTripFailed err)
  345     Just (Right ()) -> Right ()
  346 
  347 -- | A touch plus a PIN prompt. Deliberately generous: this is a human-in-the-
  348 -- loop path, and a timeout here is a failed setup, not a hung gateway.
  349 roundTripTimeoutMicros :: Int
  350 roundTripTimeoutMicros = 60 * 1000 * 1000
  351 
  352 -- ---------------------------------------------------------------------------
  353 -- Setup lock
  354 -- ---------------------------------------------------------------------------
  355 
  356 -- | Hold the setup lock for the duration of the action, or report that another
  357 -- setup holds it. Released in a bracket, so an exception cannot leave the
  358 -- operator locked out for the 30 s stale window.
  359 withSetupLock
  360   :: FilePath
  361   -> IO (Either ProvisionError a)
  362   -> IO (Either ProvisionError a)
  363 withSetupLock lockPath action = do
  364   acquired <- acquireSetupLock lockPath
  365   if acquired
  366     then action `finally` removeIfExists lockPath
  367     else pure (Left ProvisionInProgress)
  368 
  369 -- | @O_CREAT|O_EXCL@ with mode 0600. A concurrent gateway gets @EEXIST@ from
  370 -- the kernel — there is no check-then-create window. A lock older than
  371 -- 'staleLockSeconds' is assumed to be debris from a crashed process and is
  372 -- reclaimed once.
  373 acquireSetupLock :: FilePath -> IO Bool
  374 acquireSetupLock lockPath = do
  375   createDirectoryIfMissing True (takeDirectory lockPath)
  376   created <- tryCreate
  377   if created
  378     then pure True
  379     else do
  380       stale <- isStale
  381       if stale
  382         then removeIfExists lockPath >> tryCreate
  383         else pure False
  384   where
  385     tryCreate = do
  386       result <- try @IOException
  387         (openFd lockPath WriteOnly defaultFileFlags { exclusive = True, creat = Just 0o600 })
  388       case result of
  389         Left _   -> pure False
  390         Right fd -> closeFd fd >> pure True
  391 
  392     isStale = do
  393       result <- try @IOException (getFileStatus lockPath)
  394       case result of
  395         -- It vanished between the failed create and this stat: treat it as
  396         -- stale so the single retry below gets a fresh chance.
  397         Left _   -> pure True
  398         Right st -> do
  399           now <- getPOSIXTime
  400           pure (now - modificationTimeHiRes st > staleLockSeconds)
  401 
  402 staleLockSeconds :: POSIXTime
  403 staleLockSeconds = 30
  404 
  405 removeIfExists :: FilePath -> IO ()
  406 removeIfExists path = void (try @IOException (removeFile path))
  407 
  408 -- ---------------------------------------------------------------------------
  409 -- Rendering
  410 -- ---------------------------------------------------------------------------
  411 
  412 -- | The only rendering of a 'ProvisionError' allowed into a response body:
  413 -- a fixed reason plus (sometimes) a copy-pasteable next step.
  414 --
  415 -- The detail text of every constructor is deliberately dropped here — it is
  416 -- the caller's job to log it. Details can contain an identity path, a vault
  417 -- path, or @age@'s stderr.
  418 renderProvisionError :: ProvisionError -> (Text, Maybe Text)
  419 renderProvisionError = \case
  420   ProvisionAlreadyExists ->
  421     ( "a vault key is already recorded; nothing was changed"
  422     , Just "use the \"resume\" backend to finish an interrupted setup" )
  423   ProvisionInProgress ->
  424     ( "another vault setup is already running"
  425     , Just "wait a few seconds and retry" )
  426   ProvisionKeygenMissing _ ->
  427     ( "a required key-generation tool is not installed"
  428     , Just "install age-keygen or age-plugin-yubikey, then retry" )
  429   ProvisionNeedsTty command ->
  430     ( "this vault backend needs an interactive terminal"
  431     , Just command )
  432   ProvisionInvalidInput _ ->
  433     ( "the request is not a valid vault setup request"
  434     , Just "send {\"backend\":\"local\"}, \"yubikey\", \"user\" or \"resume\"" )
  435   ProvisionRoundTripFailed _ ->
  436     ( "the new key failed its encrypt/decrypt round-trip; nothing was written"
  437     , Just "check that the identity file belongs to the recipient, then retry" )
  438   ProvisionKeygenFailed _ ->
  439     ( "key generation failed; nothing was written"
  440     , Just "see the gateway log for the tool's output" )
  441   ProvisionVaultInitFailed ->
  442     ( "the vault file could not be created"
  443     , Just "check permissions on the seal config directory" )
  444   ProvisionVaultUnreadable ->
  445     ( "the vault file could not be opened with the new key; it was left in place"
  446     , Just "remove the vault file if it is a leftover from an interrupted setup" )
  447   ProvisionConfigWriteFailed _ ->
  448     ( "the vault key could not be recorded in security.toml"
  449     , Just "check permissions on the seal home directory" )
  450 
  451 vaultErrorText :: VaultError -> Text
  452 vaultErrorText err = case err of
  453   VaultBackendError msg -> msg
  454   VaultLocked -> "vault is locked"
  455   VaultNotFound -> "vault file not found"
  456   VaultAlreadyExists -> "vault already exists"
  457   VaultKeyNotFound key -> "secret not found: " <> key

  463 -- | The real seams: @age-keygen@, the non-interactive @age-plugin-yubikey@
  464 -- core, the narrow CAS writer, and the @age@ round-trip.
  465 defaultProvisionDeps :: SealPaths -> ProvisionDeps
  466 defaultProvisionDeps paths = ProvisionDeps
  467   { pdKeygen = SetupKeygen
  468       { skLocal = do
  469           found <- findExecutable "age-keygen"
  470           case found of
  471             Nothing -> pure (Left (ProvisionKeygenMissing "age-keygen"))
  472             Just _  -> either (Left . ProvisionKeygenFailed) Right
  473                          <$> setupLocalAgeKey paths "default"
  474       , skYubi = \name touch pin ->
  475           mapKeygenFailure <$> setupYubiKeyCore paths name touch pin
  476       , skUser = setupUserSuppliedFrom
  477       }
  478   , pdEncryptor = SetupEncryptor
  479       { seMk     = mkAgeEncryptor
  480       , seVerify = roundTripVerify
  481       }
  482   , pdWriteConfig = updateVaultKeySection (securityFilePath paths)
  483   , pdLockPath    = securityFilePath paths <> ".lock"
  484   }
  485   where
  486     mapKeygenFailure failure = case failure of
  487       Left (KeySetupMissing binary) -> Left (ProvisionKeygenMissing binary)
  488       Left (KeySetupNeedsTty command) -> Left (ProvisionNeedsTty command)
  489       Left (KeySetupFailed detail) -> Left (ProvisionKeygenFailed detail)
  490       Right key -> Right key
  491 
  492 -- | The nonce is fixed on purpose: this checks that the key can encrypt and
  493 -- decrypt *something*, and a fixed value makes a failure reproducible.
  494 roundTripNonce :: ByteString
  495 roundTripNonce = "seal-vault-round-trip"
  496 
  497 roundTripVerify :: VaultEncryptor -> IO (Either Text ())
  498 roundTripVerify encryptor = do
  499   encrypted <- veEncrypt encryptor roundTripNonce
  500   case encrypted of
  501     Left err -> pure (Left ("encrypt failed: " <> vaultErrorText err))
  502     Right ciphertext -> do
  503       decrypted <- veDecrypt encryptor ciphertext
  504       pure $ case decrypted of
  505         Left err -> Left ("decrypt failed: " <> vaultErrorText err)
  506         Right plain
  507           | plain == roundTripNonce -> Right ()
  508           | otherwise -> Left "decrypt returned different bytes"

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The embedding backend capability — a record-of-functions (consistent
    3 -- with the project's 'MemoryBackend' / 'SkillBackend' convention) rather
    4 -- than a typeclass, per CONTRIBUTING.md: "capability-handle records of IO
    5 -- functions over type classes".
    6 --
    7 -- The embedding backend is a top-level Seal Harness concern, not specific
    8 -- to memory. The same backend can be used for memory search, session
    9 -- search, and any future feature that needs semantic retrieval.
   10 --
   11 -- Ships with 'nullEmbeddingBackend' (no search, returns empty results).
   12 -- The engram backend (subprocess to the engram CLI) is a follow-up.
   13 module Seal.Memory.Embedding
   14   ( EmbeddingBackend (..)
   15   , SearchResult (..)
   16   , nullEmbeddingBackend
   17   ) where
   18 
   19 import Data.Text (Text)
   20 
   21 -- | A single search result from the embedding backend.
   22 data SearchResult = SearchResult
   23   { srPath    :: Text
   24     -- ^ The memory path (relative to active/ or archived/).
   25   , srContent :: Text
   26     -- ^ The memory file content.
   27   , srScore   :: Double
   28     -- ^ Relevance score (higher = more relevant).
   29   } deriving stock (Eq, Show)
   30 
   31 -- | The embedding backend capability. Each operation is IO — the engram
   32 -- backend shells out to the engram CLI; the null backend is a no-op.
   33 data EmbeddingBackend = EmbeddingBackend
   34   { ebIndex       :: FilePath -> Text -> IO ()
   35     -- ^ Add a file to the search index.
   36   , ebUnindex     :: FilePath -> IO ()
   37     -- ^ Remove a file from the search index.
   38   , ebSearch      :: Text -> Int -> IO [SearchResult]
   39     -- ^ Semantic search: query + limit, returns ranked results.
   40   , ebBackendName :: Text
   41     -- ^ Human-readable backend name (e.g. "null", "engram").
   42   }
   43 
   44 -- | The null backend: no indexing, no search. Returns empty results.
   45 -- For development without ollama/engram installed.
   46 nullEmbeddingBackend :: EmbeddingBackend
   47 nullEmbeddingBackend = EmbeddingBackend
   48   { ebIndex       = \_ _ -> pure ()
   49   , ebUnindex     = \_   -> pure ()
   50   , ebSearch      = \_ _ -> pure []
   51   , ebBackendName = "null"
   52   }

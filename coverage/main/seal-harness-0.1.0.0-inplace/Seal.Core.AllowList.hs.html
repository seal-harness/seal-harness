   23 -- | Codec for the @allow_from@ array: 'AllowAll' when absent, 'AllowOnly'
   24 -- of the listed strings when present. Shared by all channel config codecs
   25 -- (Signal, Telegram, Channels) to avoid triplication.
   26 allowListCodec :: Toml.TomlCodec (AllowList Text)
   27 allowListCodec = Toml.dimap toList fromList (Toml.dioptional (Toml.arrayOf Toml._Text "allow_from"))
   28   where
   29     toList :: AllowList Text -> Maybe [Text]
   30     toList AllowAll       = Nothing
   31     toList (AllowOnly xs) = Just (Set.toList xs)
   32     fromList :: Maybe [Text] -> AllowList Text
   33     fromList Nothing   = AllowAll
   34     fromList (Just xs) = AllowOnly (Set.fromList xs)

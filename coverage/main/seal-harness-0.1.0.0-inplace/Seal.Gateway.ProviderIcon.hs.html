    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Provider favicon resolution: fetch the root page of an OpenAI-compatible
    3 -- provider's base URL, find the first @<link rel="icon">@ (or
    4 -- @apple-touch-icon@) tag, resolve the @href@ to an absolute URL, and fetch
    5 -- the icon image bytes.
    6 --
    7 -- This is a backend proxy: the frontend's @<img src="/api/providers/:name/icon">@
    8 -- hits this module through the REST API, which fetches the provider's website
    9 -- server-side. This avoids CORS (the browser cannot read cross-origin HTML)
   10 -- and network-reachability issues (the backend can reach private endpoints).
   11 --
   12 -- Results are cached in-memory per root URL with a TTL so repeated renders
   13 -- do not refetch.
   14 module Seal.Gateway.ProviderIcon
   15   ( resolveProviderIconBytes
   16   , IconResult (..)
   17   , IconCache
   18   , newIconCache
   19   , cachedIcon
   20   , storeIcon
   21   , deriveRootUrl
   22   , extractIconHref
   23   , resolveIconUrl
   24   ) where
   25

   41 -- | The result of resolving a provider icon.
   42 data IconResult
   43   = IconFound ByteString Text
   44     -- ^ The icon image bytes and its content-type (e.g. "image/png").
   45   | IconNotFound
   46     -- ^ No icon link found on the root page, or the page could not be fetched.
   47   deriving stock (Show, Eq)
   48 
   49 -- | An in-memory cache of resolved icons, keyed by root URL.
   50 newtype IconCache = IconCache
   51   { icEntries :: IORef (Map.Map Text (IconResult, UTCTime))
   52   }
   53 
   54 -- | Create a new empty icon cache.
   55 newIconCache :: IO IconCache
   56 newIconCache = IconCache <$> newIORef Map.empty
   57 
   58 -- | Cache TTL in seconds. A provider's favicon changes rarely; 10 minutes
   59 -- keeps the badge responsive to operator config changes without refetching
   60 -- on every page load.
   61 iconCacheTtlSeconds :: NominalDiffTime
   62 iconCacheTtlSeconds = 600
   63 
   64 -- | Look up a cached icon entry. Returns 'Nothing' when the cache has no
   65 -- entry or the entry has expired.
   66 cachedIcon :: IconCache -> Text -> IO (Maybe IconResult)
   67 cachedIcon cache rootUrl = do
   68   entries <- readIORef (icEntries cache)
   69   case Map.lookup rootUrl entries of
   70     Nothing -> pure Nothing
   71     Just (result, cachedAt) -> do
   72       now <- getCurrentTime
   73       if diffUTCTime now cachedAt > iconCacheTtlSeconds
   74         then pure Nothing
   75         else pure (Just result)
   76 
   77 -- | Store an icon result in the cache.
   78 storeIcon :: IconCache -> Text -> IconResult -> IO ()
   79 storeIcon cache rootUrl result = do
   80   now <- getCurrentTime
   81   modifyIORef' (icEntries cache) (Map.insert rootUrl (result, now))
   82 
   83 -- | Resolve a provider's favicon: fetch the root page, extract the icon
   84 -- @<link>@ href, resolve it to an absolute URL, fetch the image bytes, and
   85 -- return them with the content-type. Uses the cache to avoid refetching.
   86 resolveProviderIconBytes :: IconCache -> Manager -> Text -> IO IconResult
   87 resolveProviderIconBytes cache mgr baseUrl = do
   88   let rootUrl = deriveRootUrl baseUrl
   89   case rootUrl of
   90     Nothing -> pure IconNotFound
   91     Just root -> do
   92       mCached <- cachedIcon cache root
   93       case mCached of
   94         Just result -> pure result
   95         Nothing -> do
   96           result <- fetchProviderIcon mgr root
   97           storeIcon cache root result
   98           pure result

  100 -- | Derive the root page URL (origin only) from a provider base URL.
  101 -- Strips @/v1@ and any trailing path. Returns 'Nothing' when the input is
  102 -- empty or not a parseable URL.
  103 --
  104 --   "https://api.openai.com/v1"      → "https://api.openai.com"
  105 --   "http://localhost:8000/v1/"      → "http://localhost:8000"
  106 --   "https://api.example.com/api/v2" → "https://api.example.com"
  107 deriveRootUrl :: Text -> Maybe Text
  108 deriveRootUrl baseUrl = do
  109   let trimmed = T.strip baseUrl
  110   if T.null trimmed then Nothing
  111   else case T.breakOn "://" trimmed of
  112     (scheme, rest) | not (T.null rest) ->
  113       let hostPart = T.takeWhile (\c -> c /= '/' && c /= '?') (T.drop 3 rest)
  114       in if T.null hostPart then Nothing
  115          else Just (scheme <> "://" <> hostPart)
  116     _ -> Nothing
  117 
  118 -- | Extract the first icon @<link>@ tag's @href@ value from an HTML string.
  119 -- Searches for @rel="icon"@, @rel="shortcut icon"@, and
  120 -- @rel="apple-touch-icon"@ (case-insensitive), in that preference order.
  121 -- Returns 'Nothing' when no icon link is found.
  122 extractIconHref :: ByteString -> Maybe Text
  123 extractIconHref html =
  124   case findMatchingLinks "icon" of
  125     Just href -> Just href
  126     Nothing -> case findMatchingLinks "shortcut icon" of
  127       Just href -> Just href
  128       Nothing -> findMatchingLinks "apple-touch-icon"
  129   where
  130     -- The full HTML as lowercase Text for case-insensitive scanning.
  131     htmlText = TE.decodeUtf8With TEE.lenientDecode html
  132     htmlLower = T.toLower htmlText
  133     -- Find all <link ...> tags whose rel attribute matches the pattern,
  134     -- and return the href of the first one.
  135     findMatchingLinks :: Text -> Maybe Text
  136     findMatchingLinks relPattern = findLinkTagWithRel htmlText htmlLower relPattern 0
  137 
  138 -- | Scan the HTML text for @<link>@ tags, checking each one's @rel@
  139 -- attribute (case-insensitive via the lowercase copy) against the pattern.
  140 -- Returns the @href@ value of the first match.
  141 findLinkTagWithRel :: Text -> Text -> Text -> Int -> Maybe Text
  142 findLinkTagWithRel orig lower relPattern start =
  143   case T.breakOn "<link" (T.drop start lower) of
  144     (_, "") -> Nothing  -- no more <link tags
  145     (pre, tagStart) ->
  146       let afterLink = T.drop 5 tagStart  -- skip "<link"
  147       in case T.breakOn ">" afterLink of
  148         (tagContent, _) ->
  149           let tagLower = tagContent
  150               tagOrig = T.take (T.length tagContent) (T.drop (T.length pre + 5) orig)
  151           in if containsRel relPattern tagLower
  152                then extractHrefValue tagOrig
  153                else findLinkTagWithRel orig lower relPattern (T.length pre + 5 + T.length tagContent + 1)
  154 
  155 -- | Check whether a tag's content contains @rel="...icon..."@ (already
  156 -- lowercased). We check for @rel@ followed by @=@ and the pattern.
  157 containsRel :: Text -> Text -> Bool
  158 containsRel relPattern tagContent =
  159   case T.breakOn "rel" tagContent of
  160     (_, "") -> False
  161     (_, afterRel) ->
  162       let afterRelWord = T.dropWhile (\c -> c == ' ' || c == '\t' || c == '\n') (T.drop 3 afterRel)
  163       in case T.uncons afterRelWord of
  164            Just ('=', afterEq) ->
  165              let afterEq' = T.dropWhile (\c -> c == ' ' || c == '\t' || c == '\n') afterEq
  166              in case T.uncons afterEq' of
  167                   Just ('"', afterQuote) ->
  168                     relPattern `T.isPrefixOf` T.toLower (T.takeWhile (/= '"') afterQuote)
  169                   Just ('\'', afterQuote) ->
  170                     relPattern `T.isPrefixOf` T.toLower (T.takeWhile (/= '\'') afterQuote)
  171                   _ -> relPattern `T.isPrefixOf` T.dropWhile (\c -> c == ' ' || c == '\t') afterEq'
  172            _ -> False
  173 
  174 -- | Extract the @href@ value from a tag's attribute string.
  175 -- Looks for @href="..."@ or @href='...'@.
  176 extractHrefValue :: Text -> Maybe Text
  177 extractHrefValue tag =
  178   case T.breakOn "href" (T.toLower tag) of
  179     (_, "") -> Nothing
  180     (_, afterHref) ->
  181       let afterHrefWord = T.dropWhile (\c -> c == ' ' || c == '\t' || c == '\n') (T.drop 4 afterHref)
  182       in case T.uncons afterHrefWord of
  183            Just ('=', afterEq) ->
  184              let afterEq' = T.dropWhile (\c -> c == ' ' || c == '\t' || c == '\n') afterEq
  185              in case T.uncons afterEq' of
  186                   Just ('"', afterQuote) -> Just (T.takeWhile (/= '"') afterQuote)
  187                   Just ('\'', afterQuote) -> Just (T.takeWhile (/= '\'') afterQuote)
  188                   _ -> Just (T.takeWhile (\c -> c /= ' ' && c /= '>' && c /= '\t' && c /= '\n') afterEq')
  189            _ -> Nothing

  191 -- | Resolve an icon @href@ (which may be absolute, protocol-relative, or
  192 -- root-relative) to an absolute URL using the provider's root URL as the
  193 -- base. Returns 'Nothing' when the href is empty.
  194 resolveIconUrl :: Text -> Text -> Maybe Text
  195 resolveIconUrl rootUrl href = do
  196   let h = T.strip href
  197   if T.null h then Nothing
  198   else if "//" `T.isPrefixOf` h then
  199     case T.breakOn "://" rootUrl of
  200       (scheme, _) | not (T.null scheme) -> Just (scheme <> ":" <> h)
  201       _ -> Nothing
  202   else if "://" `T.isInfixOf` h then Just h
  203   else if "/" `T.isPrefixOf` h then Just (rootUrl <> h)
  204   else Just (rootUrl <> "/" <> h)

  206 -- | Fetch the provider's root page, extract the icon, fetch the image bytes.
  207 fetchProviderIcon :: Manager -> Text -> IO IconResult
  208 fetchProviderIcon mgr rootUrl = do
  209   eHtml <- tryFetch mgr rootUrl
  210   case eHtml of
  211     Left _ -> pure IconNotFound
  212     Right resp ->
  213       let code = statusCode (responseStatus resp)
  214       in if code < 200 || code > 299
  215            then pure IconNotFound
  216            else case extractIconHref (BL.toStrict (responseBody resp)) of
  217              Nothing -> pure IconNotFound
  218              Just href -> case resolveIconUrl rootUrl href of
  219                Nothing -> pure IconNotFound
  220                Just iconUrl -> do
  221                  eIcon <- tryFetch mgr iconUrl
  222                  case eIcon of
  223                    Left _ -> pure IconNotFound
  224                    Right iconResp ->
  225                      let icCode = statusCode (responseStatus iconResp)
  226                      in if icCode < 200 || icCode > 299
  227                           then pure IconNotFound
  228                           else
  229                             let ct = contentTypeFromResponse iconResp
  230                             in pure (IconFound (BL.toStrict (responseBody iconResp)) ct)
  231 
  232 -- | Fetch a URL and return the response, or Left on any error.
  233 tryFetch :: Manager -> Text -> IO (Either SomeException (Response BL.ByteString))
  234 tryFetch mgr url = try (do
  235   req <- parseRequest (T.unpack url)
  236   let req' = req { responseTimeout = responseTimeoutMicro 5000000 }
  237   httpLbs req' mgr)
  238 
  239 -- | Extract the Content-Type header value from a response, defaulting to
  240 -- "image/png" when absent (most favicons are PNG or ICO).
  241 contentTypeFromResponse :: Response BL.ByteString -> Text
  242 contentTypeFromResponse resp =
  243   case lookup hContentType (responseHeaders resp) of
  244     Just ct -> TE.decodeUtf8With TEE.lenientDecode ct
  245     Nothing -> "image/png"

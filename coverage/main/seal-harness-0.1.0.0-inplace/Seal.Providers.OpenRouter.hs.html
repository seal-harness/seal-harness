    3 -- | OpenRouter provider using the @openai@ package's *types* (v2.5.4) with our
    4 -- own HTTP round-trip (mirroring 'Seal.Providers.Anthropic' / 'Ollama'). We do
    5 -- NOT use 'OpenAI.V1.makeMethods' (the servant-client convenience wrapper) —
    6 -- that would unwrap the 'ApiKey' to 'Text' and surface 'servant-client'
    7 -- runtime exceptions. Instead the request is built as a
    8 -- 'OpenAI.V1.Chat.Completions.CreateChatCompletion', @encode@d to JSON, and
    9 -- POSTed with 'Network.HTTP.Client'; the response is @eitherDecode@d as a
   10 -- 'OpenAI.V1.Chat.Completions.ChatCompletionObject'. The API key bytes live
   11 -- only on the @Authorization: Bearer@ header built inside the CPS 'withApiKey'
   12 -- accessor — never returned, never logged.
   13 --
   14 -- Base URL: @https://openrouter.ai/api/v1/chat/completions@
   15 -- Vault key: @OPENROUTER_API_KEY@
   16 --
   17 -- Security: capability-based — no opcode touches the agent-immutable home
   18 -- state dir; this module imports only the openai types, the HTTP client
   19 -- library, and the CPS secret accessor. It never spawns a process, never
   20 -- touches the filesystem directly, and never imports the web-client library
   21 -- whose types it borrows (see AGENTS.md §Security).

  101 -- Pure request mapping ------------------------------------------------------

  103 -- | Build the OpenAI-compatible chat-completion request from our shape. The
  104 -- system prompt becomes a leading 'Chat.System' message; 'CbThinking' blocks
  105 -- are dropped (no OpenAI equivalent); 'CbToolUse' becomes the assistant's
  106 -- @tool_calls@; 'CbToolResult' becomes a 'Chat.Tool' role message carrying
  107 -- the matching @tool_call_id@.
  108 encodeRequest :: CompletionRequest -> Chat.CreateChatCompletion
  109 encodeRequest cr = Chat._CreateChatCompletion
  110   { Chat.messages = V.fromList (systemMsgs <> concatMap msgOpenAi (crMessages cr))
  111   , Chat.model = OAM.Model (modelIdText (crModel cr))
  112   , Chat.max_completion_tokens = Just (fromIntegral (crMaxTokens cr))
  113   , Chat.tools = toolsVec
  114   , Chat.tool_choice = tc
  115   }
  116   where
  117     systemMsgs = case crSystem cr of
  118       Nothing -> []
  119       Just s  -> [ Chat.System { Chat.content = V.singleton (txt s), Chat.name = Nothing } ]
  120     toolsVec = case crTools cr of
  121       []  -> Nothing
  122       ts  -> Just (V.fromList (map toolDef ts))
  123     tc = case crToolChoice cr of
  124       ToolAuto  -> Just OAT.ToolChoiceAuto
  125       ToolNone  -> Just OAT.ToolChoiceNone
  126 
  127     toolDef (ToolDefinition (OpName n) d sch) =
  128       OAT.Tool_Function
  129         ( OAT.Function
  130             { OAT.description = Just d
  131             , OAT.name = n
  132             , OAT.parameters = Just sch
  133             , OAT.strict = Nothing
  134             }
  135         )

  137 -- | Map a single Seal 'Message' to one or more OpenAI messages. A Seal
  138 -- assistant message that mixes text + tool calls + tool results splits into a
  139 -- single 'Chat.Assistant' message (text + tool_calls) plus one 'Chat.Tool'
  140 -- message per 'CbToolResult'. (OpenAI has no per-tool-result container; each
  141 -- result is its own @tool@-role message keyed by @tool_call_id@.)
  142 msgOpenAi :: Message -> [Chat.Message (V.Vector Chat.Content)]
  143 msgOpenAi (Message role blocks) =
  144   let textContents   = [ txt t     | CbText t <- blocks ]
  145       toolCallVec    = [ mkToolCall i n inp
  146                        | CbToolUse (ToolCallId i) (OpName n) inp <- blocks ]
  147       toolResultMsgs = [ Chat.Tool
  148                           { Chat.content = V.singleton (txt (joinParts parts))
  149                           , Chat.tool_call_id = i
  150                           }
  151                        | CbToolResult (ToolCallId i) parts _ <- blocks ]
  152       mainMsg = case role of
  153         User -> Chat.User
  154           { Chat.content = V.fromList textContents
  155           , Chat.name = Nothing
  156           }
  157         Assistant -> Chat.Assistant
  158           { Chat.assistant_content =
  159               if null textContents then Nothing else Just (V.fromList textContents)
  160           , Chat.tool_calls =
  161               if null toolCallVec then Nothing else Just (V.fromList toolCallVec)
  162           , Chat.refusal = Nothing
  163           , Chat.name = Nothing
  164           , Chat.assistant_audio = Nothing
  165           }
  166   in mainMsg : toolResultMsgs
  167   where
  168     mkToolCall i n inp =
  169       OATC.ToolCall_Function
  170         { OATC.id = i
  171         , OATC.function = OATC.Function
  172             { OATC.name = n
  173             , OATC.arguments = TE.decodeUtf8 (BL.toStrict (encode inp))
  174             }
  175         }
  176     joinParts []     = ""
  177     joinParts (p:ps) = T.intercalate "\n" (map partText (p:ps))
  178     partText (TrpText t) = t

  180 -- Pure response mapping -----------------------------------------------------

  182 -- | Map a finish-reason string to our 'StopReason'. OpenRouter follows the
  183 -- OpenAI convention (@stop@ / @length@ / @tool_calls@); an unrecognized value
  184 -- (including @content_filter@) maps to 'StopOther' rather than failing — the
  185 -- caller can still use the partial response.
  186 mapFinishReason :: Text -> StopReason
  187 mapFinishReason r
  188   | r == "stop"        = StopEnd
  189   | r == "tool_calls"  = StopToolUse
  190   | r == "tool_use"     = StopToolUse
  191   | r == "length"      = StopMaxTokens
  192   | otherwise          = StopOther r
  193 
  194 -- | Decode an OpenAI 'Chat.ChatCompletionObject' into our
  195 -- 'CompletionResponse'. An empty @choices@ array is a graceful 'Left' (some
  196 -- upstream error conditions return no choices). The first choice is taken;
  197 -- its @message@ is an 'Chat.Assistant' carrying optional text + tool calls.
  198 -- Tool-call ids are taken verbatim from OpenRouter (it returns ids); the
  199 -- JSON-encoded @arguments@ text is parsed back to a 'Value'.
  200 decodeResponse :: Chat.ChatCompletionObject -> Either Text CompletionResponse
  201 decodeResponse obj =
  202   case V.toList (Chat.choices obj) of
  203     []     -> Left "OpenRouter returned no choices"
  204     (c:_)  ->
  205       let stop  = mapFinishReason (Chat.finish_reason c)
  206           blkss = assistantBlocks (Chat.message c)
  207           u     = Chat.usage obj
  208           usage = Usage
  209             { uInput  = fromIntegral (UAU.prompt_tokens u)
  210             , uOutput = fromIntegral (UAU.completion_tokens u)
  211             }
  212       in Right (CompletionResponse blkss stop usage)

  214 -- | Extract our 'ContentBlock's from an OpenAI assistant message: the text
  215 -- body becomes 'CbText' and each tool call becomes a 'CbToolUse' with the
  216 -- upstream id + name, and the JSON-encoded @arguments@ parsed back to a
  217 -- 'Value' (OpenRouter emits arguments as a JSON object encoded as a 'Text';
  218 -- if it ever fails to parse, we fall back to a placeholder 'Value' so the
  219 -- transcript still records the call).
  220 assistantBlocks :: Chat.Message Text -> [ContentBlock]
  221 assistantBlocks msg = case msg of
  222   Chat.Assistant mText _ _ _ mCalls ->
  223     let txts  = maybe [] (\t -> [CbText t]) mText
  224         calls = maybe [] (map toolCallBlock . V.toList) mCalls
  225     in txts <> calls
  226   _ -> []
  227   where
  228     toolCallBlock tc =
  229       let OATC.Function n args = OATC.function tc
  230           parsed = case eitherDecode (BL.fromStrict (TE.encodeUtf8 args)) :: Either String Value of
  231             Right v  -> v
  232             Left _e  -> Null
  233       in CbToolUse (ToolCallId (OATC.id tc)) (OpName n) parsed

  237 -- | The headers for an OpenRouter chat-completion POST. The API key bytes
  238 -- (@bytes@) live only here, inside the 'withApiKey' continuation. OpenRouter
  239 -- recommends @HTTP-Referer@ / @X-Title@ for app attribution; neither carries a
  240 -- secret.
  241 bearerHeaders :: ByteString -> RequestHeaders
  242 bearerHeaders bytes =
  243   [ ("content-type", "application/json")
  244   , ("authorization", "Bearer " <> bytes)
  245   , ("HTTP-Referer", "https://github.com/seal-harness")
  246   , ("X-Title", "seal-harness")
  247   ]

  249 -- HTTP round-trip ------------------------------------------------------------
  250 
  251 -- | POST @<baseUrl>/chat/completions@ with the given headers + request body,
  252 -- then decode the response. Non-2xx -> 'httpErrorText'; a transport error
  253 -- (connection / TLS) -> a 'Left' string. The body carries no secret. The
  254 -- @baseUrl@ is the configured one ('orBaseUrl'), so a custom/proxy endpoint
  255 -- is honored.
  256 sendChatCompletion
  257   :: Manager -> Text -> RequestHeaders -> Chat.CreateChatCompletion
  258   -> IO (Either Text CompletionResponse)
  259 sendChatCompletion mgr baseUrl hdrs req = do
  260   initReq <- parseRequest (T.unpack ("POST " <> baseUrl <> "/chat/completions"))
  261   let httpReq = initReq
  262         { requestBody = RequestBodyLBS (encode req)
  263         , requestHeaders = hdrs
  264         }
  265   result <- try (httpLbs httpReq mgr) :: IO (Either HttpException (Response BL.ByteString))
  266   case result of
  267     Left _e ->
  268       pure (Left "OpenRouter HTTP request failed (connection or transport error)")
  269     Right resp ->
  270       let code = statusCode (responseStatus resp)
  271       in if code >= 200 && code <= 299
  272            then case eitherDecode (responseBody resp) of
  273                   Left e    -> pure (Left (T.pack e))
  274                   Right obj -> pure (decodeResponse obj)
  275            else pure (Left (httpErrorText code
  276                              (TE.decodeUtf8With TEE.lenientDecode
  277                                (BL.toStrict (responseBody resp)))))
  278 
  279 -- | Render a non-2xx OpenRouter response into a user-facing message. 429
  280 -- (rate limit) and 401 (auth) get friendlier, actionable hints; other codes
  281 -- keep the full response body (the body carries no secret).
  282 httpErrorText :: Int -> Text -> Text
  283 httpErrorText 429 _ =
  284   "OpenRouter rate limit (HTTP 429) — the account behind this credential is \
  285   \being throttled. Wait a moment and retry, or switch models with /model use."
  286 httpErrorText 401 _ =
  287   "OpenRouter rejected the API key (HTTP 401) — run /provider remove openrouter \
  288   \then /provider add openrouter to re-enter the key."
  289 httpErrorText code body =
  290   "OpenRouter API returned HTTP " <> T.pack (show code) <> ": " <> body

  292 -- Provider instance ---------------------------------------------------------

  294 instance Provider OpenRouter where
  295   listModels prov = pure (Right [orDefaultModel prov])
  296   complete prov cr = case orApiKey prov of
  297     Nothing ->
  298       pure (Left "OpenRouter needs an API key \x2014 run /vault setup then /provider add openrouter")
  299     Just key ->
  300       withApiKey key $ \bytes ->
  301         sendChatCompletion (orManager prov) (orBaseUrl prov) (bearerHeaders bytes) (encodeRequest cr)

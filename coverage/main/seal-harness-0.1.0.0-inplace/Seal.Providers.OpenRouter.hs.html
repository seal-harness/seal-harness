    4 -- | OpenAI-compatible provider using the @openai@ package's *types* (v2.5.4)
    5 -- with our own HTTP round-trip (mirroring 'Seal.Providers.Anthropic' /
    6 -- 'Ollama'). We do NOT use 'OpenAI.V1.makeMethods' (the servant-client
    7 -- convenience wrapper) — that would unwrap the 'ApiKey' to 'Text' and surface
    8 -- 'servant-client' runtime exceptions. Instead the request is built as a
    9 -- 'OpenAI.V1.Chat.Completions.CreateChatCompletion', @encode@d to JSON, and
   10 -- POSTed with 'Network.HTTP.Client'; the response is @eitherDecode@d as a
   11 -- 'OpenAI.V1.Chat.Completions.ChatCompletionObject'. The API key bytes live
   12 -- only on the @Authorization: Bearer@ header built inside the CPS 'withApiKey'
   13 -- accessor — never returned, never logged.
   14 --
   15 -- This module originally began life as a fixed OpenRouter provider, but the
   16 -- 'OpenRouter' type now serves /any/ OpenAI-compatible
   17 -- @\/v1\/chat\/completions@ endpoint — a self-hosted vLLM, LM Studio,
   18 -- api.openai.com, Ollama Cloud, etc. The only difference between instances is
   19 -- the base URL the caller supplies, which is why 'Seal.Providers.Registry'
   20 -- routes both 'OpenRouterProvider' and 'OpenAICompatibleProvider' through
   21 -- here. Error messages include the base URL so a failure at an Ollama Cloud
   22 -- endpoint is not misattributed to OpenRouter.
   23 --
   24 -- OpenRouter-specific defaults: @https://openrouter.ai/api/v1@,
   25 -- @OPENROUTER_API_KEY@.
   26 --
   27 -- Security: capability-based — no opcode touches the agent-immutable home
   28 -- state dir; this module imports only the openai types, the HTTP client
   29 -- library, and the CPS secret accessor. It never spawns a process, never
   30 -- touches the filesystem directly, and never imports the web-client library
   31 -- whose types it borrows (see AGENTS.md §Security).

  133 -- | Build the OpenAI-compatible chat-completion request from our shape. The
  134 -- system prompt becomes a leading 'Chat.System' message; 'CbThinking' blocks
  135 -- are dropped (no OpenAI equivalent); 'CbToolUse' becomes the assistant's
  136 -- @tool_calls@; 'CbToolResult' becomes a 'Chat.Tool' role message carrying
  137 -- the matching @tool_call_id@.
  138 encodeRequest :: CompletionRequest -> Chat.CreateChatCompletion
  139 encodeRequest cr = Chat._CreateChatCompletion
  140   { Chat.messages = V.fromList (systemMsgs <> concatMap msgOpenAi (crMessages cr))
  141   , Chat.model = OAM.Model (modelIdText (crModel cr))
  142   , Chat.max_completion_tokens = Just (fromIntegral (crMaxTokens cr))
  143   , Chat.tools = toolsVec
  144   , Chat.tool_choice = tc
  145   }
  146   where
  147     systemMsgs = case crSystem cr of
  148       Nothing -> []
  149       Just s  -> [ Chat.System { Chat.content = V.singleton (txt s), Chat.name = Nothing } ]
  150     toolsVec = case crTools cr of
  151       []  -> Nothing
  152       ts  -> Just (V.fromList (map toolDef ts))
  153     tc = case crToolChoice cr of
  154       ToolAuto  -> Just OAT.ToolChoiceAuto
  155       ToolNone  -> Just OAT.ToolChoiceNone
  156 
  157     toolDef (ToolDefinition (OpName n) d sch) =
  158       OAT.Tool_Function
  159         ( OAT.Function
  160             { OAT.description = Just d
  161             , OAT.name = n
  162             , OAT.parameters = Just sch
  163             , OAT.strict = Nothing
  164             }
  165         )

  167 -- | Map a single Seal 'Message' to one or more OpenAI messages. A Seal
  168 -- assistant message that mixes text + tool calls + tool results splits into a
  169 -- single 'Chat.Assistant' message (text + tool_calls) plus one 'Chat.Tool'
  170 -- message per 'CbToolResult'. (OpenAI has no per-tool-result container; each
  171 -- result is its own @tool@-role message keyed by @tool_call_id@.)
  172 msgOpenAi :: Message -> [Chat.Message (V.Vector Chat.Content)]
  173 msgOpenAi (Message role blocks) =
  174   let textContents   = [ txt t     | CbText t <- blocks ]
  175       toolCallVec    = [ mkToolCall i n inp
  176                        | CbToolUse (ToolCallId i) (OpName n) inp <- blocks ]
  177       toolResultMsgs = [ Chat.Tool
  178                           { Chat.content = V.singleton (txt (joinParts parts))
  179                           , Chat.tool_call_id = i
  180                           }
  181                        | CbToolResult (ToolCallId i) parts _ <- blocks ]
  182       mainMsg = case role of
  183         User -> Chat.User
  184           { Chat.content = V.fromList textContents
  185           , Chat.name = Nothing
  186           }
  187         Assistant -> Chat.Assistant
  188           { Chat.assistant_content =
  189               if null textContents then Nothing else Just (V.fromList textContents)
  190           , Chat.tool_calls =
  191               if null toolCallVec then Nothing else Just (V.fromList toolCallVec)
  192           , Chat.refusal = Nothing
  193           , Chat.name = Nothing
  194           , Chat.assistant_audio = Nothing
  195           }
  196   in mainMsg : toolResultMsgs
  197   where
  198     mkToolCall i n inp =
  199       OATC.ToolCall_Function
  200         { OATC.id = i
  201         , OATC.function = OATC.Function
  202             { OATC.name = n
  203             , OATC.arguments = TE.decodeUtf8 (BL.toStrict (encode inp))
  204             }
  205         }
  206     joinParts []     = ""
  207     joinParts (p:ps) = T.intercalate "\n" (map partText (p:ps))
  208     partText (TrpText t) = t

  212 -- | Map a finish-reason string to our 'StopReason'. OpenRouter follows the
  213 -- OpenAI convention (@stop@ / @length@ / @tool_calls@); an unrecognized value
  214 -- (including @content_filter@) maps to 'StopOther' rather than failing — the
  215 -- caller can still use the partial response.
  216 mapFinishReason :: Text -> StopReason
  217 mapFinishReason r
  218   | r == "stop"        = StopEnd
  219   | r == "tool_calls"  = StopToolUse
  220   | r == "tool_use"     = StopToolUse
  221   | r == "length"      = StopMaxTokens
  222   | otherwise          = StopOther r

  224 -- | Decode an OpenAI 'Chat.ChatCompletionObject' into our
  225 -- 'CompletionResponse'. An empty @choices@ array is a graceful 'Left' (some
  226 -- upstream error conditions return no choices). The first choice is taken;
  227 -- its @message@ is an 'Chat.Assistant' carrying optional text + tool calls. The
  228 -- @endpoint@ label is included in error messages so a non-OpenRouter
  229 -- OpenAI-compatible instance (e.g. an Ollama Cloud endpoint) is not
  230 -- misattributed. Tool-call ids are taken verbatim from the upstream (it
  231 -- returns ids); the JSON-encoded @arguments@ text is parsed back to a 'Value'.
  232 decodeResponse :: Text -> Chat.ChatCompletionObject -> Either Text CompletionResponse
  233 decodeResponse endpoint obj =
  234   case V.toList (Chat.choices obj) of
  235     []     -> Left ("the endpoint " <> endpoint <> " returned no choices")
  236     (c:_)  ->
  237       let stop  = mapFinishReason (Chat.finish_reason c)
  238           blkss = assistantBlocks (Chat.message c)
  239           u     = Chat.usage obj
  240           usage = Usage
  241             { uInput  = fromIntegral (UAU.prompt_tokens u)
  242             , uOutput = fromIntegral (UAU.completion_tokens u)
  243             }
  244       in Right (CompletionResponse blkss stop usage)

  246 -- | Extract our 'ContentBlock's from an OpenAI assistant message: the text
  247 -- body becomes 'CbText' and each tool call becomes a 'CbToolUse' with the
  248 -- upstream id + name, and the JSON-encoded @arguments@ parsed back to a
  249 -- 'Value' (OpenRouter emits arguments as a JSON object encoded as a 'Text';
  250 -- if it ever fails to parse, we fall back to a placeholder 'Value' so the
  251 -- transcript still records the call).
  252 assistantBlocks :: Chat.Message Text -> [ContentBlock]
  253 assistantBlocks msg = case msg of
  254   Chat.Assistant mText _ _ _ mCalls ->
  255     let txts  = maybe [] (\t -> [CbText t]) mText
  256         calls = maybe [] (map toolCallBlock . V.toList) mCalls
  257     in txts <> calls
  258   _ -> []
  259   where
  260     toolCallBlock tc =
  261       let OATC.Function n args = OATC.function tc
  262           parsed = case eitherDecode (BL.fromStrict (TE.encodeUtf8 args)) :: Either String Value of
  263             Right v  -> v
  264             Left _e  -> Null
  265       in CbToolUse (ToolCallId (OATC.id tc)) (OpName n) parsed

  269 -- | The headers for an OpenAI-compatible chat-completion POST. The API key
  270 -- bytes (@bytes@) live only here, inside the 'withApiKey' continuation.
  271 -- OpenRouter recommends @HTTP-Referer@ / @X-Title@ for app attribution;
  272 -- neither carries a secret.
  273 bearerHeaders :: ByteString -> RequestHeaders
  274 bearerHeaders bytes =
  275   [ ("content-type", "application/json")
  276   , ("authorization", "Bearer " <> bytes)
  277   , ("HTTP-Referer", "https://github.com/seal-harness")
  278   , ("X-Title", "seal-harness")
  279   ]

  281 -- HTTP round-trip ------------------------------------------------------------
  282 
  283 -- | Response timeout for chat-completion requests, in microseconds. The
  284 -- @newTlsManager@ default is too short for non-streaming requests to remote
  285 -- cloud models with a large context (49k+ input tokens): the model can take
  286 -- 74+ seconds to generate the full response before returning it in one shot.
  287 -- A 5-minute timeout gives generous headroom, matching the Ollama provider's
  288 -- 'ollamaResponseTimeoutMicro'. This is a per-request override (the manager's
  289 -- default is not changed, so other HTTP clients in the process keep their own
  290 -- timeouts).
  291 chatCompletionTimeoutMicro :: Int
  292 chatCompletionTimeoutMicro = 300_000_000  -- 5 minutes
  293 
  294 -- | Render an 'HttpException' (transport-level failure) as a user-facing
  295 -- error message naming @baseUrl@. A 'ResponseTimeout' gets a distinct
  296 -- message so the user knows the endpoint is reachable but the model was too
  297 -- slow — not that the endpoint is down. Other transport errors use the
  298 -- generic wording.
  299 httpTransportErrorText :: Text -> HttpException -> Text
  300 httpTransportErrorText baseUrl = \case
  301   HttpExceptionRequest _ ResponseTimeout ->
  302     "request to " <> baseUrl <> " timed out after "
  303       <> T.pack (show (chatCompletionTimeoutMicro `div` 1_000_000))
  304       <> "s — the model likely took too long to respond (large context or \
  305         \a slow cloud model). Try reducing the conversation context, \
  306         \switching to a faster model, or send \"continue\" to retry."
  307   _ -> "HTTP request to " <> baseUrl
  308         <> " failed (connection or transport error)"
  309 
  310 -- | POST @<baseUrl>/chat/completions@ with the given headers + request body,
  311 -- then decode the response. Non-2xx -> 'httpErrorText'; a transport error
  312 -- (connection / TLS) -> a 'Left' string naming @baseUrl@ so the operator can
  313 -- see which endpoint is unreachable (this function serves all
  314 -- OpenAI-compatible providers, not just OpenRouter). The body carries no
  315 -- secret. The @baseUrl@ is the configured one ('orBaseUrl'), so a
  316 -- custom/proxy endpoint is honored.
  317 sendChatCompletion
  318   :: Manager -> Text -> RequestHeaders -> Chat.CreateChatCompletion
  319   -> IO (Either Text CompletionResponse)
  320 sendChatCompletion mgr baseUrl hdrs req = do
  321   initReq <- parseRequest (T.unpack ("POST " <> baseUrl <> "/chat/completions"))
  322   let httpReq = initReq
  323         { requestBody = RequestBodyLBS (encode req)
  324         , requestHeaders = hdrs
  325         , responseTimeout = responseTimeoutMicro chatCompletionTimeoutMicro
  326         }
  327   result <- try (httpLbs httpReq mgr) :: IO (Either HttpException (Response BL.ByteString))
  328   case result of
  329     Left e ->
  330       pure (Left (httpTransportErrorText baseUrl e))
  331     Right resp ->
  332       let code = statusCode (responseStatus resp)
  333       in if code >= 200 && code <= 299
  334            then case eitherDecode (responseBody resp) of
  335                   Left e    -> pure (Left (T.pack e))
  336                   Right obj -> pure (decodeResponse baseUrl obj)
  337            else pure (Left (httpErrorText code
  338                              (TE.decodeUtf8With TEE.lenientDecode
  339                                (BL.toStrict (responseBody resp)))))
  340 
  341 -- | Render a non-2xx response into a user-facing message. 429 (rate limit)
  342 -- and 401 (auth) get friendlier, actionable hints; other codes keep the full
  343 -- response body (the body carries no secret).
  344 httpErrorText :: Int -> Text -> Text
  345 httpErrorText 429 _ =
  346   "the provider is rate limiting this credential (HTTP 429) — wait a moment \
  347   \and retry, or switch models with /model use."
  348 httpErrorText 401 _ =
  349   "the endpoint rejected the API key (HTTP 401) — re-enter it with \
  350   \/provider remove and /provider add for this provider."
  351 httpErrorText code body =
  352   -- Capped: an endpoint that answers with an HTML error page should not paste
  353   -- the whole page into a CLI line or a wizard hint.
  354   "the provider returned HTTP " <> T.pack (show code) <> ": " <> T.take 300 body

  358 instance Provider OpenRouter where
  359   -- The endpoint's live catalogue. Real, not a static guess: for a generic
  360   -- OpenAI-compatible endpoint "is this URL right?" is answered here, and
  361   -- OpenRouter's own catalogue changes over time. With no credential the call
  362   -- is skipped and the configured default is returned, so a keyless caller
  363   -- still gets an answer without touching the network.
  364   listModels prov = case orApiKey prov of
  365     Nothing -> pure (Right [orDefaultModel prov])
  366     Just key ->
  367       withApiKey key $ \bytes ->
  368         listModelsHttp (orManager prov) (orBaseUrl prov) (bearerHeaders bytes)
  369 
  370   complete prov cr = case orApiKey prov of
  371     Nothing ->
  372       pure (Left ("the endpoint " <> orBaseUrl prov
  373         <> " needs an API key \x2014 run /vault setup then /provider add"))
  374     Just key ->
  375       withApiKey key $ \bytes ->
  376         sendChatCompletion (orManager prov) (orBaseUrl prov) (bearerHeaders bytes) (encodeRequest cr)

  378 -- | @GET {base}\/models@ — the OpenAI-compatible model-list endpoint.
  379 listModelsHttp :: Manager -> Text -> RequestHeaders -> IO (Either Text [ModelId])
  380 listModelsHttp mgr baseUrl hdrs = do
  381   initReq <- parseRequest (T.unpack (trimTrailingSlash baseUrl <> "/models"))
  382   let httpReq = initReq { requestHeaders = hdrs }
  383   result <- try (httpLbs httpReq mgr)
  384     :: IO (Either HttpException (Response BL.ByteString))
  385   case result of
  386     Left _e ->
  387       pure (Left "the provider's model-list request failed (connection or transport error)")
  388     Right resp ->
  389       let code = statusCode (responseStatus resp)
  390       in if code >= 200 && code <= 299
  391            then pure (decodeModelIds (responseBody resp))
  392            else pure (Left (httpErrorText code
  393                              (TE.decodeUtf8With TEE.lenientDecode
  394                                (BL.toStrict (responseBody resp)))))
  395 
  396 trimTrailingSlash :: Text -> Text
  397 trimTrailingSlash = T.dropWhileEnd (== '/')
  398 
  399 -- | Decode the OpenAI-compatible model list: @{"data":[{"id":"…"}]}@.
  400 decodeModelIds :: BL.ByteString -> Either Text [ModelId]
  401 decodeModelIds body = case eitherDecode body of
  402   Left err -> Left ("could not read the model list: " <> T.pack err)
  403   Right value -> case parseMaybe (withObject "model list" (.: "data")) value of
  404     Nothing -> Left "the endpoint returned no model list"
  405     Just entries -> maybe (Left "the endpoint's model list has entries without an id")
  406                            Right
  407                            (traverse entryId (entries :: [Value]))
  408   where
  409     entryId v = ModelId <$> parseMaybe (withObject "model" (.: "id")) v

    2 -- | The @/new@ command: start a fresh session in a new tab.
    3 --
    4 -- @/new@ mints a new 'SessionMeta' from the config defaults (or from
    5 -- optional @-p@\/@-m@ overrides) and inserts a new tab into the
    6 -- 'TabsHandle' — the same action the web frontend's "Start a new tab"
    7 -- form performs. The old session (if any) is kept on disk, untouched —
    8 -- still listed in @\/session list@, still resumable with @\/tab resume
    9 -- \<id\>@. When @-r@\/@--repo@ is given, the repo is cloned into the
   10 -- new session's workdir via SETUP_REPO before the first turn (same as
   11 -- the web form's "Set up repo" field).
   12 --
   13 -- The @-r@\/@--repo@ argument accepts either a registered repo ID (looked
   14 -- up in the repo registry @repos.toml@) or a raw git URL. When a registered
   15 -- ID is found, the registered URL is used for the clone. When no ID
   16 -- matches, the value is treated as a raw URL.
   17 --
   18 -- == Channel wiring
   19 --
   20 -- The command is registered as a 'CommandSpec' for the CLI and web-gateway
   21 -- paths (both track "current" via an active-session ref + a 'TabsHandle').
   22 -- The inbox channels (Signal, Telegram) handle @/new@ at the loop level
   23 -- ('Seal.Channels.Chat.Loop') instead, because the per-conversation cursor +
   24 -- conversation key aren't available to a registry 'CommandAction' — see the
   25 -- design doc.

   62 -- | Resolve the @-r@\/@--repo@ value to a git URL. First tries the value
   63 -- as a registered repo ID (looked up via 'RepoRegistryHandle'); if found,
   64 -- returns the registered 'srUrl'. If no ID matches, falls back to treating
   65 -- the value as a raw git URL. This lets the user type @\/new -r myrepo@
   66 -- (a registered id) or @\/new -r https:\/\/github.com\/foo\/bar.git@ (a raw
   67 -- URL) interchangeably.
   68 resolveRepoUrl :: RepoRegistryHandle -> Text -> IO Text
   69 resolveRepoUrl regH raw = do
   70   eRepos <- rrhList regH
   71   case eRepos of
   72     Right repos | Just repo <- matchById repos -> pure (srUrl repo)
   73     _ -> pure raw
   74   where
   75     matchById repos = do
   76       rid <- either (const Nothing) Just (mkRepoId (T.strip raw))
   77       listToMaybe [ r | r <- repos, srId r == rid ]

   79 -- | The deps a channel needs to run @/new@. Built once at startup.
   80 --
   81 -- 'ndInsertTab' is the seam where each channel plugs in its tab-insertion
   82 -- + active-session-swap logic. It receives the freshly-minted
   83 -- 'SessionMeta' and the 'ChannelCaps', inserts a new tab bound to the new
   84 -- session, swaps the active-session ref to the new meta, and returns the
   85 -- old sid (so the caller can render the confirmation line naming it).
   86 data NewDeps = NewDeps
   87   { ndPaths        :: SealPaths
   88   , ndCfg          :: IO RuntimeConfig
   89   , ndAgentDefs    :: Backends
   90   , ndChannelLabel :: Text
   91   , ndOldMeta      :: IO SessionMeta
   92     -- ^ Read the current (pre-swap) active session. Used to (a) determine
   93     -- the old sid for the confirmation line, and (b) as the source of
   94     -- default provider/model when no overrides are given.
   95   , ndInsertTab     :: ChannelCaps -> SessionMeta -> IO SessionId
   96     -- ^ Insert a new tab bound to the new session, swap the active-session
   97     -- ref, and return the old sid.
   98   , ndSetupRepo    :: Maybe (SessionId -> Text -> IO (Either Text Text))
   99     -- ^ Optional repo-setup seam. When 'Just', the @/new@ command calls it
  100     -- with the new session id + repo URL to clone a repo into the session
  101     -- workdir (dispatching SETUP_REPO). 'Nothing' in contexts without a
  102     -- dispatcher (standalone CLI, tests).
  103   , ndRepoReg      :: Maybe RepoRegistryHandle
  104     -- ^ Optional repo-registry handle for resolving @-r@ repo IDs to URLs.
  105     -- 'Nothing' in contexts without a registry (tests); the @-r@ value is
  106     -- then treated as a raw URL.
  107   }

  109 -- | The @/new@ command spec. Grouped under Sessions in @/help@. Always
  110 -- available (the operator typed it; no autonomy gate).
  111 newCommandSpec :: NewDeps -> CommandSpec
  112 newCommandSpec deps = CommandSpec
  113   { csName         = CommandName "new"
  114   , csAliases      = []
  115   , csGroup        = GroupSession
  116   , csSynopsis     = "Start a new tab with a fresh session (optional -p/-m/-r overrides)"
  117   , csParserInfo   = newParserInfo deps
  118   , csAvailability = AlwaysAvailable
  119   }

  152 -- | The @/new@ action: read the old session, mint a fresh session using
  153 -- the given args (or config defaults when not specified), insert a new
  154 -- tab, optionally clone a repo, and send the confirmation line.
  155 newCmd :: NewDeps -> Maybe Text -> Maybe Text -> Maybe Text -> CommandAction
  156 newCmd deps mProvider mModel mRepo = commandAction $ \caps -> do
  157   oldMeta <- ndOldMeta deps
  158   let args = NewArgs { naProvider = mProvider, naModel = mModel, naRepo = mRepo }
  159   meta <- mintNewSessionWith args oldMeta deps
  160   oldSid <- ndInsertTab deps caps meta
  161   case (naRepo args, ndSetupRepo deps) of
  162     (Just repoVal, Just setupFn) -> do
  163       repoUrl <- resolveRepoUrlMaybe deps repoVal
  164       eRes <- setupFn (smId meta) repoUrl
  165       case eRes of
  166         Left err -> ccSend caps ("repo setup failed: " <> err)
  167         Right _  -> pure ()
  168     _ -> pure ()
  169   ccSend caps (renderNewConfirmation meta oldSid)
  170 
  171 -- | Resolve the @-r@ value to a URL using the repo registry (if wired),
  172 -- falling back to the raw value.
  173 resolveRepoUrlMaybe :: NewDeps -> Text -> IO Text
  174 resolveRepoUrlMaybe deps raw =
  175   case ndRepoReg deps of
  176     Just regH -> resolveRepoUrl regH raw
  177     Nothing   -> pure raw

  179 -- | Mint a fresh 'SessionMeta' using the given args, falling back to the
  180 -- old session's provider/model when no override is given (so mid-session
  181 -- @\/model use@ changes survive @\/new@), and ultimately to config
  182 -- defaults. Persisted to disk via 'newSession' so @/session list@ picks it
  183 -- up. Shared by the registry path (CLI/web) and the loop-level inbox path.
  184 mintNewSessionWith :: NewArgs -> SessionMeta -> NewDeps -> IO SessionMeta
  185 mintNewSessionWith args oldMeta deps =
  186   newSession (ndPaths deps) provider model (ndChannelLabel deps) (smAgent oldMeta)
  187   where
  188     provider = fromMaybe (smProvider oldMeta) (naProvider args)
  189     model    = fromMaybe (smModel oldMeta) (naModel args)
  190 
  191 -- | Mint a fresh 'SessionMeta' from the config defaults (or the given
  192 -- overrides). Used when there is no "old session" to inherit from (e.g.
  193 -- the loop-level path on first conversation message).
  194 mintNewSession :: NewArgs -> NewDeps -> IO SessionMeta
  195 mintNewSession args deps = do
  196   cfg <- ndCfg deps
  197   (mAgent, mProv, mModel) <- resolveDefaultAgent (bAgentDefs (ndAgentDefs deps)) cfg
  198   let (cfgProv, cfgModel) = defaultSessionSelection cfg
  199       provider = fromMaybe (fromMaybe cfgProv mProv) (naProvider args)
  200       model    = fromMaybe (fromMaybe cfgModel mModel) (naModel args)
  201   newSession (ndPaths deps) provider model (ndChannelLabel deps) mAgent
  202 
  203 -- | Render the one-line confirmation. Names the old session + resume hint
  204 -- so the safety net (old session kept on disk) is user-visible, per
  205 -- PM/designer review.
  206 renderNewConfirmation :: SessionMeta -> SessionId -> Text
  207 renderNewConfirmation meta oldSid =
  208   renderNewSessionConfirmation
  209     (sessionIdText (smId meta))
  210     (smProvider meta)
  211     (smModel meta)
  212     (sessionIdText oldSid)

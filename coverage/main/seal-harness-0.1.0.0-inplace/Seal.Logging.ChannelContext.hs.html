    3 -- | Channel metadata attached to log entries via 'withChannelContext'.
    4 -- Carries the channel kind (typed 'ChannelKind' enum, not a magic string),
    5 -- a SHA-256 hash of the conversation id (first 12 hex chars — PII defense:
    6 -- phone numbers and chat ids are never logged in full to the operator
    7 -- console), and the session id (timestamp-derived, not PII).
    8 --
    9 -- The context is 'Maybe'-heavy so partial states (a reader thread with no
   10 -- session yet, a slash command before the turn starts) can still log with
   11 -- whatever metadata is available.
   12 module Seal.Logging.ChannelContext
   13   ( ChannelContext (..)
   14   , ctxFromMessageSource
   15   , ctxFromSession
   16   , hashConversationId
   17   ) where
   18 
   19 import Data.Aeson (ToJSON (..), object, (.=))
   20 import Data.Text (Text)
   21 import Data.Text qualified as T
   22 import Data.Text.Encoding qualified as TE
   23 import GHC.Generics (Generic)
   24 
   25 import Katip (LogItem (..), ToObject (..), PayloadSelection (..))
   26 
   27 import Seal.Core.ChannelKind (ChannelKind (..), channelKindToText, channelKindFromText)
   28 import Seal.Core.MessageSource (MessageSource (..), conversationIdText)
   29 import Seal.Core.Types (sessionIdText)
   30 import Seal.Security.Crypto (sha256Hash)
   31 import Seal.Session.Meta (SessionMeta (..))

   33 -- | Structured channel metadata for log entries. Attached to a 'SealLogger'
   34 -- via 'withChannelContext' at the start of each turn.
   35 data ChannelContext = ChannelContext
   36   { ccChannelKind         :: Maybe ChannelKind
   37     -- ^ The typed channel enum ('Telegram', 'Signal', 'Web', 'Cli', etc.).
   38     -- Serialized via 'channelKindToText' in the 'ToJSON' instance.
   39   , ccConversationIdHash  :: Maybe Text
   40     -- ^ SHA-256 hash (first 12 hex chars) of the conversation id. PII
   41     -- defense: phone numbers (Signal) and chat ids (Telegram) are never
   42     -- logged in full to the operator console. The full conversation id is
   43     -- available only in the per-session @seal.log@.
   44   , ccSessionId           :: Maybe Text
   45     -- ^ The session id (timestamp-derived, not PII). Logged in full so the
   46     -- operator can correlate log lines to sessions.
   47   }
   48   deriving stock (Eq, Show, Generic)
   49 
   50 instance ToJSON ChannelContext where
   51   toJSON ctx = object
   52     [ "channelKind"        .= fmap channelKindToText (ccChannelKind ctx)
   53     , "conversationIdHash" .= ccConversationIdHash ctx
   54     , "sessionId"          .= ccSessionId ctx
   55     ]
   56 
   57 instance ToObject ChannelContext
   58 
   59 instance LogItem ChannelContext where
   60   payloadKeys _ _ = AllKeys

   62 -- | Build a 'ChannelContext' from a 'MessageSource' (if available) and an
   63 -- optional 'SessionMeta'. The conversation id is hashed (never logged in
   64 -- full). Used by inbox channels (Telegram, Signal) which have a
   65 -- 'MessageSource'.
   66 ctxFromMessageSource :: Maybe MessageSource -> Maybe SessionMeta -> ChannelContext
   67 ctxFromMessageSource mSrc mMeta =
   68   ChannelContext
   69     { ccChannelKind        = fmap msChannelKind mSrc
   70     , ccConversationIdHash = fmap (hashConversationId . conversationIdText . msConversationId) mSrc
   71     , ccSessionId          = fmap (sessionIdText . smId) mMeta
   72     }
   73 
   74 -- | Build a 'ChannelContext' from a 'SessionMeta' alone (no 'MessageSource').
   75 -- Used by the CLI and web paths which don't have a 'MessageSource' with
   76 -- channel kind. The channel kind is derived from 'smChannel' if it
   77 -- matches a known 'ChannelKind'.
   78 ctxFromSession :: SessionMeta -> ChannelContext
   79 ctxFromSession meta =
   80   ChannelContext
   81     { ccChannelKind        = channelKindFromText (smChannel meta)
   82     , ccConversationIdHash = Nothing
   83     , ccSessionId          = Just (sessionIdText (smId meta))
   84     }
   85 
   86 -- | Hash a conversation id to its first 12 hex chars (SHA-256). This is
   87 -- the PII defense: the operator log carries the hash, not the full id.
   88 -- The hash is sufficient for correlation (matching log lines to a
   89 -- conversation) without exposing phone numbers or chat ids.
   90 hashConversationId :: Text -> Text
   91 hashConversationId cid =
   92   T.take 12 (T.pack (show (sha256Hash (TE.encodeUtf8 cid))))

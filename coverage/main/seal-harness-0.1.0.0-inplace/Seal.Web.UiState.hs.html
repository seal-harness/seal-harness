    4 -- | Persistent UI state for the web gateway: the last-chosen "new tab" form

   39 -- | The persisted UI state. Both fields are optional so a missing/empty
   40 -- file decodes to the empty state (the frontend treats that as "use the
   41 -- defaults").
   42 data UiState = UiState
   43   { usLastOptions  :: Maybe LastOptions
   44     -- ^ The last-chosen "new tab" form selection.
   45   , usCustomModels :: [Text]
   46     -- ^ Custom model ids the user has typed before, most-recent first,
   47     -- deduped. Kept small (capped at 'maxCustomModels') by 'addCustomModel'.
   48   , usRepoHistory :: [Text]
   49     -- ^ Repository URLs the user has "set up" before, most-recent first,
   50     -- deduped. The "set up repo" combo box offers these as suggestions.
   51     -- Kept small (capped at 'maxRepoHistory') by 'addRepoHistory'.
   52   } deriving stock (Eq, Show, Generic)
   53 
   54 -- | The last-chosen "new tab" form selection. Mirrors the frontend's
   55 -- @useNewTabSpec@ shape; only the user-selectable fields are persisted (the
   56 -- transiently-computed model list + validation state are NOT).
   57 data LastOptions = LastOptions
   58   { loKind          :: Text
   59     -- ^ @\"provider\"@ | @\"harness\"@ | @\"attach\"@
   60   , loProvider      :: Text
   61   , loModel         :: Text
   62   , loUseCustomModel :: Bool
   63   , loFlavour       :: Text
   64   , loCustomBinary  :: Text
   65   , loAttachSession :: Text
   66   , loAttachWindow  :: Text
   67   , loAttachManual  :: Bool
   68   , loRepo          :: Text
   69     -- ^ The last-entered repo URL for the "set up repo" combo box (empty
   70     -- string = no repo). Persisted so the combo box reopens with the last
   71     -- URL, and recorded into 'usRepoHistory' on submit.
   72   } deriving stock (Eq, Show, Generic)

  114 -- | The maximum number of custom model ids retained. Keeps the combobox
  115 -- list bounded — the user is unlikely to type more than 32 unique ids over
  116 -- time, and the list is for fast recall, not an audit log.
  117 maxCustomModels :: Int
  118 maxCustomModels = 32
  119 
  120 -- | The maximum number of repo URLs retained in the history. Same
  121 -- rationale as 'maxCustomModels'.
  122 maxRepoHistory :: Int
  123 maxRepoHistory = 32

  125 -- | A handle holding the in-memory copy + the on-disk path. The MVar
  126 -- serializes writes so concurrent PUTs don't interleave file writes.
  127 data UiStateHandle = UiStateHandle
  128   { uhPath  :: FilePath
  129   , uhState :: MVar UiState
  130   }
  131 
  132 -- | Create a 'UiStateHandle' by loading the file at
  133 -- @\<state\>\/ui_state.json@. Missing/unparseable file → empty state (a
  134 -- warning is emitted to stderr on a parse error so the operator knows).
  135 newUiStateHandle :: SealPaths -> IO UiStateHandle
  136 newUiStateHandle paths = do
  137   let path = spState paths </> "ui_state.json"
  138   mState <- loadUiState path
  139   ref <- newMVar mState
  140   pure UiStateHandle { uhPath = path, uhState = ref }
  141 
  142 -- | Read the current 'UiState'. Cheap — reads the in-memory copy.
  143 getUiState :: UiStateHandle -> IO UiState
  144 getUiState h = readMVar (uhState h)
  145 
  146 -- | Replace the last-chosen form selection. Persists atomically; never
  147 -- throws (a write failure logs to stderr and keeps the in-memory copy so
  148 -- the UI still works within the session).
  149 setLastOptions :: UiStateHandle -> LastOptions -> IO ()
  150 setLastOptions h opts =
  151   modifyMVar_ (uhState h) $ \s -> do
  152     let next = s { usLastOptions = Just opts }
  153     persistUiState (uhPath h) next
  154     pure next
  155 
  156 -- | Add a custom model id to the history. Dedupes, trims, caps at
  157 -- 'maxCustomModels' (most-recent first), and persists atomically. A blank
  158 -- id is a no-op. Never throws.
  159 addCustomModel :: UiStateHandle -> Text -> IO ()
  160 addCustomModel h raw =
  161   modifyMVar_ (uhState h) $ \s -> do
  162     let trimmed = T.strip raw
  163         next = if T.null trimmed
  164                  then s
  165                  else s { usCustomModels = dedupe trimmed (usCustomModels s) }
  166     persistUiState (uhPath h) next
  167     pure next
  168   where
  169     -- Move @x@ to the front, drop any earlier occurrence, cap the tail.
  170     dedupe x xs = take maxCustomModels (x : filter (/= x) xs)
  171 
  172 -- | Add a repo URL to the history. Dedupes, trims, caps at
  173 -- 'maxRepoHistory' (most-recent first), and persists atomically. A blank
  174 -- URL is a no-op. Never throws.
  175 addRepoHistory :: UiStateHandle -> Text -> IO ()
  176 addRepoHistory h raw =
  177   modifyMVar_ (uhState h) $ \s -> do
  178     let trimmed = T.strip raw
  179         next = if T.null trimmed
  180                  then s
  181                  else s { usRepoHistory = dedupe trimmed (usRepoHistory s) }
  182     persistUiState (uhPath h) next
  183     pure next
  184   where
  185     dedupe x xs = take maxRepoHistory (x : filter (/= x) xs)

  187 -- ── Internal: load/persist ─────────────────────────────────────────────

  189 -- | The empty state: no last options, no custom models, no repo history.
  190 emptyUiState :: UiState
  191 emptyUiState = UiState { usLastOptions = Nothing, usCustomModels = [], usRepoHistory = [] }
  192 
  193 -- | Load the state file. Missing file → empty state. Unparseable → empty
  194 -- state + a stderr warning (the file is replaced on the next successful
  195 -- write, so corruption self-heals).
  196 loadUiState :: FilePath -> IO UiState
  197 loadUiState path = do
  198   exists <- doesFileExist path
  199   if not exists
  200     then pure emptyUiState
  201     else do
  202       bs <- BL.readFile path
  203       case A.decode bs :: Maybe UiState of
  204         Just s  -> pure s
  205         Nothing -> do
  206           globalLogIO WarningS "could not parse ui_state.json; using empty UI state"
  207           pure emptyUiState
  208 
  209 -- | Persist the state atomically: write @.tmp@, rename over the target.
  210 -- Creates the parent dir (it exists under @\<state\>@, but the call is
  211 -- idempotent). On any IO error, logs to stderr and continues (the
  212 -- in-memory copy stays; the UI still works within the session).
  213 persistUiState :: FilePath -> UiState -> IO ()
  214 persistUiState path s = do
  215   createDirectoryIfMissing True (takeDirectory path)
  216   let tmp = path <> ".tmp"
  217   BL.writeFile tmp (A.encode s)
  218   renameFile tmp path

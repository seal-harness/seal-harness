    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Auto-setup for SearXNG: checks if a local instance is reachable,
    3 -- and if not, starts one via Docker with JSON format enabled.
    4 module Seal.Web.SearXngSetup
    5   ( ensureSearXng
    6   , searXngDefaultUrl
    7   ) where
    8 
    9 import Control.Concurrent (threadDelay)
   10 import Control.Exception (try)
   11 import Data.Text (Text)
   12 import Data.Text qualified as T
   13 import Network.HTTP.Client
   14   ( HttpException, Manager, httpLbs, parseRequest
   15   , responseStatus )
   16 import Network.HTTP.Types (statusCode)
   17 import System.Exit (ExitCode (..))
   18 import System.Process (readProcessWithExitCode)
   19 
   20 searXngDefaultUrl :: Text
   21 searXngDefaultUrl = "http://localhost:8888"
   22 
   23 -- | Check if SearXNG is reachable at the given URL. Returns 'True' if
   24 -- the instance responds with HTTP 200 on the @/search@ endpoint.
   25 isSearXngRunning :: Manager -> Text -> IO Bool
   26 isSearXngRunning mgr url = do
   27   eReq <- try (parseRequest (T.unpack (url <> "/search?q=test&format=json")))
   28   case eReq of
   29     Left (_ :: HttpException) -> pure False
   30     Right req -> do
   31       eResp <- try (httpLbs req mgr)
   32       case eResp of
   33         Left (_ :: HttpException) -> pure False
   34         Right resp -> pure (statusCode (responseStatus resp) == 200)
   35 
   36 -- | Ensure a SearXNG instance is running. If the configured URL is the
   37 -- default localhost:8888 and no instance is reachable, attempt to start
   38 -- one via Docker. Returns 'True' if an instance is reachable after setup.
   39 -- Returns 'False' if Docker is not available or the container fails to start.
   40 ensureSearXng :: Manager -> Text -> IO Bool
   41 ensureSearXng mgr url = do
   42   running <- isSearXngRunning mgr url
   43   if running
   44     then pure True
   45     else do
   46       if not (isLocalhost url)
   47         then pure False
   48         else tryDockerInstall
   49   where
   50     isLocalhost u = "localhost" `T.isInfixOf` u || "127.0.0.1" `T.isInfixOf` u
   51 
   52     tryDockerInstall :: IO Bool
   53     tryDockerInstall = do
   54       (exitCode, _, _) <- readProcessWithExitCode "docker" ["--version"] ""
   55       case exitCode of
   56         ExitFailure _ -> pure False
   57         ExitSuccess -> do
   58           (ec, out, _) <- readProcessWithExitCode "docker"
   59             ["ps", "-a", "-q", "--filter", "name=seal-harness-searxng"] ""
   60           case ec of
   61             ExitFailure _ -> pure False
   62             ExitSuccess ->
   63               if not (null (dropWhile (=='\n') out))
   64                 then do
   65                   _ <- readProcessWithExitCode "docker"
   66                     ["start", "seal-harness-searxng"] ""
   67                   threadDelay 3000000  -- 3 seconds
   68                   isSearXngRunning mgr url
   69                 else do
   70                   _ <- readProcessWithExitCode "docker"
   71                     ["pull", "searxng/searxng:latest"] ""
   72                   let dockerArgs =
   73                         [ "run", "-d"
   74                         , "--name", "seal-harness-searxng"
   75                         , "-p", "8888:8080"
   76                         , "-e", "SEARXNG_BASE_URL=http://localhost:8888/"
   77                         , "searxng/searxng:latest"
   78                         ]
   79                   (ec2, _, _) <- readProcessWithExitCode "docker" dockerArgs ""
   80                   case ec2 of
   81                     ExitFailure _ -> pure False
   82                     ExitSuccess -> do
   83                       threadDelay 5000000  -- 5 seconds
   84                       _ <- readProcessWithExitCode "docker"
   85                         ["exec", "seal-harness-searxng"
   86                         , "sh", "-c"
   87                         , "sed -i 's/formats: \\[html\\]/formats: [html, json]/' /etc/searxng/settings.yml || true"
   88                         ] ""
   89                       _ <- readProcessWithExitCode "docker"
   90                         ["restart", "seal-harness-searxng"] ""
   91                       threadDelay 3000000  -- 3 seconds
   92                       isSearXngRunning mgr url

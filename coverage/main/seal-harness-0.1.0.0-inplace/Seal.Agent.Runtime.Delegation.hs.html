    2 -- | The delegation pipeline — the Seal analog of Hermes' @delegate_task@.
    3 --
    4 -- 'runDelegate' spawns one or more child agents with isolated context, runs each
    5 -- against a goal to completion (synchronously), and returns a structured
    6 -- result per child. The parent blocks until all children finish (or time out).
    7 --
    8 -- Each child gets:
    9 --
   10 --   * a fresh 'SessionId' and its own two-file transcript nested under
   11 --     @\<parent-session\>\/agents\/\<child-id\>@ (the worker-builder opens it);
   12 --   * a focused system prompt built from the def's @adSystem@ + the goal +
   13 --     optional context (the worker-builder assembles it);
   14 --   * a narrowed tool set: the def's @adTools@ allow-list with the delegation
   15 --     blocklist always stripped (no recursive spawning, no def mutation, no
   16 --     lifecycle introspection) — the worker-builder is responsible for
   17 --     applying the blocklist when it constructs the child ISA registry.
   18 --
   19 -- Features ported from @delegate_task@:
   20 --
   21 --   * batch mode ('tasks' array) with a concurrency cap
   22 --     (@delegation.max_concurrent_children@);
   23 --   * depth cap (@delegation.max_spawn_depth@, default 1, clamped to [1,3])
   24 --     + @orchestrator_enabled@ kill switch (the role field is accepted but the
   25 --     wiring layer enforces the effective role when it builds the child's
   26 --     ISA registry);
   27 --   * timeout (@delegation.child_timeout_seconds@, default 600s, floor 30s)
   28 --     via 'System.Timeout.timeout';
   29 --   * a heartbeat thread that touches a parent-activity cell every 30s so a
   30 --     gateway inactivity timeout doesn't fire while the child works;
   31 --   * a spawn-pause flag so an operator can freeze new fan-out without
   32 --     interrupting running children;
   33 --   * per-child provider\/model override via @[delegation] provider/model/base_url@
   34 --     (the worker-builder reads the override and resolves the child provider
   35 --     from it instead of inheriting the parent's);
   36 --   * rich result aggregation: summary, status, exit_reason, tokens,
   37 --     tool_trace, files read/written (the worker reports via the
   38 --     'ChildRunHooks' accumulators).
   39 --
   40 -- Differences from @delegate_task@ (by design):
   41 --
   42 --   * Children are /transcripted/ — each child's conversation + entries land
   43 --     on disk under the parent's @agents\/\<child-id\>@ dir, surviving restart.
   44 --   * The child run is the existing 'Seal.Agent.Loop.runTurn' with the goal as
   45 --     the first user message — there is no separate child AIAgent class.

  172 -- | Resolve a loaded 'DelegationConfig' to its effective numeric knobs, with
  173 -- the documented clamps applied. Provider/model/base_url/api_key/api_mode
  174 -- overrides are read directly from the config (they're 'Maybe'-typed).
  175 resolveDelegationConfig :: DelegationConfig
  176                        -> (Int, Double, Int, Bool)
  177 resolveDelegationConfig cfg =
  178   ( maxConc
  179   , childTimeout
  180   , maxDepth
  181   , orchEnabled
  182   )
  183   where
  184     maxConc = max 1 $ fromMaybe defaultMaxConcurrentChildren (dcMaxConcurrentChildren cfg)
  185     childTimeout = max minChildTimeoutSeconds $
  186                      fromMaybe defaultChildTimeoutSeconds (dcChildTimeoutSeconds cfg)
  187     rawDepth = fromMaybe defaultMaxSpawnDepth (dcMaxSpawnDepth cfg)
  188     maxDepth = max minSpawnDepth (min maxSpawnDepthCap rawDepth)
  189     orchEnabled = fromMaybe True (dcOrchestratorEnabled cfg)

  212 -- | Opaque identifier for one live subagent. Smart-constructed from the def
  213 -- id + a short random suffix so multiple concurrent children of the same def
  214 -- don't collide.
  215 newtype SubagentId = SubagentId { unSubagentId :: Text }
  216   deriving stock (Eq, Ord, Show)
  217 
  218 -- | Mint a fresh 'SubagentId' from a def-id text + 8 hex chars. The def id is
  219 -- embedded for debuggability (the TUI can show which def a subagent ran);
  220 -- the random suffix disambiguates concurrent spawns.
  221 mkSubagentId :: Text -> IO SubagentId
  222 mkSubagentId defIdText = do
  223   suffix <- randomRIO (0 :: Int, 0xFFFFFFFF)
  224   pure (SubagentId (defIdText <> "-" <> showHexI32 suffix))
  225 
  226 subagentIdText :: SubagentId -> Text
  227 subagentIdText (SubagentId t) = t
  228 
  229 -- | Show an 'Int' as 8-char zero-padded lowercase hex.
  230 showHexI32 :: Int -> Text
  231 showHexI32 n = T.justifyRight 8 '0' (T.pack (go n))
  232   where
  233     go x
  234       | x <= 0    = "0"
  235       | otherwise = loop x ""
  236     loop y acc
  237       | y == 0    = acc
  238       | otherwise = let (q, r) = y `divMod` 16
  239                         c = if r < 10 then toEnum (r + fromEnum '0')
  240                                       else toEnum (r - 10 + fromEnum 'a')
  241                     in loop q (c : acc)

  247 -- | One task to delegate. The def id resolves the provider/model/system/tools
  248 -- template; the goal is the first user message; the context is optional
  249 -- background appended to the system prompt.
  250 data ChildTask = ChildTask
  251   { ctDefId   :: !Text
  252     -- ^ The agent def id to spawn from (must already exist via AGENT_DEF_WRITE).
  253   , ctGoal    :: !Text
  254     -- ^ The task goal — becomes the first user message in the child's turn.
  255   , ctContext :: !(Maybe Text)
  256     -- ^ Optional background context appended to the system prompt.
  257   , ctRole    :: !(Maybe Text)
  258     -- ^ Narrow-only per-task hint: @\"leaf\"@ downgrades an orchestrator
  259     -- def's child to a leaf (no spawning). Spawning capability is
  260     -- DEF-authoritative — a leaf def can never be widened by task input
  261     -- (issue #154: the effective role is computed in W2; this field is

  357 -- | A worker-builder: given a def + fresh session id + the task + the run
  358 -- hooks, run the child turn to completion and return the outcome.
  359 --
  360 -- The wiring layer (Cli.hs, Channels.Loop.hs, Gateway.Send.hs) closes over
  361 -- the per-turn 'AgentEnv' deps (caps, execBackend, appEnv, eCfg, parent
  362 -- session id, parent depth) and produces the IO action. Keeping this as a
  363 -- parameter decouples the opcode from 'Seal.Agent.Loop.runTurn' / provider
  364 -- resolution so the opcode is unit-testable with a fake worker.
  365 type AgentWorkerBuilder
  366   =  AgentDef
  367   -> SessionId
  368   -> ChildTask
  369   -> ChildRunHooks
  370   -> IO ChildWorkerOutcome

  372 -- | Thread-safe accumulators the worker-builder writes to during the child
  373 -- run. 'runDelegate' reads them after the worker returns to populate the
  374 -- 'ChildResult'. The worker appends to these via 'atomicModifyIORef'' (the
  375 -- IORefs are single-writer-per-child, but atomicModify is cheap and safe).
  376 data ChildRunHooks = ChildRunHooks

  381   , crhFilesWritten :: IORef [Text]
  382     -- ^ Append-only file-written accumulator.
  383   , crhInterrupted  :: IORef Bool
  384     -- ^ Polled by the worker between iterations; when 'True' the worker
  385     -- should stop at its next turn boundary and return 'CwoInterrupted'.
  386   }

  408 -- | A process-global flag that, when paused, makes 'runDelegate' reject new
  409 -- spawns with a structured error. Running children keep running; only NEW
  410 -- 'runDelegate' calls fail fast until 'setSpawnPaused' clears the flag.
  411 newtype SpawnPauseFlag = SpawnPauseFlag (TVar Bool)
  412 
  413 newSpawnPauseFlag :: IO SpawnPauseFlag
  414 newSpawnPauseFlag = SpawnPauseFlag <$> newTVarIO False
  415 
  416 setSpawnPaused :: SpawnPauseFlag -> Bool -> IO Bool
  417 setSpawnPaused (SpawnPauseFlag tv) paused = do
  418   atomically (writeTVar tv paused)
  419   pure paused
  420 
  421 isSpawnPaused :: SpawnPauseFlag -> IO Bool
  422 isSpawnPaused (SpawnPauseFlag tv) = readTVarIO tv

  476 -- | The top-level delegation runner. Spawns one or more child agents, runs
  477 -- each against its goal to completion (synchronously), and returns a
  478 -- 'ChildResult' per task. The parent blocks until all children finish (or
  479 -- time out). See the module header for the feature list.
  480 runDelegate
  481   :: DelegationConfig
  482   -> SpawnPauseFlag
  483   -> Maybe ParentActivity
  484   -> Int
  485      -- ^ the parent's current delegation depth (0 = top-level parent)
  486   -> DelegateInput
  487   -> (ChildTask -> IO (Either Text (AgentDef, AgentWorkerBuilder, SessionId)))
  488      -- ^ resolver: look up the def, build the worker, mint a fresh child
  489      -- 'SessionId'. 'Left' = def not found / invalid / depth would be
  490      -- exceeded. Per-task so the caller can cache or re-resolve.
  491   -> IO (Either Text [ChildResult])
  492 runDelegate cfg pauseFlag mParentActivity parentDepth input resolveTask = do
  493   paused <- isSpawnPaused pauseFlag
  494   if paused
  495     then pure (Left "Delegation spawning is paused. Clear the pause via the TUI or the AGENT_INTERRUPT RPC before retrying.")
  496     else do
  497       let (maxConc, childTimeout, maxDepth, _orchEnabled) = resolveDelegationConfig cfg
  498       if parentDepth >= maxDepth
  499         then pure (Left ("Delegation depth limit reached (depth=" <> T.pack (show parentDepth)
  500                           <> ", max_spawn_depth=" <> T.pack (show maxDepth)
  501                           <> "). Raise delegation.max_spawn_depth in config.toml if deeper nesting is required (cap: "
  502                           <> T.pack (show maxSpawnDepthCap) <> ")."))
  503         else do
  504           tasks <- case input of
  505             DiSingle t -> pure [t]
  506             DiBatch ts -> pure ts
  507           if null tasks
  508             then pure (Left "No tasks provided.")
  509             else do
  510               when (length tasks > maxConc) $
  511                 -- We don't reject oversize batches; the semaphore caps
  512                 -- parallelism. (delegate_task rejects; we cap.)
  513                 pure ()
  514               runBatch maxConc childTimeout tasks
  515   where
  516     -- Run all tasks with a concurrency cap. Single-task short-cut avoids the
  517     -- thread-pool overhead.
  518     runBatch :: Int -> Double -> [ChildTask] -> IO (Either Text [ChildResult])
  519     runBatch maxConc childTimeout tasks =
  520       case tasks of
  521         [t] -> do
  522           r <- runOne 0 t childTimeout
  523           pure (Right [r])
  524         _   -> do
  525           sem <- newTVarIO maxConc
  526           mvars <- forM (zip [0 ..] tasks) $ \(idx, task) -> do
  527             resMVar <- newEmptyMVar
  528             void (forkIO $
  529               bracketSem sem $
  530                 runOne idx task childTimeout >>= putMVar resMVar)
  531             pure resMVar
  532           results <- forM mvars readMVar
  533           pure (Right results)
  534 
  535     -- Acquire / release the counting semaphore around an IO action. The
  536     -- semaphore is a 'TVar Int' initialized to @maxConc@: each acquire
  537     -- atomically decrements (blocking if zero via 'retry'), each release
  538     -- atomically increments. This allows up to @maxConc@ concurrent holders,
  539     -- unlike the previous 'MVar'-based approach which was a binary semaphore
  540     -- (mutex) that serialized all children regardless of @maxConc@.
  541     bracketSem :: TVar Int -> IO a -> IO a
  542     bracketSem sem act = bracket acquire release (const act)
  543       where
  544         acquire = atomically $ do
  545           n <- readTVar sem
  546           if n > 0
  547             then writeTVar sem (n - 1)
  548             else retry
  549         release _ = atomically (modifyTVar' sem (+1))
  550 
  551     -- Run a single task to completion with a hard timeout and a heartbeat
  552     -- thread that touches the parent-activity cell.
  553     runOne :: Int -> ChildTask -> Double -> IO ChildResult
  554     runOne idx task childTimeout = do
  555       subagentId <- mkSubagentId (ctDefId task)
  556       let micros = round (childTimeout * 1000000) :: Int
  557       start <- getCurrentTime
  558       -- Thread-safe accumulators the worker writes to.
  559       traceRef     <- newIORef []
  560       readRef     <- newIORef []
  561       writtenRef  <- newIORef []
  562       interruptedRef <- newIORef False
  563       let hooks = ChildRunHooks traceRef readRef writtenRef interruptedRef
  564       mRes <- resolveTask task >>= \case
  565         Left err -> pure (Left err)
  566         Right (def, worker, childSid) -> do
  567           -- Start the heartbeat thread.
  568           hbStop <- newEmptyMVar
  569           hbThreadId <- forkIO (heartbeatLoop mParentActivity hbStop)
  570           let runWithCatch =
  571                 worker def childSid task hooks
  572                   `catch` \e -> pure (ChildWorkerOutcome
  573                                        (Just (T.pack (show (e :: SomeException))))
  574                                        CerError 0 0 (Just childSid))
  575           mOutcome <- timeout micros runWithCatch
  576           -- Stop the heartbeat.
  577           void (tryPutMVar hbStop ())
  578           killThread hbThreadId
  579           pure (maybe (Left "timeout") Right mOutcome)
  580       end <- getCurrentTime
  581       let dur = realToFrac (end `diffUTCTime` start) :: Double
  582       case mRes of
  583         Left err ->
  584           pure (ChildResult
  585                   { crTaskIndex = idx
  586                   , crStatus = if err == "timeout" then CsTimeout else CsError
  587                   , crSummary = Nothing
  588                   , crExitReason = if err == "timeout" then CerTimeout else CerError
  589                   , crDurationSeconds = dur
  590                   , crSubagentId = subagentId
  591                   , crTokensInput = 0
  592                   , crTokensOutput = 0
  593                   , crToolTrace = []
  594                   , crError = Just (if err == "timeout"
  595                                       then "Subagent timed out after " <> T.pack (show childTimeout) <> "s."
  596                                       else err)
  597                   , crFilesRead = []
  598                   , crFilesWritten = []
  599                   , crChildSession = Nothing
  600                   })
  601         Right outcome -> do
  602           traceList <- readIORef traceRef
  603           filesRead <- readIORef readRef
  604           filesWritten <- readIORef writtenRef
  605           pure (ChildResult
  606                   { crTaskIndex = idx
  607                   , crStatus = childStatusFor (cwoExitReason outcome)
  608                   , crSummary = cwoSummary outcome
  609                   , crExitReason = cwoExitReason outcome
  610                   , crDurationSeconds = dur
  611                   , crSubagentId = subagentId
  612                   , crTokensInput = cwoTokensInput outcome
  613                   , crTokensOutput = cwoTokensOutput outcome
  614                   , crToolTrace = reverse traceList
  615                   , crError = Nothing
  616                   , crFilesRead = reverse filesRead
  617                   , crFilesWritten = reverse filesWritten
  618                   , crChildSession = cwoChildSession outcome
  619                    })

  621 -- | The async delegation runner. Forks one worker per task (respecting
  622 -- 'max_concurrent_children'), registers each child in the 'AgentRuntime'
  623 -- at spawn time, and returns immediately with per-child 'SpawnInfo'.
  624 -- Children run in parallel in forked threads; when a child finishes, its
  625 -- 'ChildResult' is passed to the 'AgentCompletionCallback' (which the
  626 -- opcode layer uses to append the result to the parent's transcript and
  627 -- update the registry).
  628 --
  629 -- The pause check and depth check are synchronous (before any forking);
  630 -- resolve errors and worker completion are async (in the forked threads).
  631 -- Per-child timeout is governed solely by
  632 -- 'dcChildTimeoutSeconds' (default 600s).
  633 runDelegateAsync
  634   :: DelegationConfig
  635   -> SpawnPauseFlag
  636   -> Maybe ParentActivity
  637   -> Int
  638      -- ^ the parent's current delegation depth (0 = top-level parent)
  639   -> DelegateInput
  640   -> (ChildTask -> IO (Either Text (AgentDef, AgentWorkerBuilder)))
  641      -- ^ resolver: look up the def, build the worker. 'Left' = def not
  642      -- found / invalid / depth exceeded. The session id is NOT minted by
  643      -- the resolver — 'spawnOne' mints the session before calling it.
  644   -> AgentCompletionCallback
  645      -- ^ completion callback (called from forked child threads)
  646   -> SpawnCallback
  647      -- ^ spawn callback (called after resolve success, before the worker
  648      -- runs — used to register the child in the 'AgentRuntime').
  649   -> IO SessionId
  650      -- ^ mint a fresh 'SessionId' for each child (called before forking
  651      -- so the session is available in 'SpawnInfo' immediately).
  652   -> IO (Either Text [SpawnInfo])
  653 runDelegateAsync cfg pauseFlag mParentActivity parentDepth input resolveTask callback spawnCallback mintSession = do
  654   paused <- isSpawnPaused pauseFlag
  655   if paused
  656     then pure (Left "Delegation spawning is paused. Clear the pause via the TUI or the AGENT_INTERRUPT RPC before retrying.")
  657     else do
  658       let (maxConc, childTimeout, maxDepth, _orchEnabled) = resolveDelegationConfig cfg
  659       if parentDepth >= maxDepth
  660         then pure (Left ("Delegation depth limit reached (depth=" <> T.pack (show parentDepth)
  661                           <> ", max_spawn_depth=" <> T.pack (show maxDepth)
  662                           <> "). Raise delegation.max_spawn_depth in config.toml if deeper nesting is required (cap: "
  663                           <> T.pack (show maxSpawnDepthCap) <> ")."))
  664         else do
  665           tasks <- case input of
  666             DiSingle t -> pure [t]
  667             DiBatch ts -> pure ts
  668           if null tasks
  669             then pure (Left "No tasks provided.")
  670             else do
  671               spawnInfos <- runBatchAsync maxConc childTimeout tasks
  672               pure (Right spawnInfos)
  673   where
  674     -- Fork all tasks with a concurrency cap. Returns immediately after
  675     -- forking (each forked thread runs the child to completion).
  676     runBatchAsync :: Int -> Double -> [ChildTask] -> IO [SpawnInfo]
  677     runBatchAsync maxConc childTimeout tasks =
  678       case tasks of
  679         [t] -> do
  680           si <- spawnOne 0 t childTimeout Nothing
  681           pure [si]
  682         _   -> do
  683           sem <- newTVarIO maxConc
  684           forM (zip [0 ..] tasks) $ \(idx, task) -> do
  685             spawnOne idx task childTimeout (Just sem)
  686 
  687     -- Spawn one child: resolve synchronously, register via spawnCallback,
  688     -- then fork the worker execution. Returns SpawnInfo immediately.
  689     -- The session is minted INSIDE the semaphore (when present) so that
  690     -- concurrent spawns don't mint in the same microsecond — the random
  691     -- suffix in 'mintSession' makes collisions vanishingly unlikely, but
  692     -- serializing the mint through the semaphore is a belt-and-suspenders
  693     -- guarantee.
  694     spawnOne :: Int -> ChildTask -> Double -> Maybe (TVar Int) -> IO SpawnInfo
  695     spawnOne idx task childTimeout mSem = do
  696       subagentId <- mkSubagentId (ctDefId task)
  697       let micros = round (childTimeout * 1000000) :: Int
  698           mkChild :: IO SpawnInfo
  699           mkChild = do
  700             childSid <- mintSession
  701             start <- getCurrentTime
  702             traceRef     <- newIORef []
  703             readRef     <- newIORef []
  704             writtenRef  <- newIORef []
  705             interruptedRef <- newIORef False
  706             let hooks = ChildRunHooks traceRef readRef writtenRef interruptedRef
  707             mResolve <- resolveTask task
  708             case mResolve of
  709               Left err -> do
  710                 -- Resolve error: fork the error callback (async).
  711                 end <- getCurrentTime
  712                 let dur = realToFrac (end `diffUTCTime` start) :: Double
  713                     result = mkErrorResult idx subagentId dur err Nothing
  714                 void (forkIO (callback result))
  715               Right (def, worker) -> do
  716                 -- Fork the worker execution (async), then register the
  717                 -- child with the forked ThreadId so AGENT_STOP can kill
  718                 -- the correct thread (not the parent's).
  719                 childTid <- forkIO $ do
  720                   case mSem of
  721                     Nothing -> runWorker idx task childTimeout subagentId childSid micros start traceRef readRef writtenRef def worker hooks
  722                     Just sem -> bracketSem sem $
  723                       runWorker idx task childTimeout subagentId childSid micros start traceRef readRef writtenRef def worker hooks
  724                 spawnCallback subagentId def childSid childTid
  725             pure (SpawnInfo subagentId childSid idx)
  726       case mSem of
  727         Nothing -> mkChild
  728         Just sem -> bracketSem sem mkChild
  729 
  730     -- Run the worker to completion (called from a forked thread).
  731     runWorker :: Int -> ChildTask -> Double -> SubagentId -> SessionId -> Int
  732              -> UTCTime -> IORef [ToolTraceEntry] -> IORef [Text] -> IORef [Text]
  733              -> AgentDef -> AgentWorkerBuilder -> ChildRunHooks -> IO ()
  734     runWorker idx task childTimeout subagentId childSid micros start traceRef readRef writtenRef def worker hooks = do
  735       -- Start the heartbeat thread.
  736       hbStop <- newEmptyMVar
  737       hbThreadId <- forkIO (heartbeatLoop mParentActivity hbStop)
  738       let runWithCatch =
  739             worker def childSid task hooks
  740               `catch` \e -> pure (ChildWorkerOutcome
  741                                    (Just (T.pack (show (e :: SomeException))))
  742                                    CerError 0 0 (Just childSid))
  743       mOutcome <- timeout micros runWithCatch
  744       -- Stop the heartbeat.
  745       void (tryPutMVar hbStop ())
  746       killThread hbThreadId
  747       end <- getCurrentTime
  748       let dur = realToFrac (end `diffUTCTime` start) :: Double
  749       case mOutcome of
  750         Nothing -> do
  751           let result = mkTimeoutResult idx subagentId dur childSid childTimeout
  752           callback result
  753         Just outcome -> do
  754           traceList <- readIORef traceRef
  755           filesRead <- readIORef readRef
  756           filesWritten <- readIORef writtenRef
  757           let result = ChildResult
  758                 { crTaskIndex = idx
  759                 , crStatus = childStatusFor (cwoExitReason outcome)
  760                 , crSummary = cwoSummary outcome
  761                 , crExitReason = cwoExitReason outcome
  762                 , crDurationSeconds = dur
  763                 , crSubagentId = subagentId
  764                 , crTokensInput = cwoTokensInput outcome
  765                 , crTokensOutput = cwoTokensOutput outcome
  766                 , crToolTrace = reverse traceList
  767                 , crError = Nothing
  768                 , crFilesRead = reverse filesRead
  769                 , crFilesWritten = reverse filesWritten
  770                 , crChildSession = cwoChildSession outcome
  771                 }
  772           callback result
  773 
  774     -- Acquire / release the counting semaphore around an IO action.
  775     bracketSem :: TVar Int -> IO a -> IO a
  776     bracketSem sem act = bracket acquire release (const act)
  777       where
  778         acquire = atomically $ do
  779           n <- readTVar sem
  780           if n > 0
  781             then writeTVar sem (n - 1)
  782             else retry
  783         release _ = atomically (modifyTVar' sem (+1))

  830 -- | A 30s heartbeat that touches the parent-activity cell while the child
  831 -- runs. Stops when the stop-MVar is filled.
  832 heartbeatLoop :: Maybe ParentActivity -> MVar () -> IO ()
  833 heartbeatLoop mParentActivity stopRef = loop
  834   where
  835     loop = do
  836       stop <- tryTakeMVar stopRef
  837       case stop of
  838         Just ()  -> pure ()
  839         Nothing  -> do
  840           for_ mParentActivity touchParentActivity
  841           threadDelay 30000000  -- 30s
  842           loop

    2 -- | The delegation pipeline — the Seal analog of Hermes' @delegate_task@.
    3 --
    4 -- 'runDelegate' spawns one or more child agents with isolated context, runs each
    5 -- against a goal to completion (synchronously), and returns a structured
    6 -- result per child. The parent blocks until all children finish (or time out).
    7 --
    8 -- Each child gets:
    9 --
   10 --   * a fresh 'SessionId' and its own two-file transcript nested under
   11 --     @\<parent-session\>\/agents\/\<child-id\>@ (the worker-builder opens it);
   12 --   * a focused system prompt built from the def's @adSystem@ + the goal +
   13 --     optional context (the worker-builder assembles it);
   14 --   * a narrowed tool set: the def's @adTools@ allow-list with the delegation
   15 --     blocklist always stripped (no recursive spawning, no def mutation, no
   16 --     lifecycle introspection) — the worker-builder is responsible for
   17 --     applying the blocklist when it constructs the child ISA registry.
   18 --
   19 -- Features ported from @delegate_task@:
   20 --
   21 --   * batch mode ('tasks' array) with a concurrency cap
   22 --     (@delegation.max_concurrent_children@);
   23 --   * depth cap (@delegation.max_spawn_depth@, default 1, clamped to [1,3])
   24 --     + @orchestrator_enabled@ kill switch (the role field is accepted but the
   25 --     wiring layer enforces the effective role when it builds the child's
   26 --     ISA registry);
   27 --   * timeout (@delegation.child_timeout_seconds@, default 600s, floor 30s)
   28 --     via 'System.Timeout.timeout';
   29 --   * a heartbeat thread that touches a parent-activity cell every 30s so a
   30 --     gateway inactivity timeout doesn't fire while the child works;
   31 --   * a spawn-pause flag so an operator can freeze new fan-out without
   32 --     interrupting running children;
   33 --   * per-child provider\/model override via @[delegation] provider/model/base_url@
   34 --     (the worker-builder reads the override and resolves the child provider
   35 --     from it instead of inheriting the parent's);
   36 --   * rich result aggregation: summary, status, exit_reason, tokens,
   37 --     tool_trace, files read/written (the worker reports via the
   38 --     'ChildRunHooks' accumulators).
   39 --
   40 -- Differences from @delegate_task@ (by design):
   41 --
   42 --   * Children are /transcripted/ — each child's conversation + entries land
   43 --     on disk under the parent's @agents\/\<child-id\>@ dir, surviving restart.
   44 --   * The child run is the existing 'Seal.Agent.Loop.runTurn' with the goal as
   45 --     the first user message — there is no separate child AIAgent class.

  116   , dcOrchestratorEnabled :: Maybe Bool

  168 -- | Resolve a loaded 'DelegationConfig' to its effective numeric knobs, with
  169 -- the documented clamps applied. Provider/model/base_url/api_key/api_mode
  170 -- overrides are read directly from the config (they're 'Maybe'-typed).
  171 resolveDelegationConfig :: DelegationConfig
  172                        -> (Int, Double, Int, Bool)
  173 resolveDelegationConfig cfg =
  174   ( maxConc
  175   , childTimeout
  176   , maxDepth
  177   , orchEnabled
  178   )
  179   where
  180     maxConc = max 1 $ fromMaybe defaultMaxConcurrentChildren (dcMaxConcurrentChildren cfg)
  181     childTimeout = max minChildTimeoutSeconds $
  182                      fromMaybe defaultChildTimeoutSeconds (dcChildTimeoutSeconds cfg)
  183     rawDepth = fromMaybe defaultMaxSpawnDepth (dcMaxSpawnDepth cfg)
  184     maxDepth = max minSpawnDepth (min maxSpawnDepthCap rawDepth)
  185     orchEnabled = fromMaybe True (dcOrchestratorEnabled cfg)

  208 -- | Opaque identifier for one live subagent. Smart-constructed from the def
  209 -- id + a short random suffix so multiple concurrent children of the same def
  210 -- don't collide.
  211 newtype SubagentId = SubagentId { unSubagentId :: Text }
  212   deriving stock (Eq, Ord, Show)
  213 
  214 -- | Mint a fresh 'SubagentId' from a def-id text + 8 hex chars. The def id is
  215 -- embedded for debuggability (the TUI can show which def a subagent ran);
  216 -- the random suffix disambiguates concurrent spawns.
  217 mkSubagentId :: Text -> IO SubagentId
  218 mkSubagentId defIdText = do
  219   suffix <- randomRIO (0 :: Int, 0xFFFFFFFF)
  220   pure (SubagentId (defIdText <> "-" <> showHexI32 suffix))
  221 
  222 subagentIdText :: SubagentId -> Text
  223 subagentIdText (SubagentId t) = t
  224 
  225 -- | Show an 'Int' as 8-char zero-padded lowercase hex.
  226 showHexI32 :: Int -> Text
  227 showHexI32 n = T.justifyRight 8 '0' (T.pack (go n))
  228   where
  229     go x
  230       | x <= 0    = "0"
  231       | otherwise = loop x ""
  232     loop y acc
  233       | y == 0    = acc
  234       | otherwise = let (q, r) = y `divMod` 16
  235                         c = if r < 10 then toEnum (r + fromEnum '0')
  236                                       else toEnum (r - 10 + fromEnum 'a')
  237                     in loop q (c : acc)

  348 -- | A worker-builder: given a def + fresh session id + the task + the run
  349 -- hooks, run the child turn to completion and return the outcome.
  350 --
  351 -- The wiring layer (Cli.hs, Channels.Loop.hs, Gateway.Send.hs) closes over
  352 -- the per-turn 'AgentEnv' deps (caps, execBackend, appEnv, eCfg, parent
  353 -- session id, parent depth) and produces the IO action. Keeping this as a
  354 -- parameter decouples the opcode from 'Seal.Agent.Loop.runTurn' / provider
  355 -- resolution so the opcode is unit-testable with a fake worker.
  356 type AgentWorkerBuilder
  357   =  AgentDef
  358   -> SessionId
  359   -> ChildTask
  360   -> ChildRunHooks
  361   -> IO ChildWorkerOutcome
  362 
  363 -- | Thread-safe accumulators the worker-builder writes to during the child
  364 -- run. 'runDelegate' reads them after the worker returns to populate the
  365 -- 'ChildResult'. The worker appends to these via 'atomicModifyIORef'' (the
  366 -- IORefs are single-writer-per-child, but atomicModify is cheap and safe).
  367 data ChildRunHooks = ChildRunHooks
  368   { crhToolTrace    :: IORef [ToolTraceEntry]
  369     -- ^ Append-only tool-trace accumulator.
  370   , crhFilesRead    :: IORef [Text]
  371     -- ^ Append-only file-read accumulator.
  372   , crhFilesWritten :: IORef [Text]
  373     -- ^ Append-only file-written accumulator.
  374   , crhInterrupted  :: IORef Bool
  375     -- ^ Polled by the worker between iterations; when 'True' the worker
  376     -- should stop at its next turn boundary and return 'CwoInterrupted'.
  377   }
  378 
  379 -- | What the worker-builder reports back to 'runDelegate'. The builder does
  380 -- NOT fill @crTaskIndex@ / @crSubagentId@ / @crDurationSeconds@ — those are
  381 -- filled in by the orchestrator. It DOES fill in the summary, tokens, exit
  382 -- reason, and child session id it observed.
  383 data ChildWorkerOutcome = ChildWorkerOutcome
  384   { cwoSummary      :: !(Maybe Text)
  385     -- ^ The final text response from the child (or an error message).
  386   , cwoExitReason   :: !ChildExitReason
  387     -- ^ How the child's run ended (from the worker's perspective).
  388   , cwoTokensInput  :: !Int
  389   , cwoTokensOutput :: !Int
  390   , cwoChildSession :: !(Maybe SessionId)
  391     -- ^ The 'SessionId' the worker ran under (so the parent can look up the
  392     -- child's transcript later).
  393   }

  399 -- | A process-global flag that, when paused, makes 'runDelegate' reject new
  400 -- spawns with a structured error. Running children keep running; only NEW
  401 -- 'runDelegate' calls fail fast until 'setSpawnPaused' clears the flag.
  402 newtype SpawnPauseFlag = SpawnPauseFlag (TVar Bool)
  403 
  404 newSpawnPauseFlag :: IO SpawnPauseFlag
  405 newSpawnPauseFlag = SpawnPauseFlag <$> newTVarIO False
  406 
  407 setSpawnPaused :: SpawnPauseFlag -> Bool -> IO Bool
  408 setSpawnPaused (SpawnPauseFlag tv) paused = do
  409   atomically (writeTVar tv paused)
  410   pure paused
  411 
  412 isSpawnPaused :: SpawnPauseFlag -> IO Bool
  413 isSpawnPaused (SpawnPauseFlag tv) = readTVarIO tv

  415 -- ---------------------------------------------------------------------------
  416 -- Parent activity heartbeat
  417 -- ---------------------------------------------------------------------------
  418 
  419 -- | A mutable cell the parent's gateway/loop touches to record \"the parent
  420 -- is alive and working\". 'runDelegate' starts a heartbeat thread that
  421 -- periodically touches this while a child runs, so the parent's gateway
  422 -- inactivity timeout doesn't fire during a long delegation.
  423 newtype ParentActivity = ParentActivity (MVar ())
  424 
  425 newParentActivity :: IO ParentActivity
  426 newParentActivity = ParentActivity <$> newMVar ()
  427 
  428 -- | Touch the parent-activity cell (idempotent, never blocks).
  429 touchParentActivity :: ParentActivity -> IO ()
  430 touchParentActivity (ParentActivity mv) =
  431   void (tryPutMVar mv () >> tryTakeMVar mv)

  433 -- ---------------------------------------------------------------------------
  434 -- Run
  435 -- ---------------------------------------------------------------------------
  436 
  437 -- | The input to 'runDelegate' — one task to run (single mode) or many
  438 -- (batch mode). The caller (the AGENT_START opcode) normalizes the model's
  439 -- input to this before calling.
  440 data DelegateInput
  441   = DiSingle !ChildTask
  442   | DiBatch ![ChildTask]
  443 
  444 -- | The top-level delegation runner. Spawns one or more child agents, runs
  445 -- each against its goal to completion (synchronously), and returns a
  446 -- 'ChildResult' per task. The parent blocks until all children finish (or
  447 -- time out). See the module header for the feature list.
  448 runDelegate
  449   :: DelegationConfig
  450   -> SpawnPauseFlag
  451   -> Maybe ParentActivity
  452   -> Int
  453      -- ^ the parent's current delegation depth (0 = top-level parent)
  454   -> DelegateInput
  455   -> (ChildTask -> IO (Either Text (AgentDef, AgentWorkerBuilder, SessionId)))
  456      -- ^ resolver: look up the def, build the worker, mint a fresh child
  457      -- 'SessionId'. 'Left' = def not found / invalid / depth would be
  458      -- exceeded. Per-task so the caller can cache or re-resolve.
  459   -> IO (Either Text [ChildResult])
  460 runDelegate cfg pauseFlag mParentActivity parentDepth input resolveTask = do
  461   paused <- isSpawnPaused pauseFlag
  462   if paused
  463     then pure (Left "Delegation spawning is paused. Clear the pause via the TUI or the AGENT_INTERRUPT RPC before retrying.")
  464     else do
  465       let (maxConc, childTimeout, maxDepth, _orchEnabled) = resolveDelegationConfig cfg
  466       if parentDepth >= maxDepth
  467         then pure (Left ("Delegation depth limit reached (depth=" <> T.pack (show parentDepth)
  468                           <> ", max_spawn_depth=" <> T.pack (show maxDepth)
  469                           <> "). Raise delegation.max_spawn_depth in config.toml if deeper nesting is required (cap: "
  470                           <> T.pack (show maxSpawnDepthCap) <> ")."))
  471         else do
  472           tasks <- case input of
  473             DiSingle t -> pure [t]
  474             DiBatch ts -> pure ts
  475           if null tasks
  476             then pure (Left "No tasks provided.")
  477             else do
  478               when (length tasks > maxConc) $
  479                 -- We don't reject oversize batches; the semaphore caps
  480                 -- parallelism. (delegate_task rejects; we cap.)
  481                 pure ()
  482               runBatch maxConc childTimeout tasks
  483   where
  484     -- Run all tasks with a concurrency cap. Single-task short-cut avoids the
  485     -- thread-pool overhead.
  486     runBatch :: Int -> Double -> [ChildTask] -> IO (Either Text [ChildResult])
  487     runBatch maxConc childTimeout tasks =
  488       case tasks of
  489         [t] -> do
  490           r <- runOne 0 t childTimeout
  491           pure (Right [r])
  492         _   -> do
  493           sem <- newTVarIO maxConc
  494           mvars <- forM (zip [0 ..] tasks) $ \(idx, task) -> do
  495             resMVar <- newEmptyMVar
  496             void (forkIO $
  497               bracketSem sem $
  498                 runOne idx task childTimeout >>= putMVar resMVar)
  499             pure resMVar
  500           results <- forM mvars readMVar
  501           pure (Right results)
  502 
  503     -- Acquire / release the counting semaphore around an IO action. The
  504     -- semaphore is a 'TVar Int' initialized to @maxConc@: each acquire
  505     -- atomically decrements (blocking if zero via 'retry'), each release
  506     -- atomically increments. This allows up to @maxConc@ concurrent holders,
  507     -- unlike the previous 'MVar'-based approach which was a binary semaphore
  508     -- (mutex) that serialized all children regardless of @maxConc@.
  509     bracketSem :: TVar Int -> IO a -> IO a
  510     bracketSem sem act = bracket acquire release (const act)
  511       where
  512         acquire = atomically $ do
  513           n <- readTVar sem
  514           if n > 0
  515             then writeTVar sem (n - 1)
  516             else retry
  517         release _ = atomically (modifyTVar' sem (+1))
  518 
  519     -- Run a single task to completion with a hard timeout and a heartbeat
  520     -- thread that touches the parent-activity cell.
  521     runOne :: Int -> ChildTask -> Double -> IO ChildResult
  522     runOne idx task childTimeout = do
  523       subagentId <- mkSubagentId (ctDefId task)
  524       let micros = round (childTimeout * 1000000) :: Int
  525       start <- getCurrentTime
  526       -- Thread-safe accumulators the worker writes to.
  527       traceRef     <- newIORef []
  528       readRef     <- newIORef []
  529       writtenRef  <- newIORef []
  530       interruptedRef <- newIORef False
  531       let hooks = ChildRunHooks traceRef readRef writtenRef interruptedRef
  532       mRes <- resolveTask task >>= \case
  533         Left err -> pure (Left err)
  534         Right (def, worker, childSid) -> do
  535           -- Start the heartbeat thread.
  536           hbStop <- newEmptyMVar
  537           hbThreadId <- forkIO (heartbeatLoop mParentActivity hbStop)
  538           let runWithCatch =
  539                 worker def childSid task hooks
  540                   `catch` \e -> pure (ChildWorkerOutcome
  541                                        (Just (T.pack (show (e :: SomeException))))
  542                                        CerError 0 0 (Just childSid))
  543           mOutcome <- timeout micros runWithCatch
  544           -- Stop the heartbeat.
  545           void (tryPutMVar hbStop ())
  546           killThread hbThreadId
  547           pure (maybe (Left "timeout") Right mOutcome)
  548       end <- getCurrentTime
  549       let dur = realToFrac (end `diffUTCTime` start) :: Double
  550       case mRes of
  551         Left err ->
  552           pure (ChildResult
  553                   { crTaskIndex = idx
  554                   , crStatus = if err == "timeout" then CsTimeout else CsError
  555                   , crSummary = Nothing
  556                   , crExitReason = if err == "timeout" then CerTimeout else CerError
  557                   , crDurationSeconds = dur
  558                   , crSubagentId = subagentId
  559                   , crTokensInput = 0
  560                   , crTokensOutput = 0
  561                   , crToolTrace = []
  562                   , crError = Just (if err == "timeout"
  563                                       then "Subagent timed out after " <> T.pack (show childTimeout) <> "s."
  564                                       else err)
  565                   , crFilesRead = []
  566                   , crFilesWritten = []
  567                   , crChildSession = Nothing
  568                   })
  569         Right outcome -> do
  570           traceList <- readIORef traceRef
  571           filesRead <- readIORef readRef
  572           filesWritten <- readIORef writtenRef
  573           pure (ChildResult
  574                   { crTaskIndex = idx
  575                   , crStatus = childStatusFor (cwoExitReason outcome)
  576                   , crSummary = cwoSummary outcome
  577                   , crExitReason = cwoExitReason outcome
  578                   , crDurationSeconds = dur
  579                   , crSubagentId = subagentId
  580                   , crTokensInput = cwoTokensInput outcome
  581                   , crTokensOutput = cwoTokensOutput outcome
  582                   , crToolTrace = reverse traceList
  583                   , crError = Nothing
  584                   , crFilesRead = reverse filesRead
  585                   , crFilesWritten = reverse filesWritten
  586                   , crChildSession = cwoChildSession outcome
  587                   })
  588 
  589 -- | Map a worker-reported exit reason to the aggregate status.
  590 childStatusFor :: ChildExitReason -> ChildStatus
  591 childStatusFor = \case
  592   CerCompleted    -> CsCompleted
  593   CerMaxIterations -> CsFailed
  594   CerTimeout      -> CsTimeout
  595   CerInterrupted  -> CsInterrupted
  596   CerError        -> CsError

  598 -- | A 30s heartbeat that touches the parent-activity cell while the child
  599 -- runs. Stops when the stop-MVar is filled.
  600 heartbeatLoop :: Maybe ParentActivity -> MVar () -> IO ()
  601 heartbeatLoop mParentActivity stopRef = loop
  602   where
  603     loop = do
  604       stop <- tryTakeMVar stopRef
  605       case stop of
  606         Just ()  -> pure ()
  607         Nothing  -> do
  608           for_ mParentActivity touchParentActivity
  609           threadDelay 30000000  -- 30s
  610           loop

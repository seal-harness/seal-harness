   41 -- Auth types ---------------------------------------------------------------
   42 
   43 -- | How the provider authenticates. OAuth carries a mutable token cell plus
   44 -- injected refresh/persist actions so 'ensureFresh' can rotate tokens in place.
   45 data AnthropicAuth
   46   = AuthApiKey ApiKey
   47   | AuthOAuth  OAuthSession
   48 
   49 data OAuthSession = OAuthSession
   50   { osTokens :: IORef OAuthTokens
   51   , osRefresh :: OAuthTokens -> IO (Either Text OAuthTokens)
   52   , osPersist :: OAuthTokens -> IO ()
   53   }
   54 
   55 data Anthropic = Anthropic
   56   { anModel :: ModelId
   57   , anManager :: Manager
   58   , anAuth :: AnthropicAuth
   59   }
   60 
   61 mkAnthropic :: Manager -> ApiKey -> ModelId -> Anthropic
   62 mkAnthropic mgr key model = Anthropic model mgr (AuthApiKey key)
   63 
   64 mkAnthropicOAuth :: Manager -> OAuthSession -> ModelId -> Anthropic
   65 mkAnthropicOAuth mgr sess model = Anthropic model mgr (AuthOAuth sess)

   67 -- Header builders ----------------------------------------------------------
   68 
   69 -- | Refresh this many seconds BEFORE the real expiry, so a token cannot lapse
   70 -- mid-request.
   71 refreshSkew :: NominalDiffTime
   72 refreshSkew = 60
   73 
   74 apiKeyHeaders :: ByteString -> RequestHeaders
   75 apiKeyHeaders keyBytes =
   76   [ ("content-type",      "application/json")
   77   , ("anthropic-version", TE.encodeUtf8 anthropicVersion)
   78   , ("x-api-key",         keyBytes)
   79   ]
   80 
   81 oauthHeaders :: ByteString -> RequestHeaders
   82 oauthHeaders accessBytes =
   83   [ ("content-type",      "application/json")
   84   , ("anthropic-version", TE.encodeUtf8 anthropicVersion)
   85   , ("anthropic-beta",    TE.encodeUtf8 anthropicBeta)
   86   , ("authorization",     "Bearer " <> accessBytes)
   87   ]
   88 
   89 -- | Return current tokens, refreshing first if within 'refreshSkew' of expiry.
   90 -- On refresh success, update the in-memory ref AND persist.
   91 ensureFresh :: OAuthSession -> IO (Either Text OAuthTokens)
   92 ensureFresh sess = do
   93   toks <- readIORef (osTokens sess)
   94   now  <- getCurrentTime
   95   if otExpiresAt toks <= addUTCTime refreshSkew now
   96     then do
   97       r <- osRefresh sess toks
   98       case r of
   99         Left e    -> pure (Left e)
  100         Right new -> do
  101           writeIORef (osTokens sess) new
  102           osPersist sess new
  103           pure (Right new)
  104     else pure (Right toks)

  106 -- Pure request mapping -----------------------------------------------------

  108 encodeRequest :: CompletionRequest -> Value
  109 encodeRequest cr = object $
  110   [ "model"      .= crModel cr
  111   , "max_tokens" .= crMaxTokens cr
  112   , "messages"   .= map encMsg (crMessages cr)
  113   ]
  114   <> maybe [] (\s -> ["system" .= s]) (crSystem cr)
  115   <> ["tools" .= map encTool (crTools cr) | not (null (crTools cr))]
  116 
  117 encMsg :: Message -> Value
  118 encMsg (Message r blocks) = object
  119   [ "role" .= roleText r, "content" .= map encBlock blocks ]
  120 
  121 roleText :: Role -> Text
  122 roleText User = "user"
  123 roleText Assistant = "assistant"
  124 
  125 encBlock :: ContentBlock -> Value
  126 encBlock (CbText t) = object ["type" .= ("text" :: Text), "text" .= t]
  127 encBlock (CbToolUse (ToolCallId i) (OpName n) inp) =
  128   object ["type" .= ("tool_use" :: Text), "id" .= i, "name" .= n, "input" .= inp]
  129 encBlock (CbToolResult (ToolCallId i) parts isErr) =
  130   object [ "type"        .= ("tool_result" :: Text)
  131          , "tool_use_id" .= i
  132          , "is_error"    .= isErr
  133          , "content"     .= [object ["type" .= ("text" :: Text), "text" .= t] | TrpText t <- parts]
  134          ]
  135 
  136 encTool :: ToolDefinition -> Value
  137 encTool (ToolDefinition (OpName n) d sch) =
  138   -- Omit input_schema entirely when it's the on-demand stub. Anthropic
  139   -- normally requires input_schema on every tool; sending it only when the
  140   -- model has a real schema to follow keeps the stub tools (the common case
  141   -- under on-demand mode) at zero schema-token cost. The model retrieves a
  142   -- tool's real schema via OPCODE_DESCRIBE before calling it.
  143   if sch == stubSchema
  144     then object ["name" .= n, "description" .= d]
  145     else object ["name" .= n, "description" .= d, "input_schema" .= sch]
  146 
  147 -- Pure response mapping ----------------------------------------------------
  148 
  149 decodeResponse :: Value -> Either Text CompletionResponse
  150 decodeResponse = mapLeft T.pack . parseEither parseResp
  151   where mapLeft f = either (Left . f) Right
  152 
  153 parseResp :: Value -> Parser CompletionResponse
  154 parseResp = withObject "response" $ \o -> do
  155   blocks <- o .: "content" >>= mapM parseBlock
  156   stop   <- parseStop <$> o .: "stop_reason"
  157   usageV <- o .: "usage"
  158   uin    <- usageV .: "input_tokens"
  159   uout   <- usageV .: "output_tokens"
  160   pure (CompletionResponse blocks stop (Usage uin uout))
  161 
  162 parseBlock :: Value -> Parser ContentBlock
  163 parseBlock = withObject "block" $ \o -> do
  164   ty <- o .: "type" :: Parser Text
  165   case ty of
  166     "text"     -> CbText <$> o .: "text"
  167     "tool_use" -> CbToolUse . ToolCallId <$> o .: "id"
  168                   <*> (OpName <$> o .: "name")
  169                   <*> o .: "input"
  170     other      -> fail ("unknown content block type: " <> T.unpack other)
  171 
  172 parseStop :: Text -> StopReason
  173 parseStop "end_turn"   = StopEnd
  174 parseStop "tool_use"   = StopToolUse
  175 parseStop "max_tokens" = StopMaxTokens
  176 parseStop other        = StopOther other

  178 -- HTTP round-trip ----------------------------------------------------------
  179 
  180 -- | POST /v1/messages with the given auth headers; decode the response, or
  181 -- return the HTTP status + response body (the body carries no secret).
  182 sendMessages
  183   :: Manager -> RequestHeaders -> CompletionRequest
  184   -> IO (Either Text CompletionResponse)
  185 sendMessages mgr hdrs cr = do
  186   let body = encode (encodeRequest cr)
  187   initReq <- parseRequest "POST https://api.anthropic.com/v1/messages"
  188   let req = initReq { requestBody = RequestBodyLBS body, requestHeaders = hdrs }
  189   result <- try (httpLbs req mgr)
  190   case result of
  191     Left (_ :: HttpException) ->
  192       pure (Left "HTTP request to Anthropic failed (connection or transport error)")
  193     Right resp -> do
  194       let code = statusCode (responseStatus resp)
  195       if code >= 200 && code <= 299
  196         then pure $ case eitherDecode (responseBody resp) of
  197           Left e  -> Left (T.pack e)
  198           Right v -> decodeResponse v
  199         else pure $ Left $ httpErrorText code
  200           (TE.decodeUtf8With TEE.lenientDecode (BL.toStrict (responseBody resp)))
  201 
  202 -- | Render a non-2xx Anthropic response into a user-facing message. HTTP 429
  203 -- gets a friendlier hint: the subscription backend's rate-limit body is terse
  204 -- and unhelpful (e.g. @{"error":{"type":"rate_limit_error","message":"Error"}}@),
  205 -- so we explain the likely cause and next steps instead of echoing it. Every
  206 -- other status keeps the full response body, which is diagnostic and key-safe.
  207 httpErrorText :: Int -> Text -> Text
  208 httpErrorText 429 _ =
  209   "Anthropic rate limit (HTTP 429) — the account behind this credential is being \
  210   \throttled; a subscription usage cap is the common cause. Wait a moment and \
  211   \retry, or switch models with /model use."
  212 httpErrorText code body =
  213   "Anthropic API returned HTTP " <> T.pack (show code) <> ": " <> body

  215 -- Provider instance --------------------------------------------------------
  216 
  217 instance Provider Anthropic where
  218   listModels a = pure (Right [anModel a])
  219   complete a cr = case anAuth a of
  220     AuthApiKey key ->
  221       withApiKey key $ \keyBytes ->
  222         sendMessages (anManager a) (apiKeyHeaders keyBytes) cr
  223     AuthOAuth sess -> do
  224       ef <- ensureFresh sess
  225       case ef of
  226         Left e     -> pure (Left e)
  227         Right toks ->
  228           withBearerToken (otAccess toks) $ \accessBytes ->
  229             sendMessages (anManager a) (oauthHeaders accessBytes) cr

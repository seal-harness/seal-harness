    1 -- | The wire snapshot for the WS @lists@ frame and the REST @GET /api/lists@
    2 -- endpoint. Carries the partitioned session lists (mutually exclusive by
    3 -- construction via 'partitionSessions') plus the set of sessions currently
    4 -- in a @thinking@ turn (so a freshly-connected web client can hydrate its
    5 -- sidebar without waiting for the next harness-status event). The WS frame
    6 -- wraps this with @{"type": "lists", ...}@ (added by the broadcast path);
    7 -- the REST body is the bare record (no @type@ field).
    8 --
    9 -- The 'ListsSnapshotWire' type lives in 'Seal.Gateway.Types.ListsSnapshot'
   10 -- (the 'seal-gateway-types' library stanza). Only the builder
   11 -- ('buildListsSnapshot') stays here — it depends on 'TabsHandle',
   12 -- 'SealPaths', and 'SessionStore' (server-internal types).
   13 --
   14 -- Takes 'TabsHandle' + 'SealPaths' directly (NOT 'ApiDeps') so this module
   15 -- does NOT import 'Seal.Gateway.API' — avoids a source-level import cycle
   16 -- ('Seal.Gateway.API' imports this module for the /api/lists route).
   17 module Seal.Gateway.ListsSnapshot
   18   ( ListsSnapshotWire (..)
   19   , buildListsSnapshot
   20   ) where
   21 
   22 import Data.Set (Set)
   23 import qualified Data.Set as Set

   34 -- | Build the partitioned snapshot. Takes the components directly (not
   35 -- 'ApiDeps') so this module stays free of a cycle with 'Seal.Gateway.API'.
   36 -- The @thinking@ set is the broker's current in-memory view of which
   37 -- sessions are mid-turn, so a freshly-connected web client can hydrate
   38 -- its sidebar (the live activity stream only carries forward from here).
   39 buildListsSnapshot :: TabsHandle -> SealPaths -> Set SessionId -> IO ListsSnapshotWire
   40 buildListsSnapshot tabsH paths thinkingSids = do
   41   tl <- snapshotTabs tabsH
   42   let tabsJson = map tabToJson (tlTabs tl)
   43   recent   <- listSessions paths
   44   archived <- listArchivedSessions paths
   45   let ps = partitionSessions tl recent archived
   46   recentJson   <- mapM (sessionInfoJsonWithSnippet paths) (psRecentSessions ps)
   47   archivedJson <- mapM (sessionInfoJsonWithSnippet paths) (psArchivedSessions ps)
   48   tabbedJson   <- mapM (sessionInfoJsonWithSnippet paths) (psTabSessions ps)
   49   pure ListsSnapshotWire
   50     { lswTabs = tabsJson
   51     , lswRecentSessions = recentJson
   52     , lswArchivedSessions = archivedJson
   53     , lswTabSessions = tabbedJson
   54     , lswThinkingSessionIds = map sessionIdText (Set.toList thinkingSids)
   55     }

    2 -- | The streaming progress manager for chat channels. When enabled, the
    3 -- agent loop routes text deltas and tool-call notifications through this
    4 -- module, which sends progressive edits to the chat platform (Telegram via
    5 -- @editMessageText@, Signal via @send@ with @editTimestamp@). Both
    6 -- platforms support message editing, so the same edit-based algorithm
    7 -- works for both — the message identifier is an opaque 'Text' to the
    8 -- caller (a Telegram @message_id@ string or a Signal @timestamp@ string).
    9 --
   10 -- The module is in the library (not a channel-specific sub-module) because
   11 -- the streaming logic is platform-agnostic; only the 'ChannelHandle' /
   12 -- transport methods differ. Mirrors Hermes' @GatewayStreamConsumer@ but
   13 -- adapted to Seal's @ReaderT AppEnv IO@ + handle pattern.
   14 module Seal.Channels.StreamProgress

   17   , resolveStreamProgressConfig

   44 -- | Configuration for the stream progress manager. Loaded from the
   45 -- @[chat_streaming]@ section of @config.toml@. When @spcEnabled@ is
   46 -- 'False', the manager is inert (the existing behavior: no streaming,
   47 -- no tool progress, final text sent once via 'replyFanout').
   48 data StreamProgressConfig = StreamProgressConfig
   49   { spcEnabled         :: !Bool
   50     -- ^ Master switch. 'False' = existing behavior (no streaming).
   51   , spcToolProgress    :: !Bool
   52     -- ^ Send tool-call notifications as an editable progress bubble.
   53   , spcTextStreaming   :: !Bool
   54     -- ^ Progressive text edits as tokens arrive.
   55   , spcEditIntervalMs  :: !Int
   56     -- ^ Minimum milliseconds between edits (rate limiting).
   57   , spcBufferThreshold :: !Int
   58     -- ^ Codepoints accumulated before forcing an edit (debounce).
   59   , spcCursor          :: !Text
   60     -- ^ Cursor character appended to intermediate edits (removed on
   61     -- the final edit). Default: @▉@ (U+2589 LEFT ONE QUARTER BLOCK).
   62   } deriving stock (Eq, Show)
   63 
   64 -- | The default config: disabled, with sensible defaults for the other
   65 -- fields. When the operator enables @spcEnabled@, the rest of the fields
   66 -- are already populated.
   67 instance Default StreamProgressConfig where
   68   def = StreamProgressConfig
   69     { spcEnabled         = False
   70     , spcToolProgress    = True
   71     , spcTextStreaming   = True
   72     , spcEditIntervalMs  = 1500
   73     , spcBufferThreshold = 80
   74     , spcCursor          = "\x2589"
   75     }
   76 
   77 -- | The canonical default for re-export. Same as 'def'.
   78 defaultStreamProgressConfig :: StreamProgressConfig
   79 defaultStreamProgressConfig = def

   81 -- | Resolve an optional 'StreamProgressConfig' from the config file into a
   82 -- fully-populated one. 'Nothing' (the @[chat_streaming]@ section is absent)
   83 -- returns the disabled default. A present-but-partial section fills
   84 -- missing fields from 'def'. Pure.
   85 resolveStreamProgressConfig :: Maybe StreamProgressConfig -> StreamProgressConfig
   86 resolveStreamProgressConfig = fromMaybe def

   88 -- | The mutable streaming state for one turn. Lives in 'IORef's so the
   89 -- agent loop can call 'onToolCall' / 'onTextDelta' / 'finalizeText'
   90 -- without passing the state around.
   91 data StreamProgress = StreamProgress
   92   { spConfig        :: !StreamProgressConfig
   93   , spHandle        :: !ChannelHandle
   94   , spTextMsgId     :: !(IORef (Maybe Text))
   95     -- ^ The message id of the current text bubble ('Nothing' = no
   96     -- message sent yet for this segment).
   97   , spToolMsgId     :: !(IORef (Maybe Text))
   98     -- ^ The message id of the current tool-progress bubble.
   99   , spAccumulated   :: !(IORef Text)
  100     -- ^ Accumulated text deltas for the current text segment.
  101   , spToolLines     :: !(IORef [Text])
  102     -- ^ Accumulated tool-call lines for the current tool bubble.
  103   , spLastEdit      :: !(IORef (Maybe UTCTime))
  104     -- ^ Wall-clock time of the last edit (for rate limiting).
  105   , spEditSupported :: !(IORef Bool)
  106     -- ^ 'False' once edits start failing (fall back to new messages).
  107   , spSecretOps     :: !(Set OpName)
  108     -- ^ Opcodes whose input must be redacted from tool-progress lines.
  109   }

  111 -- | Create a new 'StreamProgress' for a turn. The 'secretOps' set is the
  112 -- set of opcode names whose input must not be shown in tool-progress
  113 -- messages (e.g. @SECRET_GET@). The caller passes it from the registry's
  114 -- 'secretOpcodes'.
  115 newStreamProgress :: StreamProgressConfig -> ChannelHandle -> Set OpName -> IO StreamProgress
  116 newStreamProgress cfg h secretOps = do
  117   textMsgId  <- newIORef Nothing
  118   toolMsgId  <- newIORef Nothing
  119   accum      <- newIORef ""
  120   toolLns    <- newIORef []
  121   lastEdit   <- newIORef Nothing
  122   editOk     <- newIORef True
  123   pure StreamProgress
  124     { spConfig        = cfg
  125     , spHandle        = h
  126     , spTextMsgId     = textMsgId
  127     , spToolMsgId     = toolMsgId
  128     , spAccumulated   = accum
  129     , spToolLines     = toolLns
  130     , spLastEdit      = lastEdit
  131     , spEditSupported = editOk
  132     , spSecretOps     = secretOps
  133     }

  135 -- | Called when the agent is about to dispatch a tool call. Sends or
  136 -- edits the tool-progress bubble with the new tool line appended. When
  137 -- the config has @spcToolProgress = False@ or @spcEnabled = False@, this
  138 -- is a no-op. IO.
  139 onToolCall :: StreamProgress -> OpName -> Text -> IO ()
  140 onToolCall sp opName input =
  141   when (spcToolProgress cfg && spcEnabled cfg) $ do
  142     let line = formatToolLine (spSecretOps sp) opName input
  143     currentLines <- readIORef (spToolLines sp)
  144     let newLines = currentLines <> [line]
  145     writeIORef (spToolLines sp) newLines
  146     msgId <- readIORef (spToolMsgId sp)
  147     let content = T.intercalate "\n" newLines
  148     case msgId of
  149       Nothing -> do
  150         mId <- chSendWithId (spHandle sp) content
  151         case mId of
  152           Just id' -> writeIORef (spToolMsgId sp) (Just id')
  153           Nothing  -> pure ()
  154       Just id' -> do
  155         editOk <- editOrFallback sp id' content
  156         unless editOk $ writeIORef (spEditSupported sp) False
  157   where cfg = spConfig sp
  158 
  159 -- | Called for each text delta from the provider stream. Accumulates
  160 -- the text and sends/edits the text bubble when the rate-limit /
  161 -- threshold conditions are met. IO.
  162 onTextDelta :: StreamProgress -> Text -> IO ()
  163 onTextDelta sp delta =
  164   when (spcTextStreaming cfg && spcEnabled cfg) $ do
  165     accum <- readIORef (spAccumulated sp)
  166     let accum' = accum <> delta
  167     writeIORef (spAccumulated sp) accum'
  168     now <- getCurrentTime
  169     mLastEdit <- readIORef (spLastEdit sp)
  170     msgId <- readIORef (spTextMsgId sp)
  171     when (shouldEdit cfg now mLastEdit (T.length accum')) $ do
  172       let content = addCursor cfg accum'
  173       case msgId of
  174         Nothing -> do
  175           mId <- chSendWithId (spHandle sp) content
  176           case mId of
  177             Just id' -> do
  178               writeIORef (spTextMsgId sp) (Just id')
  179               writeIORef (spLastEdit sp) (Just now)
  180             Nothing  -> pure ()
  181         Just id' -> do
  182           editOk <- editOrFallback sp id' content
  183           when editOk $ writeIORef (spLastEdit sp) (Just now)
  184   where cfg = spConfig sp

  186 -- | Finalize the text stream: edit the message without the cursor. If
  187 -- the edit fails, send the full text as a new message. Returns 'True' if
  188 -- the final text was delivered (via edit or new message), 'False' if
  189 -- nothing was sent (the caller should fall back to 'replyFanout'). IO.
  190 finalizeText :: StreamProgress -> Text -> IO Bool
  191 finalizeText sp finalText =
  192   if spcTextStreaming cfg && spcEnabled cfg
  193     then do
  194       accum <- readIORef (spAccumulated sp)
  195       let text = if T.null accum then finalText else accum
  196       msgId <- readIORef (spTextMsgId sp)
  197       case msgId of
  198         Nothing -> do
  199           mId <- chSendWithId (spHandle sp) text
  200           pure (isJust mId)
  201         Just id' -> do
  202           editOk <- editOrFallback sp id' text
  203           if editOk
  204             then pure True
  205             else do
  206               mId <- chSendWithId (spHandle sp) text
  207               pure (isJust mId)
  208     else pure False
  209   where cfg = spConfig sp

  211 -- | Signal a segment break: the current text message is finalized (edit
  212 -- without cursor) and the state is reset so the next text delta starts a
  213 -- new message below any tool-progress messages. IO.
  214 --
  215 -- The tool-progress bubble state ('spToolMsgId', 'spToolLines') is
  216 -- intentionally NOT reset: the tool bubble persists across the entire
  217 -- turn so all tool calls edit the same message, avoiding notification
  218 -- spam (one message per tool call).
  219 segmentBreak :: StreamProgress -> IO ()
  220 segmentBreak sp =
  221   when (spcTextStreaming cfg && spcEnabled cfg) $ do
  222     msgId <- readIORef (spTextMsgId sp)
  223     accum <- readIORef (spAccumulated sp)
  224     case (msgId, T.null accum) of
  225       (Just id', False) -> voidEditOrFallback sp id' accum
  226       _                 -> pure ()
  227     writeIORef (spTextMsgId sp) Nothing
  228     writeIORef (spAccumulated sp) ""
  229   where cfg = spConfig sp
  230 
  231 -- | Edit a message via 'chEditMessage'. Returns 'True' on success,
  232 -- 'False' if editing is not supported or the edit fails. IO.
  233 editOrFallback :: StreamProgress -> Text -> Text -> IO Bool
  234 editOrFallback sp msgId content =
  235   case chEditMessage (spHandle sp) of
  236     Just editFn -> editFn msgId content
  237     Nothing     -> pure False
  238 
  239 -- | Like 'editOrFallback' but discards the result. IO.
  240 voidEditOrFallback :: StreamProgress -> Text -> Text -> IO ()
  241 voidEditOrFallback sp msgId content =
  242   void (editOrFallback sp msgId content)

  244 -- | Format a tool-call progress line. Shows the opcode name and a
  245 -- truncated input. The opcode name is prefixed with a per-opcode emoji
  246 -- (see 'opEmoji'). For secret-bearing opcodes (in the 'secretOps' set),
  247 -- the input is replaced with @"<redacted>"@. Pure.
  248 formatToolLine :: Set OpName -> OpName -> Text -> Text
  249 formatToolLine secretOps (OpName name) input =
  250   opEmoji (OpName name) <> " " <> name <> " " <> inputDisplay
  251   where
  252     inputDisplay
  253       | OpName name `Set.member` secretOps = "<redacted>"
  254       | otherwise = truncateInput input
  255     truncateInput t =
  256       if T.length t > 120
  257         then T.take 120 t <> "..."
  258         else t

  302 -- | Should the manager send an edit now? Returns 'True' when enough
  303 -- time has passed since the last edit, or the buffer threshold is
  304 -- exceeded. Pure (takes the current time as an argument).
  305 shouldEdit :: StreamProgressConfig -> UTCTime -> Maybe UTCTime -> Int -> Bool
  306 shouldEdit cfg now mLastEdit accumLen =
  307   accumLen >= spcBufferThreshold cfg || timeElapsed
  308   where
  309     timeElapsed = case mLastEdit of
  310       Nothing -> True
  311       Just lastEdit ->
  312         diffUTCTime now lastEdit * 1000
  313           >= fromIntegral (spcEditIntervalMs cfg)
  314 
  315 -- | Append the cursor to the text for intermediate edits. Pure.
  316 addCursor :: StreamProgressConfig -> Text -> Text
  317 addCursor cfg text =
  318   if T.null (spcCursor cfg) then text else text <> spcCursor cfg
  319 
  320 -- | Strip the cursor from the text for the final edit. Pure.
  321 stripCursor :: StreamProgressConfig -> Text -> Text
  322 stripCursor cfg text =
  323   if T.null (spcCursor cfg)
  324     then text
  325     else fromMaybe text (T.stripSuffix (spcCursor cfg) text)

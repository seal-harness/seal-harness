    2 -- | The content-addressed agent-metadata cache: a local, sanitized snapshot
    3 -- of a remote repo's @.agents/@ tree, keyed by @(repo URL, content hash)@ so
    4 -- every session working the same repo at the same content shares one entry.
    5 --
    6 -- Why: post-clone agent-def discovery reads @agents.md@ plus one
    7 -- @agent.md@ per discovered agent over SSH — two round trips per file. On a
    8 -- remote host with slow per-call setup this crawls; a single rsync of the
    9 -- whole tree (or a cache hit) collapses it to one probe + one transfer.
   10 --
   11 -- Security posture (untrusted bytes land on the trusted harness machine):
   12 --
   13 --   * Consumers are pure Markdown parsers — nothing in a snapshot is ever
   14 --     executed. This module enforces that structurally anyway:
   15 --   * 'sanitizeSnapshot' strips exec bits (files 0600, dirs 0700) and
   16 --     DELETES any non-regular entry — symlinks above all, so a malicious
   17 --     repo cannot plant @agent.md -> ~/.ssh/id_ed25519@ and have a parser
   18 --     read through it.
   19 --   * The transfer uses a minimal fixed rsync argv (no @-a@/@-p@/@-o@/@-g@/
   20 --     @-t@ — no owner/perms/times preservation), per-file size cap, and an
   21 --     @-e@ rsh string built ONLY from our own pinned 'SshConfig' options.
   22 --   * The probe hashes the tar stream (@tar -cf - .agents | sha256sum@):
   23 --     headers make it sensitive to names/renames/deletes, and sha256 is
   24 --     required because the input is adversarial — a weak checksum would let
   25 --     a malicious repo serve different agent sets under one cache key.
   26 --   * Any failure (rsync missing remotely, probe garbage, transfer error)
   27 --     yields 'MetaFallback' and callers use the legacy per-file crawl.

   98 -- | The fixed remote probe command for a repo dir's @.agents/@ content
   99 -- hash. Anchored with @cd@ (single-quoted — the path is SafePath-derived,
  100 -- quoting is defense-in-depth), streams the whole tree through @tar@ so
  101 -- names/sizes/types participate in the hash, and pipes into @sha256sum@
  102 -- (GNU coreutils — present on Linux remotes; its absence fails the probe
  103 -- and the caller falls back to the legacy crawl).
  104 agentMetaProbeCmd :: FilePath -> Text
  105 agentMetaProbeCmd repoDir =
  106   "cd " <> shellQuoteDir (T.pack repoDir)
  107   <> " && tar -cf - .agents 2>/dev/null | sha256sum"
  108 
  109 -- | Single-quote a path for the remote shell (embedded quotes escaped via
  110 -- the standard @'\\''@ idiom).
  111 shellQuoteDir :: Text -> Text
  112 shellQuoteDir t = "'" <> T.concatMap q t <> "'"
  113   where
  114     q :: Char -> Text
  115     q '\'' = "'\\''"
  116     q c    = T.singleton c
  117 
  118 -- | Parse the probe output (@sha256sum@ prints @\<hex\>\\s\\s-@ for stdin).
  119 -- Returns 'Nothing' unless the output leads with an exact 64-hex-digit
  120 -- digest — anything else (tar errors leaking through, missing-sha256sum
  121 -- diagnostics, empty output) fails the probe.
  122 parseProbeHash :: Text -> Maybe Text
  123 parseProbeHash out =
  124   case filter (not . T.null) (T.words out) of
  125     (hex : _)
  126       | T.length hex == 64
  127           && T.all (`elem` ("0123456789abcdefABCDEF" :: String)) hex
  128       -> Just (T.toLower hex)
  129     _ -> Nothing
  130 
  131 -- | Derive the cache entry directory name:
  132 -- @\<sha256(normalizedUrl)\>-\<sha256(content)\>@ — hex and dashes only, so
  133 -- it is filesystem-safe by construction.
  134 agentMetaCacheKey :: Text -> Text -> Text
  135 agentMetaCacheKey normalizedUrl contentHash =
  136   hexOf normalizedUrl <> "-" <> hexOf contentHash
  137   where
  138     hexOf :: Text -> Text
  139     hexOf = T.toLower . TE.decodeUtf8 . sha256Hash . TE.encodeUtf8

  141 -- | Sanitize a freshly-transferred snapshot directory, in place:
  142 --
  143 --   * dirs  → 0700
  144 --   * files → 0600 (exec bit stripped — the point of the exercise)
  145 --   * symlinks, fifos, devices, sockets → DELETED (never followed). A link
  146 --     pointing outside the snapshot would otherwise hand every parser a
  147 --     readable escape hatch onto the trusted machine.
  148 sanitizeSnapshot :: FilePath -> IO ()
  149 sanitizeSnapshot = go
  150   where
  151     go dir = do
  152       setFileMode dir 0o700
  153       names <- listDirectory dir
  154       mapM_ (handleEntry dir) names
  155     handleEntry parent name = do
  156       let path = parent </> name
  157       -- lstat semantics: judge the entry itself, never what a link points at.
  158       st <- getSymbolicLinkStatus path
  159       if isSymbolicLink st
  160         then ignoreIO (removeFile path)
  161         else if isDirectory st
  162           then go path
  163           else if isRegularFile st
  164             then setFileMode path 0o600
  165             else ignoreIO (removeFile path)  -- fifo / device / socket: prune
  166     ignoreIO :: IO () -> IO ()
  167     ignoreIO action = do
  168       _ <- try @IOException action
  169       pure ()

  175 -- | Per-file transfer cap (1 MiB). Discovery parsers bound reads far below
  176 -- this; the cap just stops a hostile repo from shipping a multi-gigabyte
  177 -- file inside @.agents/@.
  178 agentMetaMaxFileSize :: Int
  179 agentMetaMaxFileSize = 1024 * 1024
  180 
  181 -- | The local shell string rsync hands @-e@. Built EXCLUSIVELY from our own
  182 -- pinned 'SshConfig' options — host/user/port travel in the destination
  183 -- argument, never here — and every value is single-quoted because rsync
  184 -- runs the rsh program via the local shell.
  185 rsyncRshString :: SshConfig -> String
  186 rsyncRshString cfg =
  187   unwords
  188     ( [ "ssh"
  189       , "-o", shq "StrictHostKeyChecking=yes"
  190       , "-o", shq "BatchMode=yes"
  191       , "-o", shq ("UserKnownHostsFile=" <> scKnownHosts cfg)
  192       ]
  193         <> portArgs
  194         <> identityArgs
  195     )
  196   where
  197     shq :: String -> String
  198     shq s = "'" <> concatMap esc s <> "'"
  199     esc :: Char -> String
  200     esc '\'' = "'\\''"
  201     esc c    = [c]
  202     portArgs = case scPort cfg of
  203       22 -> []
  204       p  -> ["-p", show p]
  205     identityArgs = case scIdentity cfg of
  206       Nothing -> []
  207       Just f  -> ["-i", shq f]

  209 -- | The fixed rsync argv pulling a remote directory tree into one local
  210 -- dir. Element 0 is the PROGRAM NAME — 'rsyncTransferIO' splits it off
  211 -- before spawning (spawning @proc "rsync" argv@ verbatim would hand rsync
  212 -- a duplicated program name as a bogus source arg).
  213 --
  214 -- Deliberately NOT @-a@: no owner/group/perms/times preservation — every
  215 -- entry arrives with fresh default perms, then 'sanitizeSnapshot' sets the
  216 -- final tight modes.
  217 rsyncArgv :: SshConfig -> FilePath -> FilePath -> [String]
  218 rsyncArgv cfg srcAbs destDir =
  219   [ "rsync"
  220   , "-r"
  221   , "--max-size=" <> show agentMetaMaxFileSize
  222   , "-e", rsyncRshString cfg
  223   , T.unpack (getSshUser (scUser cfg)) <> "@" <> T.unpack (getSshHost (scHost cfg))
  224       <> ":" <> srcAbs
  225   , destDir ++ "/"
  226   ]

  242 data AgentMetaOutcome
  243   = MetaSnapshot FilePath   -- ^ Local sanitized snapshot root for this repo's .agents
  244   | MetaFallback            -- ^ Anything failed — caller uses the legacy crawl
  245   deriving stock (Eq, Show)
  246 
  247 -- | Ensure a content-addressed snapshot exists for @(normalizedUrl, current
  248 -- remote .agents content)@ and return its path, or 'MetaFallback'.
  249 --
  250 -- @runTransfer@ receives the rsync argv and performs it (production: real
  251 -- rsync via 'withManagedProcess'; tests: a fake that records the argv and
  252 -- materializes files) so the flow stays hermetically testable.
  253 ensureAgentMetaSnapshot
  254   :: RemoteRunner                    -- ^ probe transport (pinned ssh)
  255   -> SshConfig
  256   -> FilePath                        -- ^ cache root (<state>/repos/agent-meta-cache)
  257   -> Text                            -- ^ normalized repo URL (cache key part 1)
  258   -> FilePath                        -- ^ remote repo dir (absolute, SafePath-derived)
  259   -> ([String] -> FilePath -> IO (Either ExecError ()))
  260      -- ^ runTransfer argv destTmpDir
  261   -> IO AgentMetaOutcome
  262 ensureAgentMetaSnapshot runner cfg cacheRoot normalizedUrl repoDir runTransfer = do
  263   r <- try @IOException $ do
  264     mHash <- runProbe
  265     case mHash of
  266       Nothing -> pure MetaFallback
  267       Just contentHash -> do
  268         let entry = cacheRoot </> T.unpack (agentMetaCacheKey normalizedUrl contentHash)
  269         hit <- doesFileExist (entry </> metaFileName)
  270         if hit
  271           then do
  272             amcLog ("snapshot: cache HIT " <> T.pack entry)
  273             pure (MetaSnapshot entry)   -- content-addressed cache hit: no transfer
  274           else do
  275             amcLog ("snapshot: cache MISS, transferring " <> T.pack entry)
  276             transferAndPublish entry contentHash
  277   pure (fromRight MetaFallback r)
  278   where
  279     runProbe = case mkShellCommand (agentMetaProbeCmd repoDir) of
  280       Left _    -> pure Nothing   -- unreachable: internally-built, no NULs
  281       Right cmd -> do
  282         eRes <- runRemoteShell runner cfg cmd
  283         let mh = either (const Nothing) parseProbeHash eRes
  284         when (isNothing mh) $
  285           amcLog ("snapshot: content-hash probe failed for " <> T.pack repoDir
  286                   <> " raw=" <> either (T.pack . show) id eRes)
  287         pure mh
  288 
  289     transferAndPublish entry contentHash = do
  290       createDirectoryIfMissing True cacheRoot
  291       let tmp = entry ++ ".tmp"
  292       -- A stale .tmp from a crashed prior attempt must not survive.
  293       _ <- try @IOException (removeDirectoryRecursive tmp)
  294       createDirectoryIfMissing True tmp
  295       xfer <- runTransfer (agentMetaTransferArgv cfg repoDir tmp) tmp
  296       case xfer of
  297         Left _ -> do
  298           amcLog ("snapshot: rsync failed for " <> T.pack repoDir
  299                   <> " — see [agent-meta rsync] line above if present")
  300           _ <- try @IOException (removeDirectoryRecursive tmp)
  301           pure MetaFallback
  302         Right () -> do
  303           sanitizeSnapshot tmp
  304           writeMeta tmp contentHash
  305           published <- publish entry tmp
  306           pure (MetaSnapshot published)
  307 
  308     writeMeta dir contentHash = do
  309       now <- getCurrentTime
  310       let metaPath = dir </> metaFileName
  311       writeFile metaPath (metaJson repoDir contentHash now)
  312       setFileMode metaPath 0o600
  313 
  314     -- Publish atomically: rename tmp→entry. On a concurrent-writer race,
  315     -- keep the winner and drop ours — both are sanitized equivalents of
  316     -- the same (url, content-hash).
  317     publish entry tmp = do
  318       exists <- doesFileExist (entry </> metaFileName)
  319       if exists
  320         then do removeDirectoryRecursive tmp; pure entry
  321         else do renameDirectory tmp entry; pure entry
  322 
  323 metaFileName :: FilePath
  324 metaFileName = "seal-meta.json"

  326 -- | The entry's provenance record. Hand-rolled JSON over values we fully
  327 -- control (a validated path, a hex digest, an RFC-3339 timestamp).
  328 metaJson :: FilePath -> Text -> UTCTime -> String
  329 metaJson repoDir contentHash now =
  330   "{\"repo_dir\":" <> show repoDir
  331   <> ",\"content_hash\":" <> show (T.unpack contentHash)
  332   <> ",\"synced_at\":" <> show (formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%SZ" now)
  333   <> "}"

  352   , mceSshCfg       :: SshConfig      -- ^ destination for rsync + probe anchor
  353   , mceCacheRoot    :: FilePath       -- ^ <state>/repos/agent-meta-cache
  354   , mceRunTransfer  :: [String] -> FilePath -> IO (Either ExecError ())

  359 -- | The fixed remote command reading a repo clone's origin URL — the first
  360 -- half of the cache key. One round trip; failure (not a git repo, detached
  361 -- tree) simply excludes that repo from snapshot routing.
  362 probeRepoUrlCmd :: FilePath -> Text
  363 probeRepoUrlCmd repoDir =
  364   "cd " <> shellQuoteDir (T.pack repoDir)
  365   <> " && git config --get remote.origin.url"

  367 -- | Parse @git config@ output: a single non-empty line, trimmed. Rejects
  368 -- multi-line/garbage output (fail-closed to legacy routing).
  369 parseGitConfigUrl :: Text -> Maybe Text
  370 parseGitConfigUrl out =
  371   case lines (T.unpack out) of
  372     [l] -> let t = T.strip (T.pack l)
  373            in if T.null t then Nothing else Just t
  374     _   -> Nothing

  376 -- | Read ceiling served to parsers from snapshots. Matches the remote arm's
  377 -- bounded-read cap so consumers see identical truncation behavior.
  378 metaReadCeilingBytes :: Int
  379 metaReadCeilingBytes = 131072

  381 -- | Build a WorkdirFs that routes reads falling under any successfully
  382 -- snapshotted @<repo>/.agents@ subtree to the LOCAL sanitized snapshot,
  383 -- passing everything else through to the underlying (remote) fs.
  384 --
  385 -- Structure truth stays remote: 'wfsSnapshot' is always a passthrough, so
  386 -- SETUP_REPO invalidation and repo enumeration behave exactly as before —
  387 -- only file content reads change lanes. Local serving goes through
  388 -- 'mkLocalWorkdirFs', whose SafePath confinement keeps every read inside
  389 -- its snapshot root (and snapshots contain no symlinks to escape with).
  390 buildRoutingWorkdirFs
  391   :: MetaCacheEnv          -- ^ probe/transfer deps
  392   -> WorkdirFs             -- ^ underlying (remote) fs
  393   -> Text                  -- ^ the SESSION WORKDIR absolute path (the scan's
  394                            -- own anchor — NOT the ssh workspace; sessions nest
  395                            -- at <workspace>/workdirs/<sid>/)
  396   -> [Text]                -- ^ candidate repo top-dir names from wfsSnapshot
  397   -> IO WorkdirFs
  398 buildRoutingWorkdirFs env underlying wsRootAbs topDirs = do
  399   amcLog ("routing: session-workdir=" <> wsRootAbs
  400           <> " candidates=" <> T.pack (show topDirs))
  401   routes <- concat <$> mapM routeOne topDirs
  402   if null routes
  403     then amcLog "routing: no routes established (all repos fell back)"
  404     else amcLog ("routing: " <> T.pack (show (map fst routes)))
  405   pure (applyRoutes routes underlying)
  406   where
  407     routeOne topDir = do
  408       let repoDirAbs = T.unpack wsRootAbs </> T.unpack topDir
  409       mUrl <- probeUrl repoDirAbs
  410       case mUrl of
  411         Nothing -> do
  412           amcLog ("routing: url probe failed for " <> topDir
  413                   <> " — falling back to legacy reads")
  414           pure []
  415         Just url -> do
  416           amcLog ("routing: url=" <> url <> " repo=" <> T.pack repoDirAbs)
  417           outcome <- ensureAgentMetaSnapshot (mceRunner env) (mceSshCfg env)
  418                        (mceCacheRoot env) url repoDirAbs (mceRunTransfer env)
  419           case outcome of
  420             MetaSnapshot dir -> do
  421               amcLog ("routing: snapshot OK for " <> T.pack repoDirAbs)
  422               pure [(topDir <> "/.agents", dir)]
  423             MetaFallback -> do
  424               amcLog ("routing: snapshot failed for " <> T.pack repoDirAbs
  425                       <> " — falling back to legacy reads")
  426               pure []
  427     probeUrl rd = case mkShellCommand (probeRepoUrlCmd rd) of
  428       Left _    -> pure Nothing   -- unreachable: internally built, no NULs
  429       Right cmd -> do
  430         eRes <- runRemoteShell (mceRunner env) (mceSshCfg env) cmd
  431         pure $ either (const Nothing) parseGitConfigUrl eRes

  433 -- | Apply prefix→localRoot routing over a WorkdirFs record. Matching reads
  434 -- are served by a SafePath-confined local arm rooted at the snapshot;
  435 -- everything else — including 'wfsSnapshot', the structure source of truth —
  436 -- passes through untouched. An empty route table returns the fs unchanged.
  437 applyRoutes :: [(Text, FilePath)] -> WorkdirFs -> WorkdirFs
  438 applyRoutes [] fs = fs
  439 applyRoutes routes fs = fs
  440   { wfsReadFile           = \rp ->
  441       route rp (wfsReadFile fs) serveRead
  442   , wfsDoesFileExist      = \rp ->
  443       route rp (wfsDoesFileExist fs) (existsIn wfsDoesFileExist)
  444   , wfsDoesDirectoryExist = \rp ->
  445       route rp (wfsDoesDirectoryExist fs) (existsIn wfsDoesDirectoryExist)
  446   , wfsListDirectory      = \rp ->
  447       route rp (wfsListDirectory fs) serveList
  448   }
  449   where
  450     route
  451       :: RemotePath
  452       -> (RemotePath -> IO a)
  453       -> (FilePath -> Text -> IO a)
  454       -> IO a
  455     route rp onRemote onLocal =
  456       case matchRoute (getRemotePath rp) of
  457         Nothing          -> onRemote rp
  458         Just (root, rel) -> onLocal root rel
  459     matchRoute t = foldr
  460       (\(prefix, root) acc ->
  461         if t == prefix
  462           then Just (root, ".")
  463           else if (prefix <> "/") `T.isPrefixOf` t
  464             then Just (root, T.drop (T.length prefix + 1) t)
  465             else acc)
  466       Nothing routes
  467     localFs root = mkLocalWorkdirFs (WorkspaceRoot root) metaReadCeilingBytes
  468     -- The rel paths are internally generated (prefix-stripped snapshot
  469     -- paths), so a parse failure here is unreachable; it maps to WfsNotFound
  470     -- rather than inventing a PathError.
  471     toRp :: Text -> Either WorkdirFsErr RemotePath
  472     toRp rel = either (const (Left WfsNotFound)) Right (mkRemotePath rel)
  473 
  474     serveRead root rel =
  475       case toRp rel of
  476         Left e    -> pure (Left e)
  477         Right rp  -> wfsReadFile (localFs root) rp
  478     existsIn
  479       :: (WorkdirFs -> RemotePath -> IO Bool)
  480       -> FilePath -> Text -> IO Bool
  481     existsIn sel root rel =
  482       case toRp rel of
  483         Left _   -> pure False
  484         Right rp -> sel (localFs root) rp
  485     serveList root rel =
  486       case toRp rel of
  487         Left e    -> pure (Left e)
  488         Right rp  -> wfsListDirectory (localFs root) rp

  494 -- | Startup sweep: keep only the newest @keepN@ cache entries (by entry
  495 -- dir mtime). Best-effort — every error is swallowed; a full disk of stale
  496 -- entries is untidy, not unsafe.
  497 gcAgentMetaCache :: FilePath -> Int -> IO ()
  498 gcAgentMetaCache root keepN = do
  499   _ <- try @IOException $ do
  500     names <- listDirectory root
  501     dated <- forM names $ \n -> do
  502       let p = root </> n
  503       mTime <- try @IOException (getModificationTime p)
  504       pure (either (const []) (\t -> [(t, p)]) mTime)
  505     let sorted = sortOn (Down . fst) (concat dated)
  506         stale  = drop keepN sorted
  507     mapM_ (removeDirectoryRecursive . snd) stale
  508   pure ()

  510 -- | Production transfer runner: fixed-argv rsync via the managed-process
  511 -- seam (process-group spawn, bounded output, group kill on cancel).
  512 -- Non-zero exit → 'ExecError' (the caller falls back to the legacy crawl);
  513 -- stderr is logged at debug for diagnosability.
  514 rsyncTransferIO :: SshConfig -> [String] -> FilePath -> IO (Either ExecError ())
  515 rsyncTransferIO cfg argv _destTmp = case argv of
  516   -- argv carries the program name at its head; spawn splits it off. An
  517   -- empty argv is unreachable (callers pass 'agentMetaTransferArgv' output).
  518   []                -> pure (Left ExecRemoteUnreachable)
  519   (prog : progArgs) ->
  520     let cp = (proc prog progArgs)
  521                { std_in = NoStream, std_out = CreatePipe, std_err = CreatePipe
  522                , create_group = True
  523                }
  524     in runManagedRsync cfg cp
  525 
  526 runManagedRsync :: SshConfig -> CreateProcess -> IO (Either ExecError ())
  527 runManagedRsync cfg cp = do
  528   res <- try @IOException $
  529     withManagedProcess cp $ \ph _mIn hOut hErr -> do
  530       out <- readBounded hOut agentMetaMaxFileSize
  531       err <- readBounded hErr agentMetaMaxFileSize
  532       ec  <- waitForProcess ph
  533       pure (ec, out, err)
  534   case res of
  535     Left _ioErr -> pure (Left ExecRemoteUnreachable)
  536     Right (ExitSuccess, _, _) -> pure (Right ())
  537     Right (ExitFailure _n, _, err) -> do
  538       globalLogIO K.DebugS (K.ls (T.pack
  539         ("[agent-meta rsync] failed: " <> T.unpack (T.strip (T.takeEnd 400 err))
  540          <> " host=" <> show (getSshHost (scHost cfg)))))
  541       pure (Left ExecRemoteUnreachable)

    2 -- | No-progress repeat detection for session reads (session-read hardening
    3 -- WU4).
    4 --
    5 -- The incident (session 20261022-174851-823): the agent re-issued
    6 -- increasingly large IDENTICAL @SESSION_MANAGE@ reads, each embedding the
    7 -- previous tool result verbatim — a no-progress feedback loop nothing
    8 -- detected. This module gives the agent loop a per-session sliding window
    9 -- of canonical call keys so a no-progress loop becomes observable and
   10 -- interruptible:
   11 --
   12 -- * 1st occurrence — 'RvAllow' (unchanged behavior);
   13 -- * 2nd–3rd — 'RvWarn' (dispatch proceeds; 'repeatWarnNote' is prepended
   14 --   to the result parts so the model sees the signal);
   15 -- * ≥4th — 'RvBlock' (the opcode is never dispatched; an actionable error
   16 --   is synthesized).
   17 --
   18 -- Detection is call-order based (not wall-clock), scoped to 'repeatScope'
   19 -- (@SESSION_MANAGE@ + @SESSION_GET@ only), and enabled by default. The
   20 -- registry is process-lifetime: created once per process at the same
   21 -- origination sites as 'Seal.Tools.Exec.Abort.SessionAbortRegistry' and
   22 -- keyed by 'SessionId', so the window survives across turns and requests.
   23 -- Bounded memory: each session's window holds at most
   24 -- 'repeatWindowCapacity' canonical keys; the registry mirrors the abort
   25 -- registry's map-growth policy (no cap — entries are per live session).
   26 --
   27 -- Security: the 'RepeatTrack' and 'SessionRepeatRegistry' constructors are
   28 -- NOT exported (mirrors 'Seal.Tools.Exec.Abort'). Tracks are obtained from
   29 -- 'newRepeatTrack' or 'lookupOrCreateRepeatTrack' only.

   33     -- * Per-session sliding window

   35   , newRepeatTrack
   36   , recordCall

   39   , newSessionRepeatRegistry
   40   , lookupOrCreateRepeatTrack
   41     -- * Scope + loop-facing messages
   42   , repeatScope
   43   , repeatWarnNote

   68 -- | The repeat detector's verdict for one in-scope call.
   69 data RepeatVerdict
   70   = RvAllow
   71     -- ^ First occurrence of the canonical key within the window. Dispatch
   72     -- unchanged.
   73   | RvWarn
   74     -- ^ 2nd–3rd occurrence within the window. The call is dispatched
   75     -- normally; 'repeatWarnNote' is prepended to the result parts so the
   76     -- model sees the no-progress signal.
   77   | RvBlock
   78     -- ^ ≥4th occurrence within the window. The opcode is never dispatched;
   79     -- 'repeatBlockedMessage' is synthesized as an error result so the
   80     -- model can correct course.
   81   deriving (Eq, Show)
   82 
   83 -- | Sliding-window capacity: the number of most-recent canonical call keys
   84 -- retained per session. 'repeatWindowCapacity' intervening calls evict the
   85 -- oldest entry, making an old key \"fresh\" again (detection is
   86 -- call-order based, not permanent).
   87 repeatWindowCapacity :: Int
   88 repeatWindowCapacity = 16
   89 
   90 -- | Opcodes subject to repeat detection: session reads only (the incident
   91 -- loop was a session read). Ops outside this set are untracked — zero
   92 -- behavior change.
   93 repeatScope :: Set OpName
   94 repeatScope = Set.fromList [OpName "SESSION_MANAGE", OpName "SESSION_GET"]

   96 -- | The note prepended to a warned call's tool-result parts (model-visible
   97 -- only; nothing content-bearing is added to recorded metadata).
   98 repeatWarnNote :: Text
   99 repeatWarnNote =
  100   "[note: repeated identical session read — no new information expected]"
  101 
  102 -- | The synthesized error for a blocked call. Actionable: names the opcode
  103 -- and tells the model what to do instead of re-issuing the identical call.
  104 repeatBlockedMessage :: OpName -> Text
  105 repeatBlockedMessage (OpName name) =
  106   "blocked: repeated identical " <> name
  107   <> " call — refine the request or move on; no new information is expected from repeating it"

  109 -- | Canonical identity of a call: the opcode name, a NUL separator, and the
  110 -- aeson-encoded input as 'Text'. JSON output escapes control characters, so
  111 -- a raw NUL cannot appear in either half — the pair injectively encodes as
  112 -- one 'Text'. The decode is lenient so the function is total
  113 -- (aeson output is always valid UTF-8; leniency is belt-and-braces).
  114 canonicalKey :: OpName -> Value -> Text
  115 canonicalKey (OpName name) input =
  116   name <> "\0"
  117        <> TE.decodeUtf8With lenientDecode (BL.toStrict (A.encode input))
  118 
  119 -- | Push a canonical key onto the window (most recent first), evicting
  120 -- beyond 'repeatWindowCapacity'. Bounded: the result holds at most
  121 -- 'repeatWindowCapacity' entries, so 'recordCall' is O(window) per call.
  122 stepWindow :: Text -> [Text] -> [Text]
  123 stepWindow key window = take repeatWindowCapacity (key : window)
  124 
  125 -- | Verdict for a canonical key against the window AFTER the push (the
  126 -- key itself included): 1 occurrence → 'RvAllow', 2–3 → 'RvWarn', ≥4 →
  127 -- 'RvBlock' (maintainer-confirmed thresholds, 2026-10-22).
  128 windowVerdict :: Text -> [Text] -> RepeatVerdict
  129 windowVerdict key windowAfterPush = case countIn key windowAfterPush of
  130   n | n >= 4    -> RvBlock
  131     | n >= 2    -> RvWarn
  132     | otherwise -> RvAllow
  133 
  134 -- | Count a key's occurrences in a window (bounded: callers pass windows
  135 -- of at most 'repeatWindowCapacity' entries).
  136 countIn :: Text -> [Text] -> Int
  137 countIn key = length . filter (== key)
  138 
  139 -- | Per-session sliding window of the last 'repeatWindowCapacity' canonical
  140 -- call keys (most recent first). The constructor is NOT exported — tracks
  141 -- are obtained from 'newRepeatTrack' or 'lookupOrCreateRepeatTrack'.
  142 newtype RepeatTrack = RepeatTrack (TVar [Text])
  143 
  144 -- | Build a fresh, empty track.
  145 newRepeatTrack :: IO RepeatTrack
  146 newRepeatTrack = RepeatTrack <$> newTVarIO []
  147 
  148 -- | Record a call and return its verdict. Pushes the canonical key onto the
  149 -- session's sliding window (evicting beyond 'repeatWindowCapacity'), then
  150 -- counts the key's occurrences in the updated window. O(window) per call,
  151 -- exception-free (the window is bounded; the JSON encoding is total). The
  152 -- agent loop calls this for opcodes in 'repeatScope' only — out-of-scope
  153 -- opcodes are never recorded.
  154 recordCall :: RepeatTrack -> OpName -> Value -> IO RepeatVerdict
  155 recordCall (RepeatTrack tv) name input = do
  156   let key = canonicalKey name input
  157   window <- atomically $ do
  158     old <- readTVar tv
  159     let new = stepWindow key old
  160     writeTVar tv new
  161     pure new
  162   pure (windowVerdict key window)
  163 
  164 -- | Process-lifetime repeat registry (mirrors
  165 -- 'Seal.Tools.Exec.Abort.SessionAbortRegistry'). Maps each 'SessionId' to
  166 -- its 'RepeatTrack', lazily created on first lookup via atomic
  167 -- insert-if-absent STM. Created ONCE per process at the same origination
  168 -- sites as the abort registry — a per-request registry would silently
  169 -- reset the cross-turn window. The constructor is NOT exported.
  170 newtype SessionRepeatRegistry = SessionRepeatRegistry (TVar (Map SessionId RepeatTrack))
  171 
  172 -- | Build a new empty registry.
  173 newSessionRepeatRegistry :: IO SessionRepeatRegistry
  174 newSessionRepeatRegistry = SessionRepeatRegistry <$> newTVarIO Map.empty
  175 
  176 -- | Look up the 'RepeatTrack' for a session, creating it on first use. The
  177 -- insert-if-absent is atomic via STM, so two concurrent lookups for a new
  178 -- session yield the same track (no duplicate creation). The SAME track is
  179 -- returned for the same sid across calls and turns — this is what makes
  180 -- the window process-lifetime state.
  181 lookupOrCreateRepeatTrack :: SessionRepeatRegistry -> SessionId -> IO RepeatTrack
  182 lookupOrCreateRepeatTrack (SessionRepeatRegistry tv) sid = do
  183   m <- readTVarIO tv
  184   case Map.lookup sid m of
  185     Just track -> pure track
  186     Nothing -> do
  187       track <- newRepeatTrack
  188       atomically $ do
  189         m' <- readTVar tv
  190         -- insertIfAbsent semantics: if another thread won the race, use
  191         -- its track; otherwise insert ours.
  192         let track' = fromMaybe track (Map.lookup sid m')
  193         writeTVar tv (Map.insert sid track' m')
  194         pure track'

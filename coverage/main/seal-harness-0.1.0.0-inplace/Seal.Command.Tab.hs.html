   29 -- | A hook invoked after a tab is successfully closed. The 'TabRef' is the
   30 -- closed tab's backing ref (e.g. @BoundSession sid@). Wiring sites build
   31 -- this from the 'CursorStore' + 'ReplyRegistry' so conversations attached
   32 -- to the closed tab are notified. The no-op identity is 'noTabCloseNotifier'
   33 -- (used by tests + the standalone CLI which has no other channels).
   34 type TabCloseNotifier = TabRef -> IO ()
   35 
   36 -- | The identity 'TabCloseNotifier' — does nothing. Used when there are no
   37 -- other channels to notify (tests, standalone CLI).
   38 noTabCloseNotifier :: TabCloseNotifier
   39 noTabCloseNotifier _ = pure ()

   41 -- | The @/tab@ command spec. Bare @/tab@ is intercepted by
   42 -- 'Seal.Routing.Route' as 'CurrentTab' (show the focused tab) BEFORE the
   43 -- registry is consulted, so this spec's parser only runs when a subcommand
   44 -- is present (@/tab list@, @/tab new@, etc.).
   45 tabCommandSpec :: SealPaths -> TabsHandle -> TabCloseNotifier -> CommandSpec
   46 tabCommandSpec paths h closeNotifier = CommandSpec
   47   { csName         = CommandName "tab"
   48   , csAliases      = []
   49   , csGroup        = GroupGeneral
   50   , csSynopsis     = "Show current tab, or manage tabs (list/new/close/focus/resume/rename)"
   51   , csParserInfo   = tabParserInfo paths h closeNotifier
   52   , csAvailability = AlwaysAvailable
   53   }
   54 
   55 -- | The terse-grammar synopsis entry for @/help@. A synthetic spec (no
   56 -- parser — it's handled by 'Seal.Routing.Route' before the registry); the
   57 -- synopsis is registered so @/help@ shows the @/N@ grammar.
   58 terseGrammarSpec :: CommandSpec
   59 terseGrammarSpec = CommandSpec
   60   { csName         = CommandName "N"
   61   , csAliases      = []
   62   , csGroup        = GroupGeneral
   63   , csSynopsis     = terseSynopsis
   64   , csParserInfo   = info (pure (commandAction (const (pure ())))) (progDesc "Terse tab switching")
   65   , csAvailability = AlwaysAvailable
   66   }

   68 tabParserInfo :: SealPaths -> TabsHandle -> TabCloseNotifier -> ParserInfo CommandAction
   69 tabParserInfo paths h closeNotifier =
   70   info (tabParser paths h closeNotifier <**> helper)
   71     (  progDesc "Manage tabs"
   72     <> header   "tab — manage tabs (new/list/close/focus/resume/rename)"
   73     )
   74 
   75 tabParser :: SealPaths -> TabsHandle -> TabCloseNotifier -> Parser CommandAction
   76 tabParser paths h closeNotifier = hsubparser
   77   $  command "list"   (info (pure (listCmd paths h))   (progDesc "List all tabs"))
   78   <> command "new"    (info (newCmd h <$> optional kindArg)
   79                                  (progDesc "Create a new tab (default kind: ai)"))
   80   <> command "close"  (info (closeCmd h closeNotifier <$> tabIndexArg <*> forceFlag)
   81                                  (progDesc "Close a tab by index (compacts the list)"))
   82   <> command "focus"  (info (focusCmd h <$> tabIndexArg)
   83                                  (progDesc "Focus a tab by index"))
   84   <> command "resume" (info (resumeCmd h <$> sessionArg)
   85                                  (progDesc "Resume a session into a new tab"))
   86   <> command "rename" (info (renameCmd h <$> tabIndexArg <*> nameArg)
   87                                  (progDesc "Rename a tab by index"))
   88   <> metavar "COMMAND"

   90 -- | The /tab list subcommand. Resolves each tab's display name: the
   91 -- user-set label (if any), or the first user message snippet (matching
   92 -- the web frontend's 'sessionDisplayTitle' cascade) for session-backed
   93 -- tabs with no label.
   94 listCmd :: SealPaths -> TabsHandle -> CommandAction
   95 listCmd paths h = commandAction $ \caps -> do
   96   tl <- snapshotTabs h
   97   if tabCount tl == 0
   98     then ccSend caps "no tabs"
   99     else do
  100       names <- mapM (resolveTabName paths) (tlTabs tl)
  101       mapM_ (ccSend caps . uncurry renderTabLine) (zip names (tlTabs tl))
  102 
  103 -- | Resolve a tab's display name. Returns 'Nothing' when no name is
  104 -- available (no label, no transcript, or a harness tab with no label).
  105 resolveTabName :: SealPaths -> Tab -> IO (Maybe Text)
  106 resolveTabName paths t
  107   | Just label <- tLabel t = pure (Just label)
  108   | BoundSession sid <- tRef t = do
  109       let mp = sessionMetaPath paths sid
  110       exists <- doesFileExist mp
  111       if not exists
  112         then pure Nothing
  113         else firstUserMessageSnippet paths sid
  114   | otherwise = pure Nothing
  115 
  116 -- | The /tab new subcommand. (For 6b the kind is informational; a session
  117 -- tab is the default. A harness tab needs the wizard — deferred.)
  118 newCmd :: TabsHandle -> Maybe Text -> CommandAction
  119 newCmd h _mKind = commandAction $ \caps -> do
  120   let ref = BoundSession placeholder
  121   r <- insertTabH h ref KindAi Nothing
  122   case r of
  123     Left e  -> ccSend caps ("tab new failed: " <> e)
  124     Right i -> ccSend caps ("tab " <> T.singleton (tabIndexToChar i) <> " created")
  125   where
  126     placeholder = case mkSessionId "tab-session" of
  127       Right s -> s
  128       Left _  -> error "placeholder session id"
  129 
  130 -- | The /tab close subcommand. Snapshots the tab's 'TabRef' before removing
  131 -- it so the close notifier can tell attached channels which tab was closed.
  132 closeCmd :: TabsHandle -> TabCloseNotifier -> Int -> Bool -> CommandAction
  133 closeCmd h closeNotifier idx force = commandAction $ \caps -> do
  134   case mkTabIndex idx of
  135     Left e  -> ccSend caps ("invalid index: " <> e)
  136     Right i -> do
  137       mRef <- tabRefAtIndex h i
  138       r <- removeTabH h i
  139       case r of
  140         Left e  -> if force then ccSend caps ("force close: " <> e) else ccSend caps ("close failed: " <> e)
  141         Right _ -> do
  142           ccSend caps ("tab " <> T.singleton (tabIndexToChar i) <> " closed")
  143           maybe (pure ()) closeNotifier mRef
  144 
  145 -- | Look up the 'TabRef' at a tab index ('Nothing' if out of range).
  146 tabRefAtIndex :: TabsHandle -> Seal.Handles.Tab.TabIndex -> IO (Maybe TabRef)
  147 tabRefAtIndex h i = do
  148   tl <- snapshotTabs h
  149   pure (tRef <$> lookupTab tl i)
  150   where
  151     lookupTab tl idx = go (tlTabs tl)
  152       where
  153         go [] = Nothing
  154         go (t:rest) | tIndex t == idx = Just t
  155                     | otherwise       = go rest
  156 
  157 -- | The /tab focus subcommand.
  158 focusCmd :: TabsHandle -> Int -> CommandAction
  159 focusCmd h idx = commandAction $ \caps -> do
  160   case mkTabIndex idx of
  161     Left e  -> ccSend caps ("invalid index: " <> e)
  162     Right i -> do
  163       r <- focusTabH h i
  164       case r of
  165         Left e  -> ccSend caps ("focus failed: " <> e)
  166         Right _ -> ccSend caps ("focused tab " <> T.singleton (tabIndexToChar i))
  167 
  168 -- | The /tab resume subcommand.
  169 resumeCmd :: TabsHandle -> Text -> CommandAction
  170 resumeCmd h sidText = commandAction $ \caps -> do
  171   case mkSessionId sidText of
  172     Left e  -> ccSend caps ("invalid session id: " <> e)
  173     Right s -> do
  174       r <- insertTabH h (BoundSession s) KindAi Nothing
  175       case r of
  176         Left e  -> ccSend caps ("resume failed: " <> e)
  177         Right i -> ccSend caps ("tab " <> T.singleton (tabIndexToChar i) <> " resumed " <> sessionIdText s)
  178 
  179 -- | The /tab rename subcommand.
  180 renameCmd :: TabsHandle -> Int -> Text -> CommandAction
  181 renameCmd h idx name = commandAction $ \caps -> do
  182   case mkTabIndex idx of
  183     Left e  -> ccSend caps ("invalid index: " <> e)
  184     Right i -> do
  185       r <- renameTabH h i name
  186       case r of
  187         Left e  -> ccSend caps ("rename failed: " <> e)
  188         Right _ -> ccSend caps ("tab " <> T.singleton (tabIndexToChar i) <> " renamed to " <> name)

  190 -- | One line per tab for /tab list. The display name (label or snippet) is
  191 -- resolved by 'resolveTabName' and passed as 'mName'.
  192 renderTabLine :: Maybe Text -> Tab -> Text
  193 renderTabLine mName t =
  194   T.singleton (tabIndexToChar (tIndex t)) <> "  " <> kindText (tKind t)
  195     <> maybe "" ("  " <>) mName
  196     <> "  " <> refText (tRef t)
  197   where
  198     kindText = T.pack . show
  199     refText (BoundSession s)  = "session:" <> sessionIdText s
  200     refText (BoundHarness _)  = "harness:<id>"

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | A name-indexed opcode set; derives the provider tool-definition list the
    3 -- agent is offered each turn. Tool definitions preserve registration order
    4 -- (not alphabetical) so the wiring layer controls which tools the model sees
    5 -- first — many LLMs bias toward the first tool whose description matches,
    6 -- so the order is a lever for steering tool selection (e.g. WEB_FETCH before
    7 -- BROWSER_OPEN for simple page fetches).
    8 module Seal.ISA.Registry
    9   ( Registry
   10   , mkRegistry
   11   , lookupOp
   12   , hideOpcodes
   13   , registryToolDefs
   14   , registryToolDefs'
   15   , stubSchema
   16   , secretOpNames
   17   , secretOpcodes
   18   ) where
   19

   21 import Data.Map.Strict qualified as Map
   22 import Data.Set (Set)
   23 import Data.Set qualified as Set

   25 import Seal.Core.Types (OpName (..))

   29 -- | The registry carries both a name-indexed 'Map' (O(log n) dispatch
   30 -- lookup) and the registration-ordered opcode list (for tool-definition
   31 -- emission in a stable, wiring-controlled order). The 'Set OpName' marks
   32 -- opcodes that are dispatchable but hidden from the model's tool catalog
   33 -- (e.g. legacy shims superseded by consolidated MANAGE opcodes).
   34 data Registry = Registry (Map OpName Opcode) [Opcode] (Set OpName)
   35 
   36 mkRegistry :: [Opcode] -> Registry
   37 -- | Deduplicate by opcode name, keeping the first occurrence in
   38 -- registration order. This guards the tool-definition list (and thus the
   39 -- model's tool catalog) against accidental duplicates in the wiring layer.
   40 mkRegistry ops =
   41   let dedup = dedupByName Set.empty ops
   42         where dedupByName _ [] = []
   43               dedupByName acc (o : rest)
   44                 | opName o `Set.member` acc = dedupByName acc rest
   45                 | otherwise = o : dedupByName (Set.insert (opName o) acc) rest
   46   in Registry (Map.fromList [(opName o, o) | o <- dedup]) dedup Set.empty
   47 
   48 lookupOp :: Registry -> OpName -> Maybe Opcode
   49 lookupOp (Registry m _ _) n = Map.lookup n m
   50 
   51 -- | Mark opcodes as hidden from the model's tool catalog. They remain
   52 -- dispatchable via 'lookupOp' (so @\/skill load@, transcript replay, and
   53 -- other internal callers that reference legacy opcode names still work),
   54 -- but 'registryToolDefs'' omits them — the model only sees the
   55 -- consolidated MANAGE opcodes, not the legacy shims they subsume.
   56 hideOpcodes :: Set OpName -> Registry -> Registry
   57 hideOpcodes hidden (Registry m order _) = Registry m order hidden
   58 
   59 -- | The tool definitions the provider is offered each turn, in registration
   60 -- order (the order 'mkRegistry' received). NOT alphabetical — the wiring
   61 -- layer controls the order so it can steer which tools the model tries first.
   62 registryToolDefs :: Registry -> [ToolDefinition]
   63 registryToolDefs = registryToolDefs' False
   64 
   65 -- | Like 'registryToolDefs' but with a flag to swap full @input_schema@s for
   66 -- a minimal stub. When @useStub@ is 'True', each tool definition carries
   67 -- 'stubSchema' instead of the opcode's real 'opInSchema' — the name and
   68 -- description are preserved so the model still knows what tools exist, but
   69 -- the per-opcode schema JSON (the bulk of the tokens) is omitted. The model
   70 -- is expected to call the @OPCODE_DESCRIBE@ opcode to retrieve a tool's full
   71 -- schema before calling it. When @useStub@ is 'False' this is identical to
   72 -- 'registryToolDefs'. The provider encoders OMIT the @input_schema@ field
   73 -- entirely when it equals 'stubSchema' (see 'Seal.Providers.Class'), so the
   74 -- stub costs zero tokens on the wire rather than the few a real stub object
   75 -- would cost.
   76 registryToolDefs' :: Bool -> Registry -> [ToolDefinition]
   77 registryToolDefs' useStub (Registry _ order hidden) =
   78   [ ToolDefinition (opName o) (opDesc o) (if useStub then stubSchema else opInSchema o)
   79   | o <- order
   80   , not (opName o `Set.member` hidden)
   81   ]
   82 
   83 -- | The set of opcode names whose tool results may carry secrets and must be
   84 -- redacted from the on-disk @conversation.jsonl@. Only opcodes that return a
   85 -- vault secret value in 'orParts' belong here — currently just 'SECRET_GET'.
   86 -- Other opcodes (MEMORY_READ, FILE_READ, SHELL_EXEC, etc.) return
   87 -- agent-visible data that is safe to persist verbatim and display in the
   88 -- frontend. Using trust level as a proxy was wrong: MEMORY_READ is Trusted
   89 -- but returns memory content (not vault secrets), so redacting it hid
   90 -- harmless output behind @<redacted:secret>@.
   91 secretOpNames :: Registry -> Set OpName
   92 secretOpNames (Registry m _ _) =
   93   Set.fromList [ opName o | o <- Map.elems m, opName o `Set.member` secretOpcodes ]
   94 
   95 -- | The static set of opcode names that return secret values in 'orParts'.
   96 secretOpcodes :: Set OpName
   97 secretOpcodes = Set.fromList [ OpName "SECRET_GET" ]

   66 -- | The scheduler configuration. The 'csFireDelegate' fires a job and
   67 -- returns the spawned session id. The 'csNotifyHuman' callback is used
   68 -- for SHOW_HUMAN notifications (e.g. auto-disable after failures).
   69 -- The 'csClock' is injectable for deterministic testing.
   70 data CronScheduler = CronScheduler
   71   { csStore         :: CronStore
   72   , csFireDelegate  :: CronFireDelegate
   73   , csClock         :: IO UTCTime
   74   , csTickInterval  :: Int               -- ^ seconds
   75   , csStopSignal    :: TVar Bool
   76   , csMaxConcurrent :: Int
   77   , csNotifyHuman   :: Text -> IO ()     -- ^ SHOW_HUMAN callback
   78   }

   80 -- | Start the cron scheduler. Forks a thread that ticks every
   81 -- 'csTickInterval' seconds, checks for due jobs, and fires them.
   82 -- Returns immediately (the scheduler runs in a forked thread).
   83 startCronScheduler :: CronScheduler -> IO ()
   84 startCronScheduler scheduler = do
   85   stop <- readTVarIO (csStopSignal scheduler)
   86   unless stop $ do
   87     tick scheduler
   88     threadDelay (csTickInterval scheduler * 1_000_000)
   89     startCronScheduler scheduler
   90 
   91 -- | One tick: check for due jobs and fire them.
   92 tick :: CronScheduler -> IO ()
   93 tick scheduler = do
   94   now <- csClock scheduler
   95   jobs <- csList (csStore scheduler) True  -- enabled only
   96   mapM_ (fireIfDue scheduler now) jobs
   97 
   98 -- | Fire a job if it's due (next_fire <= now) and not overlapping.
   99 -- The next fire time is computed from the job's last fire time (or
  100 -- creation time if never fired), not from @now@ — this is the standard
  101 -- cron pattern: "the scheduled time for the next run has arrived."
  102 -- Computing from @now@ would always return a future time (the next
  103 -- minute boundary after now), so no job would ever be due.
  104 fireIfDue :: CronScheduler -> UTCTime -> CronJob -> IO ()
  105 fireIfDue scheduler now job = do
  106   let refTime = fromMaybe (cjCreatedAt job) (cjLastFire job)
  107   mNextFire <- computeNextFire (cjSchedule job) (cjTimezone job) refTime
  108   case mNextFire of
  109     Nothing -> pure ()
  110     Just nextFire
  111       | nextFire > now -> pure ()  -- next scheduled fire is still in the future
  112       | otherwise -> do
  113           -- Check overlap: previous run still pending/running
  114           runs <- csListRuns (csStore scheduler) (cjId job) 1
  115           case runs of
  116             (r:_) | isRunActive now r -> pure ()  -- overlap, skip
  117             _ -> do
  118               -- Check max concurrent
  119               activeCount <- countActiveRuns scheduler now
  120               when (activeCount < csMaxConcurrent scheduler) $ do
  121                 fireJob scheduler now job
  122 
  123 -- | Count currently active runs (Pending or Running) across all jobs.
  124 countActiveRuns :: CronScheduler -> UTCTime -> IO Int
  125 countActiveRuns scheduler now = do
  126   allJobs <- csList (csStore scheduler) True
  127   activeRuns <- mapM countJobActive allJobs
  128   pure (sum activeRuns)
  129   where
  130     countJobActive job = do
  131       runs <- csListRuns (csStore scheduler) (cjId job) 10
  132       pure (length (filter (isRunActive now) runs))
  133 
  134 -- | Fire a single job: record a Pending run, call the fire delegate,
  135 -- update the run status on completion.
  136 fireJob :: CronScheduler -> UTCTime -> CronJob -> IO ()
  137 fireJob scheduler now job = do
  138   let jid = cjId job
  139   -- Record a Pending run
  140   let pendingRun = CronRun
  141         { crJobId = jid
  142         , crSessionId = ""  -- stamped once the delegate returns a session id
  143         , crStartedAt = now
  144         , crEndedAt = Nothing
  145         , crStatus = CrPending
  146         , crError = Nothing
  147         }
  148   -- Persist the run BEFORE firing. The dispatcher's ACK-before-execute
  149   -- principle applies here for the same reason: a completion callback can
  150   -- arrive as soon as the turn starts, and it can only update a row that
  151   -- already exists. (The original code only wrote a row after the delegate
  152   -- returned, so a fast turn's completion matched nothing and the job wedged.)
  153   csRecordRun (csStore scheduler) jid pendingRun
  154   -- Fire the job
  155   eResult <- csFireDelegate scheduler job
  156   case eResult of
  157     Left err -> do
  158       -- Fire failed: move the pending row to Failed
  159       csUpdateRun (csStore scheduler) jid
  160         (applyRunOutcome now (Left err))
  161       -- Check consecutive failures for auto-disable
  162       checkAutoDisable (csStore scheduler) (csNotifyHuman scheduler) now job
  163     Right sessionId -> do
  164       -- Fire succeeded: stamp the session id and move Pending -> Running.
  165       -- If a completion already landed, markRunning leaves the terminal row
  166       -- alone (terminal rows are never regressed).
  167       csUpdateRun (csStore scheduler) jid
  168         (markRunning (SCT.sessionIdText sessionId) now)
  169       -- Update job's lastFire and runCount
  170       updateJobAfterFire scheduler now job
  171       -- One-shot auto-disable
  172       when (cjOneShot job) $ do
  173         mJob <- csRead (csStore scheduler) jid
  174         case mJob of
  175           Just j -> do
  176             now' <- csClock scheduler
  177             csSave (csStore scheduler) j { cjEnabled = False, cjUpdatedAt = now' }
  178           Nothing -> pure ()

  180 -- | Is this run still occupying a concurrency slot at @now@?
  181 --
  182 -- A run counts as active only while it is genuinely in flight. Two things make
  183 -- that false: a terminal status, and a run older than 'cronStaleAfterSeconds'
  184 -- (whose completion callback was lost, or which belonged to a process that
  185 -- died). Without the age bound, a single lost completion wedges scheduling
  186 -- until restart; that is the bug this predicate exists to prevent.
  187 isRunActive :: UTCTime -> CronRun -> Bool
  188 isRunActive now r =
  189   (crStatus r == CrPending || crStatus r == CrRunning)
  190     && diffUTCTime now (crStartedAt r) < fromIntegral cronStaleAfterSeconds
  191 
  192 -- | How long a Pending/Running run is trusted before it is treated as stale.
  193 -- Six hours: long enough that no realistic turn is cut off (a turn that runs
  194 -- longer has its own problems), short enough that a lost completion recovers
  195 -- without an operator restarting the daemon.
  196 cronStaleAfterSeconds :: Int64
  197 cronStaleAfterSeconds = 6 * 60 * 60
  198 
  199 -- | Apply a turn outcome to a run row. Idempotent: an already-terminal row is
  200 -- returned unchanged, so a duplicate completion callback cannot regress a
  201 -- completed run back to failed.
  202 applyRunOutcome :: UTCTime -> Either Text () -> CronRun -> CronRun
  203 applyRunOutcome now outcome r
  204   | isTerminal (crStatus r) = r
  205   | otherwise = case outcome of
  206       Right () -> r { crStatus = CrCompleted, crEndedAt = Just now }
  207       Left e   -> r { crStatus = CrFailed, crEndedAt = Just now, crError = Just e }
  208 
  209 -- | Stamp the session id and move @Pending -> Running@. Leaves a terminal row
  210 -- alone: a completion can arrive before the delegate returns, and the late
  211 -- "Running" write must not overwrite it.
  212 markRunning :: Text -> UTCTime -> CronRun -> CronRun
  213 markRunning sid _now r
  214   | isTerminal (crStatus r) = r
  215   | otherwise = r { crSessionId = sid, crStatus = CrRunning }
  216 
  217 isTerminal :: CronRunStatus -> Bool
  218 isTerminal CrCompleted = True
  219 isTerminal CrFailed    = True
  220 isTerminal _           = False
  221 
  222 -- | Mark every run left @Pending@/@Running@ as failed, returning how many were
  223 -- repaired. Called once at startup: a row still active at process start cannot
  224 -- belong to the current process, so no turn can be in flight for it. Without
  225 -- this, a store wedged by the completion bug would stay wedged after the fix
  226 -- ships.
  227 reconcileStaleRuns :: CronStore -> UTCTime -> IO Int
  228 reconcileStaleRuns store now = do
  229   jobs <- csList store False
  230   counts <- mapM (reconcileJob store now) jobs
  231   pure (sum counts)
  232 
  233 -- | Reconcile one job's run history. Every row still marked Pending/Running is
  234 -- repairable regardless of age: at startup the current process has performed no
  235 -- fires, so *any* active row belongs to a previous process, and no turn can be
  236 -- in flight for it. 'csUpdateRun' maps the transformation over the whole
  237 -- history, which is what repairs a store wedged by repeated fires rather than
  238 -- only its newest row.
  239 reconcileJob :: CronStore -> UTCTime -> CronJob -> IO Int
  240 reconcileJob store now job = do
  241   runs <- csListRuns store (cjId job) maxRetainedRuns
  242   let stale = length (filter isStillActive runs)
  243   when (stale > 0) $
  244     csUpdateRun store (cjId job) (applyRunOutcome now (Left reconcileMessage))
  245   pure stale
  246   where
  247     isStillActive r = crStatus r == CrPending || crStatus r == CrRunning
  248 
  249 -- | The store's run-retention window. Reconciliation inspects this many rows so
  250 -- no row the store would show is left unrepaired.
  251 maxRetainedRuns :: Int
  252 maxRetainedRuns = 100
  253 
  254 reconcileMessage :: Text
  255 reconcileMessage =
  256   "reconciled at startup: run did not complete before the previous process exited"
  257 
  258 -- | Update job's lastFire and runCount after a successful fire.
  259 updateJobAfterFire :: CronScheduler -> UTCTime -> CronJob -> IO ()
  260 updateJobAfterFire scheduler now job = do
  261   csSave (csStore scheduler) job
  262     { cjLastFire = Just now
  263     , cjRunCount = cjRunCount job + 1
  264     , cjUpdatedAt = now
  265     }
  266 
  267 -- | Check if a job should be auto-disabled after consecutive failures.
  268 checkAutoDisable :: CronStore -> (Text -> IO ()) -> UTCTime -> CronJob -> IO ()
  269 checkAutoDisable store notifyHuman now job = do
  270   runs <- csListRuns store (cjId job) maxConsecutiveFailures
  271   let recentFailures = length (filter (\r -> crStatus r == CrFailed) runs)
  272   when (recentFailures >= maxConsecutiveFailures) $ do
  273     -- Auto-disable the job
  274     csSave store job
  275       { cjEnabled = False
  276       , cjUpdatedAt = now
  277       , cjConsecutiveFailures = recentFailures
  278       }
  279     -- Notify operator via SHOW_HUMAN
  280     notifyHuman
  281       ("cron job '" <> cjName job <> "' (id: " <> cronJobIdText (cjId job)
  282        <> ") auto-disabled after " <> T.pack (show maxConsecutiveFailures) <> " consecutive failures")

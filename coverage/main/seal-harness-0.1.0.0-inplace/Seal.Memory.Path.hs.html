    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The memory path model. 'MemoryPath' is a smart-constructed newtype that
    3 -- validates directory-hierarchy paths for memory files under
    4 -- @~\/.seal\/memory\/active\/@. Paths are relative, @\/@-separated, and use
    5 -- the charset @[A-Za-z0-9_-]+@ per segment (the same charset as the former
    6 -- 'MemoryId', extended to allow @\/@ as a directory separator).
    7 --
    8 -- The path the agent provides does NOT include the @.md@ extension — the
    9 -- store appends it. So @"projects\/pureclaw\/architecture"@ maps to
   10 -- @active\/projects\/pureclaw\/architecture.md@ on disk.
   11 --
   12 -- Validation rejects:
   13 --   * empty paths
   14 --   * leading or trailing slashes
   15 --   * @..@ or @.@ segments (path traversal)
   16 --   * segments with leading dots (e.g. @.hidden@)
   17 --   * segments with characters outside @[A-Za-z0-9_-]+@
   18 --   * double slashes (empty segments)
   19 module Seal.Memory.Path
   20   ( MemoryPath (..)
   21   , mkMemoryPath
   22   , isValidMemoryPath
   23   , memoryPathText
   24   , memoryPathSegments
   25   ) where
   26 
   27 import Data.Text (Text)
   28 import Data.Text qualified as T
   29 
   30 -- | Opaque memory path. Smart-constructed via 'mkMemoryPath'; the charset
   31 -- predicate guards every segment.
   32 newtype MemoryPath = MemoryPath Text
   33   deriving stock (Eq, Ord, Show)
   34 
   35 -- | The valid character set for each path segment (same as the former
   36 -- 'MemoryId' charset).
   37 segmentChars :: [Char]
   38 segmentChars = ['A' .. 'Z'] <> ['a' .. 'z'] <> ['0' .. '9'] <> "_-"
   39 
   40 -- | Validate a segment: non-empty, no leading dot, only allowed chars.
   41 isValidSegment :: Text -> Bool
   42 isValidSegment seg =
   43   not (T.null seg)
   44     && T.head seg /= '.'
   45     && T.all (`elem` segmentChars) seg
   46 
   47 -- | Validate a full memory path. See module docs for the rules.
   48 isValidMemoryPath :: Text -> Bool
   49 isValidMemoryPath t =
   50   not (T.null t)
   51     && not (T.isPrefixOf "/" t)
   52     && not (T.isSuffixOf "/" t)
   53     && not ("//" `T.isInfixOf` t)
   54     && all isValidSegment (T.splitOn "/" t)
   55     && not (any (\seg -> seg == ".." || seg == ".") (T.splitOn "/" t))
   56 
   57 -- | Smart constructor for 'MemoryPath'.
   58 mkMemoryPath :: Text -> Either Text MemoryPath
   59 mkMemoryPath t
   60   | isValidMemoryPath t = Right (MemoryPath t)
   61   | otherwise           = Left ("invalid memory path: " <> t)
   62 
   63 -- | Extract the raw 'Text' from a 'MemoryPath'.
   64 memoryPathText :: MemoryPath -> Text
   65 memoryPathText (MemoryPath t) = t
   66 
   67 -- | Split a 'MemoryPath' into its directory segments.
   68 memoryPathSegments :: MemoryPath -> [Text]
   69 memoryPathSegments (MemoryPath t) = T.splitOn "/" t

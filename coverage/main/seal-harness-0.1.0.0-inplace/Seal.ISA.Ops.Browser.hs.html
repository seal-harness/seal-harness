    2 -- | BROWSER_MANAGE (Untrusted): consolidated action-based entry point for
    3 -- all browser automation via the @agent-browser@ CLI
    4 -- (<https://github.com/vercel-labs/agent-browser>). The opcode shells out
    5 -- to @agent-browser@ via 'uioBinExec' — the existing 'UntrustedIO' capability
    6 -- seam for named binaries with validated argv. This module never imports
    7 -- 'System.Process'; all IO goes through the capability handle.
    8 --
    9 -- agent-browser manages its own daemon + Chromium lifecycle. Seal Harness
   10 -- treats it as a black-box CLI that returns text or JSON on stdout. The
   11 -- @--json@ flag is always passed so output is structured. The @--max-output@
   12 -- flag bounds the response size (operator-configurable, default 15000 chars).
   13 module Seal.ISA.Ops.Browser
   14   ( browserManageOp
   15   , BrowserAction (..)
   16   , NetworkFilters (..)
   17   , HarAction (..)
   18   , parseBrowserAction

   36 -- | The action enum — discriminates between browser operations.
   37 data BrowserAction
   38   = BaOpen       Text           -- ^ url
   39   | BaSnapshot
   40   | BaClick      Text           -- ^ ref (@eN) or CSS selector
   41   | BaFill       Text Text      -- ^ ref/selector, text
   42   | BaType       Text Text      -- ^ ref/selector, text
   43   | BaPress      Text           -- ^ key (Enter, Tab, etc.)
   44   | BaScroll     Text           -- ^ direction (up/down/left/right)
   45   | BaRead       (Maybe Text)   -- ^ optional URL
   46   | BaScreenshot (Maybe Text)   -- ^ optional path
   47   | BaEval       Text           -- ^ JavaScript
   48   | BaWait       Text           -- ^ selector or condition
   49   | BaClose
   50   | BaSession
   51   | BaNetworkRequests NetworkFilters  -- ^ network requests with optional filters
   52   | BaNetworkRequest  Text            -- ^ requestId for a single request detail
   53   | BaNetworkHar       HarAction      -- ^ HAR recording start/stop
   54   deriving stock (Eq, Show)
   55 
   56 -- | Optional filter flags for @network requests@.
   57 data NetworkFilters = NetworkFilters
   58   { nfFilter :: Maybe Text   -- ^ @--filter <pattern>@: URL substring or pattern
   59   , nfType   :: Maybe Text   -- ^ @--type <csv>@: resource type (script, image, xhr, fetch, ...)
   60   , nfMethod :: Maybe Text   -- ^ @--method <method>@: HTTP method (GET, POST, ...)
   61   , nfStatus :: Maybe Text   -- ^ @--status <status>@: exact status, family (2xx), or range
   62   , nfClear  :: Bool          -- ^ @--clear@: clear the tracked request log
   63   } deriving stock (Eq, Show)
   64 
   65 -- | Discriminates between HAR start and stop sub-actions.
   66 data HarAction
   67   = HarStart (Maybe Text)  -- ^ start recording; optional @--content@ mode (text, all, none)
   68   | HarStop  (Maybe Text)  -- ^ stop recording; optional output file path
   69   deriving stock (Eq, Show)

   71 -- | The default max-output ceiling (characters). Operator-configurable
   72 -- via config in a future task.
   73 defaultMaxOutput :: Int
   74 defaultMaxOutput = 15000

   76 -- | BROWSER_MANAGE opcode. Input: @{ action: string, ...action-specific fields }@.
   77 browserManageOp :: SecurityPolicy -> Opcode
   78 browserManageOp policy = UntrustedOpcode
   79   { uoName = OpName "BROWSER_MANAGE"
   80   , uoDesc = "Browser automation via agent-browser CLI. Action: open (url), snapshot (interactive elements), click (ref/selector), fill (ref+text), type (ref+text), press (key), scroll (dir), read (url?), screenshot (path?), eval (js), wait (selector), close, session (list current), network-requests (filters?), network-request (requestId), network-har (harAction)."
   81   , uoInSchema = browserManageSchema
   82   , uoOutSchema = object []
   83   , uoAuthorize = \v ->
   84       case parseBrowserAction v of
   85         Left e -> Left e
   86         Right _ -> checkAutonomy
   87   , uoRun = \v -> do
   88       let recorded = object [ "action" .= actionField v ]
   89       case parseBrowserAction v of
   90         Left e -> pure (OpResult [TrpText e] True recorded)
   91         Right action -> do
   92           let mSession = sessionField v
   93               maxOut = defaultMaxOutput
   94           case buildBrowserArgs action mSession maxOut of
   95             Left e -> pure (OpResult [TrpText e] True recorded)
   96             Right (binName, binArgs) -> do
   97               res <- uioBinExec binName binArgs Nothing
   98               pure $ case res of
   99                 Left err -> OpResult [TrpText (renderUntrustedErr err)] True recorded
  100                 Right out -> OpResult [TrpText out] False recorded
  101   }
  102   where
  103     checkAutonomy = case spAutonomy policy of
  104       Deny -> Left "BROWSER_MANAGE denied by autonomy policy"
  105       _    -> Right ()

  107 -- | Parse the action + action-specific fields from the input JSON.
  108 parseBrowserAction :: Value -> Either Text BrowserAction
  109 parseBrowserAction v =
  110   case actionField v of
  111     Nothing -> Left "BROWSER_MANAGE requires {action:string}"
  112     Just a
  113       | a == "open"             -> BaOpen <$> requireField a v "url"
  114       | a == "snapshot"         -> Right BaSnapshot
  115       | a == "click"            -> BaClick <$> refOrSelector a v
  116       | a == "fill"             -> BaFill <$> refOrSelector a v <*> requireField a v "text"
  117       | a == "type"             -> BaType <$> refOrSelector a v <*> requireField a v "text"
  118       | a == "press"            -> BaPress <$> requireField a v "key"
  119       | a == "scroll"           -> BaScroll <$> requireField a v "dir"
  120       | a == "read"             -> Right (BaRead (optionalField v "url"))
  121       | a == "screenshot"       -> Right (BaScreenshot (optionalField v "path"))
  122       | a == "eval"             -> BaEval <$> requireField a v "js"
  123       | a == "wait"             -> BaWait <$> requireField a v "selector"
  124       | a == "close"            -> Right BaClose
  125       | a == "session"          -> Right BaSession
  126       | a == "network-requests" -> Right (BaNetworkRequests (parseNetworkFilters v))
  127       | a == "network-request"  -> BaNetworkRequest <$> requireField a v "requestId"
  128       | a == "network-har"      -> BaNetworkHar <$> parseHarAction v
  129       | otherwise               -> Left ("BROWSER_MANAGE: unknown action \"" <> a <> "\"")

  131 -- | Build the agent-browser argv from the parsed action.
  132 -- Returns (binary name, argv args) or an error if a BinArg fails validation.
  133 buildBrowserArgs :: BrowserAction -> Maybe Text -> Int
  134                 -> Either Text (BinName, [BinArg])
  135 buildBrowserArgs action mSession maxOut = do
  136   binName <- mkBinName "agent-browser"
  137   baseArgs <- traverse mkBinArg baseArgTexts
  138   cmdArgs <- traverse mkBinArg cmdArgTexts
  139   Right (binName, baseArgs ++ cmdArgs)
  140   where
  141     baseArgTexts :: [Text]
  142     baseArgTexts =
  143       [ "--json", "--max-output", T.pack (show maxOut) ]
  144       ++ sessionArgs
  145 
  146     sessionArgs :: [Text]
  147     sessionArgs = case mSession of
  148       Just s -> ["--session", s]
  149       Nothing -> []
  150 
  151     cmdArgTexts :: [Text]
  152     cmdArgTexts = case action of
  153       BaOpen url         -> ["open", url]
  154       BaSnapshot         -> ["snapshot", "-i"]
  155       BaClick ref        -> ["click", ref]
  156       BaFill ref text    -> ["fill", ref, text]
  157       BaType ref text    -> ["type", ref, text]
  158       BaPress key        -> ["press", key]
  159       BaScroll dir       -> ["scroll", dir]
  160       BaRead mUrl        -> case mUrl of
  161                              Just url -> ["read", url]
  162                              Nothing  -> ["read"]
  163       BaScreenshot mPath -> case mPath of
  164                              Just path -> ["screenshot", path]
  165                              Nothing   -> ["screenshot"]
  166       BaEval js          -> ["eval", js]
  167       BaWait sel         -> ["wait", sel]
  168       BaClose            -> ["close"]
  169       BaSession          -> ["session", "list"]
  170       BaNetworkRequests f -> ["network", "requests"] ++ filterArgTexts f
  171       BaNetworkRequest rid -> ["network", "request", rid]
  172       BaNetworkHar hAction -> case hAction of
  173         HarStart mContent -> ["network", "har", "start"] ++ contentArgTexts mContent
  174         HarStop  mPath    -> ["network", "har", "stop"]  ++ maybeToList mPath

  271 -- | Parse the optional filter fields for @network-requests@.
  272 parseNetworkFilters :: Value -> NetworkFilters
  273 parseNetworkFilters v = NetworkFilters
  274   { nfFilter = optionalField v "filter"
  275   , nfType   = optionalField v "type"
  276   , nfMethod = optionalField v "method"
  277   , nfStatus = optionalField v "status"
  278   , nfClear  = boolField v "clear"
  279   }
  280 
  281 -- | Parse the @harAction@ discriminator and its sub-fields for @network-har@.
  282 parseHarAction :: Value -> Either Text HarAction
  283 parseHarAction v =
  284   case optionalField v "harAction" of
  285     Nothing -> Left "BROWSER_MANAGE: network-har requires {harAction:string}"
  286     Just ha
  287       | ha == "start" -> Right (HarStart (optionalField v "content"))
  288       | ha == "stop"  -> Right (HarStop (optionalField v "path"))
  289       | otherwise     -> Left ("BROWSER_MANAGE: network-har requires harAction \"start\" or \"stop\", got \"" <> ha <> "\"")

  291 -- | Build the @--flag value@ argv tokens from 'NetworkFilters'.
  292 filterArgTexts :: NetworkFilters -> [Text]
  293 filterArgTexts f = concat
  294   [ maybe [] (\x -> ["--filter", x]) (nfFilter f)
  295   , maybe [] (\x -> ["--type", x])   (nfType f)
  296   , maybe [] (\x -> ["--method", x]) (nfMethod f)
  297   , maybe [] (\x -> ["--status", x]) (nfStatus f)
  298   , ["--clear" | nfClear f]
  299   ]
  300 
  301 -- | Build the @--content@ argv token from an optional content mode.
  302 contentArgTexts :: Maybe Text -> [Text]
  303 contentArgTexts = maybe [] (\c -> ["--content", c])

  307 actionField :: Value -> Maybe Text
  308 actionField = parseMaybe (withObject "in" (.: "action"))
  309 
  310 sessionField :: Value -> Maybe Text
  311 sessionField v = case parseMaybe (withObject "in" (.:? "session")) v :: Maybe (Maybe Text) of
  312   Just (Just s) | not (T.null s) -> Just s
  313   _                              -> Nothing
  314 
  315 requireField :: Text -> Value -> Text -> Either Text Text
  316 requireField act v field =
  317   case parseMaybe (withObject "in" (.: fromText field)) v of
  318     Nothing -> Left ("BROWSER_MANAGE: " <> act <> " requires {" <> field <> ":string}")
  319     Just s
  320       | T.null s -> Left ("BROWSER_MANAGE: " <> field <> " is empty")
  321       | otherwise -> Right s
  322 
  323 optionalField :: Value -> Text -> Maybe Text
  324 optionalField v field =
  325   case parseMaybe (withObject "in" (.:? fromText field)) v :: Maybe (Maybe Text) of
  326     Just (Just s) | not (T.null s) -> Just s
  327     _                              -> Nothing
  328 
  329 -- | Read a boolean field from the input JSON (defaults to False).
  330 boolField :: Value -> Text -> Bool
  331 boolField v field =
  332   case parseMaybe (withObject "in" (.:? fromText field)) v :: Maybe (Maybe Bool) of
  333     Just (Just b) -> b
  334     _             -> False
  335 
  336 -- | Get the ref or selector for click/fill/type actions.
  337 refOrSelector :: Text -> Value -> Either Text Text
  338 refOrSelector act v =
  339   case optionalField v "ref" of
  340     Just r -> Right r
  341     Nothing -> case optionalField v "selector" of
  342       Just s -> Right s
  343       Nothing -> Left ("BROWSER_MANAGE: " <> act <> " requires {ref:string} or {selector:string}")

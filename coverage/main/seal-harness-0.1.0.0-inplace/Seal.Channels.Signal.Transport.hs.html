   45 -- | The testability seam over signal-cli.
   46 data SignalTransport = SignalTransport
   47   { stReceive :: IO (Either Text Value)
   48     -- ^ Next inbound JSON value (one line from signal-cli's stdout).
   49   , stSend    :: Text -> Text -> IO ()
   50     -- ^ Send a message: recipient, body. Writes a JSON-RPC @send@ frame
   51     -- (fire-and-forget notification — no response is read).
   52   , stSendWithId :: Text -> Text -> IO (Maybe Text)
   53     -- ^ Send a message and return the timestamp from the JSON-RPC
   54     -- response. Blocks until the matching response arrives (with a
   55     -- timeout). 'Nothing' on failure or timeout. The timestamp is the
   56     -- message's identity in the Signal protocol — pass it to
   57     -- 'stEditMessage' to edit or 'stDeleteMessage' to delete.
   58   , stEditMessage :: Text -> Text -> Text -> IO Bool
   59     -- ^ Edit a previously sent message: recipient, timestamp, new
   60     -- content. Sends a JSON-RPC @send@ with @editTimestamp@ set. Returns
   61     -- 'True' on success.
   62   , stDeleteMessage :: Text -> Text -> IO Bool
   63     -- ^ Delete a previously sent message: recipient, timestamp. Calls
   64     -- the JSON-RPC @remoteDelete@ method. Returns 'True' on success.
   65   , stClose   :: IO ()
   66   }

   68 -- | A mock transport backed by a 'TQueue' of inbound 'Value's and an 'IORef'
   69 -- of captured sends. 'stReceive' pops the next inbound (or returns
   70 -- @Left "inbox empty"@); 'stSend' appends @(recipient, body)@ to the
   71 -- capture; 'stClose' is a no-op (idempotent).
   72 mkMockSignalTransport :: [Value] -> IO (SignalTransport, IO [(Text, Text)])
   73 mkMockSignalTransport scripted = do
   74   q <- newTQueueIO
   75   mapM_ (atomically . writeTQueue q) scripted
   76   capRef <- newIORef []
   77   tsRef <- newIORef (1000 :: Int)
   78   let transport = SignalTransport
   79         { stReceive = do
   80             m <- atomically (tryReadTQueue q)
   81             case m of
   82               Just v  -> pure (Right v)
   83               Nothing -> pure (Left "signal inbox empty")
   84         , stSend = \r b -> modifyIORef' capRef ((r, b) :)
   85         , stSendWithId = \_r _b -> do
   86             n <- readIORef tsRef
   87             let n' = n + 1
   88             writeIORef tsRef n'
   89             pure (Just (T.pack (show n')))
   90         , stEditMessage = \_r _ts _content -> pure True
   91         , stDeleteMessage = \_r _ts -> pure True
   92         , stClose = pure ()
   93         }
   94       getCaptured = reverse <$> readIORef capRef
   95   pure (transport, getCaptured)

   97 -- | Spawn @signal-cli --output=json --trust-new-identities=always
   98 -- -u <account> jsonRpc@ as a child process via "System.Process", line-
   99 -- buffered JSON-RPC over stdin/stdout. 'stReceive' reads one line from
  100 -- stdout and decodes it; 'stSend' writes a JSON-RPC @send@ frame to stdin;
  101 -- 'stClose' terminates the child (with a 5s graceful wait). A preflight
  102 -- @signal-cli --version@ probe fails fast if the binary is absent.
  103 --
  104 -- The '<account>' is the smart-constructed validated 'SignalAccount' from
  105 -- "Seal.Signal.Config" — option injection fails to compile. This module
  106 -- takes it as 'Text' here to avoid a cycle (Signal.Config imports nothing
  107 -- from this module; the constructor is enforced at the call site).
  108 mkRealSignalTransport :: Text -> IO (Either Text SignalTransport)
  109 mkRealSignalTransport account = do
  110   versionOk <- probeSignalCli
  111   if not versionOk
  112     then pure (Left "signal-cli not installed or on PATH")
  113     else do
  114       eStarted <- try @IOException $
  115         createProcess
  116           ( (proc "signal-cli"
  117               [ "--output=json"
  118               , "--trust-new-identities=always"
  119               , "-u", T.unpack account
  120               , "jsonRpc"
  121               ])
  122               { std_in = CreatePipe, std_out = CreatePipe, std_err = Inherit }
  123           )
  124       case eStarted of
  125         Left e -> pure (Left ("signal-cli launch failed: " <> T.pack (show e)))
  126         Right (mIn, mOut, _err, ph) -> do
  127           (hIn, hOut) <- case (mIn, mOut) of
  128             (Just a, Just b) -> pure (a, b)
  129             _ -> error "mkRealSignalTransport: pipe creation failed (unreachable)"
  130           hSetBuffering hIn (BlockBuffering Nothing)
  131           hSetBuffering hOut LineBuffering
  132           inbox <- newTQueueIO
  133           readerDead <- newTVarIO False
  134           idRef <- newIORef (0 :: Int)
  135           respMap <- newIORef (Map.empty :: Map.Map Int (MVar (Maybe Value)))
  136           -- Spawn the demux reader thread: reads all stdout lines and
  137           -- routes responses (have id) to waiting callers via MVars,
  138           -- notifications (no id) to the inbox TQueue.
  139           _ <- forkIO (demuxReader hOut inbox respMap readerDead)
  140           -- Serialize all writes to hIn: the three operations (hPutStr +
  141           -- hPutStrLn + hFlush) are not atomic together, so concurrent
  142           -- sends from the forked agent-turn thread (tool-progress
  143           -- bubbles) and the main loop thread (e.g. /tab focus
  144           -- confirmation) can interleave bytes, corrupting JSON-RPC
  145           -- frames. The lock is held only during the write itself, NOT
  146           -- while waiting for a JSON-RPC response (so a stSendWithId
  147           -- call does not block a concurrent stSend).
  148           sendLock <- newMVar ()
  149           let writeFrame :: Value -> IO ()
  150               writeFrame frame = do
  151                 withMVar sendLock $ \_ -> do
  152                   BL.hPutStr hIn (A.encode frame)
  153                   TIO.hPutStrLn hIn ""
  154                   hFlush hIn
  155           let sendRequest :: Value -> IO (Maybe Value)
  156               sendRequest frame = do
  157                 rid <- atomicModifyIORef' idRef (\n -> (n + 1, n))
  158                 mv <- newEmptyMVar
  159                 _ <- atomicModifyIORef' respMap (\m -> (Map.insert rid mv m, m))
  160                 let ridVal = A.Number (fromIntegral rid)
  161                     -- Merge the id into the frame object (frame is always
  162                     -- an A.object, so we use KeyMap.insert).
  163                     framed = case frame of
  164                       A.Object o -> A.Object (KeyMap.insert (Key.fromString "id") ridVal o)
  165                       _ -> frame
  166                 writeFrame framed
  167                 mResult <- timeout 10000000 (takeMVar mv)
  168                 _ <- atomicModifyIORef' respMap (\m -> (Map.delete rid m, m))
  169                 pure (case mResult of
  170                   Just (Just v) -> Just v
  171                   _ -> Nothing)
  172           pure (Right SignalTransport
  173             { stReceive = receiveBlocking inbox readerDead
  174             , stSend = \recipient body -> do
  175                 let frame = A.object
  176                       [ "jsonrpc" A..= ("2.0" :: Text)
  177                       , "method"  A..= ("send" :: Text)
  178                       , "params"  A..= A.object
  179                           [ "recipient" A..= [recipient]
  180                           , "message"   A..= body
  181                           ]
  182                       ]
  183                 writeFrame frame
  184             , stSendWithId = \recipient body -> do
  185                 let frame = A.object
  186                       [ "jsonrpc" A..= ("2.0" :: Text)
  187                       , "method"  A..= ("send" :: Text)
  188                       , "params"  A..= A.object
  189                           [ "recipient" A..= [recipient]
  190                           , "message"   A..= body
  191                           ]
  192                       ]
  193                 mResp <- sendRequest frame
  194                 pure (extractTimestamp =<< mResp)
  195             , stEditMessage = \recipient ts content -> do
  196                 let frame = A.object
  197                       [ "jsonrpc" A..= ("2.0" :: Text)
  198                       , "method"  A..= ("send" :: Text)
  199                       , "params"  A..= A.object
  200                           [ "recipient" A..= [recipient]
  201                           , "message"   A..= content
  202                           , "editTimestamp" A..= (read (T.unpack ts) :: Int)
  203                           ]
  204                       ]
  205                 mResp <- sendRequest frame
  206                 pure (case mResp of
  207                   Just _  -> True
  208                   Nothing -> False)
  209             , stDeleteMessage = \recipient ts -> do
  210                 let frame = A.object
  211                       [ "jsonrpc" A..= ("2.0" :: Text)
  212                       , "method"  A..= ("remoteDelete" :: Text)
  213                       , "params"  A..= A.object
  214                           [ "recipient" A..= [recipient]
  215                           , "targetTimestamp" A..= (read (T.unpack ts) :: Int)
  216                           ]
  217                       ]
  218                 mResp <- sendRequest frame
  219                 pure (case mResp of
  220                   Just _  -> True
  221                   Nothing -> False)
  222             , stClose = do
  223                 _ <- try @IOException (hClose hIn)
  224                 terminateProcess ph
  225                 _ <- timeout 5000000 (waitForProcess ph)
  226                 pure ()
  227             })

  229 -- | The background demux reader: reads all lines from signal-cli's
  230 -- stdout and classifies each:
  231 -- * JSON-RPC response (has @result@ or @error@ + @id@) → route to the
  232 --   waiting caller via the MVar keyed by @id@ in 'respMap'.
  233 -- * JSON-RPC notification (has @method@ + @params@, no @id@) → push to
  234 --   the inbox 'TQueue' (picked up by 'stReceive').
  235 -- On a read error or EOF, the reader sets @readerDead@ and exits (the
  236 -- transport is closing).
  237 demuxReader
  238   :: Handle -> TQueue Value
  239   -> IORef (Map.Map Int (MVar (Maybe Value))) -> TVar Bool
  240   -> IO ()
  241 demuxReader hOut inbox respMap readerDead = go
  242   where
  243     go = do
  244       eLine <- try @IOException (hGetLine hOut)
  245       case eLine of
  246         Left _ -> atomically (writeTVar readerDead True)  -- EOF or error
  247         Right line -> do
  248           case A.decode (BL.fromStrict (TE.encodeUtf8 (T.pack line))) of
  249             Nothing -> go  -- malformed line: skip
  250             Just v -> do
  251               case extractResponseId v of
  252                 Just rid -> do
  253                   -- JSON-RPC response: deliver to the waiting MVar
  254                   m <- readIORef respMap
  255                   case Map.lookup rid m of
  256                     Just mv -> putMVar mv (Just v)
  257                     Nothing -> pure ()  -- stale: no waiting caller
  258                 Nothing -> do
  259                   -- JSON-RPC notification: push to inbox
  260                   atomically (writeTQueue inbox v)
  261           go
  262 
  263 -- | Block until the demux reader's inbox yields a value; return 'Left'
  264 -- only when the reader is dead AND the inbox is drained (EOF). The
  265 -- channel's reader loop treats any 'Left' as fatal, so this must NOT
  266 -- return 'Left' while signal-cli is merely slow to emit — the pre-fix
  267 -- non-blocking variant returned @Left "signal inbox empty"@ the instant
  268 -- the inbox was momentarily empty, killing the channel at startup.
  269 -- Implemented as an STM orElse: @readTQueue@ (blocks while empty) racing
  270 -- the EOF check (@readerDead@ + queue empty → 'Left'). A live reader
  271 -- therefore blocks forever until an envelope arrives; a dead reader
  272 -- delivers any remaining envelopes first, then EOF.
  273 receiveBlocking :: TQueue Value -> TVar Bool -> IO (Either Text Value)
  274 receiveBlocking inbox readerDead =
  275   atomically $
  276     (Right <$> readTQueue inbox)
  277       `orElse` do
  278         dead <- readTVar readerDead
  279         check dead
  280         isEmpty <- isEmptyTQueue inbox
  281         check isEmpty
  282         pure (Left "signal inbox empty")

  284 -- | Extract the @id@ field from a JSON-RPC response. Returns 'Nothing' for
  285 -- notifications (which have no @id@). Pure.
  286 extractResponseId :: Value -> Maybe Int
  287 extractResponseId v =
  288   case v of
  289     A.Object o -> case KeyMap.lookup (Key.fromString "id") o of
  290       Just (A.Number n) -> Just (round n)
  291       _ -> Nothing
  292     _ -> Nothing
  293 
  294 -- | Extract the @result.timestamp@ from a JSON-RPC @send@ response. Returns
  295 -- 'Nothing' if the response is malformed or the field is absent. Pure.
  296 extractTimestamp :: Value -> Maybe Text
  297 extractTimestamp v =
  298   case v of
  299     A.Object o -> case KeyMap.lookup (Key.fromString "result") o of
  300       Just (A.Object ro) -> case KeyMap.lookup (Key.fromString "timestamp") ro of
  301         Just (A.Number n) -> Just (T.pack (show (round n :: Int)))
  302         _ -> Nothing
  303       _ -> Nothing
  304     _ -> Nothing

  306 -- | Preflight @signal-cli --version@. Returns 'True' if the binary is on
  307 -- PATH and exits successfully. Mirrors 'Seal.Security.Vault.Age's
  308 -- @age --version@ preflight.
  309 probeSignalCli :: IO Bool
  310 probeSignalCli = do
  311   r <- try @IOException $
  312         withCreateProcess
  313           ( (proc "signal-cli" ["--version"])
  314               { std_out = NoStream, std_err = NoStream }
  315           ) $ \_ _ _ ph -> waitForProcess ph
  316   pure $ case r of
  317     Right ExitSuccess -> True
  318     _                 -> False

  324 -- | Split a message into chunks of at most 'limit' characters, preferring
  325 -- paragraph boundaries (@\\n\\n@), then line boundaries (@\\n@), hard-cut
  326 -- as a last resort. **Chunks carry their trailing separator** (except the
  327 -- last chunk, which has none), so 'T.concat' of the chunks is identity:
  328 --
  329 -- > T.concat (chunkMessage limit t) == t
  330 --
  331 -- This makes the chunks the literal bytes to send, in order — the receiver
  332 -- concatenates and sees the original text. Pure.
  333 chunkMessage :: Int -> Text -> [Text]
  334 chunkMessage limit t
  335   | limit < 1 = error "chunkMessage: limit must be >= 1"
  336   | T.null t  = []
  337   | otherwise = go t
  338   where
  339     go s
  340       | T.null s       = []
  341       | T.length s <= limit = [s]
  342       | otherwise =
  343           let (chunk, rest) = nextChunk limit s
  344           in chunk : if T.null rest then [] else go rest
  345 
  346 -- | Extract the next chunk (with its trailing separator) and the remaining
  347 -- text. Prefers a paragraph boundary (@\\n\\n@) within the limit, then a
  348 -- line boundary (@\\n@), then a hard cut at the limit.
  349 nextChunk :: Int -> Text -> (Text, Text)
  350 nextChunk limit s =
  351   case findParagraphBreak limit s of
  352     Just n -> T.splitAt n s
  353     Nothing -> case findLineBreak limit s of
  354       Just n -> T.splitAt n s
  355       Nothing -> T.splitAt limit s
  356 
  357 -- | The 1-based cut position (inclusive of the @\\n\\n@ separator) of the
  358 -- last @\\n\\n@ within the first 'limit' characters, or 'Nothing' if there
  359 -- is none. The chunk will then be @take n s@ (which ends in @\\n\\n@).
  360 findParagraphBreak :: Int -> Text -> Maybe Int
  361 findParagraphBreak limit s =
  362   -- Search the first `limit` characters for the rightmost "\n\n".
  363   let window = T.take limit s
  364   in case T.breakOnEnd "\n\n" window of
  365        (pre, _post) | not (T.null pre) -> Just (T.length pre)
  366        _ -> Nothing
  367 
  368 -- | The 1-based cut position (inclusive of the @\\n@ separator) of the last
  369 -- @\\n@ within the first 'limit' characters, or 'Nothing' if there is none.
  370 findLineBreak :: Int -> Text -> Maybe Int
  371 findLineBreak limit s =
  372   let window = T.take limit s
  373   in case T.breakOnEnd "\n" window of
  374        (pre, _post) | not (T.null pre) -> Just (T.length pre)
  375        _ -> Nothing

  381 -- | A parsed inbound signal-cli envelope: the peer-derived conversation id,
  382 -- the sender's user id, and the message body. The conversation id is
  383 -- **server-derived from authenticated transport metadata** (the peer's
  384 -- phone number + UUID), never read from the message body — enforced
  385 -- structurally by 'conversationIdForSignal' taking only the peer fields.
  386 data SignalEnvelope = SignalEnvelope
  387   { seConversationId :: ConversationId
  388   , seSender         :: UserId
  389   , seBody           :: Text
  390   } deriving stock (Eq, Show)
  391 
  392 -- | Derive the server-side 'ConversationId' from the peer's authenticated
  393 -- transport metadata. signal-cli envelopes carry both a phone number
  394 -- (@source@) and a UUID (@sourceUuid@); the conversation id is
  395 -- @sig:<source>:<sourceUuid>@ (or @sig:<source>@ when the UUID is absent).
  396 -- Never reads the message body. A missing or empty @source@ is rejected
  397 -- (a peer phone number is required).
  398 conversationIdForSignal :: Maybe Text -> Maybe Text -> Either Text ConversationId
  399 conversationIdForSignal mSource mUuid =
  400   case mSource of
  401     Nothing        -> Left "signal envelope missing source (peer phone number)"
  402     Just src
  403       | T.null src -> Left "signal envelope source is empty"
  404       | otherwise  -> mkConversationId full
  405       where
  406         full = case mUuid of
  407           Nothing   -> "sig:" <> src
  408           Just uuid -> "sig:" <> src <> ":" <> uuid
  409 
  410 -- | Parse a raw signal-cli JSON value into a 'SignalEnvelope'. Handles both
  411 -- raw envelopes (@{"envelope": {...}}@) and JSON-RPC-wrapped messages
  412 -- (@{"jsonrpc":"2.0","method":"receive","params":{"envelope":{...}}}@).
  413 -- Returns 'Left' on a malformed value, a missing peer field, or a
  414 -- conversation-id smart-constructor failure. The body is taken from the
  415 -- envelope's @dataMessage.message@; a @conversationId@ key in the body is
  416 -- IGNORED (the conversation id is always server-derived via
  417 -- 'conversationIdForSignal').
  418 parseSignalEnvelope :: Value -> Either Text SignalEnvelope
  419 parseSignalEnvelope v = do
  420   env <- unwrapEnvelope v
  421   source <- fieldText "source" env
  422   let mUuid = fieldTextMaybe "sourceUuid" env
  423   cid <- conversationIdForSignal (Just source) mUuid
  424   uid <- case mkUserId source of
  425     Right u  -> Right u
  426     Left err -> Left ("signal sender not a valid UserId: " <> err)
  427   let body = extractBody env
  428   Right SignalEnvelope
  429     { seConversationId = cid
  430     , seSender         = uid
  431     , seBody           = body
  432     }

  434 -- | Unwrap a raw envelope (@{"envelope": {...}}@) or a JSON-RPC-wrapped
  435 -- message (@{"params": {"envelope": {...}}}@) to the inner envelope object.
  436 -- Returns 'Left' if neither shape is present or the value is not an object.
  437 unwrapEnvelope :: Value -> Either Text Value
  438 unwrapEnvelope v =
  439   case v of
  440     A.Object o ->
  441       case KeyMap.lookup (Key.fromString "envelope") o of
  442         Just env -> Right env
  443         Nothing -> case KeyMap.lookup (Key.fromString "params") o of
  444           Just (A.Object p) -> case KeyMap.lookup (Key.fromString "envelope") p of
  445             Just env -> Right env
  446             Nothing  -> Left "signal envelope: no envelope in params"
  447           _ -> Left "signal envelope: no envelope and no params"
  448     _ -> Left "signal envelope: not an object"
  449 
  450 -- | Extract a required text field from an object, failing on absence or
  451 -- non-text value.
  452 fieldText :: Text -> Value -> Either Text Text
  453 fieldText key v =
  454   case v of
  455     A.Object o -> case KeyMap.lookup (Key.fromString (T.unpack key)) o of
  456       Just (A.String t) -> Right t
  457       _ -> Left ("signal envelope missing or non-text field: " <> key)
  458     _ -> Left ("signal envelope field " <> key <> ": not an object")
  459 
  460 -- | Extract an optional text field from an object.
  461 fieldTextMaybe :: Text -> Value -> Maybe Text
  462 fieldTextMaybe key v = do
  463   inner <- fieldValueMaybe key v
  464   case inner of
  465     A.String t -> Just t
  466     _ -> Nothing

  468 -- | Extract an optional sub-value from an object.
  469 fieldValueMaybe :: Text -> Value -> Maybe Value
  470 fieldValueMaybe key v =
  471   case v of
  472     A.Object o -> KeyMap.lookup (Key.fromString (T.unpack key)) o
  473     _ -> Nothing

  475 -- | Extract the message body from @dataMessage.message@. Empty when absent
  476 -- (a non-data envelope, e.g. a receipt — the caller drops these).
  477 extractBody :: Value -> Text
  478 extractBody v = fromMaybe "" $ do
  479   dm <- fieldValueMaybe "dataMessage" v
  480   fieldTextMaybe "message" dm

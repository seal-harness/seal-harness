   42 -- | The runnable action a successfully-parsed command performs on its channel.
   43 -- Returns an optional follow-up turn text: 'Nothing' for commands that do not
   44 -- trigger a turn (the vast majority); @'Just' t@ when the command wants the
   45 -- agent loop to run a turn with text @t@ immediately after the command
   46 -- completes (e.g. @\/skill load@ after writing the skill body to
   47 -- @conversation.jsonl@). The follow-up turn is forked by each channel's
   48 -- dispatch site so the channel loop continues reading input while the LLM
   49 -- runs.
   50 newtype CommandAction = CommandAction { runCommandAction :: ChannelCaps -> IO (Maybe Text) }
   51 
   52 -- | Construct a 'CommandAction' from an @IO ()@ action that never triggers a
   53 -- follow-up turn. This is the common case — most commands produce output via
   54 -- 'ccSend' and return 'Nothing'. Commands that DO want to trigger a turn
   55 -- (e.g. @\/skill load@) construct 'CommandAction' directly and return
   56 -- @'Just' text@.
   57 commandAction :: (ChannelCaps -> IO ()) -> CommandAction
   58 commandAction act = CommandAction (\caps -> void (act caps) >> pure Nothing)
   59 
   60 -- | Run a 'CommandAction' and return the optional follow-up-turn signal.
   61 -- Convenience alias for 'runCommandAction'.
   62 runCommandActionMaybe :: CommandAction -> ChannelCaps -> IO (Maybe Text)
   63 runCommandActionMaybe = runCommandAction

   74 -- | NOTE: /help is NOT a registered spec — it is a meta-operation handled by
   75 -- Seal.Command.Help / Seal.Command.Parse over the Registry (avoids the
   76 -- registry-needs-itself knot). Feature modules build their own CommandSpec
   77 -- and the startup wiring assembles the Registry.
   78 newtype Registry = Registry { registrySpecs :: [CommandSpec] }
   79 
   80 mkRegistry :: [CommandSpec] -> Registry

   83 -- | Case-insensitive lookup by primary name or any alias.
   84 lookupSpec :: Registry -> CommandName -> Maybe CommandSpec
   85 lookupSpec (Registry specs) (CommandName needle) =
   86   find matchesAny specs
   87   where
   88     lower = T.toCaseFold needle
   89     nameEq (CommandName n) = T.toCaseFold n == lower
   90     matchesAny spec = nameEq (csName spec) || any nameEq (csAliases spec)

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | A strict, bounded DER decoder for the one structure WebAuthn needs: the
    3 -- @Ecdsa-Sig-Value@ that an authenticator returns for an ES256 assertion,
    4 --
    5 -- > SEQUENCE { r INTEGER, s INTEGER }
    6 --
    7 -- crypton's 'Crypto.PubKey.ECC.ECDSA.verify' takes a pair of 'Integer's, and
    8 -- nothing in the existing dependency set parses DER, so this module is the
    9 -- third small decoder the design calls for.
   10 --
   11 -- Bounds: the whole input must be consumed, lengths must be definite and
   12 -- satisfiable by the remaining bytes, and both integers must be non-empty and
   13 -- non-negative. No partial functions.
   14 module Seal.Security.WebAuthn.Der
   15   ( decodeEcdsaSignature
   16   , maxInput
   17   ) where
   18 
   19 import Data.Bits (shiftL, (.&.))
   20 import Data.ByteString (ByteString)
   21 import Data.ByteString qualified as BS
   22 import Data.Text (Text)
   23 import Data.Word (Word8)
   24 
   25 -- | Upper bound on a signature we will parse (512 bytes is generous for two
   26 -- 256-bit integers).
   27 maxInput :: Int
   28 maxInput = 512
   29 
   30 -- | Parse an ECDSA signature into @(r, s)@.
   31 decodeEcdsaSignature :: ByteString -> Either Text (Integer, Integer)
   32 decodeEcdsaSignature bs
   33   | BS.length bs > maxInput = Left "DER input too large"
   34   | BS.null bs              = Left "empty DER input"
   35   | otherwise = do
   36       (tag, rest0) <- take1 bs
   37       if tag /= 0x30
   38         then Left "DER: expected a SEQUENCE (0x30)"
   39         else do
   40           (len, rest1) <- takeLength rest0
   41           (body, rest2) <- takeExact len rest1
   42           if not (BS.null rest2)
   43             then Left "DER: trailing bytes after SEQUENCE"
   44             else do
   45               (r, rest3) <- takeInteger body
   46               (s, rest4) <- takeInteger rest3
   47               if BS.null rest4
   48                 then Right (r, s)
   49                 else Left "DER: trailing bytes after the integers"
   50 
   51 take1 :: ByteString -> Either Text (Word8, ByteString)
   52 take1 bs = case BS.uncons bs of
   53   Nothing      -> Left "DER: truncated input"
   54   Just (x, xs) -> Right (x, xs)
   55 
   56 takeLength :: ByteString -> Either Text (Int, ByteString)
   57 takeLength bs = do
   58   (b0, rest) <- take1 bs
   59   if b0 < 0x80
   60     then Right (fromIntegral b0, rest)
   61     else if b0 == 0x80
   62       then Left "DER: indefinite lengths are not accepted"
   63       else do
   64         let n = fromIntegral (b0 .&. 0x7f)
   65         if n == 0 || n > 3
   66           then Left "DER: unsupported length width"
   67           else do
   68             (raw, rest') <- takeExact n rest
   69             Right (foldl (\acc w -> acc `shiftL` 8 + fromIntegral w) 0 (BS.unpack raw), rest')
   70 
   71 takeExact :: Int -> ByteString -> Either Text (ByteString, ByteString)
   72 takeExact n bs
   73   | n < 0 = Left "DER: negative length"
   74   | n > BS.length bs = Left "DER: length exceeds remaining input"
   75   | otherwise = Right (BS.splitAt n bs)
   76 
   77 takeInteger :: ByteString -> Either Text (Integer, ByteString)
   78 takeInteger bs = do
   79   (tag, rest0) <- take1 bs
   80   if tag /= 0x02
   81     then Left "DER: expected an INTEGER (0x02)"
   82     else do
   83       (len, rest1) <- takeLength rest0
   84       if len <= 0
   85         then Left "DER: empty INTEGER"
   86         else do
   87           (raw, rest2) <- takeExact len rest1
   88           case BS.uncons raw of
   89             Nothing -> Left "DER: empty INTEGER"
   90             Just (b0, _)
   91               -- WebAuthn ES256 signatures are always positive; a negative r or
   92               -- s is a malformed signature, not a value to reinterpret.
   93               | b0 .&. 0x80 /= 0 -> Left "DER: negative INTEGER in an ECDSA signature"
   94               | otherwise ->
   95                   Right (foldl (\acc w -> acc `shiftL` 8 + fromIntegral w) 0 (BS.unpack raw), rest2)

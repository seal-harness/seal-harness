   43 data PathError
   44   = PathEscapesWorkspace FilePath
   45   | PathIsBlocked Text
   46   | PathDoesNotExist FilePath
   47   | PathInsecureMode FilePath
   48   deriving stock (Eq, Show)

   50 blockedNames :: [FilePath]
   51 blockedNames = [".env", ".ssh", ".gnupg", ".netrc", ".seal"]

   53 -- | Lexically collapse @.@ and @..@ segments without touching the filesystem.
   54 -- A @..@ removes the preceding ordinary segment; if there is none (or the
   55 -- predecessor is itself an un-collapsible @..@) the @..@ is retained, so an
   56 -- attempt to climb above the root survives into the containment check and is
   57 -- rejected there rather than being silently normalised away.
   58 lexicalCollapse :: [FilePath] -> [FilePath]
   59 lexicalCollapse = reverse . foldl' step []
   60   where
   61     step acc "." = acc
   62     step (x : xs) ".." | x /= ".." = xs
   63     step acc seg = seg : acc

   65 -- | Validate that a requested path is safely confined within the workspace.
   66 --
   67 -- Steps (in order):
   68 --   1. Reject if any segment of the requested path is a blocked name.
   69 --   2. Lexically collapse @..@/@.@ in the joined path (no filesystem access)
   70 --      and reject with 'PathEscapesWorkspace' if it leaves the root. This is a
   71 --      purely lexical determination, so a @..@ escape is caught BEFORE any
   72 --      filesystem call — independent of whether the target exists.
   73 --   3. Canonicalize the joined path (follows symlinks); a missing final
   74 --      component makes @canonicalizePath@ throw, caught as 'PathDoesNotExist'.
   75 --   4. Re-run the containment check on the canonical path to catch symlinks
   76 --      that resolve outside the workspace.
   77 --   5. Confirm the path exists; return 'SafePath' on success.
   78 --
   79 -- Containment check approach: component-wise prefix test via
   80 -- @splitDirectories canonRoot \`isPrefixOf\` splitDirectories candidate@,
   81 -- which correctly rejects @\/ws\/barbaz@ as not inside @\/ws\/bar@.
   82 mkSafePath :: WorkspaceRoot -> FilePath -> IO (Either PathError SafePath)
   83 mkSafePath (WorkspaceRoot root) requested = do
   84   canonRoot <- canonicalizePath root
   85   -- 1. Blocked-name check (pure, on the segments of the requested path).
   86   if any (`elem` blockedNames) (splitDirectories requested)
   87     then pure $ Left $ PathIsBlocked $ T.pack $ "path touches a blocked location: " <> requested
   88     else do
   89       let joined = if isAbsolute requested then requested else canonRoot </> requested
   90           rootDirs = splitDirectories canonRoot
   91           -- 2. Lexically resolve `..`/`.` and check containment, no filesystem access.
   92           lexicalDirs = lexicalCollapse (splitDirectories joined)
   93       if not (rootDirs `isPrefixOf` lexicalDirs)
   94         then pure $ Left $ PathEscapesWorkspace (joinPath lexicalDirs)
   95         else resolveAndCheck rootDirs joined (joinPath lexicalDirs)
   96 
   97   where
   98     resolveAndCheck rootDirs joined lexical = do
   99       -- 3. Canonicalize (follows symlinks). Throws IOException if path is missing.
  100       canonResult <- try (canonicalizePath joined) :: IO (Either IOException FilePath)
  101       case canonResult of
  102         Left _ -> pure $ Left $ PathDoesNotExist lexical
  103         Right canon ->
  104           -- 4. Post-canonicalization containment check: catches symlink escapes.
  105           if not (rootDirs `isPrefixOf` splitDirectories canon)
  106             then pure $ Left $ PathEscapesWorkspace canon
  107             else do
  108               exists <- doesPathExist canon
  109               if not exists
  110                 then pure $ Left $ PathDoesNotExist canon
  111                 else pure $ Right $ SafePath canon
  112 
  113 -- | Like 'mkSafePath' but allows the final path component to NOT exist
  114 -- (for FILE_WRITE: the file may be created fresh). Steps 1-2 (blocked-name
  115 -- + lexical collapse + containment) are identical. Step 3 canonicalizes
  116 -- the PARENT (which must exist), then re-joins the final component. Step 4
  117 -- confirms the parent is contained; the final component's existence is
  118 -- irrelevant (it may or may not exist).
  119 mkSafePathForWrite :: WorkspaceRoot -> FilePath -> IO (Either PathError SafePath)
  120 mkSafePathForWrite (WorkspaceRoot root) requested = do
  121   canonRoot <- canonicalizePath root
  122   if any (`elem` blockedNames) (splitDirectories requested)
  123     then pure $ Left $ PathIsBlocked $ T.pack $ "path touches a blocked location: " <> requested
  124     else do
  125       let joined = if isAbsolute requested then requested else canonRoot </> requested
  126           rootDirs = splitDirectories canonRoot
  127           lexicalDirs = lexicalCollapse (splitDirectories joined)
  128       if not (rootDirs `isPrefixOf` lexicalDirs)
  129         then pure $ Left $ PathEscapesWorkspace (joinPath lexicalDirs)
  130         else do
  131           -- Canonicalize the parent (must exist), then re-join the final
  132           -- component. If the parent doesn't exist, the canonicalization
  133           -- fails → PathDoesNotExist (the parent must exist for a write).
  134           let lexicalPath = joinPath lexicalDirs
  135           case splitDirectories lexicalPath of
  136             [] -> pure $ Left $ PathEscapesWorkspace lexicalPath
  137             [_] ->
  138               -- The path is a single component (the root itself); writing
  139               -- the root is not meaningful, but the root exists.
  140               resolveParentAndCheck rootDirs lexicalPath canonRoot
  141             segments ->
  142               let parentLex = joinPath (init segments)
  143               in resolveParentAndCheck rootDirs lexicalPath
  144                    =<< canonicalizePath parentLex
  145   where
  146     resolveParentAndCheck rootDirs lexicalPath canonParent = do
  147       -- Step 4: confirm the canonical parent is contained.
  148       if not (rootDirs `isPrefixOf` splitDirectories canonParent)
  149         then pure $ Left $ PathEscapesWorkspace canonParent
  150         else do
  151           parentExists <- doesPathExist canonParent
  152           if not parentExists
  153             then pure $ Left $ PathDoesNotExist canonParent
  154             else pure $ Right $ SafePath lexicalPath

  160 -- | Lexically validate a workspace-relative path for the REMOTE untrusted
  161 -- executor. This is the remote-safe counterpart of 'mkSafePath'/'mkSafePathForWrite':
  162 -- it performs ONLY the lexical confinement steps (blocked-name check +
  163 -- '..'/@.@ collapse + component-wise containment under the root) and skips
  164 -- the local-FS 'canonicalizePath' (the file lives on the remote machine;
  165 -- the harness cannot canonicalize it locally). The validated 'SafePath' is
  166 -- the lexically-resolved path anchored under the (remote) workspace root.
  167 --
  168 -- Use this for all remote-arm 'UntrustedIO' file methods. The returned
  169 -- 'SafePath' is what gets passed to the SSH call as a single argv element
  170 -- after @--@, so no shell on the remote ever interprets it as a flag.
  171 --
  172 -- Pure: no IO, no symlink resolution. Symlink confinement on the remote is
  173 -- the remote OS's responsibility (the workspace root is the agent's
  174 -- confined directory; the SSH user has no access outside it by policy).
  175 mkSafePathRemote :: WorkspaceRoot -> FilePath -> Either PathError SafePath
  176 mkSafePathRemote (WorkspaceRoot root) requested =
  177   if any (`elem` blockedNames) (splitDirectories requested)
  178     then Left $ PathIsBlocked $ T.pack $ "path touches a blocked location: " <> requested
  179     else
  180       let joined      = if isAbsolute requested then requested else root </> requested
  181           rootDirs    = splitDirectories root
  182           lexicalDirs = lexicalCollapse (splitDirectories joined)
  183       in if not (rootDirs `isPrefixOf` lexicalDirs)
  184            then Left $ PathEscapesWorkspace (joinPath lexicalDirs)
  185            else Right $ SafePath (joinPath lexicalDirs)

  189 --
  190 -- These are the same as 'mkSafePath' / 'mkSafePathForWrite' /
  191 -- 'mkSafePathRemote' but with one difference: when the requested path is
  192 -- ABSOLUTE, the workspace-containment check is SKIPPED. The blocked-name
  193 -- check, lexical '..'/@.@ collapse, canonicalization (local arm), and
  194 -- existence checks all still apply. Relative paths are confined exactly as
  195 -- in the original functions.

  197 -- Use these for the agent-facing FILE_READ / FILE_WRITE / FILE_PATCH
  198 -- opcodes so an agent can operate on files outside the session workdir
  199 -- (e.g. persistent files across sessions). Internal harness operations
  200 -- (WorkdirFs, skills, memory, agent defs) continue to use the confining
  201 -- originals.

  204 -- | Like 'mkSafePath' but allows absolute paths to escape the workspace.
  205 -- Relative paths are confined as usual; absolute paths are canonicalized
  206 -- and existence-checked but NOT workspace-confined.
  207 mkSafePathAllowAbs :: WorkspaceRoot -> FilePath -> IO (Either PathError SafePath)
  208 mkSafePathAllowAbs (WorkspaceRoot root) requested = do
  209   canonRoot <- canonicalizePath root
  210   if any (`elem` blockedNames) (splitDirectories requested)
  211     then pure $ Left $ PathIsBlocked $ T.pack $ "path touches a blocked location: " <> requested
  212     else if isAbsolute requested
  213       then resolveAbs
  214       else do
  215         let joined = canonRoot </> requested
  216             rootDirs = splitDirectories canonRoot
  217             lexicalDirs = lexicalCollapse (splitDirectories joined)
  218         if not (rootDirs `isPrefixOf` lexicalDirs)
  219           then pure $ Left $ PathEscapesWorkspace (joinPath lexicalDirs)
  220           else resolveAndCheck rootDirs joined (joinPath lexicalDirs)
  221   where
  222     resolveAbs = do
  223       canonResult <- try (canonicalizePath requested) :: IO (Either IOException FilePath)
  224       case canonResult of
  225         Left _ -> pure $ Left $ PathDoesNotExist requested
  226         Right canon -> do
  227           exists <- doesPathExist canon
  228           if not exists
  229             then pure $ Left $ PathDoesNotExist canon
  230             else pure $ Right $ SafePath canon
  231     resolveAndCheck rootDirs joined lexical = do
  232       canonResult <- try (canonicalizePath joined) :: IO (Either IOException FilePath)
  233       case canonResult of
  234         Left _ -> pure $ Left $ PathDoesNotExist lexical
  235         Right canon ->
  236           if not (rootDirs `isPrefixOf` splitDirectories canon)
  237             then pure $ Left $ PathEscapesWorkspace canon
  238             else do
  239               exists <- doesPathExist canon
  240               if not exists
  241                 then pure $ Left $ PathDoesNotExist canon
  242                 else pure $ Right $ SafePath canon

  291           parentExists <- doesPathExist canonParent

  296 -- | Like 'mkSafePathRemote' but allows absolute paths to escape the
  297 -- workspace. Relative paths are confined as usual; absolute paths are
  298 -- lexically collapsed and returned as-is (no containment check). Pure.
  299 mkSafePathRemoteAllowAbs :: WorkspaceRoot -> FilePath -> Either PathError SafePath
  300 mkSafePathRemoteAllowAbs (WorkspaceRoot root) requested
  301   | any (`elem` blockedNames) (splitDirectories requested)
  302   = Left $ PathIsBlocked $ T.pack $ "path touches a blocked location: " <> requested
  303   | isAbsolute requested
  304   = Right $ SafePath (joinPath (lexicalCollapse (splitDirectories requested)))
  305   | otherwise
  306   = let joined      = root </> requested
  307         rootDirs    = splitDirectories root
  308         lexicalDirs = lexicalCollapse (splitDirectories joined)
  309     in if not (rootDirs `isPrefixOf` lexicalDirs)
  310          then Left $ PathEscapesWorkspace (joinPath lexicalDirs)
  311          else Right $ SafePath (joinPath lexicalDirs)

  320 -- | Create (mkdir -p) and harden (chmod 0700) a keys directory, returning a

  328 -- | An absolute path that has been verified to be safely confined within a
  329 -- 'KeysRoot' directory (no @..@ escape, no symlink escape, and — if the file
  330 -- already exists — owned by the effective user with mode 0600 or 0400). The
  331 -- constructor is intentionally not exported; obtain a value via 'mkSafeKeyPath'.
  332 newtype SafeKeyPath = SafeKeyPath FilePath

  342 -- | Validate that a requested path is safely confined within the 'KeysRoot'.
  343 --
  344 -- Steps (in order):
  345 --   1. Lexically collapse @..@\/@.@ and check containment under the root —
  346 --      identical to 'mkSafePath' (reuses 'lexicalCollapse' + component-wise
  347 --      @splitDirectories@ prefix check).
  348 --   2. Canonicalize (follows symlinks); re-run the containment check to catch
  349 --      symlink escapes.
  350 --   3. If the target does not yet exist (key to be written later): return
  351 --      @Right@ with the lexically-resolved path.
  352 --   4. If the target exists: require @fileOwner == getEffectiveUserID@ and
  353 --      @fileMode & 0o777 ∈ {0o600, 0o400}@; otherwise return
  354 --      @Left (PathInsecureMode path)@.
  355 mkSafeKeyPath :: KeysRoot -> FilePath -> IO (Either PathError SafeKeyPath)
  356 mkSafeKeyPath (KeysRoot root) requested = do
  357   canonRoot <- canonicalizePath root
  358   let joined = if isAbsolute requested then requested else canonRoot </> requested
  359       rootDirs = splitDirectories canonRoot
  360       lexicalDirs = lexicalCollapse (splitDirectories joined)
  361   if not (rootDirs `isPrefixOf` lexicalDirs)
  362     then pure $ Left $ PathEscapesWorkspace (joinPath lexicalDirs)
  363     else do
  364       canonResult <- try (canonicalizePath joined) :: IO (Either IOException FilePath)
  365       case canonResult of
  366         Left _ ->
  367           -- Path does not exist yet — allowed; the caller will create it.
  368           pure $ Right $ SafeKeyPath (joinPath lexicalDirs)
  369         Right canon ->
  370           if not (rootDirs `isPrefixOf` splitDirectories canon)
  371             then pure $ Left $ PathEscapesWorkspace canon
  372             else do
  373               exists <- doesPathExist canon
  374               if not exists
  375                 -- canonicalizePath resolved the parent but the file is absent —
  376                 -- still allowed; the caller will create it.
  377                 then pure $ Right $ SafeKeyPath canon
  378                 else checkSecurity canon
  379   where
  380     checkSecurity canon = do
  381       status <- getFileStatus canon
  382       euid <- getEffectiveUserID
  383       let owner = fileOwner status
  384           mode = fileMode status `intersectFileModes` 0o777
  385       if (owner /= euid) || (mode `notElem` [0o600, 0o400])
  386         then pure $ Left $ PathInsecureMode canon
  387         else pure $ Right $ SafeKeyPath canon

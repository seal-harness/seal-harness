   30 -- | Program information for 'runWithConfiguration'. Provides @--config-file@,
   31 -- @--print-config@, and @--help@ automatically.

   36 -- | Dispatch on the selected command, running it in 'App'. The 'SealLogger'
   37 -- is built once via 'withSealLogger' (a bracket that closes the scribe on
   38 -- exit) and threaded into 'Env' so 'App'-level 'logMsg' and 'IO'-level
   39 -- 'logIO' reach the same scribe.
   40 dispatch :: Config -> IO ()
   41 dispatch cfg =
   42   withSealLogger (view config_logLevel cfg) (Just "seal.log") $ \logger -> do
   43     setGlobalLogger logger
   44     env <- mkEnv logger cfg
   45     runApp env $ case _config_command cfg of
   46       CommandNoOp   -> pure ()
   47       CommandTui autonomy -> liftIO (Seal.Tui.runTui autonomy logger)
   48       CommandServe autonomy mStaticDir noEnrollToken -> liftIO (Seal.Command.Serve.runServeMain autonomy mStaticDir noEnrollToken logger)
   49       CommandAccessControl action -> do
   50         -- The command returns its code; exiting stays here so every code is
   51         -- assertable in-process (design §12).
   52         code <- liftIO (Seal.Command.AccessControl.runAccessControl action)
   53         liftIO (exitWith code)
   54       CommandGenOpenApi -> liftIO (BL.hPutStr stdout Seal.Gateway.OpenApi.encodeOpenApi)
   55 
   56 -- | Map the process arguments so that an empty argument list behaves as if
   57 -- @--help@ was passed. Running @seal@ with no arguments should print usage
   58 -- rather than silently doing nothing. Any non-empty argument list is passed
   59 -- through unchanged.
   60 withDefaultArgs :: [String] -> [String]
   61 withDefaultArgs [] = ["--help"]
   62 withDefaultArgs as = as

   64 -- | Raise the soft FD limit to the hard limit. The default macOS soft limit
   65 -- is 256, which the gateway's @buildListsSnapshot@ hot path blows through
   66 -- (hundreds of small files per request, historically via lazy IO that left
   67 -- handles open for the GC). Even with strict reads, raising the soft limit
   68 -- to the hard ceiling is cheap defense-in-depth against FD exhaustion. The
   69 -- raise is best-effort: a failure (e.g. a sandboxed runner that blocks
   70 -- @setrlimit@) is swallowed so the binary still starts.
   71 raiseFdLimit :: IO ()
   72 raiseFdLimit = do
   73   limits <- getResourceLimit ResourceOpenFiles
   74   case hardLimit limits of
   75     ResourceLimit _hard | softLimit limits /= hardLimit limits ->
   76       setResourceLimit ResourceOpenFiles limits { softLimit = hardLimit limits }
   77     _ -> pure ()

   79 -- | Entry point: parse defaults + config file + CLI flags, then dispatch.
   80 -- With no arguments, fall back to @--help@.
   81 appMain :: IO ()
   82 appMain = do
   83   raiseFdLimit
   84   args <- getArgs
   85   withArgs (withDefaultArgs (reorderGlobalArgs args)) (CUtils.runWithConfiguration programInfo dispatch)

   87 -- | Move global options (those defined at the @pConfig@ level, outside any
   88 -- @hsubparser@) before the subcommand keyword. @Options.Applicative@'s
   89 -- @hsubparser@ creates a hard parser boundary: once the subcommand keyword
   90 -- (e.g. @serve@) is seen, only subcommand-specific options are recognised.
   91 -- A global option like @--log-level@ that appears after @serve@ is rejected.
   92 --
   93 -- This preprocessing lets @make serve ARGS="--yolo --log-level debug
   94 -- --no-enrollment-token"@ work: @--log-level@ is hoisted before @serve@,
   95 -- and the subcommand options (@--yolo@, @--no-enrollment-token@) stay after.
   96 -- Options that are already before the subcommand are left in place.
   97 -- @--flag=value@ syntax is a single token and passes through unchanged
   98 -- (it does not match the bare @--flag@ name).
   99 reorderGlobalArgs :: [String] -> [String]
  100 reorderGlobalArgs args =
  101   case break isSubcommand args of
  102     (_, []) -> args
  103     (pre, cmd : post) -> pre ++ hoistGlobals post ++ [cmd] ++ stripGlobals post
  104   where
  105     subcommands = ["serve", "tui", "access-control", "gen-openapi"]
  106     isSubcommand s = s `elem` subcommands
  107 
  108     -- Global option names (defined in pConfig, outside any hsubparser). Each
  109     -- takes exactly one argument. The gateway's bind host/port live in
  110     -- security.toml, so there are no server flags to hoist.
  111     globalOpts = ["--log-level"]
  112     isGlobalOpt s = s `elem` globalOpts
  113 
  114     -- Extract global option pairs (flag + value) from the post-subcommand
  115     -- args, preserving order, for hoisting before the subcommand keyword.
  116     hoistGlobals = go []
  117       where
  118         go acc [] = reverse acc
  119         go acc (f : v : rest) | isGlobalOpt f = go (v : f : acc) rest
  120         go acc (_ : rest) = go acc rest
  121 
  122     -- Remove global option pairs from the post-subcommand args, leaving
  123     -- only subcommand-specific options.
  124     stripGlobals = go
  125       where
  126         go [] = []
  127         go (f : _v : rest) | isGlobalOpt f = go rest
  128         go (x : rest) = x : go rest

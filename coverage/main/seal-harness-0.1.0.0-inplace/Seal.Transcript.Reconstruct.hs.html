    2 -- | Pure reconstruction from the new indexed transcript format
    3 -- (@conversation.jsonl@ + @entries.jsonl@) back to the old
    4 -- 'TranscriptEntry' stream. The reconstructed entries are byte-identical to
    5 -- what the old @transcript.jsonl@ format would have stored, modulo the
    6 -- intentional uncapping improvement (the new format stores full messages
    7 -- where the old format may have size-capped payloads).
    8 --
    9 -- A 'Request' entry @i@ is reconstructed with:
   10 --
   11 -- * @messages = conversation[0 .. convLen_i)@ (the prefix of conversation lines
   12 --   in effect at that turn);
   13 -- * the left-folded effective envelope at @i@ (model / system / tools /
   14 --   toolChoice / maxTokens), wrapped into a 'CompletionRequest'-shaped
   15 --   'Value';
   16 -- * @teModel@ = the effective envelope's model at @i@ (the model that
   17 --   generated the entry's messages — the web transcript view labels each
   18 --   LLM message with it);
   19 -- * the same direction + meta as the stored entry.
   20 --
   21 -- A 'Response' entry is reconstructed by re-inserting the assistant content
   22 -- blocks from @conversation.jsonl@ (the lines between the prior request's
   23 -- @convLen@ and this response's @convLen@) into the stored response envelope.
   24 --
   25 -- 'EKHarness' and 'EKCompaction' events carry no envelope; they round-trip
   26 -- their metadata + a 'Request'/'Response' direction derived from their kind
   27 -- (a harness turn is a 'Request' from the user side; a compaction is a
   28 -- boundary marker with 'Request' direction).
   29 module Seal.Transcript.Reconstruct
   30   ( reconstruct
   31   , reconstructStreaming
   32   , reconstructStreamingPage
   33   ) where

   49 -- | Maximum conversation lines to read per entry when reconstructing.
   50 -- Entries whose delta (erConvLen - priorConvLen) exceeds this are
   51 -- truncated to the last N messages — the most recent ones, which are
   52 -- what the frontend displays. This bounds total memory to
   53 -- pageEntries × maxLinesPerEntry × avgLineSize regardless of the
   54 -- conversation's total size or entry distribution. Without this cap, a
   55 -- session where the last 49 entries span 709K of 724K conversation lines
   56 -- would load ~3.6GB into memory, causing OOM.
   57 maxLinesPerEntry :: Int
   58 maxLinesPerEntry = 100
   59 
   60 -- | Cap the read range to the last 'maxLinesPerEntry' lines.
   61 capStart :: Int -> Int -> Int
   62 capStart start end = max start (end - maxLinesPerEntry)

   64 -- | Reconstruct the old 'TranscriptEntry' stream from the two new files.
   65 --
   66 -- The input is the full conversation (the @conversation.jsonl@ contents decoded
   67 -- into a @[Message]@) and the full entry stream (the @entries.jsonl@ contents
   68 -- decoded into an @[EntryRecord]@). The output is one 'TranscriptEntry' per
   69 -- 'EntryRecord', in order, with payloads rebuilt from the conversation prefix
   70 -- and the effective envelope.
   71 --
   72 -- The first request's envelope is the initial envelope passed in (there is no
   73 -- prior to delta against, so the first request's full envelope is the
   74 -- baseline). Subsequent requests fold deltas on top.
   75 --
   76 -- Both 'EKRequest' and 'EKResponse' entries carry ONLY the NEW messages added
   77 -- since the prior turn (@conv[start:end]@), NOT the cumulative conversation
   78 -- prefix. This is the whole point of the indexed transcript delta format: the on-disk
   79 -- @conversation.jsonl@ already stores each message exactly once, and
   80 -- re-embedding the full history into every request entry would be O(N²) in
   81 -- the conversation length (a 146-turn session would ship ~5,000 redundant
   82 -- message copies, turning 280KB on disk into ~18MB on the wire). The frontend
   83 -- extracts only the last message of each request for display, so the chat
   84 -- view is unaffected; the "View raw JSON" modal shows the delta payload (the
   85 -- new messages + the effective envelope), which is what the user actually
   86 -- wants to inspect when debugging a single turn.
   87 reconstruct :: [Message] -> [EntryRecord] -> [TranscriptEntry]
   88 reconstruct conv = go 0 Nothing
   89   where
   90     -- @start@ is the conversation-line index where the current turn's new
   91     -- messages begin (i.e. the prior response's convLen, or 0 for the first
   92     -- turn). @mEnv@ is the effective envelope at the most recent request (used
   93     -- to reconstruct response payloads, which carry the request's model).
   94     go :: Int -> Maybe Envelope -> [EntryRecord] -> [TranscriptEntry]
   95     go _ _       [] = []
   96     go start mEnv (e : es) =
   97       case erKind e of
   98         EKRequest ->
   99           let env = effectiveAtE e mEnv
  100               end = erConvLen e
  101               -- Delta-only: the NEW messages added since the prior turn
  102               -- (conv[start:end]), NOT the cumulative prefix.
  103               msgs = take (end - start) (drop start conv)
  104               -- Only include envelope fields when they CHANGED from the
  105               -- prior effective envelope. The frontend deduplicates System
  106               -- Prompt and Tools rows independently, so shipping the same
  107               -- 36KB system prompt + 14KB tools array in all 73 request
  108               -- entries of a 146-turn session wastes ~3.6MB. Omitting
  109               -- unchanged fields lets the frontend skip the rows for those
  110               -- entries (it only renders the first occurrence of each
  111               -- unique system prompt / tools pair).
  112               sys = if envSystem env /= (envSystem =<< mEnv)
  113                       then envSystem env
  114                       else Nothing
  115               tools = if envTools env /= maybe [] envTools mEnv
  116                         then Just (envTools env)
  117                         else Nothing
  118               payload = requestPayload env sys tools msgs
  119               entry = toEntry e Request (modelOf env) payload
  120           in entry : go end (Just env) es
  121         EKResponse ->
  122           let end = erConvLen e
  123               msgs = take (end - start) (drop start conv)
  124               payload = responsePayload mEnv msgs e
  125               entry = toEntry e Response (mEnv >>= modelOf) payload
  126           in entry : go end mEnv es
  127         EKHarness ->
  128           -- A harness entry (opcode invocation) adds no conversation lines
  129           -- (erConvLen = 0), so the conversation cursor is preserved — the
  130           -- next response entry must slice from the same @start@, not from 0.
  131           -- Resetting the cursor here would make the next response's
  132           -- @take end (drop 0 conv)@ return the ENTIRE conversation, which
  133           -- the frontend renders as one giant duplicate response row.
  134           let payload = harnessPayload (take (erConvLen e) (drop start conv)) e
  135               entry = toEntry e Request Nothing payload
  136           in entry : go start mEnv es
  137         EKCompaction ->
  138           let entry = toEntry e Request Nothing Null
  139           in entry : go (erConvLen e) mEnv es

  141 -- | The effective envelope at a request entry. The first request has no
  142 -- prior envelope, so its delta is folded against a default baseline; in
  143 -- practice the writer always emits a full envelope on the first request,
  144 -- so the delta carries every field. We fall back to a sensible default
  145 -- only if the delta is absent (a malformed entry).
  146 effectiveAtE :: EntryRecord -> Maybe Envelope -> Envelope
  147 effectiveAtE e mEnv =
  148   let baseline = fromMaybe defaultEnv mEnv
  149   in case erEnvelope e of
  150        Nothing -> baseline
  151        Just d  -> applyDelta baseline d
  152 
  153 -- | A neutral baseline envelope used when no prior envelope exists.
  154 defaultEnv :: Envelope
  155 defaultEnv = Envelope (ModelId "") Nothing [] ToolAuto 0
  156 
  157 -- | The model to attribute to an entry: the envelope's model, or 'Nothing'
  158 -- when the envelope carries no model.
  159 --
  160 -- A request written without an envelope delta reconstructs against
  161 -- 'defaultEnv', whose model is the empty string. The frontend uses this
  162 -- value as the message label, so an empty label must never be emitted —
  163 -- 'Nothing' lets the frontend fall back to its own label instead.
  164 modelOf :: Envelope -> Maybe ModelId
  165 modelOf env = case envModel env of
  166   m@(ModelId t)
  167     | not (T.null t) -> Just m
  168     | otherwise      -> Nothing
  169 
  170 -- | Build the old 'Request' payload: a 'CompletionRequest'-shaped JSON object
  171 -- carrying the model, system, tools, toolChoice, maxTokens, and the message
  172 -- prefix in effect at this turn. The @sys@ and @tools@ arguments are the
  173 -- system prompt and tools to include — 'Nothing' means "omit the field"
  174 -- (used when the value is unchanged from the prior request, so the frontend
  175 -- skips the redundant row). Mirrors the old 'ToJSON CompletionRequest'
  176 -- encoding so a "view raw" reproduces the bytes the old format stored.
  177 requestPayload :: Envelope -> Maybe Text -> Maybe [ToolDefinition] -> [Message] -> Value
  178 requestPayload env sys mTools msgs = object $
  179   [ "model"      .= envModel env
  180   , "messages"   .= msgs
  181   ] <> toolsField
  182     <> [ "toolChoice" .= envToolChoice env
  183        , "maxTokens"  .= envMaxTokens env
  184        ] <> systemField
  185   where
  186     systemField = case sys of
  187       Nothing -> []  -- unchanged from prior; omit so frontend skips
  188       Just s  -> ["system" .= s]
  189     toolsField = case mTools of
  190       Nothing -> []  -- unchanged from prior; omit so frontend skips
  191       Just ts -> ["tools" .= ts]
  192 
  193 -- | Build the old 'Response' payload: the assistant content blocks (drawn
  194 -- from the conversation lines added since the prior turn) plus the usage /
  195 -- stop / duration metadata stored on the entry.
  196 responsePayload :: Maybe Envelope -> [Message] -> EntryRecord -> Value
  197 responsePayload mEnv msgs e = object
  198   [ "model"    .= (envModel <$> mEnv)
  199   , "content"  .= concatMap msgContent msgs
  200   , "usage"    .= erUsage e
  201   , "stop"     .= erStop e
  202   , "durationMs" .= erDurationMs e
  203   ]

  205 -- | A harness turn's payload: the conversation lines added since the prior
  206 -- turn (the harness's input/output as messages), plus the harness name and
  207 -- the opcode metadata. The @op@ field is always included in the base
  208 -- payload (v1: so the frontend filter 'Seal.Gateway.Transcript.reconEntryToFrontend'
  209 -- can whitelist @op.name == "SKILL_LOAD"@ and surface /skill load
  210 -- invocations as distinct harness entries). The @input@, @result@, and
  211 -- @channel@ fields are included when present in @erMeta@ (the 'SKILL_LOAD'
  212 -- result entry recorded by 'Seal.ISA.Dispatch.recordSkillLoadResult' carries
  213 -- input + result + the originating channel label, so the frontend can render
  214 -- the skill body in a collapsible tool-call box AND surface channel
  215 -- provenance in the source label). When the entry carries approval metadata
  216 -- (erMeta has an "approval" key), the payload also includes the approval
  217 -- info so the frontend can render the confirmation decision.
  218 harnessPayload :: [Message] -> EntryRecord -> Value
  219 harnessPayload msgs e =
  220   let base = object
  221         [ "messages" .= msgs
  222         , "harness"  .= erHarness e
  223         , "op"       .= (Map.lookup "op" (erMeta e) :: Maybe Value)
  224         , "input"    .= (Map.lookup "input" (erMeta e) :: Maybe Value)
  225         , "result"   .= (Map.lookup "result" (erMeta e) :: Maybe Value)
  226         , "channel"  .= (Map.lookup "channel" (erMeta e) :: Maybe Value)
  227         ]
  228   in case Map.lookup "approval" (erMeta e) of
  229        Nothing -> base
  230        Just approvalVal -> object
  231          [ "messages" .= msgs
  232          , "harness"  .= erHarness e
  233          , "op"       .= (Map.lookup "op" (erMeta e) :: Maybe Value)
  234          , "input"    .= (Map.lookup "input" (erMeta e) :: Maybe Value)
  235          , "result"   .= (Map.lookup "result" (erMeta e) :: Maybe Value)
  236          , "channel"  .= (Map.lookup "channel" (erMeta e) :: Maybe Value)
  237          , "approval" .= approvalVal
  238          ]

  240 -- | Lift an 'EntryRecord' into a 'TranscriptEntry' with the given direction,
  241 -- model, and payload. The id, timestamp, duration, correlation, and meta are
  242 -- carried over.
  243 --
  244 -- The @model@ is the model that generated (or, for a request, was asked to
  245 -- generate) the entry's messages, taken from the effective envelope at that
  246 -- turn. It reaches the web transcript view as the frontend entry's top-level
  247 -- @model@ field, which labels each LLM message with the model that produced
  248 -- it instead of a fixed "Assistant". Opcode-invocation ('EKHarness') and
  249 -- compaction entries are not LLM messages, so they carry 'Nothing'.
  250 toEntry :: EntryRecord -> Direction -> Maybe ModelId -> Value -> TranscriptEntry
  251 toEntry e dir model payload = TranscriptEntry
  252   { teId = erId e
  253   , teTimestamp = erTimestamp e
  254   , teModel = model
  255   , teDirection = dir
  256   , tePayload = payload
  257   , teDurationMs = erDurationMs e
  258   , teCorrelation = erCorrelation e
  259   , teMeta = erMeta e
  260   }

  262 -- | Streaming variant of 'reconstruct': reads conversation lines on demand
  263 -- via the index instead of requiring the full @[Message]@ list in memory.
  264 -- The fold state (start, mEnv) is identical to the pure 'reconstruct' go
  265 -- function. For each 'EntryRecord', reads only the conversation lines
  266 -- @[start, end)@ via 'readConvLines'.
  267 --
  268 -- On a 'readConvLines' error (Left), returns an empty list for that entry
  269 -- (matching the skip-malformed-line behavior of the pure path).
  270 reconstructStreaming
  271   :: FilePath       -- ^ conversation.jsonl path
  272   -> FilePath       -- ^ conversation.idx path
  273   -> [EntryRecord]  -- ^ all entries (small file, read fully)
  274   -> IO [TranscriptEntry]
  275 reconstructStreaming convPath idxPath entries = do
  276   lc <- convLineCount idxPath
  277   withConvIndex convPath idxPath $ \hs -> go hs lc 0 Nothing entries
  278   where
  279     go :: ConvIndexHandles -> Int -> Int -> Maybe Envelope -> [EntryRecord] -> IO [TranscriptEntry]
  280     go _ _ _ _       [] = pure []
  281     go hs lc start mEnv (e : es) =
  282       case erKind e of
  283         EKRequest -> do
  284           let env = effectiveAtE e mEnv
  285               end = erConvLen e
  286           eMsgs <- readMsgsH hs lc (capStart start end) end
  287           let sys = if envSystem env /= (envSystem =<< mEnv)
  288                       then envSystem env
  289                       else Nothing
  290               tools = if envTools env /= maybe [] envTools mEnv
  291                         then Just (envTools env)
  292                         else Nothing
  293               payload = requestPayload env sys tools (msgsFromEither eMsgs)
  294               entry = toEntry e Request (modelOf env) payload
  295           rest <- go hs lc end (Just env) es
  296           pure (entry : rest)
  297         EKResponse -> do
  298           let end = erConvLen e
  299           eMsgs <- readMsgsH hs lc (capStart start end) end
  300           let payload = responsePayload mEnv (msgsFromEither eMsgs) e
  301               entry = toEntry e Response (mEnv >>= modelOf) payload
  302           rest <- go hs lc end mEnv es
  303           pure (entry : rest)
  304         EKHarness -> do
  305           eMsgs <- readMsgsH hs lc (capStart start (erConvLen e)) (erConvLen e)
  306           let payload = harnessPayload (msgsFromEither eMsgs) e
  307               entry = toEntry e Request Nothing payload
  308           rest <- go hs lc start mEnv es
  309           pure (entry : rest)
  310         EKCompaction -> do
  311           let entry = toEntry e Request Nothing Null
  312           rest <- go hs lc (erConvLen e) mEnv es
  313           pure (entry : rest)
  314 
  315     readMsgsH :: ConvIndexHandles -> Int -> Int -> Int -> IO (Either a [Message])
  316     readMsgsH hs lc s e
  317       | s >= e = pure (Right [])
  318       | otherwise = do
  319           result <- readConvLinesWith hs lc s e
  320           pure (case result of
  321                    Right ms -> Right ms
  322                    Left _   -> Right [])
  323 
  324     msgsFromEither :: Either a [Message] -> [Message]
  325     msgsFromEither (Right ms) = ms
  326     msgsFromEither (Left _)   = []

  328 -- | Paginated streaming reconstruction: folds envelopes over ALL entries
  329 -- (pure, no IO — needed for correct envelope deltas), but only reads
  330 -- conversation lines (IO via the index) for the entries in the requested
  331 -- page [@startEntry, startEntry + limit@). This avoids processing 1775
  332 -- entries' conversation lines when only the last 5 are needed.
  333 --
  334 -- Returns the reconstructed TranscriptEntries for the page only.
  335 reconstructStreamingPage
  336   :: FilePath       -- ^ conversation.jsonl path
  337   -> FilePath       -- ^ conversation.idx path
  338   -> [EntryRecord]  -- ^ all entries (small file, read fully)
  339   -> Int            -- ^ start entry index (0-based, inclusive)
  340   -> Int            -- ^ limit (max entries to process)
  341   -> IO [TranscriptEntry]
  342 reconstructStreamingPage convPath idxPath entries startEntry limit = do
  343   lc <- convLineCount idxPath
  344   -- Phase 1: fold envelopes over all entries to find the effective
  345   -- envelope + conversation cursor at the start of the page.
  346   let (startConv, mEnv, _) = foldToStart entries 0 Nothing startEntry
  347       pageEntries = take limit (drop startEntry entries)
  348       endConv = case reverse pageEntries of
  349         [] -> startConv
  350         (lastE : _) -> erConvLen lastE
  351   putStrLn $ "[reconstruct] page entries=" <> show (length pageEntries)
  352            <> " startEntry=" <> show startEntry
  353            <> " convLines=" <> show (endConv - startConv)
  354            <> " convRange=[" <> show startConv <> ".." <> show endConv <> "]"
  355   -- Phase 2: read conversation lines only for the page entries.
  356   withConvIndex convPath idxPath $ \hs -> go hs lc startConv mEnv pageEntries
  357   where
  358     -- Fold over entries [0, startEntry) to find the conversation cursor
  359     -- and effective envelope at the page start. Pure — no IO.
  360     foldToStart :: [EntryRecord] -> Int -> Maybe Envelope -> Int -> (Int, Maybe Envelope, [EntryRecord])
  361     foldToStart [] start mEnv _ = (start, mEnv, [])
  362     foldToStart (e : es) start mEnv n
  363       | n <= 0    = (start, mEnv, e : es)
  364       | otherwise =
  365           case erKind e of
  366             EKRequest ->
  367               let env = effectiveAtE e mEnv
  368               in foldToStart es (erConvLen e) (Just env) (n - 1)
  369             EKResponse ->
  370               foldToStart es (erConvLen e) mEnv (n - 1)
  371             EKHarness ->
  372               foldToStart es start mEnv (n - 1)
  373             EKCompaction ->
  374               foldToStart es (erConvLen e) mEnv (n - 1)
  375 
  376     -- Same go as reconstructStreaming, reused for the page entries.
  377     go :: ConvIndexHandles -> Int -> Int -> Maybe Envelope -> [EntryRecord] -> IO [TranscriptEntry]
  378     go _ _ _ _       [] = pure []
  379     go hs lc start mEnv (e : es) =
  380       case erKind e of
  381         EKRequest -> do
  382           let env = effectiveAtE e mEnv
  383               end = erConvLen e
  384           eMsgs <- readMsgsH hs lc (capStart start end) end
  385           let sys = if envSystem env /= (envSystem =<< mEnv)
  386                       then envSystem env
  387                       else Nothing
  388               tools = if envTools env /= maybe [] envTools mEnv
  389                         then Just (envTools env)
  390                         else Nothing
  391               payload = requestPayload env sys tools (msgsFromEither eMsgs)
  392               entry = toEntry e Request (modelOf env) payload
  393           rest <- go hs lc end (Just env) es
  394           pure (entry : rest)
  395         EKResponse -> do
  396           let end = erConvLen e
  397           eMsgs <- readMsgsH hs lc (capStart start end) end
  398           let payload = responsePayload mEnv (msgsFromEither eMsgs) e
  399               entry = toEntry e Response (mEnv >>= modelOf) payload
  400           rest <- go hs lc end mEnv es
  401           pure (entry : rest)
  402         EKHarness -> do
  403           eMsgs <- readMsgsH hs lc (capStart start (erConvLen e)) (erConvLen e)
  404           let payload = harnessPayload (msgsFromEither eMsgs) e
  405               entry = toEntry e Request Nothing payload
  406           rest <- go hs lc start mEnv es
  407           pure (entry : rest)
  408         EKCompaction -> do
  409           let entry = toEntry e Request Nothing Null
  410           rest <- go hs lc (erConvLen e) mEnv es
  411           pure (entry : rest)

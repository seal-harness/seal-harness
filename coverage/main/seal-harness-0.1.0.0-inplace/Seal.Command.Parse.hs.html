   31 -- Internal state for the tokenizer state machine.
   32 data TokSt
   33   = Outside              -- ^ Between tokens
   34   | InWord  [Char]       -- ^ Inside a bare token (reversed chars accumulated)
   35   | InQuote [Char]       -- ^ Inside a double-quoted span (reversed chars; continues
   36                          --   any preceding bare chars so adjacent quoted/unquoted
   37                          --   sections form a single token, e.g. foo"bar" -> "foobar")
   38 
   39 -- | Quote-aware shell-words tokenizer. Supports double-quoted tokens so that
   40 -- @\/vault add "my key"@ correctly produces three tokens. Returns 'Left' with
   41 -- an error message if a double-quote is never closed.
   42 tokenize :: Text -> Either Text [Text]
   43 tokenize input = go Outside (T.unpack input) []
   44   where
   45     -- Flush the current InWord accumulator as a completed token.
   46     flushWord :: [Char] -> [Text] -> [Text]
   47     flushWord cs acc = T.pack (reverse cs) : acc
   48 
   49     -- Finalize the state machine after all characters have been consumed.
   50     finish :: TokSt -> [Text] -> Either Text [Text]
   51     finish Outside     acc = Right (reverse acc)
   52     finish (InWord cs) acc = Right (reverse (flushWord cs acc))
   53     finish (InQuote _) _   = Left "unterminated double-quote"
   54 
   55     go :: TokSt -> [Char] -> [Text] -> Either Text [Text]
   56     go st          []     acc = finish st acc
   57     -- Outside: skip whitespace; open quote starts quoted token;
   58     -- any other char starts a bare word.
   59     go Outside     (c:cs) acc
   60       | c == '"'             = go (InQuote []) cs acc
   61       | isSpace c            = go Outside cs acc
   62       | otherwise            = go (InWord [c]) cs acc
   63     -- InWord: whitespace terminates this token; quote opens an adjacent
   64     -- quoted span (still building the *same* token); other chars extend.
   65     go (InWord ws) (c:cs) acc
   66       | c == '"'             = go (InQuote ws) cs acc
   67       | isSpace c            = go Outside cs (flushWord ws acc)
   68       | otherwise            = go (InWord (c:ws)) cs acc
   69     -- InQuote: closing quote returns to InWord (bare continuation allowed
   70     -- immediately after); other chars extend the quoted span.
   71     go (InQuote ws) (c:cs) acc
   72       | c == '"'             = go (InWord ws) cs acc
   73       | otherwise            = go (InQuote (c:ws)) cs acc
   74

   79 data ParseOutcome
   80   = ParsedAction CommandAction
   81   | ParseHelp    (Maybe CommandName)
   82   | ParseFailure Text
   83 
   84 -- | Parse a full slash-command line; input MUST begin with @\/@.
   85 --
   86 -- Routing rules (in order):
   87 --   1. head word == "help" (case-insensitive)     -> 'ParseHelp'
   88 --   2. "--help"/"-h" is the command's SOLE argument -> 'ParseHelp' (Just head):
   89 --      top-level help for the command itself.
   90 --   3. head word found in registry                 -> 'execParserPure' -> 'ParsedAction'
   91 --      or 'ParseFailure'. A help flag that follows a subcommand token (e.g.
   92 --      @provider add -h@) falls through to here: 'execParserPure' handles
   93 --      @--help@/@-h@ natively at every level (the command parsers use
   94 --      'hsubparser', which gives each subcommand its own help), so the
   95 --      subcommand's help is rendered — as 'ParseFailure' text, which the caller
   96 --      echoes identically to any other help. Rule 2 is deliberately narrow so
   97 --      it never swallows a subcommand-scoped help flag.
   98 --   4. head word not in registry                   -> 'ParseFailure'
   99 --
  100 -- 'CompletionInvoked' from 'execParserPure' is reserved for future
  101 -- shell-completion integration (seal --bash-completion-*); the TUI
  102 -- never triggers it, so it maps to @ParseFailure ""@ (empty, not shown).
  103 parseSlash :: Registry -> Text -> ParseOutcome
  104 parseSlash registry fullLine =
  105   let line = T.drop 1 fullLine           -- strip leading '/'
  106   in case tokenize line of
  107     Left err     -> ParseFailure ("parse error: " <> err)
  108     Right []     -> ParseFailure "empty command"
  109     Right (h:rest) ->
  110       let headName = CommandName h
  111           -- case-insensitive "help" check
  112           isHelp   = T.toCaseFold h == "help"
  113           -- A help flag that is the command's SOLE argument requests the
  114           -- command's own (top-level) help. A help flag with other tokens
  115           -- present (a subcommand) is left for execParserPure to dispatch to
  116           -- the right subcommand — see rule 3 in the Haddock above.
  117           isTopLevelHelp = rest == ["--help"] || rest == ["-h"]
  118       in if isHelp
  119          then ParseHelp (case rest of
  120                 []    -> Nothing
  121                 (n:_) -> Just (CommandName n))
  122          else if isTopLevelHelp
  123               then ParseHelp (Just headName)
  124               else if T.toCaseFold h == "call" && isJust (lookupSpec registry headName)
  125                    then parseCallRaw registry h line
  126                    else case lookupSpec registry headName of
  127                      Nothing   -> ParseFailure ("unknown command: " <> h)
  128                      Just spec ->
  129                        -- execParserPure expects [String], not [Text]
  130                        let args  = map T.unpack rest
  131                            -- defaultPrefs: enables --help/--version, no disambiguation,
  132                            -- single-line error context. CompletionInvoked is reserved
  133                            -- (see note above).
  134                            prefs = defaultPrefs
  135                        in case execParserPure prefs (csParserInfo spec) args of
  136                          Success act         -> ParsedAction act
  137                          Failure f           ->
  138                            let (msg, _) = renderFailure f (T.unpack h)
  139                            in ParseFailure (T.pack msg)
  140                          CompletionInvoked _ -> ParseFailure ""
  141 
  142 -- | Special-case parser for @/call@: the JSON payload may contain
  143 -- double-quotes and spaces that the shell-words tokenizer would strip or
  144 -- split. To preserve the raw JSON, we bypass the tokenizer for the JSON
  145 -- portion: split the line (post-/@/@) into the command name, the opcode
  146 -- name, and the raw rest-of-line as a single string. The @call@ parser
  147 -- then receives @[OP, JSON-raw]@ as its argv.
  148 parseCallRaw :: Registry -> Text -> Text -> ParseOutcome
  149 parseCallRaw registry h line =
  150   -- line is the full command minus the leading '/'. Drop the command name
  151   -- ("call") to reach "OP JSON-raw".
  152   let afterCall = T.dropWhile (== ' ') (T.drop (T.length h) line)
  153       -- Split into OP and the raw JSON rest on the first run of whitespace.
  154       (opText, jsonRest) = T.break isSpace afterCall
  155       jsonRest' = T.dropWhile isSpace jsonRest
  156       mSpec = lookupSpec registry (CommandName h)
  157       args = case T.null jsonRest' of
  158                True  -> [T.unpack opText]                 -- /call OP
  159                False -> [T.unpack opText, T.unpack jsonRest']  -- /call OP {json...}
  160   in case mSpec of
  161        Nothing   -> ParseFailure ("unknown command: " <> h)
  162        Just spec ->
  163          case execParserPure defaultPrefs (csParserInfo spec) args of
  164            Success act         -> ParsedAction act
  165            Failure f           ->
  166              let (msg, _) = renderFailure f (T.unpack h)
  167              in ParseFailure (T.pack msg)
  168            CompletionInvoked _ -> ParseFailure ""

   93 -- | The unified capability handle. Each field is an IO action the opcode
   94 -- calls; the smart constructors wire the local or remote implementation.
   95 -- The constructor is NOT exported — only the smart constructors are.
   96 --
   97 -- File methods take a 'RemotePath' (a workspace-relative path). Internally
   98 -- each arm re-anchors the path against its own 'WorkspaceRoot' and runs

  205 -- | The structured error an 'UntrustedIO' method returns. Wraps the
  206 -- path-error, the bounded-overflow, and the executor-layer errors so the
  207 -- opcode can render a single, consistent error message.
  208 data UntrustedErr
  209   = UePath PathError
  210     -- ^ SafePath confinement failed (blocked name, @..@ escape, missing
  211     -- parent, symlink escape).
  212   | UeBounded Int
  213     -- ^ The call exceeded the operator ceiling (write/read/search). The
  214     -- 'Int' is the ceiling for the error message.
  215   | UeExec ExecError
  216     -- ^ The executor failed (remote unreachable, host-key mismatch, not
  217     -- implemented).
  218   | UeIo Text
  219     -- ^ An otherwise-uncategorized IO error (caught 'IOException',
  220     -- rendered as 'Text').
  221   deriving stock (Eq, Show)

  223 -- | Render an 'UntrustedErr' as a human-readable 'Text' for the opcode's
  224 -- 'orParts' (the model-visible error message).
  225 renderUntrustedErr :: UntrustedErr -> Text
  226 renderUntrustedErr = \case
  227   UePath pe   -> "path error: " <> T.pack (show pe)
  228   UeBounded n -> "content exceeds operator ceiling (" <> T.pack (show n) <> " bytes)"
  229   UeExec ee   -> renderExecError ee
  230   UeIo msg    -> "io error: " <> msg

  232 -- | Render an 'ExecError' as a human-readable 'Text'. The structured
  233 -- 'ExecError' ADT's 'Show' instance gives compact names like
  234 -- @"ExecRemoteUnreachable"@; this function expands them to full sentences
  235 -- so the model + user sees the /meaning/, not the constructor name.
  236 renderExecError :: ExecError -> Text

  243   ExecHostKeyUnknown -> "the remote host's key is not in known_hosts. Add it manually or use host-key adoption."

  251 -- | The local untrusted executor. Workspace files live on the local FS;
  252 -- commands run via @/bin/sh -c@ (shell) or @proc@ (bin). Absent under
  253 -- @-f remote-only-untrusted@.
  254 mkLocalUntrustedIO :: WorkspaceRoot -> UntrustedIO
  255 mkLocalUntrustedIO wsRoot =
  256   -- The workdir root itself, resolved once as the SafePath for @.@. This
  257   -- is the single source of truth for the \"untrusted opcodes share one
  258   -- workspace root\" invariant: 'uioShellExec' / 'uioBinExec' default to
  259   -- it when the caller omits a cwd, so every opcode agrees on the same
  260   -- root without any opcode needing to remember to pass it.
  261   let wsRootPath = case mkSafePathRemote wsRoot "." of
  262         Right sp -> getSafePath sp
  263         Left _   -> "/nonexistent-workdir-fail-closed"
  264   in UntrustedIO
  265   { uioReadFile = \rp scanBytes -> do
  266       let rel = T.unpack (getRemotePath rp)
  267       eSafe <- mkSafePathAllowAbs wsRoot rel
  268       case eSafe of
  269         Left pe  -> pure (Left (UePath pe))
  270         Right sp -> do
  271           -- Bounded read: read at most scanBytes from the file, decode to
  272           -- Text, split to lines, build a LineWindow covering all read
  273           -- lines. The 'lwTruncated' flag is set when the file exceeds the
  274           -- scan ceiling (the read stopped before EOF). This matches the
  275           -- remote arm's 'lineWindowFromText' — both arms return the same
  276           -- shape, and the opcode does the offset/limit windowing purely.
  277           eContent <- try (readBoundedLocal (getSafePath sp) scanBytes)
  278                         :: IO (Either IOException (BS.ByteString, Bool))
  279           pure $ case eContent of
  280             Left ioErr   -> Left (UeIo (T.pack (show ioErr)))
  281             Right (bs, truncated) ->
  282               -- When truncated, the trailing partial line (no newline) is
  283               -- NOT counted as a complete line — matching the original
  284               -- 'readLineWindow' semantics (lwTotal is a lower bound). When
  285               -- not truncated, 'T.lines' counts a trailing partial line
  286               -- (matching Data.Text.lines).
  287               let txt = TE.decodeUtf8Lenient bs
  288                   rawLines = T.lines txt
  289                   -- Drop the trailing partial line if truncated AND it has
  290                   -- no newline (i.e. it's a partial). T.lines splits on \n;
  291                   -- a trailing partial line is one where the original bytes
  292                   -- didn't end with \n.
  293                   endsWithNl = not (BS.null bs) && BS.last bs == 0x0A
  294                   ls = if truncated && not endsWithNl
  295                          then init rawLines   -- drop the partial
  296                          else rawLines
  297                   n = length ls
  298               in Right LineWindow
  299                    { lwLines     = ls
  300                    , lwStart     = 0
  301                    , lwEnd       = n
  302                    , lwTotal     = n
  303                    , lwHasMore   = truncated
  304                    , lwTruncated = truncated
  305                    }
  306   , uioWriteFile = \rp content mode ceiling' -> do
  307       let rel       = T.unpack (getRemotePath rp)
  308           byteCount = BS.length (TE.encodeUtf8 content)
  309       if byteCount > ceiling'
  310         then pure (Left (UeBounded ceiling'))
  311         else do
  312           eSafe <- mkSafePathForWriteAllowAbs wsRoot rel
  313           case eSafe of
  314             Left pe  -> pure (Left (UePath pe))
  315             Right sp -> do
  316               eUnit <- try (writeLocal sp content mode) :: IO (Either IOException ())
  317               pure $ case eUnit of
  318                 Left ioErr -> Left (UeIo (T.pack (show ioErr)))
  319                 Right _    -> Right byteCount
  320   , uioPatchFile = \rp patch -> do
  321       let rel = T.unpack (getRemotePath rp)
  322       eSafe <- mkSafePathAllowAbs wsRoot rel
  323       case eSafe of
  324         Left pe  -> pure (Left (UePath pe))
  325         Right sp -> patchLocal sp patch
  326   , uioShellExec = \cmd mCwd ->
  327       let argv = ["/bin/sh", "-c", T.unpack (textShellCommand cmd)]
  328       in case mCwd of
  329            Nothing -> runLocalFixedArgv False argv (Just wsRootPath)
  330            Just rp -> case mkSafePathRemote wsRoot (T.unpack (getRemotePath rp)) of
  331              Left pe  -> pure (Left (UePath pe))
  332              Right sp -> runLocalFixedArgv False argv (Just (getSafePath sp))
  333   , uioBinExec = \bin bargs mCwd ->
  334       let argv = T.unpack (textBinName bin) : map (T.unpack . textBinArg) bargs
  335       in case resolveBinCwd wsRoot mCwd of
  336            Left pe       -> pure (Left (UePath pe))
  337            Right cwdAbs -> runLocalFixedArgv True argv (Just cwdAbs)
  338   , uioProcessList =
  339       case mkShellCommand "ps -o pid=,cmd=" of
  340         Left _   -> pure (Left (UeExec ExecNotImplemented))
  341         Right sh -> uioShellExec (mkLocalUntrustedIO wsRoot) sh Nothing
  342   , uioProcessKill = \pid ->
  343       case mkShellCommand ("kill " <> T.pack (show pid)) of
  344         Left _   -> pure (Left (UeExec ExecNotImplemented))
  345         Right sh -> do
  346           res <- uioShellExec (mkLocalUntrustedIO wsRoot) sh Nothing
  347           pure (const (Right ()) =<< res)
  348   , uioSearchFiles = \pat mPath _limit -> do
  349       let mRel = maybe "" (T.unpack . getRemotePath) mPath

  351         Left pe  -> pure (Left (UePath pe))

  353           Left _   -> pure (Left (UeExec ExecNotImplemented))
  354           Right sh -> uioShellExec (mkLocalUntrustedIO wsRoot) sh Nothing
  355   , uioShellExecEnv = \extras cmd mCwd ->

  358            Nothing -> runLocalFixedArgvEnv False argv (Just wsRootPath) extras
  359            Just rp -> case mkSafePathRemote wsRoot (T.unpack (getRemotePath rp)) of

  361              Right sp -> runLocalFixedArgvEnv False argv (Just (getSafePath sp)) extras

  390 -- | Read at most @maxBytes@ from a local file. Returns the bytes read and
  391 -- a 'Bool' indicating whether the file was truncated (the file is larger
  392 -- than @maxBytes@ — the read stopped at the ceiling).
  393 readBoundedLocal :: FilePath -> Int -> IO (BS.ByteString, Bool)
  394 readBoundedLocal path maxBytes = do
  395   bs <- BS.readFile path
  396   let len = BS.length bs
  397   if len <= maxBytes
  398     then pure (bs, False)
  399     else pure (BS.take maxBytes bs, True)

  401 -- | Write content to the local FS via the validated 'SafePath'. The temp
  402 -- file + rename is used for @'WMWrite'@ so the write is atomic (a crash
  403 -- mid-write leaves the old file intact); @'WMAppend'@ appends directly
  404 -- (the file already exists or is created fresh by 'BS.appendFile').
  405 writeLocal :: SafePath -> Text -> WriteMode -> IO ()
  406 writeLocal sp content mode =
  407   let path  = getSafePath sp
  408       bytes = TE.encodeUtf8 content
  409   in case mode of
  410        WMAppend -> BS.appendFile path bytes
  411        WMWrite  -> do
  412          let tmp = path <> ".seal-write-tmp"
  413          BS.writeFile tmp bytes
  414          renameFile tmp path

  416 -- | Apply a unified diff to the local file at the validated 'SafePath':
  417 -- read → apply in-process → atomic temp+rename write.
  418 patchLocal :: SafePath -> Text -> IO (Either UntrustedErr ())
  419 patchLocal sp patch = do
  420   eContent <- try (BS.readFile (getSafePath sp)) :: IO (Either IOException BS.ByteString)
  421   case eContent of
  422     Left ioErr -> pure (Left (UeIo (T.pack (show ioErr))))
  423     Right content -> case applyUnifiedDiff (TE.decodeUtf8Lenient content) patch of
  424       Left applyErr -> pure (Left (UeIo applyErr))
  425       Right newContent -> do
  426         let path = getSafePath sp
  427             tmp  = path <> ".seal-patch-tmp"
  428         eUnit <- try (BS.writeFile tmp (TE.encodeUtf8 newContent) >> renameFile tmp path)
  429                    :: IO (Either IOException ())
  430         pure $ case eUnit of
  431           Left ioErr -> Left (UeIo (T.pack (show ioErr)))
  432           Right _    -> Right ()

  436 -- (the binary is not on PATH) when True; otherwise 127 is a normal
  437 -- command-not-found failure, returned via 'Right' with the exit code
  438 -- annotation. Any 'IOException' becomes 'Left ExecNotImplemented'.

  442 runLocalFixedArgv treat127AsMissing argv mCwd =

  445 -- | Run a fixed-argv program locally with env overrides. The @extras@
  446 -- are MERGED over the inherited environment ('getEnvironment'): the
  447 -- process inherits the harness env (PATH, HOME, etc.) plus the overrides
  448 -- (e.g. @SSH_AUTH_SOCK@, @GIT_SSH_COMMAND@, @GIT_TERMINAL_PROMPT=0@). This
  449 -- is the local-arm env-override seam (W2) — the secret-free auth env for
  450 -- deploy-key ops. The overrides are NEVER secret bytes (only paths +
  451 -- non-secret values); the secret lives only in the forwarded agent's
  452 -- memory.
  453 runLocalFixedArgvEnv
  454   :: Bool -> [String] -> Maybe String -> [(String, String)]
  455   -> IO (Either UntrustedErr Text)
  456 runLocalFixedArgvEnv treat127AsMissing argv mCwd extras = do
  457   let (program, args) = case argv of
  458         (p : as) -> (p, as)
  459         []       -> error "runLocalFixedArgvEnv: empty argv (unreachable)"
  460   logExecDebug "[local]" argv mCwd extras
  461   inherited <- getEnvironment
  462   let env' = Just (mergeEnv inherited extras)
  463       cp = (proc program args)
  464               { std_in = NoStream, std_out = CreatePipe, std_err = CreatePipe
  465               , cwd = mCwd, env = env'
  466               , create_group = True
  467               }
  468       -- ^ @create_group = True@ puts the child in its own POSIX process
  469       -- group so 'withManagedProcess' can kill the whole group on cleanup
  470       -- (SIGTERM → grace → SIGKILL). This prevents orphans when the
  471       -- dispatch wrapper cancels the Haskell worker thread.
  472       maxOutput = 50_000  -- bounded output cap (Task 4, matches ttcMaxOutputBytes)
  473   res <- try @IOException
  474          (withManagedProcess cp $ \ph _mIn hOut hErr -> do
  475             out <- readBounded hOut maxOutput
  476             err <- readBounded hErr maxOutput
  477             ec  <- waitForProcess ph
  478             pure (ec, out, err))
  479   pure $ case res of
  480     Left _ioErr                     -> Left (UeExec ExecNotImplemented)

  488 -- a user's ambient @SSH_AUTH_SOCK@ must NOT override the per-op agent's

  542 -- | Emit a debug-level log line showing the exact command the UntrustedIO
  543 -- executor is about to run. Covers both the local arm (a fixed-argv
  544 -- subprocess) and the remote arm (an @ssh@ argv, including @-A@ for
  545 -- agent-forwarding git ops). The @tag@ prefixes the line so the reader
  546 -- can distinguish local vs remote vs remote-with-agent at a glance:
  547 --
  548 --   * @"[local]"@        — a local subprocess (@runLocalFixedArgvEnv@)
  549 --   * @"[remote ssh]"@   — a remote SSH exec (no agent forwarding)
  550 --   * @"[remote ssh -A]"@ — a remote SSH exec with agent forwarding
  551 --                          (deploy-key git ops)
  552 --
  553 -- For the remote arm, the full SSH argv is logged (e.g.
  554 -- @ssh -o StrictHostKeyChecking=yes ... agent@exec.internal -- cd ... && git pull@),
  555 -- so the complete command — including the SSH portion — is visible when
  556 -- @mode=remote@. Env overrides (e.g. @SSH_AUTH_SOCK=...@,
  557 -- @GIT_SSH_COMMAND=...@) are shown as @KEY=VAL@ tokens. The cwd is shown
  558 -- as @cd <path> &&@ when set.
  559 --
  560 -- Secret-bearing env keys (see 'secretEnvKeys') are redacted via
  561 -- 'redactEnv' before the @KEY=VAL@ tokens are rendered, so a value like
  562 -- @GH_TOKEN=ghp_secret...@ appears as @GH_TOKEN=<redacted>@ in the log.
  563 -- This is a defense-in-depth seam on 'logExecDebug' itself, not caller
  564 -- discipline — the redaction fires regardless of which arm produced the
  565 -- call (the remote arm passes the secret extras separately via
  566 -- 'runRemoteShellTextEnv' so the same redaction path covers it).
  567 logExecDebug :: String -> [String] -> Maybe String -> [(String, String)] -> IO ()
  568 logExecDebug tag argv mCwd extras =
  569   globalLogIO DebugS (K.ls msg)
  570   where
  571     msg = T.pack (unwords (filter (not . null) parts))
  572     parts =
  573       [ tag ]
  574       <> envPart
  575       <> cwdPart
  576       <> [ unwords (map shellQuoteArgv argv) ]
  577     -- Secret values NEVER reach the log: 'redactEnv' replaces the values
  578     -- of 'secretEnvKeys' with <redacted> before rendering. Non-secret
  579     -- keys (SSH_AUTH_SOCK, GIT_TERMINAL_PROMPT, ...) pass through so
  580     -- operators can still diagnose credential-injection plumbing.
  581     envPart = case redactEnv extras of
  582       [] -> []
  583       xs -> [ unwords (map (\(k, v) -> k <> "=" <> v) xs) ]
  584     cwdPart = case mCwd of
  585       Just c  -> ["cd " <> c <> " &&"]
  586       Nothing -> []
  587     -- Quote an argv token for log readability: wrap in single quotes if
  588     -- it contains spaces or shell metacharacters. This is display-only
  589     -- (the actual subprocess argv is passed verbatim, never shell-parsed).
  590     shellQuoteArgv s
  591       | any (\c -> c == ' ' || c == '\'' || c == '"' || c == '$' || c == '`') s = "'" <> s <> "'"
  592       | otherwise = s

  606 -- | Same as 'mkRemoteUntrustedIO' (the two names document the same
  607 -- constructor). Provided for the wiring site, which threads the runner
  608 -- explicitly.
  609 mkRemoteUntrustedIOFromRunner :: SshConfig -> RemoteRunner -> UntrustedIO
  610 mkRemoteUntrustedIOFromRunner sshCfg runner =
  611   -- The workdir root itself, resolved once as the SafePath for @.@. This
  612   -- is the single source of truth for the \"untrusted opcodes share one
  613   -- workspace root\" invariant on the remote arm: 'uioShellExec' /
  614   -- 'uioBinExec' default to it when the caller omits a cwd, so every
  615   -- opcode agrees on the same root (the remote @scWorkspace@), not the
  616   -- remote user's home or wherever the SSH server drops it.
  617   let wsRootPath = case mkSafePathRemote (wsRootFromCfg sshCfg) "." of
  618         Right sp -> getSafePath sp
  619         Left _   -> "/nonexistent-workdir-fail-closed"
  620   in UntrustedIO
  621   { uioReadFile = \rp scanBytes ->
  622       let rel = T.unpack (getRemotePath rp)
  623       in case mkSafePathRemoteAllowAbs (wsRootFromCfg sshCfg) rel of
  624            Left pe  -> pure (Left (UePath pe))
  625            Right sp -> do
  626              -- ssh ... -- head -c <scanBytes> <abspath>  (bounded read).
  627              -- The SafePath is the workspace-anchored absolute path on the
  628              -- remote machine; the SSH command reads from that path, not
  629              -- the remote user's home CWD.
  630              let absPath = getSafePath sp
  631                  cmd = T.pack ("head -c " <> show scanBytes <> " " <> shellQuote absPath)
  632              res <- runRemoteShellText runner sshCfg cmd
  633              pure (Right . lineWindowFromText =<< res)
  634   , uioWriteFile = \rp content mode _ceiling' -> do
  635       let rel       = T.unpack (getRemotePath rp)
  636           byteCount = BS.length (TE.encodeUtf8 content)
  637       case mkSafePathRemoteAllowAbs (wsRootFromCfg sshCfg) rel of
  638         Left pe  -> pure (Left (UePath pe))
  639         Right sp -> do
  640           -- ssh ... -- tee [-a] <abspath>   with content on stdin.
  641           -- The SafePath is the workspace-anchored absolute path.
  642           let absPath  = getSafePath sp
  643               teeFlag = case mode of WMWrite -> "" ; WMAppend -> "-a "
  644               cmd  = T.pack ("tee " <> teeFlag <> shellQuote absPath)
  645               argv = sshExecArgv sshCfg cmd
  646           res <- runRemoteStdin runner argv (TE.encodeUtf8 content)
  647           pure (either (Left . UeExec) (const (Right byteCount)) res)
  648   , uioPatchFile = \rp patch -> do
  649       let rel = T.unpack (getRemotePath rp)
  650       case mkSafePathRemoteAllowAbs (wsRootFromCfg sshCfg) rel of
  651         Left pe  -> pure (Left (UePath pe))
  652         Right sp -> do
  653           let absPath = getSafePath sp
  654           -- Read remote (cat) → apply diff in-process → write remote via
  655           -- a single SSH exec with stdin. The patched content is piped to
  656           -- a remote sh -c that writes the temp + mv (atomic).
  657           let readCmd = T.pack ("cat " <> shellQuote absPath)
  658           rRead <- runRemoteShellText runner sshCfg readCmd
  659           case rRead of
  660             Left e       -> pure (Left e)
  661             Right oldTxt -> case applyUnifiedDiff oldTxt patch of
  662               Left applyErr -> pure (Left (UeIo applyErr))
  663               Right newContent -> do
  664                 let tmpPath = absPath <> ".seal-patch-tmp"
  665                     remoteSh = T.pack
  666                       ("sh -c 'cat > " <> shellQuote tmpPath
  667                        <> " && mv " <> shellQuote tmpPath
  668                        <> " " <> shellQuote absPath <> "'")
  669                     argv = sshExecArgv sshCfg remoteSh
  670                 res <- runRemoteStdin runner argv (TE.encodeUtf8 newContent)
  671                 pure (either (Left . UeExec) (const (Right ())) res)
  672   , uioShellExec = \cmd mCwd ->
  673       let prefixCd p = "cd " <> shellQuote p <> " && "
  674       in case mCwd of
  675         Nothing -> runRemoteShellText runner sshCfg
  676                      (T.pack (prefixCd wsRootPath
  677                               <> T.unpack (textShellCommand cmd)))
  678         Just rp ->
  679           case mkSafePathRemote (wsRootFromCfg sshCfg) (T.unpack (getRemotePath rp)) of
  680             Left pe  -> pure (Left (UePath pe))
  681             Right sp ->
  682               let cdCmd = "cd " <> shellQuote (getSafePath sp)
  683                           <> " && " <> T.unpack (textShellCommand cmd)
  684               in runRemoteShellText runner sshCfg (T.pack cdCmd)
  685   , uioBinExec = \bin bargs mCwd ->
  686       let argv' = T.unpack (textBinName bin) : map (T.unpack . textBinArg) bargs
  687       in case resolveBinCwd (wsRootFromCfg sshCfg) mCwd of
  688            Left pe  -> pure (Left (UePath pe))
  689            Right cwdPath ->
  690              let cmd = T.pack ("cd " <> shellQuote cwdPath <> " && "
  691                              <> T.unpack (T.intercalate " " (map (T.pack . shellQuote) argv')))
  692              in runRemoteShellText runner sshCfg cmd
  693   , uioProcessList =
  694       runRemoteShellText runner sshCfg "ps -o pid=,cmd="
  695    , uioProcessKill = \pid -> do
  696        res <- runRemoteShellText runner sshCfg ("kill " <> T.pack (show pid))
  697        pure (const (Right ()) =<< res)
  698    , uioSearchFiles = \pat mPath _limit ->
  699        let mRel = maybe "" (T.unpack . getRemotePath) mPath
  700        in case mkSafePathRemote (wsRootFromCfg sshCfg) mRel of
  701             Left pe  -> pure (Left (UePath pe))
  702             Right sp -> runRemoteShellText runner sshCfg (buildRgCmd pat (Just sp))
  703     , uioShellExecEnv = \extras cmd mCwd ->
  704        let prefixCd p = "cd " <> shellQuote p <> " && "
  705            cmdText = T.pack (T.unpack (textShellCommand cmd))
  706        in case mCwd of
  707          Nothing ->
  708            if hasSecretExtras extras
  709              then runRemoteShellStdinEnv runner sshCfg
  710                     (T.pack (prefixCd wsRootPath)) extras cmdText
  711              else runRemoteShellTextEnv runner sshCfg
  712                     (T.pack (prefixCd wsRootPath)) extras cmdText
  713          Just rp ->
  714            case mkSafePathRemote (wsRootFromCfg sshCfg) (T.unpack (getRemotePath rp)) of
  715              Left pe  -> pure (Left (UePath pe))
  716              Right sp ->
  717                let cdPart = T.pack ("cd " <> shellQuote (getSafePath sp) <> " && ")
  718                in if hasSecretExtras extras
  719                     then runRemoteShellStdinEnv runner sshCfg cdPart extras cmdText
  720                     else runRemoteShellTextEnv runner sshCfg cdPart extras cmdText
  721     , uioShellExecGitEnv = \extras _mKnownHosts cmd mCwd ->
  722         -- Remote deploy-key path (simple approach): ssh -A forwards the
  723         -- SEAL agent to the remote machine. The remote git clone uses the
  724         -- default ssh + the forwarded agent + the remote machine's default
  725         -- known_hosts. No GIT_SSH_COMMAND, no remote known_hosts file.
  726         -- Strip SSH_AUTH_SOCK/SSH_AGENT_PID from the REMOTE env prefix
  727         -- (the forwarded agent replaces them). Keep them in the LOCAL
  728         -- ssh -A process env so the right agent is forwarded.
  729         let localEnv = [ (k, v) | (k, v) <- extras
  730                       , k == "SSH_AUTH_SOCK" || k == "SSH_AGENT_PID"
  731                       ]
  732             remoteExtras = [ (k, v) | (k, v) <- extras
  733                            , k /= "SSH_AUTH_SOCK"
  734                            , k /= "SSH_AGENT_PID"
  735                            ]
  736             remoteEnvPrefix = T.pack (renderEnvPrefix remoteExtras)
  737         in case mCwd of
  738           Nothing ->
  739             let prefixCd = "cd " <> shellQuote wsRootPath <> " && "
  740                 fullCmd = T.pack prefixCd <> remoteEnvPrefix <> T.pack (T.unpack (textShellCommand cmd))
  741             in runRemoteShellForwardingEnvText runner sshCfg localEnv fullCmd
  742           Just rp ->
  743             case mkSafePathRemote (wsRootFromCfg sshCfg) (T.unpack (getRemotePath rp)) of
  744               Left pe  -> pure (Left (UePath pe))
  745               Right sp ->
  746                 let cdCmd = "cd " <> shellQuote (getSafePath sp) <> " && "
  747                     fullCmd = T.pack cdCmd <> remoteEnvPrefix <> T.pack (T.unpack (textShellCommand cmd))
  748                 in runRemoteShellForwardingEnvText runner sshCfg localEnv fullCmd
  749      , uioBinExecEnv = \extras bin bargs mCwd ->
  750          let argv' = T.unpack (textBinName bin) : map (T.unpack . textBinArg) bargs
  751          in case resolveBinCwd (wsRootFromCfg sshCfg) mCwd of
  752             Left pe  -> pure (Left (UePath pe))
  753             Right cwdPath ->
  754               let cdPart = T.pack ("cd " <> shellQuote cwdPath <> " && ")
  755                   cmd = T.pack (T.unpack (T.intercalate " " (map (T.pack . shellQuote) argv')))
  756               in if hasSecretExtras extras
  757                    then runRemoteShellStdinEnv runner sshCfg cdPart extras cmd
  758                    else runRemoteShellTextEnv runner sshCfg cdPart extras cmd
  759     , uioBinExecGitEnv = \extras _mKnownHosts bin bargs mCwd ->
  760         -- Remote deploy-key path: mirrors uioShellExecGitEnv. Strip
  761         -- SSH_AUTH_SOCK/SSH_AGENT_PID from the REMOTE env prefix (the
  762         -- forwarded agent replaces them). Keep them in the LOCAL ssh -A
  763         -- process env so the right agent is forwarded.
  764         let localEnv = [ (k, v) | (k, v) <- extras
  765                       , k == "SSH_AUTH_SOCK" || k == "SSH_AGENT_PID"
  766                       ]
  767             remoteExtras = [ (k, v) | (k, v) <- extras
  768                            , k /= "SSH_AUTH_SOCK"
  769                            , k /= "SSH_AGENT_PID"
  770                            ]
  771             argv' = T.unpack (textBinName bin) : map (T.unpack . textBinArg) bargs
  772             remoteEnvPrefix = T.pack (renderEnvPrefix remoteExtras)
  773         in case resolveBinCwd (wsRootFromCfg sshCfg) mCwd of
  774            Left pe  -> pure (Left (UePath pe))
  775            Right cwdPath ->
  776              let cdPart = T.pack ("cd " <> shellQuote cwdPath <> " && ")
  777                  cmd = T.pack (T.unpack (T.intercalate " " (map (T.pack . shellQuote) argv')))
  778              in runRemoteShellForwardingEnvText runner sshCfg localEnv (cdPart <> remoteEnvPrefix <> cmd)

  830 -- | Resolve an optional cwd ('RemotePath') for BIN_EXEC into an absolute
  831 -- 'FilePath' the executor can pass to the subprocess. Unlike
  832 -- 'uioShellExec' (which confines ALL paths — relative and absolute —
  833 -- under the workspace root via 'mkSafePathRemote'), BIN_EXEC treats an
  834 -- ABSOLUTE cwd as a pass-through (the operator explicitly chose an
  835 -- out-of-workspace directory); a RELATIVE cwd is SafePath-confined under
  836 -- the workspace root (the default untrusted-confinement contract).
  837 --
  838 --   * 'Nothing' → the workspace root itself (the one-root invariant).
  839 --   * 'Just rp' (absolute) → the path verbatim (no confinement).
  840 --   * 'Just rp' (relative) → 'mkSafePathRemote'-confined under the root;
  841 --     a @..@ escape or blocked name is rejected with 'Left'.
  842 resolveBinCwd :: WorkspaceRoot -> Maybe RemotePath -> Either PathError FilePath
  843 resolveBinCwd wsRoot mCwd =
  844   case mCwd of
  845     Nothing   -> Right wsRootPath
  846     Just rp ->
  847       let rel = T.unpack (getRemotePath rp)
  848       in if isAbsolute rel
  849            then Right rel
  850            else getSafePath <$> mkSafePathRemote wsRoot rel
  851   where
  852     wsRootPath = case mkSafePathRemote wsRoot "." of
  853       Right sp -> getSafePath sp
  854       Left _   -> "/nonexistent-workdir-fail-closed"

  856 -- | Single-quote a 'String' for the remote shell (the path is already
  857 -- SafePath-validated, but quoting is defense-in-depth against any
  858 -- metacharacters the validator permits, e.g. spaces). Embedded single
  859 -- quotes are escaped with the standard @'\''@ idiom so a value containing
  860 -- a quote cannot break out of its single-quoted argv token — e.g.
  861 -- @foo'bar@ becomes @'foo'\''bar'@, which the shell parses back to the
  862 -- literal @foo'bar@ as a single token. Applied uniformly to all remote
  863 -- paths and search patterns.
  864 shellQuote :: String -> String
  865 shellQuote s = "'" <> go s <> "'"
  866   where
  867     go []         = []
  868     go ('\'':rest) = "'\\''" <> go rest
  869     go (c:rest)    = c : go rest

  871 -- | Render the env-override prefix for the remote arm: @env VAR='val'
  872 -- VAR2='val2' @ (each value single-quoted via 'shellQuote' so a value
  873 -- containing spaces / quotes / @$()@ is passed as a single token — the
  874 -- @env@ command sets them before the command runs). The @env@ prefix is
  875 -- portable (@ssh -A@ only forwards the agent socket; @SendEnv@/@SetEnv@
  876 -- need server @AcceptEnv@ which is @none@ by default — design §4.4). The
  877 -- caller appends the command after the prefix.
  878 renderEnvPrefix :: [(String, String)] -> String
  879 renderEnvPrefix extras =
  880   if null extras
  881     then ""
  882     else "env " <> unwords [ k <> "=" <> shellQuote v | (k, v) <- extras ] <> " "

  911 -- 'shellCmd' parse failure is lifted to 'UeExec'; the 'runRemoteShell'
  912 -- 'ExecError' is lifted to 'UeExec'. Uniform across the remote arm.

  916   case shellCmd cmdText of
  917     Left e    -> pure (Left e)
  918     Right cmd -> do

  924 -- | Like 'runRemoteShellText' but with env overrides applied to the
  925 -- SUBPROCESS. The caller supplies the @cd@ prefix and the command SEPARATELY;
  926 -- the composed remote command is @\<cdPart\>env K='V' … \<cmd\>@ so the vars
  927 -- scope to the trailing command — NOT to the @cd@.
  928 --
  929 -- Placement is security-relevant (the PAT-clone bug): a POSIX shell scopes a
  930 -- leading @env@ assignment only to the single command that follows it, so an
  931 -- env prefix BEFORE @\<cd\> && \<cmd\>@ binds nothing for @\<cmd\>@. This
  932 -- mirrors the deploy-key path ('uioShellExecGitEnv'), which composes
  933 -- @cd … && env … git …@ for the same reason.
  934 --
  935 -- The 'logExecDebug' call receives the command WITHOUT the env prefix and
  936 -- the extras SEPARATELY so 'redactEnv' can redact secret-bearing keys before
  937 -- rendering (design §3.7): the logged message must not contain any value
  938 -- from a key in 'secretEnvKeys'.
  939 runRemoteShellTextEnv
  940   :: RemoteRunner -> SshConfig -> Text -> [(String, String)] -> Text
  941   -> IO (Either UntrustedErr Text)
  942 runRemoteShellTextEnv runner cfg cdPart extras cmd =
  943   let envPrefix = T.pack (renderEnvPrefix extras)
  944       fullCmdText = cdPart <> envPrefix <> cmd
  945       logCmdText = cdPart <> cmd
  946   in case shellCmd fullCmdText of
  947        Left e     -> pure (Left e)
  948        Right full -> do
  949          logCmd <- case shellCmd logCmdText of
  950            Left _   -> pure full
  951            Right l  -> pure l
  952          let argv = sshExecArgv cfg (textShellCommand logCmd)
  953          logExecDebug "[remote ssh]" argv Nothing extras
  954          res <- runRemoteShell runner cfg full
  955          pure (either (Left . UeExec) Right res)

  957 -- | Run a remote command with secret-bearing env vars piped over SSH
  958 -- stdin (NOT embedded in the command string). Used by
  959 -- 'uioShellExecEnv'/'uioBinExecEnv' when the extras contain any key in
  960 -- 'secretEnvKeys' (GH_TOKEN, GIT_CONFIG_PARAMETERS, etc.).
  961 --
  962 -- The remote command is wrapped in a POSIX @sh -c@ that reads one
  963 -- newline-separated value per @read -r KEY@, exports them, then @exec@s
  964 -- the target command. The stdin payload is the raw values joined by
  965 -- newlines (no shell quoting — @read -r@ handles arbitrary bytes except
  966 -- NUL/newline, which tokens never contain).
  967 --
  968 -- Security: the secret values travel over the SSH channel's stdin pipe
  969 -- (encrypted in transit, kernel-owned memory on both ends). They never
  970 -- appear in the command string, so they are NOT visible in:
  971 --   * the local @ssh@ process's argv (@\/proc\/<pid>\/cmdline@)
  972 --   * the remote shell's command line (@\/proc@ on the untrusted machine)
  973 --   * shell history (SSH commands are non-interactive)
  974 --
  975 -- The @cd@ prefix runs BEFORE the @sh -c@ wrapper (it's in the outer
  976 -- command string, not inside the @sh -c@), so the cwd is set by the
  977 -- remote shell before the inner @sh -c@ execs the target command. The
  978 -- @exec@ replaces the inner shell, so the target command inherits the
  979 -- cwd + the exported env vars.
  980 --
  981 -- The 'logExecDebug' call receives the command WITHOUT the secret
  982 -- values (they're on stdin, not in the command string) and the extras
  983 -- with redacted values (so the log shows which keys were set, not their
  984 -- values). The stdin payload itself is NEVER logged.
  985 runRemoteShellStdinEnv
  986   :: RemoteRunner -> SshConfig -> Text -> [(String, String)] -> Text
  987   -> IO (Either UntrustedErr Text)
  988 runRemoteShellStdinEnv runner cfg cdPart extras cmd =
  989   let keys = map fst extras
  990       readCmds = T.intercalate "; " (map (\k -> T.pack ("read -r " <> k)) keys)
  991       exportList = T.unwords (map T.pack keys)
  992       -- Escape single quotes in cmd so it can be embedded inside the
  993       -- outer sh -c '...' wrapper (each ' becomes '\'' — the standard
  994       -- POSIX single-quote escape idiom).
  995       escapedCmd = T.replace "'" "'\\''" cmd
  996       innerCmd = "sh -c '" <> readCmds <> "; export " <> exportList
  997                  <> "; exec " <> escapedCmd <> "'"
  998       fullCmdText = cdPart <> innerCmd
  999       stdinPayload = BS.intercalate (BS.singleton 10) (map (TE.encodeUtf8 . T.pack . snd) extras) <> BS.singleton 10
 1000   in case shellCmd fullCmdText of
 1001        Left e     -> pure (Left e)
 1002        Right full -> do
 1003          let argv = sshExecArgv cfg (textShellCommand full)
 1004          logExecDebug "[remote ssh -stdin]" argv Nothing extras
 1005          res <- runRemoteStdin runner argv stdinPayload
 1006          pure (either (Left . UeExec) Right res)

 1013 -- | Apply a minimal unified diff to the original content. Returns
 1014 -- @Left errMsg@ if the patch is malformed or the context doesn't match;
 1015 -- @Right newContent@ on success. This is the same applier that lived in
 1016 -- 'Seal.ISA.Ops.File.applyUnifiedDiff' — lifted here so both arms share
 1017 -- one implementation.
 1018 --
 1019 -- The applier is **content-based**, not position-based: for each hunk it
 1020 -- extracts the old lines (context @ @ + removed @-@), searches for that
 1021 -- sequence in the file starting near the hunk header's line-number hint,
 1022 -- and splices in the new lines (context @ @ + added @+@). The line number
 1023 -- is a hint to disambiguate duplicate matches, NOT a trusted insertion
 1024 -- index — this is the design used by OpenCode (@seek@ in patch.ts) and
 1025 -- Hermes (@fuzzy_find_and_replace@). A position-based applier that
 1026 -- blindly @splitAt@s at the header's oldStart corrupts the file when:
 1027 --   * the model's line number is stale (file changed since it read it),
 1028 --   * an earlier hunk in the same patch shifted line numbers, or
 1029 --   * the model miscounted lines.
 1030 -- All three were observed in session 20260810-123133-477 ("The patch is
 1031 -- inserting lines at wrong positions").
 1032 applyUnifiedDiff :: Text -> Text -> Either Text Text
 1033 applyUnifiedDiff original patch =
 1034   let origLines  = T.lines original
 1035       patchLines = T.lines patch
 1036   in go origLines patchLines
 1037   where
 1038     -- All hunks apply against the ORIGINAL file's line numbering; we
 1039     -- thread the already-patched lines through 'go' so successive hunks
 1040     -- see a consistent view. The search-start hint is carried in the
 1041     -- accumulator so a later hunk's line-number hint still points into
 1042     -- the original's coordinate space (we never let an earlier hunk's
 1043     -- insertions/deletions shift a later hunk's target).
 1044     go origLines [] = Right (joinLines origLines)
 1045     go origLines (h : rest)
 1046       | T.isPrefixOf "@@ " h  = applyHunk origLines h rest
 1047       | T.isPrefixOf "--- " h = go origLines rest
 1048       | T.isPrefixOf "+++ " h = go origLines rest
 1049       | T.null h              = go origLines rest
 1050       | otherwise             = Left ("unexpected line in patch: " <> h)
 1051 
 1052     applyHunk origLines header rest =
 1053       case parseHunkHeader header of
 1054         Left err -> Left err
 1055         Right (oldStart, mOldLen) ->
 1056           let (hunkLines, remainingPatch) = span isHunkLine rest
 1057               (oldLines, _newLines) = splitHunk hunkLines
 1058               -- The line-number hint is a 0-based index into the file
 1059               -- pointing at where the hunk's old lines BEGIN. For a
 1060               -- normal hunk (oldLen >= 1) that's oldStart-1 (1-based ->
 1061               -- 0-based). For a PURE-INSERT hunk (oldLen == 0) the
 1062               -- header's oldStart is the line AFTER which to insert
 1063               -- (1-based), so the 0-based insertion index is oldStart
 1064               -- itself (insert after that line). The @@ -0,0 +1,N @@
 1065               -- form (insert at start of empty file) gives oldStart=0,
 1066               -- which clamps to index 0.
 1067               isPureInsert = case mOldLen of
 1068                 Just 0  -> True
 1069                 Nothing -> null oldLines   -- short-form header; infer
 1070                 _       -> False
 1071               hint = if isPureInsert
 1072                        then max 0 oldStart        -- insert after line oldStart
 1073                        else max 0 (oldStart - 1) -- replace starting at oldStart
 1074           in case findMatch origLines oldLines hint of
 1075                Left err -> Left err
 1076                Right idx ->
 1077                  let (before, atAndAfter) = splitAt idx origLines
 1078                      matched = take (length oldLines) atAndAfter
 1079                      after   = drop (length oldLines) atAndAfter
 1080                      -- Build the replacement: walk the hunk body and the
 1081                      -- matched file lines in parallel. For context (' ')
 1082                      -- lines we prefer the FILE's actual text (so a
 1083                      -- fuzzy match doesn't drag the patch's trailing
 1084                      -- whitespace into the file); for '+' lines we use
 1085                      -- the patch's text; '-' lines are dropped.
 1086                      replacement = spliceHunk hunkLines matched
 1087                  in go (before ++ replacement ++ after) remainingPatch
 1088 
 1089     -- | Walk the hunk body and the matched file lines in parallel,
 1090     -- producing the replacement line list. Context (' ') lines take
 1091     -- the FILE's line (preserving the file's actual whitespace);
 1092     -- '+' lines take the patch's text; '-' and '\\' lines are dropped.
 1093     spliceHunk :: [Text] -> [Text] -> [Text]
 1094     spliceHunk = go'
 1095       where
 1096         go' [] _ = []
 1097         go' (l : ls) fileLines =
 1098           case T.uncons l of
 1099             Just (' ', _) -> case fileLines of
 1100               (f : fs) -> f : go' ls fs
 1101               []       -> go' ls []   -- shouldn't happen; findMatch guards
 1102             Just ('+', body') -> body' : go' ls fileLines
 1103             Just ('-', _) -> go' ls (drop 1 fileLines)
 1104             Just ('\\', _) -> go' ls fileLines
 1105             _ -> go' ls fileLines   -- empty/malformed: skip, keep file pos
 1106 
 1107     -- | Split hunk body lines into (oldLines, newLines). Context lines
 1108     -- (' ' prefix) appear in BOTH; removed lines ('-') in old only;
 1109     -- added lines ('+') in new only. '\\' (no-newline-at-eof) and empty
 1110     -- lines are ignored.
 1111     splitHunk :: [Text] -> ([Text], [Text])
 1112     splitHunk ls =
 1113       let go' []         = ([], [])
 1114           go' (l : rest) =
 1115             case T.uncons l of
 1116               Just (' ', body) -> let (o, n) = go' rest in (body : o, body : n)
 1117               Just ('-', body) -> let (o, n) = go' rest in (body : o, n)
 1118               Just ('+', body) -> let (o, n) = go' rest in (o, body : n)
 1119               Just ('\\', _)   -> go' rest
 1120               _                -> go' rest  -- empty or malformed: skip
 1121       in go' ls
 1122 
 1123     -- | Find the index in @lines@ where the @pattern@ sequence matches,
 1124     -- searching near the @hint@ first, then the whole file. Tries four
 1125     -- comparison strategies in order (exact -> rstrip -> trim ->
 1126     -- normalized) so trailing-whitespace and Unicode punctuation
 1127     -- differences don't cause a hard failure — mirrors OpenCode's
 1128     -- @seek@ fallback ladder. An empty pattern (pure-insert hunk) lands
 1129     -- at the hint index (insert after the hinted line).
 1130     findMatch :: [Text] -> [Text] -> Int -> Either Text Int
 1131     findMatch _    []     hint = Right hint
 1132     findMatch xs pat    hint =
 1133       case searchNear xs pat hint of
 1134         Just idx -> Right idx
 1135         Nothing  ->
 1136           case searchAll xs pat of
 1137             Just idx -> Right idx
 1138             Nothing  ->
 1139               Left ("FILE_PATCH: context not found near line "
 1140                       <> T.pack (show (hint + 1)) <> ":\n"
 1141                       <> T.intercalate "\n" pat)
 1142 
 1143     searchNear xs pat hint =
 1144       let n      = length xs
 1145           plen   = length pat
 1146           -- Search outward from the hint: try hint, hint±1, hint±2, ...
 1147           -- so the closest match to the claimed line wins (disambiguates
 1148           -- duplicates by proximity to the hint).
 1149           offsets = concatMap (\d -> [hint + d, hint - d]) [0..(n - 1)]
 1150           valid  = filter (\i -> i >= 0 && i <= n - plen) offsets
 1151       in firstMatch xs pat valid
 1152 
 1153     searchAll xs pat =
 1154       let n    = length xs
 1155           plen = length pat
 1156       in firstMatch xs pat [0 .. n - plen]
 1157 
 1158     firstMatch _    _   []       = Nothing
 1159     firstMatch xs pat (i : is) =
 1160       if matchesAt xs pat i
 1161         then Just i
 1162         else firstMatch xs pat is
 1163 
 1164     matchesAt xs pat i =
 1165       any (\cmp -> and (zipWith cmp (drop i xs) pat))
 1166            [exactEq, rstripEq, trimEq, normalizedEq]
 1167 
 1168     exactEq a b = a == b
 1169     rstripEq a b = T.dropWhileEnd isSpace a == T.dropWhileEnd isSpace b
 1170     trimEq a b = T.dropWhile isSpace (T.dropWhileEnd isSpace a)
 1171               == T.dropWhile isSpace (T.dropWhileEnd isSpace b)
 1172     normalizedEq a b = normalize (trimLine a) == normalize (trimLine b)
 1173       where
 1174         trimLine = T.dropWhile isSpace . T.dropWhileEnd isSpace
 1175         normalize = T.map (\c -> if c == '\t' then ' ' else c)
 1176 
 1177     isHunkLine l =
 1178       T.null l
 1179       || T.isPrefixOf " " l
 1180       || T.isPrefixOf "-" l
 1181       || T.isPrefixOf "+" l
 1182       || T.isPrefixOf "\\" l
 1183 
 1184     -- | Join lines back into text, preserving the trailing newline if
 1185     -- the original had one (T.lines drops it; we re-add it so a file
 1186     -- ending in a newline round-trips).
 1187     joinLines [] = ""
 1188     joinLines ls = T.intercalate "\n" ls <> "\n"
 1189 
 1190     parseHunkHeader h =
 1191       case T.stripPrefix "@@ -" h of
 1192         Nothing -> Left ("malformed hunk header: " <> h)
 1193         Just rest0 ->
 1194           let (oldStartStr, afterOld) = breakNum rest0
 1195               (oldLenStr, afterOldLen) = breakLen afterOld
 1196               afterPlus0 = T.dropWhile (/= '+') afterOldLen
 1197               afterPlus  = T.drop 1 afterPlus0
 1198               (newStartStr, _) = breakNum afterPlus
 1199               oldStart = readMaybe (T.unpack oldStartStr) :: Maybe Int
 1200               _newStart = readMaybe (T.unpack newStartStr) :: Maybe Int
 1201           in case oldStart of
 1202                Just os_ -> Right (os_, readMaybe (T.unpack oldLenStr) :: Maybe Int)
 1203                Nothing  -> Left ("malformed hunk header numbers: " <> h)
 1204       where
 1205         breakNum :: Text -> (Text, Text)
 1206         breakNum s =
 1207           let (digits, rest) = T.span isDigit s
 1208           in case T.uncons rest of
 1209                Just (',', rest') -> (digits, T.dropWhile isDigit rest')
 1210                _                 -> (digits, rest)
 1211         -- | After the oldStart number, an optional @,len@ may follow.
 1212         -- Return the len string (empty if the short form was used) and
 1213         -- the rest after the @,len@ (or after the start number if short
 1214         -- form). The length is informational (the content search is the
 1215         -- source of truth); we keep it only for diagnostics.
 1216         breakLen :: Text -> (Text, Text)
 1217         breakLen s =
 1218           case T.uncons s of
 1219             Just (',', rest) -> T.span isDigit rest
 1220             _                -> ("", s)
 1221         readMaybe :: String -> Maybe Int
 1222         readMaybe s = case reads s :: [(Int, String)] of
 1223           [(n, _)] -> Just n
 1224           _        -> Nothing

    2 -- | The cron store backend. Disk is canonical: jobs live as JSON files
    3 -- under @config\/cron\/jobs\/\<id\>.json@ and run history as append-only
    4 -- JSONL at @config\/cron\/runs\/\<id\>.jsonl@ (rotated to last 100).
    5 module Seal.Cron.Store
    6   ( CronStore (..)
    7   , fileCronStore
    8   , memoryCronStore
    9   ) where

   35 data CronStore = CronStore
   36   { csCreate    :: CronJob -> IO ()
   37   , csRead      :: CronJobId -> IO (Maybe CronJob)
   38   , csList      :: Bool -> IO [CronJob]
   39   , csSave      :: CronJob -> IO ()
   40   , csDelete    :: CronJobId -> IO ()
   41   , csRecordRun :: CronJobId -> CronRun -> IO ()
   42   , csListRuns  :: CronJobId -> Int -> IO [CronRun]
   43   , csUpdateRun :: CronJobId -> (CronRun -> CronRun) -> IO ()
   44     -- ^ Apply a pure transformation to every run row of a job. Keyed on the job,
   45     -- not the session: a row exists from the moment a fire starts (before it
   46     -- has a session id), and reconciliation must repair a whole history, not
   47     -- just its newest row. Both callers pass a terminal-respecting
   48     -- transformation, so mapping all rows is idempotent. A job with no runs is
   49     -- a no-op.
   50   , csLock      :: MVar ()
   51     -- ^ Serializes every mutation. The file backend does whole-file
   52     -- read-modify-write on both jobs (one file) and runs (one JSONL), so two
   53     -- writers (the scheduler tick and a turn-completion callback) would
   54     -- otherwise lose updates — and a lost update is exactly what re-wedges a
   55     -- job. 'seal serve' is single-process per SEAL_HOME, so an in-process lock
   56     -- is sufficient.
   57   }
   58 
   59 fileCronStore :: FilePath -> ConfigRepo -> IO CronStore
   60 fileCronStore dir repo = do
   61   let jobsDir = dir </> "jobs"
   62       runsDir = dir </> "runs"
   63   createDirectoryIfMissing True jobsDir
   64   createDirectoryIfMissing True runsDir
   65   lock <- newMVar ()
   66   pure CronStore
   67     { csCreate    = \j -> withMVar lock (\_ -> writeJobFile jobsDir repo j)
   68     , csRead      = readJobFile jobsDir
   69     , csList      = listJobs jobsDir
   70     , csSave      = \j -> withMVar lock (\_ -> writeJobFile jobsDir repo j)
   71     , csDelete    = \jid -> withMVar lock (\_ -> deleteJobFile jobsDir repo jid)
   72     , csRecordRun = \jid run -> withMVar lock (\_ -> recordRun runsDir repo jid run)
   73     , csListRuns  = listRuns runsDir
   74     , csUpdateRun = \jid f -> withMVar lock (\_ -> updateRunFile runsDir repo jid f)
   75     , csLock      = lock
   76     }
   77 
   78 writeJobFile :: FilePath -> ConfigRepo -> CronJob -> IO ()
   79 writeJobFile jobsDir repo job = do
   80   let path = jobFilePath jobsDir (cjId job)
   81       tmp  = path <.> "tmp"
   82   createDirectoryIfMissing True (takeDirectory path)
   83   TIO.writeFile tmp (TL.toStrict (TLE.decodeUtf8 (encode job)))
   84   setFileMode tmp 0o600
   85   renameFile tmp path
   86   let rel = "cron/jobs" </> T.unpack (cronJobIdText (cjId job)) <.> "json"
   87   _ <- gitCommitAll repo rel ("seal: CRON write " <> cronJobIdText (cjId job))
   88   pure ()
   89 
   90 readJobFile :: FilePath -> CronJobId -> IO (Maybe CronJob)
   91 readJobFile jobsDir jid = do
   92   let path = jobFilePath jobsDir jid
   93   exists <- doesFileExist path
   94   if not exists
   95     then pure Nothing
   96     else do
   97       content <- TIO.readFile path
   98       case eitherDecode (TLE.encodeUtf8 (TL.fromStrict content)) of
   99         Left _    -> pure Nothing
  100         Right job -> pure (Just job)
  101 
  102 listJobs :: FilePath -> Bool -> IO [CronJob]
  103 listJobs jobsDir enabledOnly = do
  104   exists <- doesDirectoryExist jobsDir
  105   if not exists
  106     then pure []
  107     else do
  108       entries <- listDirectory jobsDir
  109       let jsonFiles = filter (isSuffixOf ".json") entries
  110       jobs <- mapM (readJobByName jobsDir) jsonFiles
  111       let allJobs = catMaybes jobs
  112           filtered = if enabledOnly
  113                        then filter cjEnabled allJobs
  114                        else allJobs
  115       pure (sortOn (cronJobIdText . cjId) filtered)
  116 
  117 deleteJobFile :: FilePath -> ConfigRepo -> CronJobId -> IO ()
  118 deleteJobFile jobsDir repo jid = do
  119   let path = jobFilePath jobsDir jid
  120   exists <- doesFileExist path
  121   when exists $ do
  122     removeFile path
  123     let rel = "cron/jobs" </> T.unpack (cronJobIdText jid) <.> "json"
  124     _ <- gitCommitAll repo rel ("seal: CRON delete " <> cronJobIdText jid)
  125     pure ()
  126 
  127 recordRun :: FilePath -> ConfigRepo -> CronJobId -> CronRun -> IO ()
  128 recordRun runsDir repo jid run = do
  129   let path = runFilePath runsDir jid
  130   createDirectoryIfMissing True (takeDirectory path)
  131   existing <- readRunFile path
  132   let updated = take 100 (run : existing)
  133   writeRunFile path updated
  134   let rel = "cron/runs" </> T.unpack (cronJobIdText jid) <.> "jsonl"
  135   _ <- gitCommitAll repo rel ("seal: CRON run record " <> cronJobIdText jid)
  136   pure ()

  138 -- | Apply a transformation to every run row of a job (order preserved, 100-run
  139 -- cap preserved). Both callers pass a transformation that respects terminal
  140 -- rows, so mapping the whole history is idempotent and is what lets startup
  141 -- reconciliation repair every row a previous process wedged — not just the
  142 -- newest. When nothing changes the file is not rewritten and no commit is made,
  143 -- so the common case costs one read.
  144 updateRunFile :: FilePath -> ConfigRepo -> CronJobId -> (CronRun -> CronRun) -> IO ()
  145 updateRunFile runsDir repo jid f = do
  146   let path = runFilePath runsDir jid
  147   exists <- doesFileExist path
  148   when exists $ do
  149     existing <- readRunFile path
  150     let updated = map f existing
  151     if updated == existing
  152       then pure ()
  153       else do
  154         writeRunFile path updated
  155         let rel = "cron/runs" </> T.unpack (cronJobIdText jid) <.> "jsonl"
  156         _ <- gitCommitAll repo rel ("seal: CRON run update " <> cronJobIdText jid)
  157         pure ()
  158 
  159 listRuns :: FilePath -> CronJobId -> Int -> IO [CronRun]
  160 listRuns runsDir jid limit = do
  161   let path = runFilePath runsDir jid
  162   exists <- doesFileExist path
  163   if not exists
  164     then pure []
  165     else do
  166       runs <- readRunFile path
  167       pure (take limit runs)
  168 
  169 jobFilePath :: FilePath -> CronJobId -> FilePath
  170 jobFilePath jobsDir jid = jobsDir </> T.unpack (cronJobIdText jid) <.> "json"
  171 
  172 runFilePath :: FilePath -> CronJobId -> FilePath
  173 runFilePath runsDir jid = runsDir </> T.unpack (cronJobIdText jid) <.> "jsonl"
  174 
  175 readRunFile :: FilePath -> IO [CronRun]
  176 readRunFile path = do
  177   exists <- doesFileExist path
  178   if not exists
  179     then pure []
  180     else do
  181       content <- TIO.readFile path
  182       let lines' = filter (not . T.null) (T.lines content)
  183           decoded = [ eitherDecode (TLE.encodeUtf8 (TL.fromStrict line))
  184                     | line <- lines' ]
  185       pure (rights decoded)
  186 
  187 writeRunFile :: FilePath -> [CronRun] -> IO ()
  188 writeRunFile path runs = do
  189   let tmp = path <.> "tmp"
  190       content = T.intercalate "\n"
  191                   [ TL.toStrict (TLE.decodeUtf8 (encode run))
  192                   | run <- runs ]
  193   TIO.writeFile tmp content
  194   setFileMode tmp 0o600
  195   renameFile tmp path
  196 
  197 readJobByName :: FilePath -> FilePath -> IO (Maybe CronJob)
  198 readJobByName jobsDir name = do
  199   let path = jobsDir </> name
  200   content <- TIO.readFile path
  201   case eitherDecode (TLE.encodeUtf8 (TL.fromStrict content)) of
  202     Left _    -> pure Nothing
  203     Right job -> pure (Just job)

  205 memoryCronStore :: IO CronStore
  206 memoryCronStore = do
  207   jobsRef <- newIORef (Map.empty :: Map CronJobId CronJob)
  208   runsRef <- newIORef (Map.empty :: Map CronJobId [CronRun])
  209   lock <- newMVar ()
  210   pure CronStore
  211     { csCreate = \job -> modifyIORef' jobsRef (Map.insert (cjId job) job)
  212     , csRead   = \jid -> do
  213         m <- readIORef jobsRef
  214         pure (Map.lookup jid m)
  215     , csList   = \enabledOnly -> do
  216         m <- readIORef jobsRef
  217         let allJobs = Map.elems m
  218             filtered = if enabledOnly
  219                          then filter cjEnabled allJobs
  220                          else allJobs
  221         pure (sortOn (cronJobIdText . cjId) filtered)
  222     , csSave   = \job -> modifyIORef' jobsRef (Map.insert (cjId job) job)
  223     , csDelete = modifyIORef' jobsRef . Map.delete
  224     , csRecordRun = \jid run -> withMVar lock (\_ -> modifyIORef' runsRef
  225         (Map.insertWith (\_new old -> take 100 (run : old)) jid [run]))
  226     , csListRuns = \jid limit -> do
  227         m <- readIORef runsRef
  228         pure (take limit (fromMaybe [] (Map.lookup jid m)))
  229     , csUpdateRun = \jid f -> withMVar lock (\_ -> do
  230         m <- readIORef runsRef
  231         case Map.lookup jid m of
  232           Nothing -> pure ()
  233           Just rs  -> modifyIORef' runsRef (Map.insert jid (map f rs)))
  234     , csLock = lock
  235     }

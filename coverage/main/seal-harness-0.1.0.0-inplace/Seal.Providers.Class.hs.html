   52 -- | A single block of content in a message.
   53 --
   54 -- __Partial field selectors warning:__ The record fields 'cbId', 'cbName',
   55 -- 'cbInput', 'cbForId', 'cbParts', and 'cbIsError' are only defined on their
   56 -- respective constructors ('CbToolUse' and 'CbToolResult'). They are partial
   57 -- across the full type and will throw a runtime error if applied to the wrong
   58 -- constructor. Always access them inside a constructor-specific pattern match;
   59 -- never use them as bare accessor functions.
   60 --
   61 -- JSON keys (via 'stripPrefixToJSON'): @cbId@→@id@, @cbName@→@name@,
   62 -- @cbInput@→@input@, @cbForId@→@forId@, @cbParts@→@parts@,
   63 -- @cbIsError@→@isError@. The constructor tag (@CbText@/@CbToolUse@/
   64 -- @CbToolResult@) is emitted by aeson's default 'TaggedObject' sum encoding
   65 -- unchanged.
   66 -- @CbThinking@ is serialized as @{"tag":"CbThinking","contents":"..."}@
   67 -- and rewritten to @{"type":"thinking","thinking":"..."}@ for the frontend.
   68 data ContentBlock
   69   = CbText Text
   70   | CbThinking Text
   71   | CbToolUse    { cbId :: ToolCallId, cbName :: OpName, cbInput :: Value }
   72   | CbToolResult { cbForId :: ToolCallId, cbParts :: [ToolResultPart], cbIsError :: Bool }
   73   deriving stock (Eq, Show, Generic)
   74 instance ToJSON ContentBlock where toJSON = stripPrefixToJSON
   75 instance FromJSON ContentBlock where parseJSON = stripPrefixParseJSON

   77 -- | Provenance of a single conversation message: the channel it arrived on
   78 -- and, when that channel authenticates one, the sender.
   79 --
   80 -- This is transcript metadata, not provider payload: the concrete provider
   81 -- encoders ('Seal.Providers.Anthropic', 'Seal.Providers.Ollama',
   82 -- 'Seal.Providers.OpenRouter') pattern-match the role and content blocks
   83 -- only, so an origin never reaches the wire as a structured field. What the
   84 -- model sees instead is the rendered tag produced by
   85 -- 'Seal.Transcript.Origin.annotateOrigins' when the loop builds the request.
   86 --
   87 -- 'moChannel' is the lowercase channel tag (see
   88 -- 'Seal.Core.ChannelKind.channelKindToText': @\"signal\"@, @\"telegram\"@, …);
   89 -- 'moUserId' is the authenticated sender id exactly as the channel reported
   90 -- it (a Signal phone/UUID, a Telegram numeric id) — already validated and
   91 -- bounded by 'Seal.Core.MessageSource.mkMessageSource', never raw transport
   92 -- text; 'moConversationId' is the server-derived conversation key.
   93 --
   94 -- Every field is persisted in @conversation.jsonl@, so it must stay
   95 -- secret-free.
   96 data MessageOrigin = MessageOrigin
   97   { moChannel :: Text
   98   , moUserId :: Maybe Text
   99   , moConversationId :: Maybe Text
  100   , moPasskeyId :: Maybe Text
  101   , moClientIp :: Maybe Text
  102   } deriving stock (Eq, Show, Generic)
  103 
  104 -- JSON keys: @channel@, @userId@, @conversationId@; absent fields are omitted
  105 -- (not null) so a per-message origin stays compact.
  106 instance ToJSON MessageOrigin where
  107   toJSON o = object $ ["channel" .= moChannel o] <> userIdField <> conversationIdField <> passkeyIdField <> clientIpField
  108     where
  109       userIdField = case moUserId o of
  110         Nothing -> []
  111         Just u  -> ["userId" .= u]
  112       conversationIdField = case moConversationId o of
  113         Nothing -> []
  114         Just c  -> ["conversationId" .= c]
  115       passkeyIdField = case moPasskeyId o of
  116         Nothing -> []
  117         Just p  -> ["passkeyId" .= p]
  118       clientIpField = case moClientIp o of
  119         Nothing -> []
  120         Just i  -> ["clientIp" .= i]
  121 
  122 instance FromJSON MessageOrigin where
  123   parseJSON = withObject "MessageOrigin" $ \o -> MessageOrigin
  124     <$> o .: "channel"
  125     <*> o .:? "userId"
  126     <*> o .:? "conversationId"
  127     <*> o .:? "passkeyId"
  128     <*> o .:? "clientIp"

  130 -- | One conversation message: a role, its content blocks, and — for a message
  131 -- that arrived from an authenticated chat channel — its 'MessageOrigin'.
  132 --
  133 -- The 'Message' pattern is the two-argument view of this type, and it is what
  134 -- nearly every call site uses: it builds a message with no origin and matches
  135 -- one while ignoring the origin. Provider encoders, tool dispatch, and the
  136 -- content helpers therefore keep working unchanged — provenance rides
  137 -- alongside the payload, it is not part of it. Attach an origin with a record
  138 -- update (@m { msgOrigin = Just o }@), which needs no constructor access.
  139 data Message = MkMessage
  140   { msgRole :: Role
  141   , msgContent :: [ContentBlock]
  142   , msgOrigin :: Maybe MessageOrigin
  143   } deriving stock (Eq, Show, Generic)

  145 -- | Build (and match) a message with no origin.
  146 pattern Message :: Role -> [ContentBlock] -> Message
  147 pattern Message r c <- MkMessage r c _
  148   where
  149     Message r c = MkMessage r c Nothing

  153 -- JSON keys: @role@, @content@, and @origin@ when present. An origin-less
  154 -- message encodes exactly as it did before origins existed, so
  155 -- @conversation.jsonl@ lines written by an older binary still decode (a
  156 -- missing @origin@ means 'Nothing') and new web/CLI turns keep the old shape.
  157 instance ToJSON Message where
  158   toJSON (MkMessage r c mo) = object $
  159     ["role" .= r, "content" .= c] <> originField mo
  160     where
  161       originField = \case
  162         Nothing -> []
  163         Just o  -> ["origin" .= o]
  164 
  165 instance FromJSON Message where
  166   parseJSON = withObject "Message" $ \o ->
  167     MkMessage <$> o .: "role" <*> o .: "content" <*> o .:? "origin"
  168 
  169 textMsg :: Role -> Text -> Message
  170 textMsg r t = Message r [CbText t]

  176 -- | Custom 'ToJSON': emits @name@ + @description@ always, and
  177 -- @input_schema@ only when it differs from 'stubSchema'. The keys follow the
  178 -- Anthropic wire shape (the @tools@ array is an external-facing payload, so
  179 -- the external convention applies — not the internal strip-prefix camelCase).
  180 -- This keeps the on-disk transcript envelope (@edTools@) and the debug
  181 -- @requests.jsonl@ consistent with the provider encoders (which also use
  182 -- @input_schema@ / @function.parameters@). The stub is omitted entirely so
  183 -- no @input_schema@ token cost leaks in on-demand mode.
  184 instance ToJSON ToolDefinition where
  185   toJSON (ToolDefinition n d sch) =
  186     if sch == stubSchema
  187       then object ["name" .= n, "description" .= d]
  188       else object ["name" .= n, "description" .= d, "input_schema" .= sch]
  189 
  190 -- | Custom 'FromJSON': @name@ + @description@ required; @input_schema@
  191 -- OPTIONAL, defaulting to 'stubSchema' when the key is ABSENT. The 'ToJSON'
  192 -- instance omits @input_schema@ for stub tools, so a required-field reader
  193 -- would reject those rows and — worse — take down the whole 'EnvelopeDelta'
  194 -- parse (aeson's 'parseJSON' is sequential), losing the system prompt from
  195 -- the reconstructed envelope. Key PRESENCE (not value null-ness)
  196 -- distinguishes "stored stub, omitted by ToJSON" from "explicitly null": an
  197 -- absent key defaults to 'stubSchema' (closing the stub round-trip), while a
  198 -- present 'null' stays 'Null' (so the arbitrary-data round-trip property
  199 -- still holds).
  200 instance FromJSON ToolDefinition where
  201   parseJSON = withObject "ToolDefinition" $ \o -> ToolDefinition
  202     <$> o .: "name"
  203     <*> o .: "description"
  204     <*> pure (fromMaybe stubSchema (KeyMap.lookup (fromString "input_schema") o))
  205 
  206 -- | The minimal placeholder @input_schema@ emitted when on-demand schema
  207 -- loading is enabled. Both Anthropic and Ollama normally require an
  208 -- @input_schema@ / @parameters@ field on every tool definition; when a
  209 -- 'ToolDefinition' carries this exact value, the provider encoders OMIT
  210 -- the field entirely rather than sending it inline — saving the few tokens
  211 -- the stub would otherwise cost on every tool, every turn. The model is
  212 -- expected to call @OPCODE_DESCRIBE@ to retrieve a tool's real schema
  213 -- before calling it.
  214 stubSchema :: Value
  215 stubSchema = object ["type" .= ("object" :: Text)]

  248 class Provider p where
  249   complete   :: p -> CompletionRequest -> IO (Either Text CompletionResponse)
  250   listModels :: p -> IO (Either Text [ModelId])
  251   -- | Stream a completion request, invoking the callback for each 'StreamEvent'
  252   -- as it arrives. The callback returns 'True' to continue streaming or 'False'
  253   -- to abort early (e.g. the consumer is no longer interested). The final
  254   -- 'StreamOutcome' carries the aggregated stop reason + usage. Providers
  255   -- that do not support streaming can fall back to 'complete' + emit a single
  256   -- 'StreamContentChunk' + 'StreamDone' (the default implementation does this).
  257   streamComplete
  258     :: p -> CompletionRequest -> (StreamEvent -> IO Bool) -> IO (Either Text StreamOutcome)
  259   streamComplete p cr k = do
  260     -- Default: use the non-streaming 'complete' and emit events for each
  261     -- content block (text + tool calls), then a final 'StreamDone'.
  262     eresp <- complete p cr
  263     case eresp of
  264       Left err -> pure (Left err)
  265       Right resp -> do
  266         mapM_ (emitBlock k) (rsContent resp)
  267         _ <- k (StreamDone (rsStop resp) (rsUsage resp))
  268         pure (Right (StreamOutcome (rsStop resp) (rsUsage resp)))
  269 
  270 -- | Emit 'StreamEvent's for a single 'ContentBlock' (used by the default
  271 -- 'streamComplete' implementation that wraps the non-streaming 'complete').
  272 emitBlock :: (StreamEvent -> IO Bool) -> ContentBlock -> IO ()
  273 emitBlock k block = case block of
  274   CbText t -> unless (T.null t) $
  275     void (k (StreamTextChunk t))
  276   CbThinking t -> unless (T.null t) $
  277     void (k (StreamThinkingChunk t))
  278   CbToolUse tcid (OpName name) args ->
  279     void (k (StreamToolStart tcid (OpName name)) >>
  280           k (StreamToolEnd tcid (OpName name) args))
  281   CbToolResult{} -> pure ()  -- tool results are not model output
  282 
  283 data SomeProvider = forall p. Provider p => SomeProvider p

  285 -- | A single streaming event emitted by 'streamComplete'. Mirrors the
  286 -- 'LLMEvent' contract from OpenCode: text deltas, tool-call lifecycle
  287 -- (start/delta/end), and a final done signal carrying the stop reason +
  288 -- usage. The events arrive incrementally as the model generates; the
  289 -- consumer aggregates them into 'ContentBlock's and writes to the
  290 -- transcript live.
  291 data StreamEvent
  292   = StreamTextChunk Text
  293   -- ^ A chunk of assistant text. Multiple chunks concatenate to form the
  294   -- final 'CbText' block(s).
  295   | StreamThinkingChunk Text
  296   -- ^ A chunk of model reasoning (extended thinking). Multiple chunks
  297   -- concatenate to form the final 'CbThinking' block.
  298   | StreamToolStart ToolCallId OpName
  299   -- ^ The model began emitting a tool call. The id is synthesized by the
  300   -- provider (Ollama carries no ids; the provider assigns sequential ones).
  301   | StreamToolDelta ToolCallId Text
  302   -- ^ A chunk of the tool-call's JSON arguments. Multiple deltas
  303   -- concatenate to form the final 'cbInput' 'Value' (parsed at 'StreamToolEnd').
  304   | StreamToolEnd ToolCallId OpName Value
  305   -- ^ A tool call completed. Carries the fully-parsed 'cbInput'.
  306   | StreamDone StopReason Usage
  307   -- ^ The generation finished. Carries the final stop reason + usage. This
  308   -- is always the last event (unless the stream was aborted by the callback
  309   -- returning 'False' or a transport error).
  310   deriving stock (Eq, Show, Generic)
  311 instance ToJSON StreamEvent
  312 
  313 -- | The aggregated outcome of a completed stream: the final stop reason +
  314 -- usage. The loop uses this (alongside the accumulated 'ContentBlock's) to
  315 -- decide whether to continue (tool calls / truncation) or finish.
  316 data StreamOutcome = StreamOutcome
  317   { soStop :: StopReason
  318   , soUsage :: Usage
  319   } deriving stock (Eq, Show, Generic)
  320 instance ToJSON StreamOutcome

  322 -- | Retry policy for transient provider failures during streaming (same as
  323 -- the non-streaming path: up to 2 retries, 50ms exponential backoff).
  324 streamRetryPolicy :: RetryPolicyM IO
  325 streamRetryPolicy =
  326   exponentialBackoff 50_000 <> limitRetries 2

  328 -- | Call 'streamComplete' with retry on transient errors. Mirrors
  329 -- 'providerCompleteWithRetry' in the non-streaming path. The transient-error
  330 -- predicate is the same ('isTransientError' lives in the loop; here we
  331 -- accept a predicate so the streaming provider can reuse it).
  332 providerStreamWithRetry
  333   :: SomeProvider -> (Text -> Bool) -> CompletionRequest
  334   -> (StreamEvent -> IO Bool) -> IO (Either Text StreamOutcome)
  335 providerStreamWithRetry (SomeProvider p) isTransient cr k =
  336   retrying streamRetryPolicy shouldRetry $ \_ -> streamComplete p cr k
  337   where
  338     shouldRetry _ (Left err) = pure (isTransient err)
  339     shouldRetry _ (Right _)  = pure False

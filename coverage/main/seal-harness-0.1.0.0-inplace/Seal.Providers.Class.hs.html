   49 -- | A single block of content in a message.
   50 --
   51 -- __Partial field selectors warning:__ The record fields 'cbId', 'cbName',
   52 -- 'cbInput', 'cbForId', 'cbParts', and 'cbIsError' are only defined on their
   53 -- respective constructors ('CbToolUse' and 'CbToolResult'). They are partial
   54 -- across the full type and will throw a runtime error if applied to the wrong
   55 -- constructor. Always access them inside a constructor-specific pattern match;
   56 -- never use them as bare accessor functions.
   57 --
   58 -- JSON keys (via 'stripPrefixToJSON'): @cbId@→@id@, @cbName@→@name@,
   59 -- @cbInput@→@input@, @cbForId@→@forId@, @cbParts@→@parts@,
   60 -- @cbIsError@→@isError@. The constructor tag (@CbText@/@CbToolUse@/
   61 -- @CbToolResult@) is emitted by aeson's default 'TaggedObject' sum encoding
   62 -- unchanged.
   63 data ContentBlock
   64   = CbText Text
   65   | CbToolUse    { cbId :: ToolCallId, cbName :: OpName, cbInput :: Value }
   66   | CbToolResult { cbForId :: ToolCallId, cbParts :: [ToolResultPart], cbIsError :: Bool }
   67   deriving stock (Eq, Show, Generic)
   68 instance ToJSON ContentBlock where toJSON = stripPrefixToJSON
   69 instance FromJSON ContentBlock where parseJSON = stripPrefixParseJSON
   70

   84 -- | Custom 'ToJSON': emits @name@ + @description@ always, and
   85 -- @input_schema@ only when it differs from 'stubSchema'. The keys follow the
   86 -- Anthropic wire shape (the @tools@ array is an external-facing payload, so
   87 -- the external convention applies — not the internal strip-prefix camelCase).
   88 -- This keeps the on-disk transcript envelope (@edTools@) and the debug
   89 -- @requests.jsonl@ consistent with the provider encoders (which also use
   90 -- @input_schema@ / @function.parameters@). The stub is omitted entirely so
   91 -- no @input_schema@ token cost leaks in on-demand mode.
   92 instance ToJSON ToolDefinition where
   93   toJSON (ToolDefinition n d sch) =
   94     if sch == stubSchema
   95       then object ["name" .= n, "description" .= d]
   96       else object ["name" .= n, "description" .= d, "input_schema" .= sch]
   97 
   98 -- | Custom 'FromJSON': @name@ + @description@ required; @input_schema@
   99 -- OPTIONAL, defaulting to 'stubSchema' when the key is ABSENT. The 'ToJSON'
  100 -- instance omits @input_schema@ for stub tools, so a required-field reader
  101 -- would reject those rows and — worse — take down the whole 'EnvelopeDelta'
  102 -- parse (aeson's 'parseJSON' is sequential), losing the system prompt from
  103 -- the reconstructed envelope. Key PRESENCE (not value null-ness)
  104 -- distinguishes "stored stub, omitted by ToJSON" from "explicitly null": an
  105 -- absent key defaults to 'stubSchema' (closing the stub round-trip), while a
  106 -- present 'null' stays 'Null' (so the arbitrary-data round-trip property
  107 -- still holds).
  108 instance FromJSON ToolDefinition where
  109   parseJSON = withObject "ToolDefinition" $ \o -> ToolDefinition
  110     <$> o .: "name"
  111     <*> o .: "description"
  112     <*> pure (fromMaybe stubSchema (KeyMap.lookup (fromString "input_schema") o))
  113 
  114 -- | The minimal placeholder @input_schema@ emitted when on-demand schema
  115 -- loading is enabled. Both Anthropic and Ollama normally require an
  116 -- @input_schema@ / @parameters@ field on every tool definition; when a
  117 -- 'ToolDefinition' carries this exact value, the provider encoders OMIT
  118 -- the field entirely rather than sending it inline — saving the few tokens
  119 -- the stub would otherwise cost on every tool, every turn. The model is
  120 -- expected to call @OPCODE_DESCRIBE@ to retrieve a tool's real schema
  121 -- before calling it.
  122 stubSchema :: Value
  123 stubSchema = object ["type" .= ("object" :: Text)]

  156 class Provider p where
  157   complete   :: p -> CompletionRequest -> IO (Either Text CompletionResponse)
  158   listModels :: p -> IO (Either Text [ModelId])
  159   -- | Stream a completion request, invoking the callback for each 'StreamEvent'
  160   -- as it arrives. The callback returns 'True' to continue streaming or 'False'
  161   -- to abort early (e.g. the consumer is no longer interested). The final
  162   -- 'StreamOutcome' carries the aggregated stop reason + usage. Providers
  163   -- that do not support streaming can fall back to 'complete' + emit a single
  164   -- 'StreamContentChunk' + 'StreamDone' (the default implementation does this).
  165   streamComplete
  166     :: p -> CompletionRequest -> (StreamEvent -> IO Bool) -> IO (Either Text StreamOutcome)
  167   streamComplete p cr k = do
  168     -- Default: use the non-streaming 'complete' and emit events for each
  169     -- content block (text + tool calls), then a final 'StreamDone'.
  170     eresp <- complete p cr
  171     case eresp of
  172       Left err -> pure (Left err)
  173       Right resp -> do
  174         mapM_ (emitBlock k) (rsContent resp)
  175         _ <- k (StreamDone (rsStop resp) (rsUsage resp))
  176         pure (Right (StreamOutcome (rsStop resp) (rsUsage resp)))
  177 
  178 -- | Emit 'StreamEvent's for a single 'ContentBlock' (used by the default
  179 -- 'streamComplete' implementation that wraps the non-streaming 'complete').
  180 emitBlock :: (StreamEvent -> IO Bool) -> ContentBlock -> IO ()
  181 emitBlock k block = case block of
  182   CbText t -> unless (T.null t) $
  183     void (k (StreamTextChunk t))
  184   CbToolUse tcid (OpName name) args ->
  185     void (k (StreamToolStart tcid (OpName name)) >>
  186           k (StreamToolEnd tcid (OpName name) args))
  187   CbToolResult{} -> pure ()  -- tool results are not model output
  188 
  189 data SomeProvider = forall p. Provider p => SomeProvider p
  190 
  191 -- | A single streaming event emitted by 'streamComplete'. Mirrors the
  192 -- 'LLMEvent' contract from OpenCode: text deltas, tool-call lifecycle
  193 -- (start/delta/end), and a final done signal carrying the stop reason +
  194 -- usage. The events arrive incrementally as the model generates; the
  195 -- consumer aggregates them into 'ContentBlock's and writes to the
  196 -- transcript live.
  197 data StreamEvent
  198   = StreamTextChunk Text
  199   -- ^ A chunk of assistant text. Multiple chunks concatenate to form the
  200   -- final 'CbText' block(s).
  201   | StreamToolStart ToolCallId OpName
  202   -- ^ The model began emitting a tool call. The id is synthesized by the
  203   -- provider (Ollama carries no ids; the provider assigns sequential ones).
  204   | StreamToolDelta ToolCallId Text
  205   -- ^ A chunk of the tool-call's JSON arguments. Multiple deltas
  206   -- concatenate to form the final 'cbInput' 'Value' (parsed at 'StreamToolEnd').
  207   | StreamToolEnd ToolCallId OpName Value
  208   -- ^ A tool call completed. Carries the fully-parsed 'cbInput'.
  209   | StreamDone StopReason Usage
  210   -- ^ The generation finished. Carries the final stop reason + usage. This
  211   -- is always the last event (unless the stream was aborted by the callback
  212   -- returning 'False' or a transport error).
  213   deriving stock (Eq, Show)
  214 
  215 -- | The aggregated outcome of a completed stream: the final stop reason +
  216 -- usage. The loop uses this (alongside the accumulated 'ContentBlock's) to
  217 -- decide whether to continue (tool calls / truncation) or finish.
  218 data StreamOutcome = StreamOutcome
  219   { soStop :: StopReason
  220   , soUsage :: Usage
  221   } deriving stock (Eq, Show)

  223 -- | Retry policy for transient provider failures during streaming (same as
  224 -- the non-streaming path: up to 2 retries, 50ms exponential backoff).
  225 streamRetryPolicy :: RetryPolicyM IO
  226 streamRetryPolicy =
  227   exponentialBackoff 50_000 <> limitRetries 2
  228 
  229 -- | Call 'streamComplete' with retry on transient errors. Mirrors
  230 -- 'providerCompleteWithRetry' in the non-streaming path. The transient-error
  231 -- predicate is the same ('isTransientError' lives in the loop; here we
  232 -- accept a predicate so the streaming provider can reuse it).
  233 providerStreamWithRetry
  234   :: SomeProvider -> (Text -> Bool) -> CompletionRequest
  235   -> (StreamEvent -> IO Bool) -> IO (Either Text StreamOutcome)
  236 providerStreamWithRetry (SomeProvider p) isTransient cr k =
  237   retrying streamRetryPolicy shouldRetry $ \_ -> streamComplete p cr k
  238   where
  239     shouldRetry _ (Left err) = pure (isTransient err)
  240     shouldRetry _ (Right _)  = pure False

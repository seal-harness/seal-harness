    1 -- | The @conversation.jsonl@ model: one raw 'Message' per line, in order — the
    2 -- pure content list of the session. Reading is @mapMaybe decode . lines@ →
    3 -- @[Message]@, fed straight to the provider with zero transformation. Appending
    4 -- a turn appends only the *new* message lines, so the file grows by deltas
    5 -- rather than re-serializing the whole prior conversation each turn (the old
    6 -- @transcript.jsonl@ format's O(N²) growth). No timestamps or ids live here —
    7 -- those are derived from @entries.jsonl@ ('Seal.Transcript.Entries').
    8 module Seal.Transcript.Conv
    9   ( ConvLine (..)
   10   , encodeConvLine
   11   , readConversation
   12   , appendMessages
   13   , diffNew
   14   ) where
   15 
   16 import Data.Aeson (decode, encode)
   17 import Data.ByteString (ByteString)
   18 import Data.ByteString qualified as BS
   19 import Data.ByteString.Lazy qualified as BL
   20 import Data.Maybe (fromMaybe, mapMaybe)
   21 
   22 import Seal.Providers.Class (Message (..))
   23 
   24 -- | A single line of @conversation.jsonl@: a raw 'Message' (role + content
   25 -- blocks). The on-disk form is the canonical aeson encoding of 'Message' with
   26 -- no trailing newline; the writer appends the newline.
   27 newtype ConvLine = ConvLine { clMessage :: Message }
   28   deriving stock (Eq, Show)
   29 
   30 -- | Canonical strict encoding of one line, no trailing newline.
   31 encodeConvLine :: ConvLine -> ByteString
   32 encodeConvLine (ConvLine m) = BL.toStrict (encode m)
   33 
   34 -- | Read @conversation.jsonl@ into the message list. Malformed lines are
   35 -- skipped (a torn tail leaves at most one bad line, already tolerated here).
   36 readConversation :: ByteString -> [Message]
   37 readConversation bs = mapMaybe decodeLine (splitLines bs)
   38   where
   39     decodeLine l = decode (BL.fromStrict l)
   40 
   41 -- | Split the file body into its non-empty component lines. A trailing newline
   42 -- produces a final empty chunk that we drop; interior empty lines (which a
   43 -- well-formed conversation file never has) are also dropped.
   44 splitLines :: ByteString -> [ByteString]
   45 splitLines = filter (not . BS.null) . BS.split 0x0a
   46 
   47 -- | The minimal append: the messages from @incoming@ that are not already
   48 -- present in @written@, taken as a suffix of the written list. The writer
   49 -- calls this to find what to append, then writes one line per result.
   50 --
   51 -- Invariant (caller responsibility): @incoming@ is the full message list for
   52 -- the turn (prior messages + new ones); @written@ is the conversation as it
   53 -- exists on disk. The new messages are the suffix of @incoming@ beyond the
   54 -- length of @written@, after verifying @written@ is a prefix of @incoming@.
   55 -- If @written@ is not a prefix of @incoming@ (a divergence — should not happen
   56 -- under the single-writer daemon), the whole @incoming@ is returned (a safe,
   57 -- if redundant, fallback).
   58 diffNew :: [Message] -> [Message] -> [Message]
   59 diffNew incoming written = fromMaybe incoming (stripPrefixMessages written incoming)
   60 
   61 -- | Append @newLines@ to @written@, returning the combined list. Kept as a
   62 -- separate function so the writer's "diff then append" flow is one named step.
   63 appendMessages :: [Message] -> [Message] -> [Message]
   64 appendMessages written newLines = written <> newLines
   65 
   66 -- | Like 'Data.List.stripPrefix' but compares 'Message's for equality. Returns
   67 -- @Just remainder@ when @written@ is a leading prefix of @incoming@ (so the
   68 -- remainder is the new suffix to append), or 'Nothing' on divergence.
   69 stripPrefixMessages :: [Message] -> [Message] -> Maybe [Message]
   70 stripPrefixMessages [] incoming              = Just incoming
   71 stripPrefixMessages _   []                   = Nothing
   72 stripPrefixMessages (w:ws) (i:is)
   73   | w == i    = stripPrefixMessages ws is
   74   | otherwise = Nothing

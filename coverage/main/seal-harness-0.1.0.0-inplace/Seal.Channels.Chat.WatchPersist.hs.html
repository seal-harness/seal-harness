    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Watch-state persistence — the per-conversation watch-all-tabs map
    3 -- survives a @seal serve@ restart. Written atomically (0600) to
    4 -- @\<state\>\/watch_state.json@ on every watch-mode mutation via
    5 -- 'Seal.Util.AtomicJson.saveJsonAtomic'; loaded at boot by 'loadWatchMap'
    6 -- so a conversation's watch mode is restored instead of reset to off.
    7 --
    8 -- The map is encoded as a JSON array of
    9 -- @{"key": {"channel": "...", "conv": "..."}, "enabled": true}@ pairs.
   10 -- 'ConversationKey' is a record, not a JSON-string-keyable type, so the
   11 -- map is serialized as a list of wire objects (mirroring
   12 -- 'Seal.Channels.Cursor.Persist.CursorWire').
   13 module Seal.Channels.Chat.WatchPersist
   14   ( saveWatchMap
   15   , loadWatchMap
   16   ) where
   17 
   18 import Data.Aeson qualified as A
   19 import Data.Aeson.Key qualified as Key
   20 import Data.Aeson.KeyMap qualified as KeyMap
   21 import Data.Map.Strict (Map)
   22 import Data.Map.Strict qualified as Map
   23 import System.Directory (doesFileExist)
   24 
   25 import Katip (Severity (..))
   26 import Seal.Channels.Chat.Types (ConversationKey (..))
   27 import Seal.Logging.Global (globalLogIO)
   28 import Seal.Util.AtomicJson (saveJsonAtomic)
   29 import Seal.Util.StrictIO (decodeFileStrict)
   30 
   31 -- | The on-disk wire type: a key/enabled pair. The map is converted to/from
   32 -- this list on save/load.
   33 data WatchWire = WatchWire
   34   { wwKey     :: ConversationKey
   35   , wwEnabled :: Bool
   36   }
   37 
   38 instance A.ToJSON WatchWire where
   39   toJSON w = A.object
   40     [ "key"     A..= ckToJSON (wwKey w)
   41     , "enabled" A..= wwEnabled w
   42     ]
   43     where
   44       ckToJSON (ConversationKey ch cv) = A.object
   45         [ "channel" A..= ch
   46         , "conv"    A..= cv
   47         ]
   48 
   49 instance A.FromJSON WatchWire where
   50   parseJSON = A.withObject "WatchWire" $ \o -> do
   51     keyVal <- o A..: "key"
   52     enabled <- o A..: "enabled"
   53     case keyVal of
   54       A.Object ko ->
   55         case (asText (KeyMap.lookup (Key.fromText "channel") ko),
   56               asText (KeyMap.lookup (Key.fromText "conv") ko)) of
   57           (Just ch, Just cv) -> pure (WatchWire (ConversationKey ch cv) enabled)
   58           _ -> fail "missing channel or conv field"
   59       _ -> fail "key is not an object"
   60     where
   61       asText (Just (A.String t)) = Just t
   62       asText _ = Nothing
   63 
   64 -- | Save a watch-state map to @path@ atomically (0600, MVar-serialized via
   65 -- 'saveJsonAtomic'). Never throws; a write failure is the caller's
   66 -- responsibility to 'catch' (the persisting watch state does so and logs
   67 -- a warning, matching the cursor-store pattern).
   68 saveWatchMap :: FilePath -> Map ConversationKey Bool -> IO ()
   69 saveWatchMap path m =
   70   saveJsonAtomic path (A.encode (map (uncurry WatchWire) (Map.toList m)))
   71 
   72 -- | Load the watch-state map from @path@. Missing file -> 'Nothing' (fresh
   73 -- empty store). Corrupt JSON -> 'Nothing' + a stderr warning (no
   74 -- conversation content in the file — just channel/conversation ids + a
   75 -- boolean). Entries with a missing or malformed key are dropped
   76 -- (defense-in-depth against a tampered file), mirroring
   77 -- 'Seal.Channels.Cursor.Persist.filterValid'.
   78 loadWatchMap :: FilePath -> IO (Maybe (Map ConversationKey Bool))
   79 loadWatchMap path = do
   80   exists <- doesFileExist path
   81   if not exists
   82     then pure Nothing
   83     else do
   84       mWires <- decodeFileStrict path :: IO (Maybe [WatchWire])
   85       case mWires of
   86         Nothing -> do
   87           globalLogIO WarningS "could not parse watch_state.json; using empty watch state"
   88           pure Nothing
   89         Just ws ->
   90           pure (Just (Map.fromList [ (wwKey w, wwEnabled w) | w <- ws ]))

   30 data CronAction
   31   = CaCreate | CaRead | CaList | CaUpdate | CaDelete | CaTrigger | CaEnable | CaDisable
   32 
   33 parseCronAction :: Value -> Either Text CronAction
   34 parseCronAction v =
   35   case parseMaybe (withObject "in" (.: "action")) v of
   36     Just t -> case t :: Text of
   37       "create"  -> Right CaCreate
   38       "read"    -> Right CaRead
   39       "list"    -> Right CaList
   40       "update"  -> Right CaUpdate
   41       "delete"  -> Right CaDelete
   42       "trigger" -> Right CaTrigger
   43       "enable"  -> Right CaEnable
   44       "disable" -> Right CaDisable
   45       other     -> Left ("unknown cron action: " <> other)
   46     Nothing -> Left "missing or invalid action field"
   47 
   48 idField :: Value -> Maybe Text
   49 idField = parseMaybe (withObject "in" (.: "id"))
   50 
   51 nameField :: Value -> Maybe Text
   52 nameField = parseMaybe (withObject "in" (.: "name"))
   53 
   54 scheduleField :: Value -> Maybe Text
   55 scheduleField = parseMaybe (withObject "in" (.: "schedule"))
   56 
   57 timezoneField :: Value -> Maybe Text
   58 timezoneField v = joinMaybe (parseMaybe (withObject "in" (.:? "timezone")) v)
   59 
   60 promptField :: Value -> Maybe Text
   61 promptField = parseMaybe (withObject "in" (.: "prompt"))
   62 
   63 agentDefIdField :: Value -> Maybe Text
   64 agentDefIdField v = joinMaybe (parseMaybe (withObject "in" (.:? "agent_def_id")) v)
   65 
   66 providerField :: Value -> Maybe Text
   67 providerField v = joinMaybe (parseMaybe (withObject "in" (.:? "provider")) v)
   68 
   69 modelField :: Value -> Maybe Text
   70 modelField v = joinMaybe (parseMaybe (withObject "in" (.:? "model")) v)
   71 
   72 enabledField :: Value -> Bool
   73 enabledField v = fromMaybe False (joinMaybe (parseMaybe (withObject "in" (.:? "enabled")) v))
   74 
   75 oneShotField :: Value -> Bool
   76 oneShotField v = fromMaybe False (joinMaybe (parseMaybe (withObject "in" (.:? "one_shot")) v))
   77 
   78 enabledOnlyField :: Value -> Bool
   79 enabledOnlyField v = fromMaybe False (joinMaybe (parseMaybe (withObject "in" (.:? "enabled_only")) v))
   80 
   81 enabledField' :: Value -> Maybe Bool
   82 enabledField' v = joinMaybe (parseMaybe (withObject "in" (.:? "enabled")) v)
   83 
   84 oneShotField' :: Value -> Maybe Bool
   85 oneShotField' v = joinMaybe (parseMaybe (withObject "in" (.:? "one_shot")) v)

   87 joinMaybe :: Maybe (Maybe a) -> Maybe a

   90 handleCreate :: CronStore -> Value -> App OpResult
   91 handleCreate store v = do
   92   case (nameField v, scheduleField v, promptField v) of
   93     (Nothing, _, _) -> pure (errResult "create requires {name:string}")
   94     (_, Nothing, _) -> pure (errResult "create requires {schedule:string}")
   95     (_, _, Nothing) -> pure (errResult "create requires {prompt:string}")
   96     (Just name, Just sched, Just prompt) -> do
   97       let mTzStr = timezoneField v
   98           mAgentDefId = agentDefIdField v
   99           enabled = enabledField v
  100           oneShot = oneShotField v
  101           mProv = providerField v
  102           mModel = modelField v
  103       eJob <- liftIO (createJob store name sched mTzStr prompt mAgentDefId mProv mModel enabled oneShot)
  104       case eJob of
  105         Left e -> pure (errResult e)
  106         Right job -> do
  107           now <- liftIO getCurrentTime
  108           mNextFire <- liftIO (computeNextFire (cjSchedule job) (cjTimezone job) now)
  109           let slug = cronJobIdText (cjId job)
  110           pure (OpResult
  111             [ TrpText ("created job '" <> name <> "' (id: " <> slug
  112                        <> "), next fire: " <> tshowUTC mNextFire) ]
  113             False
  114             (object [ "action" .= ("create" :: Text), "id" .= slug
  115                     , "name" .= name, "schedule" .= sched
  116                     , "enabled" .= enabled, "one_shot" .= oneShot
  117                     , "next_fire" .= mNextFire ]))
  118 
  119 handleRead :: CronStore -> Value -> App OpResult
  120 handleRead store v = do
  121   case idField v >>= either (const Nothing) Just . mkCronJobId of
  122     Nothing -> pure (errResult "read requires {id:string}")
  123     Just jid -> do
  124       mJob <- liftIO (csRead store jid)
  125       case mJob of
  126         Nothing -> pure (OpResult
  127           [ TrpText ("cron job not found: " <> cronJobIdText jid) ] True
  128           (object [ "action" .= ("read" :: Text), "id" .= cronJobIdText jid, "found" .= False ]))
  129         Just job -> pure (OpResult
  130           [ TrpText (renderJob job) ] False
  131           (object [ "action" .= ("read" :: Text), "id" .= cronJobIdText jid, "found" .= True
  132                   , "name" .= cjName job, "schedule" .= cronExprText (cjSchedule job)
  133                   , "enabled" .= cjEnabled job, "one_shot" .= cjOneShot job ]))
  134 
  135 handleList :: CronStore -> Value -> App OpResult
  136 handleList store v = do
  137   let enabledOnly = enabledOnlyField v
  138   jobs <- liftIO (csList store enabledOnly)
  139   let rendered = case jobs of
  140         [] -> "(no cron jobs defined)"
  141         _ -> T.intercalate "\n"
  142                [ cronJobIdText (cjId j)
  143                  <> " [" <> (if cjEnabled j then "enabled" else "disabled" :: Text) <> "]: "
  144                  <> cjName j <> " — " <> cronExprText (cjSchedule j)
  145                | j <- jobs ]
  146   pure (OpResult [TrpText rendered] False
  147          (object [ "action" .= ("list" :: Text), "count" .= length jobs
  148                  , "ids" .= map (cronJobIdText . cjId) jobs
  149                  , "enabled_count" .= length (filter cjEnabled jobs) ]))
  150 
  151 handleUpdate :: CronStore -> Value -> App OpResult
  152 handleUpdate store v = do
  153   case idField v >>= either (const Nothing) Just . mkCronJobId of
  154     Nothing -> pure (errResult "update requires {id:string}")
  155     Just jid -> do
  156       eUpdated <- liftIO (updateJob store jid (CronUpdate
  157         { cuName = nameField v
  158         , cuSchedule = scheduleField v
  159         , cuTimezone = timezoneField v
  160         , cuPrompt = promptField v
  161         , cuAgentDefId = agentDefIdField v
  162         , cuProvider = providerField v
  163         , cuModel = modelField v
  164         , cuEnabled = enabledField' v
  165         , cuOneShot = oneShotField' v
  166         }))
  167       case eUpdated of
  168         Left e -> pure (errResult e)
  169         Right updatedJob -> do
  170           now <- liftIO getCurrentTime
  171           mNextFire <- liftIO (computeNextFire (cjSchedule updatedJob) (cjTimezone updatedJob) now)
  172           pure (OpResult
  173             [ TrpText ("updated job '" <> cronJobIdText jid <> "'") ] False
  174             (object [ "action" .= ("update" :: Text), "id" .= cronJobIdText jid
  175                     , "next_fire" .= mNextFire ]))
  176 
  177 handleDelete :: CronStore -> Value -> App OpResult
  178 handleDelete store v = do
  179   case idField v >>= either (const Nothing) Just . mkCronJobId of
  180     Nothing -> pure (errResult "delete requires {id:string}")
  181     Just jid -> do
  182       mJob <- liftIO (csRead store jid)
  183       _ <- liftIO (deleteJob store jid)
  184       let existed = isJust mJob
  185       pure (OpResult
  186         [ TrpText (if existed then "deleted" else "deleted (was not present)") ] False
  187         (object [ "action" .= ("delete" :: Text), "id" .= cronJobIdText jid, "existed" .= existed ]))
  188 
  189 handleTrigger :: Maybe CronFireDelegate -> CronStore -> Value -> App OpResult
  190 handleTrigger mFireDelegate store v = do
  191   case idField v >>= either (const Nothing) Just . mkCronJobId of
  192     Nothing -> pure (errResult "trigger requires {id:string}")
  193     Just jid -> do
  194       mJob <- liftIO (csRead store jid)
  195       case mJob of
  196         Nothing -> pure (OpResult
  197           [ TrpText ("cron job not found: " <> cronJobIdText jid) ] True
  198           (object [ "action" .= ("trigger" :: Text), "id" .= cronJobIdText jid, "found" .= False ]))
  199         Just job -> case mFireDelegate of
  200           Nothing -> pure (errResult "trigger not available (no fire delegate configured)")
  201           Just fire -> do
  202             eResult <- liftIO (fire job)
  203             case eResult of
  204               Left err -> pure (errResult ("trigger failed: " <> err))
  205               Right sid -> pure (OpResult
  206                 [ TrpText ("triggered job '" <> cronJobIdText jid
  207                            <> "', session: " <> sessionIdText sid) ] False
  208                 (object [ "action" .= ("trigger" :: Text), "id" .= cronJobIdText jid
  209                         , "session_id" .= sessionIdText sid ]))
  210 
  211 handleEnable :: CronStore -> Value -> App OpResult
  212 handleEnable store v = do
  213   case idField v >>= either (const Nothing) Just . mkCronJobId of
  214     Nothing -> pure (errResult "enable requires {id:string}")
  215     Just jid -> do
  216       eJob <- liftIO (setJobEnabled store jid True)
  217       case eJob of
  218         Left _ -> pure (OpResult
  219           [ TrpText ("cron job not found: " <> cronJobIdText jid) ] True
  220           (object [ "action" .= ("enable" :: Text), "id" .= cronJobIdText jid, "found" .= False ]))
  221         Right job -> do
  222           now <- liftIO getCurrentTime
  223           mNextFire <- liftIO (computeNextFire (cjSchedule job) (cjTimezone job) now)
  224           pure (OpResult
  225             [ TrpText ("enabled job '" <> cronJobIdText jid <> "', next fire: " <> tshowUTC mNextFire) ] False
  226             (object [ "action" .= ("enable" :: Text), "id" .= cronJobIdText jid, "next_fire" .= mNextFire ]))
  227 
  228 handleDisable :: CronStore -> Value -> App OpResult
  229 handleDisable store v = do
  230   case idField v >>= either (const Nothing) Just . mkCronJobId of
  231     Nothing -> pure (errResult "disable requires {id:string}")
  232     Just jid -> do
  233       eJob <- liftIO (setJobEnabled store jid False)
  234       case eJob of
  235         Left _ -> pure (OpResult
  236           [ TrpText ("cron job not found: " <> cronJobIdText jid) ] True
  237           (object [ "action" .= ("disable" :: Text), "id" .= cronJobIdText jid, "found" .= False ]))
  238         Right _job -> do
  239           pure (OpResult
  240             [ TrpText ("disabled job '" <> cronJobIdText jid <> "'") ] False
  241             (object [ "action" .= ("disable" :: Text), "id" .= cronJobIdText jid ]))

  243 authorizeCronManage :: Value -> Either Text ()
  244 authorizeCronManage v =
  245   case parseCronAction v of
  246     Left e -> Left e
  247     Right action -> case action of
  248       CaCreate -> do
  249         case (nameField v, scheduleField v, promptField v) of
  250           (Just name, Just sched, Just _prompt) -> do
  251             whenB (T.null name || T.length name > 100) (Left "name must be 1-100 characters")
  252             case mkCronExpr sched of
  253               Left e -> Left e
  254               Right _ -> case timezoneField v of
  255                 Just tzText -> const (Right ()) =<< mkCronTimeZone tzText
  256                 Nothing -> Right ()
  257           _ -> Left "create requires {name, schedule, prompt}"
  258       CaRead    -> checkId v "read"
  259       CaList    -> Right ()
  260       CaUpdate  -> do
  261         _ <- checkId v "update"
  262         let hasField = any isJust [ nameField v, scheduleField v, timezoneField v
  263                                   , promptField v, providerField v, modelField v ]
  264                       || isJust (enabledField' v) || isJust (oneShotField' v)
  265         if hasField then Right () else Left "update requires at least one field to update"
  266       CaDelete  -> checkId v "delete"
  267       CaTrigger -> checkId v "trigger"
  268       CaEnable  -> checkId v "enable"
  269       CaDisable -> checkId v "disable"
  270 
  271 checkId :: Value -> Text -> Either Text ()
  272 checkId v actionName =
  273   case idField v of
  274     Nothing -> Left (actionName <> " requires {id:string}")
  275     Just t -> const (Right ()) =<< mkCronJobId t

  277 cronManageOp :: Maybe CronFireDelegate -> CronStore -> SessionId -> Opcode
  278 cronManageOp mFireDelegate store _session = TrustedOpcode
  279   { toName = OpName "CRON_MANAGE"
  280   , toTrust = Audited
  281   , toDesc = "Manage cron jobs. Use action to select: create (new job), read (by id), list (all jobs), update (modify by id), delete (by id, idempotent), trigger (fire immediately), enable (enable job), disable (pause job)."
  282   , toInSchema = object
  283       [ "type" .= ("object" :: Text)
  284       , "properties" .= object
  285           [ fromText "action" .= object
  286               [ "type" .= ("string" :: Text)
  287               , "enum" .= (["create","read","list","update","delete","trigger","enable","disable"] :: [Text])
  288               , "description" .= ("Operation to perform." :: Text) ]
  289           , fromText "id" .= object
  290               [ "type" .= ("string" :: Text)
  291               , "description" .= ("Cron job id ([a-z0-9-]+) (read, update, delete, trigger, enable, disable)." :: Text) ]
  292           , fromText "name" .= object
  293               [ "type" .= ("string" :: Text)
  294               , "description" .= ("Human-readable job name, max 100 chars (create, update)." :: Text) ]
  295           , fromText "schedule" .= object
  296               [ "type" .= ("string" :: Text)
  297               , "description" .= ("5-field cron expression, e.g. \"0 9 * * 1-5\" (create, update)." :: Text) ]
  298           , fromText "timezone" .= object
  299               [ "type" .= ("string" :: Text)
  300               , "description" .= ("IANA timezone, e.g. \"America/New_York\" (create, update)." :: Text) ]
  301           , fromText "prompt" .= object
  302               [ "type" .= ("string" :: Text)
  303               , "description" .= ("Prompt to run on each fire, max 4000 chars (create, update)." :: Text) ]
  304           , fromText "agent_def_id" .= object
  305               [ "type" .= ("string" :: Text)
  306               , "description" .= ("Agent def id to use (create, update). Default: default agent." :: Text) ]
  307           , fromText "provider" .= object
  308               [ "type" .= ("string" :: Text)
  309               , "description" .= ("Provider override for this job (create, update). Omit or send \"\" to use Seal's configured default provider." :: Text) ]
  310           , fromText "model" .= object
  311               [ "type" .= ("string" :: Text)
  312               , "description" .= ("Model override for this job (create, update). Omit or send \"\" to use Seal's configured default model." :: Text) ]
  313           , fromText "enabled" .= object
  314               [ "type" .= ("boolean" :: Text)
  315               , "description" .= ("Initial enabled state (create, update). Default: false on create." :: Text) ]
  316           , fromText "one_shot" .= object
  317               [ "type" .= ("boolean" :: Text)
  318               , "description" .= ("If true, auto-disable after first run (create, update). Default: false." :: Text) ]
  319           , fromText "enabled_only" .= object
  320               [ "type" .= ("boolean" :: Text)
  321               , "description" .= ("List only enabled jobs (list). Default: false." :: Text) ]
  322           ]
  323       , "required" .= (["action"] :: [Text])
  324       ]
  325   , toOutSchema = object []
  326   , toAuthorize = authorizeCronManage
  327   , toBlocking = False
  328   , toRun = \_ v ->
  329       case parseCronAction v of
  330         Left e -> pure (errResult e)
  331         Right action -> case action of
  332           CaCreate  -> handleCreate store v
  333           CaRead    -> handleRead store v
  334           CaList    -> handleList store v
  335           CaUpdate  -> handleUpdate store v
  336           CaDelete  -> handleDelete store v
  337           CaTrigger -> handleTrigger mFireDelegate store v
  338           CaEnable  -> handleEnable store v
  339           CaDisable -> handleDisable store v
  340   }

  342 errResult :: Text -> OpResult
  343 errResult msg = OpResult [TrpText msg] True (object [])

  345 renderJob :: CronJob -> Text
  346 renderJob j = T.intercalate "\n"
  347   [ "id: " <> cronJobIdText (cjId j)
  348   , "name: " <> cjName j
  349   , "schedule: " <> cronExprText (cjSchedule j)
  350   , "timezone: " <> maybe "(host local)" cronTimeZoneText (cjTimezone j)
  351   , "provider: " <> fromMaybe "(configured default)" (cjProvider j)
  352   , "model: " <> fromMaybe "(configured default)" (cjModel j)
  353   , "prompt: " <> cjPrompt j
  354   , "enabled: " <> (if cjEnabled j then "true" else "false" :: Text)
  355   , "one_shot: " <> (if cjOneShot j then "true" else "false" :: Text)
  356   , "run_count: " <> T.pack (show (cjRunCount j))
  357   , "consecutive_failures: " <> T.pack (show (cjConsecutiveFailures j))
  358   ]
  359 
  360 tshowUTC :: Maybe UTCTime -> Text
  361 tshowUTC = maybe "unknown" (T.pack . show)

  363 whenB :: Bool -> Either Text () -> Either Text ()
  364 whenB True e = e

    2 -- | Session search capability — a record-of-functions handle (per
    3 -- CONTRIBUTING.md: "capability-handle records of IO functions over type
    4 -- classes") providing full-transcript search across all sessions.
    5 --
    6 -- Three search tiers, tried in order by 'resolveSessionSearchBackend':
    7 --
    8 -- 1. **Engram** ('engramSessionSearchBackend') — semantic search via the
    9 --    engram CLI subprocess. Uses the 'EmbeddingBackend' resolved from
   10 --    config. Falls back to in-memory scan when engram is unavailable
   11 --    or returns no results.
   12 -- 2. **Ripgrep** ('ripgrepSessionSearchBackend') — literal substring
   13 --    search via the @rg@ CLI subprocess with fixed argv (no shell).
   14 --    Fast for literal queries across many session transcripts.
   15 -- 3. **In-memory** ('inMemorySessionSearchBackend') — pure Haskell
   16 --    full-transcript scan. No external dependencies. Scans all messages
   17 --    and all content blocks (text, thinking, tool calls, tool results)
   18 --    in every session's @conversation.jsonl@.
   19 --
   20 -- The previous implementation only searched the session description and
   21 -- the first 120 characters of the first user message. This module fixes
   22 -- that by searching the full transcript in every tier.
   23 module Seal.Session.Search
   24   ( SessionSearchBackend (..)
   25   , inMemorySessionSearchBackend
   26   , ripgrepSessionSearchBackend
   27   , engramSessionSearchBackend
   28   , resolveSessionSearchBackend
   29   ) where

   58 -- | A session search result: the session metadata + a snippet showing
   59 -- the match context.
   60 type SessionSearchResult = (SessionMeta, Text)
   61 
   62 -- | The session search capability handle.
   63 data SessionSearchBackend = SessionSearchBackend
   64   { ssbSearch :: SealPaths -> Text -> Bool -> IO [SessionSearchResult]
   65     -- ^ Search sessions: query (case-insensitive substring), archived flag.
   66     -- Returns matched sessions with a context snippet.
   67   , ssbBackendName :: Text
   68     -- ^ Human-readable backend name (e.g. "in-memory", "ripgrep", "engram").
   69   }
   70 
   71 -- ---------------------------------------------------------------------------
   72 -- In-memory search (fallback — no external dependencies)
   73 -- ---------------------------------------------------------------------------
   74 
   75 -- | In-memory full-transcript search. Reads each session's
   76 -- @conversation.jsonl@, decodes all messages, and checks every content
   77 -- block (text, thinking, tool calls, tool results) for the query.
   78 -- Also checks the session description.
   79 inMemorySessionSearchBackend :: SessionSearchBackend
   80 inMemorySessionSearchBackend = SessionSearchBackend
   81   { ssbSearch = inMemorySearch
   82   , ssbBackendName = "in-memory"
   83   }
   84 
   85 inMemorySearch :: SealPaths -> Text -> Bool -> IO [SessionSearchResult]
   86 inMemorySearch paths query archived = do
   87   let qLower = T.toCaseFold (T.strip query)
   88   if T.null qLower
   89     then pure []
   90     else do
   91       metas <- if archived then listArchivedSessions paths else listSessions paths
   92       results <- mapM (searchSessionFull paths qLower) metas
   93       pure [ (m, snip) | (m, Just snip) <- results ]
   94 
   95 -- | Search a single session's full transcript for the query. Returns
   96 -- 'Just snippet' if any content block matches, 'Nothing' otherwise.
   97 searchSessionFull :: SealPaths -> Text -> SessionMeta -> IO (SessionMeta, Maybe Text)
   98 searchSessionFull paths qLower meta = do
   99   let descLower = maybe "" T.toCaseFold (smDescription meta)
  100       descMatch = qLower `T.isInfixOf` descLower
  101   mSnippet <- fullTranscriptSnippet paths (smId meta) qLower
  102   if descMatch
  103     then pure (meta, Just (fromMaybe (fromMaybe "(no description)" (smDescription meta)) mSnippet))
  104     else case mSnippet of
  105       Just snip -> pure (meta, Just snip)
  106       Nothing   -> pure (meta, Nothing)
  107 
  108 -- | Read a session's full conversation and find the first content block
  109 -- that contains the query (case-insensitive). Returns a truncated snippet
  110 -- of the matching text.
  111 fullTranscriptSnippet :: SealPaths -> SessionId -> Text -> IO (Maybe Text)
  112 fullTranscriptSnippet paths sid qLower = do
  113   let convPath = sessionConversationPath paths sid
  114   exists <- doesFileExist convPath
  115   if not exists
  116     then pure Nothing
  117     else do
  118       raw <- readFileTextStrict convPath
  119       let msgs = mapMaybe decodeMsg (T.lines raw)
  120           rendered = concatMap renderMsgForSearch msgs
  121       pure (firstMatchSnippet qLower rendered)
  122 
  123 -- | Render a message into searchable text blocks (role label + each
  124 -- content block rendered as text).
  125 renderMsgForSearch :: Message -> [Text]
  126 renderMsgForSearch msg =
  127   let roleStr = case msgRole msg of
  128         User      -> "User"
  129         Assistant -> "Assistant"
  130       blockText (CbText t)          = [t]
  131       blockText (CbThinking t)      = ["[thinking] " <> t]
  132       blockText (CbToolUse{cbName = OpName n}) = ["[tool call: " <> n <> "]"]
  133       blockText (CbToolResult{cbParts = parts, cbIsError = isErr}) =
  134         let label = if isErr then "[tool error]" else "[tool result]"
  135             content = T.intercalate "\n" [t | TrpText t <- parts]
  136         in [if T.null content then label else label <> " " <> content]
  137   in [roleStr] <> concatMap blockText (msgContent msg)
  138 
  139 -- | Find the first text block that contains the query (case-insensitive)
  140 -- and return a truncated snippet of the matching text.
  141 firstMatchSnippet :: Text -> [Text] -> Maybe Text
  142 firstMatchSnippet qLower blocks =
  143   case [b | b <- blocks, qLower `T.isInfixOf` T.toCaseFold b] of
  144     (b : _) -> Just (truncateSnippet 200 b)
  145     []      -> Nothing
  146 
  147 -- | Truncate a snippet to at most @n@ characters, appending an ellipsis.
  148 truncateSnippet :: Int -> Text -> Text
  149 truncateSnippet n t
  150   | T.length t <= n = t
  151   | otherwise       = T.take n t <> "\x2026"
  152 
  153 -- | Decode a JSON line into a Message.
  154 decodeMsg :: Text -> Maybe Message
  155 decodeMsg line = A.decode (BL.fromStrict (TE.encodeUtf8 line))
  156 
  157 -- ---------------------------------------------------------------------------
  158 -- Ripgrep search (literal subprocess — fast for many sessions)
  159 -- ---------------------------------------------------------------------------
  160 
  161 -- | Ripgrep-based session search. Uses @rg -i -l -- <query> <root>@ to
  162 -- find matching @conversation.jsonl@ files, then loads metadata for
  163 -- matching sessions and extracts a snippet via a second @rg@ pass.
  164 -- Also checks session descriptions (which are in @session.json@, not
  165 -- @conversation.jsonl@). Falls back to in-memory search if @rg@ fails.
  166 ripgrepSessionSearchBackend :: SessionSearchBackend
  167 ripgrepSessionSearchBackend = SessionSearchBackend
  168   { ssbSearch = ripgrepSearch
  169   , ssbBackendName = "ripgrep"
  170   }
  171 
  172 ripgrepSearch :: SealPaths -> Text -> Bool -> IO [SessionSearchResult]
  173 ripgrepSearch paths query archived = do
  174   let qStr = T.unpack (T.strip query)
  175       qLower = T.toCaseFold (T.strip query)
  176   if null qStr
  177     then pure []
  178     else do
  179       let root = sessionsRoot paths
  180       exists <- doesDirectoryExist root
  181       if not exists
  182         then pure []
  183         else do
  184           eFiles <- tryRg ["-i", "-l", "--", qStr, root]
  185           case eFiles of
  186             Left _ -> ssbSearch inMemorySessionSearchBackend paths query archived
  187             Right out -> do
  188               let filePaths = filter (not . T.null) (T.lines (TE.decodeUtf8With lenient out))
  189                   sessionIds = mapMaybe pathToSessionId filePaths
  190               allMetas <- if archived then listArchivedSessions paths else listSessions paths
  191               let matchedMetas = [ m | m <- allMetas, smId m `elem` sessionIds ]
  192                   descMetas = [ m | m <- allMetas
  193                               , qLower `T.isInfixOf` maybe "" T.toCaseFold (smDescription m)
  194                               , m `notElem` matchedMetas ]
  195                   allMatched = matchedMetas <> descMetas
  196               mapM (withSnippet paths qStr) allMatched
  197 
  198 -- | Get a snippet for a matched session. Tries ripgrep for a content
  199 -- snippet, falls back to the session description.
  200 withSnippet :: SealPaths -> String -> SessionMeta -> IO SessionSearchResult
  201 withSnippet paths qStr meta = do
  202   let convPath = sessionConversationPath paths (smId meta)
  203   exists <- doesFileExist convPath
  204   mSnip <- if not exists
  205     then pure Nothing
  206     else do
  207       eSnip <- tryRg ["-i", "-o", "--", qStr, convPath]
  208       case eSnip of
  209         Right out | not (BS.null out) ->
  210           pure (Just (truncateSnippet 200 (TE.decodeUtf8With lenient out)))
  211         _ -> pure Nothing
  212   let snip = fromMaybe (fromMaybe "(no description)" (smDescription meta)) mSnip
  213   pure (meta, snip)

  215 -- | Run @rg@ with fixed argv (no shell). Returns stdout on success.
  216 -- @rg@ exit code 1 (no matches) is treated as success with empty output.
  217 tryRg :: [String] -> IO (Either Text BS.ByteString)
  218 tryRg args = do
  219   let cp = (proc "rg" args)
  220         { std_in = CreatePipe, std_out = CreatePipe, std_err = CreatePipe }
  221   withCreateProcess cp $ \mIn mOut mErr ph -> do
  222     for_ mIn hClose
  223     case (mOut, mErr) of
  224       (Just out, Just _err) -> do
  225         outBs <- BS.hGetContents out
  226         let !_ = BS.length outBs
  227         ec <- waitForProcess ph
  228         case ec of
  229           ExitSuccess     -> pure (Right outBs)
  230           ExitFailure 1   -> pure (Right outBs)  -- rg returns 1 when no matches
  231           ExitFailure _   -> pure (Left "rg failed")
  232       _ -> do
  233         _ <- waitForProcess ph
  234         pure (Left "rg: missing handles")

  236 -- | Extract a SessionId from a conversation.jsonl file path.
  237 -- Paths look like @<root>/<session-id>/conversation.jsonl@.
  238 pathToSessionId :: Text -> Maybe SessionId
  239 pathToSessionId filePath =
  240   let dir = takeDirectory (T.unpack filePath)
  241       sidText = T.pack (takeBaseName dir)
  242   in either (const Nothing) Just (mkSessionId sidText)

  244 -- ---------------------------------------------------------------------------
  245 -- Engram search (semantic subprocess — requires engram + ollama)
  246 -- ---------------------------------------------------------------------------
  247 
  248 -- | Engram-based session search. When the embedding backend is non-null
  249 -- (i.e. engram is configured), this backend attempts semantic search.
  250 -- Since session transcripts may not be indexed in engram, it falls back
  251 -- to in-memory full-transcript scan to ensure all matches are found.
  252 -- The engram tier is a placeholder for future incremental indexing —
  253 -- once session transcripts are indexed on write, the semantic search
  254 -- path will return ranked results without scanning every file.
  255 engramSessionSearchBackend :: SessionSearchBackend
  256 engramSessionSearchBackend = SessionSearchBackend
  257   { ssbSearch = engramSearch
  258   , ssbBackendName = "engram"
  259   }
  260 
  261 engramSearch :: SealPaths -> Text -> Bool -> IO [SessionSearchResult]
  262 engramSearch paths query archived = do
  263   let qStr = T.strip query
  264   if T.null qStr
  265     then pure []
  266     else ssbSearch inMemorySessionSearchBackend paths query archived

  268 -- ---------------------------------------------------------------------------
  269 -- Resolution — pick the best available backend
  270 -- ---------------------------------------------------------------------------
  271 
  272 -- | Resolve the best available session search backend.
  273 --
  274 -- If the embedding backend's name is @"engram"@, the engram search backend
  275 -- is selected (it falls back to in-memory when engram results are empty).
  276 -- Otherwise, if @rg@ is available (checked via the @findBin@ action),
  277 -- the ripgrep backend is selected. Otherwise, the in-memory backend is
  278 -- used as the final fallback.
  279 resolveSessionSearchBackend
  280   :: EmbeddingBackend
  281   -> SealPaths
  282   -> (FilePath -> IO (Maybe FilePath))
  283   -- ^ A lookup action to find a binary on PATH (e.g. 'findExecutable').
  284   -> IO SessionSearchBackend
  285 resolveSessionSearchBackend embedding _paths findBin =
  286   if ebBackendName embedding == "engram"
  287     then pure engramSessionSearchBackend
  288     else do
  289       mRg <- findBin "rg"
  290       case mRg of
  291         Just _  -> pure ripgrepSessionSearchBackend
  292         Nothing -> pure inMemorySessionSearchBackend

   15 -- | The subcommand selected on the command line. 'CommandNoOp' is the
   16 -- harmless placeholder carried by 'defaultConfig'; it is not exposed as a
   17 -- subcommand (the subcommands are @tui@ and @serve@) and is excluded from
   18 -- the config-file 'FromJSON'/'ToJSON' instances.
   19 -- Each subcommand carries the autonomy level selected via @--yolo@
   20 -- ('Full' — bypass the human-confirmation gate) vs the default
   21 -- ('Supervised' — prompt before running any Untrusted opcode).
   22 data Command
   23   = CommandNoOp
   24   | CommandTui AutonomyLevel
   25   | CommandServe AutonomyLevel (Maybe FilePath) Bool
   26   | CommandAccessControl AccessControlAction
   27   | CommandGenOpenApi
   28   deriving (Eq, Show)
   29 
   30 -- | The @seal access-control@ subcommands. All of them are human-driven admin
   31 -- actions against @~\/.seal@; none is an opcode.
   32 --
   33 -- @enroll@ mints the one-time authority that adds a credential, @disable@ is
   34 -- the only way to turn access control off (there is no HTTP path that can),
   35 -- @reset@ is the lost-every-device escape hatch, and @migrate-credentials@
   36 -- finishes the pre-split → @credentials.toml@ move. @configure@ persists the
   37 -- gateway coordinates before the first passkey is enrolled.
   38 data AccessControlAction
   39   = AcStatus
   40   | AcEnroll !EnrollOptions
   41   | AcConfigure !ConfigureOptions
   42   | AcDisable !Bool
   43     -- ^ @--yes@: disabling hands the gateway to anyone who can reach it, so it
   44     -- is never implicit. Kept as a field rather than a parse-time requirement so
   45     -- a missing @--yes@ is a usage error (exit 2) the command can report.
   46   | AcReset
   47   | AcMigrateCredentials !Bool
   48     -- ^ @--force@: re-run the table removal even though @credentials.toml@
   49     -- exists. Never overwrites the store itself.
   50   deriving (Eq, Show)
   51 
   52 -- | The @configure@ flag set. Every field is optional (except that at least
   53 -- one must be present, which the command enforces): the operator changes the
   54 -- coordinates they name and nothing else. @coPort@ is raw text so a bad value
   55 -- is a usage error (exit 2) the command reports, not an optparse failure.
   56 data ConfigureOptions = ConfigureOptions
   57   { coHost     :: !(Maybe Text)
   58   , coPort     :: !(Maybe Text)
   59   , coTls      :: !Bool
   60   , coNoTls    :: !Bool
   61   , coTlsCert  :: !(Maybe FilePath)
   62   , coTlsKey   :: !(Maybe FilePath)
   63   , coTlsHosts :: ![Text]
   64   , coOrigin   :: !(Maybe Text)
   65   , coRpId     :: !(Maybe Text)
   66   } deriving (Eq, Show)
   67

   69 -- | The @enroll@ flag matrix. Validation lives in the command, not the parser,
   70 -- so every illegal combination is reachable and testable as a usage error.
   71 data EnrollOptions = EnrollOptions
   72   { eoTtl     :: !(Maybe String)
   73     -- ^ @--ttl N[s|m|h]@, unparsed: the command clamps and rejects.
   74   , eoNoToken :: !Bool
   75     -- ^ @--no-token@: mint a loopback-confined window grant instead of a token.
   76   , eoForce   :: !Bool
   77     -- ^ @--force@: replace an outstanding grant.
   78   , eoCancel  :: !Bool
   79     -- ^ @--cancel@: drop the outstanding grant, its claims and its token.
   80   } deriving (Eq, Show)

   82 pCommand :: Parser Command
   83 pCommand = hsubparser
   84   $  command "tui" (info (CommandTui <$> pAutonomy)
   85                          (progDesc "Start the interactive terminal UI (TUI)"))
   86   <> command "serve" (info (CommandServe <$> pAutonomy <*> pStaticDir <*> pNoEnrollmentToken)
   87                             (progDesc "Run the web gateway server"))
   88   <> command "access-control"
   89        (info (pAccessControl <**> helper)
   90              (progDesc "Manage passkey access control (security.toml, credentials.toml)"))
   91   <> command "gen-openapi" (info (pure CommandGenOpenApi)
   92                                   (progDesc "Print the OpenAPI 3.1 spec (JSON) to stdout"))
   93 
   94 -- | @access-control status | enroll | disable | reset | migrate-credentials@.
   95 pAccessControl :: Parser Command
   96 pAccessControl = CommandAccessControl <$> hsubparser
   97   (  command "status"
   98        (info (pure AcStatus)
   99              (progDesc "Print the access-control mode, origin and enrolled credentials"))
  100   <> command "enroll"
  101        (info (AcEnroll <$> pEnroll)
  102              (progDesc "Mint a one-time enrollment grant (token on stdout, hints on stderr)"))
  103   <> command "configure"
  104        (info (AcConfigure <$> pConfigure)
  105              (progDesc "Set the gateway bind host/port, TLS and passkey origin before first enrollment"))
  106   <> command "disable"
  107        (info (AcDisable <$> pDisableConfirm)
  108              (progDesc "Turn access control off, keeping credentials (requires --yes)"))
  109   <> command "reset"
  110        (info (AcReset <$ pResetConfirm)
  111              (progDesc "Remove the credentials and the section so you can re-enroll (requires --yes)"))
  112   <> command "migrate-credentials"
  113        (info (AcMigrateCredentials <$> pMigrateForce <* pMigrateConfirm)
  114              (progDesc "Move passkeys out of security.toml into credentials.toml (requires --yes)"))
  115   )

  117 -- | The @enroll@ flags. Mutually exclusive combinations are rejected by the
  118 -- command with exit code 2.
  119 pEnroll :: Parser EnrollOptions
  120 pEnroll = EnrollOptions
  121   <$> optional (strOption
  122         ( long "ttl"
  123        <> metavar "N[s|m|h]"
  124        <> help "Grant lifetime, e.g. 90s, 10m, 1h (default 10m, window 5m; 30s-24h)"
  125        ))
  126   <*> switch
  127         ( long "no-token"
  128        <> help "Mint a loopback-only enrollment window instead of a token (bootstrap only)"
  129         )
  130   <*> switch
  131         ( long "force"
  132        <> help "Replace an outstanding grant instead of refusing"
  133         )
  134   <*> switch
  135         ( long "cancel"
  136        <> help "Drop the outstanding grant, its claims and its token file"
  137         )

  139 -- | The @configure@ flags. All optional; the command requires at least one and
  140 -- validates combinations (exit 2) before touching the filesystem.
  141 pConfigure :: Parser ConfigureOptions
  142 pConfigure = ConfigureOptions
  143   <$> optionalText "host" "HOST" "Bind host (a name, an address, or 0.0.0.0 for every interface)"
  144   <*> optionalText "port" "PORT"
  145         "Bind port (1-65535); must match the port clients use"
  146   <*> switch
  147         ( long "tls"
  148        <> help "Serve HTTPS (with --tls-cert/--tls-key, or a generated self-signed pair)"
  149         )
  150   <*> switch
  151         ( long "no-tls"
  152        <> help "Serve HTTP; clears tls_cert, tls_key and tls_hosts"
  153         )
  154   <*> optionalFilePath "tls-cert" "PATH" "PEM certificate chain (implies TLS)"
  155   <*> optionalFilePath "tls-key" "PATH" "PEM private key (implies TLS)"
  156   <*> many (T.pack <$> strOption
  157         ( long "tls-host"
  158        <> metavar "NAME"
  159        <> help "Replace the generated certificate's SAN names (repeatable; requires TLS)"
  160         ))
  161   <*> optionalText "origin" "URL"
  162         "External origin clients use (https://name[:port]); also allowlisted for the WebSocket"
  163   <*> optionalText "rp-id" "DOMAIN"
  164         "WebAuthn relying-party id (defaults to the origin host)"
  165   where
  166     optionalText longName meta desc =
  167       (T.pack <$>) <$> optional (strOption
  168         ( long longName <> metavar meta <> help desc ))
  169     optionalFilePath longName meta desc =
  170       optional (strOption
  171         ( long longName <> metavar meta <> help desc ))

  173 -- | @--force@ re-runs the table removal. It never overwrites @credentials.toml@.
  174 pMigrateForce :: Parser Bool

  176   ( long "force"
  177  <> help "Drop the stale tables even if credentials.toml already exists (the store is never overwritten)"

  180 -- | @--yes@ is mandatory: a migration rewrites the file the next boot depends on.

  187 -- | @--yes@ is mandatory: resetting access control hands the gateway to anyone

  189 pResetConfirm :: Parser AccessControlAction

  195 -- | @--yes@ is mandatory: with access control off, anyone who can reach the

  199   ( long "yes"

  203 -- | @--yolo@ sets 'Full' autonomy (bypass the untrusted-opcode approval
  204 -- gate); absent defaults to 'Supervised'.
  205 pAutonomy :: Parser AutonomyLevel
  206 pAutonomy = flag Supervised Full
  207   ( long "yolo"
  208   <> help "Bypass the approval gate so untrusted opcodes run without prompting (ACK audit still recorded)"
  209   )

  211 -- | @--static-dir DIR@ overrides the @[gateway] static_dir@ in
  212 -- @~\/.seal\/security.toml@, serving UI assets from DIR instead of the embedded
  213 -- bundle (dev mode). Absent = @Nothing@ (use the security config or the
  214 -- embedded default).
  215 pStaticDir :: Parser (Maybe FilePath)
  216 pStaticDir = optional (strOption
  217   ( long "static-dir"
  218   <> metavar "DIR"
  219   <> help "Serve static UI assets from DIR instead of the embedded bundle (dev mode)"
  220   ))

    2 -- | The agent-definition model. 'AgentDefId' is a smart-constructed newtype with
    3 -- the same charset predicate as 'Seal.Core.Types.SessionId'
    4 -- (@[A-Za-z0-9_-]+@, non-empty, no leading dot). An 'AgentDef' is a named
    5 -- configuration (provider + model + system prompt + tool exposure) that a

   33 -- | Opaque agent-definition key. Smart-constructed via 'mkAgentDefId'.
   34 newtype AgentDefId = AgentDefId Text
   35   deriving stock (Eq, Ord, Show)
   36   deriving newtype (ToJSON, FromJSON)
   37 
   38 -- | @[A-Za-z0-9_-]+@, non-empty, no leading dot. Mirrors 'isValidMemoryId'.
   39 isValidAgentDefId :: Text -> Bool
   40 isValidAgentDefId t =
   41   not (T.null t)
   42     && T.head t /= '.'
   43     && T.all (`elem` chars) t
   44   where
   45     chars = ['A' .. 'Z'] <> ['a' .. 'z'] <> ['0' .. '9'] <> "_-"
   46 
   47 mkAgentDefId :: Text -> Either Text AgentDefId
   48 mkAgentDefId t
   49   | isValidAgentDefId t = Right (AgentDefId t)
   50   | otherwise           = Left ("invalid agent def id: " <> T.pack (show t))
   51 
   52 agentDefIdText :: AgentDefId -> Text
   53 agentDefIdText (AgentDefId t) = t

   55 -- | One agent definition. 'adProvider' is a provider label (e.g. @\"ollama\"@);
   56 -- 'adTools' is the opcode allow-list (which opcodes this agent may call). The
   57 -- system prompt and tool list are agent-visible data (not vault secrets); they
   58 -- are recorded in full in both the session transcript and the Audited log.
   59 -- 'adSession' is the originating session (provenance). 'adUpdatedAt' is bumped
   60 -- on each 'AGENT_DEF_UPDATE'.
   61 --
   62 -- 'adGroup' is an optional category for display grouping, mirroring 'skGroup'
   63 -- on 'Seal.Skills.Types.Skill'. It is derived from the on-disk parent
   64 -- directory name (@config\/agents\/\<group\>\/\<id\>.md@) and may be
   65 -- overridden via a @group:@ frontmatter key. 'Nothing' means the def belongs
   66 -- to the default (ungrouped) section. The def /id/ stays flat regardless of
   67 -- group (the charset predicate forbids @\/@), so group is purely display
   68 -- metadata — opcodes, file mapping, and lookups all key on 'adId'.
   69 data AgentDef = AgentDef
   70   { adId        :: AgentDefId
   71   , adName      :: Text
   72   , adProvider  :: Text
   73   , adModel     :: ModelId
   74   , adSystem    :: Maybe Text
   75   , adTools     :: AllowList OpName
   76   , adGroup     :: Maybe Text
   77   , adRole      :: Maybe Text
   78     -- ^ @\"orchestrator\"@ | @\"leaf\"@ | 'Nothing' (≡ leaf). Authoritative —
   79     -- the def author (or operator) decides; AGENT_START's per-task role may
   80     -- only NARROW (orchestrator → leaf), never widen. Validated at
   81     -- AGENT_DEF_WRITE (rejects anything else) and passed through the
   82     -- field validators at decode time.
   83   , adDescription :: Maybe Text
   84     -- ^ One-line human/model-facing summary rendered into the
   85     -- @\<available_agents\>@ catalog and AGENT_DEF_LIST output.
   86     -- Sanitized (single line, no control chars, no catalog-fence tokens,
   87     -- capped) — the same injection defense as every other def field.
   88   , adCreatedAt :: UTCTime
   89   , adUpdatedAt :: UTCTime
   90   , adSession   :: SessionId
   91   } deriving stock (Eq, Show, Generic)

   93 -- | The small per-field cap (role, description, group, provider, model):
   94 -- these render into prompt catalogs / tool output, so they are bounded
   95 -- tightly. 256 chars.
   96 agentFieldCapSmall :: Int
   97 agentFieldCapSmall = 256
   98 
   99 -- | The name cap. 1024 chars (names may be longer than the small cap).
  100 agentFieldCapName :: Int
  101 agentFieldCapName = 1024
  102 
  103 -- | Catalog-fence tokens a def field must never carry (they would let a
  104 -- def field forge or close the @\<available_agents\>@ /
  105 -- @\<available_skills\>@ catalog block, or forge frontmatter fences on
  106 -- re-encode).
  107 agentFenceTokens :: [Text]
  108 agentFenceTokens = ["</available_agents>", "</available_skills>", "---"]
  109 
  110 -- | Sanitize one agent-def text field for prompt/catalog rendering:
  111 -- newlines, carriage returns, and tabs become spaces, C0 control
  112 -- characters are stripped, the catalog fence tokens become underscores,
  113 -- and the result is truncated at @cap@ characters with a truncation
  114 -- marker. Pure; used by both decode paths and AGENT_DEF_WRITE (single
  115 -- chokepoint each).
  116 sanitizeAgentTextField :: Int -> Text -> Text
  117 sanitizeAgentTextField cap = truncateField cap . T.strip . replaceFences . stripControl . singleLine
  118   where
  119     singleLine = T.replace "\r" " " . T.replace "\n" " " . T.replace "\t" " "
  120     stripControl = T.filter (>= ' ')
  121     replaceFences t = foldl' (\acc tok -> T.replace tok "_" acc) t agentFenceTokens
  122     truncateField n txt
  123       | T.length txt <= n = txt
  124       | otherwise = T.take n txt <> "[...truncated]"
  125 
  126 -- | Sanitize every renderable field of an 'AgentDef' (role, description,
  127 -- group, provider, model, name) with the per-field cap matrix:
  128 -- @agentFieldCapSmall@ (256) for role/description/group/provider/model,
  129 -- @agentFieldCapName@ (1024) for name. Pure; applied by both decode paths
  130 -- and AGENT_DEF_WRITE so no unsanitized field can reach a prompt or tool
  131 -- output.
  132 sanitizeAgentDefFields :: AgentDef -> AgentDef
  133 sanitizeAgentDefFields d = d
  134   { adName        = sanitizeAgentTextField agentFieldCapName (adName d)
  135   , adProvider    = sanitizeAgentTextField agentFieldCapSmall (adProvider d)
  136   , adModel       = ModelId (sanitizeAgentTextField agentFieldCapSmall m)
  137   , adGroup       = sanitizeMaybe agentFieldCapSmall (adGroup d)
  138   , adRole        = sanitizeMaybe agentFieldCapSmall (adRole d)
  139   , adDescription = sanitizeMaybe agentFieldCapSmall (adDescription d)
  140   }
  141   where
  142     m = case adModel d of ModelId t -> t
  143     sanitizeMaybe cap = fmap (sanitizeAgentTextField cap)

  145 -- | Encode an 'AllowList OpName' as a JSON value: @\"all\"@ for 'AllowAll', or
  146 -- an array of opcode-name strings for 'AllowOnly'.
  147 allowListToValue :: AllowList OpName -> Value
  148 allowListToValue AllowAll       = String "all"
  149 allowListToValue (AllowOnly xs) = toJSON (map unOpNameText (Set.toList xs))
  150   where
  151     unOpNameText (OpName t) = t
  152 
  153 -- | Decode an 'AllowList OpName' from @\"all\"@ or an array of opcode-name
  154 -- strings. Unknown shapes default to 'AllowAll' (fail-closed would block the
  155 -- agent from calling anything; the wiring layer validates input before
  156 -- construction, so a malformed stored def is treated permissively).
  157 allowListFromValue :: Value -> AllowList OpName
  158 allowListFromValue (String "all") = AllowAll
  159 allowListFromValue (Array xs)     = AllowOnly (Set.fromList [ OpName t | String t <- V.toList xs ])
  160 allowListFromValue _               = AllowAll

  162 instance ToJSON AgentDef where
  163   toJSON d = object
  164     [ "id"         .= adId d
  165     , "name"       .= adName d
  166     , "provider"   .= adProvider d
  167     , "model"      .= adModel d
  168     , "system"     .= adSystem d
  169     , "tools"      .= allowListToValue (adTools d)
  170     , "group"      .= adGroup d
  171     , "role"       .= adRole d
  172     , "description" .= adDescription d
  173     , "created_at" .= adCreatedAt d
  174     , "updated_at" .= adUpdatedAt d
  175     , "session"    .= adSession d
  176     ]
  177 
  178 instance FromJSON AgentDef where
  179   parseJSON = withObject "AgentDef" $ \o -> AgentDef
  180     <$> o .:  "id"
  181     <*> o .:  "name"
  182     <*> o .:  "provider"
  183     <*> o .:  "model"
  184     <*> o .:? "system"
  185     <*> (allowListFromValue <$> o .: "tools")
  186     <*> o .:? "group"
  187     <*> o .:? "role"
  188     <*> o .:? "description"
  189     <*> o .:  "created_at"
  190     <*> o .:  "updated_at"
  191     <*> o .:  "session"

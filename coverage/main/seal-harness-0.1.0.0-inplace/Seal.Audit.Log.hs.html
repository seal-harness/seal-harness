    2 -- | Cross-session audit log for trusted mutating opcodes.
    3 --
    4 -- The audit log is a monthly-rotated, append-only text file under
    5 -- @~\/.seal\/audit\/audit-YYYY-MM.log@. Each line records one mutating
    6 -- opcode invocation: timestamp, session id, opcode name, action, target
    7 -- id, and the config repo's git commit hash at the time of the write.
    8 --
    9 -- The log is intentionally concise — full diffs and content live in the
   10 -- config git repo (for skills and agent defs) or the session transcript
   11 -- (for vault and memory). The audit log is the single cross-session
   12 -- timeline an operator can @tail -F@ or @grep@ to see what changed and
   13 -- when.
   14 --
   15 -- /Process-global pattern/: like 'Seal.Logging.Global', the audit logger
   16 -- is set once at startup via 'setAuditLogger' and read via
   17 -- 'globalAuditLog'. This avoids threading an 'AuditLogger' through the
   18 -- dispatcher and every opcode call site. If 'setAuditLogger' has not been
   19 -- called, 'globalAuditLog' is a no-op (best-effort — the audit log must
   20 -- never crash the caller, especially inside the dispatch path).
   21 module Seal.Audit.Log
   22   ( AuditLogger
   23   , AuditSink (..)
   24   , fileAuditSink
   25   , fileAuditSinkWithGit
   26   , setAuditLogger
   27   , unsetAuditLogger
   28   , globalAuditLog
   29   , globalAuditWrite
   30   , withAuditLogger
   31   , auditWrite
   32   , auditWriteIf
   33   , formatAuditLine
   34   , auditFileName
   35   ) where

   49 -- | Where audit lines are written. The production sink writes to a
   50 -- monthly file under @~\/.seal\/audit\/@ and optionally fetches the
   51 -- config repo's git HEAD hash on each write; tests inject a simple
   52 -- temp-directory sink.
   53 data AuditSink = AuditSink
   54   { asWriteLine :: Text -> IO ()
   55     -- ^ Write one complete line (without trailing newline) to the sink.
   56   , asGitHash :: IO (Maybe Text)
   57     -- ^ Fetch the current git HEAD short hash for the config repo.
   58     -- Returns 'Nothing' if git is unavailable or the repo has no commits.
   59     -- Best-effort: failures are swallowed and rendered as @-@.
   60   }
   61 
   62 -- | A file-based audit sink that writes to @\<dir\>\/audit-YYYY-MM.log@.
   63 -- No git hash fetching (the hash field in log lines will be @-@). Used by
   64 -- tests.
   65 fileAuditSink :: FilePath -> AuditSink
   66 fileAuditSink dir = AuditSink
   67   { asWriteLine = writeLineToFile dir
   68   , asGitHash = pure Nothing
   69   }
   70 
   71 -- | A file-based audit sink with git hash fetching. The directory is
   72 -- created (recursively) on construction. Each write appends one line to
   73 -- the current month's file. The git hash is fetched via the supplied IO
   74 -- action (typically @git -C \<configRoot\> rev-parse --short HEAD@).
   75 fileAuditSinkWithGit :: FilePath -> IO (Maybe Text) -> AuditSink
   76 fileAuditSinkWithGit dir gitHashAction = AuditSink
   77   { asWriteLine = writeLineToFile dir
   78   , asGitHash = gitHashAction
   79   }
   80 
   81 -- | Internal: write one line to the monthly file under @dir@.
   82 writeLineToFile :: FilePath -> Text -> IO ()
   83 writeLineToFile dir line = do
   84   now <- getCurrentTimeSafe
   85   let fname = "audit-" <> formatTime defaultTimeLocale "%Y-%m" now <> ".log"
   86       path = dir </> fname
   87   appendFile path (T.unpack line <> "\n")
   88     `catch` \(_ :: SomeException) -> pure ()
   89 
   90 -- | A safe 'getCurrentTime' that never throws (best-effort).
   91 getCurrentTimeSafe :: IO UTCTime
   92 getCurrentTimeSafe = getCurrentTime
   93   `catch` \(_ :: SomeException) -> pure (UTCTime (toEnum 0) 0)
   94 
   95 -- | The audit logger handle. Carries the sink and is set once at startup.
   96 newtype AuditLogger = AuditLogger
   97   { alSink :: AuditSink
   98   }
   99 
  100 -- | The process-global audit logger ref. Initialized to 'Nothing' (no-op).
  101 -- Set once at startup via 'setAuditLogger'.
  102 globalAuditLoggerRef :: IORef (Maybe AuditLogger)
  103 globalAuditLoggerRef = unsafePerformIO (newIORef Nothing)
  104 {-# NOINLINE globalAuditLoggerRef #-}
  105 
  106 -- | Set the process-global audit logger. Called once at startup.
  107 setAuditLogger :: AuditSink -> IO ()
  108 setAuditLogger sink = writeIORef globalAuditLoggerRef (Just (AuditLogger sink))
  109 
  110 -- | Unset the process-global audit logger (set the ref to 'Nothing'). Used
  111 -- by tests that install a logger and want to restore the no-op default
  112 -- after the test so subsequent tests are unaffected.
  113 unsetAuditLogger :: IO ()
  114 unsetAuditLogger = writeIORef globalAuditLoggerRef Nothing
  115 
  116 -- | Emit an audit line via the process-global logger. If no logger has
  117 -- been set, this is a no-op. Best-effort: if the sink throws, the error is
  118 -- silently swallowed (the audit log must never crash the dispatch path).
  119 globalAuditLog :: Text -> IO ()
  120 globalAuditLog line = do
  121   mLogger <- readIORef globalAuditLoggerRef
  122   case mLogger of
  123     Nothing -> pure ()
  124     Just logger -> asWriteLine (alSink logger) line
  125       `catch` \(_ :: SomeException) -> pure ()
  126 
  127 -- | Emit a structured audit entry via the process-global logger. Fetches
  128 -- the git hash from the sink, formats the line, and writes it. If no
  129 -- logger has been set, this is a no-op. Best-effort. This is the function
  130 -- the agent loop calls after a successful mutating opcode dispatch.
  131 globalAuditWrite
  132   :: UTCTime -> Maybe SessionId -> OpName -> Maybe Text
  133   -> Text -> IO ()
  134 globalAuditWrite ts mSession opName mAction targetId = do
  135   mLogger <- readIORef globalAuditLoggerRef
  136   case mLogger of
  137     Nothing -> pure ()
  138     Just logger -> do
  139       mGitHash <- asGitHash (alSink logger) `catch` \(_ :: SomeException) -> pure Nothing
  140       let line = formatAuditLine ts mSession opName mAction targetId mGitHash
  141       asWriteLine (alSink logger) line
  142         `catch` \(_ :: SomeException) -> pure ()
  143 
  144 -- | Bracket the audit logger's lifetime. Sets the logger for the duration
  145 -- of the action, unsets it on exit (even on exception). Used by tests; the
  146 -- production code uses 'setAuditLogger' (the logger lives for the process
  147 -- lifetime).
  148 withAuditLogger :: AuditSink -> (AuditLogger -> IO a) -> IO a
  149 withAuditLogger sink = bracket before after
  150   where
  151     before = do
  152       let logger = AuditLogger sink
  153       writeIORef globalAuditLoggerRef (Just logger)
  154       pure logger
  155     after _ = writeIORef globalAuditLoggerRef Nothing
  156 
  157 -- | Write an audit entry for a mutating opcode invocation. Called by
  158 -- tests that want explicit control over the logger. The timestamp is
  159 -- supplied by the caller (so tests can use a fixed time); in production
  160 -- the agent loop uses 'globalAuditWrite' which fetches the current time
  161 -- and git hash internally.
  162 auditWrite
  163   :: AuditLogger -> UTCTime -> Maybe SessionId -> OpName -> Maybe Text
  164   -> Text -> Maybe Text -> IO ()
  165 auditWrite logger ts mSession opName mAction targetId mGitHash = do
  166   let line = formatAuditLine ts mSession opName mAction targetId mGitHash
  167   asWriteLine (alSink logger) line
  168     `catch` \(_ :: SomeException) -> pure ()
  169 
  170 -- | Like 'auditWrite' but only writes if the predicate says this
  171 -- opcode+action should be audited. This is the function tests call to
  172 -- verify the predicate gating.
  173 auditWriteIf
  174   :: AuditLogger -> (OpName -> Maybe Text -> Bool)
  175   -> UTCTime -> Maybe SessionId -> OpName -> Maybe Text
  176   -> Text -> Maybe Text -> IO ()
  177 auditWriteIf logger predicate ts mSession opName mAction targetId mGitHash =
  178   when (predicate opName mAction) $
  179     auditWrite logger ts mSession opName mAction targetId mGitHash

  181 -- | Format one audit log line. The format is:
  182 --
  183 -- @
  184 -- YYYY-MM-DDThh:mm:ss.sssZ session=\<id\> op=\<name\> action=\<action\> id=\<target\> git=\<hash\>
  185 -- @
  186 --
  187 -- Missing fields are rendered as @-@ (never omitted, so the line is
  188 -- greppable by field). The timestamp is ISO-8601 UTC with milliseconds.
  189 formatAuditLine
  190   :: UTCTime -> Maybe SessionId -> OpName -> Maybe Text
  191   -> Text -> Maybe Text -> Text
  192 formatAuditLine ts mSession (OpName opName) mAction targetId mGitHash =
  193   T.intercalate " "
  194     [ T.pack (formatTime defaultTimeLocale "%Y-%m-%dT%H:%M:%S.%3qZ" ts)
  195     , "session=" <> maybe "-" sessionIdText mSession
  196     , "op=" <> opName
  197     , "action=" <> fromMaybe "-" mAction
  198     , "id=" <> targetId
  199     , "git=" <> fromMaybe "-" mGitHash
  200     ]
  201 
  202 -- | The audit filename for a given timestamp: @audit-YYYY-MM.log@.
  203 auditFileName :: UTCTime -> FilePath
  204 auditFileName ts =
  205   "audit-" <> formatTime defaultTimeLocale "%Y-%m" ts <> ".log"

    6 -- | Generate an OpenAPI 3.1 specification from the 'SealRoute' declarative
    7 -- route spec (design §2.2 + Phase 4 — the §1a UC3 follow-up).
    8 --
    9 -- This module is pure: no 'IO', no 'ApiDeps'. It builds a
   10 -- 'Data.OpenApi.OpenApi' document by feeding an OpenAPI-flavored 'Meta' table
   11 -- (parallel to the runtime 'Seal.Gateway.Route.routeMeta') to
   12 -- 'Trasa.OpenApi.toOpenApi'. Every path is prefixed with @\/api@ (matching the
   13 -- gateway's @("api" : _) -> apiApp@ dispatch in 'Seal.Gateway.Server').
   14 --
   15 -- The handlers build JSON 'Value's directly (the 'Resp'/'Req' types are
   16 -- opaque markers), so request and response bodies are typed as free-form JSON
   17 -- objects for v1. The spec documents routes, methods, capture names, and media
   18 -- types honestly without overpromising a typed body shape the handlers don't
   19 -- yet enforce. High-value bodies can later get dedicated 'ToSchema' instances.

   46 -- | The complete Seal Harness OpenAPI 3.1 document, generated from
   47 -- 'allRoutes' + 'openApiMeta'. The 'Data.OpenApi.Info' is stamped with the
   48 -- project title + version (the 'mempty' default has empty strings).
   49 sealOpenApi :: OpenApi
   50 sealOpenApi = baseDoc
   51   { _openApiInfo = (_openApiInfo baseDoc)
   52       { _infoTitle = "Seal Harness Gateway API"
   53       , _infoVersion = "0.1.0.0"
   54       } }
   55   where
   56     baseDoc = Trasa.OpenApi.toOpenApi openApiMeta allRoutes

   58 -- | The OpenAPI document encoded as compact JSON. This is what
   59 -- @seal gen-openapi@ writes to stdout and what @GET \/api\/openapi.json@ serves.
   60 encodeOpenApi :: ByteString
   61 encodeOpenApi = A.encode sealOpenApi

  135 -- | Prefix a path with @match "api"@ so the rendered OpenAPI paths include the
  136 -- @\/api@ prefix the client sees (the gateway strips it before dispatching to
  137 -- the manual router).
  138 api
  139   :: Path OpenApiCaptureCodec caps
  140   -> Rec (Query OpenApiCaptureCodec) qrys
  141   -> RequestBody (Many OpenApiBodyCodec) req
  142   -> ResponseBody (Many OpenApiBodyCodec) resp
  143   -> M.Method
  144   -> Meta OpenApiCaptureCodec OpenApiCaptureCodec
  145            (Many OpenApiBodyCodec) (Many OpenApiBodyCodec)
  146            caps qrys req resp
  147 api p = Meta (match "api" ./ p)

  177 -- | The free-form JSON body codec for the opaque marker types ('Resp', 'Req',

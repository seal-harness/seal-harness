    1 -- | Strict, resource-safe file-reading helpers built on @conduit@ + @resourcet@.
    2 --
    3 -- The lazy 'Data.ByteString.Lazy.readFile' / 'Data.Text.IO.readFile' helpers
    4 -- keep the file handle open until the lazy structure is fully consumed, which
    5 -- is GC-dependent. The 'Seal.Gateway.ListsSnapshot.buildListsSnapshot' hot
    6 -- path reads hundreds of small files per request (111 sessions × multiple
    7 -- files each) and the lazy handles pile up faster than the GC reclaims them,
    8 -- hitting the OS soft FD limit ('EMFILE' / @resource exhausted (Too many
    9 -- open files)@).
   10 --
   11 -- @conduit@'s 'sourceFile' opens the handle inside a 'ResourceT' scope, so the
   12 -- handle is released as soon as 'runConduitRes' returns — regardless of when
   13 -- (or whether) the resulting lazy 'ByteString' is forced. The bytes are
   14 -- materialized eagerly by 'sinkLazy' before the 'ResourceT' closes the handle,
   15 -- so the decode sees the full file with the fd already released.
   16 --
   17 -- Also provides 'streamFirstMatch' for early-termination line streaming —
   18 -- reads a file line-by-line via conduit and returns the first line where a
   19 -- predicate returns 'Just', without materializing the entire file. Used by
   20 -- 'firstUserMessageSnippetFast' to avoid reading multi-MB conversation files
   21 -- when only the first user message is needed.
   22 module Seal.Util.StrictIO
   23   ( readFileStrict
   24   , readFileTextStrict
   25   , streamFirstMatch
   26   , streamDecodeLines
   27   , streamDecodeAfter
   28   , decodeFileStrict
   29   ) where
   30

   39 -- | Read a file strictly into a lazy 'ByteString', closing the underlying
   40 -- handle before the helper returns. The handle is acquired and released by
   41 -- the 'ResourceT' scope of 'runConduitRes'; the bytes are materialized by
   42 -- 'sinkLazy' before the scope exits. Callers 'doesFileExist'-guard the path
   43 -- first (the lazy variants they replace did the same).
   44 readFileStrict :: FilePath -> IO BL.ByteString
   45 readFileStrict path = runConduitRes (sourceFile path .| sinkLazy)
   46 
   47 -- | Read a file strictly as 'Text' by reading bytes strictly then decoding.
   48 -- The handle is closed before the decode runs.
   49 readFileTextStrict :: FilePath -> IO Text
   50 readFileTextStrict path = TE.decodeUtf8 . BL.toStrict <$> readFileStrict path

   52 -- | Stream lines from a file, applying a predicate to each line. Returns
   53 -- the first line where the predicate returns 'Just a', stopping the stream
   54 -- early. The file handle is released as soon as the match is found (or the
   55 -- file is exhausted) by conduit's 'ResourceT' scope. Each line is a strict
   56 -- 'ByteString' without the trailing newline.
   57 --
   58 -- This avoids materializing the entire file when only an early line is
   59 -- needed — e.g. @conversation.jsonl@ files averaging 2.7 MB where the
   60 -- first user message is typically in the first few lines. The line
   61 -- splitting is done in-stream (accumulating chunks until a newline is
   62 -- found) so no intermediate @conduit-extra@ dependency is needed.
   63 streamFirstMatch :: (ByteString -> Maybe a) -> FilePath -> IO (Maybe a)
   64 streamFirstMatch test path = runConduitRes
   65   (sourceFile path .| lineLoop BS.empty)
   66   where
   67     -- | Accumulate chunks until a newline (0x0A) is found, then apply the
   68     -- predicate to the complete line. Stops early on the first match.
   69     lineLoop acc = do
   70       mChunk <- await
   71       case mChunk of
   72         Nothing -> processRemaining acc
   73         Just chunk -> do
   74           let combined = acc <> chunk
   75           case BS.elemIndex 10 combined of
   76             Nothing -> lineLoop combined
   77             Just idx ->
   78               let line = BS.take idx combined
   79                   rest = BS.drop (idx + 1) combined
   80               in case test line of
   81                 Just a  -> pure (Just a)
   82                 Nothing -> lineLoop rest
   83 
   84     -- | Process remaining lines in the accumulator after EOF. The
   85     -- accumulator may contain multiple newline-separated lines if the
   86     -- source yielded a large chunk on the last await.
   87     processRemaining bs
   88       | BS.null bs = pure Nothing
   89       | otherwise = case BS.elemIndex 10 bs of
   90           Nothing -> case test bs of
   91             Just a  -> pure (Just a)
   92             Nothing -> pure Nothing
   93           Just idx ->
   94             let line = BS.take idx bs
   95                 rest = BS.drop (idx + 1) bs
   96             in case test line of
   97               Just a  -> pure (Just a)
   98               Nothing -> processRemaining rest

  100 -- | Read and decode a JSON file strictly. The handle is closed before the
  101 -- decode runs. Returns 'Nothing' on a decode failure (the caller is
  102 -- expected to 'doesFileExist'-guard the path first).
  103 decodeFileStrict :: FromJSON a => FilePath -> IO (Maybe a)
  104 decodeFileStrict path = decode <$> readFileStrict path
  105 -- | Stream a file line-by-line via conduit, decoding each non-empty line as
  106 -- JSON. Unlike 'readFileTextStrict' which materializes the entire file as a
  107 -- single contiguous 'Text' (potentially hundreds of MB for large transcripts),
  108 -- this reads and decodes one line at a time. The file handle is released by
  109 -- the 'ResourceT' scope when the stream completes.
  110 --
  111 -- Empty lines (blank lines between entries) are skipped. Unparseable lines
  112 -- are silently dropped (matching the @mapMaybe A.decode@ pattern used by the
  113 -- existing 'readTranscriptEntriesTimed').
  114 streamDecodeLines :: FromJSON a => FilePath -> IO [a]
  115 streamDecodeLines path = runConduitRes
  116   (sourceFile path .| lineDecodeLoop BS.empty)
  117   where
  118     lineDecodeLoop acc = do
  119       mChunk <- await
  120       case mChunk of
  121         Nothing -> processRemaining acc
  122         Just chunk -> splitLines (acc <> chunk)
  123     -- | Split and decode all complete lines in the buffer, then await more.
  124     -- When no newline is found in the buffer, await the next chunk. This
  125     -- handles the case where a single chunk contains multiple lines (the
  126     -- common case when @sourceFile@ delivers a large chunk).
  127     splitLines bs =
  128       case BS.elemIndex 10 bs of
  129         Nothing -> lineDecodeLoop bs
  130         Just idx ->
  131           let line = BS.take idx bs
  132               rest = BS.drop (idx + 1) bs
  133           in decodeLine line (splitLines rest)
  134     processRemaining bs
  135       | BS.null bs = pure []
  136       | otherwise = decodeLine bs (pure [])
  137     decodeLine line cont =
  138       if BS.null line
  139         then cont
  140         else case decode (BL.fromStrict line) of
  141                Just a  -> fmap (a :) cont
  142                Nothing -> cont
  143 
  144 -- | Stream a file line-by-line, skipping the first @skipCount@ non-empty
  145 -- lines, then decoding the remaining lines as JSON. This avoids both (a)
  146 -- materializing the entire file as a single 'Text' and (b) decoding lines
  147 -- that will be discarded. Used by 'readTranscriptEntriesTimed' when a
  148 -- 'ReadLastN' limit is set: the first @startConvLen@ conversation lines are
  149 -- skipped without decoding, and only the tail is decoded.
  150 --
  151 -- Empty lines are not counted toward @skipCount@ (matching the
  152 -- @filter (not . T.null) (T.lines raw)@ semantics of the existing code).
  153 streamDecodeAfter :: FromJSON a => Int -> FilePath -> IO [a]
  154 streamDecodeAfter skipCount path = runConduitRes
  155   (sourceFile path .| skipThenDecode BS.empty skipCount)
  156   where
  157     skipThenDecode acc skip
  158       | skip <= 0 = decodeLoop acc
  159       | otherwise = do
  160           mChunk <- await
  161           case mChunk of
  162             Nothing -> processRemainingSkip acc skip
  163             Just chunk -> splitSkip (acc <> chunk) skip
  164     -- | Split lines in the buffer, skipping non-empty ones until skip is
  165     -- exhausted, then switching to decode mode. Handles multi-line chunks.
  166     splitSkip bs skip
  167       | skip <= 0 = splitDecode bs
  168       | otherwise =
  169           case BS.elemIndex 10 bs of
  170             Nothing -> skipThenDecode bs skip
  171             Just idx ->
  172               let line = BS.take idx bs
  173                   rest = BS.drop (idx + 1) bs
  174               in if BS.null line
  175                    then splitSkip rest skip
  176                    else splitSkip rest (skip - 1)
  177     processRemainingSkip bs skip
  178       | skip <= 0 = processRemainingDecode bs
  179       | BS.null bs = pure []
  180       | otherwise =
  181           case BS.elemIndex 10 bs of
  182             Nothing -> pure []  -- remaining bytes are one last skipped line
  183             Just idx ->
  184               let line = BS.take idx bs
  185                   rest = BS.drop (idx + 1) bs
  186               in if BS.null line
  187                    then processRemainingSkip rest skip
  188                    else processRemainingSkip rest (skip - 1)
  189     -- Decoding phase (skip exhausted): same as streamDecodeLines' loop.
  190     decodeLoop acc = do
  191       mChunk <- await
  192       case mChunk of
  193         Nothing -> processRemainingDecode acc
  194         Just chunk -> splitDecode (acc <> chunk)
  195     -- | Split and decode all complete lines in the buffer, then await more.
  196     splitDecode bs =
  197       case BS.elemIndex 10 bs of
  198         Nothing -> decodeLoop bs
  199         Just idx ->
  200           let line = BS.take idx bs
  201               rest = BS.drop (idx + 1) bs
  202           in decodeLine line (splitDecode rest)
  203     processRemainingDecode bs
  204       | BS.null bs = pure []
  205       | otherwise = decodeLine bs (pure [])
  206     decodeLine line cont =
  207       if BS.null line
  208         then cont
  209         else case decode (BL.fromStrict line) of
  210                Just a  -> fmap (a :) cont
  211                Nothing -> cont

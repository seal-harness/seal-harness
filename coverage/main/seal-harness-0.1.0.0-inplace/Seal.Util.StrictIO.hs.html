    1 -- | Strict, resource-safe file-reading helpers built on @conduit@ + @resourcet@.
    2 --
    3 -- The lazy 'Data.ByteString.Lazy.readFile' / 'Data.Text.IO.readFile' helpers
    4 -- keep the file handle open until the lazy structure is fully consumed, which
    5 -- is GC-dependent. The 'Seal.Gateway.ListsSnapshot.buildListsSnapshot' hot
    6 -- path reads hundreds of small files per request (111 sessions × multiple
    7 -- files each) and the lazy handles pile up faster than the GC reclaims them,
    8 -- hitting the OS soft FD limit ('EMFILE' / @resource exhausted (Too many
    9 -- open files)@).
   10 --
   11 -- @conduit@'s 'sourceFile' opens the handle inside a 'ResourceT' scope, so the
   12 -- handle is released as soon as 'runConduitRes' returns — regardless of when
   13 -- (or whether) the resulting lazy 'ByteString' is forced. The bytes are
   14 -- materialized eagerly by 'sinkLazy' before the 'ResourceT' closes the handle,
   15 -- so the decode sees the full file with the fd already released.
   16 --
   17 -- Also provides 'streamFirstMatch' for early-termination line streaming —
   18 -- reads a file line-by-line via conduit and returns the first line where a
   19 -- predicate returns 'Just', without materializing the entire file. Used by
   20 -- 'firstUserMessageSnippetFast' to avoid reading multi-MB conversation files
   21 -- when only the first user message is needed.
   22 module Seal.Util.StrictIO
   23   ( readFileStrict
   24   , readFileTextStrict
   25   , streamFirstMatch
   26   , decodeFileStrict
   27   ) where
   28 
   29 import Conduit ( await, runConduitRes, sinkLazy, sourceFile, (.|) )
   30 import Data.Aeson (FromJSON, decode)
   31 import Data.ByteString (ByteString)
   32 import Data.ByteString qualified as BS
   33 import Data.ByteString.Lazy qualified as BL
   34 import Data.Text (Text)
   35 import Data.Text.Encoding qualified as TE

   37 -- | Read a file strictly into a lazy 'ByteString', closing the underlying
   38 -- handle before the helper returns. The handle is acquired and released by
   39 -- the 'ResourceT' scope of 'runConduitRes'; the bytes are materialized by
   40 -- 'sinkLazy' before the scope exits. Callers 'doesFileExist'-guard the path
   41 -- first (the lazy variants they replace did the same).
   42 readFileStrict :: FilePath -> IO BL.ByteString
   43 readFileStrict path = runConduitRes (sourceFile path .| sinkLazy)
   44 
   45 -- | Read a file strictly as 'Text' by reading bytes strictly then decoding.
   46 -- The handle is closed before the decode runs.
   47 readFileTextStrict :: FilePath -> IO Text
   48 readFileTextStrict path = TE.decodeUtf8 . BL.toStrict <$> readFileStrict path

   50 -- | Stream lines from a file, applying a predicate to each line. Returns
   51 -- the first line where the predicate returns 'Just a', stopping the stream
   52 -- early. The file handle is released as soon as the match is found (or the
   53 -- file is exhausted) by conduit's 'ResourceT' scope. Each line is a strict
   54 -- 'ByteString' without the trailing newline.
   55 --
   56 -- This avoids materializing the entire file when only an early line is
   57 -- needed — e.g. @conversation.jsonl@ files averaging 2.7 MB where the
   58 -- first user message is typically in the first few lines. The line
   59 -- splitting is done in-stream (accumulating chunks until a newline is
   60 -- found) so no intermediate @conduit-extra@ dependency is needed.
   61 streamFirstMatch :: (ByteString -> Maybe a) -> FilePath -> IO (Maybe a)
   62 streamFirstMatch test path = runConduitRes
   63   (sourceFile path .| lineLoop BS.empty)
   64   where
   65     -- | Accumulate chunks until a newline (0x0A) is found, then apply the
   66     -- predicate to the complete line. Stops early on the first match.
   67     lineLoop acc = do
   68       mChunk <- await
   69       case mChunk of
   70         Nothing -> processRemaining acc
   71         Just chunk -> do
   72           let combined = acc <> chunk
   73           case BS.elemIndex 10 combined of
   74             Nothing -> lineLoop combined
   75             Just idx ->
   76               let line = BS.take idx combined
   77                   rest = BS.drop (idx + 1) combined
   78               in case test line of
   79                 Just a  -> pure (Just a)
   80                 Nothing -> lineLoop rest
   81 
   82     -- | Process remaining lines in the accumulator after EOF. The
   83     -- accumulator may contain multiple newline-separated lines if the
   84     -- source yielded a large chunk on the last await.
   85     processRemaining bs
   86       | BS.null bs = pure Nothing
   87       | otherwise = case BS.elemIndex 10 bs of
   88           Nothing -> case test bs of
   89             Just a  -> pure (Just a)
   90             Nothing -> pure Nothing
   91           Just idx ->
   92             let line = BS.take idx bs
   93                 rest = BS.drop (idx + 1) bs
   94             in case test line of
   95               Just a  -> pure (Just a)
   96               Nothing -> processRemaining rest
   97 
   98 -- | Read and decode a JSON file strictly. The handle is closed before the
   99 -- decode runs. Returns 'Nothing' on a decode failure (the caller is
  100 -- expected to 'doesFileExist'-guard the path first).
  101 decodeFileStrict :: FromJSON a => FilePath -> IO (Maybe a)
  102 decodeFileStrict path = decode <$> readFileStrict path

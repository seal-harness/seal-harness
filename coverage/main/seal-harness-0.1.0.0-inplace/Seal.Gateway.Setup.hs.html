    2 -- | The setup endpoints: status, vault creation, vault reopen, and the
    3 -- provider Test-connection ping.
    4 --
    5 -- == Trust boundary (read this before changing anything here)
    6 --
    7 -- These are human-driven admin endpoints on the local control plane, in the
    8 -- same class as @\/vault setup@ and @POST \/api\/secrets@. They add no opcode,
    9 -- no 'UntrustedIO', and no ISA path. Two protections apply, both already in
   10 -- place before this module is reached:
   11 --
   12 -- * the mutation origin guard in 'Seal.Gateway.API.apiApp' (cross-origin
   13 --   requests are refused before dispatch, so a hostile page cannot drive a
   14 --   vault creation);
   15 -- * the passkey middleware in "Seal.Gateway.AccessControl", which is
   16 --   default-deny — every route here answers @403 access_control_required@
   17 --   until access control is configured, and @401@ without a session or
   18 --   service token.
   19 --
   20 -- Residual classes, stated rather than hidden: holders of the per-run service
   21 -- token, an explicit @mode = "none"@ waiver, and same-UID local processes.
   22 -- The last is not a widening: such a process can already write
   23 -- @~\/.seal\/security.toml@ or run @\/vault setup@ in @seal tui@.
   24 --
   25 -- Nothing here ever returns key material. Creation and reopen responses carry
   26 -- the public recipient and the key-type label; errors carry a fixed reason and
   27 -- an actionable hint ('Seal.Vault.Provision.renderProvisionError').
   28 --
   29 -- This module never imports 'Seal.Gateway.API' (that module imports this one),
   30 -- so the helpers and handlers are self-contained.

   91 -- | The setup endpoints' injected dependencies.
   92 --
   93 -- Two seams, both injected rather than imported so the handlers stay testable
   94 -- and so "Test connection" pings what is on disk now, not what was there at
   95 -- boot: the provisioning core (which shells out to @age-keygen@ /
   96 -- @age-plugin-yubikey@ / @age@) and the provider resolver (which needs a

  103     -- OpenAI-compatible instance) is what "Test connection" actually pings.

  119 -- | Inert dependencies for the 'ApiDeps' literals in tests that do not
  120 -- exercise the setup endpoints. Every seam fails closed with a message that
  121 -- names the stub, so a test that accidentally reaches one says so.

  123 stubSetupDeps = SetupDeps
  124   { sdProvision = ProvisionDeps
  125       { pdKeygen = SetupKeygen
  126           { skLocal = pure (Left stubError)

  134       , pdWriteConfig = \_ -> pure (Left "stub setup deps")

  147 -- | The wizard's state. Never fails: a missing file degrades to "nothing
  148 -- configured", and an unparseable one becomes the V0 blocked row.
  149 handleSetupState :: SetupEnv -> IO Response
  150 handleSetupState env = do
  151   security <- loadSecurityConfig (seSecurityPath env)
  152   runtime <- loadRuntimeConfig (seConfigPath env)
  153   mhandle <- readIORef (seHandleRef env)
  154   let sec = fromRight defaultSecurityConfig security
  155       rt = fromRight defaultRuntimeConfig runtime
  156       vaultPath = maybe (vaultFilePath (sePaths env)) T.unpack (scVaultPath sec)
  157   vaultExists <- doesFileExist vaultPath
  158   identityReadable <- case scVaultIdentity sec of
  159     Nothing -> pure Nothing
  160     Just raw -> Just <$> doesFileExist (T.unpack raw)
  161   state <- deriveSetupState SetupInputs
  162     { siPaths = sePaths env
  163     , siRuntime = rt
  164     , siSecurity = sec
  165     , siSecurityProblem = either (Just . summariseConfigProblem) (const Nothing) security
  166     , siHandle = mhandle
  167     , siVaultFileExists = vaultExists
  168     , siIdentityReadable = identityReadable
  169     , siConfiguredProviders = seProviders env
  170     -- The seal home's parent is the user's home for the default ~/.seal
  171     -- layout, which is what the tilde form in a detail is measured from.
  172     , siHome = takeDirectory (spHome (sePaths env))
  173     }
  174   pure (setupJsonNoStore status200 (setupStateJson state))
  175 
  176 setupStateJson :: SetupState -> Value
  177 setupStateJson state = object
  178   [ "required_complete" .= ssRequiredComplete state
  179   , "steps" .= map stepJson (ssSteps state)
  180   , "providers" .= map cardJson (ssProviders state)
  181   ]
  182   where
  183     stepJson step = object
  184       [ "id" .= stId step
  185       , "required" .= stRequired step
  186       , "status" .= statusText (stStatus step)
  187       , "detail" .= stDetail step
  188       , "actions" .= stActions step
  189       ]
  190     cardJson card = object
  191       [ "id" .= pcId card
  192       , "label" .= pcLabel card
  193       , "configured" .= pcConfigured card
  194       , "requires_base_url" .= pcRequiresBaseUrl card
  195       , "requires_name" .= pcRequiresName card
  196       , "isDefault" .= pcIsDefault card
  197       , "default_model" .= pcDefaultModel card
  198       , "models" .= pcModels card
  199       ]
  200 
  201 statusText :: StepStatus -> Text
  202 statusText status = case status of
  203   StepReady -> "ready"
  204   StepPending -> "pending"
  205   StepBlocked -> "blocked"
  206 
  207 -- | A config diagnostic is summarised to its first line and capped: the raw
  208 -- tomland rendering is multi-line, and a detail row is one sentence.
  209 summariseConfigProblem :: Text -> Text
  210 summariseConfigProblem raw =
  211   let firstLine = case T.lines raw of
  212         [] -> "unparseable"
  213         (line:_) -> line
  214   in T.take 200 (T.strip firstLine)

  220 handleVaultCreate :: SetupEnv -> Request -> IO Response
  221 handleVaultCreate env req
  222   | not (isJsonRequest req) =
  223       pure (setupErr status415 "content type must be application/json" Nothing)
  224   | otherwise = do
  225       body <- collectBodyLimited maxSetupBodyBytes req
  226       case body of
  227         Left () -> pure (setupErr status413 "request body exceeds 8 KiB" Nothing)
  228         Right raw -> case A.eitherDecodeStrict' raw of
  229           Left _ -> pure (setupErr status400 "request body is not valid JSON" Nothing)
  230           Right value -> do
  231             parsed <- parseVaultSpec value
  232             case parsed of
  233               Left err -> pure (provisionErrorResponse err)
  234               Right spec -> do
  235                 outcome <- provisionVault
  236                   (sePaths env) (seHandleRef env) (sdProvision (seDeps env)) spec
  237                 pure $ case outcome of
  238                   Left err -> provisionErrorResponse err
  239                   Right done -> setupJsonNoStore status201 (object
  240                     [ "recipient" .= poRecipient done
  241                     , "key_type" .= poKeyType done
  242                     ])

  244 -- | The status mapping the design fixes (§9.2). 409 for "someone else already
  245 -- holds this state", 422 for "your environment cannot do this", 500 for "we
  246 -- could not write it".
  247 provisionErrorResponse :: ProvisionError -> Response
  248 provisionErrorResponse err =
  249   let (reason, mHint) = renderProvisionError err
  250   in setupErr (statusForProvisionError err) reason mHint
  251 
  252 statusForProvisionError :: ProvisionError -> Status
  253 statusForProvisionError err = case err of
  254   ProvisionInvalidInput _      -> status400
  255   ProvisionAlreadyExists       -> status409
  256   ProvisionInProgress          -> status409
  257   ProvisionKeygenMissing _     -> status422
  258   ProvisionNeedsTty _          -> status422
  259   ProvisionRoundTripFailed _   -> status422
  260   ProvisionKeygenFailed _      -> status422
  261   ProvisionVaultInitFailed     -> status500
  262   ProvisionVaultUnreadable     -> status500
  263   ProvisionConfigWriteFailed _ -> status500

  269 -- | Open (and, in @startup@ mode, unlock) the vault that is already configured
  270 -- on disk, under the same setup lock provisioning uses — a reopen racing a
  271 -- create would otherwise open a half-written file.
  272 --
  273 -- It cannot call @Seal.Command.Serve@'s @tryOpenVault@ (import cycle), so this
  274 -- is the same short composition spelled out: load → resolveEncryptor →
  275 -- openVault → vhUnlock (startup) → writeIORef.
  276 handleVaultReopen :: SetupEnv -> IO Response
  277 handleVaultReopen env = do
  278   let lockPath = pdLockPath (sdProvision (seDeps env))
  279   result <- withSetupLock lockPath (reopenAction env)
  280   pure (either provisionErrorResponse id result)
  281 
  282 reopenAction :: SetupEnv -> IO (Either ProvisionError Response)
  283 reopenAction env = do
  284   security <- loadSecurityConfig (seSecurityPath env)
  285   case security of
  286     Left _ ->
  287       pure (Left (ProvisionConfigWriteFailed "security.toml could not be read"))
  288     Right sec -> case (scVaultRecipient sec, scVaultIdentity sec) of
  289       (Just recipient, Just identity) ->
  290         reopenConfigured env sec recipient identity
  291       _ -> pure (Right (setupErr status409
  292         "no vault key is recorded"
  293         (Just "run /vault setup in `seal tui`, then Re-check")))
  294 
  295 reopenConfigured
  296   :: SetupEnv -> SecurityConfig -> Text -> Text
  297   -> IO (Either ProvisionError Response)
  298 reopenConfigured env sec recipient identity = do
  299   let vaultPath = maybe (vaultFilePath (sePaths env)) T.unpack (scVaultPath sec)
  300       keyType = fromMaybe "unknown" (scVaultKeyType sec)
  301       mode = parseUnlockMode (scVaultUnlock sec)
  302   exists <- doesFileExist vaultPath
  303   if not exists
  304     then pure (Right (setupErr status409
  305       "the vault file does not exist yet"
  306       (Just "use the \"resume\" backend to create it")))
  307     else do
  308       -- The same seam production resolves through ('mkAgeEncryptor'): it
  309       -- validates both credentials and stays injectable in tests.
  310       encryptor <- seMk (pdEncryptor (sdProvision (seDeps env)))
  311         (AgeRecipient recipient) (AgeIdentity identity)
  312       case encryptor of
  313         Left _ -> pure (Right (setupErr status422
  314           "the recorded vault key cannot be loaded"
  315           (Just "restore the identity file named in ~/.seal/security.toml, or re-run /vault setup in `seal tui`")))
  316         Right enc -> do
  317           handle <- openVault
  318             (VaultConfig { vcPath = vaultPath, vcKeyType = keyType, vcUnlock = mode })
  319             enc
  320           -- startup mode is the only one that is genuinely locked in-process.
  321           when (mode == UnlockStartup) (void (vhUnlock handle))
  322           listResult <- vhList handle
  323           case listResult of
  324             Left _ -> pure (Right (setupErr status422
  325               "the vault file cannot be opened with the recorded key"
  326               (Just "if it holds no secrets yet: mv ~/.seal/config/vault/vault.age \
  327                     \~/.seal/config/vault/vault.age.bak, then Re-check; otherwise \
  328                     \restore the matching identity file")))
  329             Right _ -> do
  330               writeIORef (seHandleRef env) (Just handle)
  331               pure (Right (setupJson status200 (object
  332                 [ "opened" .= True
  333                 , "key_type" .= keyType
  334                 , "locked" .= (False :: Bool)
  335                 ])))

  341 -- | Ping the provider's *stored* configuration. A failed ping is not an HTTP
  342 -- error (the browser renders it next to the field); only an unknown provider
  343 -- (400) or a vault that is not open (500) is.
  344 handleProviderTest :: SetupEnv -> Text -> Request -> IO Response
  345 handleProviderTest env rawName req = do
  346   runtime <- loadRuntimeConfig (seConfigPath env)
  347   let cfg = fromRight defaultRuntimeConfig runtime
  348       known = isJust (parseProvider rawName)
  349                 || rawName `elem` customProviderNames cfg
  350   if not known
  351     then pure (setupErr status400
  352       ("unknown provider: " <> rawName)
  353       (Just "known providers: anthropic, openrouter, ollama, or a configured OpenAI-compatible instance"))
  354     else do
  355       mhandle <- readIORef (seHandleRef env)
  356       case mhandle of
  357         -- The design decides the vault-unavailable case by reading the handle
  358         -- first, never by matching on the resolver's error text.
  359         Nothing -> pure (setupErr status500
  360           "the vault is not open"
  361           (Just "create the vault (or reopen it) before testing a provider"))
  362         Just _ -> do
  363           model <- testModel env rawName req
  364           resolved <- sdResolveProvider (seDeps env) rawName model
  365           case resolved of
  366             Left err -> pure (pingFailed err)
  367             Right provider -> do
  368               -- What this can prove, precisely: the credential resolves into
  369               -- a working provider object, and (for providers whose
  370               -- listModels actually calls the network — Ollama does,
  371               -- Anthropic returns its static catalogue) that call succeeded.
  372               -- It is not a completion round-trip: that would spend tokens and
  373               -- take seconds on every click. The copy says "credentials
  374               -- resolved" rather than "reachable" for that reason.
  375               listed <- listSome provider
  376               pure $ case listed of
  377                 Left err -> pingFailed err
  378                 Right models -> setupJson status200 (object
  379                   [ "ok" .= True
  380                   , "detail" .= ("credentials resolved" :: Text)
  381                   , "models" .= length models
  382                   ])
  383 
  384 -- | A failed ping is not an HTTP error: the wizard renders it inline next to
  385 -- the field (design §9.4).
  386 pingFailed :: Text -> Response
  387 pingFailed err = setupJson status200 (object
  388   [ "ok" .= False
  389   , "error" .= err
  390   ])

  392 -- | The model the ping uses: the request's @model@ when given, else what a new
  393 -- session would use (global default, then the provider's configured default,
  394 -- then the provider's hardcoded default).
  395 testModel :: SetupEnv -> Text -> Request -> IO ModelId
  396 testModel env name req = do
  397   raw <- collectBodyLimited maxSetupBodyBytes req
  398   let fromBody = case raw of
  399         Right bytes -> case A.eitherDecodeStrict' bytes of
  400           Right (A.Object o) -> case KeyMap.lookup (Key.fromText "model") o of
  401             Just (A.String m) | not (T.null (T.strip m)) -> Just (T.strip m)
  402             _ -> Nothing
  403           _ -> Nothing
  404         Left () -> Nothing
  405   case fromBody of
  406     Just m -> pure (ModelId m)
  407     Nothing -> do
  408       runtime <- loadRuntimeConfig (seConfigPath env)
  409       let cfg = fromRight defaultRuntimeConfig runtime
  410           ModelId fallback = resolveDefaultModelForName cfg name
  411       pure $ ModelId $ fromMaybe fallback (rcDefaultModel cfg)

  417 -- | Setup request bodies are small JSON documents (a backend tag, a recipient,
  418 -- a path). 8 KiB is two orders of magnitude above the largest legitimate body
  419 -- and keeps a hostile peer from buffering a gigabyte.
  420 maxSetupBodyBytes :: Int
  421 maxSetupBodyBytes = 8192
  422 
  423 -- | Read a request body, counting bytes as they arrive.
  424 --
  425 -- 'Left' means the body exceeded the limit; the caller answers @413@. Reading
  426 -- stops at the cap, so the excess is never buffered (the reason to prefer this
  427 -- over a @Content-Length@ check alone, which a chunked request can omit).
  428 collectBodyLimited :: Int -> Request -> IO (Either () ByteString)
  429 collectBodyLimited limit req = go 0 []
  430   where
  431     go !bytes acc = do
  432       chunk <- getRequestBodyChunk req
  433       if BC.null chunk
  434         then pure (Right (BS.concat (reverse acc)))
  435         else
  436           let bytes' = bytes + BS.length chunk
  437           in if bytes' > limit
  438                then pure (Left ())
  439                else go bytes' (chunk : acc)
  440 
  441 isJsonRequest :: Request -> Bool
  442 isJsonRequest req = case lookup hContentType (requestHeaders req) of
  443   Nothing -> False
  444   Just raw ->
  445     let mediaType = T.toCaseFold (T.strip (T.takeWhile (/= ';') (decodeHeader raw)))
  446     in mediaType == "application/json"
  447   where
  448     decodeHeader = TE.decodeUtf8Lenient

  450 -- | A JSON response for a setup handler (no CORS grant, matching the rest of
  451 -- the mutation surface after the guard landed).
  452 setupJson :: Status -> Value -> Response
  453 setupJson st = responseLBS st [jsonHeader] . A.encode
  454 
  455 -- | A JSON response that must never be cached, and never sniffed: the wizard's
  456 -- state describes system configuration, and a stale copy would tell the user to
  457 -- redo a finished step.
  458 setupJsonNoStore :: Status -> Value -> Response
  459 setupJsonNoStore st =
  460   responseLBS st [jsonHeader, noStoreHeader, noSniffHeader] . A.encode
  461 
  462 -- | The shared error envelope: @{"error": …, "hint": …?}@. The hint is a
  463 -- copy-pasteable next step (a command, a path to repair); it never carries key
  464 -- material or an identity file's contents.
  465 setupErr :: Status -> Text -> Maybe Text -> Response
  466 setupErr st err mHint = setupJson st $ object $
  467   ["error" .= err] <> maybe [] (\h -> ["hint" .= h]) mHint
  468 
  469 jsonHeader :: Header
  470 jsonHeader = (mkHN "Content-Type", "application/json")
  471 
  472 noStoreHeader :: Header
  473 noStoreHeader = (mkHN "Cache-Control", "no-store")
  474 
  475 noSniffHeader :: Header
  476 noSniffHeader = (mkHN "X-Content-Type-Options", "nosniff")
  477 
  478 mkHN :: String -> HeaderName
  479 mkHN = CI.mk . BC.pack

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | A validated path to an age identity file.
    3 --
    4 -- The value travels to a subprocess argv (@age --decrypt --identity=\<path\>@)
    5 -- and is stored in @security.toml@, so it is a smart-constructed newtype rather
    6 -- than raw 'FilePath' (AGENTS.md invariant 2). The constructor is deliberately
    7 -- not exported: the ONLY way to obtain a 'VaultIdentityPath' is
    8 -- 'mkVaultIdentityPath', so no code path can hand @age@ an unvalidated path.
    9 --
   10 -- This module imports neither "Seal.Vault.Provision" nor "Seal.Vault.Backend"
   11 -- nor "Seal.Config.Security", which is what lets all three depend on it without
   12 -- an import cycle.
   13 module Seal.Vault.IdentityPath
   14   ( VaultIdentityPath
   15   , mkVaultIdentityPath
   16   , vaultIdentityPathText
   17   , vaultIdentityPathDisplay
   18   ) where
   19 
   20 import Control.Exception (IOException, try)
   21 import Data.Text (Text)
   22 import Data.Text qualified as T
   23 import System.Directory (doesFileExist)
   24 import System.FilePath (isAbsolute, takeFileName)
   25 
   26 -- | An absolute, existing, regular, readable file path with no control
   27 -- characters and no leading @-@ (which a subprocess could mistake for a flag).
   28 newtype VaultIdentityPath = VaultIdentityPath FilePath
   29   deriving stock (Eq, Show)
   30 
   31 -- | Validate a user-supplied path.
   32 --
   33 -- Rejections, in order: empty, a control character (including newline and NUL),
   34 -- a leading @-@ (option injection), a relative path, a path that does not
   35 -- exist, a path that exists but is not a regular file.
   36 --
   37 -- The path is NOT canonicalised: the operator named it, and storing the exact
   38 -- string they gave keeps @security.toml@ legible. Symlinks are followed only
   39 -- for the existence/regular-file checks.
   40 mkVaultIdentityPath :: FilePath -> IO (Either Text VaultIdentityPath)
   41 mkVaultIdentityPath raw
   42   | null raw = pure (Left "identity path is empty")
   43   | any isControlChar raw =
   44       pure (Left "identity path contains a control character")
   45   | "-" `isPrefix` raw =
   46       pure (Left "identity path must not start with '-'")
   47   | not (isAbsolute raw) =
   48       pure (Left "identity path must be absolute")
   49   | otherwise = do
   50       exists <- doesFileExist raw
   51       if not exists
   52         then pure (Left ("identity file does not exist: " <> T.pack raw))
   53         else do
   54           -- doesFileExist is False for directories, so this also rejects them.
   55           regular <- try @IOException (doesFileExist raw)
   56           pure $ case regular of
   57             Left err  -> Left ("identity path is not readable: " <> T.pack (show err))
   58             Right ok
   59               | ok        -> Right (VaultIdentityPath raw)
   60               | otherwise -> Left ("identity path is not a regular file: " <> T.pack raw)
   61   where
   62     isPrefix prefix s = take (length prefix) s == prefix
   63     -- Anything below space, plus DEL. Tab is rejected too: it never appears in
   64     -- a legitimate path and hiding one in a prompt or a log is a trick.
   65     isControlChar c = c < ' ' || c == '\DEL'
   66 
   67 -- | The raw path, for the one place that must hand it to @age@ or store it.
   68 vaultIdentityPathText :: VaultIdentityPath -> FilePath
   69 vaultIdentityPathText (VaultIdentityPath p) = p
   70 
   71 -- | A short, log-safe rendering: the file name only, never the directory.
   72 -- Used in user-facing messages so an error never discloses the operator's
   73 -- home layout.
   74 vaultIdentityPathDisplay :: VaultIdentityPath -> Text
   75 vaultIdentityPathDisplay (VaultIdentityPath p) = T.pack (takeFileName p)

   59 -- ---------------------------------------------------------------------------
   60 -- ChannelId — the validated [channels.<id>] key + URL capture + vault-key
   61 -- component. Smart-constructed: non-empty, no leading dash (option-injection
   62 -- defense), charset [A-Za-z0-9_-]. Constructor exported (like RepoId);
   63 -- mkChannelId is used at trust boundaries (POST body, URL capture, TOML key
   64 -- decode via keyChannelIdBiMap).
   65 -- ---------------------------------------------------------------------------
   66 
   67 newtype ChannelId = ChannelId Text
   68   deriving stock (Eq, Ord, Show)
   69 
   70 -- | @[A-Za-z0-9_-]+@, non-empty, no leading dash. Mirrors the 'RepoId'
   71 -- charset but forbids a leading @-@ (option-injection defense).
   72 isValidChannelId :: Text -> Bool
   73 isValidChannelId t =
   74   not (T.null t)
   75     && T.head t /= '-'
   76     && T.all validIdChar t
   77   where
   78     validIdChar :: Char -> Bool
   79     validIdChar c = c `Set.member` idChars
   80     idChars :: Set Char
   81     idChars = Set.fromList
   82       $ ['A'..'Z'] <> ['a'..'z'] <> ['0'..'9'] <> "_-"
   83 
   84 -- | Smart-construct a 'ChannelId'. Rejects empty, leading dash, and any
   85 -- character outside @[A-Za-z0-9_-]@.
   86 mkChannelId :: Text -> Either Text ChannelId
   87 mkChannelId t
   88   | isValidChannelId t = Right (ChannelId t)
   89   | otherwise          = Left ("invalid channel id: " <> T.pack (show t))
   90 
   91 channelIdText :: ChannelId -> Text
   92 channelIdText (ChannelId t) = t

   94 -- ---------------------------------------------------------------------------
   95 -- ChannelConfig — the [channels.<id>] config section
   96 -- ---------------------------------------------------------------------------
   97 
   98 -- | The @[channels.\<id\>]@ config section. All fields optional at the file
   99 -- level except 'ccKind' (required, restricted to Signal/Telegram by the
  100 -- codec). 'ccAccount' is Signal-specific; 'ccToken' is the Telegram
  101 -- config-level token fallback (mirrors 'tcToken' in the legacy
  102 -- 'TelegramConfig'); the vault token takes precedence at resolution time.
  103 data ChannelConfig = ChannelConfig
  104   { ccKind           :: ChannelKind
  105   , ccLabel          :: Maybe Text
  106   , ccAccount        :: Maybe Text
  107   , ccToken          :: Maybe Text
  108   , ccTextChunkLimit :: Maybe Int
  109   , ccAllowFrom      :: AllowList Text
  110   } deriving stock (Eq, Show)
  111 
  112 -- ---------------------------------------------------------------------------
  113 -- ResolvedChannel — the validated, ready-to-fork channel
  114 -- ---------------------------------------------------------------------------
  115 
  116 -- | A channel config that has been validated and smart-constructed into a
  117 -- ready-to-fork form. 'ResolvedSignal' carries the validated 'SignalAccount';
  118 -- 'ResolvedTelegram' carries the validated 'TelegramToken'. Both carry the
  119 -- chunk limit and 'AllowList' 'UserId' for the sender check.
  120 data ResolvedChannel
  121   = ResolvedSignal SignalAccount Int (AllowList UserId)
  122   | ResolvedTelegram TelegramToken Int (AllowList UserId)
  123   deriving stock (Eq, Show)

  125 -- ---------------------------------------------------------------------------
  126 -- TOML codec (bidirectional, restricted kind)
  127 -- ---------------------------------------------------------------------------
  128 
  129 -- | Restrict a TOML @kind@ text value to only @signal@ or @telegram@. All
  130 -- other 'ChannelKind' constructors (Cli, Web, Background, Other) and unknown
  131 -- strings are rejected — 'ResolvedChannel' only has Signal/Telegram arms.
  132 restrictChannelKind :: Text -> Either Text ChannelKind
  133 restrictChannelKind t = case channelKindFromText t of
  134   Just Signal   -> Right Signal
  135   Just Telegram -> Right Telegram
  136   Just _        -> Left ("invalid channel kind: " <> t <> " (only signal/telegram allowed)")
  137   Nothing       -> Left ("unknown channel kind: " <> t)
  138 
  139 -- | Bidirectional tomland codec for a single @[channels.\<id\>]@ table. The
  140 -- @kind@ field is restricted to @signal@/@telegram@ via 'restrictChannelKind'
  141 -- (fail-closed on cli/web/background/other). @allow_from@ uses the shared
  142 -- 'allowListCodec' from 'Seal.Core.AllowList'.
  143 channelConfigCodec :: Toml.TomlCodec ChannelConfig
  144 channelConfigCodec = ChannelConfig
  145   <$> Toml.textBy channelKindToText restrictChannelKind "kind"  .= ccKind
  146   <*> Toml.dioptional (Toml.text "label")                      .= ccLabel
  147   <*> Toml.dioptional (Toml.text "account")                    .= ccAccount
  148   <*> Toml.dioptional (Toml.text "token")                      .= ccToken
  149   <*> Toml.dioptional (Toml.int "text_chunk_limit")            .= ccTextChunkLimit
  150   <*> allowListCodec                                            .= ccAllowFrom
  151 
  152 -- | Bidirectional 'TomlBiMap' between TOML 'Key's and 'ChannelId's. @forward@
  153 -- (Key → ChannelId) runs 'mkChannelId' on the rendered key text; a bad id is
  154 -- a 'Failure' (fail-closed). @backward@ (ChannelId → Key) re-parses the id
  155 -- text (guaranteed valid by 'mkChannelId'). Mirrors 'keyRepoBiMap'.
  156 keyChannelIdBiMap :: TomlBiMap Key ChannelId
  157 keyChannelIdBiMap = BiMap
  158   { forward  = first (ArbitraryError . ("invalid channel id key: " <>) . T.pack . show)
  159                 . mkChannelId . prettyKey
  160   , backward = first (ArbitraryError . unTomlParseError) . parseKey . channelIdText
  161   }

  163 -- ---------------------------------------------------------------------------
  164 -- Vault keys for multi-channel Telegram
  165 -- ---------------------------------------------------------------------------
  166 
  167 -- | The vault key for a Telegram channel's bot token. The legacy id
  168 -- @telegram@ maps to the unsuffixed key @TELEGRAM_BOT_TOKEN@ (the key the
  169 -- CLI wizard writes), preserving zero-migration. All other ids use the
  170 -- suffixed form @TELEGRAM_BOT_TOKEN_\<id\>@.
  171 telegramVaultKeyFor :: ChannelId -> Text
  172 telegramVaultKeyFor cid =
  173   if channelIdText cid == "telegram"
  174     then "TELEGRAM_BOT_TOKEN"
  175     else "TELEGRAM_BOT_TOKEN_" <> channelIdText cid

  177 -- ---------------------------------------------------------------------------
  178 -- Resolution (pure) — config section + pre-fetched vault token -> ResolvedChannel
  179 -- ---------------------------------------------------------------------------
  180 
  181 -- | Resolve a 'ChannelConfig' into a 'ResolvedChannel'. Pure — takes a
  182 -- pre-fetched vault token (if any) rather than a 'VaultHandle', so it is
  183 -- unit-testable without vault I/O. The caller does the @vhGet@ and passes
  184 -- the result here.
  185 --
  186 -- * Signal: delegates to 'resolveSignalConfig' with the account from
  187 --   'ccAccount'. No vault (account is in config). Fails if 'ccAccount' is
  188 --   unset.
  189 -- * Telegram: the vault token (pre-fetched by the caller) takes precedence;
  190 --   if absent, falls back to 'ccToken' (the config-level token, preserved
  191 --   from legacy 'tcToken'); if both absent, fails. Delegates to
  192 --   'resolveTelegramConfig'.
  193 -- * Other kinds: rejected (only Signal/Telegram are valid channel types).
  194 resolveChannelConfig
  195   :: ChannelId
  196   -> ChannelConfig
  197   -> Maybe Text
  198   -> Either Text ResolvedChannel
  199 resolveChannelConfig _ cc mVaultToken =
  200   case ccKind cc of
  201     Signal -> do
  202       (acct, limit, allow) <- resolveSignalConfig (Just (signalConfigFromChannel cc)) Nothing
  203       Right (ResolvedSignal acct limit allow)
  204     Telegram -> do
  205       (token, limit, allow) <- resolveTelegramConfig (Just (telegramConfigFromChannel cc)) mVaultToken
  206       Right (ResolvedTelegram token limit allow)
  207     _ -> Left "unsupported channel kind (only signal/telegram allowed)"
  208 
  209 -- | Build a 'SignalConfig' from a 'ChannelConfig' for delegation to
  210 -- 'resolveSignalConfig'. Maps 'ccAccount'/'ccTextChunkLimit'/'ccAllowFrom'
  211 -- onto the legacy section shape.
  212 signalConfigFromChannel :: ChannelConfig -> SignalConfig
  213 signalConfigFromChannel cc = defaultSignalConfig
  214   { scAccount        = ccAccount cc
  215   , scTextChunkLimit = ccTextChunkLimit cc
  216   , scAllowFrom      = ccAllowFrom cc
  217   }
  218 
  219 -- | Build a 'TelegramConfig' from a 'ChannelConfig' for delegation to
  220 -- 'resolveTelegramConfig'. Maps 'ccToken'/'ccTextChunkLimit'/'ccAllowFrom'
  221 -- onto the legacy section shape.
  222 telegramConfigFromChannel :: ChannelConfig -> TelegramConfig
  223 telegramConfigFromChannel cc = defaultTelegramConfig
  224   { tcToken         = ccToken cc
  225   , tcTextChunkLimit = ccTextChunkLimit cc
  226   , tcAllowFrom     = ccAllowFrom cc
  227   }

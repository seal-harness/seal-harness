    2 -- | Validated argv newtypes (Invariant 2 — type-guaranteed argument
    3 -- sanitization). Every value derived from user/LLM input that reaches a
    4 -- subprocess argv must be carried by one of these validated,
    5 -- smart-constructed newtypes — never raw 'Text'/'String'. The exec
    6 -- wrappers accept ONLY these types in their signatures, so passing
    7 -- unsanitized input fails to compile.

    9 -- Defense against option injection is built into every smart constructor:

   38 -- | A single argv token (a program argument). Rejects leading-dash (option
   39 -- injection), NUL, newlines, and control chars.
   40 newtype ShellArg = ShellArg Text
   41   deriving stock (Eq, Show)
   42 
   43 mkShellArg :: Text -> Either Text ShellArg
   44 mkShellArg t
   45   | T.null t            = Left "shell arg is empty"
   46   | T.head t == '-'     = Left "shell arg must not start with '-' (option injection)"
   47   | T.any isArgBad t    = Left "shell arg contains NUL, newline, or control char"
   48   | otherwise           = Right (ShellArg t)
   49   where
   50     isArgBad c = c == '\0' || c == '\n' || isControl c
   51 
   52 textShellArg :: ShellArg -> Text
   53 textShellArg (ShellArg t) = t

   55 -- | A shell command string — the single argument passed to @\/bin\/sh -c@.
   56 -- Rejects NUL (which would terminate the C-string early). Does NOT reject
   57 -- leading dash (a command may begin with @-@; the executor passes it as a
   58 -- single arg to @-c@, not as a flag). Does NOT reject newlines (a shell
   59 -- command legitimately spans lines).
   60 newtype ShellCommand = ShellCommand Text
   61   deriving stock (Eq, Show)
   62 
   63 mkShellCommand :: Text -> Either Text ShellCommand
   64 mkShellCommand t
   65   | T.null t            = Left "shell command is empty"
   66   | T.any (== '\0') t   = Left "shell command contains NUL"
   67   | otherwise           = Right (ShellCommand t)
   68 
   69 textShellCommand :: ShellCommand -> Text
   70 textShellCommand (ShellCommand t) = t

   72 -- | A binary name (e.g. @python3@, @node@, @ls@). The executor resolves
   73 -- it on PATH via 'System.Process.proc' (RawCommand — no shell), so the
   74 -- value never passes through a shell. Minimal validation: reject empty
   75 -- and NUL (which would truncate the C-string). Leading dashes are
   76 -- permitted (a binary *may* legitimately begin with @-@), and the
   77 -- argv-passing model means leading-dash args are passed as argv tokens,
   78 -- not flags to a shell. Path separators are permitted (the binary may be
   79 -- an absolute or relative path). Control chars other than NUL are
   80 -- permitted (the executor passes them verbatim through argv).
   81 newtype BinName = BinName Text
   82   deriving stock (Eq, Show)
   83

   86   | T.null t          = Left "binary name is empty"
   87   | T.any (== '\0') t = Left "binary name contains NUL"
   88   | otherwise         = Right (BinName t)
   89 
   90 textBinName :: BinName -> Text
   91 textBinName (BinName t) = t

  131 -- | A git branch name destined for a subprocess argv — typically
  132 -- @git clone --branch=<b>@ or @git fetch origin <b>:refs\/heads\/<b>@.
  133 --
  134 -- The constructor is deliberately NOT exported (the same shape as
  135 -- 'Seal.Core.Types.SessionId'): a 'BranchName' can only be produced by
  136 -- 'mkBranchName', so unvalidated 'Text' cannot reach the exec wrappers.
  137 --
  138 -- Validation runs on the input with surrounding whitespace stripped (so
  139 -- @\"  dev  \"@ becomes @\"dev\"@) and rejects:
  140 --
  141 --   * the empty name;
  142 --   * @HEAD@, which names a commit rather than a branch;
  143 --   * a leading @-@ (option injection, as in 'mkShellArg');
  144 --   * characters git forbids in a refname (space, @~@, @^@, @:@, @?@, @*@,
  145 --     @[@, backslash) and any control character or NUL;
  146 --   * @..@ and @{@-sequences, a leading or trailing @\/@, @\/\/@, a trailing
  147 --     @.@, and a @.lock@ suffix.
  148 --
  149 -- Error messages are distinct per rule; callers prefix them with
  150 -- @\"invalid branch: \"@.
  151 newtype BranchName = BranchName Text
  152   deriving stock (Eq, Show)
  153 
  154 mkBranchName :: Text -> Either Text BranchName
  155 mkBranchName raw
  156   | T.null t = Left "branch name is empty"
  157   | t == "HEAD" = Left "branch name must not be HEAD (HEAD names a commit, not a branch)"
  158   | T.head t == '-' = Left "branch name must not start with '-' (option injection)"
  159   | T.any isRefnameBad t = Left "branch name contains a character git forbids in refnames (space, ~, ^, :, ?, *, [, backslash, control char, or NUL)"
  160   | ".." `T.isInfixOf` t = Left "branch name must not contain '..'"
  161   | "@{" `T.isInfixOf` t = Left "branch name must not contain '@{'"
  162   | "/" `T.isPrefixOf` t = Left "branch name must not start with '/'"
  163   | "/" `T.isSuffixOf` t = Left "branch name must not end with '/'"
  164   | "//" `T.isInfixOf` t = Left "branch name must not contain '//'"
  165   | "." `T.isSuffixOf` t = Left "branch name must not end with '.'"
  166   | ".lock" `T.isSuffixOf` t = Left "branch name must not end with '.lock'"
  167   | otherwise = Right (BranchName t)
  168   where
  169     t = T.strip raw
  170     isRefnameBad c =
  171       c `elem` (" ~^:?*[\\" :: String) || c == '\0' || isControl c
  172 
  173 branchNameText :: BranchName -> Text
  174 branchNameText (BranchName t) = t

    3 -- | The Telegram channel: a 'TelegramChannel' record (resolved allow-list +
    4 -- chunk limit + inbox 'TQueue' + transport + last-sender 'IORef' + logger)

    6 -- allow-lists the sender, and pushes @(MessageSource, body)@ to the inbox;
    7 -- sends are chunked via 'Seal.Channels.Telegram.Transport.chunkMessage' to the

    9 -- "Seal.Channels.Signal" in structure.

   36 -- | The live Telegram channel state.
   37 data TelegramChannel = TelegramChannel
   38   { tcgAllowList   :: AllowList UserId   -- ^ resolved sender allow-list
   39   , tcgChunkLimit  :: Int                -- ^ chunk limit for sends
   40   , tcgInbox       :: TQueue (MessageSource, Text)
   41   , tcgTransport   :: TelegramTransport
   42   , tcgLastChat    :: IORef (Maybe Text)  -- ^ last sender's chat id (for replies)
   43   , tcgReaderAlive :: IORef Bool
   44     -- ^ 'True' while the reader thread is running. 'chReceive' stops
   45     -- returning once the inbox drains and this is 'False'.
   46   , tcgLogger      :: SealLogger
   47     -- ^ The shared logger for structured katip logging.
   48   }
   49 
   50 instance Channel TelegramChannel where
   51   toHandle ch = ChannelHandle
   52     { chLabel       = "telegram"
   53     , chSend         = sendChunked ch
   54     , chSendError    = \t -> sendChunked ch ("error: " <> t)
   55     , chSendChunk    = sendRaw ch
   56     , chPrompt       = \_ -> pure (Left Deferred)  -- Telegram can't answer inline
   57     , chPromptSecret = \_ -> pure (Left Deferred)
   58     , chStreaming    = False  -- per-token messages flood the chat; send accumulated text once
   59     , chReadSecret   = pure Nothing
   60     , chReceive      = receiveFromInbox ch
   61     , chLastChatId   = readIORef (tcgLastChat ch)
   62     }

   64 -- | Run the reader thread with cleanup. Spawns a background thread that
   65 -- loops 'tgReceive', allow-lists the sender, and pushes
   66 -- @(MessageSource, body)@ to 'tcgInbox'. On transport close or exception,
   67 -- the thread exits. The action runs with the channel; on exit the transport
   68 -- is closed and the reader is killed. Mirrors 'withSignalChannel'.
   69 withTelegramChannel
   70   :: (AllowList UserId, Int)
   71   -> TelegramTransport
   72   -> SealLogger
   73   -> (TelegramChannel -> IO a)
   74   -> IO a
   75 withTelegramChannel (allow, chunkLimit) transport logger action =
   76   bracket before after (action . snd)
   77   where
   78     before = do
   79       inbox   <- newTQueueIO
   80       lastChat <- newIORef Nothing
   81       alive   <- newIORef True
   82       let ch = TelegramChannel
   83             { tcgAllowList   = allow
   84             , tcgChunkLimit  = chunkLimit
   85             , tcgInbox       = inbox
   86             , tcgTransport   = transport
   87             , tcgLastChat    = lastChat
   88             , tcgReaderAlive = alive
   89             , tcgLogger      = logger
   90             }
   91       tid <- forkIO (readerLoop ch)
   92       pure (tid, ch)
   93     after (tid, _ch) = do
   94       killThread tid
   95       tgClose transport
   96 
   97 -- | The background reader: loop 'tgReceive', allow-list the sender, and push
   98 -- to the inbox. A malformed update or a non-allow-listed sender is logged
   99 -- via 'logIO' and dropped (not fatal). Exits when 'tgReceive' returns 'Left'
  100 -- (the transport is closing). 'ThreadKilled' is logged at 'InfoS' (normal
  101 -- shutdown); other 'AsyncException's at 'WarningS'; synchronous exceptions
  102 -- at 'ErrorS'.
  103 readerLoop :: TelegramChannel -> IO ()
  104 readerLoop ch = go
  105   where
  106     logger = tcgLogger ch
  107     go = do
  108       eUpd <- try @SomeException (tgReceive (tcgTransport ch))
  109       case eUpd of
  110         Left e -> do
  111           case (fromException e :: Maybe AsyncException) of
  112             Just ThreadKilled ->
  113               logIO logger InfoS "reader thread stopped (channel shutting down)"
  114             Just _ ->
  115               logIO logger WarningS ("reader thread terminated by async exception: " <> ls (T.pack (show e)))
  116             Nothing ->
  117               logIO logger ErrorS ("reader thread exception: " <> ls (T.pack (show e)))
  118           writeIORef (tcgReaderAlive ch) False
  119         Right (Left err) -> do
  120           logIO logger ErrorS ("reader exiting: " <> ls err)
  121           writeIORef (tcgReaderAlive ch) False
  122         Right (Right upd) -> do
  123           let sender = tuSender upd
  124           if isAllowed sender (tcgAllowList ch)
  125             then do
  126               writeIORef (tcgLastChat ch) (Just (tuChatId upd))
  127               -- Acknowledge a callback_query (button tap) so the button's
  128               -- loading spinner dismisses, and remove the inline keyboard
  129               -- so the buttons can't be re-clicked. Both best-effort.
  130               case tuCallbackId upd of
  131                 Just cbId -> do
  132                   void (try @SomeException (tgAnswerCallback (tcgTransport ch) cbId))
  133                   case tuCallbackMessageId upd of
  134                     Just msgId -> void (try @SomeException (tgEditReplyMarkup (tcgTransport ch) (tuChatId upd) msgId))
  135                     Nothing    -> pure ()
  136                 Nothing   -> pure ()
  137               case mkMessageSource (tuConversationId upd) Telegram (Just sender) mempty of
  138                 Left err -> logIO logger WarningS ("MessageSource construction failed: " <> ls err)
  139                 Right ms -> atomically (writeTQueue (tcgInbox ch) (ms, tuBody upd))
  140             else logIO logger InfoS ("dropped non-allow-listed sender: " <> ls (userIdText sender))
  141           go

  143 -- | Send a message, chunked to 'tcgChunkLimit', addressed to the last chat.
  144 -- If no chat has been seen yet, the send is dropped with a warning log.
  145 sendChunked :: TelegramChannel -> Text -> IO ()
  146 sendChunked ch t = do
  147   mChat <- readIORef (tcgLastChat ch)
  148   case mChat of
  149     Nothing -> logIO (tcgLogger ch) WarningS "telegram: dropping send — no last chat yet"
  150     Just _  -> mapM_ (sendRaw ch) (chunkMessage (tcgChunkLimit ch) t)
  151 
  152 -- | Send one chunk verbatim to the last chat (no further splitting).
  153 sendRaw :: TelegramChannel -> Text -> IO ()
  154 sendRaw ch t = do
  155   mChat <- readIORef (tcgLastChat ch)
  156   case mChat of
  157     Nothing -> logIO (tcgLogger ch) WarningS "telegram: dropping chunk — no last chat yet"
  158     Just chatId -> tgSend (tcgTransport ch) chatId t

  160 -- | Pull the next @(MessageSource, body)@ from the inbox. Non-blocking:
  161 -- returns @(Nothing, "")@ when the inbox is empty AND the reader thread has
  162 -- exited. Returns @(Nothing, "")@ immediately in that state so the loop can
  163 -- terminate rather than block forever. Mirrors 'receiveFromInbox' in Signal.
  164 receiveFromInbox :: TelegramChannel -> IO (Maybe MessageSource, Text)
  165 receiveFromInbox ch = do
  166   mNext <- atomically (tryReadTQueue (tcgInbox ch))
  167   case mNext of
  168     Just (ms, body) -> pure (Just ms, body)
  169     Nothing -> do
  170       alive <- readIORef (tcgReaderAlive ch)
  171       if alive
  172         then do
  173           threadDelay 1000  -- 1ms
  174           receiveFromInbox ch
  175         else pure (Nothing, "")

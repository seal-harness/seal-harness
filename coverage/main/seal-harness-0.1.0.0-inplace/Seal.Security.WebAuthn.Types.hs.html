   14 import Data.ByteString (ByteString)
   15 import Data.Text (Text)
   16 import Data.Text qualified as T
   17 import Data.Time (UTCTime)
   18 
   19 -- | The operator's access-control decision, as recorded in
   20 -- @~\/.seal\/security.toml@.
   21 --
   22 -- 'AccessUnconfigured' is also the state of a @mode = \"passkey\"@ file whose
   23 -- credential list is empty — nobody could authenticate, so the gateway must be
   24 -- treatable as still awaiting setup. 'Seal.Config.Security' normalises that
   25 -- case.
   26 data AccessMode
   27   = AccessUnconfigured
   28   | AccessOpen
   29   | AccessPasskey
   30   deriving stock (Eq, Show)
   31 
   32 -- | The wire spelling of an 'AccessMode', as reported by
   33 -- @\/api\/auth\/status@ and consumed by the SPA.
   34 --
   35 -- This is deliberately NOT the config spelling: the config says @none@ (the
   36 -- operator's waiver), while the reported state says @open@ (what the gateway is
   37 -- actually doing). Keeping them distinct means "access control is off" can be
   38 -- stated in the API without the UI having to know that @none@ means open.
   39 accessModeText :: AccessMode -> Text
   40 accessModeText mode = case mode of
   41   AccessUnconfigured -> "unconfigured"
   42   AccessOpen         -> "open"
   43   AccessPasskey      -> "passkey"
   44 
   45 -- | Parse the config spelling. Anything other than @\"none\"@ or
   46 -- @\"passkey\"@ is rejected — fail-closed, so a typo can never silently mean
   47 -- "open".
   48 parseAccessMode :: Text -> Either Text AccessMode
   49 parseAccessMode t = case T.strip (T.toLower t) of
   50   "none"    -> Right AccessOpen
   51   "passkey" -> Right AccessPasskey
   52   other     -> Left ("invalid access_control.mode \"" <> other
   53                       <> "\" (expected \"passkey\" or \"none\")")
   54 
   55 -- | A P-256 public key as the 64-byte uncompressed @x || y@ pair taken from a
   56 -- COSE EC2 key. A newtype so callers cannot mistake it for the SEC1
   57 -- @0x04@-prefixed or DER forms.
   58 newtype CosePublicKey = CosePublicKey ByteString
   59   deriving stock (Eq, Show)
   60 
   61 -- | One enrolled passkey.
   62 data Credential = Credential
   63   { credId        :: ByteString
   64     -- ^ The raw WebAuthn credential id.
   65   , credPublicKey :: CosePublicKey
   66   , credSignCount :: Int
   67     -- ^ The last sign counter accepted. 0 means the authenticator does not
   68     -- implement one.
   69   , credLabel     :: Text
   70   , credCreatedAt :: UTCTime
   71   , credRpId      :: Text
   72     -- ^ The rpId this credential was created for. A credential only validates
   73     -- against its own rpId, so this is compared with the boot-resolved rpId to
   74     -- detect an orphaned credential.
   75   , credAddedVia  :: Maybe Text
   76     -- ^ How the credential was added — the @added_via@ vocabulary
   77     -- (@cli_token@, @cli_window@, @assertion@, @legacy@). 'Text' rather than
   78     -- 'Seal.Security.EnrollmentGrant.GrantSource' because that module imports
   79     -- this one; the spelling is the storage format either way, and the UI
   80     -- renders it as human copy.
   81   } deriving stock (Eq, Show)

  206 -- | Build the handle from an already-decided 'BootPlan'.
  207 --
  208 -- The plan is the whole input on purpose: the handle never reads
  209 -- @security.toml@ or @credentials.toml@ again, so editing either file under a
  210 -- running gateway cannot change what it trusts. Minting follows
  211 -- 'bpMintGrant' — nothing is minted when a usable credential exists, because
  212 -- there is already an authority the operator can present.
  213 newAccessControlHandle :: SealPaths -> BootPlan -> IO NewAccess
  214 newAccessControlHandle paths plan = do
  215   serviceToken <- generateToken 32
  216   now <- getCurrentTime
  217   let creds = bpCreds plan
  218       mOrigin = bpOrigin plan
  219       mRpId = bpRpId plan
  220   -- Boot never clobbers an outstanding grant (design §7.3, §16.9). The CLI
  221   -- mints while the daemon runs, so a restart that minted unconditionally would
  222   -- invalidate a token the operator is holding. Minting is also skipped
  223   -- entirely when a usable credential exists: an assertion is already the
  224   -- authority.
  225   (mCode, naBanner) <- case bpMintGrant plan of
  226     Nothing -> pure (Nothing, Nothing)
  227     Just source -> do
  228       outstanding <- loadGrantFile (enrollmentGrantPath paths)
  229       case outstanding of
  230         Just grant | grantLiveAt now grant ->
  231           pure (Nothing, Just (banner (grSource grant) now grant False))
  232         _ -> do
  233           minted <- enrollGrant paths source (defaultTtlFor source) True
  234           case minted of
  235             Right result -> do
  236               _ <- Audit.appendAudit paths now
  237                      (emptyAuditFields Audit.GrantMinted)
  238                        { afVia = Just (grSource (mgGrant result))
  239                        , afRemainingSeconds = Just (round (defaultTtlFor source))
  240                        }
  241               pure ()
  242             Left _ -> pure ()
  243           pure $ case minted of
  244             Left _ -> (Nothing, Nothing)
  245             Right result ->
  246               ( mgToken result
  247               , Just (banner (grSource (mgGrant result)) now (mgGrant result) True) )
  248   auditGate <- newAuditGate
  249   st <- newMVar AccessControlState
  250     { acsMode = bpMode plan
  251     , acsCredentials = creds
  252     , acsChallenges = Map.empty
  253     , acsSessions = Map.empty
  254     , acsSessionGrants = Map.empty
  255     , acsSetupCode = mCode
  256     , acsSetupHash = tokenKey <$> mCode
  257     , acsServiceToken = serviceToken
  258     , acsServiceHash = tokenKey serviceToken
  259     , acsOrigin = mOrigin
  260     , acsRpId = mRpId
  261     , acsChallengeTokens = challengeBurst
  262     , acsChallengeRefillAt = now
  263     , acsFailures = 0
  264     , acsRetryAfter = now
  265     }
  266   pure NewAccess
  267     { naHandle       = AccessControlHandle st paths auditGate
  268     , naToken        = mCode
  269     , naServiceToken = serviceToken
  270     , naOrigin       = mOrigin
  271     , naGrant        = naBanner
  272     }
  273   where
  274     banner source stamp grant fresh = GrantBanner
  275       { gbSource    = source
  276       , gbRemaining = diffUTCTime (grExpiresAt grant) stamp
  277       , gbFresh     = fresh
  278       }

  292 -- | The single authorization decision (design §7). Used by the HTTP middleware
  293 -- and by the WebSocket handshake.
  294 accessControlAuthorize :: AccessControlHandle -> Request -> IO AuthDecision
  295 accessControlAuthorize h req = do
  296   now <- getCurrentTime
  297   st <- withMVar (achState h) (pure . refillChallenges now)
  298   pure (decide now st)
  299   where
  300     path = pathInfo req
  301     decide now st
  302       | requestMethod req == methodOptions = AuthAllow
  303       | path == ["api", "health"] = AuthAllow
  304       | otherwise = case acsMode st of
  305           AccessOpen -> AuthAllow
  306           AccessUnconfigured
  307             | path `elem` unconfiguredAllow -> AuthAllow
  308             | hasServiceToken st req && not (isAuthPath path) -> AuthAllow
  309             | otherwise -> AuthSetupRequired
  310           AccessPasskey
  311             | path `elem` passkeyAllow -> AuthAllow
  312             | isRegisterPath path && isJustSetup st -> AuthAllow
  313             | hasServiceToken st req && not (isAuthPath path) -> AuthAllow
  314             | sessionValid now st (sessionKeyFromRequest req) -> AuthAllow
  315             | otherwise -> AuthUnauthorized

  318 unconfiguredAllow =

  346   | isAuthPath (pathInfo req) = authApp h req respond

  360 -- | The @\/api\/auth\/status@ payload (design §11): snake case, one @state@
  361 -- enum, and no secret in it.
  362 accessControlStatusJson :: AccessControlHandle -> Request -> IO Value
  363 accessControlStatusJson h req = do
  364   now <- getCurrentTime
  365   st <- withMVar (achState h) pure
  366   -- The file grant is read per request, never cached (design §7.3): the CLI
  367   -- mints while the daemon runs and the UI polls this endpoint.
  368   filed <- loadGrantFile (enrollmentGrantPath (achPaths h))
  369   let liveFile = filed >>= \grant -> if grantLiveAt now grant then Just grant else Nothing
  370       sessKey = sessionKeyFromRequest req
  371       sessGrant = sessKey >>= \key -> Map.lookup key (acsSessionGrants st)
  372       mismatch = rpMismatch (acsRpId st) (acsCredentials st)
  373       authenticated = acsMode st == AccessOpen
  374         || sessionValid now st sessKey
  375       -- A timing signal ("authority is armed"), not a boolean the UI acts on.
  376       enrollmentVia = case (liveFile, sessGrant) of
  377         (Just grant, _)   -> grantSourceText (grSource grant)
  378         (Nothing, Just grant) -> grantSourceText (grSource grant)
  379         _                 -> "none"
  380       usable = case acsRpId st of
  381         Nothing  -> Map.size (acsCredentials st)
  382         Just rp  -> length [ () | cred <- Map.elems (acsCredentials st)
  383                                 , Just (credRpId cred) == Just rp ]
  384   pure $ object
  385     [ "state" .= authStatusState st
  386     , "authenticated" .= authenticated
  387     , "origin" .= acsOrigin st
  388     , "rp_id" .= acsRpId st
  389     , "rp_mismatch" .= mismatch
  390     , "setup_required" .= (acsMode st == AccessUnconfigured || mismatch)
  391     , "credentials" .= Map.size (acsCredentials st)
  392     -- Design §8.3: no `enrollment_available` boolean — rev 3 keyed a button on
  393     -- the wrong one.
  394     , "enrollment_via" .= enrollmentVia
  395     , "usable_credentials" .= usable
  396     , "orphaned_credentials" .= (Map.size (acsCredentials st) - usable)
  397     ]

  406 tokenKey :: Text -> Text
  407 tokenKey = TE.decodeUtf8 . sha256Hash . TE.encodeUtf8

  409 cookieName :: Text
  410 cookieName = "seal_session"

  412 sessionKeyFromRequest :: Request -> Maybe Text
  413 sessionKeyFromRequest req =
  414   case cookieValue cookieName req of
  415     Just tok -> Just (tokenKey tok)
  416     Nothing  -> bearerToken req >>= (Just . tokenKey)

  418 bearerToken :: Request -> Maybe Text
  419 bearerToken req =
  420   case lookup hAuthorization (requestHeaders req) of
  421     Nothing -> Nothing
  422     Just raw -> case T.stripPrefix "Bearer " (T.strip (TE.decodeUtf8Lenient raw)) of
  423       Just t | not (T.null t) -> Just t
  424       _                       -> Nothing
  425 
  426 hAuthorization :: HeaderName
  427 hAuthorization = "Authorization"
  428 
  429 cookieValue :: Text -> Request -> Maybe Text
  430 cookieValue name req = do
  431   raw <- lookup hCookie (requestHeaders req)
  432   let pieces = map T.strip (T.splitOn ";" (TE.decodeUtf8Lenient raw))
  433   lookup name (map splitPair pieces)
  434   where
  435     splitPair p = case T.breakOn "=" p of
  436       (k, v) | T.null v   -> (k, "")
  437              | otherwise  -> (k, T.drop 1 v)

  439 hasServiceToken :: AccessControlState -> Request -> Bool
  440 hasServiceToken st req = case bearerToken req of
  441   Nothing  -> False
  442   Just tok -> constantTimeEq (TE.encodeUtf8 (tokenKey tok))
  443                               (TE.encodeUtf8 (acsServiceHash st))

  445 sessionValid :: UTCTime -> AccessControlState -> Maybe Text -> Bool
  446 sessionValid now st mKey = case mKey >>= (`Map.lookup` acsSessions st) of
  447   Nothing  -> False
  448   Just ses -> now < sesIdleExpiry ses
  449     && now < addUTCTime (fromIntegral sessionAbsoluteSeconds) (sesCreatedAt ses)

  451 -- | Mint a session and return the raw token (the caller sets the cookie).
  452 mintSession :: AccessControlHandle -> IO Text
  453 mintSession h = do
  454   raw <- generateToken 32
  455   now <- getCurrentTime
  456   modifyMVar_ (achState h) $ \st ->
  457     pure st { acsSessions = Map.insert (tokenKey raw) Session
  458                        { sesIdleExpiry = addUTCTime (fromIntegral sessionIdleSeconds) now
  459                        , sesCreatedAt = now
  460                        }
  461                        (Map.filter (stillLive now) (acsSessions st))
  462             }
  463   pure raw
  464   where
  465     stillLive now ses = now < sesIdleExpiry ses
  466       && now < addUTCTime (fromIntegral sessionAbsoluteSeconds) (sesCreatedAt ses)

  490 refillChallenges :: UTCTime -> AccessControlState -> AccessControlState
  491 refillChallenges now st =
  492   let elapsed = realToFrac (diffUTCTime now (acsChallengeRefillAt st)) :: Double
  493       tokens = min challengeBurst (acsChallengeTokens st + elapsed * challengeRefillPerSecond)
  494   in st { acsChallengeTokens = tokens, acsChallengeRefillAt = now }
  495 
  496 challengeFresh :: UTCTime -> Challenge -> Bool
  497 challengeFresh now ch =
  498   realToFrac (diffUTCTime now (chIssuedAt ch)) < (fromIntegral challengeTtlSeconds :: Double)

  500 -- | Issue a challenge, or fail when the issuance budget or the map cap is
  501 -- exhausted (never evicting a live entry).
  502 issueChallenge :: AccessControlHandle -> ChallengeKind -> Maybe Text -> IO (Either AuthErr Text)
  503 issueChallenge h kind mSession = do
  504   now <- getCurrentTime
  505   modifyMVar (achState h) $ \st0 -> do
  506     let st = refillChallenges now st0
  507         live = Map.filter (challengeFresh now) (acsChallenges st)
  508         st' = st { acsChallenges = live }
  509     if acsChallengeTokens st' < 1
  510       then pure (st', Left (AuthErr status429 "rate_limited"
  511                      "too many challenge requests; try again in a moment"))
  512       else if Map.size live >= maxChallenges
  513         then pure (st', Left (AuthErr status429 "rate_limited"
  514                        "too many outstanding challenges; try again in a moment"))
  515         else do
  516           raw <- getRandomBytes 32
  517           let ch = TE.decodeUtf8 (B64Url.encodeUnpadded raw)
  518           pure ( st' { acsChallenges = Map.insert ch (Challenge now kind mSession) live
  519                      , acsChallengeTokens = acsChallengeTokens st' - 1 }
  520                , Right ch )

  522 -- | Remove a challenge (consume-on-attempt) and report whether it was present,
  523 -- fresh and of the right ceremony.
  524 consumeChallenge :: AccessControlHandle -> Text -> ChallengeKind -> Maybe Text -> IO Bool
  525 consumeChallenge h presented kind mSession = do
  526   now <- getCurrentTime
  527   modifyMVar (achState h) $ \st ->
  528     case Map.lookup presented (acsChallenges st) of
  529       -- A step-up challenge belongs to exactly one session: a different session
  530       -- presenting it consumes nothing, so a stolen challenge string is not
  531       -- enough to redeem it.
  532       Just ch | chKind ch == kind
  533               , chSession ch == mSession
  534               , challengeFresh now ch ->
  535         pure (st { acsChallenges = Map.delete presented (acsChallenges st) }, True)
  536       _ -> pure (st, False)

  571 authErrorWith :: Status -> Text -> Text -> Value -> Response

  573   Object o -> Object (KeyMap.insert (Key.fromText "error") (String msg)
  574            (KeyMap.insert (Key.fromText "code") (String code) o))
  575   other    -> object ["error" .= msg, "code" .= code, "detail" .= other]
  576 
  577 respondResult :: [Header] -> Either AuthErr Value -> Response
  578 respondResult extra result = case result of
  579   Left (AuthErr st code msg)              -> authError st code msg
  580   Left (AuthErrDetail st code msg detail) -> authErrorWith st code msg detail
  581   Right v                                 -> authOk extra v

  583 -- | Like 'respondResult', for handlers that mint a session and therefore carry
  584 -- a @Set-Cookie@ header back.
  585 respondSession :: Either AuthErr (Value, [Header]) -> Response
  586 respondSession result = case result of
  587   Left (AuthErr st code msg)              -> authError st code msg
  588   Left (AuthErrDetail st code msg detail) -> authErrorWith st code msg detail
  589   Right (v, extra)                        -> authOk extra v

  595 authApp :: AccessControlHandle -> Application
  596 authApp h req respond =
  597   case (requestMethod req, pathInfo req) of
  598     (m, ["api", "auth", "status"]) | m == methodGet ->
  599       accessControlStatusJson h req >>= respond . authOk []
  600 
  601     (m, ["api", "auth", "register", "options"]) | m == methodPost -> do
  602       result <- withEnrollment h req (registerOptions h)
  603       respond (respondResult [] result)
  604 
  605     (m, ["api", "auth", "register", "verify"]) | m == methodPost -> do
  606       result <- withEnrollment h req (verifyRegistrationEndpoints h False)
  607       respond (respondSession result)
  608 
  609     (m, ["api", "auth", "login", "options"]) | m == methodPost -> do
  610       result <- loginOptions h
  611       respond (respondResult [] result)
  612 
  613     (m, ["api", "auth", "login", "verify"]) | m == methodPost -> do
  614       result <- loginVerify h req
  615       respond (respondSession result)
  616 
  617     (m, ["api", "auth", "logout"]) | m == methodPost -> do
  618       logout h req
  619       -- Clearing the cookie (not just forgetting the session) matters: the
  620       -- cookie is persistent (24 h idle), so a browser would otherwise keep
  621       -- presenting a token the server ignores for the rest of the day.
  622       cleared <- clearedSessionCookie h
  623       respond (authOk [cleared] (object ["ok" .= True]))
  624 
  625     -- The step-up endpoint pair (design §8.1): a live session confirms itself
  626     -- with an existing passkey and receives a short-lived assertion grant.
  627     (m, ["api", "auth", "enroll", "options"]) | m == methodPost -> do
  628       result <- enrollOptions h req
  629       respond (respondResult [] result)
  630 
  631     (m, ["api", "auth", "enroll", "verify"]) | m == methodPost -> do
  632       result <- enrollVerify h req
  633       respond (respondResult [] result)
  634 
  635     (m, ["api", "auth", "credentials", "options"]) | m == methodPost -> do
  636       result <- withEnrollment h req (registerOptions h)
  637       respond (respondResult [] result)
  638 
  639     (m, ["api", "auth", "credentials", "verify"]) | m == methodPost -> do
  640       result <- withEnrollment h req (verifyRegistrationEndpoints h True)
  641       respond (respondSession result)
  642 
  643     (m, ["api", "auth", "credentials"]) | m == methodGet -> do
  644       result <- withCredentialAdmin h req (listCredentials h)
  645       respond (respondResult [] result)
  646 
  647     (m, ["api", "auth", "credentials", cid]) | m == methodDelete -> do
  648       result <- withCredentialAdmin h req (const (deleteCredential h cid))
  649       respond (respondResult [] result)
  650 
  651     (m, ["api", "auth", "credentials", cid]) | m == methodPut -> do
  652       result <- withCredentialAdmin h req (renameCredential h cid)
  653       respond (respondResult [] result)
  654 
  655     (m, ["api", "auth", "waive"]) | m == methodPost -> do
  656       result <- withEnrollment h req (waiveAccess h)
  657       respond (respondResult [] result)
  658 
  659     _ -> respond (authError status404 "not_found" "no such endpoint")

  665 -- | The header a client presents enrollment authority in.
  666 --
  667 -- Replaces @X-Seal-Setup-Token@: the setup code was a boot-time bearer token
  668 -- with no expiry, and it is now a grant like any other (design §7.2, §8.2).
  669 enrollTokenHeader :: Request -> Maybe Text
  670 enrollTokenHeader req =
  671   case lookup "X-Seal-Enroll-Token" (requestHeaders req) of
  672     Nothing  -> Nothing
  673     Just raw -> let t = T.strip (TE.decodeUtf8Lenient raw) in if T.null t then Nothing else Just t

  675 -- | Resolve the authority this request presents, then ask whether this boot
  676 -- state accepts it.
  677 --
  678 -- The file-backed grant is re-read on every call and never cached: the CLI
  679 -- mints while the daemon runs, so a cached copy would be invisible until
  680 -- restart (design §7.3). Only session-carried assertion grants live in memory.
  681 --
  682 -- A presented token that matches nothing is 'GrantInvalid' — it must not be
  683 -- downgraded to "no authority presented", or a wrong token would be as good as
  684 -- an absent one (gate round 4, B-1).
  685 resolveEnrollment
  686   :: AccessControlHandle -> AccessControlState -> Request
  687   -> IO (Either GrantDenied EnrollmentAuthority)
  688 resolveEnrollment h st req = do
  689   now <- getCurrentTime
  690   filed <- loadGrantFile (enrollmentGrantPath (achPaths h))
  691   let liveFile = filed >>= \grant -> if grantLiveAt now grant then Just grant else Nothing
  692       presented = enrollTokenHeader req
  693       sessKey = sessionKeyFromRequest req
  694       sessGrant = sessKey >>= \key -> Map.lookup key (acsSessionGrants st)
  695       ctx = GrantContext
  696         { gcMode = acsMode st
  697         , gcAvailability = credentialAvailability (acsRpId st) (acsCredentials st)
  698         , gcAuthorized = Nothing
  699         , gcHasSession = sessionValid now st sessKey
  700         }
  701   pure $ case findAuthority presented liveFile sessGrant (isJust sessGrant) of
  702     Right grant -> enrollmentAuthority ctx { gcAuthorized = Just (grSource grant) }
  703     Left GrantInvalid | isJust presented -> Left GrantInvalid
  704     Left _ -> enrollmentAuthority ctx
  705 
  706 -- | The wire response for each refusal. The constructors exist for this.
  707 enrollRefusal :: GrantDenied -> AuthErr
  708 enrollRefusal denied = case denied of
  709   GrantRequired ->
  710     AuthErr status403 "grant_required"
  711       "enrollment needs an enrollment token; run `seal access-control enroll`"
  712   StepUpRequired ->
  713     AuthErr status403 "step_up_required"
  714       "confirm it's you with an existing passkey to add another"
  715   GrantInvalid ->
  716     AuthErr status403 "grant_invalid"
  717       ("that enrollment token is not valid for this gateway; use the token the "
  718         <> "current `seal serve` printed, or run `seal access-control enroll --force`")

  720 -- | Gate an enrollment endpoint on 'resolveEnrollment', handing the accepted
  721 -- authority to the handler.
  722 withEnrollment
  723   :: AccessControlHandle -> Request
  724   -> (EnrollmentAuthority -> Request -> IO (Either AuthErr a))
  725   -> IO (Either AuthErr a)
  726 withEnrollment h req action = do
  727   st <- withMVar (achState h) pure
  728   decision <- resolveEnrollment h st req
  729   case decision of
  730     Left denied -> do
  731       auditRefusal h denied
  732       pure (Left (enrollRefusal denied))
  733     Right authority -> action authority req

  735 -- | Record a refused enrollment attempt, aggregated.
  736 --
  737 -- Aggregation is what keeps a pre-auth flood from rotating the state-changing
  738 -- evidence away (design §15): the key is the refusal code, never anything the
  739 -- client chose.
  740 auditRefusal :: AccessControlHandle -> GrantDenied -> IO ()
  741 auditRefusal h denied = do
  742   now <- getCurrentTime
  743   let code = case denied of
  744         GrantRequired   -> "grant_required"
  745         StepUpRequired  -> "step_up_required"
  746         GrantInvalid    -> "grant_invalid"
  747   admitted <- admitRefusal (achAudit h) now code
  748   when admitted $ do
  749     _ <- appendAudit (achPaths h) now (emptyAuditFields Audit.GrantRefused)
  750     pure ()
  751 
  752 -- | Record a state-changing event. Failures are swallowed on purpose: the audit
  753 -- log is detective, and a full disk must not fail a login.
  754 auditEvent :: AccessControlHandle -> AuditFields -> IO ()
  755 auditEvent h fields = do
  756   now <- getCurrentTime
  757   _ <- appendAudit (achPaths h) now fields
  758   pure ()

  760 -- | Who may list and delete credentials (design §8.1).

  762   = AdminSession
  763   | AdminCliToken
  764   | AdminOpen
  765   | AdminDenied
  766   deriving stock (Eq, Show)

  768 -- | Credential administration is session-or-CLI-token, plus unauthenticated in
  769 -- @open@ mode (which is fully open anyway, and is what lets @\/access@ work
  770 -- there at all). Window and assertion grants are enrollment-only.
  771 credentialAdminFor
  772   :: AccessControlHandle -> AccessControlState -> Request -> IO CredentialAdmin
  773 credentialAdminFor h st req = do
  774   now <- getCurrentTime
  775   if acsMode st == AccessOpen
  776     then pure AdminOpen
  777     else if sessionValid now st (sessionKeyFromRequest req)
  778       then pure AdminSession
  779       else do
  780         filed <- loadGrantFile (enrollmentGrantPath (achPaths h))
  781         let liveFile = filed >>= \grant -> if grantLiveAt now grant then Just grant else Nothing
  782             tokenOk = case (enrollTokenHeader req, liveFile) of
  783               (Just raw, Just grant) | grSource grant == CliToken ->
  784                 case grTokenSha grant of
  785                   Just digest -> constantTimeEq digest (tokenDigest raw)
  786                   Nothing     -> False
  787               _ -> False
  788         pure (if tokenOk then AdminCliToken else AdminDenied)

  801 -- | Bind the outstanding file grant to a challenge, exclusively.
  802 --
  803 -- @O_CREAT|O_EXCL@ on the claim file decides which ceremony wins, and the grant
  804 -- is written back with its bound challenge so a grant bound to A cannot redeem
  805 -- B (design §7.3). A grant in a session ('Assertion') is not bound here: it is
  806 -- single-use through the challenge itself.
  807 --
  808 -- Retries supersede an abandoned claim: see 'bindClaimSuperseding'.
  809 bindGrant :: SealPaths -> Grant -> ChallengeId -> IO Bool
  810 bindGrant paths grant challenge = do
  811   now <- getCurrentTime
  812   let GrantId rawId = grId grant
  813       claim = GrantClaim
  814         { clGrantId = grId grant
  815         , clChallengeId = challenge
  816         , clDeadline = effectiveDeadline grant now (fromIntegral challengeTtlSeconds)
  817         }
  818   won <- bindClaimSuperseding (enrollmentClaimPath paths rawId) claim
  819   if not won
  820     then pure False
  821     else do
  822       saved <- saveGrantFile (enrollmentGrantPath paths) grant { grPhase = GrantBound challenge }
  823       pure (either (const False) (const True) saved)
  824 
  825 -- | Does the outstanding grant permit redeeming this challenge?
  826 --
  827 -- 'True' when there is no file grant (the authority was an assertion, carried
  828 -- by the session), so the assertion path is unaffected by the file protocol.
  829 grantPermits :: SealPaths -> ChallengeId -> IO Bool
  830 grantPermits paths challenge = do
  831   filed <- loadGrantFile (enrollmentGrantPath paths)
  832   case filed of
  833     Nothing -> pure True
  834     Just grant -> do
  835       let GrantId rawId = grId grant
  836       held <- loadClaimFile (enrollmentClaimPath paths rawId)
  837       now <- getCurrentTime
  838       pure (grantRedeems challenge grant && maybe False (claimLiveAt now) held)

  844 registerOptions
  845   :: AccessControlHandle -> EnrollmentAuthority -> Request
  846   -> IO (Either AuthErr Value)
  847 registerOptions h authority _req = do
  848   st <- withMVar (achState h) pure
  849   case (acsOrigin st, acsRpId st) of
  850     (_, Nothing) -> pure (Left (AuthErr status400 "origin_mismatch"
  851                          "this gateway has no relying-party id configured"))
  852     (Nothing, _) -> pure (Left (AuthErr status400 "origin_mismatch"
  853                          "this gateway has no origin configured"))
  854     (Just _, Just rpId) -> do
  855       now <- getCurrentTime
  856       challenge <- issueChallenge h ChallengeCreate Nothing
  857       case challenge of
  858         Left e      -> pure (Left e)
  859         Right value -> do
  860           -- Bind the outstanding grant to this challenge before handing out
  861           -- the options (design §7.3): the O_EXCL claim is what makes one grant
  862           -- redeemable exactly once, even across processes. An assertion grant
  863           -- lives in the session and needs no file binding.
  864           bound <- case authority of
  865             Assertion -> pure True
  866             _ -> do
  867               filed <- loadGrantFile (enrollmentGrantPath (achPaths h))
  868               case filed of
  869                 Just grant | grantLiveAt now grant ->
  870                   bindGrant (achPaths h) grant (ChallengeId value)
  871                 -- The token was live when 'resolveEnrollment' looked, so this is
  872                 -- a race with --cancel/--force: the ceremony must not proceed.
  873                 _ -> pure False
  874           if not bound
  875             then pure (Left (enrollRefusal GrantInvalid))
  876             else do
  877               userRaw <- getRandomBytes 16
  878               pure (Right (object
  879                 [ "challenge" .= value
  880                 , "rp" .= object ["id" .= rpId, "name" .= ("Seal Harness" :: Text)]
  881                 , "user" .= object
  882                     [ "id" .= TE.decodeUtf8 (B64Url.encodeUnpadded userRaw)
  883                     , "name" .= ("operator" :: Text)
  884                     , "displayName" .= ("Seal Harness operator" :: Text)
  885                     ]
  886                 , "pubKeyCredParams" .=
  887                     [ object ["type" .= ("public-key" :: Text), "alg" .= (-7 :: Int)] ]
  888                 , "timeout" .= (120000 :: Int)
  889                 , "attestation" .= ("none" :: Text)
  890                 , "authenticatorSelection" .= object
  891                     [ "userVerification" .= ("required" :: Text)
  892                     , "residentKey" .= ("preferred" :: Text)
  893                     ]
  894                 , "excludeCredentials" .=
  895                     [ object [ "type" .= ("public-key" :: Text)
  896                              , "id" .= cid ] | cid <- Map.keys (acsCredentials st) ]
  897                 ]))

  919 -- | Verify a registration response. @isAdd@ distinguishes the
  920 -- authenticated "add another passkey" path from first-run enrollment.
  921 verifyRegistrationEndpoints
  922   :: AccessControlHandle -> Bool -> EnrollmentAuthority -> Request
  923   -> IO (Either AuthErr (Value, [Header]))
  924 verifyRegistrationEndpoints h isAdd authority req = do
  925   body <- readJsonBody req
  926   case body of
  927     Left e -> pure (Left e)
  928     Right obj -> do
  929       st <- withMVar (achState h) pure
  930       case (,,) <$> b64Field obj "clientDataJSON"
  931                 <*> b64Field obj "attestationObject"
  932                 <*> b64Field obj "rawId" of
  933         Right (clientData, attestation, rawId) -> do
  934           now <- getCurrentTime
  935           let presentedChallenge = clientChallenge clientData
  936           -- The grant must still be bound to this challenge and its claim must
  937           -- still be inside the deadline; only then is the challenge consumed.
  938           permits <- case (authority, presentedChallenge) of
  939             (Assertion, Just c) -> pure (Just c)
  940             (_, Just c) -> do
  941               ok <- grantPermits (achPaths h) (ChallengeId c)
  942               pure (if ok then Just c else Nothing)
  943             (_, Nothing) -> pure Nothing
  944           consumed <- case permits of
  945             Nothing -> pure False
  946             Just c  -> consumeChallenge h c ChallengeCreate Nothing
  947           if not consumed
  948             then pure (Left
  949                    (case permits of
  950                       Nothing | isJust presentedChallenge -> enrollRefusal GrantInvalid
  951                       _ -> AuthErr status400 "challenge_invalid"
  952                              "the challenge is missing, expired or already used"))
  953             else
  954               case (acsOrigin st, acsRpId st) of
  955                 (Just origin, Just rpId) ->
  956                   case verifyRegistration RegistrationInput
  957                         { riClientDataJSON = clientData
  958                         , riAttestationObject = attestation
  959                         , riRawId = rawId
  960                         , riExpectedChallenge = fromMaybeText presentedChallenge
  961                         , riExpectedOrigin = origin
  962                         , riExpectedRpId = rpId
  963                         } of
  964                     Left e -> pure (Left (webAuthnFailure e))
  965                     Right result -> persistRegistration h authority isAdd result now
  966                 _ -> pure (Left (AuthErr status400 "origin_mismatch"
  967                             "this gateway has no origin configured"))
  968         Left e -> pure (Left e)

  970 persistRegistration
  971   :: AccessControlHandle -> EnrollmentAuthority -> Bool -> RegistrationResult -> UTCTime
  972   -> IO (Either AuthErr (Value, [Header]))
  973 persistRegistration h authority isAdd result now = do
  974   st <- withMVar (achState h) pure
  975   let rawId = TE.decodeUtf8 (B64Url.encodeUnpadded (rrCredentialId result))
  976       cred = Credential
  977         { credId = rrCredentialId result
  978         , credPublicKey = rrPublicKey result
  979         , credSignCount = rrSignCount result
  980         , credLabel = "passkey"
  981         , credCreatedAt = now
  982         , credRpId = fromMaybeText (acsRpId st)
  983         , credAddedVia = Just (grantSourceText authority)
  984         }
  985   -- Raise the mode and write the credential as ONE operation (design §6.1):
  986   -- two files, so neither is atomic alone, and two interleaved ceremonies could
  987   -- otherwise leave a credential on a gateway whose mode still says "none".
  988   -- 'enrollCredential' takes the write lock exactly once around both.
  989   written <- enrollCredential (securityFilePath (achPaths h))
  990                               (credentialsFilePath (achPaths h))
  991                               (Map.singleton rawId (credentialToFile cred))
  992   case written of
  993     Left err -> pure (Left (AuthErr status500 "write_failed" err))
  994     Right () -> do
  995           modifyMVar_ (achState h) $ \s ->
  996             let creds' = Map.insert rawId cred (acsCredentials s)
  997             in pure s { acsMode = AccessPasskey
  998                       , acsCredentials = creds'
  999                       , acsSetupCode = Nothing
 1000                       , acsSetupHash = Nothing
 1001                       }
 1002           -- Success consumes the authority: file grant, claim marker and the
 1003           -- plaintext token all go, and any session-carried grant with them
 1004           -- (design §7.2). Sessions themselves are untouched.
 1005           clearOutstandingGrants h
 1006           auditEvent h (emptyAuditFields Audit.CredentialAdded)
 1007             { afCredentialId = Just rawId
 1008             , afLabel = Just (credLabel cred)
 1009             , afRpId = Just (credRpId cred)
 1010             , afOrigin = acsOrigin st
 1011             , afVia = Just authority
 1012             }
 1013           when (acsMode st /= AccessPasskey) $
 1014             auditEvent h (emptyAuditFields Audit.ModeEnabled) { afOrigin = acsOrigin st }
 1015           auditEvent h (emptyAuditFields Audit.GrantConsumed) { afOrigin = acsOrigin st }
 1016           token <- mintSession h
 1017           cookie <- sessionCookie h token
 1018           pure (Right ( object
 1019             [ "ok" .= True
 1020             , "credential" .= rawId
 1021             , "added" .= isAdd
 1022             ]
 1023             , [cookie] ))

 1026 loginVerify :: AccessControlHandle -> Request -> IO (Either AuthErr (Value, [Header]))
 1027 loginVerify h req = do
 1028   body <- readJsonBody req
 1029   case body of
 1030     Left e -> pure (Left e)
 1031     Right obj -> do
 1032       now <- getCurrentTime
 1033       allowed <- backoffAllows h now
 1034       if not allowed
 1035         then pure (Left (AuthErr status429 "rate_limited" "too many failed attempts; try again shortly"))
 1036         else do
 1037           st <- withMVar (achState h) pure
 1038           case (,,,) <$> b64Field obj "clientDataJSON"
 1039                      <*> b64Field obj "authenticatorData"
 1040                      <*> b64Field obj "signature"
 1041                      <*> b64Field obj "rawId" of
 1042             Right (clientData, authData, signature, rawId) -> do
 1043               let claimedId = TE.decodeUtf8 (B64Url.encodeUnpadded rawId)
 1044               case Map.lookup claimedId (acsCredentials st) of
 1045                 Nothing -> do
 1046                   recordFailure h now
 1047                   pure (Left (AuthErr status400 "unknown_credential" "no credential with that id is enrolled"))
 1048                 Just cred -> do
 1049                   consumed <- maybe (pure False)
 1050                                     (\c -> consumeChallenge h c ChallengeGet Nothing)
 1051                                     (clientChallenge clientData)
 1052                   if not consumed
 1053                     then pure (Left (AuthErr status400 "challenge_invalid"
 1054                                "the challenge is missing, expired or already used"))
 1055                     else
 1056                       case (acsOrigin st, acsRpId st) of
 1057                         (Just origin, Just rpId) ->
 1058                           case verifyAssertion (credPublicKey cred) AssertionInput
 1059                                 { aiClientDataJSON = clientData
 1060                                 , aiAuthenticatorData = authData
 1061                                 , aiSignature = signature
 1062                                 , aiExpectedChallenge = fromMaybeText (clientChallenge clientData)
 1063                                 , aiExpectedOrigin = origin
 1064                                 , aiExpectedRpId = rpId
 1065                                 , aiStoredSignCount = credSignCount cred
 1066                                 } of
 1067                             Left e -> do
 1068                               recordFailure h now
 1069                               pure (Left (webAuthnFailure e))
 1070                             Right newCount -> do
 1071                               bumped <- claimSignCount h claimedId (credSignCount cred) newCount
 1072                               if not bumped
 1073                                 then pure (Left (AuthErr status400 "sign_count_regression"
 1074                                            "this assertion was already used"))
 1075                                 else do
 1076                                   -- Persist, then mint. A counter that did not
 1077                                   -- reach disk must not produce a session: the
 1078                                   -- next boot would accept a replay of the
 1079                                   -- assertion we just used.
 1080                                   persisted <- persistSignCount h claimedId newCount
 1081                                   case persisted of
 1082                                     Left err -> pure (Left (AuthErr status500 "write_failed" err))
 1083                                     Right () -> do
 1084                                       resetBackoff h
 1085                                       token <- mintSession h
 1086                                       cookie <- sessionCookie h token
 1087                                       pure (Right ( object
 1088                                         ["ok" .= True, "credential" .= claimedId]
 1089                                         , [cookie] ))
 1090                         _ -> pure (Left (AuthErr status400 "origin_mismatch"
 1091                                     "this gateway has no origin configured"))
 1092             Left e -> pure (Left e)

 1098 -- | @enroll/options@ (design §8.1): the assertion that authorises adding a
 1099 -- second passkey.
 1100 --
 1101 -- Unlike @login/options@ this is not a session gate but a mode+availability
 1102 -- gate: it exists only in a healthy @passkey@ boot with at least one usable
 1103 -- credential, because there is nothing to prove anywhere else. The challenge is
 1104 -- single-use and bound to the calling session, so a challenge lifted from
 1105 -- another session cannot be redeemed here.
 1106 enrollOptions :: AccessControlHandle -> Request -> IO (Either AuthErr Value)
 1107 enrollOptions h req = do
 1108   st <- withMVar (achState h) pure
 1109   now <- getCurrentTime
 1110   let sessKey = sessionKeyFromRequest req
 1111   case (acsMode st, credentialAvailability (acsRpId st) (acsCredentials st),
 1112         sessionValid now st sessKey, sessKey) of
 1113     (AccessPasskey, Usable, True, Just key) -> do
 1114       challenge <- issueChallenge h ChallengeStepUp (Just key)
 1115       case challenge of
 1116         Left e      -> pure (Left e)
 1117         Right value -> pure (Right (object
 1118           [ "challenge" .= value
 1119           , "rpId" .= fromMaybeText (acsRpId st)
 1120           , "timeout" .= (120000 :: Int)
 1121           , "userVerification" .= ("required" :: Text)
 1122           -- Only usable credentials: the picker must never offer one that

 1124           , "allowCredentials" .=

 1135 -- | @enroll/verify@: check the step-up assertion and return the grant the
 1136 -- client presents to @credentials/*@.
 1137 --
 1138 -- The grant is an 'Assertion' grant: 120 s, digest stored in this session only,
 1139 -- and presented as @X-Seal-Enroll-Token@ on the retried @credentials/options@.
 1140 -- No failure body carries @grant_token@ or @grant_expires_at@.
 1141 enrollVerify :: AccessControlHandle -> Request -> IO (Either AuthErr Value)
 1142 enrollVerify h req = do
 1143   body <- readJsonBody req
 1144   case body of
 1145     Left e -> pure (Left e)
 1146     Right obj ->
 1147       -- Order is part of the contract (design §4.1). The type checks, the
 1148       -- mode/session gate and the outstanding-grant check all run before
 1149       -- 'consumeChallenge', so a malformed body, an absent session or an
 1150       -- already-armed grant costs the operator no WebAuthn prompt and burns no
 1151       -- sign counter.
 1152       case (parseGrantMode obj, boolField obj "force") of
 1153         (Left e, _) -> pure (Left e)
 1154         (_, Left e) -> pure (Left e)
 1155         (Right portable, Right force) -> do
 1156           now <- getCurrentTime
 1157           st <- withMVar (achState h) pure
 1158           let sessKey = sessionKeyFromRequest req
 1159           if not (sessionValid now st sessKey)
 1160                || acsMode st /= AccessPasskey
 1161                || credentialAvailability (acsRpId st) (acsCredentials st) /= Usable
 1162             then pure (Left (enrollRefusal GrantRequired))
 1163             else do
 1164               armed <- if portable && not force
 1165                          then liveFileGrant h now
 1166                          else pure Nothing
 1167               case armed of
 1168                 Just grant -> pure (Left (grantOutstandingRefusal grant))
 1169                 Nothing    -> verifyStepUp h obj st sessKey portable

 1171 -- | The assertion half of @enroll/verify@: the body's base64url fields, the
 1172 -- challenge, the signature, the sign counter and the mint. Split out only so
 1173 -- the checks above read as the ordered contract they are.
 1174 verifyStepUp
 1175   :: AccessControlHandle -> Value -> AccessControlState -> Maybe Text -> Bool
 1176   -> IO (Either AuthErr Value)
 1177 verifyStepUp h obj st sessKey portable =
 1178   case (,,,) <$> b64Field obj "clientDataJSON"
 1179              <*> b64Field obj "authenticatorData"
 1180              <*> b64Field obj "signature"
 1181              <*> b64Field obj "rawId" of
 1182     Left e -> pure (Left e)
 1183     Right (clientData, authData, signature, rawId) -> do
 1184       let claimedId = TE.decodeUtf8 (B64Url.encodeUnpadded rawId)
 1185       case Map.lookup claimedId (acsCredentials st) of
 1186         Nothing -> pure (Left (enrollRefusal GrantInvalid))
 1187         Just cred
 1188           -- The picker only offers usable credentials, so naming an orphaned
 1189           -- one is forged or racing (design §8.1).
 1190           | Just (credRpId cred) /= acsRpId st ->
 1191               pure (Left (enrollRefusal GrantInvalid))
 1192           | otherwise -> do
 1193               consumed <- maybe (pure False)
 1194                 (\c -> consumeChallenge h c ChallengeStepUp sessKey)
 1195                 (clientChallenge clientData)
 1196               if not consumed
 1197                 then pure (Left (AuthErr status400 "challenge_invalid"
 1198                            "the challenge is missing, expired or already used"))
 1199                 else case (acsOrigin st, acsRpId st) of
 1200                   (Just origin, Just rpId) ->
 1201                     case verifyAssertion (credPublicKey cred) AssertionInput
 1202                           { aiClientDataJSON = clientData
 1203                           , aiAuthenticatorData = authData
 1204                           , aiSignature = signature
 1205                           , aiExpectedChallenge = fromMaybeText (clientChallenge clientData)
 1206                           , aiExpectedOrigin = origin
 1207                           , aiExpectedRpId = rpId
 1208                           , aiStoredSignCount = credSignCount cred
 1209                           } of
 1210                       Left e -> pure (Left (webAuthnFailure e))
 1211                       Right newCount -> do
 1212                         bumped <- claimSignCount h claimedId (credSignCount cred) newCount
 1213                         if not bumped
 1214                           then pure (Left (AuthErr status400 "sign_count_regression"
 1215                                     "this assertion was already used"))
 1216                           else do
 1217                             persisted <- persistSignCount h claimedId newCount
 1218                             case persisted of
 1219                               Left err -> pure (Left (AuthErr status500 "write_failed" err))
 1220                               -- One step-up, one mint: the portable branch
 1221                               -- deliberately does not also write a session
 1222                               -- grant, so there is never a live same-device
 1223                               -- token for the client to discard.
 1224                               Right ()
 1225                                 | portable  -> mintPortableGrant h
 1226                                 | otherwise -> mintAssertionGrant h claimedId sessKey
 1227                   _ -> pure (Left (AuthErr status400 "origin_mismatch"
 1228                               "this gateway has no origin configured"))

 1391 deleteCredential :: AccessControlHandle -> Text -> IO (Either AuthErr Value)
 1392 deleteCredential h cid = do
 1393   st <- withMVar (achState h) pure
 1394   let rpId = acsRpId st
 1395       -- With a wildcard bind every credential is usable (design §9.4).
 1396       usable cred = case rpId of
 1397         Nothing -> True
 1398         Just rp -> cfcRpId cred == rp
 1399   -- The guard runs inside the writer, on the freshly loaded map, so two
 1400   -- concurrent deletes cannot each see "one other credential remains" and both
 1401   -- proceed. A read-then-write in the handler could.
 1402   written <- updateCredentialsFile (credentialsFilePath (achPaths h)) $ \creds ->
 1403     let remaining = Map.delete cid creds
 1404         usableLeft = length (filter usable (Map.elems remaining))
 1405     in if Map.member cid creds && usableLeft == 0
 1406          then Left "that is the last passkey that can sign in here; add or fix another before removing it"
 1407          else Right remaining
 1408   case written of
 1409     Left err -> pure (Left (AuthErr status409 "last_credential" err))
 1410     Right () -> do
 1411       modifyMVar_ (achState h) $ \s ->
 1412         pure s { acsCredentials = Map.delete cid (acsCredentials s) }
 1413       auditEvent h (emptyAuditFields Audit.CredentialRemoved)
 1414         { afCredentialId = Just cid
 1415         , afRpId = rpId
 1416         , afOrigin = acsOrigin st
 1417         }
 1418       pure (Right (object ["ok" .= True, "removed" .= cid]))

 1523   where
 1524     go acc chunks = do
 1525       chunk <- getRequestBodyChunk req

   44 -- | All paths derived from the seal home directory.
   45 --
   46 -- * 'spConfig' — version-controllable config tree; ordinary directory
   47 -- * 'spState'  — mutable runtime state; ordinary directory
   48 -- * 'spKeys'   — key material; created mode 0700, never version-controlled
   49 data SealPaths = SealPaths
   50   { spHome   :: FilePath   -- ^ @SEAL_HOME@ env var or @~\/.seal@
   51   , spConfig :: FilePath   -- ^ @\<home\>\/config@
   52   , spState  :: FilePath   -- ^ @\<home\>\/state@
   53   , spKeys   :: FilePath   -- ^ @\<home\>\/keys@
   54   , spCache  :: FilePath   -- ^ @\<home\>\/cache@ (per-session workdirs)
   55   } deriving stock (Eq, Show)
   56 
   57 -- | Resolve the seal home directory.
   58 --
   59 -- Returns the value of @SEAL_HOME@ when the variable is set; otherwise
   60 -- returns @~\/.seal@ via 'getHomeDirectory'.
   61 resolveSealHome :: IO FilePath
   62 resolveSealHome = do
   63   mEnv <- lookupEnv "SEAL_HOME"
   64   case mEnv of
   65     Just h  -> pure h
   66     Nothing -> do
   67       home <- getHomeDirectory
   68       pure (home </> ".seal")
   69 
   70 -- | Compute all sub-paths under the seal home directory without touching
   71 -- the filesystem.
   72 getSealPaths :: IO SealPaths
   73 getSealPaths = do
   74   home <- resolveSealHome
   75   pure SealPaths
   76     { spHome   = home
   77     , spConfig = home </> "config"
   78     , spState  = home </> "state"
   79     , spKeys   = home </> "keys"
   80     , spCache  = home </> "cache"
   81     }

   83 -- | Create the seal directory tree, setting restrictive permissions on the
   84 -- keys directory.
   85 --
   86 -- * @config\/@ and @state\/@ are created with default (umask-governed) mode.
   87 -- * @keys\/@ is created and then explicitly set to mode @0700@.
   88 --
   89 -- Calling this function when the directories already exist is safe
   90 -- ('createDirectoryIfMissing' is idempotent; 'setFileMode' is idempotent).
   91 ensureSealDirs :: SealPaths -> IO ()
   92 ensureSealDirs paths = do
   93   createDirectoryIfMissing True (spConfig paths)
   94   createDirectoryIfMissing True (spState  paths)
   95   createDirectoryIfMissing True (spCache  paths)
   96   createDirectoryIfMissing True (spKeys   paths)
   97   -- The vault lives in a subdirectory of config/ ('vaultFilePath'); create it
   98   -- so the atomic vault write has an existing parent for its .tmp file.
   99   createDirectoryIfMissing True (takeDirectory (vaultFilePath paths))
  100   -- The persistent ssh-agent registry lives in a subdirectory of state/repos;
  101   -- create it so the registry file has an existing parent (#88).
  102   createDirectoryIfMissing True (sshAgentsDir paths)
  103   setFileMode (spKeys paths) 0o700

  127 -- | Private state dir for clone-time temp files (GIT_ASKPASS helper scripts,
  128 -- deploy-key keyfiles, managed known_hosts). Security invariants: created
  129 -- mode @0700@, never under @\/tmp@ (so secrets never touch a world-writable
  130 -- temp dir), never version-controlled (so secret bytes never land in git).
  131 repoCloneStateDir :: SealPaths -> FilePath
  132 repoCloneStateDir paths = spState paths </> "repos"

  134 -- | The harness-private dir holding the encrypted deploy-key keyfiles (rev 3
  135 -- of the design: the encrypted keyfile stays on the harness disk as
  136 -- ciphertext; the passphrase lives in the vault; @ssh-add@ decrypts the
  137 -- keyfile into agent memory at git-op time using the passphrase piped to its
  138 -- stdin). Path: @\<state\>\/repos\/keys@. Created mode @0700@ by the caller
  139 -- (the @ssh-keygen@ invocation in W5 creates it on first key generation).
  140 -- The encrypted keyfile is the per-repo artifact named by 'repoIdText'.
  141 repoKeysDir :: SealPaths -> FilePath
  142 repoKeysDir paths = repoCloneStateDir paths </> "keys"

  216 -- | Directory for a sub-agent's transcript, nested under its parent session:
  217 -- @\<state\>\/sessions\/\<parent-id\>\/agents\/\<child-id\>@. Each forked agent
  218 -- instance gets its own two-file transcript here so the parent's
  219 -- @conversation.jsonl@ \/ @entries.jsonl@ stay uncontaminated (the two-file
  220 -- format's @erConvLen@ and envelope-delta fold are per-session; mixing a
  221 -- sub-agent's entries into the parent's files would corrupt reconstruction).
  222 -- The Audited log is shared (cross-session and canonical).
  223 agentSessionDir :: SealPaths -> SessionId -> SessionId -> FilePath
  224 agentSessionDir paths parentSid childSid =
  225   sessionDir paths parentSid </> "agents" </> T.unpack (sessionIdText childSid)
  226 
  227 -- | Locate a child session's directory by scanning every parent session's
  228 -- @agents\/@ subtree: @\<state\>\/sessions\/\<parent\>\/agents\/\<sid\>@.
  229 -- Returns the first @parentSid@ whose @agents\/\<sid\>\/conversation.jsonl@
  230 -- exists, paired with the child directory path. Returns 'Nothing' when no
  231 -- parent nests such a child. Used by the session opcodes to make forked
  232 -- sub-agent transcripts visible to SESSION_GET / SESSION_SEARCH / SESSION_LIST.
  233 resolveChildSessionPath :: SealPaths -> SessionId -> IO (Maybe (SessionId, FilePath))
  234 resolveChildSessionPath paths sid = do
  235   let root = sessionsRoot paths
  236       sidText = sessionIdText sid
  237   exists <- doesDirectoryExist root
  238   if not exists
  239     then pure Nothing
  240     else do
  241       entries <- listDirectory root
  242       dirs <- filterM (doesDirectoryExist . (root </>)) entries
  243       matches <- forM dirs $ \e -> do
  244         let parentDir = root </> e
  245             agentsDir = parentDir </> "agents"
  246             childDir = agentsDir </> T.unpack sidText
  247             convPath = childDir </> "conversation.jsonl"
  248         agentsOk <- doesDirectoryExist agentsDir
  249         if not agentsOk
  250           then pure Nothing
  251           else do
  252             convOk <- doesFileExist convPath
  253             if not convOk
  254               then pure Nothing
  255               else case mkSessionId (T.pack e) of
  256                 Left _ -> pure Nothing
  257                 Right parentSid -> pure (Just (parentSid, childDir))
  258       pure (listToMaybe (catMaybes matches))

  260 -- | Root for per-session working directories: @\<cache\>\/workdirs@.
  261 workdirsRoot :: SealPaths -> FilePath
  262 workdirsRoot paths = spCache paths </> "workdirs"
  263 
  264 -- | One session's working directory: @\<cache\>\/workdirs\/\<session-id\>@.
  265 -- Asserts 'isValidSessionId' as defense-in-depth (the constructor is
  266 -- locked down, but this catches any future raw-constructor leak before
  267 -- the path reaches mkdir or rm -rf).
  268 sessionWorkdir :: SealPaths -> SessionId -> FilePath
  269 sessionWorkdir paths sid
  270   | isValidSessionId (sessionIdText sid) =
  271       workdirsRoot paths </> T.unpack (sessionIdText sid)
  272   | otherwise =
  273       error ("sessionWorkdir: invalid session id: " <> T.unpack (sessionIdText sid))

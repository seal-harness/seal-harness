    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Shared atomic JSON persistence for small state files
    3 -- (@\<state\>\/tabs.json@, @\<state\>\/cursors.json@).
    4 --
    5 -- The pattern — write @path.tmp@, chmod 0600, rename over the target, all
    6 -- serialized by a module-level MVar so concurrent writes to the same file
    7 -- cannot interleave — was duplicated across 'Seal.Tabs.Persist.saveTabList'
    8 -- and is now needed by 'Seal.Channels.Cursor.Persist.saveCursorMap'. This
    9 -- helper centralizes it so the two do not drift.
   10 --
   11 -- Note: 'Seal.Session.Store.saveSessionMeta' uses a different temp-file
   12 -- strategy ('openBinaryTempFile' for a unique name per call) because it
   13 -- guards against concurrent saves of the /same/ session file from multiple
   14 -- threads without a shared lock; it is intentionally NOT consolidated here
   15 -- (see the design doc's §6). This helper is for files where a single
   16 -- module-level MVar is an acceptable serialization point.
   17 module Seal.Util.AtomicJson
   18   ( saveJsonAtomic
   19   ) where
   20 
   21 import Control.Concurrent.MVar (MVar, modifyMVar_, newMVar)
   22 import Data.ByteString.Lazy qualified as BL
   23 import System.Directory (createDirectoryIfMissing, renameFile)
   24 import System.FilePath (takeDirectory)
   25 import System.IO.Unsafe (unsafePerformIO)
   26 import System.Posix.Files (setFileMode, unionFileModes, ownerReadMode, ownerWriteMode)
   27 
   28 -- | Module-level write lock — serializes concurrent 'saveJsonAtomic' calls
   29 -- so two mutations cannot interleave file writes (a bare forkIO TVar
   30 -- listener would race; the explicit MVar guarantees atomicity). One lock
   31 -- per process, shared across all callers. The critical sections are short
   32 -- (a single file write + rename) so this is not a contention concern in
   33 -- practice — these files mutate at human, not machine, rate.
   34 writeLock :: MVar ()
   35 writeLock = unsafePerformIO (newMVar ())
   36 {-# NOINLINE writeLock #-}
   37 
   38 -- | Atomically write a lazy 'BL.ByteString' (typically @Aeson.encode@ output)
   39 -- to @path@: create the parent dir, write @path.tmp@, chmod 0600, rename
   40 -- over the target. Serialized via 'writeLock' so concurrent calls from
   41 -- multiple threads (channel loop + gateway API + slash command) don't
   42 -- collide on the shared @.tmp@ path. Never throws (the caller is expected
   43 -- to 'catch' log a warning on failure, matching the
   44 -- 'Seal.Tabs.Persist' / 'Seal.Channels.Cursor.Persist' pattern — the
   45 -- in-memory handle stays authoritative within the session and the next
   46 -- successful mutation retries by writing the full current state, so a
   47 -- missed save self-heals).
   48 saveJsonAtomic :: FilePath -> BL.ByteString -> IO ()
   49 saveJsonAtomic path bs = modifyMVar_ writeLock $ \_ -> do
   50   createDirectoryIfMissing True (takeDirectory path)
   51   let tmp = path <> ".tmp"
   52   BL.writeFile tmp bs
   53   setFileMode tmp (unionFileModes ownerReadMode ownerWriteMode)  -- 0600
   54   renameFile tmp path

   40 -- | The dependencies a channel turn needs to have full parity with the web
   41 -- and CLI paths. Built once at startup (in 'Seal.Command.Serve') and shared
   42 -- across all turns. The reply registry, write locks, abort registry, and
   43 -- exec cache created inside 'newChannelDeps' are shared with the web send
   44 -- handler ('Seal.Gateway.Send.SendDeps') so turns from all surfaces
   45 -- (web, chat channels) hit the same stores.
   46 data ChannelDeps = ChannelDeps
   47   { cdPaths      :: SealPaths
   48   , cdVault      :: VaultRuntime
   49   , cdRepoReg    :: RepoRegistryHandle
   50   , cdAgentReg   :: AgentRegistryHandle
   51     -- ^ The shared ssh-agent registry (one per process). Threaded through
   52     -- 'TurnDeps' so all git-op call sites share the same 'arhLive' set.
   53   , cdProvider   :: ProviderRuntime
   54   , cdBackends   :: Backends
   55   , cdAutonomy   :: Policy.AutonomyLevel
   56   , cdBroker     :: Maybe StreamBroker
   57     -- ^ The WS broker for pushing live transcript entries to the frontend.
   58     -- 'Nothing' in standalone modes (no web frontend); 'Just' under
   59     -- @seal serve@ so channel turns surface in the web UI in real time.
   60   , cdHarnessRegistry :: HarnessRegistry
   61   , cdTmuxRunner  :: TmuxRunner
   62   , cdHttpManager :: Maybe Manager
   63   , cdApprovals   :: ApprovalCache
   64   , cdCursors     :: CursorStore
   65     -- ^ Per-conversation tab cursors. Each conversation (Telegram chat,
   66     -- Signal conversation) has its own cursor into the shared tab list.
   67   , cdReplies     :: ReplyRegistry
   68     -- ^ Per-session reply fan-out registry. Channels subscribe their
   69     -- 'ChannelHandle' when they focus a tab; replies are fanned out to
   70     -- all subscribed handles after each turn.
   71   , cdLocks       :: SessionLocks
   72     -- ^ Per-session write locks. Serializes concurrent turns on the same
   73     -- session to prevent transcript corruption.
   74   , cdAbortReg     :: SessionAbortRegistry
   75     -- ^ Per-session abort registry. The channel @\/stop@ command calls
   76     -- 'setSessionAbort' on this; the turn path looks up the per-session
   77     -- 'AbortFlag' via 'lookupOrCreateAbortFlag' and passes it into
   78     -- 'mkSessionAgentEnv' as 'aeAbortFlag'. Mirrors 'cdLocks'.
   79   , cdTabs        :: TabsHandle
   80     -- ^ The shared, unified tab handle. Under @seal serve@, this is the SAME
   81     -- handle as the web gateway's 'adTabsHandle', so a tab inserted by any
   82     -- channel is visible in the web sidebar.
   83   , cdConfig      :: IO RuntimeConfig
   84     -- ^ Load the current config (re-read per turn so config changes take
   85     -- effect without a restart). Used for default provider/model/agent
   86     -- when creating a new session for a conversation.
   87   , cdIsRemote    :: Bool
   88     -- ^ Whether the untrusted executor runs commands over SSH (remote
   89     -- mode from the security config). Set once at startup from
   90     -- @isJust (untrustedExecConfigFromSecurity secCfg)@. Threaded into
   91     -- 'CloneDeps' so the deploy-key clone path knows to use agent
   92     -- forwarding (@ssh -A@) + a remote @known_hosts@ temp file.
   93   , cdLogger      :: SealLogger
   94     -- ^ The shared logger for structured katip logging. Built once at
   95     -- startup via 'withSealLogger', threaded through all channel turns.
   96   , cdExecCache   :: SessionExecCache
   97     -- ^ The per-process session-exec + workdir-discovery cache (created by
   98     -- 'newChannelDeps'). Under @seal serve@ the SAME instance backs the
   99     -- web 'SendDeps'/'ApiDeps' so a scan runs once per session across all
  100     -- surfaces.
  101   }

  103 -- | Build a 'ChannelDeps' with fresh reply/lock/abort stores and the given
  104 -- config loader. Used by 'Seal.Command.Serve'. The 'tabsH' is the
  105 -- shared/unified handle. The 'cursors' is supplied by the caller: a
  106 -- persisting store ('Seal.Channels.Cursor.newPersistingCursorStore') under
  107 -- @seal serve@ so the conversation→tab bindings survive a restart, or a
  108 -- non-persisting 'newCursorStore' in tests. The caller is responsible for
  109 -- loading + seeding the cursor store at boot (see
  110 -- 'Seal.Channels.Cursor.Persist.loadCursorMap' + 'seedCursorStore'); this
  111 -- function does NOT touch disk for cursors.
  112 newChannelDeps
  113   :: SealPaths -> VaultRuntime -> RepoRegistryHandle -> AgentRegistryHandle -> ProviderRuntime -> Backends
  114   -> Policy.AutonomyLevel -> Maybe StreamBroker
  115   -> HarnessRegistry -> TmuxRunner -> Maybe Manager
  116   -> ApprovalCache -> IO RuntimeConfig
  117   -> Bool
  118   -> TabsHandle
  119   -> SealLogger
  120   -> CursorStore
  121   -> IO ChannelDeps
  122 newChannelDeps paths vault repoReg agentReg provider backends autonomy broker
  123                harnessReg tmux httpMgr approvals loadCfg isRemote tabsH logger cursors = do
  124   replies <- newReplyRegistry
  125   locks   <- newSessionLocks
  126   abortReg <- newSessionAbortRegistry
  127   execCache <- newSessionExecCache
  128   pure ChannelDeps
  129     { cdPaths      = paths
  130     , cdVault      = vault
  131     , cdRepoReg    = repoReg
  132     , cdAgentReg   = agentReg
  133     , cdProvider   = provider
  134     , cdBackends   = backends
  135     , cdAutonomy   = autonomy
  136     , cdBroker     = broker
  137     , cdHarnessRegistry = harnessReg
  138     , cdTmuxRunner  = tmux
  139     , cdHttpManager = httpMgr
  140     , cdApprovals   = approvals
  141     , cdCursors     = cursors
  142     , cdReplies     = replies
  143     , cdLocks       = locks
  144     , cdAbortReg    = abortReg
  145     , cdTabs        = tabsH
  146     , cdConfig      = loadCfg
  147     , cdIsRemote    = isRemote
  148     , cdLogger      = logger
  149     , cdExecCache   = execCache
  150     }

  152 -- | Build a 'TabCloseNotifier' from the shared cursor store + reply
  153 -- registry. When a tab is closed, every conversation whose cursor points
  154 -- at the closed tab's session is notified (via 'replyFanout') and the
  155 -- cursor is cleared (so the next message mints a fresh tab). For
  156 -- 'BoundHarness' tabs there are no channel subscriptions, so this is a
  157 -- no-op. Used by the slash-command registry's @\/tab close@ and the REST
  158 -- @POST \/api\/tabs\/:index\/close@ path.
  159 mkTabCloseNotifier :: CursorStore -> ReplyRegistry -> TabCloseNotifier
  160 mkTabCloseNotifier cursors replies ref = case ref of
  161   BoundSession sid -> do
  162     replyFanout replies sid (msg sid)
  163     cursorClearAll cursors ref
  164   BoundHarness _ -> pure ()
  165   where
  166     msg sid = "tab closed (session " <> sessionIdText sid <> "); a new tab will be created on your next message"

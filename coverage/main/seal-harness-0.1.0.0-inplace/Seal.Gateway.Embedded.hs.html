    1 {-# LANGUAGE OverloadedStrings #-}
    2 {-# LANGUAGE TemplateHaskell #-}
    3 -- | The web frontend's build output (frontend/dist), embedded into the
    4 -- binary at compile time so @seal serve@ needs no on-disk static dir and
    5 -- no config. Same philosophy as "Seal.Skills.Builtins": the UI is always
    6 -- the version the binary shipped with — self-contained, no first-run
    7 -- materialization, no install-layout dependency, no staleness across
    8 -- upgrades.
    9 module Seal.Gateway.Embedded
   10   ( embeddedAssets
   11   , embeddedAssetCount
   12   , distManifest
   13   ) where
   14 
   15 import Data.ByteString (ByteString)
   16 import Data.FileEmbed (embedDir, embedOneStringFileOf)
   17 import Data.Map.Strict (Map)
   18 import Data.Map.Strict qualified as Map
   19 import Language.Haskell.TH.Syntax (runIO)
   20 import System.Directory (doesDirectoryExist)
   21 
   22 -- | Frontend bytes keyed by forward-slash path relative to the dist root
   23 -- (e.g. @"index.html"@, @"assets\/index-De3F1a.js"@).
   24 --
   25 -- @embedDir@ properties (verified in file-embed source):
   26 --   * Recursive, sorted output.
   27 --   * Dot-prefixed entries skipped (so @.seal-manifest@ never appears).
   28 --   * Every embedded file registered with @qAddDependentFile@ (called
   29 --     inside @pairToExp@ in file-embed's @embedDir@ implementation).
   30 --   * Keys are relative paths joined with @System.FilePath.(<\/>)@ — which
   31 --     uses @\"\/\"@ on Unix (verified: @fileList'@ builds keys via
   32 --     @top <\/> filename@ where @top@ starts as @\"\"@). This matches the
   33 --     @T.intercalate \"\/\"@ lookup in 'Seal.Gateway.Server.serveEmbedded'.
   34 --     Target platforms are Unix-only (nix, macOS, Linux).
   35 embeddedAssets :: Map FilePath ByteString
   36 embeddedAssets = Map.fromList
   37   $(do exists <- runIO (doesDirectoryExist "frontend/dist")
   38        if exists
   39          then embedDir "frontend/dist"
   40          else [| [] |]
   41   )
   42 
   43 -- | Sentinel written by @make frontend-build@ (a hash over the dist tree).
   44 -- Changes on every UI build, forcing this module to recompile so the
   45 -- @embedDir@ splice re-runs. This closes file-embed's documented caveat
   46 -- that /newly added/ files don't reliably retrigger recompilation — which
   47 -- is the default case for us, since Vite emits new hash-named files each
   48 -- build. Fallback file (committed) keeps @cabal build@ working from a
   49 -- clean checkout with no dist.
   50 --
   51 -- Exported (not just a side-effect splice) to avoid @-Wunused-top-binds@
   52 -- under @-Wall -Werror@. Also useful for debugging: the manifest hash
   53 -- identifies which UI build is embedded in a running binary.
   54 distManifest :: String
   55 distManifest = $(embedOneStringFileOf
   56   [ "frontend/dist/.seal-manifest"
   57   , "config/frontend-dist-absent.txt"
   58   ])
   59 
   60 -- | Number of embedded assets. Equal to 'Map.size' of 'embeddedAssets'.
   61 embeddedAssetCount :: Int
   62 embeddedAssetCount = Map.size embeddedAssets

    1 {-# LANGUAGE OverloadedStrings #-}
    2 {-# LANGUAGE TemplateHaskell #-}
    3 -- | First-run scaffolding for the two operator-editable configuration files.
    4 --
    5 -- A fresh install has no @config.toml@ and no @security.toml@, and both loaders
    6 -- treat an absent file as "use the built-in defaults". Seal works that way, but
    7 -- the option surface is then invisible until someone reads the source. This
    8 -- module ships the surface: each template lists the keys its codec accepts,
    9 -- with the value Seal would use — written out where a written value resolves to
   10 -- the same behaviour as an absent key, commented out where it does not (the
   11 -- default is "unset", the value names something only the operator knows, or the
   12 -- value is machine-absolute).
   13 --
   14 -- The templates are embedded with 'file-embed', the mechanism
   15 -- 'Seal.Skills.Builtins' already uses for built-in skills, so a binary always
   16 -- carries the version of the file it shipped with: no install-layout
   17 -- dependency, no first-run download, no staleness across upgrades.
   18 --
   19 -- Two invariants, both load-bearing:
   20 --
   21 -- * __Never overwrite.__ An existing file is the operator's (or an earlier
   22 --   run's, or a program's). A scaffold that clobbered it would undo hand edits
   23 --   on every boot.
   24 -- * __Never change behaviour.__ Writing a template has to resolve to exactly
   25 --   the configuration of leaving both files absent.
   26 --   @Seal.Config.TemplatesSpec@ asserts both. The @security.toml@ half is also
   27 --   skipped while @config.toml@ still carries pre-split security fields:
   28 --   creating the file there would make 'Seal.Config.Migrate' read the install as
   29 --   already migrated — its "security.toml wins" branch — and drop the operator's
   30 --   vault settings.
   31 module Seal.Config.Templates
   32   ( configTomlTemplate
   33   , securityTomlTemplate
   34   , ensureConfigTemplates
   35   ) where
   36

   59 -- | The bytes of @config\/templates\/config.toml@, embedded at compile time.
   60 configTomlTemplate :: Text
   61 configTomlTemplate = TE.decodeUtf8 $(do
   62   qAddDependentFile "config/templates/config.toml"
   63   embedFile "config/templates/config.toml")
   64 
   65 -- | The bytes of @config\/templates\/security.toml@, embedded at compile time.
   66 securityTomlTemplate :: Text
   67 securityTomlTemplate = TE.decodeUtf8 $(do
   68   qAddDependentFile "config/templates/security.toml"
   69   embedFile "config/templates/security.toml")
   70 
   71 -- | How a template is written.
   72 --
   73 -- @security.toml@ holds boot-critical state, so it is created @0600@ with
   74 -- @O_NOFOLLOW@ through 'writeSecretFile' — the same primitive its other writers
   75 -- use. The runtime config is ordinary data and is written like
   76 -- 'Seal.Config.File.saveRuntimeConfig' writes it.
   77 data FileSensitivity = SecretFile | PlainFile
   78   deriving stock (Eq, Show)
   79 
   80 -- | Write whichever of the two templates is missing.
   81 --
   82 -- Returns the files it created and the problems it hit. A failure is per-file:
   83 -- an unwritable @config.toml@ does not stop @security.toml@ from being written,
   84 -- and neither failure is thrown at the caller — a missing template must never
   85 -- stop the harness from booting.
   86 ensureConfigTemplates :: SealPaths -> IO ([FilePath], [Text])
   87 ensureConfigTemplates paths = do
   88   cfgOutcome <- seedFile (configFilePath paths) configTomlTemplate PlainFile
   89   secOutcome <- seedSecurity paths
   90   let created = catMaybes (rights [cfgOutcome, secOutcome])
   91       problems = lefts [cfgOutcome, secOutcome]
   92   pure (created, problems)
   93 
   94 -- | @security.toml@, unless a migration still owns the operator's settings.
   95 seedSecurity :: SealPaths -> IO (Either Text (Maybe FilePath))
   96 seedSecurity paths = do
   97   carries <- configCarriesLegacyFields (configFilePath paths)
   98   case carries of
   99     Left err -> pure (Left err)
  100     -- Defer. `seal serve` and `seal tui` run 'Seal.Config.Migrate.migrateSecurityConfig'
  101     -- before this, and the migration moves those fields into security.toml.
  102     -- Creating the file first would flip the migration's file-existence test to
  103     -- its "security.toml wins" branch and silently drop them.
  104     Right True  -> pure (Right Nothing)
  105     Right False -> seedFile (securityFilePath paths) securityTomlTemplate SecretFile
  106 
  107 -- | Does the existing @config.toml@ still carry pre-split security fields?
  108 configCarriesLegacyFields :: FilePath -> IO (Either Text Bool)
  109 configCarriesLegacyFields path = do
  110   exists <- doesFileExist path
  111   if not exists
  112     then pure (Right False)
  113     else do
  114       read' <- try (TIO.readFile path) :: IO (Either IOException Text)
  115       pure $ case read' of
  116         Left err       -> Left (describeReadError path err)
  117         Right contents -> Right (configHasLegacySecurityFields contents)
  118 
  119 -- | Create @path@ with @contents@ if nothing is there. 'Nothing' means there
  120 -- was already a file; 'Left' means the write failed.
  121 seedFile :: FilePath -> Text -> FileSensitivity -> IO (Either Text (Maybe FilePath))
  122 seedFile path contents sensitivity = do
  123   outcome <- try go :: IO (Either IOException (Maybe FilePath))
  124   pure $ case outcome of
  125     Left err      -> Left (describeWriteError path err)
  126     Right created -> Right created
  127   where
  128     go :: IO (Maybe FilePath)
  129     go = do
  130       exists <- doesFileExist path
  131       if exists
  132         then pure Nothing
  133         else do
  134           -- The parent directory normally exists by now ('Seal.Config.Paths.ensureSealDirs'
  135           -- runs first at every call site); creating it here keeps the whole
  136           -- write inside one handler, so a missing directory is a reported
  137           -- failure rather than an exception.
  138           createDirectoryIfMissing True (takeDirectory path)
  139           -- Never rename over the destination: a file created between the
  140           -- existence check above and this install must survive. Write a
  141           -- unique temp, then hard-link it into place; EEXIST is a benign lost
  142           -- race, not a write failure.
  143           bracketOnError (openUniqueTemp path) cleanup $ \(tmp, h) -> do
  144             BS.hPutStr h (TE.encodeUtf8 contents)
  145             hClose h
  146             case sensitivity of
  147               SecretFile -> setFileMode tmp 0o600
  148               PlainFile  -> pure ()
  149             installed <- try (createLink tmp path) :: IO (Either IOException ())
  150             ignoreIO (removeFile tmp)
  151             case installed of
  152               Right () -> pure (Just path)
  153               Left err
  154                 | isAlreadyExistsError err -> pure Nothing
  155                 | otherwise               -> ioError err
  156 
  157     cleanup (tmp, h) = do
  158       ignoreIO (hClose h)
  159       ignoreIO (removeFile tmp)
  160 
  161     ignoreIO action = void (try action :: IO (Either IOException ()))
  162 
  163 describeWriteError :: FilePath -> IOException -> Text
  164 describeWriteError path err =
  165   "could not write " <> T.pack path <> ": " <> T.pack (displayException err)
  166 
  167 describeReadError :: FilePath -> IOException -> Text
  168 describeReadError path err =
  169   "could not read " <> T.pack path <> ": " <> T.pack (displayException err)

    1 -- | The Dynamic Retrieval page-sizer and a generic 'paginate'.
    2 --
    3 -- File- and ISA-agnostic. The sizer computes a page size as
    4 -- @clamp floor ceiling (round (coeff * sqrt total))@, with a per-call
    5 -- explicit limit overriding the computed size (clamped to
    6 -- @[1, ppCeiling]@ so a caller can never request an unbounded window).
    7 module Seal.Core.Paging
    8   ( PageParams (..)
    9   , Page (..)
   10   , clamp
   11   , pageSize
   12   , windowSize
   13   , paginate
   14   , defaultPageParams
   15   ) where
   16 
   17 -- | Sizer parameters. Invariants: @1 <= ppFloor <= ppCeiling@ and
   18 -- @ppCoeff >= 0@.
   19 data PageParams = PageParams
   20   { ppFloor   :: !Int   -- ^ minimum page size (invariant: @1 <= ppFloor <= ppCeiling@)
   21   , ppCeiling :: !Int   -- ^ maximum page size
   22   , ppCoeff   :: !Double -- ^ @A@ in @round(A * sqrt total)@ (invariant: @>= 0@)
   23   } deriving stock (Eq, Show)
   24 
   25 -- | A page of results plus the metadata a model needs to page forward.
   26 data Page a = Page
   27   { pgItems   :: [a]   -- ^ the windowed items, in input order
   28   , pgOffset  :: !Int  -- ^ 0-based offset this page starts at (clamped to @[0,total]@)
   29   , pgTotal   :: !Int  -- ^ total item count (== length of the input list)
   30   , pgHasMore :: !Bool -- ^ @pgOffset + length pgItems < pgTotal@
   31   } deriving stock (Eq, Show)
   32 
   33 -- | @clamp lo hi x = max lo (min hi x)@. Value last, matching
   34 -- @Data.Ord.clamp (lo,hi) x@.
   35 clamp :: Int -> Int -> Int -> Int
   36 clamp lo hi x = max lo (min hi x)
   37 
   38 -- | @pageSize params total = clamp ppFloor ppCeiling (round (ppCoeff * sqrt total))@.
   39 -- Result is always within @[ppFloor, ppCeiling]@.
   40 pageSize :: PageParams -> Int -> Int
   41 pageSize (PageParams floor' ceiling' coeff) total =
   42   clamp floor' ceiling' (round (coeff * sqrt (fromIntegral total :: Double)))
   43 
   44 -- | The single source of truth for "how many items to return", shared by
   45 -- 'paginate' (list path) and 'Seal.Text.LineFile.readLineWindow'
   46 -- (streaming path) so they cannot drift.
   47 --
   48 -- @windowSize params total mLimit = maybe (pageSize params total) (clamp 1 ppCeiling) mLimit@
   49 windowSize :: PageParams -> Int -> Maybe Int -> Int
   50 windowSize params@(PageParams _ ceiling' _) total mLimit =
   51   case mLimit of
   52     Nothing  -> pageSize params total
   53     Just lim -> clamp 1 ceiling' lim
   54 
   55 -- | @paginate params offset mLimit items@, where @total = length items@.
   56 --
   57 --   * @offset'  = clamp 0 total offset@
   58 --   * @size     = windowSize params total mLimit@
   59 --   * @window   = take size (drop offset' items)@
   60 paginate :: PageParams -> Int -> Maybe Int -> [a] -> Page a
   61 paginate params offset mLimit items =
   62   let total   = length items
   63       offset' = clamp 0 total offset
   64       size    = windowSize params total mLimit
   65       window  = take size (drop offset' items)
   66   in Page
   67        { pgItems   = window
   68        , pgOffset  = offset'
   69        , pgTotal   = total
   70        , pgHasMore = offset' + length window < total
   71        }
   72 
   73 -- | 'PageParams' used everywhere in this milestone.
   74 -- @PageParams { ppFloor = 500, ppCeiling = 2000, ppCoeff = 0.0 }@.
   75 -- A flat 500-line default (matching Hermes' read_file), with a 2000-line
   76 -- hard ceiling. @ppCoeff = 0@ means @pageSize = clamp 500 2000 0 = 500@
   77 -- regardless of total line count — no dynamic sqrt scaling.
   78 defaultPageParams :: PageParams
   79 defaultPageParams = PageParams { ppFloor = 500, ppCeiling = 2000, ppCoeff = 0.0 }

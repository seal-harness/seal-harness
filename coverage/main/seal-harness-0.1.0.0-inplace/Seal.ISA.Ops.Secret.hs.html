    2 -- | SECRET_MANAGE (Audited): full CRUD for vault secrets via a single
    3 -- action-based opcode. Actions: @get@ (read a secret), @put@ (create or
    4 -- update a secret — upsert), @delete@ (remove a secret, ALWAYS with human
    5 -- approval), @list@ (enumerate key names only).
    6 --
    7 -- Security: secret values returned by @get@ are passed to the model via
    8 -- 'orParts' but NEVER appear in 'orRecorded' (the transcript payload).
    9 -- The @put@ action likewise never records the value. The @delete@ action
   10 -- prompts the human operator via 'ChannelCaps.ccPrompt' before proceeding;
   11 -- if the human denies, the secret is left intact and an error result is
   12 -- returned. The @list@ action returns only key names, never values.
   13 module Seal.ISA.Ops.Secret

   15   , vaultGetByName
   16   , vaultPutByName
   17   , vaultDeleteByName

   43 data SecretAction
   44   = SecretGet
   45   | SecretPut
   46   | SecretDelete
   47   | SecretList
   48 
   49 parseSecretAction :: Value -> Either Text SecretAction
   50 parseSecretAction v =
   51   case parseMaybe (withObject "in" (.: "action")) v of
   52     Just t -> case t :: Text of
   53       "get"    -> Right SecretGet
   54       "put"    -> Right SecretPut
   55       "delete" -> Right SecretDelete
   56       "list"   -> Right SecretList
   57       other    -> Left ("unknown secret action: " <> other)
   58     Nothing -> Left "missing or invalid action field"
   59

   74 -- | Look up a secret by name from the vault handle in the runtime.
   75 -- Returns 'Left' with a human-readable message if the vault is unconfigured
   76 -- or the key is absent; 'Right' with the UTF-8-decoded value otherwise.
   77 -- The decoded 'Text' is passed directly to 'orParts' and must not reach
   78 -- 'orRecorded'.
   79 vaultGetByName :: VaultRuntime -> Text -> IO (Either Text Text)
   80 vaultGetByName rt key = do
   81   mh <- readIORef (vrHandleRef rt)
   82   case mh of
   83     Nothing -> pure (Left "vault not configured — run /vault setup")
   84     Just h -> do
   85       r <- vhGet h key
   86       pure $ case r of
   87         Left e  -> Left (T.pack (show e))
   88         Right bs -> Right (TE.decodeUtf8Lenient bs)
   89 
   90 -- | Store a secret value under the given key (upsert). Returns 'Left' on
   91 -- vault errors. The value is never included in the returned 'Text'.
   92 vaultPutByName :: VaultRuntime -> Text -> Text -> IO (Either Text ())
   93 vaultPutByName rt key val = do
   94   mh <- readIORef (vrHandleRef rt)
   95   case mh of
   96     Nothing -> pure (Left "vault not configured — run /vault setup")
   97     Just h -> do
   98       r <- vhPut h key (TE.encodeUtf8 val)
   99       pure $ case r of
  100         Left e  -> Left (T.pack (show e))
  101         Right _ -> Right ()
  102 
  103 -- | Delete a secret by key. Returns 'Left' if the vault is unconfigured or
  104 -- the key is absent.
  105 vaultDeleteByName :: VaultRuntime -> Text -> IO (Either Text ())
  106 vaultDeleteByName rt key = do
  107   mh <- readIORef (vrHandleRef rt)
  108   case mh of
  109     Nothing -> pure (Left "vault not configured — run /vault setup")
  110     Just h -> do
  111       r <- vhDelete h key
  112       pure $ case r of
  113         Left e  -> Left (T.pack (show e))
  114         Right _ -> Right ()
  115 
  116 -- | List all secret key names in the vault. Values are never returned.
  117 vaultListNames :: VaultRuntime -> IO (Either Text [Text])
  118 vaultListNames rt = do
  119   mh <- readIORef (vrHandleRef rt)
  120   case mh of
  121     Nothing -> pure (Left "vault not configured — run /vault setup")
  122     Just h -> do
  123       r <- vhList h
  124       pure $ case r of
  125         Left e   -> Left (T.pack (show e))
  126         Right ks -> Right ks

  128 -- ---------------------------------------------------------------------------
  129 -- Authorization
  130 -- ---------------------------------------------------------------------------
  131 
  132 authorizeGet :: Value -> Either Text ()
  133 authorizeGet v =
  134   maybe (Left "get requires {name:string}") (const (Right ())) (nameField v)
  135 
  136 authorizePut :: Value -> Either Text ()
  137 authorizePut v =
  138   case nameField v of
  139     Nothing -> Left "put requires {name:string}"
  140     Just _ -> case valueField v of
  141       Nothing -> Left "put requires {value:string}"
  142       Just val | T.null val -> Left "put requires non-empty value"
  143                | otherwise  -> Right ()
  144 
  145 authorizeDelete :: Value -> Either Text ()
  146 authorizeDelete v =
  147   maybe (Left "delete requires {name:string}") (const (Right ())) (nameField v)
  148 
  149 authorizeList :: Value -> Either Text ()
  150 authorizeList _ = Right ()
  151 
  152 -- | Authorize gate for SECRET_MANAGE — dispatches per-action validation.
  153 authorizeManage :: Value -> Either Text ()
  154 authorizeManage v =
  155   case parseSecretAction v of
  156     Left e        -> Left e
  157     Right action  -> case action of
  158       SecretGet    -> authorizeGet v
  159       SecretPut    -> authorizePut v
  160       SecretDelete -> authorizeDelete v
  161       SecretList   -> authorizeList v
  162 
  163 -- ---------------------------------------------------------------------------
  164 -- Handlers
  165 -- ---------------------------------------------------------------------------
  166 
  167 handleGet :: VaultRuntime -> Value -> App OpResult
  168 handleGet rt v = do
  169   let key = fromMaybe "" (nameField v)
  170   result <- liftIO (vaultGetByName rt key)
  171   pure $ case result of
  172     Left err ->
  173       OpResult [TrpText err] True (object ["name" .= key])
  174     Right secret ->
  175       OpResult [TrpText secret] False (object ["name" .= key])
  176 
  177 handlePut :: VaultRuntime -> Value -> App OpResult
  178 handlePut rt v = do
  179   let key = fromMaybe "" (nameField v)
  180       val = fromMaybe "" (valueField v)
  181   result <- liftIO (vaultPutByName rt key val)
  182   pure $ case result of
  183     Left err ->
  184       OpResult [TrpText err] True (object ["name" .= key])
  185     Right _ ->
  186       OpResult [TrpText "stored"] False (object ["name" .= key])
  187 
  188 handleDelete :: VaultRuntime -> ChannelCaps -> Value -> App OpResult
  189 handleDelete rt caps v = do
  190   let key = fromMaybe "" (nameField v)
  191       prompt = "Delete vault secret \"" <> key <> "\"? This cannot be undone."
  192   ans <- liftIO (ccPrompt caps (AskPrompt prompt []))
  193   -- The confirmation gate (web UI) returns "once"/"for_session"/"always"
  194   -- for approval and "rejected" for denial. The CLI returns typed text.
  195   -- Treat any non-empty, non-rejection response as approval.
  196   if T.toLower (T.strip ans) `elem` ["rejected", "", "no", "n", "cancel", "denied"]
  197     then pure (OpResult [TrpText "delete denied by operator"] True (object ["name" .= key]))
  198     else do
  199       result <- liftIO (vaultDeleteByName rt key)
  200       pure $ case result of
  201         Left err ->
  202           OpResult [TrpText err] True (object ["name" .= key])
  203         Right _ ->
  204           OpResult [TrpText "deleted"] False (object ["name" .= key])
  205 
  206 handleList :: VaultRuntime -> Value -> App OpResult
  207 handleList rt _v = do
  208   result <- liftIO (vaultListNames rt)
  209   pure $ case result of
  210     Left err ->
  211       OpResult [TrpText err] True (object ["count" .= (0 :: Int)])
  212     Right [] ->
  213       OpResult [TrpText "No secrets."] False (object ["count" .= (0 :: Int)])
  214     Right keys -> do
  215       let rendered = T.intercalate "\n" keys
  216             <> "\n---\n" <> T.pack (show (length keys)) <> " secrets"
  217       OpResult [TrpText rendered] False (object ["count" .= length keys])

  223 -- | SECRET_MANAGE: action-based entry point for all vault secret operations.
  224 -- The @action@ field discriminates between @get@, @put@, @delete@, and
  225 -- @list@. The @delete@ action ALWAYS prompts the human operator for
  226 -- approval before proceeding — it is a blocking opcode.
  227 secretManageOp :: VaultRuntime -> ChannelCaps -> Opcode
  228 secretManageOp rt caps = TrustedOpcode
  229   { toName = OpName "SECRET_MANAGE"
  230   , toTrust = Trusted
  231   , toDesc = "Manage vault secrets. Use action to select: get (fetch a secret value by key name), put (create or update a secret — upsert), delete (remove a secret — ALWAYS prompts for human approval), list (enumerate key names only)."
  232   , toInSchema = object
  233       [ "type" .= ("object" :: Text)
  234       , "properties" .= object
  235           [ fromText "action" .= object
  236               [ "type" .= ("string" :: Text)
  237               , "enum" .= (["get", "put", "delete", "list"] :: [Text])
  238               , "description" .= ("Operation to perform." :: Text)
  239               ]
  240           , fromText "name" .= object
  241               [ "type" .= ("string" :: Text)
  242               , "description" .= ("The vault key name of the secret (get, put, delete)." :: Text)
  243               ]
  244           , fromText "value" .= object
  245               [ "type" .= ("string" :: Text)
  246               , "description" .= ("The secret value to store (put only). Never recorded in the transcript." :: Text)
  247               ]
  248           ]
  249       , "required" .= (["action"] :: [Text])
  250       ]
  251   , toOutSchema = object []
  252   , toAuthorize = authorizeManage
  253   , toBlocking = True
  254   , toRun = \_ v ->
  255       case parseSecretAction v of
  256         Left e -> pure (OpResult [TrpText e] True (object []))
  257         Right action -> case action of
  258           SecretGet    -> handleGet rt v
  259           SecretPut    -> handlePut rt v
  260           SecretDelete -> handleDelete rt caps v
  261           SecretList   -> handleList rt v
  262   }

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | The per-flavour harness observer: classify a captured screen + the
    3 -- tmux marker map into a 'Liveness'. Pure (the capture is IO, the
    4 -- classification is pure). Only the Claude Code screen-capture heuristic
    5 -- lands in 6a; a codex/generic observer is a follow-up.
    6 module Seal.Harness.Observer
    7   ( observeClaudeCode
    8   , livenessToActivity
    9   ) where
   10 
   11 import Data.Map.Strict (Map)
   12 import Data.Map.Strict qualified as Map
   13 import Data.Text (Text)
   14 import Data.Text qualified as T
   15 
   16 import Seal.Harness.Registry (Liveness (..))
   17 
   18 -- | Classify a captured Claude Code screen + the marker map into a
   19 -- 'Liveness'. Heuristic:
   20 --
   21 -- * 'LvExited' when the @seal_id marker is gone AND the pane is dead
   22 --   (the @pane_dead@ marker is present).
   23 -- * 'LvOrphaned' when the pane exists but the @seal_id marker is gone
   24 --   (someone renamed the window out from under us, or the harness crashed
   25 --   and the pane is in a stale state).
   26 -- * 'LvThinking' when the screen shows a spinner / "Thinking…" / "Working…".
   27 -- * 'LvAwaitingInput' when the screen shows the Claude Code prompt (@> @
   28 --   at the bottom + no spinner) or a yes/no confirmation.
   29 -- * 'LvIdle' otherwise (at the main prompt but not awaiting input).
   30 observeClaudeCode :: Text -> Map Text Text -> Liveness
   31 observeClaudeCode screen markers
   32   | not hasSealId && paneDead        = LvExited
   33   | not hasSealId                    = LvOrphaned
   34   | T.isInfixOf "Thinking" stripped  = LvThinking
   35   | T.isInfixOf "Working" stripped   = LvThinking
   36   | isAwaiting screen                 = LvAwaitingInput
   37   | otherwise                        = LvIdle
   38   where
   39     hasSealId = Map.member "seal_id" markers
   40     paneDead  = Map.lookup "pane_dead" markers == Just "1"
   41     stripped  = T.strip screen
   42     isAwaiting s =
   43       let ll = T.strip (lastLine s)
   44       in ll == ">" || ll == ">"
   45          || T.isInfixOf "? (y/n)" s
   46          || T.isInfixOf "? Yes/No" s
   47          || T.isInfixOf "Press Enter" s
   48 
   49 -- | The last non-empty line of a captured screen.
   50 lastLine :: Text -> Text
   51 lastLine s = case filter (not . T.null) (T.lines s) of
   52   []  -> ""
   53   xs  -> last xs
   54 
   55 -- | Map a 'Liveness' to the activity-stream tag the frontend (Phase 7)
   56 -- consumes.
   57 livenessToActivity :: Liveness -> Text
   58 livenessToActivity = \case
   59   LvIdle          -> "idle"
   60   LvThinking      -> "thinking"
   61   LvAwaitingInput -> "awaiting_input"
   62   LvExited        -> "exited"
   63   LvOrphaned      -> "orphaned"

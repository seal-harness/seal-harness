    2 -- | Pure context window management: token estimation and message truncation.
    3 --
    4 -- The agent loop ('Seal.Agent.Loop') calls these functions before building a
    5 -- 'CompletionRequest' to ensure the conversation fits within the model's
    6 -- context window. All functions are pure — no IO, no 'AgentEnv' dependency —
    7 -- so they can be tested independently.
    8 module Seal.Agent.ContextTruncation
    9   ( estimateTokenCount
   10   , truncateMessages
   11   , aggressiveTruncate

   13   , defaultTruncationConfig
   14   , aggressiveKeepCount
   15   , truncationNotice
   16   ) where

   25 -- | Configuration for the truncation strategy.
   26 data TruncationConfig = TruncationConfig
   27   { tcKeepRecent :: Int
   28     -- ^ The number of most-recent messages to always preserve (the "active
   29     -- context window"). Default: 20.
   30   , tcMinKeep :: Int
   31     -- ^ The minimum number of messages to keep, even if they exceed the
   32     -- budget. Prevents pathological cases where even the last few messages
   33     -- are too large. Default: 2.
   34   }
   35 
   36 -- | Default truncation config: keep 20 recent messages, minimum 2.
   37 defaultTruncationConfig :: TruncationConfig
   38 defaultTruncationConfig = TruncationConfig
   39   { tcKeepRecent = 20
   40   , tcMinKeep = 2
   41   }
   42 
   43 -- | The number of messages to keep in aggressive truncation mode (used
   44 -- for prompt-too-long error recovery).
   45 aggressiveKeepCount :: Int
   46 aggressiveKeepCount = 5
   47 
   48 -- | The synthetic message inserted when older context is truncated, so the
   49 -- model knows history is missing. Uses 'User' role because the provider-
   50 -- agnostic 'Message' type has no @System@ role (system prompt is a separate
   51 -- 'crSystem' field).
   52 truncationNotice :: Text
   53 truncationNotice =
   54   "[System: Older conversation context was truncated to fit the model's \
   55   \context window. Some earlier messages are no longer available.]"
   56 
   57 -- | Estimate the token count for a completion request's components: system
   58 -- prompt + tool definitions + messages. Uses the chars/4 approximation
   59 -- (the standard rough estimate; a proper tokenizer is out of scope for MVP).
   60 estimateTokenCount :: Maybe Text -> [Message] -> [ToolDefinition] -> Int
   61 estimateTokenCount mSystem msgs tools =
   62   systemTokens + toolTokens + messageTokens
   63   where
   64     systemTokens = maybe 0 textTokens mSystem
   65     toolTokens = sum (map toolTokens' tools)
   66     messageTokens = sum (map messageTokens' msgs)
   67     -- chars / 4, minimum 1 for non-empty text
   68     textTokens t = if T.null t then 0 else max 1 (T.length t `div` 4)
   69     -- Estimate tool tokens from JSON encoding (tool definitions are already
   70     -- ToJSON; their wire size is what the provider sees).
   71     toolTokens' td =
   72       let bs = A.encode td
   73           charLen = BL.length bs
   74       in if charLen == 0 then 0 else max 1 (fromIntegral charLen `div` 4)
   75     -- Sum the text/thinking/tool content of a message.
   76     messageTokens' (Message _ blocks) =
   77       sum (map blockTokens' blocks)
   78     blockTokens' (CbText t) = textTokens t
   79     blockTokens' (CbThinking t) = textTokens t
   80     blockTokens' (CbToolUse _ _ v) =
   81       let bs = A.encode v
   82           charLen = BL.length bs
   83       in if charLen == 0 then 0 else max 1 (fromIntegral charLen `div` 4)
   84     blockTokens' (CbToolResult _ parts _) =
   85       sum (map partTokens' parts)
   86     partTokens' (TrpText t) = textTokens t
   87 
   88 -- | Truncate the message list to fit within the given token budget (for
   89 -- the message portion only — system and tools are accounted for separately
   90 -- by the caller). When messages fit, they are returned unchanged. When they
   91 -- don't, the oldest messages are dropped from the middle, a synthetic
   92 -- truncation notice is inserted, and the most recent messages are preserved.
   93 -- The last tool-use + tool-result pair is never split (if the tool-result
   94 -- is kept, the preceding tool-use is also kept).
   95 truncateMessages :: TruncationConfig -> Int -> [Message] -> [Message]
   96 truncateMessages cfg budget msgs
   97   | currentTokens <= budget = msgs
   98   | otherwise = truncated
   99   where
  100     currentTokens = estimateTokenCount Nothing msgs []
  101     -- Take the last tcKeepRecent messages, but ensure we don't split a
  102     -- tool-use/tool-result pair. If the first kept message is a tool-result
  103     -- (User with CbToolResult), include the preceding message too.
  104     keepN = tcKeepRecent cfg
  105     rawKept = takeEnd keepN msgs
  106     kept = alignToolPair (dropEnd keepN msgs) rawKept
  107     -- Check if the kept messages fit; if not, progressively drop from front
  108     keptTokens = estimateTokenCount Nothing kept []
  109     finalKept = if keptTokens <= budget
  110                   then kept
  111                   else shrinkToFit budget (tcMinKeep cfg) kept
  112     -- Insert the truncation notice before the kept messages
  113     truncated = truncationNoticeMsg : finalKept
  114 
  115 -- | Aggressively truncate: keep only the last 'aggressiveKeepCount' messages.
  116 -- Used for prompt-too-long error recovery, where we need to shrink the
  117 -- context as much as possible in one shot. No truncation notice is inserted
  118 -- (the caller handles messaging).
  119 aggressiveTruncate :: [Message] -> [Message]
  120 aggressiveTruncate msgs
  121   | length msgs <= aggressiveKeepCount = msgs
  122   | otherwise = alignToolPair dropped (takeEnd aggressiveKeepCount msgs)
  123   where
  124     dropped = dropEnd aggressiveKeepCount msgs
  125 
  126 -- | If the first message in @kept@ is a tool-result (User role with
  127 -- CbToolResult blocks), include the preceding message from @dropped@ (the
  128 -- Assistant message with the matching tool-use). This prevents splitting a
  129 -- tool-use/tool-result pair across the truncation boundary.
  130 alignToolPair :: [Message] -> [Message] -> [Message]
  131 alignToolPair dropped kept =
  132   case kept of
  133     (m : _) | isToolResult m ->
  134       case takeEnd 1 dropped of
  135         (prev : _) | isToolUse prev -> prev : kept
  136         _ -> kept
  137     _ -> kept
  138 
  139 -- | Shrink the message list from the front until it fits within the budget,
  140 -- but never below minKeep messages.
  141 shrinkToFit :: Int -> Int -> [Message] -> [Message]
  142 shrinkToFit budget minKeep = go
  143   where
  144     go ms
  145       | length ms <= minKeep = ms
  146       | estimateTokenCount Nothing ms [] <= budget = ms
  147       | otherwise = go (dropOne ms)
  148 
  149 -- | Drop one message from the front. If the dropped message was a tool-use
  150 -- whose matching tool-result is the next message, drop both to avoid
  151 -- leaving an orphaned tool-result.
  152 dropOne :: [Message] -> [Message]
  153 dropOne [] = []
  154 dropOne (m : rest)
  155   | isToolUse m = case rest of
  156       (m2 : _) | isToolResult m2 -> drop 1 rest
  157       _ -> rest
  158   | otherwise = rest
  159 
  160 -- | The truncation notice as a 'Message' (User role).
  161 truncationNoticeMsg :: Message
  162 truncationNoticeMsg = textMsg User truncationNotice
  163 
  164 -- | Check if a message is a tool-result (User role with CbToolResult blocks).
  165 isToolResult :: Message -> Bool
  166 isToolResult (Message User blocks) = any isResultBlock blocks
  167   where isResultBlock CbToolResult{} = True
  168         isResultBlock _ = False
  169 isToolResult _ = False
  170 
  171 -- | Check if a message is a tool-use (Assistant role with CbToolUse blocks).
  172 isToolUse :: Message -> Bool
  173 isToolUse (Message Assistant blocks) = any isUseBlock blocks
  174   where isUseBlock CbToolUse{} = True
  175         isUseBlock _ = False
  176 isToolUse _ = False
  177 
  178 -- | Take the last @n@ elements of a list.
  179 takeEnd :: Int -> [a] -> [a]
  180 takeEnd n xs = drop (max 0 (length xs - n)) xs
  181 
  182 -- | Drop the last @n@ elements of a list.
  183 dropEnd :: Int -> [a] -> [a]
  184 dropEnd n xs = take (max 0 (length xs - n)) xs

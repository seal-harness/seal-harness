    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | Standalone symmetric-crypto primitives: AES-256-CTR encryption, SHA-256
    3 -- hashing, constant-time comparison, and random-byte / token generation.
    4 --
    5 -- __Important — confidentiality only, no authentication.__
    6 -- The 'encrypt' and 'decrypt' functions use AES-256 in CTR mode, which
    7 -- provides /confidentiality/ but /not/ integrity or authentication.  The
    8 -- ciphertext is unauthenticated and malleable: bit-flips to the ciphertext
    9 -- produce bit-flips in the decrypted output with no error raised.  Callers
   10 -- that need tamper-detection must pair this with a MAC, switch to an AEAD
   11 -- construction, or — for at-rest secrets — use the vault API, which is backed
   12 -- by @age@ (an authenticated encryption scheme).
   13 module Seal.Security.Crypto
   14   ( getRandomBytes
   15   , sha256Hash
   16   , constantTimeEq
   17   , generateToken
   18   , encrypt
   19   , decrypt
   20   ) where

   40 -- | Cryptographically secure random bytes from system entropy.
   41 getRandomBytes :: Int -> IO ByteString
   42 getRandomBytes = R.getRandomBytes

   44 -- | Lowercase-hex SHA-256 of the input (64 ASCII bytes).
   45 -- crypton-1.x wraps GHC's native ByteArray, which is not an instance of
   46 -- memory's ByteArrayAccess, so we go through the Show instance (which renders
   47 -- lowercase hex) rather than convertToBase.
   48 sha256Hash :: ByteString -> ByteString
   49 sha256Hash bs = BC.pack $ show (hash bs :: Digest SHA256)
   50 
   51 -- | Constant-time equality, for comparing secrets/tokens/hashes.
   52 constantTimeEq :: ByteString -> ByteString -> Bool
   53 constantTimeEq = constEq

   55 -- | A hex token of @n@ random bytes (so 2n hex chars).
   56 generateToken :: Int -> IO Text
   57 generateToken n = do
   58   raw <- getRandomBytes n
   59   pure (TE.decodeUtf8 (convertToBase Base16 raw))

   61 -- | AES-256-CTR encrypt. A fresh random 16-byte IV is generated and prepended
   62 -- to the ciphertext. The key must be exactly 32 bytes.
   63 --
   64 -- __Warning — confidentiality only, NOT integrity\/authentication.__
   65 -- The ciphertext is unauthenticated and malleable: bit-flips in transit go
   66 -- undetected.  Callers needing tamper-detection must add a MAC or use an AEAD
   67 -- construction; for at-rest secrets prefer the vault API, which uses an
   68 -- authenticated scheme end-to-end.
   69 encrypt :: SecretKey -> ByteString -> IO (Either Text ByteString)
   70 encrypt sk plaintext = withSecretKey sk $ \keyBytes ->
   71   case cipherInit keyBytes :: CryptoFailable AES256 of
   72     CryptoFailed _      -> pure (Left "key must be 32 bytes for AES-256")
   73     CryptoPassed cipher -> do
   74       ivBytes <- getRandomBytes ivLength
   75       case makeIV ivBytes of
   76         Nothing -> pure (Left "could not construct a 16-byte IV")
   77         Just iv -> pure (Right (ivBytes <> ctrCombine cipher iv plaintext))
   78 
   79 -- | Inverse of 'encrypt'. Pure: CTR needs no entropy to decrypt.
   80 --
   81 -- __Warning — confidentiality only, NOT integrity\/authentication.__
   82 -- No MAC is checked; a tampered ciphertext decrypts silently to corrupted
   83 -- plaintext.  See 'encrypt' for guidance on when to use a MAC or AEAD instead.
   84 decrypt :: SecretKey -> ByteString -> Either Text ByteString
   85 decrypt sk blob = withSecretKey sk $ \keyBytes ->
   86   if BS.length blob < ivLength
   87     then Left "ciphertext shorter than the IV it must carry"
   88     else
   89       let (ivBytes, ciphertext) = BS.splitAt ivLength blob
   90       in case cipherInit keyBytes :: CryptoFailable AES256 of
   91            CryptoFailed _      -> Left "key must be 32 bytes for AES-256"
   92            CryptoPassed cipher ->
   93              case makeIV ivBytes of
   94                Nothing -> Left "could not construct a 16-byte IV"
   95                Just iv -> Right (ctrCombine cipher iv ciphertext)

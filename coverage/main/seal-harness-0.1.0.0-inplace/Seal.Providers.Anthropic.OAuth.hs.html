   57 -- | A PKCE verifier/challenge pair. In this flow @state@ == the verifier.
   58 data Pkce = Pkce
   59   { pkceVerifier :: Text
   60   , pkceChallenge :: Text
   61   } deriving stock (Eq, Show)

   63 -- | OAuth credentials. Secrets stay opaque; there is deliberately NO 'Show' or
   64 -- 'ToJSON' instance. The only serialization path is 'serializeTokens' (added
   65 -- with the codec), which writes into the encrypted vault.
   66 data OAuthTokens = OAuthTokens
   67   { otAccess :: BearerToken
   68   , otRefresh :: RefreshToken
   69   , otExpiresAt :: UTCTime
   70   }

   95 -- PKCE ---------------------------------------------------------------------
   96 
   97 -- | @base64url-nopad(SHA-256(bytes))@. The argument is the ASCII bytes of the
   98 -- verifier string.
   99 --
  100 -- Note: crypton 1.x stores 'Digest' in GHC's native @ByteArray#@, which is
  101 -- not an instance of @memory@'s 'ByteArrayAccess'. We extract the raw digest
  102 -- bytes by decoding the lowercase-hex 'Show' instance, mirroring the approach
  103 -- used in "Seal.Security.Crypto".
  104 codeChallenge :: ByteString -> Text
  105 codeChallenge bs =
  106   TE.decodeUtf8 (convertToBase Base64URLUnpadded (sha256Raw bs))
  107   where
  108     sha256Raw :: ByteString -> ByteString
  109     sha256Raw b =
  110       let hexBs = TE.encodeUtf8 $ T.pack $ show (hash b :: Digest SHA256)
  111       in case convertFromBase Base16 hexBs :: Either String ByteString of
  112            Right raw -> raw
  113            Left e -> error ("sha256Raw: impossible hex decode: " <> e)
  114 
  115 -- | Build the authorize URL. Query params are emitted in the required order and
  116 -- percent-encoded by 'renderSimpleQuery' (space -> %20, ':' -> %3A, etc.).
  117 buildAuthorizeUrl :: Pkce -> Text
  118 buildAuthorizeUrl pkce =
  119   oauthAuthorizeUrl <> TE.decodeUtf8 (renderSimpleQuery True params)
  120   where
  121     params =
  122       [ ("response_type",        "code")
  123       , ("client_id",            TE.encodeUtf8 oauthClientId)
  124       , ("redirect_uri",         TE.encodeUtf8 oauthRedirectUri)
  125       , ("scope",                TE.encodeUtf8 oauthScope)
  126       , ("state",                TE.encodeUtf8 (pkceVerifier pkce))
  127       , ("code_challenge",       TE.encodeUtf8 (pkceChallenge pkce))
  128       , ("code_challenge_method","S256")
  129       ]

  131 -- | Split the pasted @CODE#STATE@ on the FIRST '#'. With no '#', the whole
  132 -- string is the code and the state is empty.
  133 parsePastedCode :: Text -> (Text, Text)
  134 parsePastedCode t =
  135   let (code, rest) = T.breakOn "#" t
  136   in (code, T.drop 1 rest)
  137 
  138 -- | Fresh PKCE pair: 32 random bytes -> base64url-nopad verifier (43 chars);
  139 -- challenge = 'codeChallenge' of the verifier's ASCII bytes.
  140 newPkce :: IO Pkce
  141 newPkce = do
  142   raw <- getRandomBytes 32
  143   let verifier = TE.decodeUtf8 (convertToBase Base64URLUnpadded raw)
  144   pure (Pkce verifier (codeChallenge (TE.encodeUtf8 verifier)))

  146 -- Token-response parser + vault-blob codec ------------------------------------
  147 
  148 -- | Parse the token endpoint's JSON response. @now@ is the reference time used
  149 -- to turn the relative @expires_in@ (seconds) into an absolute expiry.
  150 -- @token_type@ and @scope@ are ignored.
  151 parseTokenResponse :: UTCTime -> Value -> Either Text OAuthTokens
  152 parseTokenResponse now = mapLeft T.pack . parseEither parse
  153   where
  154     mapLeft f = either (Left . f) Right
  155     parse = withObject "token response" $ \o -> do
  156       acc     <- o .: "access_token"
  157       ref     <- o .: "refresh_token"
  158       expires <- o .: "expires_in"
  159       pure OAuthTokens
  160         { otAccess    = mkBearerToken (TE.encodeUtf8 acc)
  161         , otRefresh   = mkRefreshToken (TE.encodeUtf8 ref)
  162         , otExpiresAt = addUTCTime (fromIntegral (expires :: Int)) now
  163         }
  164 
  165 -- | Encode tokens as the vault blob. This is the ONLY place token bytes are
  166 -- serialized; the target is the encrypted vault.
  167 serializeTokens :: OAuthTokens -> ByteString
  168 serializeTokens ts =
  169   withBearerToken (otAccess ts) $ \acc ->
  170     withRefreshToken (otRefresh ts) $ \ref ->
  171       BL.toStrict $ A.encode $ object
  172         [ "access_token"  .= TE.decodeUtf8 acc
  173         , "refresh_token" .= TE.decodeUtf8 ref
  174         , "expires_at"    .= (round (utcTimeToPOSIXSeconds (otExpiresAt ts)) :: Int)
  175         ]
  176 
  177 -- | Decode the vault blob back into tokens.
  178 deserializeTokens :: ByteString -> Either Text OAuthTokens
  179 deserializeTokens bs =
  180   case eitherDecodeStrict' bs of
  181     Left e  -> Left (T.pack e)
  182     Right v -> mapLeft T.pack (parseEither parse v)
  183   where
  184     mapLeft f = either (Left . f) Right
  185     parse = withObject "oauth tokens blob" $ \o -> do
  186       acc  <- o .: "access_token"
  187       ref  <- o .: "refresh_token"
  188       exp' <- o .: "expires_at"
  189       pure OAuthTokens
  190         { otAccess    = mkBearerToken (TE.encodeUtf8 acc)
  191         , otRefresh   = mkRefreshToken (TE.encodeUtf8 ref)
  192         , otExpiresAt = posixSecondsToUTCTime (fromIntegral (exp' :: Int))
  193         }

  195 -- HTTP: request-body builders + code exchange + refresh -----------------------
  196 
  197 -- | JSON body for the authorization_code grant (the code exchange).
  198 authorizationCodeBody :: Pkce -> Text -> Text -> Value
  199 authorizationCodeBody pkce code state = object
  200   [ "grant_type"    .= ("authorization_code" :: Text)
  201   , "client_id"     .= oauthClientId
  202   , "code"          .= code
  203   , "state"         .= state
  204   , "redirect_uri"  .= oauthRedirectUri
  205   , "code_verifier" .= pkceVerifier pkce
  206   ]
  207 
  208 -- | JSON body for the refresh_token grant. Carries the refresh secret into the
  209 -- request body via the CPS accessor (the token endpoint needs it).
  210 refreshTokenBody :: OAuthTokens -> Value
  211 refreshTokenBody ts =
  212   withRefreshToken (otRefresh ts) $ \ref -> object
  213     [ "grant_type"    .= ("refresh_token" :: Text)
  214     , "client_id"     .= oauthClientId
  215     , "refresh_token" .= TE.decodeUtf8 ref
  216     ]
  217 
  218 -- | POST the token endpoint with the given JSON body, then parse the response
  219 -- with @now@ as the expiry reference. Non-2xx returns Left with status + body
  220 -- (no secret is present in the body).
  221 postToken :: Manager -> UTCTime -> Value -> IO (Either Text OAuthTokens)
  222 postToken mgr now body = do
  223   initReq <- parseRequest ("POST " <> T.unpack oauthTokenUrl)
  224   let req = initReq
  225         { requestBody    = RequestBodyLBS (A.encode body)
  226         , requestHeaders =
  227             [ ("content-type",      "application/json")
  228             , ("anthropic-version", TE.encodeUtf8 anthropicVersion)
  229             , ("anthropic-beta",    TE.encodeUtf8 anthropicBeta)
  230             ]
  231         }
  232   result <- try (httpLbs req mgr) :: IO (Either SomeException (Response BL.ByteString))
  233   pure $ case result of
  234     Left _     -> Left "OAuth token request failed (connection or transport error)"
  235     Right resp ->
  236       let code = statusCode (responseStatus resp)
  237       in if code >= 200 && code <= 299
  238          then case A.eitherDecode (responseBody resp) of
  239                 Left e  -> Left (T.pack e)
  240                 Right v -> parseTokenResponse now v
  241          else Left $ "OAuth token endpoint returned HTTP " <> T.pack (show code)
  242                 <> ": " <> lazyBodyText resp
  243   where
  244     lazyBodyText =
  245       TE.decodeUtf8With TEE.lenientDecode . BL.toStrict . responseBody
  246 
  247 -- | Exchange an authorization code (+ state) for tokens.
  248 exchangeCode :: Manager -> Pkce -> Text -> Text -> IO (Either Text OAuthTokens)
  249 exchangeCode mgr pkce code state = do
  250   now <- getCurrentTime
  251   postToken mgr now (authorizationCodeBody pkce code state)
  252 
  253 -- | Refresh an access token; the response rotates the refresh token.
  254 refreshTokens :: Manager -> OAuthTokens -> IO (Either Text OAuthTokens)
  255 refreshTokens mgr ts = do
  256   now <- getCurrentTime
  257   postToken mgr now (refreshTokenBody ts)

    1 {-# LANGUAGE OverloadedStrings #-}
    2 -- | A strict, bounded CBOR decoder — just enough of RFC 8949 to read a
    3 -- WebAuthn @attestationObject@ and its COSE @EC2@ public key, and hostile to
    4 -- everything else.
    5 --
    6 -- Deliberate restrictions (all fail closed with @Left@):
    7 --
    8 -- * definite lengths only — indefinite-length items (additional info 31) are
    9 --   rejected, so a decoder never has to buffer an unbounded stream;
   10 -- * no tags, no floats, no @undefined@;
   11 -- * nesting depth ≤ 'maxDepth' and array/map arity ≤ 'maxItems';
   12 -- * every declared length is checked against the bytes that remain, so a
   13 --   4-gigabyte array header cannot make the decoder spin;
   14 -- * the whole input is capped at 'maxInput' bytes, which bounds item count
   15 --   structurally;
   16 -- * 'decodeCbor' rejects trailing bytes.
   17 --
   18 -- There are no partial functions here: every length prefix is validated
   19 -- before it is used.
   20 module Seal.Security.WebAuthn.Cbor
   21   ( Cbor (..)
   22   , decodeCbor
   23   , decodeCborPrefix
   24   , cborLookupInt
   25   , maxInput
   26   ) where
   27 
   28 import Data.Bits (shiftL, shiftR, (.&.))
   29 import Data.ByteString (ByteString)
   30 import Data.ByteString qualified as BS
   31 import Data.Text (Text)
   32 import Data.Text qualified as T
   33 import Data.Text.Encoding qualified as TE
   34 import Data.Word (Word8)

   36 -- | A decoded CBOR value. Only the shapes WebAuthn actually uses are
   37 -- representable.
   38 data Cbor
   39   = CborInt !Integer
   40   | CborBytes !ByteString
   41   | CborText !Text
   42   | CborArray ![Cbor]
   43   | CborMap ![(Cbor, Cbor)]
   44   | CborBool !Bool
   45   | CborNull
   46   deriving stock (Eq, Show)
   47 
   48 -- | Upper bound on an input we will parse at all (64 KiB). WebAuthn
   49 -- attestation objects are a few hundred bytes; anything near this is an
   50 -- attack.
   51 maxInput :: Int
   52 maxInput = 65536
   53 
   54 maxDepth :: Int
   55 maxDepth = 32
   56 
   57 maxItems :: Int
   58 maxItems = 1024
   59 
   60 -- | Decode exactly one CBOR value; trailing bytes are an error.
   61 decodeCbor :: ByteString -> Either Text Cbor
   62 decodeCbor bs = do
   63   (v, rest) <- decodeCborPrefix bs
   64   if BS.null rest
   65     then Right v
   66     else Left "trailing bytes after CBOR value"
   67 
   68 -- | Decode one CBOR value and return it with the unconsumed remainder. Used
   69 -- for the COSE key that is followed by optional extension data inside
   70 -- @attestedCredentialData@.
   71 decodeCborPrefix :: ByteString -> Either Text (Cbor, ByteString)
   72 decodeCborPrefix bs
   73   | BS.length bs > maxInput =
   74       Left ("CBOR input larger than " <> tshow maxInput <> " bytes")
   75   | otherwise = decodeItem 0 bs
   76 
   77 decodeItem :: Int -> ByteString -> Either Text (Cbor, ByteString)
   78 decodeItem depth bs
   79   | depth > maxDepth = Left "CBOR nesting too deep"
   80   | otherwise = do
   81       (h, rest) <- uncons "truncated CBOR value"
   82       let major = h `shiftR` 5
   83           ai    = h .&. 0x1f
   84       case major of
   85         0 -> do
   86           (n, rest') <- readLen ai rest
   87           Right (CborInt n, rest')
   88         1 -> do
   89           (n, rest') <- readLen ai rest
   90           Right (CborInt (negate n - 1), rest')
   91         2 -> do
   92           (n, rest') <- readLen ai rest
   93           (b, rest2) <- takeBytes n rest'
   94           Right (CborBytes b, rest2)
   95         3 -> do
   96           (n, rest') <- readLen ai rest
   97           (b, rest2) <- takeBytes n rest'
   98           case TE.decodeUtf8' b of
   99             Left _  -> Left "CBOR text string is not valid UTF-8"
  100             Right t -> Right (CborText t, rest2)
  101         4 -> do
  102           (n, rest') <- readLen ai rest
  103           checkArity n rest'
  104           decodeList (depth + 1) n rest'
  105         5 -> do
  106           (n, rest') <- readLen ai rest
  107           checkArity n rest'
  108           decodeMap (depth + 1) n rest'
  109         6 -> Left "CBOR tags are not accepted"
  110         7 -> case ai of
  111                20 -> Right (CborBool False, rest)
  112                21 -> Right (CborBool True, rest)
  113                22 -> Right (CborNull, rest)
  114                _  -> Left "unsupported CBOR simple/float value"
  115         _ -> Left "invalid CBOR major type"
  116   where
  117     uncons msg = case BS.uncons bs of
  118       Nothing     -> Left ("truncated CBOR: " <> msg)
  119       Just (x, xs) -> Right (x, xs)
  120 
  121 -- | Read the length payload for an additional-info nibble.
  122 readLen :: Word8 -> ByteString -> Either Text (Integer, ByteString)
  123 readLen ai rest
  124   | ai <= 23 = Right (fromIntegral ai, rest)
  125   | ai == 24 = readN 1 rest
  126   | ai == 25 = readN 2 rest
  127   | ai == 26 = readN 4 rest
  128   | ai == 27 = readN 8 rest
  129   | otherwise = Left "CBOR indefinite or reserved length is not accepted"
  130   where
  131     readN n bs
  132       | BS.length bs < n = Left "truncated CBOR length"
  133       | otherwise = do
  134           let (raw, rest') = BS.splitAt n bs
  135           Right (foldl (\acc w -> acc `shiftL` 8 + fromIntegral w) 0 (BS.unpack raw), rest')
  136 
  137 -- | Take @n@ bytes, refusing a length that exceeds the remaining input.
  138 takeBytes :: Integer -> ByteString -> Either Text (ByteString, ByteString)
  139 takeBytes n bs
  140   | n < 0 = Left "negative CBOR length"
  141   | n > fromIntegral (BS.length bs) = Left "CBOR length exceeds remaining input"
  142   | otherwise = Right (BS.splitAt (fromIntegral n) bs)
  143 
  144 -- | Reject an arity that could not possibly be satisfied by the remaining
  145 -- bytes (every element needs at least one byte; a map entry needs two).
  146 checkArity :: Integer -> ByteString -> Either Text ()
  147 checkArity n rest
  148   | n < 0 = Left "negative CBOR arity"
  149   | n > fromIntegral maxItems = Left ("CBOR container arity exceeds " <> tshow maxItems)
  150   | n > fromIntegral (BS.length rest) = Left "CBOR arity exceeds remaining input"
  151   | otherwise = Right ()
  152 
  153 decodeList :: Int -> Integer -> ByteString -> Either Text (Cbor, ByteString)
  154 decodeList depth n = go n []
  155   where
  156     go 0 acc rest = Right (CborArray (reverse acc), rest)
  157     go k acc rest = do
  158       (v, rest') <- decodeItem depth rest
  159       go (k - 1) (v : acc) rest'
  160 
  161 decodeMap :: Int -> Integer -> ByteString -> Either Text (Cbor, ByteString)
  162 decodeMap depth n = go n []
  163   where
  164     go 0 acc rest = Right (CborMap (reverse acc), rest)
  165     go k acc rest = do
  166       (key, rest1) <- decodeItem depth rest
  167       (val, rest2) <- decodeItem depth rest1
  168       go (k - 1) ((key, val) : acc) rest2
  169 
  170 -- | Look up an integer key in a CBOR map. COSE keys are small integers, some
  171 -- negative (@-1@..@-3@), so this is the only lookup the verifier needs.
  172 cborLookupInt :: Integer -> Cbor -> Maybe Cbor
  173 cborLookupInt k (CborMap pairs) = lookup (CborInt k) pairs
  174 cborLookupInt _ _               = Nothing

  177 tshow = T.pack . show

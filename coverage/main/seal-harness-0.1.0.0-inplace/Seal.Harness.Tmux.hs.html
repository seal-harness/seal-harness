   61 -- | A validated tmux identifier (session name, window name, pane id).
   62 -- Smart-constructed: rejects empty, leading dash, control chars, and @:@
   63 -- (tmux's separator — a @:@ in a name would break coordinate parsing).
   64 newtype TmuxIdent = TmuxIdent Text
   65   deriving stock (Eq, Show)
   66 
   67 -- | The charset a tmux ident may contain: @A-Za-z0-9_.%@@ (no @:@ — tmux's
   68 -- separator; no control chars; no leading dash — option injection). @%@
   69 -- is used by tmux pane ids (@%5@).
   70 tmuxIdentChars :: Set Char
   71 tmuxIdentChars = Set.fromList
   72   $ ['A'..'Z'] <> ['a'..'z'] <> ['0'..'9'] <> "_.%-"
   73 
   74 -- | Smart constructor for 'TmuxIdent'. 'Left' on a malformed ident.
   75 mkTmuxIdent :: Text -> Either Text TmuxIdent
   76 mkTmuxIdent t = case validateTmuxIdent t of
   77   Left err -> Left err
   78   Right _  -> Right (TmuxIdent t)
   79 
   80 -- | The bare predicate: 'Right ()' if valid, 'Left' with an error otherwise.
   81 validateTmuxIdent :: Text -> Either Text ()
   82 validateTmuxIdent t
   83   | T.null t              = Left "tmux ident is empty"
   84   | T.head t == '-'       = Left "tmux ident must not start with '-' (option injection)"
   85   | T.any (== ':') t      = Left "tmux ident must not contain ':' (tmux separator)"
   86   | not (T.all validChar t) = Left "tmux ident has invalid characters"
   87   | otherwise             = Right ()
   88   where
   89     validChar c = c `Set.member` tmuxIdentChars
   90 
   91 tmuxIdentText :: TmuxIdent -> Text
   92 tmuxIdentText (TmuxIdent t) = t

  101 -- | @send-keys -t <target> -l -- <text>@ — send literal text. The @-l@ +
  102 -- @--@ separator guards against option injection: a payload starting with
  103 -- @-@ is sent as literal text, not interpreted as a flag. 'sendEnterNamedArgs'
  104 -- is separate so a payload like @"Enter"@ is never parsed as the Enter key.
  105 sendKeysNamedArgs :: TmuxIdent -> Text -> [String]
  106 sendKeysNamedArgs target text =
  107   [ "send-keys", "-t", T.unpack (tmuxIdentText target), "-l", "--", T.unpack text ]
  108 
  109 -- | @send-keys -t <target> Enter@ — send the Enter keystroke. No @-l@/@--@
  110 -- (the key name IS the token).
  111 sendEnterNamedArgs :: TmuxIdent -> [String]
  112 sendEnterNamedArgs target =
  113   [ "send-keys", "-t", T.unpack (tmuxIdentText target), "Enter" ]

  163 -- | The process-execution seam: a function that runs @tmux@ with a given
  164 -- argv and returns its stdout. The real implementation uses
  165 -- "System.Process" (no stdin, capture stdout); tests supply a fake that
  166 -- records the argv and returns scripted output.
  167 newtype TmuxRunner = TmuxRunner { runTmux :: [String] -> IO (Either HarnessError Text) }
  168 
  169 -- | The real tmux runner via System.Process. Preflight @tmux --version@
  170 -- (fail-closes with 'HeTmuxMissing' if absent).
  171 mkRealTmuxRunner :: IO TmuxRunner
  172 mkRealTmuxRunner = do
  173   ok <- probeTmux
  174   pure (TmuxRunner (if ok then runReal else const (pure (Left HeTmuxMissing))))
  175   where
  176     runReal args = do
  177       res <- try @IOException (readTmuxNoInput args)
  178       case res of
  179         Right (ExitSuccess, out, _err) -> pure (Right (TE.decodeUtf8Lenient out))
  180         Right (ExitFailure 127, _, _) -> pure (Left HeTmuxMissing)
  181         Right (ExitFailure n, _, err) -> pure (Left (HeCaptureFailed ("tmux exited " <> T.pack (show n) <> ": " <> TE.decodeUtf8Lenient err)))
  182         Left _e                        -> pure (Left HeTmuxMissing)  -- launch failure = not on PATH
  183 
  184 -- | Run tmux with no stdin, capturing stdout/stderr as strict ByteStrings.
  185 readTmuxNoInput :: [String] -> IO (ExitCode, BS.ByteString, BS.ByteString)
  186 readTmuxNoInput args =
  187   withCreateProcess
  188     ( (proc "tmux" args)
  189         { std_in = NoStream, std_out = CreatePipe, std_err = CreatePipe }
  190     ) $ \_ mOut mErr ph -> do
  191       (hOut, hErr) <- case (mOut, mErr) of
  192         (Just a, Just b) -> pure (a, b)
  193         _ -> error "readTmuxNoInput: pipe creation failed (unreachable)"
  194       out <- BS.hGetContents hOut
  195       err <- BS.hGetContents hErr
  196       ec  <- waitForProcess ph
  197       let !_ = BS.length out
  198           !_ = BS.length err
  199       pure (ec, out, err)
  200 
  201 -- | Preflight @tmux --version@.
  202 probeTmux :: IO Bool
  203 probeTmux = do
  204   r <- try @IOException (readTmuxNoInput ["--version"])
  205   case r of
  206     Right (ExitSuccess, _, _) -> pure True
  207     _                         -> pure False

  209 -- | Run a tmux command that's expected to produce no useful stdout (just

  211 runVoid :: TmuxRunner -> [String] -> IO (Either HarnessError ())
  212 runVoid runner args = either Left (const (Right ())) <$> runTmux runner args

  219 -- | Add a harness window to a session, stamp the @seal_id marker.
  220 -- @new-window -t <session> -n <name>@ then @set-option -t <target> @seal_id <id>@.
  221 addHarnessWindowNamed
  222   :: TmuxRunner -> TmuxIdent -> TmuxIdent -> HarnessId -> IO (Either HarnessError ())
  223 addHarnessWindowNamed runner session name hid = do
  224   r1 <- runVoid runner (newWindowNamedArgs session name)
  225   case r1 of
  226     Left e -> pure (Left e)
  227     Right _ -> runVoid runner (setWindowMarkerArgs name "seal_id" (harnessIdToText hid))

  252 -- | Read the tmux user options (markers) for a window.
  253 -- @show-options -t <target>@ → parse @@"key" "value"@ lines into a map.
  254 readMarkers :: TmuxRunner -> TmuxIdent -> IO (Either HarnessError (Map Text Text))
  255 readMarkers runner target = do
  256   r <- runTmux runner ["show-options", "-t", T.unpack (tmuxIdentText target)]
  257   pure $ case r of
  258     Left e  -> Left e
  259     Right t -> Right (parseMarkers t)
  260 
  261 -- | Parse tmux @show-options@ output: lines like @@"seal_id" "abc-uuid"@.
  262 -- Strip the leading @@, unquote the key + value.
  263 parseMarkers :: Text -> Map Text Text
  264 parseMarkers = Map.fromList . mapMaybe parseLine . T.lines
  265   where
  266     parseLine ln =
  267       case T.words ln of
  268         [k, v] -> case T.uncons k of
  269           Just ('@', rest) -> Just (rest, unquote v)
  270           _                -> Nothing
  271         _ -> Nothing
  272     unquote v = fromMaybe v (T.stripSuffix "\"" (fromMaybe v (T.stripPrefix "\"" v)))
  273

  298 -- | A fromMaybe that's imported here to avoid an extra import line.
  299 fromMaybe :: a -> Maybe a -> a
  300 fromMaybe d Nothing  = d
  301 fromMaybe _ (Just x) = x
  302 
  303 -- | A mapMaybe (local — avoids a Data.Maybe import for one helper).
  304 mapMaybe :: (a -> Maybe b) -> [a] -> [b]
  305 mapMaybe _ []     = []
  306 mapMaybe f (x:xs) = case f x of
  307   Just y  -> y : mapMaybe f xs
  308   Nothing -> mapMaybe f xs
